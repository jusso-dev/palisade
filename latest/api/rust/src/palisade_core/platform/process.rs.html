1//! Walking a process's ancestry safely when process IDs are reused.
2
3use alloc::vec::Vec;
4
5use super::{ErrorKind, ProcessInfo, ProcessKey, ProcessRecord, Result};
6
7/// The most ancestors [`ancestry`] returns (#59 records up to 32).
8pub const MAX_ANCESTRY: usize = 32;
9
10/// The ancestors of `process`, parent first, up to [`MAX_ANCESTRY`].
11///
12/// The walk stops at the first ancestor that cannot be trusted to be the
13/// real parent:
14///
15/// - the recorded parent ID has no running process (the parent exited);
16/// - the process now using that ID was created **after** the child, so it
17///   reused the ID of the real, exited parent;
18/// - the chain loops back to a process already visited.
19///
20/// So a reused process ID never makes an unrelated process look like an
21/// ancestor. Errors other than [`ErrorKind::NotFound`] (such as access
22/// denied) are returned, because they say nothing about whether the
23/// ancestor exists.
24///
25/// # Errors
26///
27/// Any error of [`ProcessInfo::get`] for `process` itself, and errors other
28/// than [`ErrorKind::NotFound`] while walking.
29pub fn ancestry<P: ProcessInfo + ?Sized>(
30    processes: &P,
31    process: ProcessKey,
32) -> Result<Vec<ProcessRecord>> {
33    let mut out: Vec<ProcessRecord> = Vec::new();
34    let mut child = processes.get(process)?;
35    while out.len() < MAX_ANCESTRY {
36        let Some(parent_pid) = child.parent_pid else {
37            break;
38        };
39        let parent = match processes.by_pid(parent_pid) {
40            Ok(parent) => parent,
41            Err(e) if e.kind() == ErrorKind::NotFound => break,
42            Err(e) => return Err(e),
43        };
44        // A parent is always created before its child. A process created
45        // later took over the exited parent's ID.
46        if parent.key.created > child.key.created {
47            break;
48        }
49        if parent.key == process || out.iter().any(|a| a.key == parent.key) {
50            break;
51        }
52        out.push(parent.clone());
53        child = parent;
54    }
55    Ok(out)
56}
