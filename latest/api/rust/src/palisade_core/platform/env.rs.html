1//! `%NAME%` environment-variable expansion, as Windows does it.
2
3use alloc::string::String;
4use core::fmt;
5
6/// The longest expansion accepted, in characters: the Windows limit for an
7/// environment block value.
8pub const MAX_EXPANDED_CHARS: usize = 32_767;
9
10/// An expansion failed.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum ExpandError {
13    /// The result would be longer than [`MAX_EXPANDED_CHARS`].
14    TooLong,
15}
16
17impl fmt::Display for ExpandError {
18    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
19        match self {
20            Self::TooLong => write!(
21                f,
22                "the expansion is longer than {MAX_EXPANDED_CHARS} characters"
23            ),
24        }
25    }
26}
27
28impl core::error::Error for ExpandError {}
29
30/// Replaces each `%NAME%` in `input` with `lookup(NAME)`.
31///
32/// The rules, which the Windows implementation and the mocks share:
33///
34/// - The text between a `%` and the next `%` is a name. A name that
35///   `lookup` does not know is left as written, `%NAME%`, and scanning
36///   continues after its closing `%`.
37/// - `%%` (an empty name) and a `%` with no closing `%` are left as written.
38/// - Values are inserted literally: a `%` inside a value is not expanded
39///   again, so a user-controlled variable cannot pull in another one.
40///
41/// Names are passed to `lookup` exactly as written; case-insensitive lookup
42/// is the caller's (the [`Environment`](super::Environment) implementation's)
43/// job.
44///
45/// # Errors
46///
47/// [`ExpandError::TooLong`] if the result exceeds [`MAX_EXPANDED_CHARS`].
48pub fn expand_env_vars(
49    input: &str,
50    lookup: impl Fn(&str) -> Option<String>,
51) -> Result<String, ExpandError> {
52    let mut out = String::new();
53    let mut chars = 0usize;
54    let mut push = |out: &mut String, s: &str| {
55        chars = chars.saturating_add(s.chars().count());
56        if chars > MAX_EXPANDED_CHARS {
57            return Err(ExpandError::TooLong);
58        }
59        out.push_str(s);
60        Ok(())
61    };
62    let mut rest = input;
63    while let Some(start) = rest.find('%') {
64        let (before, from_percent) = rest.split_at(start);
65        push(&mut out, before)?;
66        let after = from_percent.get(1..).unwrap_or_default();
67        let Some(end) = after.find('%') else {
68            // No closing `%`: the rest is literal.
69            push(&mut out, from_percent)?;
70            return Ok(out);
71        };
72        let name = after.get(..end).unwrap_or_default();
73        let value = if name.is_empty() { None } else { lookup(name) };
74        if let Some(value) = value {
75            push(&mut out, &value)?;
76        } else {
77            push(&mut out, "%")?;
78            push(&mut out, name)?;
79            push(&mut out, "%")?;
80        }
81        rest = after.get(end + 1..).unwrap_or_default();
82    }
83    push(&mut out, rest)?;
84    Ok(out)
85}
