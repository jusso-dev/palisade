27/// A value could not be parsed from its text form.
28///
29/// The message is static, so parsing never allocates and the type works in
30/// `no_std` builds.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub struct ParseError(&'static str);
33
34impl ParseError {
35    /// What was expected.
36    #[must_use]
37    pub const fn message(&self) -> &'static str {
38        self.0
39    }
40}
41
42impl fmt::Display for ParseError {
43    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
44        f.write_str(self.0)
45    }
46}
47
48impl core::error::Error for ParseError {}

50/// Implements serde for a type through its `Display` and `FromStr` impls, so
51/// the text form has exactly one definition.
52macro_rules! string_serde {
53    ($ty:ty, $expecting:literal) => {
54        #[cfg(feature = "serde")]
55        impl serde::Serialize for $ty {
56            fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
57                serializer.collect_str(self)
58            }
59        }
60
61        #[cfg(feature = "serde")]
62        impl<'de> serde::Deserialize<'de> for $ty {
63            fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
64                struct Visitor;
65                impl serde::de::Visitor<'_> for Visitor {
66                    type Value = $ty;
67                    fn expecting(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
68                        f.write_str($expecting)
69                    }
70                    fn visit_str<E: serde::de::Error>(self, v: &str) -> Result<$ty, E> {
71                        v.parse()
72                            .map_err(|e: $crate::ParseError| E::custom(e.message()))
73                    }
74                }
75                deserializer.deserialize_str(Visitor)
76            }
77        }
78    };
79}
80pub(crate) use string_serde;
