3/// How a policy's decisions take effect (#36).
4///
5/// The variants are declared from weakest to strongest, so the derived
6/// ordering ranks them. The "strictest" merge in #36 treats `learning` and
7/// `audit` as equally strong; that rule belongs to the merge code, not to this
8/// ordering.
9#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
10#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
11#[cfg_attr(feature = "serde", serde(rename_all = "snake_case"))]
12#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
13pub enum Mode {
14    /// The engine is not consulted. The device is reported as unprotected.
15    Disabled,
16    /// Evaluate fully, allow everything, and report would-block decisions.
17    Monitor,
18    /// Like `monitor`, with a full event for every decision.
19    Audit,
20    /// Audit, and report every would-block object as a learning candidate.
21    Learning,
22    /// Decisions take effect: blocked means blocked.
23    Enforce,
24}
