1//! The NTFS upcase table (#332).
2//!
3//! NTFS compares two names by upper-casing each UTF-16 code unit through
4//! the volume's upcase table (`$UpCase`), written when the volume is
5//! formatted. [`upcase_unit`] is the table that Windows 7 and every later
6//! version write, and that the OS itself upper-cases with
7//! (`RtlUpcaseUnicodeChar`; `palisade-fileid`'s Windows tests check every
8//! unit against it). [`fold`] upper-cases a name with it.
9//!
10//! It is not Unicode's upper-case mapping. It never maps a character
11//! outside ASCII into ASCII (`ı`, `ſ` and the Kelvin sign stay as they
12//! are, so `ſetup.exe` and `SETUP.EXE` are two names), it leaves
13//! characters outside the Basic Multilingual Plane alone, and it predates
14//! case pairs that later Unicode versions added. A volume formatted by an
15//! older Windows holds an older table; telling those apart is
16//! `palisade-fileid`'s job.
17//!
18//! Pure and `no_std`.
19
20use alloc::string::String;

22/// The CRC-64 that NTFS records in `$UpCase:$Info` for the table
23/// [`upcase_unit`] implements: the one Windows 7 and later write when they

171/// Upper-cases one UTF-16 code unit with the standard NTFS table.
172#[must_use]
173pub fn upcase_unit(unit: u16) -> u16 {
174    let i = RUNS.partition_point(|&(first, ..)| first <= unit);
175    let Some(&(first, last, step, add)) = i.checked_sub(1).and_then(|i| RUNS.get(i)) else {
176        return unit;
177    };
178    if unit <= last && (unit - first).is_multiple_of(step) {
179        unit.wrapping_add(add)
180    } else {
181        unit
182    }
183}
184
185/// Upper-cases one character with [`upcase_unit`]. A character outside
186/// the Basic Multilingual Plane is unchanged, as NTFS leaves each half of
187/// its surrogate pair unchanged.
188#[must_use]
189pub fn upcase_char(c: char) -> char {
190    u16::try_from(u32::from(c))
191        .ok()
192        .and_then(|u| char::from_u32(u32::from(upcase_unit(u))))
193        .unwrap_or(c)
194}
195
196/// Upper-cases every character of `s` with [`upcase_char`]: the form two
197/// names that NTFS treats as one share.
198#[must_use]
199pub fn fold(s: &str) -> String {
200    s.chars().map(upcase_char).collect()
201}

203#[cfg(test)]
204mod tests {
205    use alloc::vec::Vec;
206
207    use super::*;

209    /// The CRC-64 NTFS records in `$UpCase:$Info` (`mkntfs`'s `crc64`):
210    /// reflected, polynomial `0x9A6C9329AC4BC9B5`, initial value and final
211    /// XOR all ones.
212    fn crc64(data: &[u8]) -> u64 {
213        let mut crc = u64::MAX;
214        for &b in data {
215            crc ^= u64::from(b);
216            for _ in 0..8 {
217                crc = if crc & 1 == 1 {
218                    0x9A6C_9329_AC4B_C9B5 ^ (crc >> 1)
219                } else {
220                    crc >> 1
221                };
222            }
223        }
224        !crc
225    }
226
227    fn table() -> Vec<u16> {
228        (0..=u16::MAX).map(upcase_unit).collect()
229    }
230
231    #[test]
232    fn the_table_is_the_one_windows_records() {
233        let bytes: Vec<u8> = table().iter().flat_map(|u| u.to_le_bytes()).collect();
234        assert_eq!(crc64(&bytes), STANDARD_UPCASE_CRC);
235        assert_eq!(crc64(b""), 0);
236    }
237
238    #[test]
239    fn runs_are_sorted_and_disjoint() {
240        for w in RUNS.windows(2) {
241            assert!(w[0].1 < w[1].0, "{w:?}");
242        }
243        for (first, last, step, _) in RUNS {
244            assert!(first <= last && step >= 1);
245        }
246    }
247
248    #[test]
249    fn upcasing_is_stable_and_keeps_ascii_apart() {
250        let t = table();
251        for u in 0..=u16::MAX {
252            let up = t[usize::from(u)];
253            // Upper-casing an upper-case unit changes nothing.
254            assert_eq!(t[usize::from(up)], up, "{u:04X}");
255            // ASCII upper-cases as ASCII, and nothing else reaches it.
256            match u8::try_from(u) {
257                Ok(b) if b.is_ascii() => assert_eq!(up, u16::from(b.to_ascii_uppercase())),
258                _ => assert!(up >= 0x80, "{u:04X}"),
259            }
260            // Surrogates are left alone.
261            if (0xD800..0xE000).contains(&u) {
262                assert_eq!(up, u);
263            }
264        }
265    }
266
267    #[test]
268    fn examples() {
269        assert_eq!(fold("Stra\u{df}e.exe"), "STRA\u{df}E.EXE");
270        assert_eq!(fold("\u{1c6}"), "\u{1c4}");
271        assert_eq!(fold("\u{fb01}"), "\u{fb01}");
272        assert_eq!(fold("\u{e9}"), "\u{c9}");
273        assert_eq!(fold("\u{ff}"), "\u{178}");
274        assert_eq!(fold("\u{ff41}"), "\u{ff21}");
275        // Greek with a combining iota: NTFS folds these, though Unicode's
276        // full upper case is two characters.
277        assert_eq!(fold("\u{1f80}"), "\u{1f88}");
278        assert_eq!(fold("\u{1ff3}"), "\u{1ffc}");
279        // Unicode maps these into ASCII; NTFS does not.
280        assert_eq!(fold("\u{131}\u{17f}\u{212a}"), "\u{131}\u{17f}\u{212a}");
281        // Outside the Basic Multilingual Plane: unchanged.
282        assert_eq!(fold("\u{10428}"), "\u{10428}");
283        assert_eq!(upcase_char('\u{10428}'), '\u{10428}');
284        assert_eq!(upcase_unit(0), 0);
285        assert_eq!(upcase_unit(0xFFFF), 0xFFFF);
286        assert_eq!(upcase_unit(0x149), 0x149);
287    }
