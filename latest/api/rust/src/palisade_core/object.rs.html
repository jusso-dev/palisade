8/// The kind of script, for [`ObjectType::Script`].
9#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
10#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
11#[cfg_attr(feature = "serde", serde(rename_all = "snake_case"))]
12#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
13pub enum ScriptKind {
14    /// `.ps1`, `.psm1`, `.psd1`.
15    Powershell,
16    /// `.bat`.
17    Batch,
18    /// `.cmd`.
19    Cmd,
20    /// `.vbs`, `.vbe`.
21    Vbscript,
22    /// `.js`, `.jse` run by the OS script host.
23    Jscript,
24    /// `.wsf`.
25    Wsf,
26    /// `.wsc`, `.sct` scriptlets.
27    Wsc,
28    /// `.py`, `.pyw`, `.pyc`, `.pyo`.
29    Python,
30    /// `.jar`, `.class`.
31    Java,
32    /// `.reg` registry scripts.
33    Reg,
34    /// Any other interpreted script.
35    Other,
36}

69        }
70    }

73/// The class of object being evaluated (ISM-1657, plus drivers for ISM-1658).
74///
75/// The text form is `snake_case`, and scripts are `script.<kind>`, for
76/// example `executable` or `script.powershell`. The same spelling is used by
77/// the per-type mode keys in #36. How a file's class is detected (content
78/// first, extension second) is decided by #3 and #4.
79#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
80pub enum ObjectType {
81    /// Any PE image launched as a process, whatever its extension.
82    Executable,
83    /// A module loaded into a process (`.dll`, `.ocx`, `.pyd`, `.cpl` as a module).
84    Library,
85    /// A kernel image (`.sys`).
86    Driver,
87    /// An interpreted script.
88    Script(ScriptKind),
89    /// An installer package or patch (`.msi`, `.msp`, `.mst`, `.msu`, packaged-app install).
90    Installer,
91    /// Compiled HTML help (`.chm`).
92    CompiledHtml,
93    /// An HTML application (`.hta`).
94    HtmlApplication,
95    /// A control panel applet opened through the control panel host.
96    ControlPanelApplet,
97    /// Packaged-app activation.
98    PackagedApp,
99    /// An interpreter command or argument string evaluated as an object (#7).
100    Command,
101    /// COM class activation (reserved).
102    ComObject,
103}
104
105impl ObjectType {
106    /// Every object type, in declaration order, with every script kind.
107    pub const ALL: [Self; 21] = [
108        Self::Executable,
109        Self::Library,
110        Self::Driver,
111        Self::Script(ScriptKind::Powershell),
112        Self::Script(ScriptKind::Batch),
113        Self::Script(ScriptKind::Cmd),
114        Self::Script(ScriptKind::Vbscript),
115        Self::Script(ScriptKind::Jscript),
116        Self::Script(ScriptKind::Wsf),
117        Self::Script(ScriptKind::Wsc),
118        Self::Script(ScriptKind::Python),
119        Self::Script(ScriptKind::Java),
120        Self::Script(ScriptKind::Reg),
121        Self::Script(ScriptKind::Other),
122        Self::Installer,
123        Self::CompiledHtml,
124        Self::HtmlApplication,
125        Self::ControlPanelApplet,
126        Self::PackagedApp,
127        Self::Command,
128        Self::ComObject,
129    ];
130
131    /// The name. For a script this is the prefix `script`, and the full
132    /// text form adds the kind.
133    const fn name(self) -> &'static str {
134        match self {
135            Self::Executable => "executable",
136            Self::Library => "library",
137            Self::Driver => "driver",
138            Self::Script(_) => "script",
139            Self::Installer => "installer",
140            Self::CompiledHtml => "compiled_html",
141            Self::HtmlApplication => "html_application",
142            Self::ControlPanelApplet => "control_panel_applet",
143            Self::PackagedApp => "packaged_app",
144            Self::Command => "command",
145            Self::ComObject => "com_object",
146        }
147    }
148}

150impl fmt::Display for ObjectType {
151    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
152        match self {
153            Self::Script(kind) => write!(f, "script.{}", kind.as_str()),
154            other => f.write_str(other.name()),
155        }
156    }
157}
158
159impl FromStr for ObjectType {
160    type Err = ParseError;
161
162    fn from_str(s: &str) -> Result<Self, ParseError> {
163        if let Some(kind) = s.strip_prefix("script.") {
164            return ScriptKind::ALL
165                .into_iter()
166                .find(|k| k.as_str() == kind)
167                .map(Self::Script)
168                .ok_or(ParseError("unknown script kind"));
169        }
170        Self::ALL
171            .into_iter()
172            .find(|t| !matches!(t, Self::Script(_)) && t.name() == s)
173            .ok_or(ParseError("unknown object type"))
174    }
175}
