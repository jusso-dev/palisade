1//! The SNTP packet codec (RFC 4330, the client side of RFC 5905): builds a
2//! request, checks a reply and computes the clock offset.
3//!
4//! The server uses it for its clock health check (#224: warn when the
5//! offset from NTP is over 1 s, ISM-0988), and the agent can use it for its
6//! trusted-time checks (#220). It does no I/O: the caller sends the
7//! request, receives the reply and reads its own clock.
8//!
9//! A reply is untrusted network input, so [`Reply::parse`] checks every
10//! field it uses and refuses everything else:
11//!
12//! - exactly 48 bytes (no extension fields or MAC, which SNTP clients do
13//!   not use);
14//! - version 3 or 4, mode 4 (server);
15//! - leap indicator not 3 (the server's clock is not synchronised);
16//! - stratum 1 to 15: 0 is a kiss-o'-death or unspecified, 16 and above
17//!   unsynchronised;
18//! - the originate timestamp equals the transmit timestamp of the request,
19//!   so a reply to someone else's request, or a replayed one, is refused;
20//! - receive and transmit timestamps are non-zero, and the server did not
21//!   send before it received.
22//!
23//! NTP timestamps are seconds since 1900 in 32.32 fixed point and wrap
24//! every 2^32 seconds (in 2036). Each is placed in the era that puts it
25//! closest to the local clock, so the codec keeps working across the wrap
26//! as long as the local clock is within 68 years of the truth.
27

32/// The length of an SNTP packet without extensions.
33pub const PACKET_LEN: usize = 48;
34
35/// The NTP port.
36pub const PORT: u16 = 123;
37
38/// Seconds from 1900-01-01 (the NTP epoch) to 1970-01-01 (the Unix epoch).
39const NTP_UNIX_OFFSET: i128 = 2_208_988_800;
40const MICROS_PER_SECOND: i128 = 1_000_000;
41/// One NTP era, 2^32 seconds, in microseconds.
42const ERA_MICROS: i128 = (1_i128 << 32) * MICROS_PER_SECOND;
43
44const LI_UNSYNCHRONISED: u8 = 3;
45const MODE_CLIENT: u8 = 3;
46const MODE_SERVER: u8 = 4;
47const VERSION: u8 = 4;
48
49/// A 64-bit NTP timestamp: seconds since 1900 in the high 32 bits, the
50/// fraction in the low 32.
51#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
52pub struct NtpTimestamp(pub u64);
53
54impl NtpTimestamp {
55    /// The NTP timestamp of `unix_micros` (microseconds since 1970),
56    /// wrapped into its era. [`Self::to_unix_micros`] gives back exactly
57    /// `unix_micros`.
58    #[must_use]
59    pub fn from_unix_micros(unix_micros: i64) -> Self {
60        let ntp_micros = i128::from(unix_micros) + NTP_UNIX_OFFSET * MICROS_PER_SECOND;
61        let in_era = ntp_micros.rem_euclid(ERA_MICROS);
62        let seconds = in_era / MICROS_PER_SECOND;
63        let micros = in_era % MICROS_PER_SECOND;
64        // Rounded up, so `to_unix_micros` (which rounds down) gives back
65        // the same microsecond. micros < 10^6, so the fraction stays below
66        // 2^32.
67        let fraction = ((micros << 32) + MICROS_PER_SECOND - 1) / MICROS_PER_SECOND;
68        let raw = (seconds << 32) | fraction;
69        Self(u64::try_from(raw).unwrap_or(u64::MAX))
70    }
71
72    /// Microseconds since 1970, in the era that puts the result closest to
73    /// `near_unix_micros`. The fraction is rounded down to a microsecond.
74    #[must_use]
75    pub fn to_unix_micros(self, near_unix_micros: i64) -> i64 {
76        let seconds = i128::from(self.0 >> 32);
77        let fraction = i128::from(self.0 & 0xffff_ffff);
78        let in_era = seconds * MICROS_PER_SECOND + ((fraction * MICROS_PER_SECOND) >> 32);
79        let near = i128::from(near_unix_micros) + NTP_UNIX_OFFSET * MICROS_PER_SECOND;
80        // The era whose start is closest below `near - ERA/2`.
81        let era = (near - in_era + ERA_MICROS / 2).div_euclid(ERA_MICROS);
82        let unix = in_era + era * ERA_MICROS - NTP_UNIX_OFFSET * MICROS_PER_SECOND;
83        i64::try_from(unix).unwrap_or(if unix < 0 { i64::MIN } else { i64::MAX })
84    }
85}

87/// An SNTP client request whose transmit timestamp is `transmit`. Keep
88/// `transmit`: the reply must echo it.
89#[must_use]
90pub fn request(transmit: NtpTimestamp) -> [u8; PACKET_LEN] {
91    let mut packet = [0u8; PACKET_LEN];
92    let [first, .., t0, t1, t2, t3, t4, t5, t6, t7] = &mut packet;
93    *first = (VERSION << 3) | MODE_CLIENT;
94    for (slot, byte) in [t0, t1, t2, t3, t4, t5, t6, t7]
95        .into_iter()
96        .zip(transmit.0.to_be_bytes())
97    {
98        *slot = byte;
99    }
100    packet
101}

103/// A checked server reply.
104#[derive(Clone, Copy, Debug, PartialEq, Eq)]
105pub struct Reply {
106    /// The server's stratum, 1 to 15.
107    pub stratum: u8,
108    /// When the server received the request.
109    pub receive: NtpTimestamp,
110    /// When the server sent the reply.
111    pub transmit: NtpTimestamp,
112}
113
114impl Reply {
115    /// Checks `packet` as the reply to the request sent with `sent`.
116    ///
117    /// # Errors
118    ///
119    /// [`SntpError`] for each rule in the module docs.
120    pub fn parse(packet: &[u8], sent: NtpTimestamp) -> Result<Self, SntpError> {
121        let packet: &[u8; PACKET_LEN] = packet.try_into().map_err(|_| SntpError::Length)?;
122        let [first, stratum, ..] = *packet;
123        let leap = first >> 6;
124        let version = (first >> 3) & 0b111;
125        let mode = first & 0b111;
126        if !matches!(version, 3 | 4) {
127            return Err(SntpError::Version(version));
128        }
129        if mode != MODE_SERVER {
130            return Err(SntpError::Mode(mode));
131        }
132        if stratum == 0 {
133            return Err(SntpError::KissOfDeath);
134        }
135        if leap == LI_UNSYNCHRONISED || stratum > 15 {
136            return Err(SntpError::Unsynchronised);
137        }
138        let ts = |at: usize| {
139            let mut b = [0u8; 8];
140            if let Some(field) = packet.get(at..at.saturating_add(8)) {
141                b.copy_from_slice(field);
142            }
143            NtpTimestamp(u64::from_be_bytes(b))
144        };
145        let originate = ts(24);
146        let receive = ts(32);
147        let transmit = ts(40);
148        if originate != sent {
149            return Err(SntpError::NotOurs);
150        }
151        if receive.0 == 0 || transmit.0 == 0 {
152            return Err(SntpError::ZeroTimestamp);
153        }
154        // Within one era a later time has a larger raw value; across the
155        // 2036 wrap the wrapping difference is still small and positive.
156        if transmit.0.wrapping_sub(receive.0) > (1 << 63) {
157            return Err(SntpError::SentBeforeReceived);
158        }
159        Ok(Self {
160            stratum,
161            receive,
162            transmit,
163        })
164    }
165
166    /// The local clock's offset from the server in microseconds, positive
167    /// when the local clock is behind: `((T2 - T1) + (T3 - T4)) / 2`, with
168    /// `t1` the local time the request was sent and `t4` the local time the
169    /// reply arrived, both microseconds since 1970.
170    #[must_use]
171    pub fn offset_micros(&self, t1: i64, t4: i64) -> i64 {
172        let t2 = i128::from(self.receive.to_unix_micros(t1));
173        let t3 = i128::from(self.transmit.to_unix_micros(t4));
174        let offset = ((t2 - i128::from(t1)) + (t3 - i128::from(t4))) / 2;
175        i64::try_from(offset).unwrap_or(if offset < 0 { i64::MIN } else { i64::MAX })
176    }
177
178    /// The round-trip delay in microseconds, `(T4 - T1) - (T3 - T2)`,
179    /// never below zero.
180    #[must_use]
181    pub fn delay_micros(&self, t1: i64, t4: i64) -> i64 {
182        let t2 = i128::from(self.receive.to_unix_micros(t1));
183        let t3 = i128::from(self.transmit.to_unix_micros(t4));
184        let delay = (i128::from(t4) - i128::from(t1)) - (t3 - t2);
185        i64::try_from(delay.max(0)).unwrap_or(i64::MAX)
186    }
187}

189/// Why a reply was refused.
190#[derive(Clone, Copy, Debug, PartialEq, Eq)]
191pub enum SntpError {
192    /// Not exactly [`PACKET_LEN`] bytes.
193    Length,
194    /// Not version 3 or 4.
195    Version(u8),
196    /// Not mode 4 (server).
197    Mode(u8),
198    /// Stratum 0: a kiss-o'-death (the server asks the client to stop or
199    /// slow down) or unspecified.
200    KissOfDeath,
201    /// The server's clock is not synchronised (leap indicator 3, or
202    /// stratum 16 or above).
203    Unsynchronised,
204    /// The originate timestamp is not the one the request sent.
205    NotOurs,
206    /// The receive or transmit timestamp is zero.
207    ZeroTimestamp,
208    /// The transmit timestamp is before the receive timestamp.
209    SentBeforeReceived,
210}
211
212impl fmt::Display for SntpError {
213    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
214        match self {
215            Self::Length => write!(f, "the reply is not {PACKET_LEN} bytes"),
216            Self::Version(v) => write!(f, "unsupported NTP version {v}"),
217            Self::Mode(m) => write!(f, "the reply has mode {m}, not server (4)"),
218            Self::KissOfDeath => f.write_str("the server sent a kiss-o'-death (stratum 0)"),
219            Self::Unsynchronised => f.write_str("the server's clock is not synchronised"),
220            Self::NotOurs => f.write_str("the reply does not answer this request"),
221            Self::ZeroTimestamp => f.write_str("the reply has a zero timestamp"),
222            Self::SentBeforeReceived => f.write_str("the reply was sent before it was received"),
223        }
224    }
225}

236    const SECOND: i64 = 1_000_000;
237    /// 2026-11-24T00:00:00Z.
238    const NOW: i64 = 1_795_478_400 * SECOND;
239
240    /// A valid reply to `sent`, with the server's receive and transmit
241    /// times at `t2` and `t3` (Unix microseconds).
242    fn reply(sent: NtpTimestamp, t2: i64, t3: i64) -> [u8; PACKET_LEN] {
243        let mut p = [0u8; PACKET_LEN];
244        p[0] = (VERSION << 3) | MODE_SERVER;
245        p[1] = 2;
246        p[24..32].copy_from_slice(&sent.0.to_be_bytes());
247        p[32..40].copy_from_slice(&NtpTimestamp::from_unix_micros(t2).0.to_be_bytes());
248        p[40..48].copy_from_slice(&NtpTimestamp::from_unix_micros(t3).0.to_be_bytes());
249        p
250    }

252    #[test]
253    fn known_timestamps() {
254        // The Unix epoch is NTP second 2,208,988,800.
255        assert_eq!(NtpTimestamp::from_unix_micros(0).0, 2_208_988_800 << 32);
256        // Half a second is fraction 2^31 (exact, so not rounded).
257        assert_eq!(
258            NtpTimestamp::from_unix_micros(SECOND / 2).0,
259            (2_208_988_800 << 32) | (1 << 31)
260        );
261        assert_eq!(
262            NtpTimestamp((2_208_988_800 << 32) | (1 << 31)).to_unix_micros(0),
263            SECOND / 2
264        );
265    }

267    #[test]
268    fn timestamps_round_trip_across_the_2036_wrap() {
269        // 2036-02-07T06:28:16Z is where NTP era 0 ends.
270        let wrap = (1_i64 << 32) * SECOND - 2_208_988_800 * SECOND;
271        for t in [
272            NOW,
273            0,
274            -SECOND,
275            wrap - 1,
276            wrap,
277            wrap + 1,
278            wrap + 10 * 365 * 86_400 * SECOND,
279            NOW + 123_456,
280        ] {
281            let ntp = NtpTimestamp::from_unix_micros(t);
282            assert_eq!(ntp.to_unix_micros(t), t, "{t}");
283            // A local clock a few hours off still picks the right era.
284            assert_eq!(ntp.to_unix_micros(t + 3 * 3600 * SECOND), t, "{t}");
285            assert_eq!(ntp.to_unix_micros(t - 3 * 3600 * SECOND), t, "{t}");
286        }
287        assert_eq!(NtpTimestamp::from_unix_micros(wrap).0 >> 32, 0);
288        // Extreme inputs saturate instead of overflowing.
289        let _ = NtpTimestamp(u64::MAX).to_unix_micros(i64::MAX);
290        let _ = NtpTimestamp(0).to_unix_micros(i64::MIN);
291        let _ = NtpTimestamp::from_unix_micros(i64::MIN);
292    }

303    #[test]
304    fn offset_and_delay() {
305        let t1 = NOW;
306        let sent = NtpTimestamp::from_unix_micros(t1);
307        // The local clock is 1.5 s behind; 40 ms each way; 10 ms in the server.
308        let behind = 1_500_000;
309        let t2 = t1 + 40_000 + behind;
310        let t3 = t2 + 10_000;
311        let t4 = t1 + 90_000;
312        let r = Reply::parse(&reply(sent, t2, t3), sent).unwrap();
313        assert_eq!(r.stratum, 2);
314        assert_eq!(r.offset_micros(t1, t4), behind);
315        assert_eq!(r.delay_micros(t1, t4), 80_000);
316        // Ahead by 2 s.
317        let r = Reply::parse(&reply(sent, t1 - 2 * SECOND, t1 - 2 * SECOND), sent).unwrap();
318        assert_eq!(r.offset_micros(t1, t1), -2 * SECOND);
319        // A reply that claims more server time than the round trip has no
320        // negative delay.
321        let r = Reply::parse(&reply(sent, t1, t1 + SECOND), sent).unwrap();
322        assert_eq!(r.delay_micros(t1, t1), 0);
323    }

325    #[test]
326    fn every_rule_is_checked() {
327        let sent = NtpTimestamp::from_unix_micros(NOW);
328        let good = reply(sent, NOW, NOW);
329        Reply::parse(&good, sent).unwrap();
330        let with = |f: fn(&mut [u8; PACKET_LEN])| {
331            let mut p = good;
332            f(&mut p);
333            Reply::parse(&p, sent)
334        };
335        assert_eq!(Reply::parse(&good[..47], sent), Err(SntpError::Length));
336        assert_eq!(
337            Reply::parse(&[good.as_slice(), &[0]].concat(), sent),
338            Err(SntpError::Length)
339        );
340        assert_eq!(with(|p| p[0] = (2 << 3) | 4), Err(SntpError::Version(2)));
341        assert_eq!(with(|p| p[0] = (5 << 3) | 4), Err(SntpError::Version(5)));
342        with(|p| p[0] = (3 << 3) | 4).unwrap();
343        assert_eq!(with(|p| p[0] = (4 << 3) | 3), Err(SntpError::Mode(3)));
344        assert_eq!(with(|p| p[0] = (4 << 3) | 5), Err(SntpError::Mode(5)));
345        assert_eq!(with(|p| p[1] = 0), Err(SntpError::KissOfDeath));
346        assert_eq!(with(|p| p[1] = 16), Err(SntpError::Unsynchronised));
347        with(|p| p[1] = 15).unwrap();
348        assert_eq!(with(|p| p[0] |= 0b11 << 6), Err(SntpError::Unsynchronised));
349        // Leap second warnings (1 and 2) are fine.
350        with(|p| p[0] |= 0b01 << 6).unwrap();
351        with(|p| p[0] |= 0b10 << 6).unwrap();
352        assert_eq!(with(|p| p[31] ^= 1), Err(SntpError::NotOurs));
353        assert_eq!(with(|p| p[32..40].fill(0)), Err(SntpError::ZeroTimestamp));
354        assert_eq!(with(|p| p[40..48].fill(0)), Err(SntpError::ZeroTimestamp));
355        let sent_early = reply(sent, NOW, NOW - SECOND);
356        assert_eq!(
357            Reply::parse(&sent_early, sent),
358            Err(SntpError::SentBeforeReceived)
359        );
360        for e in [
361            SntpError::Length,
362            SntpError::Version(2),
363            SntpError::Mode(3),
364            SntpError::KissOfDeath,
365            SntpError::Unsynchronised,
366            SntpError::NotOurs,
367            SntpError::ZeroTimestamp,
368            SntpError::SentBeforeReceived,
369        ] {
370            assert!(!e.to_string().is_empty());
371        }
372    }
