12/// A key ID: the first 16 bytes of the SHA-256 of the key's DER
13/// `SubjectPublicKeyInfo`, written as 32 lowercase hex digits.
14///
15/// The ID is derived from the public key, never assigned, so two records
16/// with the same `kid` always describe the same key.
17#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
18pub struct Kid([u8; 16]);
19
20impl Kid {
21    /// Wraps raw key-ID bytes.
22    #[must_use]
23    pub const fn from_bytes(bytes: [u8; 16]) -> Self {
24        Self(bytes)
25    }
26
27    /// The raw key-ID bytes.
28    #[must_use]
29    pub const fn as_bytes(&self) -> &[u8; 16] {
30        &self.0
31    }
32
33    /// The key ID of a DER `SubjectPublicKeyInfo`.
34    #[must_use]
35    pub fn of_spki(spki_der: &[u8]) -> Self {
36        let digest = crypto::sha256(spki_der);
37        let mut kid = [0u8; 16];
38        for (k, d) in kid.iter_mut().zip(digest) {
39            *k = d;
40        }
41        Self(kid)
42    }
43}
44
45impl fmt::Display for Kid {
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        Hex::from_bytes(self.0).fmt(f)
48    }
49}
50
51impl fmt::Debug for Kid {
52    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
53        write!(f, "Kid({self})")
54    }
55}

57impl FromStr for Kid {
58    type Err = KeyError;
59
60    /// Accepts exactly 32 lowercase hex digits.
61    fn from_str(s: &str) -> Result<Self, KeyError> {
62        Hex::<16>::from_str(s)
63            .map(|h| Self(*h.as_bytes()))
64            .map_err(|_| KeyError::InvalidKid)
65    }
66}

68/// The SHA-256 of a DER `SubjectPublicKeyInfo`: the form of a server CA pin
69/// and of a device certificate's key fingerprint.
70#[must_use]
71pub fn spki_sha256(spki_der: &[u8]) -> Sha256Hex {
72    Sha256Hex::from_bytes(crypto::sha256(spki_der))

75/// Declares a fieldless enum with a stable text label and a stable one-byte
76/// wire code for each variant.
77macro_rules! labelled_enum {
78    (
79        $(#[$meta:meta])*
80        pub enum $name:ident ($what:literal) {
81            $( $(#[$vmeta:meta])* $variant:ident = ($code:literal, $label:literal), )+
82        }
83    ) => {
84        $(#[$meta])*
85        #[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
86        pub enum $name {
87            $( $(#[$vmeta])* $variant, )+
88        }
89
90        impl $name {
91            /// Every variant, in declaration order.
92            pub const ALL: &'static [Self] = &[$(Self::$variant),+];
93
94            /// The text label.
95            #[must_use]
96            pub const fn label(self) -> &'static str {
97                match self { $(Self::$variant => $label,)+ }
98            }
99
100            /// The one-byte code used in signed manifests. Codes are never
101            /// reused.
102            #[must_use]
103            pub const fn code(self) -> u8 {
104                match self { $(Self::$variant => $code,)+ }
105            }
106
107            /// The variant with this wire code.
108            #[must_use]
109            pub const fn from_code(code: u8) -> Option<Self> {
110                match code { $($code => Some(Self::$variant),)+ _ => None }
111            }
112        }
113
114        impl fmt::Display for $name {
115            fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116                f.write_str(self.label())
117            }
118        }
119
120        impl FromStr for $name {
121            type Err = KeyError;
122
123            fn from_str(s: &str) -> Result<Self, KeyError> {
124                match s {
125                    $($label => Ok(Self::$variant),)+
126                    _ => Err(KeyError::UnknownLabel($what)),
127                }
128            }
129        }
130    };
131}

195labelled_enum! {
196    /// A key's lifecycle state.
197    ///
198    /// Legal transitions:
199    ///
200    /// | From | To |
201    /// |---|---|
202    /// | `pending` | `active`, `retired` (never used), `compromised` |
203    /// | `active` | `retiring`, `compromised` |
204    /// | `retiring` | `retired`, `active` (rotation aborted), `compromised` |
205    /// | `retired` | `compromised` |
206    /// | `compromised` | none |
207    ///
208    /// Only `active` keys sign. `active` and `retiring` keys verify, which
209    /// gives rotations their overlap. `compromised` is final.
210    pub enum KeyStatus ("key status") {
211        /// Generated or imported, not yet in use.
212        Pending = (1, "pending"),
213        /// Signs and verifies.
214        Active = (2, "active"),
215        /// Verifies only, while its successor takes over.
216        Retiring = (3, "retiring"),
217        /// No longer used.
218        Retired = (4, "retired"),
219        /// Must never be trusted again.
220        Compromised = (5, "compromised"),
221    }
222}

224impl KeyPurpose {
225    /// Whether a key of this purpose may use `alg`.
226    #[must_use]
227    pub const fn allows(self, alg: KeyAlgorithm) -> bool {
228        match self {
229            Self::PolicySigning | Self::UnlockToken => {
230                matches!(alg, KeyAlgorithm::EcdsaP384Sha384 | KeyAlgorithm::Ed25519)
231            }
232            Self::Escrow => matches!(alg, KeyAlgorithm::EcdhP384),
233            Self::RootCa
234            | Self::IssuingCa
235            | Self::ServerTls
236            | Self::TrustSet
237            | Self::UpdateSigning
238            | Self::AuditChain => matches!(alg, KeyAlgorithm::EcdsaP384Sha384),
239        }
240    }
241}

243impl KeyAlgorithm {
244    /// The signature algorithm, or `None` for an encryption key.
245    #[must_use]
246    pub const fn signature(self) -> Option<SignatureAlgorithm> {
247        match self {
248            Self::EcdsaP384Sha384 => Some(SignatureAlgorithm::EcdsaP384Sha384),
249            Self::Ed25519 => Some(SignatureAlgorithm::Ed25519),
250            Self::EcdhP384 => None,
251        }
252    }
253
254    /// The type of public key this algorithm uses.
255    #[must_use]
256    pub const fn key_type(self) -> KeyType {
257        match self {
258            Self::EcdsaP384Sha384 | Self::EcdhP384 => KeyType::P384,
259            Self::Ed25519 => KeyType::Ed25519,
260        }
261    }
262}

264impl SignatureAlgorithm {
265    /// The key algorithm that produces this signature.
266    #[must_use]
267    pub const fn key_algorithm(self) -> KeyAlgorithm {
268        match self {
269            Self::EcdsaP384Sha384 => KeyAlgorithm::EcdsaP384Sha384,
270            Self::Ed25519 => KeyAlgorithm::Ed25519,
271        }
272    }
273
274    /// The exact length of a signature, in bytes.
275    #[must_use]
276    pub const fn signature_len(self) -> usize {
277        match self {
278            Self::EcdsaP384Sha384 => 96,
279            Self::Ed25519 => 64,
280        }
281    }

284impl KeyStatus {
285    /// Whether moving from `self` to `next` is a legal transition.
286    #[must_use]
287    pub const fn can_become(self, next: Self) -> bool {
288        matches!(
289            (self, next),
290            (
291                Self::Pending,
292                Self::Active | Self::Retired | Self::Compromised
293            ) | (Self::Active, Self::Retiring | Self::Compromised)
294                | (
295                    Self::Retiring,
296                    Self::Retired | Self::Active | Self::Compromised
297                )
298                | (Self::Retired, Self::Compromised)
299        )
300    }
301
302    /// Whether a key in this state may make new signatures.
303    #[must_use]
304    pub const fn signs(self) -> bool {
305        matches!(self, Self::Active)
306    }
307
308    /// Whether signatures by a key in this state are still accepted.
309    #[must_use]
310    pub const fn verifies(self) -> bool {
311        matches!(self, Self::Active | Self::Retiring)
312    }
313}

315/// The type of a public key.
316#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
317pub enum KeyType {
318    /// An elliptic-curve key on NIST P-384.
319    P384,
320    /// An Ed25519 key.
321    Ed25519,
322}
323
324/// DER prefix of a P-384 `SubjectPublicKeyInfo` (id-ecPublicKey with the
325/// named curve secp384r1, then the BIT STRING header). The uncompressed
326/// point (`04 ‖ x ‖ y`, 97 bytes) follows.
327const P384_SPKI_PREFIX: [u8; 23] = [
328    0x30, 0x76, 0x30, 0x10, 0x06, 0x07, 0x2a, 0x86, 0x48, 0xce, 0x3d, 0x02, 0x01, 0x06, 0x05, 0x2b,
329    0x81, 0x04, 0x00, 0x22, 0x03, 0x62, 0x00,
330];
331const P384_POINT_LEN: usize = 97;
332
333/// DER prefix of an Ed25519 `SubjectPublicKeyInfo` (RFC 8410). The 32-byte
334/// key follows.
335const ED25519_SPKI_PREFIX: [u8; 12] = [
336    0x30, 0x2a, 0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70, 0x03, 0x21, 0x00,
337];
338const ED25519_KEY_LEN: usize = 32;
339
340/// The longest `SubjectPublicKeyInfo` accepted, in bytes (P-384).
341pub const MAX_SPKI_LEN: usize = P384_SPKI_PREFIX.len() + P384_POINT_LEN;
342
343/// A public key, held as its DER `SubjectPublicKeyInfo`.
344///
345/// Only two encodings are accepted, each byte for byte: a P-384 key with the
346/// named curve and an uncompressed point, and an Ed25519 key. Every key
347/// therefore has exactly one encoding and so exactly one [`Kid`]. A point
348/// that is not on the curve is caught at verification, which then fails.
349#[derive(Clone, PartialEq, Eq, Hash)]
350pub struct PublicKey {
351    key_type: KeyType,
352    spki: Vec<u8>,
353}
354
355impl PublicKey {
356    /// Reads a DER `SubjectPublicKeyInfo`.
357    ///
358    /// # Errors
359    ///
360    /// [`KeyError::UnsupportedPublicKey`] for anything but the two accepted
361    /// encodings.
362    pub fn from_spki_der(der: &[u8]) -> Result<Self, KeyError> {
363        let key_type = if der.len() == P384_SPKI_PREFIX.len() + P384_POINT_LEN
364            && der.starts_with(&P384_SPKI_PREFIX)
365            && der.get(P384_SPKI_PREFIX.len()) == Some(&0x04)
366        {
367            KeyType::P384
368        } else if der.len() == ED25519_SPKI_PREFIX.len() + ED25519_KEY_LEN
369            && der.starts_with(&ED25519_SPKI_PREFIX)
370        {
371            KeyType::Ed25519
372        } else {
373            return Err(KeyError::UnsupportedPublicKey);
374        };
375        Ok(Self {
376            key_type,
377            spki: der.to_vec(),
378        })
379    }
380
381    /// Builds the `SubjectPublicKeyInfo` around a raw key: the uncompressed
382    /// P-384 point, or the 32-byte Ed25519 key.
383    pub(crate) fn from_raw(key_type: KeyType, raw: &[u8]) -> Result<Self, KeyError> {
384        let prefix: &[u8] = match key_type {
385            KeyType::P384 => &P384_SPKI_PREFIX,
386            KeyType::Ed25519 => &ED25519_SPKI_PREFIX,
387        };
388        let mut der = Vec::with_capacity(prefix.len() + raw.len());
389        der.extend_from_slice(prefix);
390        der.extend_from_slice(raw);
391        let key = Self::from_spki_der(&der)?;
392        if key.key_type == key_type {
393            Ok(key)
394        } else {
395            Err(KeyError::UnsupportedPublicKey)
396        }
397    }
398
399    /// The key type.
400    #[must_use]
401    pub const fn key_type(&self) -> KeyType {
402        self.key_type
403    }
404
405    /// The DER `SubjectPublicKeyInfo`.
406    #[must_use]
407    pub fn spki_der(&self) -> &[u8] {
408        &self.spki
409    }
410
411    /// The key ID.
412    #[must_use]
413    pub fn kid(&self) -> Kid {
414        Kid::of_spki(&self.spki)
415    }
416
417    /// The SHA-256 of the `SubjectPublicKeyInfo`.
418    #[must_use]
419    pub fn spki_sha256(&self) -> Sha256Hex {
420        spki_sha256(&self.spki)
421    }
422
423    /// The raw key after the fixed prefix.
424    pub(crate) fn raw(&self) -> &[u8] {
425        let prefix_len = match self.key_type {
426            KeyType::P384 => P384_SPKI_PREFIX.len(),
427            KeyType::Ed25519 => ED25519_SPKI_PREFIX.len(),
428        };
429        self.spki.get(prefix_len..).unwrap_or_default()
430    }
431
432    /// Verifies a signature over `domain`'s prefix followed by `payload`.
433    ///
434    /// Returns `false` if the algorithm does not fit this key, or the
435    /// signature is invalid or has the wrong length.
436    #[must_use]
437    pub fn verify(
438        &self,
439        alg: SignatureAlgorithm,
440        domain: super::Domain,
441        payload: &[u8],
442        signature: &[u8],
443    ) -> bool {
444        if alg.key_algorithm().key_type() != self.key_type || signature.len() != alg.signature_len()
445        {
446            return false;
447        }
448        crypto::verify(alg, self.raw(), &domain.message(payload), signature)
449    }
450}

454        write!(f, "PublicKey({:?}, kid {})", self.key_type, self.kid())
455    }
456}

458/// One key in the inventory: its public half and everything recorded about
459/// it. The private half lives in a [`Signer`](super::Signer) back end.
460#[derive(Clone, Debug, PartialEq, Eq)]
461pub struct KeyRecord {
462    public_key: PublicKey,
463    purpose: KeyPurpose,
464    algorithm: KeyAlgorithm,
465    created_at: Timestamp,
466    validity: Validity,
467    status: KeyStatus,
468    backend: KeyBackend,
469}
470
471impl KeyRecord {
472    /// A new record, in state [`KeyStatus::Pending`].
473    ///
474    /// # Errors
475    ///
476    /// [`KeyError::AlgorithmNotAllowed`] if the purpose does not allow the
477    /// algorithm (ADR 0008, #224), and [`KeyError::KeyTypeMismatch`] if the
478    /// public key is not of the algorithm's type.
479    pub fn new(
480        public_key: PublicKey,
481        purpose: KeyPurpose,
482        algorithm: KeyAlgorithm,
483        created_at: Timestamp,
484        validity: Validity,
485        backend: KeyBackend,
486    ) -> Result<Self, KeyError> {
487        if !purpose.allows(algorithm) {
488            return Err(KeyError::AlgorithmNotAllowed { purpose, algorithm });
489        }
490        if algorithm.key_type() != public_key.key_type() {
491            return Err(KeyError::KeyTypeMismatch);
492        }
493        Ok(Self {
494            public_key,
495            purpose,
496            algorithm,
497            created_at,
498            validity,
499            status: KeyStatus::Pending,
500            backend,
501        })
502    }
503
504    /// The key ID.
505    #[must_use]
506    pub fn kid(&self) -> Kid {
507        self.public_key.kid()
508    }
509
510    /// The public key.
511    #[must_use]
512    pub const fn public_key(&self) -> &PublicKey {
513        &self.public_key
514    }
515
516    /// The purpose.
517    #[must_use]
518    pub const fn purpose(&self) -> KeyPurpose {
519        self.purpose
520    }
521
522    /// The algorithm.
523    #[must_use]
524    pub const fn algorithm(&self) -> KeyAlgorithm {
525        self.algorithm
526    }
527
528    /// When the key was generated or imported.
529    #[must_use]
530    pub const fn created_at(&self) -> Timestamp {
531        self.created_at
532    }
533
534    /// The validity window.
535    #[must_use]
536    pub const fn validity(&self) -> Validity {
537        self.validity
538    }
539
540    /// The lifecycle state.
541    #[must_use]
542    pub const fn status(&self) -> KeyStatus {
543        self.status
544    }
545
546    /// Where the private key lives.
547    #[must_use]
548    pub const fn backend(&self) -> KeyBackend {
549        self.backend
550    }
551
552    /// Moves the key to `next`.
553    ///
554    /// # Errors
555    ///
556    /// [`KeyError::IllegalTransition`] if [`KeyStatus::can_become`] forbids
557    /// it. The record is unchanged.
558    pub fn transition(&mut self, next: KeyStatus) -> Result<(), KeyError> {
559        if self.status.can_become(next) {
560            self.status = next;
561            Ok(())
562        } else {
563            Err(KeyError::IllegalTransition {
564                from: self.status,
565                to: next,
566            })
567        }
568    }
569
570    /// Whether the key may sign at `now`: it is a signing key (not the
571    /// escrow encryption key), active, and inside its validity window, with
572    /// no skew allowance. Signing happens on the server, whose clock is
573    /// synchronised.
574    #[must_use]
575    pub fn can_sign(&self, now: Timestamp) -> bool {
576        self.algorithm.signature().is_some() && self.status.signs() && self.validity.contains(now)
577    }
578}

580/// A device or relay certificate as the server enrolled it: its serial and
581/// the SHA-256 of its public key (threat: T-ca-02).
582///
583/// A presented certificate matches only if **both** agree, so a certificate
584/// minted by a stolen issuing CA with a reused serial and a new key is
585/// refused. Wiring this into the mTLS handshake is #223.
586#[derive(Clone, Debug, PartialEq, Eq)]
587pub struct CertificateBinding {
588    serial: Vec<u8>,
589    spki_sha256: Sha256Hex,
590}
591
592impl CertificateBinding {
593    /// The longest serial accepted, in bytes (RFC 5280 allows 20 octets).
594    pub const MAX_SERIAL_LEN: usize = 20;
595
596    /// Records a certificate's serial (the DER INTEGER content octets) and
597    /// public key.
598    ///
599    /// # Errors
600    ///
601    /// [`KeyError::InvalidSerial`] if the serial is empty or longer than
602    /// [`Self::MAX_SERIAL_LEN`].
603    pub fn new(serial: &[u8], spki_der: &[u8]) -> Result<Self, KeyError> {
604        if serial.is_empty() || serial.len() > Self::MAX_SERIAL_LEN {
605            return Err(KeyError::InvalidSerial);
606        }
607        Ok(Self {
608            serial: serial.to_vec(),
609            spki_sha256: spki_sha256(spki_der),
610        })
611    }
612
613    /// The SHA-256 of the enrolled public key.
614    #[must_use]
615    pub const fn spki_sha256(&self) -> &Sha256Hex {
616        &self.spki_sha256
617    }
618
619    /// Whether a presented certificate's serial and public key both match.
620    #[must_use]
621    pub fn matches(&self, serial: &[u8], spki_der: &[u8]) -> bool {
622        self.serial == serial && self.spki_sha256 == spki_sha256(spki_der)
623    }
624}

626/// A key-inventory value is invalid.
627#[derive(Clone, Copy, Debug, PartialEq, Eq)]
628pub enum KeyError {
629    /// A key ID is not 32 lowercase hex digits.
630    InvalidKid,
631    /// A text label is not one of the known values; names which kind.
632    UnknownLabel(&'static str),
633    /// The public key is not a P-384 or Ed25519 key in the accepted encoding.
634    UnsupportedPublicKey,
635    /// The purpose does not allow the algorithm.
636    AlgorithmNotAllowed {
637        /// The key's purpose.
638        purpose: KeyPurpose,
639        /// The rejected algorithm.
640        algorithm: KeyAlgorithm,
641    },
642    /// The public key is not of the algorithm's key type.
643    KeyTypeMismatch,
644    /// The lifecycle does not allow this transition.
645    IllegalTransition {
646        /// The current state.
647        from: KeyStatus,
648        /// The requested state.
649        to: KeyStatus,
650    },
651    /// A certificate serial is empty or too long.
652    InvalidSerial,
653}
654
655impl fmt::Display for KeyError {
656    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
657        match self {
658            Self::InvalidKid => f.write_str("a key ID is 32 lowercase hexadecimal digits"),
659            Self::UnknownLabel(what) => write!(f, "unknown {what}"),
660            Self::UnsupportedPublicKey => {
661                f.write_str("expected a P-384 or Ed25519 SubjectPublicKeyInfo")
662            }
663            Self::AlgorithmNotAllowed { purpose, algorithm } => {
664                write!(f, "a {purpose} key may not use {algorithm}")
665            }
666            Self::KeyTypeMismatch => f.write_str("the public key does not fit the algorithm"),
667            Self::IllegalTransition { from, to } => {
668                write!(f, "a key cannot go from {from} to {to}")
669            }
670            Self::InvalidSerial => f.write_str("a certificate serial is 1 to 20 bytes"),
671        }
672    }
673}
