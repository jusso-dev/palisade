1//! Key inventory, signed trust sets and signing for Palisade's internal PKI
2//! (part of #224).
3//!
4//! Both the agent and the server need this code: the server builds and signs
5//! trust sets, and the agent verifies them. ADR 0003 puts code shared by the
6//! sibling binaries in `palisade-proto`. It cannot go in `palisade-core`,
7//! which has no third-party dependencies and builds `no_std` for the kernel
8//! driver, while signatures need `ring` (ADR 0008).
9//!
10//! - [`key`]: key IDs, purposes, algorithms, the key lifecycle and the
11//!   device-certificate binding.
12//! - [`trust_set`]: the signed trust-set manifest, its canonical encoding,
13//!   and the agent's acceptance rule.
14//! - [`signer`]: the [`Signer`] interface and an in-memory software back end.
15//! - [`lifecycle`]: renewal, expiry-alert and rotation-overlap schedules,
16//!   and trust-version adoption.
17//! - [`time`]: validity windows with clock-skew tolerance, and the clock.
18//! - `issue` (feature `issuance`): X.509 certificates and revocation lists
19//!   for the CA hierarchy, signed through a [`Signer`].
20//!
21//! Every signature covers a [`Domain`] prefix followed by the payload, so a
22//! signature made for one purpose can never verify as another
23//! (threat: T-signer-02). The only module that calls `ring` is the
24//! crate-private `crypto` module.

57/// What a signature is for. Each domain has its own prefix, and only keys of
58/// the matching purposes may sign for it (threats: T-signer-02, T-signer-06).
59#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
60pub enum Domain {
61    /// A trust-set manifest. Signed by `root-ca` or `trust-set` keys.
62    TrustSet,
63    /// A policy bundle (#225). Signed by `policy-signing` keys.
64    Policy,
65    /// A break-glass unlock token (#222). Signed by `unlock-token` keys.
66    Unlock,
67}
68
69impl Domain {
70    /// Every domain.
71    pub const ALL: &'static [Self] = &[Self::TrustSet, Self::Policy, Self::Unlock];
72
73    /// The domain-separation prefix, including its terminating NUL byte.
74    #[must_use]
75    pub const fn prefix(self) -> &'static [u8] {
76        match self {
77            Self::TrustSet => b"palisade-trustset-v1\0",
78            Self::Policy => b"palisade-policy-v1\0",
79            Self::Unlock => b"palisade-unlock-v1\0",
80        }
81    }
82
83    /// The prefix without its NUL, for messages.
84    #[must_use]
85    pub const fn label(self) -> &'static str {
86        match self {
87            Self::TrustSet => "palisade-trustset-v1",
88            Self::Policy => "palisade-policy-v1",
89            Self::Unlock => "palisade-unlock-v1",
90        }
91    }
92
93    /// Whether a key of `purpose` may sign for this domain.
94    #[must_use]
95    pub const fn signed_by(self, purpose: KeyPurpose) -> bool {
96        matches!(
97            (self, purpose),
98            (Self::TrustSet, KeyPurpose::RootCa | KeyPurpose::TrustSet)
99                | (Self::Policy, KeyPurpose::PolicySigning)
100                | (Self::Unlock, KeyPurpose::UnlockToken)
101        )
102    }
103
104    /// The signed message: the prefix followed by `payload`.
105    pub(crate) fn message(self, payload: &[u8]) -> Vec<u8> {
106        let prefix = self.prefix();
107        let mut message = Vec::with_capacity(prefix.len().saturating_add(payload.len()));
108        message.extend_from_slice(prefix);
109        message.extend_from_slice(payload);
110        message
111    }
112}
113
114impl fmt::Display for Domain {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116        f.write_str(self.label())
117    }
118}
