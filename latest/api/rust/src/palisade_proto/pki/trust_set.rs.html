1//! The signed trust-set manifest and the agent's acceptance rule.
2//!
3//! A trust set tells an agent which keys it trusts: the keys that may sign
4//! the next trust set, the server CA public-key pins, the policy-signing and
5//! unlock-token keys, the signature algorithms it accepts, and a pointer to
6//! the update-signing root metadata (#226). It is signed by the offline root
7//! or by a trust-set key that an earlier trust set endorsed.
8//!
9//! # Acceptance rule
10//!
11//! [`TrustAnchor::accept`] takes a new signed trust set only if all of these
12//! hold (threats: T-signer-06, T-ca-01):
13//!
14//! 1. The envelope and manifest decode under the strict rules below.
15//! 2. At least one signature verifies under a key the agent **already**
16//!    trusts: the pinned root, or a `root-ca` or `trust-set` key in the
17//!    *current* trust set. A key listed only in the new set never counts,
18//!    so a set cannot endorse itself.
19//! 3. A current-set key only counts if its validity window contains `now`,
20//!    allowing for clock skew. The pinned root has no window: it is trusted
21//!    until the device is re-enrolled (R-11).
22//! 4. Every signature by a key the agent knows (the pinned root or a current
23//!    trust key) must name that key's own algorithm, an algorithm the
24//!    current set accepts, and must verify, whether or not the key is inside
25//!    its validity window. A bad or mislabelled signature from a known key is
26//!    tampering, not noise, and rejects the whole set. Signatures from
27//!    unknown keys are ignored, which lets a set carry signatures for agents
28//!    on different trust versions during a rotation.
29//! 5. `trust_version` is strictly greater than the current one, so an
30//!    older or replayed set is refused (rollback), and at most
31//!    [`MAX_VERSION_STEP`] greater. Without the cap, a stolen trust-set key
32//!    could publish version `u64::MAX` and no later set, not even one signed
33//!    by the root, could ever be newer (fast-forward). The cap applies to
34//!    root-signed sets too, so recovering agents that a stolen key moved
35//!    far ahead of the rest may need intermediate sets.
36//! 6. `issued_at` is not earlier than the current set's, and not later than
37//!    `now` plus the skew tolerance.
38//!
39//! `now` must be trusted time on the agent (#220). Storing the accepted set
40//! and the last two versions in the protected data directory is also #220.
41//!
42//! # Canonical encoding
43//!
44//! The signed payload is a fixed-layout binary encoding with one valid
45//! encoding per trust set. The decoder enforces that: every list is sorted
46//! strictly ascending (so duplicates are refused), every count and length is
47//! bounded, every code is known, and no bytes may follow the last field.
48//! `encode(decode(bytes)) == bytes` for every accepted input.
49//!
50//! All integers are big-endian. Timestamps are signed 64-bit microseconds
51//! since 1970-01-01T00:00:00Z, within years 0001 to 9999.
52//!
53//! ```text
54//! payload    = version:u8 (= 1)
55//!              trust_version:u64 (>= 1)
56//!              issued_at:i64
57//!              n_algs:u8 (1..=2)   alg:u8 * n_algs          ascending
58//!              n_trust:u8 (1..=64) key * n_trust            ascending kid
59//!              n_pins:u8 (0..=64)  sha256:[u8;32] * n_pins  ascending
60//!              n_policy:u8 (0..=64) key * n_policy          ascending kid
61//!              n_unlock:u8 (0..=64) key * n_unlock          ascending kid
62//!              update_root:u8 (0 | 1)
63//!                [version:u64 sha256:[u8;32]]               if update_root = 1
64//! key        = purpose:u8 status:u8 alg:u8
65//!              not_before:i64 not_after:i64
66//!              spki_len:u8 spki:[u8; spki_len]
67//!
68//! envelope   = version:u8 (= 1)
69//!              payload_len:u32 (1..=65536) payload
70//!              n_sigs:u8 (1..=8) sig * n_sigs               ascending kid
71//! sig        = kid:[u8;16] alg:u8 bytes:[u8; length fixed by alg]
72//! ```
73//!
74//! Codes are [`KeyPurpose::code`], [`KeyStatus::code`] and
75//! [`SignatureAlgorithm::code`]. A key ID appears at most once across all
76//! three key lists, so no key serves two purposes. Each list holds only its
77//! own purposes: `root-ca` or `trust-set`, then `policy-signing`, then
78//! `unlock-token`. Listed keys are `active` or `retiring`: a retired or
79//! compromised key is revoked by leaving it out of the next trust set.
80//!
81//! The issue suggests reusing the policy bundle's canonical CBOR (#225).
82//! That encoder does not exist yet. This format is versioned, so a later
83//! version can move to it without breaking agents on version 1.

97/// The payload and envelope format version this module reads and writes.
98pub const FORMAT_VERSION: u8 = 1;
99/// The most `trust_version` may grow in one accepted update. Servers
100/// increment it by one per publication, so an agent can miss up to a million
101/// publications and still catch up in one step.
102pub const MAX_VERSION_STEP: u64 = 1_000_000;
103/// Largest envelope accepted, in bytes: the largest payload, its header
104/// and the most signatures.
105pub const MAX_ENVELOPE_LEN: usize = 1 + 4 + MAX_PAYLOAD_LEN + 1 + MAX_SIGNATURES * (16 + 1 + 96);
106/// Most keys in each key list.
107pub const MAX_KEYS: usize = 64;
108/// Most server CA pins.
109pub const MAX_PINS: usize = 64;
110/// Most signatures on one trust set.
111pub const MAX_SIGNATURES: usize = 8;
112/// Largest payload accepted, in bytes. A manifest with every list full is about 29 KiB.
113pub const MAX_PAYLOAD_LEN: usize = 64 * 1024;

115/// A key listed in a trust set.
116#[derive(Clone, Debug, PartialEq, Eq)]
117pub struct TrustedKey {
118    public_key: PublicKey,
119    purpose: KeyPurpose,
120    algorithm: SignatureAlgorithm,
121    status: KeyStatus,
122    validity: Validity,
123}
124
125impl TrustedKey {
126    /// A listed key.
127    ///
128    /// # Errors
129    ///
130    /// [`TrustSetError::Key`] if the purpose does not allow the algorithm or
131    /// the public key is of the wrong type, and
132    /// [`TrustSetError::StatusNotTrusted`] unless the status is `active` or
133    /// `retiring`.
134    pub fn new(
135        public_key: PublicKey,
136        purpose: KeyPurpose,
137        algorithm: SignatureAlgorithm,
138        status: KeyStatus,
139        validity: Validity,
140    ) -> Result<Self, TrustSetError> {
141        let key_alg = algorithm.key_algorithm();
142        if !purpose.allows(key_alg) {
143            return Err(TrustSetError::Key(KeyError::AlgorithmNotAllowed {
144                purpose,
145                algorithm: key_alg,
146            }));
147        }
148        if key_alg.key_type() != public_key.key_type() {
149            return Err(TrustSetError::Key(KeyError::KeyTypeMismatch));
150        }
151        if !status.verifies() {
152            return Err(TrustSetError::StatusNotTrusted(status));
153        }
154        Ok(Self {
155            public_key,
156            purpose,
157            algorithm,
158            status,
159            validity,
160        })
161    }

210/// Which key list a key belongs in.
211#[derive(Clone, Copy, Debug, PartialEq, Eq)]
212enum List {
213    Trust,
214    Policy,
215    Unlock,
216}

218impl List {
219    const fn holds(self, purpose: KeyPurpose) -> bool {
220        matches!(
221            (self, purpose),
222            (Self::Trust, KeyPurpose::RootCa | KeyPurpose::TrustSet)
223                | (Self::Policy, KeyPurpose::PolicySigning)
224                | (Self::Unlock, KeyPurpose::UnlockToken)
225        )
226    }
227
228    const fn name(self) -> &'static str {
229        match self {
230            Self::Trust => "trust keys",
231            Self::Policy => "policy keys",
232            Self::Unlock => "unlock keys",
233        }
234    }
235
236    const fn for_domain(domain: Domain) -> Self {
237        match domain {
238            Domain::TrustSet => Self::Trust,
239            Domain::Policy => Self::Policy,
240            Domain::Unlock => Self::Unlock,
241        }
242    }
243}

245/// A trust-set manifest. The constructors check every rule the decoder
246/// checks, so every value has exactly one encoding, except that a set
247/// without trust keys is refused by [`Self::encode`] and by
248/// [`TrustAnchor::with_current`].
249#[derive(Clone, Debug, PartialEq, Eq)]
250pub struct TrustSet {
251    trust_version: u64,
252    issued_at: Timestamp,
253    algorithms: BTreeSet<SignatureAlgorithm>,
254    trust_keys: BTreeMap<Kid, TrustedKey>,
255    server_pins: BTreeSet<Sha256Hex>,
256    policy_keys: BTreeMap<Kid, TrustedKey>,
257    unlock_keys: BTreeMap<Kid, TrustedKey>,
258    update_root: Option<UpdateRootPointer>,
259}

261impl TrustSet {
262    /// An empty trust set. Add at least one trust key before encoding.
263    ///
264    /// # Errors
265    ///
266    /// [`TrustSetError::ZeroVersion`] if `trust_version` is 0 (the version an
267    /// agent holds before its first trust set), and
268    /// [`TrustSetError::NoAlgorithms`] if `algorithms` is empty.
269    pub fn new(
270        trust_version: u64,
271        issued_at: Timestamp,
272        algorithms: &[SignatureAlgorithm],
273    ) -> Result<Self, TrustSetError> {
274        if trust_version == 0 {
275            return Err(TrustSetError::ZeroVersion);
276        }
277        if algorithms.is_empty() {
278            return Err(TrustSetError::NoAlgorithms);
279        }
280        Ok(Self {
281            trust_version,
282            issued_at,
283            algorithms: algorithms.iter().copied().collect(),
284            trust_keys: BTreeMap::new(),
285            server_pins: BTreeSet::new(),
286            policy_keys: BTreeMap::new(),
287            unlock_keys: BTreeMap::new(),
288            update_root: None,
289        })
290    }

319    /// Adds a server CA pin: the SHA-256 of an accepted CA's
320    /// `SubjectPublicKeyInfo` (#223).
321    ///
322    /// # Errors
323    ///
324    /// [`TrustSetError::TooMany`] or [`TrustSetError::DuplicatePin`].
325    pub fn add_server_pin(&mut self, pin: Sha256Hex) -> Result<(), TrustSetError> {
326        if self.server_pins.len() >= MAX_PINS {
327            return Err(TrustSetError::TooMany("server pins"));
328        }
329        if !self.server_pins.insert(pin) {
330            return Err(TrustSetError::DuplicatePin(pin));
331        }
332        Ok(())
333    }

335    /// Sets the update-signing root metadata pointer.
336    pub fn set_update_root(&mut self, pointer: UpdateRootPointer) {
337        self.update_root = Some(pointer);
338    }

340    fn add_key(&mut self, list: List, key: TrustedKey) -> Result<(), TrustSetError> {
341        if !list.holds(key.purpose) {
342            return Err(TrustSetError::WrongList {
343                list: list.name(),
344                purpose: key.purpose,
345            });
346        }
347        if !self.algorithms.contains(&key.algorithm) {
348            return Err(TrustSetError::AlgorithmNotListed(key.algorithm));
349        }
350        let kid = key.kid();
351        if self.trust_keys.contains_key(&kid)
352            || self.policy_keys.contains_key(&kid)
353            || self.unlock_keys.contains_key(&kid)
354        {
355            return Err(TrustSetError::DuplicateKey(kid));
356        }
357        let map = match list {
358            List::Trust => &mut self.trust_keys,
359            List::Policy => &mut self.policy_keys,
360            List::Unlock => &mut self.unlock_keys,
361        };
362        if map.len() >= MAX_KEYS {
363            return Err(TrustSetError::TooMany(list.name()));
364        }
365        map.insert(kid, key);
366        Ok(())
367    }

419    /// Verifies a signature for `domain` against this set: the signing key
420    /// must be listed for that domain, its algorithm must be accepted, and
421    /// its validity window must contain `now`, allowing for `skew`.
422    ///
423    /// This is the check that policy bundles (#225) and unlock tokens (#222)
424    /// will use. For trust sets, use [`TrustAnchor::accept`] instead, which
425    /// also trusts the pinned root and enforces the version rules.
426    ///
427    /// # Errors
428    ///
429    /// See [`VerifyError`].
430    pub fn verify(
431        &self,
432        domain: Domain,
433        payload: &[u8],
434        signature: &Signature,
435        now: Timestamp,
436        skew: SkewTolerance,
437    ) -> Result<&TrustedKey, VerifyError> {
438        let key = self.listed(domain, &signature.kid())?;
439        if !self.algorithms.contains(&signature.algorithm()) {
440            return Err(VerifyError::AlgorithmNotAccepted(signature.algorithm()));
441        }
442        if signature.algorithm() != key.algorithm {
443            return Err(VerifyError::AlgorithmMismatch);
444        }
445        key.validity.check(now, skew).map_err(VerifyError::Time)?;
446        if key
447            .public_key
448            .verify(signature.algorithm(), domain, payload, signature.bytes())
449        {
450            Ok(key)
451        } else {
452            Err(VerifyError::BadSignature)
453        }
454    }

456    fn listed(&self, domain: Domain, kid: &Kid) -> Result<&TrustedKey, VerifyError> {
457        let map = match List::for_domain(domain) {
458            List::Trust => &self.trust_keys,
459            List::Policy => &self.policy_keys,
460            List::Unlock => &self.unlock_keys,
461        };
462        map.get(kid).ok_or(VerifyError::UnknownKey(*kid))
463    }

465    /// The canonical payload bytes, the input to signing.
466    ///
467    /// # Errors
468    ///
469    /// [`TrustSetError::NoTrustKeys`] if no trust key is listed. The pinned
470    /// root is outside the set, so a set without trust keys could still be
471    /// followed by a root-signed one, but an empty list is almost always a
472    /// mistake, so the format refuses it.
473    pub fn encode(&self) -> Result<Vec<u8>, TrustSetError> {
474        if self.trust_keys.is_empty() {
475            return Err(TrustSetError::NoTrustKeys);
476        }
477        let mut out = Vec::with_capacity(1024);
478        out.push(FORMAT_VERSION);
479        out.extend_from_slice(&self.trust_version.to_be_bytes());
480        out.extend_from_slice(&self.issued_at.unix_micros().to_be_bytes());
481        push_count(&mut out, self.algorithms.len());
482        for alg in &self.algorithms {
483            out.push(alg.code());
484        }
485        encode_keys(&mut out, &self.trust_keys);
486        push_count(&mut out, self.server_pins.len());
487        for pin in &self.server_pins {
488            out.extend_from_slice(pin.as_bytes());
489        }
490        encode_keys(&mut out, &self.policy_keys);
491        encode_keys(&mut out, &self.unlock_keys);
492        match self.update_root {
493            None => out.push(0),
494            Some(p) => {
495                out.push(1);
496                out.extend_from_slice(&p.version.to_be_bytes());
497                out.extend_from_slice(p.sha256.as_bytes());
498            }
499        }
500        Ok(out)
501    }

503    /// Decodes canonical payload bytes. Signatures are not checked here.
504    ///
505    /// # Errors
506    ///
507    /// [`DecodeError`] for any input that is not the canonical encoding of a
508    /// valid trust set.
509    pub fn decode(bytes: &[u8]) -> Result<Self, DecodeError> {
510        if bytes.len() > MAX_PAYLOAD_LEN {
511            return Err(DecodeError::TooLong);
512        }
513        let mut r = Reader(bytes);
514        let version = r.u8()?;
515        if version != FORMAT_VERSION {
516            return Err(DecodeError::UnsupportedVersion(version));
517        }
518        let trust_version = r.u64()?;
519        let issued_at = r.timestamp()?;
520
521        let n_algs = r.count(1, SignatureAlgorithm::ALL.len(), "algorithms")?;
522        let mut algorithms = Vec::with_capacity(n_algs);
523        for _ in 0..n_algs {
524            let code = r.u8()?;
525            let alg = SignatureAlgorithm::from_code(code)
526                .ok_or(DecodeError::UnknownCode("signature algorithm"))?;
527            if algorithms.last().is_some_and(|prev| *prev >= alg) {
528                return Err(DecodeError::NotAscending("algorithms"));
529            }
530            algorithms.push(alg);
531        }
532        let mut set = Self::new(trust_version, issued_at, &algorithms).map_err(DecodeError::Set)?;
533
534        decode_keys(&mut r, &mut set, List::Trust)?;
535
536        let n_pins = r.count(0, MAX_PINS, "server pins")?;
537        let mut last: Option<Sha256Hex> = None;
538        for _ in 0..n_pins {
539            let pin = Sha256Hex::from_bytes(r.array()?);
540            if last.is_some_and(|prev| prev >= pin) {
541                return Err(DecodeError::NotAscending("server pins"));
542            }
543            set.add_server_pin(pin).map_err(DecodeError::Set)?;
544            last = Some(pin);
545        }
546
547        decode_keys(&mut r, &mut set, List::Policy)?;
548        decode_keys(&mut r, &mut set, List::Unlock)?;
549
550        match r.u8()? {
551            0 => {}
552            1 => {
553                let version = r.u64()?;
554                let sha256 = Sha256Hex::from_bytes(r.array()?);
555                set.set_update_root(UpdateRootPointer { version, sha256 });
556            }
557            _ => return Err(DecodeError::UnknownCode("update root presence")),
558        }
559        r.finish()?;
560        Ok(set)
561    }
562}

564fn push_count(out: &mut Vec<u8>, n: usize) {
565    // Every list is capped at 64 entries, so the count fits a byte.
566    out.push(u8::try_from(n).unwrap_or(u8::MAX));
567}
568
569fn encode_keys(out: &mut Vec<u8>, keys: &BTreeMap<Kid, TrustedKey>) {
570    push_count(out, keys.len());
571    for key in keys.values() {
572        out.push(key.purpose.code());
573        out.push(key.status.code());
574        out.push(key.algorithm.code());
575        out.extend_from_slice(&key.validity.not_before().unix_micros().to_be_bytes());
576        out.extend_from_slice(&key.validity.not_after().unix_micros().to_be_bytes());
577        let spki = key.public_key.spki_der();
578        push_count(out, spki.len());
579        out.extend_from_slice(spki);
580    }
581}

583fn decode_keys(r: &mut Reader<'_>, set: &mut TrustSet, list: List) -> Result<(), DecodeError> {
584    let min = usize::from(list == List::Trust);
585    let n = r.count(min, MAX_KEYS, list.name())?;
586    let mut last: Option<Kid> = None;
587    for _ in 0..n {
588        let purpose =
589            KeyPurpose::from_code(r.u8()?).ok_or(DecodeError::UnknownCode("key purpose"))?;
590        let status = KeyStatus::from_code(r.u8()?).ok_or(DecodeError::UnknownCode("key status"))?;
591        let algorithm = SignatureAlgorithm::from_code(r.u8()?)
592            .ok_or(DecodeError::UnknownCode("signature algorithm"))?;
593        let not_before = r.timestamp()?;
594        let not_after = r.timestamp()?;
595        let validity = Validity::new(not_before, not_after).map_err(DecodeError::Time)?;
596        let spki_len = usize::from(r.u8()?);
597        if spki_len > MAX_SPKI_LEN {
598            return Err(DecodeError::TooLong);
599        }
600        let public_key = PublicKey::from_spki_der(r.take(spki_len)?)
601            .map_err(|e| DecodeError::Set(TrustSetError::Key(e)))?;
602        let key = TrustedKey::new(public_key, purpose, algorithm, status, validity)
603            .map_err(DecodeError::Set)?;
604        let kid = key.kid();
605        if last.is_some_and(|prev| prev >= kid) {
606            return Err(DecodeError::NotAscending(list.name()));
607        }
608        set.add_key(list, key).map_err(DecodeError::Set)?;
609        last = Some(kid);
610    }
611    Ok(())
612}

614/// A trust set with its signatures: the unit delivered to agents.

616pub struct SignedTrustSet {
617    payload: Vec<u8>,
618    signatures: Vec<Signature>,
619}

621impl SignedTrustSet {
622    /// Encodes `set` and signs it with each key in `kids`, in the
623    /// [`Domain::TrustSet`] domain.
624    ///
625    /// During a trust-key rotation, sign with both the old and the new key,
626    /// so agents on either side of the rotation accept the set.
627    ///
628    /// # Errors
629    ///
630    /// [`TrustSetError::Signer`] if signing fails (including a key whose
631    /// purpose may not sign trust sets), [`TrustSetError::SignatureCount`]
632    /// for no keys or more than [`MAX_SIGNATURES`], and
633    /// [`TrustSetError::DuplicateKey`] if a key is given twice.
634    pub fn sign<S: Signer + ?Sized>(
635        set: &TrustSet,
636        signer: &S,
637        kids: &[Kid],
638    ) -> Result<Self, TrustSetError> {
639        if kids.is_empty() || kids.len() > MAX_SIGNATURES {
640            return Err(TrustSetError::SignatureCount);
641        }
642        let payload = set.encode()?;
643        let mut signatures: BTreeMap<Kid, Signature> = BTreeMap::new();
644        for kid in kids {
645            let sig = signer
646                .sign(kid, Domain::TrustSet, &payload)
647                .map_err(TrustSetError::Signer)?;
648            if signatures.insert(*kid, sig).is_some() {
649                return Err(TrustSetError::DuplicateKey(*kid));
650            }
651        }
652        Ok(Self {
653            payload,
654            signatures: signatures.into_values().collect(),
655        })
656    }
657
658    /// Joins a payload and signatures made elsewhere, for example by an
659    /// offline root ceremony.
660    ///
661    /// # Errors
662    ///
663    /// As [`Self::decode`] would report for the joined envelope.
664    pub fn from_parts(payload: Vec<u8>, signatures: Vec<Signature>) -> Result<Self, DecodeError> {
665        let mut sigs = signatures;
666        sigs.sort_by_key(Signature::kid);
667        let joined = Self {
668            payload,
669            signatures: sigs,
670        };
671        Self::decode(&joined.encode())
672    }
673
674    /// The signed payload bytes.
675    #[must_use]
676    pub fn payload(&self) -> &[u8] {
677        &self.payload
678    }
679
680    /// The signatures, ascending by key ID.
681    #[must_use]
682    pub fn signatures(&self) -> &[Signature] {
683        &self.signatures
684    }
685
686    /// The envelope bytes.
687    #[must_use]
688    pub fn encode(&self) -> Vec<u8> {
689        let mut out = Vec::with_capacity(self.payload.len().saturating_add(256));
690        out.push(FORMAT_VERSION);
691        // The payload is at most 64 KiB, so its length fits a u32.
692        let len = u32::try_from(self.payload.len()).unwrap_or(u32::MAX);
693        out.extend_from_slice(&len.to_be_bytes());
694        out.extend_from_slice(&self.payload);
695        push_count(&mut out, self.signatures.len());
696        for sig in &self.signatures {
697            out.extend_from_slice(sig.kid().as_bytes());
698            out.push(sig.algorithm().code());
699            out.extend_from_slice(sig.bytes());
700        }
701        out
702    }
703
704    /// Decodes envelope bytes. The payload is checked to be canonical, but
705    /// signatures are not verified: use [`TrustAnchor::accept`].
706    ///
707    /// # Errors
708    ///
709    /// [`DecodeError`] for anything but a canonical envelope around a
710    /// canonical payload.
711    pub fn decode(bytes: &[u8]) -> Result<Self, DecodeError> {
712        Self::decode_parts(bytes).map(|(signed, _)| signed)
713    }
714
715    /// Decodes envelope bytes and returns the decoded payload too, so it is
716    /// parsed once.
717    fn decode_parts(bytes: &[u8]) -> Result<(Self, TrustSet), DecodeError> {
718        if bytes.len() > MAX_ENVELOPE_LEN {
719            return Err(DecodeError::TooLong);
720        }
721        let mut r = Reader(bytes);
722        let version = r.u8()?;
723        if version != FORMAT_VERSION {
724            return Err(DecodeError::UnsupportedVersion(version));
725        }
726        let len = usize::try_from(r.u32()?).map_err(|_| DecodeError::TooLong)?;
727        if len == 0 {
728            return Err(DecodeError::Truncated);
729        }
730        if len > MAX_PAYLOAD_LEN {
731            return Err(DecodeError::TooLong);
732        }
733        let payload = r.take(len)?.to_vec();
734        // The payload must be canonical before anyone looks at signatures.
735        let set = TrustSet::decode(&payload)?;
736        let n = r.count(1, MAX_SIGNATURES, "signatures")?;
737        let mut signatures: Vec<Signature> = Vec::with_capacity(n);
738        for _ in 0..n {
739            let kid = Kid::from_bytes(r.array()?);
740            let alg = SignatureAlgorithm::from_code(r.u8()?)
741                .ok_or(DecodeError::UnknownCode("signature algorithm"))?;
742            let sig_bytes = r.take(alg.signature_len())?.to_vec();
743            if signatures.last().is_some_and(|prev| prev.kid() >= kid) {
744                return Err(DecodeError::NotAscending("signatures"));
745            }
746            let sig = Signature::new(kid, alg, sig_bytes)
747                .map_err(|e| DecodeError::Set(TrustSetError::Signer(e)))?;
748            signatures.push(sig);
749        }
750        r.finish()?;
751        Ok((
752            Self {
753                payload,
754                signatures,
755            },
756            set,
757        ))
758    }
759}

761/// What an agent trusts: the root pinned at enrolment, and the trust set it
762/// last accepted.
763#[derive(Clone, Debug)]
764pub struct TrustAnchor {
765    pinned_root: PublicKey,
766    current: Option<TrustSet>,
767    skew: SkewTolerance,
768}
769
770impl TrustAnchor {
771    /// An anchor with only the pinned root, as at enrolment.
772    ///
773    /// # Errors
774    ///
775    /// [`TrustSetError::Key`] unless the root is a P-384 key, the only
776    /// algorithm a `root-ca` key may use.
777    pub fn new(pinned_root: PublicKey, skew: SkewTolerance) -> Result<Self, TrustSetError> {
778        if pinned_root.key_type() != KeyType::P384 {
779            return Err(TrustSetError::Key(KeyError::KeyTypeMismatch));
780        }
781        Ok(Self {
782            pinned_root,
783            current: None,
784            skew,
785        })
786    }
787
788    /// An anchor restored from storage: the pinned root and the last
789    /// accepted trust set.
790    ///
791    /// # Errors
792    ///
793    /// As [`Self::new`], plus [`TrustSetError::NoTrustKeys`] if `current`
794    /// lists no trust key, which no decoded trust set can.
795    pub fn with_current(
796        pinned_root: PublicKey,
797        current: TrustSet,
798        skew: SkewTolerance,
799    ) -> Result<Self, TrustSetError> {
800        if current.trust_keys.is_empty() {
801            return Err(TrustSetError::NoTrustKeys);
802        }
803        let mut anchor = Self::new(pinned_root, skew)?;
804        anchor.current = Some(current);
805        Ok(anchor)
806    }
807
808    /// The pinned root key.
809    #[must_use]
810    pub const fn pinned_root(&self) -> &PublicKey {
811        &self.pinned_root
812    }
813
814    /// The last accepted trust set.
815    #[must_use]
816    pub const fn current(&self) -> Option<&TrustSet> {
817        self.current.as_ref()
818    }
819
820    /// The current trust version, 0 before the first trust set.
821    #[must_use]
822    pub fn trust_version(&self) -> u64 {
823        self.current.as_ref().map_or(0, TrustSet::trust_version)
824    }
825
826    /// Checks a signed trust set against the rule in the [module
827    /// docs](self), without changing the anchor.
828    ///
829    /// # Errors
830    ///
831    /// The first [`Rejection`] that applies.
832    pub fn check(&self, envelope: &[u8], now: Timestamp) -> Result<TrustSet, Rejection> {
833        let (signed, offered) =
834            SignedTrustSet::decode_parts(envelope).map_err(Rejection::Malformed)?;
835
836        let mut trusted_signature = false;
837        let mut out_of_window: Option<Rejection> = None;
838        for sig in &signed.signatures {
839            let Some((key, window)) = self.known_signer(sig)? else {
840                continue;
841            };
842            if !key.verify(
843                sig.algorithm(),
844                Domain::TrustSet,
845                &signed.payload,
846                sig.bytes(),
847            ) {
848                return Err(Rejection::BadSignature(sig.kid()));
849            }
850            match window.map(|w| w.check(now, self.skew)) {
851                None | Some(Ok(())) => trusted_signature = true,
852                Some(Err(e)) => {
853                    out_of_window.get_or_insert(Rejection::SignerNotValid(sig.kid(), e));
854                }
855            }
856        }
857        if !trusted_signature {
858            return Err(out_of_window.unwrap_or(Rejection::NoTrustedSignature));
859        }
860
861        let current_version = self.trust_version();
862        if offered.trust_version <= current_version {
863            return Err(Rejection::NotNewer {
864                current: current_version,
865                offered: offered.trust_version,
866            });
867        }
868        if offered.trust_version - current_version > MAX_VERSION_STEP {
869            return Err(Rejection::VersionJump {
870                current: current_version,
871                offered: offered.trust_version,
872            });
873        }
874        if let Some(current) = &self.current {
875            if offered.issued_at < current.issued_at {
876                return Err(Rejection::IssuedBeforeCurrent);
877            }
878        }
879        if !Validity::not_in_future(offered.issued_at, now, self.skew) {
880            return Err(Rejection::IssuedInFuture);
881        }
882        Ok(offered)
883    }
884
885    /// Checks a signed trust set and, if it passes, makes it current.
886    /// On rejection the anchor is unchanged.
887    ///
888    /// # Errors
889    ///
890    /// As [`Self::check`].
891    pub fn accept(&mut self, envelope: &[u8], now: Timestamp) -> Result<&TrustSet, Rejection> {
892        let accepted = self.check(envelope, now)?;
893        Ok(self.current.insert(accepted))
894    }
895
896    /// The key behind `sig` if the agent knows it, with its validity window
897    /// (`None` for the pinned root, which has none). `Ok(None)` for a key
898    /// the agent does not know.
899    ///
900    /// # Errors
901    ///
902    /// [`Rejection::AlgorithmNotAccepted`] if a known key's signature names
903    /// an algorithm that is not the key's own or that the current set does
904    /// not accept.
905    fn known_signer(
906        &self,
907        sig: &Signature,
908    ) -> Result<Option<(&PublicKey, Option<Validity>)>, Rejection> {
909        let kid = sig.kid();
910        if kid == self.pinned_root.kid() {
911            // A root-ca key uses ECDSA P-384 only.
912            return if sig.algorithm() == SignatureAlgorithm::EcdsaP384Sha384 {
913                Ok(Some((&self.pinned_root, None)))
914            } else {
915                Err(Rejection::AlgorithmNotAccepted(sig.algorithm()))
916            };
917        }
918        let Some(current) = &self.current else {
919            return Ok(None);
920        };
921        let Some(key) = current.trust_keys.get(&kid) else {
922            return Ok(None);
923        };
924        if !current.algorithms.contains(&sig.algorithm()) || sig.algorithm() != key.algorithm {
925            return Err(Rejection::AlgorithmNotAccepted(sig.algorithm()));
926        }
927        Ok(Some((&key.public_key, Some(key.validity))))
928    }
929}

931/// Why an agent refused a trust set.
932#[derive(Clone, Debug, PartialEq, Eq)]
933pub enum Rejection {
934    /// The envelope or payload is not canonical.
935    Malformed(DecodeError),
936    /// No signature is by a key the agent already trusts.
937    NoTrustedSignature,
938    /// A trusted key's signature does not verify.
939    BadSignature(Kid),
940    /// A trusted key signed with an algorithm the current set does not
941    /// accept, or one that is not the key's own.
942    AlgorithmNotAccepted(SignatureAlgorithm),
943    /// A trusted key's validity window does not contain `now`, even
944    /// allowing for skew.
945    SignerNotValid(Kid, TimeError),
946    /// `trust_version` is not greater than the current one (rollback or
947    /// replay).
948    NotNewer {
949        /// The agent's current version.
950        current: u64,
951        /// The offered version.
952        offered: u64,
953    },
954    /// `trust_version` is more than [`MAX_VERSION_STEP`] above the current
955    /// one (fast-forward).
956    VersionJump {
957        /// The agent's current version.
958        current: u64,
959        /// The offered version.
960        offered: u64,
961    },
962    /// `issued_at` is earlier than the current set's.
963    IssuedBeforeCurrent,
964    /// `issued_at` is later than `now` plus the skew tolerance.
965    IssuedInFuture,
966}
967
968impl fmt::Display for Rejection {
969    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
970        match self {
971            Self::Malformed(e) => write!(f, "malformed trust set: {e}"),
972            Self::NoTrustedSignature => f.write_str("no signature by a trusted key"),
973            Self::BadSignature(kid) => write!(f, "the signature by {kid} does not verify"),
974            Self::AlgorithmNotAccepted(alg) => write!(f, "{alg} is not accepted for this key"),
975            Self::SignerNotValid(kid, e) => write!(f, "signing key {kid}: {e}"),
976            Self::NotNewer { current, offered } => write!(
977                f,
978                "trust version {offered} is not newer than the current {current}"
979            ),
980            Self::VersionJump { current, offered } => write!(
981                f,
982                "trust version {offered} is more than {MAX_VERSION_STEP} above the current {current}"
983            ),
984            Self::IssuedBeforeCurrent => f.write_str("issued before the current trust set"),
985            Self::IssuedInFuture => f.write_str("issued in the future"),
986        }
987    }
988}

1125/// A bounds-checked reader over untrusted bytes. Every read either returns
1126/// the bytes or fails; nothing indexes.
1127struct Reader<'a>(&'a [u8]);
1128
1129impl<'a> Reader<'a> {
1130    fn take(&mut self, n: usize) -> Result<&'a [u8], DecodeError> {
1131        let (head, rest) = self.0.split_at_checked(n).ok_or(DecodeError::Truncated)?;
1132        self.0 = rest;
1133        Ok(head)
1134    }
1135
1136    fn array<const N: usize>(&mut self) -> Result<[u8; N], DecodeError> {
1137        let mut out = [0u8; N];
1138        out.copy_from_slice(self.take(N)?);
1139        Ok(out)
1140    }
1141
1142    fn u8(&mut self) -> Result<u8, DecodeError> {
1143        Ok(u8::from_be_bytes(self.array()?))
1144    }
1145
1146    fn u32(&mut self) -> Result<u32, DecodeError> {
1147        Ok(u32::from_be_bytes(self.array()?))
1148    }
1149
1150    fn u64(&mut self) -> Result<u64, DecodeError> {
1151        Ok(u64::from_be_bytes(self.array()?))
1152    }
1153
1154    fn timestamp(&mut self) -> Result<Timestamp, DecodeError> {
1155        Timestamp::from_unix_micros(i64::from_be_bytes(self.array()?))
1156            .ok_or(DecodeError::InvalidTimestamp)
1157    }
1158
1159    /// A one-byte count from `min` to `max`.
1160    fn count(&mut self, min: usize, max: usize, what: &'static str) -> Result<usize, DecodeError> {
1161        let n = usize::from(self.u8()?);
1162        if n < min || n > max {
1163            return Err(DecodeError::BadCount(what));
1164        }
1165        Ok(n)
1166    }
1167
1168    fn finish(self) -> Result<(), DecodeError> {
1169        if self.0.is_empty() {
1170            Ok(())
1171        } else {
1172            Err(DecodeError::TrailingBytes)
1173        }
1174    }
