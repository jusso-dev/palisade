1//! Renewal, rotation and expiry schedules (#224).
2//!
3//! These are pure functions of a validity window, the current time and, for
4//! renewal, a caller-supplied random number. Nothing here reads a clock or a
5//! random source, so every schedule can be tested by time travel.
6//!
7//! - [`RenewalPolicy`]: when a device certificate renews (at 2/3 of its
8//!   lifetime, up to 10% earlier at random), how failed attempts back off
9//!   (exponentially, up to 6 hours), and when a failing renewal alerts (less
10//!   than 7 days left).
11//! - [`ExpiryAlerts`]: the console's expiry alert thresholds (30, 14, 5 and 1
12//!   days by default).
13//! - [`RotationPlan`]: a key rotation with overlap: the new key is staged,
14//!   the service switches to it, and the old key is removed after the
15//!   overlap. Constructors enforce the minimum overlaps the issue sets for
16//!   server TLS, issuing CA and unlock-token rotations.
17//! - [`Adoption`]: how many devices report a trust version, which gates
18//!   retiring an old policy-signing key.
19//!
20//! Every rule leans towards keeping the old key longer, or renewing earlier,
21//! never the reverse (ADR 0018): a missing fact never shortens an overlap.
22

30const DAY: Duration = Duration::from_secs(86_400);

32/// `duration` in whole microseconds, saturating at `i64::MAX`.
33fn micros(duration: Duration) -> i64 {
34    i64::try_from(duration.as_micros()).unwrap_or(i64::MAX)
35}
36
37/// `t + micros`, clamped to the timestamp range.
38fn add(t: Timestamp, micros: i64) -> Timestamp {
39    let sum = t
40        .unix_micros()
41        .saturating_add(micros)
42        .clamp(Timestamp::MIN.unix_micros(), Timestamp::MAX.unix_micros());
43    Timestamp::from_unix_micros(sum).unwrap_or(Timestamp::MAX)
44}
45
46/// `later - earlier` in microseconds. Never overflows: timestamps span less
47/// than 2^59 microseconds.
48fn span(earlier: Timestamp, later: Timestamp) -> i64 {
49    later.unix_micros().saturating_sub(earlier.unix_micros())
50}

52/// How device certificates renew (#224).
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub struct RenewalPolicy {
55    initial_backoff: Duration,
56    max_backoff: Duration,
57    alert_window: Duration,
58}
59
60impl RenewalPolicy {
61    /// Renew at 2/3 of the lifetime, retry after 1 minute doubling up to
62    /// 6 hours, and alert when renewal is failing with less than 7 days left.
63    pub const DEFAULT: Self = Self {
64        initial_backoff: Duration::from_secs(60),
65        max_backoff: Duration::from_secs(6 * 3600),
66        alert_window: Duration::from_secs(7 * 86_400),
67    };
68
69    /// The fraction of the lifetime at which renewal is due, as
70    /// numerator and denominator.
71    pub const RENEW_AT: (i64, i64) = (2, 3);
72
73    /// The most renewal may be brought forward at random, as a fraction of
74    /// the lifetime (numerator, denominator).
75    pub const MAX_JITTER: (i64, i64) = (1, 10);
76
77    /// A policy with the given retry and alert settings.
78    ///
79    /// # Errors
80    ///
81    /// [`LifecycleError::BadBackoff`] if the initial back-off is zero or
82    /// longer than the maximum.
83    pub fn new(
84        initial_backoff: Duration,
85        max_backoff: Duration,
86        alert_window: Duration,
87    ) -> Result<Self, LifecycleError> {
88        if initial_backoff.is_zero() || initial_backoff > max_backoff {
89            return Err(LifecycleError::BadBackoff);
90        }
91        Ok(Self {
92            initial_backoff,
93            max_backoff,
94            alert_window,
95        })
96    }
97
98    /// When a certificate valid for `validity` should renew.
99    ///
100    /// That is 2/3 of the way through the lifetime, brought forward by up to
101    /// 10% of the lifetime: `jitter` is a uniformly random `u32` from the
102    /// caller, and the offset is `lifetime / 10 * jitter / 2^32`. Jitter
103    /// only ever moves renewal earlier, so it spreads a fleet's renewals
104    /// without ever leaving less time than the 2/3 point would.
105    #[must_use]
106    pub fn renewal_time(&self, validity: Validity, jitter: u32) -> Timestamp {
107        let lifetime = i128::from(span(validity.not_before(), validity.not_after()));
108        let (num, den) = Self::RENEW_AT;
109        let base = lifetime * i128::from(num) / i128::from(den);
110        let (jnum, jden) = Self::MAX_JITTER;
111        let max_jitter = lifetime * i128::from(jnum) / i128::from(jden);
112        let offset = (max_jitter * i128::from(jitter)) >> 32;
113        // Both terms are between 0 and the lifetime, which fits an i64.
114        let at = i64::try_from(base - offset).unwrap_or(0);
115        add(validity.not_before(), at)
116    }
117
118    /// How long to wait before the next attempt after `failures` consecutive
119    /// failed attempts: zero after none, then the initial back-off doubling
120    /// each time, capped at the maximum.
121    #[must_use]
122    pub fn retry_delay(&self, failures: u32) -> Duration {
123        if failures == 0 {
124            return Duration::ZERO;
125        }
126        let shift = failures.saturating_sub(1).min(63);
127        let factor = 1u64 << shift;
128        let initial = u64::try_from(self.initial_backoff.as_millis()).unwrap_or(u64::MAX);
129        let delay = Duration::from_millis(initial.saturating_mul(factor));
130        delay.min(self.max_backoff)
131    }
132
133    /// Where a certificate stands at `now`, given its renewal time (from
134    /// [`Self::renewal_time`]) and the number of consecutive failed renewal
135    /// attempts.
136    #[must_use]
137    pub fn status(
138        &self,
139        validity: Validity,
140        renew_at: Timestamp,
141        now: Timestamp,
142        failures: u32,
143    ) -> RenewalStatus {
144        if now > validity.not_after() {
145            return RenewalStatus::Expired;
146        }
147        if now < renew_at {
148            return RenewalStatus::Wait;
149        }
150        let remaining = span(now, validity.not_after());
151        if failures > 0 && remaining < micros(self.alert_window) {
152            RenewalStatus::Alert
153        } else {
154            RenewalStatus::Due
155        }
156    }
157}
158
159impl Default for RenewalPolicy {
160    fn default() -> Self {
161        Self::DEFAULT
162    }
163}

165/// Where a certificate stands in its renewal cycle.
166#[derive(Clone, Copy, Debug, PartialEq, Eq)]
167pub enum RenewalStatus {
168    /// Renewal is not due yet.
169    Wait,
170    /// Renewal is due: try now, or after the retry delay.
171    Due,
172    /// Renewal is due, has been failing, and less than the alert window is
173    /// left. Keep trying and raise an alert.
174    Alert,
175    /// The certificate has expired. It cannot renew itself: the device must
176    /// re-enrol.
177    Expired,
178}

180/// The expiry alert thresholds for certificates and keys (#224): 30, 14, 5
181/// and 1 days by default.
182#[derive(Clone, Debug, PartialEq, Eq)]
183pub struct ExpiryAlerts {
184    // Strictly descending, non-empty, every entry non-zero.
185    thresholds: Vec<Duration>,
186}
187
188impl ExpiryAlerts {
189    /// Most thresholds accepted.
190    pub const MAX_THRESHOLDS: usize = 16;
191
192    /// The default thresholds: 30, 14, 5 and 1 days.
193    #[must_use]
194    pub fn standard() -> Self {
195        Self {
196            thresholds: vec![DAY * 30, DAY * 14, DAY * 5, DAY],
197        }
198    }
199
200    /// Custom thresholds, in any order.
201    ///
202    /// # Errors
203    ///
204    /// [`LifecycleError::BadThresholds`] if the list is empty, longer than
205    /// [`Self::MAX_THRESHOLDS`], or holds a zero or a duplicate. An empty
206    /// list would silence every expiry alert, so it is refused.
207    pub fn new(thresholds: &[Duration]) -> Result<Self, LifecycleError> {
208        if thresholds.is_empty() || thresholds.len() > Self::MAX_THRESHOLDS {
209            return Err(LifecycleError::BadThresholds);
210        }
211        let mut sorted = thresholds.to_vec();
212        sorted.sort_unstable_by(|a, b| b.cmp(a));
213        let duplicate = sorted.windows(2).any(|w| w.first() == w.get(1));
214        if duplicate || sorted.iter().any(Duration::is_zero) {
215            return Err(LifecycleError::BadThresholds);
216        }
217        Ok(Self { thresholds: sorted })
218    }
219
220    /// The thresholds, longest first.
221    #[must_use]
222    pub fn thresholds(&self) -> &[Duration] {
223        &self.thresholds
224    }
225
226    /// The alert for something that expires at `not_after`, seen at `now`.
227    #[must_use]
228    pub fn alert(&self, not_after: Timestamp, now: Timestamp) -> ExpiryAlert {
229        let remaining = span(now, not_after);
230        if remaining < 0 {
231            return ExpiryAlert::Expired;
232        }
233        self.thresholds
234            .iter()
235            .rev()
236            .find(|t| remaining <= micros(**t))
237            .map_or(ExpiryAlert::None, |t| ExpiryAlert::Within(*t))
238    }
239}
240
241impl Default for ExpiryAlerts {
242    fn default() -> Self {
243        Self::standard()
244    }
245}

247/// An expiry alert level.
248#[derive(Clone, Copy, Debug, PartialEq, Eq)]
249pub enum ExpiryAlert {
250    /// More time is left than the longest threshold.
251    None,
252    /// No more than this threshold is left: the shortest threshold crossed.
253    Within(Duration),
254    /// Already expired.
255    Expired,
256}

258/// The stages of a rotation with overlap.
259#[derive(Clone, Copy, Debug, PartialEq, Eq)]
260pub enum RotationPhase {
261    /// Before staging: only the old key is in use or published.
262    Planned,
263    /// The new key is published (pinned, listed) but the old key is still
264    /// the one in use.
265    Staged,
266    /// The new key is in use. The old key is retiring and still trusted.
267    Overlap,
268    /// The old key may be removed from the trust set and retired.
269    Complete,
270}
271
272/// A key rotation: stage the new key, switch to it, then remove the old one.
273///
274/// The constructors enforce a minimum lead (staging to switch) and a minimum
275/// overlap (switch to removal), so a plan that would strand agents cannot be
276/// built.
277#[derive(Clone, Copy, Debug, PartialEq, Eq)]
278pub struct RotationPlan {
279    staged_at: Timestamp,
280    switch_at: Timestamp,
281    retire_at: Timestamp,
282}
283
284impl RotationPlan {
285    /// The least time a new server TLS key or certificate is pinned in the
286    /// trust set before the server switches to it (#224).
287    pub const SERVER_TLS_MIN_LEAD: Duration = Duration::from_secs(14 * 86_400);
288
289    /// A plan with at least `min_lead` between staging and switching and at
290    /// least `min_overlap` between switching and removing the old key.
291    ///
292    /// # Errors
293    ///
294    /// [`LifecycleError::LeadTooShort`] or [`LifecycleError::OverlapTooShort`].
295    pub fn new(
296        staged_at: Timestamp,
297        switch_at: Timestamp,
298        retire_at: Timestamp,
299        min_lead: Duration,
300        min_overlap: Duration,
301    ) -> Result<Self, LifecycleError> {
302        if span(staged_at, switch_at) < micros(min_lead) {
303            return Err(LifecycleError::LeadTooShort);
304        }
305        if span(switch_at, retire_at) < micros(min_overlap) {
306            return Err(LifecycleError::OverlapTooShort);
307        }
308        Ok(Self {
309            staged_at,
310            switch_at,
311            retire_at,
312        })
313    }
314
315    /// A server TLS rotation: the new pin is staged at least 14 days before
316    /// the switch, and the old pin is removed no earlier than the switch.
317    ///
318    /// # Errors
319    ///
320    /// As [`Self::new`].
321    pub fn server_tls(
322        staged_at: Timestamp,
323        switch_at: Timestamp,
324        retire_at: Timestamp,
325    ) -> Result<Self, LifecycleError> {
326        Self::new(
327            staged_at,
328            switch_at,
329            retire_at,
330            Self::SERVER_TLS_MIN_LEAD,
331            Duration::ZERO,
332        )
333    }
334
335    /// An issuing CA rotation: the new CA issues from `switch_at`, and the
336    /// old CA stays trusted for the longest device-certificate lifetime plus
337    /// `grace`, so every certificate it issued expires or renews first.
338    ///
339    /// # Errors
340    ///
341    /// [`LifecycleError::LeadTooShort`] if `staged_at` is after `switch_at`.
342    pub fn issuing_ca(
343        staged_at: Timestamp,
344        switch_at: Timestamp,
345        max_cert_lifetime: Duration,
346        grace: Duration,
347    ) -> Result<Self, LifecycleError> {
348        if switch_at < staged_at {
349            return Err(LifecycleError::LeadTooShort);
350        }
351        // Clamped at the end of time, so an absurd overlap keeps the old CA
352        // trusted rather than failing.
353        let overlap = max_cert_lifetime.saturating_add(grace);
354        Ok(Self {
355            staged_at,
356            switch_at,
357            retire_at: add(switch_at, micros(overlap)),
358        })
359    }
360
361    /// An unlock-token key rotation: tokens carry their `kid`, and the old
362    /// key stays accepted until the longest-lived token it signed has
363    /// expired.
364    #[must_use]
365    pub fn unlock_token(switch_at: Timestamp, max_token_lifetime: Duration) -> Self {
366        Self {
367            staged_at: switch_at,
368            switch_at,
369            retire_at: add(switch_at, micros(max_token_lifetime)),
370        }
371    }
372
373    /// When the new key is published.
374    #[must_use]
375    pub const fn staged_at(&self) -> Timestamp {
376        self.staged_at
377    }
378
379    /// When the new key comes into use.
380    #[must_use]
381    pub const fn switch_at(&self) -> Timestamp {
382        self.switch_at
383    }
384
385    /// When the old key may be removed.
386    #[must_use]
387    pub const fn retire_at(&self) -> Timestamp {
388        self.retire_at
389    }
390
391    /// The phase at `now`. Each boundary belongs to the later phase.
392    #[must_use]
393    pub fn phase(&self, now: Timestamp) -> RotationPhase {
394        if now >= self.retire_at {
395            RotationPhase::Complete
396        } else if now >= self.switch_at {
397            RotationPhase::Overlap
398        } else if now >= self.staged_at {
399            RotationPhase::Staged
400        } else {
401            RotationPhase::Planned
402        }
403    }
404}

406/// How many active devices report at least a target trust version (#224).
407///
408/// An old policy-signing key is retired only once every active device has
409/// the trust set that lists its successor. A device that has not reported a
410/// version counts as not adopted.
411#[derive(Clone, Copy, Debug, PartialEq, Eq)]
412pub struct Adoption {
413    target: u64,
414    adopted: usize,
415    total: usize,
416}
417
418impl Adoption {
419    /// Counts the devices whose reported trust version is at least `target`.
420    /// `reported` has one entry per active device: `None` for a device that
421    /// has not reported.
422    pub fn measure<I>(target: u64, reported: I) -> Self
423    where
424        I: IntoIterator<Item = Option<u64>>,
425    {
426        let mut adopted = 0usize;
427        let mut total = 0usize;
428        for version in reported {
429            total = total.saturating_add(1);
430            if version.is_some_and(|v| v >= target) {
431                adopted = adopted.saturating_add(1);
432            }
433        }
434        Self {
435            target,
436            adopted,
437            total,
438        }
439    }
440
441    /// The target trust version.
442    #[must_use]
443    pub const fn target(&self) -> u64 {
444        self.target
445    }
446
447    /// Devices at or past the target.
448    #[must_use]
449    pub const fn adopted(&self) -> usize {
450        self.adopted
451    }
452
453    /// Active devices counted.
454    #[must_use]
455    pub const fn total(&self) -> usize {
456        self.total
457    }
458
459    /// Whether every active device has adopted the target, so the old key
460    /// can be retired without stranding a device. True for an empty fleet.
461    #[must_use]
462    pub const fn complete(&self) -> bool {
463        self.adopted == self.total
464    }
465}

467/// A schedule value is invalid.
468#[derive(Clone, Copy, Debug, PartialEq, Eq)]
469pub enum LifecycleError {
470    /// The initial back-off is zero or longer than the maximum.
471    BadBackoff,
472    /// The alert thresholds are empty, too many, zero or duplicated.
473    BadThresholds,
474    /// The new key would be staged for less than the minimum lead time.
475    LeadTooShort,
476    /// The old key would be removed before the minimum overlap.
477    OverlapTooShort,
478}
479
480impl fmt::Display for LifecycleError {
481    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
482        match self {
483            Self::BadBackoff => {
484                f.write_str("the initial back-off must be non-zero and at most the maximum")
485            }
486            Self::BadThresholds => {
487                f.write_str("expiry alert thresholds must be 1 to 16 distinct, non-zero durations")
488            }
489            Self::LeadTooShort => {
490                f.write_str("the new key would be staged for less than the minimum lead time")
491            }
492            Self::OverlapTooShort => {
493                f.write_str("the old key would be removed before the minimum overlap")
494            }
495        }
496    }
497}
