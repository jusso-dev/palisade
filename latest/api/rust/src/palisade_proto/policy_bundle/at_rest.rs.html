1//! Policy cache files at rest (#326): AES-256-GCM sealing under a data key,
2//! with the nonce and seal-count rules of ADR 0017.
3//!
4//! The agent seals each cache file (a signed bundle envelope, #325) before
5//! writing it, and opens it after reading. Sealing adds confidentiality and
6//! an integrity check, but it never replaces signature verification: the
7//! opened bytes are the signed envelope, which the agent verifies as if it
8//! had never been sealed. A file that does not open is unreadable, never
9//! trusted.
10//!
11//! # Formats
12//!
13//! All integers are big-endian.
14//!
15//! ```text
16//! sealed  = "PLCS" version:u8 (= 1) key_id:[u8;16] nonce:[u8;12] ciphertext tag:[u8;16]
17//! keyring = "PLKR" version:u8 (= 1) seals:u64 key_id:[u8;16] key:[u8;32]
18//!           retired:u8 (0 or 1) [key_id:[u8;16] key:[u8;32]]
19//! ```
20//!
21//! A sealed file's additional authenticated data is
22//! `"palisade-policy-cache-v1\0"` followed by its first 21 bytes (magic,
23//! version and key ID), so the key ID cannot be changed without failing the
24//! tag.
25//!
26//! # Keys and nonces (ADR 0017)
27//!
28//! [`KeyRing`] holds the current data key and, after a rotation, the one
29//! before it (to open files sealed before the rotation). Every nonce is
30//! drawn from the OS secure random source inside [`KeyRing::seal`]; no
31//! caller can pass one. A key seals at most [`KeyRing::SEAL_LIMIT`] (2^32)
32//! messages, which keeps the chance of any repeated nonce under one key at
33//! or below 2^-33. The count is persisted **before** a seal is used:
34//! [`KeyRing::reserve`] raises it (rotating first if the key is used up),
35//! the caller persists [`KeyRing::encode`], and only then does
36//! [`KeyRing::seal`] use the reservation. A keyring that is missing or does
37//! not decode is never repaired: the caller generates a new one, which
38//! "rotates now".
39//!
40//! The keyring holds raw keys, so the agent protects it before writing it
41//! (machine-scope DPAPI on Windows). `ring` does not wipe key memory on
42//! drop; the agent process is trusted with the keys for its life.
43

50/// The sealed file magic.
51pub const SEALED_MAGIC: [u8; 4] = *b"PLCS";
52/// The keyring magic.
53pub const KEYRING_MAGIC: [u8; 4] = *b"PLKR";
54/// The format version both formats use.
55pub const FORMAT_VERSION: u8 = 1;
56
57const DOMAIN: &[u8] = b"palisade-policy-cache-v1\0";
58const KEY_ID_LEN: usize = 16;
59const KEY_LEN: usize = 32;
60/// Magic, version and key ID.
61const HEADER_LEN: usize = 4 + 1 + KEY_ID_LEN;
62/// The bytes sealing adds to a plaintext.
63pub const OVERHEAD: usize = HEADER_LEN + NONCE_LEN + TAG_LEN;
64const KEYRING_ONE: usize = 4 + 1 + 8 + KEY_ID_LEN + KEY_LEN + 1;
65const KEYRING_TWO: usize = KEYRING_ONE + KEY_ID_LEN + KEY_LEN;
66
67/// Why sealing or opening failed. Every failure means the file is not
68/// used.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70pub enum AtRestError {
71    /// The bytes are not in the format (wrong magic, version or length).
72    Malformed,
73    /// The file names a key this keyring does not hold.
74    UnknownKey,
75    /// The tag does not verify: the file or its header was changed.
76    Authentication,
77    /// [`KeyRing::seal`] was called without a reservation.
78    NotReserved,
79    /// The OS secure random source failed.
80    Random,
81}
82
83impl fmt::Display for AtRestError {
84    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
85        f.write_str(match self {
86            Self::Malformed => "not a sealed policy cache file or keyring",
87            Self::UnknownKey => "sealed with a data key this keyring does not hold",
88            Self::Authentication => "the sealed file does not authenticate",
89            Self::NotReserved => "no seal was reserved",
90            Self::Random => "the secure random source failed",
91        })
92    }
93}

97/// One data key.
98struct DataKey {
99    id: [u8; KEY_ID_LEN],
100    raw: [u8; KEY_LEN],
101    key: AeadKey,
102}
103
104impl DataKey {
105    fn new(id: [u8; KEY_ID_LEN], raw: [u8; KEY_LEN]) -> Result<Self, AtRestError> {
106        let key = AeadKey::new(&raw).map_err(|_| AtRestError::Malformed)?;
107        Ok(Self { id, raw, key })
108    }
109
110    fn generate() -> Result<Self, AtRestError> {
111        let mut id = [0u8; KEY_ID_LEN];
112        let mut raw = [0u8; KEY_LEN];
113        crypto::random(&mut id).map_err(|_| AtRestError::Random)?;
114        crypto::random(&mut raw).map_err(|_| AtRestError::Random)?;
115        Self::new(id, raw)
116    }
117}

119/// The data keys and the current key's persisted seal count. `Debug` never
120/// prints a key.
121pub struct KeyRing {
122    current: DataKey,
123    retired: Option<DataKey>,
124    /// Seals reserved under `current`, persisted or about to be.
125    seals: u64,
126    /// Reservations not used yet, in this process only.
127    reserved: u64,
128    limit: u64,
129}
130
131impl fmt::Debug for KeyRing {
132    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
133        f.debug_struct("KeyRing")
134            .field("seals", &self.seals)
135            .field("retired", &self.retired.is_some())
136            .finish_non_exhaustive()
137    }
138}

140impl KeyRing {
141    /// The most messages one data key seals (ADR 0017).
142    pub const SEAL_LIMIT: u64 = 1 << 32;
143
144    /// A new keyring with one fresh key and no seals.
145    ///
146    /// # Errors
147    ///
148    /// [`AtRestError::Random`] if the secure random source fails.
149    pub fn generate() -> Result<Self, AtRestError> {
150        Ok(Self {
151            current: DataKey::generate()?,
152            retired: None,
153            seals: 0,
154            reserved: 0,
155            limit: Self::SEAL_LIMIT,
156        })
157    }
158
159    /// The same keyring, rotating once `limit` seals were reserved. For
160    /// tests of rotation; values above [`Self::SEAL_LIMIT`] are lowered to
161    /// it.
162    #[must_use]
163    pub fn with_limit(mut self, limit: u64) -> Self {
164        self.limit = limit.min(Self::SEAL_LIMIT);
165        self
166    }
167
168    /// Seals reserved under the current key.
169    #[must_use]
170    pub const fn seals(&self) -> u64 {
171        self.seals
172    }
173
174    /// Reserves one seal. If the current key is used up, it is retired
175    /// first and a new key becomes current; the key retired before is then
176    /// dropped, so files sealed under it no longer open. Returns whether it
177    /// rotated. Persist [`Self::encode`] before calling [`Self::seal`].
178    ///
179    /// # Errors
180    ///
181    /// [`AtRestError::Random`] if a rotation needs a key and the secure
182    /// random source fails.
183    pub fn reserve(&mut self) -> Result<bool, AtRestError> {
184        let rotate = self.seals >= self.limit;
185        if rotate {
186            let next = DataKey::generate()?;
187            self.retired = Some(std::mem::replace(&mut self.current, next));
188            self.seals = 0;
189            self.reserved = 0;
190        }
191        self.seals = self.seals.saturating_add(1);
192        self.reserved = self.reserved.saturating_add(1);
193        Ok(rotate)
194    }
195
196    /// Seals `plaintext` under the current key with a fresh random nonce,
197    /// using one reservation.
198    ///
199    /// # Errors
200    ///
201    /// [`AtRestError::NotReserved`] without a reservation, and
202    /// [`AtRestError::Random`] if the secure random source fails.
203    pub fn seal(&mut self, plaintext: &[u8]) -> Result<Vec<u8>, AtRestError> {
204        if self.reserved == 0 {
205            return Err(AtRestError::NotReserved);
206        }
207        self.reserved -= 1;
208        let header = header(&self.current.id);
209        let body = self
210            .current
211            .key
212            .seal(&aad(&header), plaintext)
213            .map_err(|_| AtRestError::Random)?;
214        let mut out = Vec::with_capacity(header.len().saturating_add(body.len()));
215        out.extend_from_slice(&header);
216        out.extend_from_slice(&body);
217        Ok(out)
218    }
219
220    /// Opens a sealed file under whichever key it names.
221    ///
222    /// # Errors
223    ///
224    /// [`AtRestError::Malformed`], [`AtRestError::UnknownKey`] or
225    /// [`AtRestError::Authentication`].
226    pub fn open(&self, sealed: &[u8]) -> Result<Vec<u8>, AtRestError> {
227        let (head, body) = sealed
228            .split_at_checked(HEADER_LEN)
229            .ok_or(AtRestError::Malformed)?;
230        let id = sealed_key_id(head).ok_or(AtRestError::Malformed)?;
231        let key = [Some(&self.current), self.retired.as_ref()]
232            .into_iter()
233            .flatten()
234            .find(|k| k.id == id)
235            .ok_or(AtRestError::UnknownKey)?;
236        if body.len() < NONCE_LEN + TAG_LEN {
237            return Err(AtRestError::Malformed);
238        }
239        key.key
240            .open(&aad(head), body)
241            .ok_or(AtRestError::Authentication)
242    }
243
244    /// The keyring bytes, to protect and persist.
245    #[must_use]
246    pub fn encode(&self) -> Vec<u8> {
247        let mut out = Vec::with_capacity(KEYRING_TWO);
248        out.extend_from_slice(&KEYRING_MAGIC);
249        out.push(FORMAT_VERSION);
250        out.extend_from_slice(&self.seals.to_be_bytes());
251        out.extend_from_slice(&self.current.id);
252        out.extend_from_slice(&self.current.raw);
253        match &self.retired {
254            None => out.push(0),
255            Some(k) => {
256                out.push(1);
257                out.extend_from_slice(&k.id);
258                out.extend_from_slice(&k.raw);
259            }
260        }
261        out
262    }
263
264    /// Decodes keyring bytes, strictly.
265    ///
266    /// # Errors
267    ///
268    /// [`AtRestError::Malformed`] for anything [`Self::encode`] would not
269    /// write.
270    pub fn decode(bytes: &[u8]) -> Result<Self, AtRestError> {
271        const M: AtRestError = AtRestError::Malformed;
272        if bytes.len() != KEYRING_ONE && bytes.len() != KEYRING_TWO {
273            return Err(M);
274        }
275        let mut r = Reader(bytes);
276        if r.array::<4>()? != KEYRING_MAGIC || r.array::<1>()? != [FORMAT_VERSION] {
277            return Err(M);
278        }
279        let seals = u64::from_be_bytes(r.array()?);
280        let current = DataKey::new(r.array()?, r.array()?)?;
281        let retired = match (r.array::<1>()?, bytes.len()) {
282            ([0], KEYRING_ONE) => None,
283            ([1], KEYRING_TWO) => Some(DataKey::new(r.array()?, r.array()?)?),
284            _ => return Err(M),
285        };
286        if retired.as_ref().is_some_and(|k| k.id == current.id) {
287            return Err(M);
288        }
289        Ok(Self {
290            current,
291            retired,
292            seals,
293            reserved: 0,
294            limit: Self::SEAL_LIMIT,
295        })
296    }
297}

299fn header(id: &[u8; KEY_ID_LEN]) -> [u8; HEADER_LEN] {
300    let mut h = [0u8; HEADER_LEN];
301    for (o, b) in h
302        .iter_mut()
303        .zip(SEALED_MAGIC.iter().chain([FORMAT_VERSION].iter()).chain(id))
304    {
305        *o = *b;
306    }
307    h
308}
309
310fn aad(header: &[u8]) -> Vec<u8> {
311    let mut a = DOMAIN.to_vec();
312    a.extend_from_slice(header);
313    a
314}

316/// The key ID of a sealed file's header, if the magic and version match.
317fn sealed_key_id(head: &[u8]) -> Option<[u8; KEY_ID_LEN]> {
318    let rest = head.strip_prefix(&SEALED_MAGIC)?;
319    let id = rest.strip_prefix(&[FORMAT_VERSION])?;
320    id.try_into().ok()
321}

323/// Whether `bytes` start like a sealed file. Says nothing about whether it
324/// opens.
325#[must_use]
326pub fn looks_sealed(bytes: &[u8]) -> bool {
327    bytes.starts_with(&SEALED_MAGIC)
328}

330struct Reader<'a>(&'a [u8]);
331
332impl Reader<'_> {
333    fn array<const N: usize>(&mut self) -> Result<[u8; N], AtRestError> {
334        let (head, rest) = self.0.split_at_checked(N).ok_or(AtRestError::Malformed)?;
335        self.0 = rest;
336        head.try_into().map_err(|_| AtRestError::Malformed)
337    }
338}
