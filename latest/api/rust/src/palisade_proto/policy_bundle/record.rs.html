1//! The anti-rollback record and its copies.
2//!
3//! The record holds, for this device's target (ADR 0007, which is still
4//! `proposed`):
5//!
6//! - the **floor**: the highest `policy_version` the agent has made
7//!   `active`. It never exceeds the version of the bundle in `active`;
8//! - the **fallback**: the version of the `previous` bundle that the current
9//!   `active` directly replaced, or none. Only that bundle may load under the
10//!   one-step N-1 exception, and it is always below the floor.
11//!
12//! The agent keeps several copies (#225: a DPAPI-protected file and a
13//! registry mirror, later a TPM-sealed blob). Each copy is sealed on its own
14//! with HMAC-SHA-256 under a device key that the agent holds in protected
15//! storage (#220), so a copy cannot be edited without the key. A copy can
16//! still be deleted or replaced by an older sealed copy, which is why
17//! [`RecordCopies`] compares them.
18//!
19//! # Encoding
20//!
21//! ```text
22//! copy     = format:u8 (= 1)
23//!            tenant:[u8;16] scope:u8 (1 group | 2 device) scope_id:[u8;16]
24//!            floor:u64
25//!            has_fallback:u8 (0 | 1) [fallback:u64]   if has_fallback = 1
26//!            tag:[u8;32]
27//! tag      = HMAC-SHA-256(key, "palisade-policy-record-v1\0" ‖ every byte before the tag)
28//! ```
29//!
30//! Integers are big-endian, IDs are UUIDv7 bytes. The tag is checked before
31//! any field is read, and a copy for another target is refused, so a sealed
32//! copy cannot be moved between targets.
33

42/// The record format version this module reads and writes.
43pub const FORMAT_VERSION: u8 = 1;
44
45/// The MAC domain-separation prefix, including its NUL.
46const MAC_DOMAIN: &[u8] = b"palisade-policy-record-v1\0";
47
48const TAG_LEN: usize = 32;
49const BODY_LEN: usize = 1 + 16 + 1 + 16 + 8 + 1;
50/// Length of a sealed copy with no fallback.
51pub const SEALED_LEN: usize = BODY_LEN + TAG_LEN;
52/// Length of a sealed copy with a fallback.
53pub const SEALED_LEN_WITH_FALLBACK: usize = BODY_LEN + 8 + TAG_LEN;

55/// The device's record key: 32 bytes that the agent keeps in protected
56/// storage (#220). `Debug` never prints it.
57pub struct RecordKey(MacKey);

59impl RecordKey {
60    /// A key from its 32 bytes.
61    #[must_use]
62    pub fn from_bytes(key: &[u8; 32]) -> Self {
63        Self(MacKey::new(key))
64    }
65}

73/// The anti-rollback record for one target.
74#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
75pub struct Record {
76    target: Target,
77    floor: u64,
78    fallback: Option<u64>,
79}
80
81impl Record {
82    /// The record before any bundle was activated: floor 0, no fallback.
83    #[must_use]
84    pub const fn empty(target: Target) -> Self {
85        Self {
86            target,
87            floor: 0,
88            fallback: None,
89        }
90    }
91
92    /// A record.
93    ///
94    /// # Errors
95    ///
96    /// [`RecordError::InvalidFallback`] unless the fallback, if any, is
97    /// from 1 to below the floor.
98    pub const fn new(
99        target: Target,
100        floor: u64,
101        fallback: Option<u64>,
102    ) -> Result<Self, RecordError> {
103        if let Some(f) = fallback
104            && (f == 0 || f >= floor)
105        {
106            return Err(RecordError::InvalidFallback);
107        }
108        Ok(Self {
109            target,
110            floor,
111            fallback,
112        })
113    }
114
115    /// The target.
116    #[must_use]
117    pub const fn target(&self) -> Target {
118        self.target
119    }
120
121    /// The highest version made `active`.
122    #[must_use]
123    pub const fn floor(&self) -> u64 {
124        self.floor
125    }
126
127    /// The `previous` version the current `active` directly replaced.
128    #[must_use]
129    pub const fn fallback(&self) -> Option<u64> {
130        self.fallback
131    }
132
133    /// The same record without its fallback.
134    #[must_use]
135    pub const fn without_fallback(self) -> Self {
136        Self {
137            fallback: None,
138            ..self
139        }
140    }
141
142    /// One sealed copy.
143    #[must_use]
144    pub fn seal(&self, key: &RecordKey) -> Vec<u8> {
145        let mut out = Vec::with_capacity(SEALED_LEN_WITH_FALLBACK);
146        out.push(FORMAT_VERSION);
147        out.extend_from_slice(self.target.tenant.0.as_bytes());
148        let (code, id) = match self.target.scope {
149            Scope::Group(id) => (1u8, id),
150            Scope::Device(host) => (2u8, host.0),
151        };
152        out.push(code);
153        out.extend_from_slice(id.as_bytes());
154        out.extend_from_slice(&self.floor.to_be_bytes());
155        match self.fallback {
156            None => out.push(0),
157            Some(f) => {
158                out.push(1);
159                out.extend_from_slice(&f.to_be_bytes());
160            }
161        }
162        let tag = key.0.tag(&mac_input(&out));
163        out.extend_from_slice(&tag);
164        out
165    }
166
167    /// Reads one sealed copy, checking its tag first.
168    ///
169    /// # Errors
170    ///
171    /// [`RecordError::Malformed`] for a wrong length or layout,
172    /// [`RecordError::BadTag`] if the tag does not verify, and
173    /// [`RecordError::WrongTarget`] for a copy sealed for another target.
174    pub fn unseal(bytes: &[u8], key: &RecordKey, expected: Target) -> Result<Self, RecordError> {
175        if bytes.len() != SEALED_LEN && bytes.len() != SEALED_LEN_WITH_FALLBACK {
176            return Err(RecordError::Malformed);
177        }
178        let (body, tag) = bytes
179            .split_at_checked(bytes.len() - TAG_LEN)
180            .ok_or(RecordError::Malformed)?;
181        if !key.0.verify(&mac_input(body), tag) {
182            return Err(RecordError::BadTag);
183        }
184        let record = decode(body)?;
185        if record.target != expected {
186            return Err(RecordError::WrongTarget);
187        }
188        Ok(record)
189    }
190}
191
192fn mac_input(body: &[u8]) -> Vec<u8> {
193    let mut m = Vec::with_capacity(MAC_DOMAIN.len() + body.len());
194    m.extend_from_slice(MAC_DOMAIN);
195    m.extend_from_slice(body);
196    m
197}
198
199fn decode(body: &[u8]) -> Result<Record, RecordError> {
200    const M: RecordError = RecordError::Malformed;
201    let (&format, rest) = body.split_first().ok_or(M)?;
202    if format != FORMAT_VERSION {
203        return Err(M);
204    }
205    let (tenant, rest) = rest.split_first_chunk::<16>().ok_or(M)?;
206    let (&scope, rest) = rest.split_first().ok_or(M)?;
207    let (scope_id, rest) = rest.split_first_chunk::<16>().ok_or(M)?;
208    let (floor, rest) = rest.split_first_chunk::<8>().ok_or(M)?;
209    let (&has_fallback, rest) = rest.split_first().ok_or(M)?;
210    let (fallback, rest) = match has_fallback {
211        0 => (None, rest),
212        1 => {
213            let (f, rest) = rest.split_first_chunk::<8>().ok_or(M)?;
214            (Some(u64::from_be_bytes(*f)), rest)
215        }
216        _ => return Err(M),
217    };
218    if !rest.is_empty() {
219        return Err(M);
220    }
221    let tenant = TenantId(Uuid7::from_bytes(*tenant).ok_or(M)?);
222    let id = Uuid7::from_bytes(*scope_id).ok_or(M)?;
223    let scope = match scope {
224        1 => Scope::Group(id),
225        2 => Scope::Device(HostId(id)),
226        _ => return Err(M),
227    };
228    Record::new(
229        Target { tenant, scope },
230        u64::from_be_bytes(*floor),
231        fallback,
232    )
233    .map_err(|_| M)
234}

236/// The state of one copy as read from storage.
237#[derive(Clone, Copy, Debug, PartialEq, Eq)]
238pub enum CopyState {
239    /// The copy does not exist.
240    Missing,
241    /// The copy exists but does not unseal.
242    Corrupt(RecordError),
243    /// The copy unsealed.
244    Intact(Record),
245}
246
247/// Every copy of the record, read and compared.
248#[derive(Clone, Debug, PartialEq, Eq)]
249pub struct RecordCopies {
250    target: Target,
251    copies: Vec<CopyState>,
252}
253
254impl RecordCopies {
255    /// Unseals each copy (`None` for a copy that does not exist).
256    #[must_use]
257    pub fn read(copies: &[Option<&[u8]>], key: &RecordKey, target: Target) -> Self {
258        let copies = copies
259            .iter()
260            .map(|c| match c {
261                None => CopyState::Missing,
262                Some(bytes) => match Record::unseal(bytes, key, target) {
263                    Ok(r) => CopyState::Intact(r),
264                    Err(e) => CopyState::Corrupt(e),
265                },
266            })
267            .collect();
268        Self { target, copies }
269    }
270
271    /// The target the copies were read for.
272    #[must_use]
273    pub const fn target(&self) -> Target {
274        self.target
275    }
276
277    /// Each copy's state, in the order given.
278    #[must_use]
279    pub fn states(&self) -> &[CopyState] {
280        &self.copies
281    }
282
283    fn intact(&self) -> impl Iterator<Item = &Record> {
284        self.copies.iter().filter_map(|c| match c {
285            CopyState::Intact(r) => Some(r),
286            _ => None,
287        })
288    }
289
290    /// The highest floor among intact copies, or 0.
291    #[must_use]
292    pub fn floor(&self) -> u64 {
293        self.intact().map(|r| r.floor).max().unwrap_or(0)
294    }
295
296    /// The fallback, only if every intact copy records the same one.
297    #[must_use]
298    pub fn fallback(&self) -> Option<u64> {
299        let mut intact = self.intact();
300        let first = intact.next()?.fallback;
301        intact
302            .all(|r| r.fallback == first)
303            .then_some(first)
304            .flatten()
305    }
306
307    /// The record as read: [`Self::floor`] and [`Self::fallback`].
308    #[must_use]
309    pub fn record(&self) -> Record {
310        // The fallback agrees across copies, so it is below that copy's
311        // floor and so below the highest floor.
312        Record::new(self.target, self.floor(), self.fallback())
313            .unwrap_or_else(|_| Record::empty(self.target))
314    }
315
316    /// Whether any copy is missing.
317    #[must_use]
318    pub fn any_missing(&self) -> bool {
319        self.copies.iter().any(|c| matches!(c, CopyState::Missing))
320    }
321
322    /// Whether any copy exists but does not unseal.
323    #[must_use]
324    pub fn any_corrupt(&self) -> bool {
325        self.copies
326            .iter()
327            .any(|c| matches!(c, CopyState::Corrupt(_)))
328    }
329
330    /// Whether the intact copies hold different values.
331    #[must_use]
332    pub fn disagree(&self) -> bool {
333        let mut intact = self.intact();
334        intact
335            .next()
336            .is_some_and(|first| intact.any(|r| r != first))
337    }
338
339    /// Whether a disagreement is exactly what an interrupted activation
340    /// leaves (ADR 0007): every copy intact, the highest floor equal to the
341    /// version of the verified `active` bundle that loaded, and exactly two
342    /// distinct values, `older` and `newer`, where `older` has no fallback
343    /// and either
344    ///
345    /// - step 3 was interrupted: the same floor, and only `newer` (the
346    ///   value before step 3) has a fallback; or
347    /// - step 5 was interrupted: `older` has the lower floor, and `newer`'s
348    ///   fallback is none or `older`'s floor (the bundle step 2 copied).
349    ///
350    /// The second condition is stricter than ADR 0007's wording, so that an
351    /// old sealed copy restored into one store does not pass as an
352    /// interrupted activation.
353    #[must_use]
354    pub fn interrupted_activation(&self, active_version: Option<u64>) -> bool {
355        if self.any_missing() || self.any_corrupt() || active_version != Some(self.floor()) {
356            return false;
357        }
358        let mut distinct: Vec<&Record> = Vec::new();
359        for r in self.intact() {
360            if !distinct.contains(&r) {
361                distinct.push(r);
362            }
363        }
364        match distinct.as_slice() {
365            [a, b] => {
366                let key = |r: &Record| (r.floor, r.fallback.is_some());
367                let (older, newer) = if key(a) <= key(b) { (a, b) } else { (b, a) };
368                older.fallback.is_none()
369                    && (older.floor == newer.floor
370                        || newer.fallback.is_none_or(|f| f == older.floor))
371            }
372            _ => false,
373        }
374    }
375}

377/// A record or copy is invalid.
378#[derive(Clone, Copy, Debug, PartialEq, Eq)]
379pub enum RecordError {
380    /// The fallback is 0 or not below the floor.
381    InvalidFallback,
382    /// The copy has the wrong length, format version or layout.
383    Malformed,
384    /// The copy's tag does not verify under this device's key.
385    BadTag,
386    /// The copy was sealed for another target.
387    WrongTarget,
388}
389
390impl fmt::Display for RecordError {
391    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
392        f.write_str(match self {
393            Self::InvalidFallback => "the fallback must be from 1 to below the floor",
394            Self::Malformed => "the record copy is malformed",
395            Self::BadTag => "the record copy's MAC does not verify",
396            Self::WrongTarget => "the record copy is for another target",
397        })
398    }
399}
