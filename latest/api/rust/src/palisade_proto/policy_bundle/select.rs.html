1//! Which bundle the agent runs, and whether a new bundle is accepted.
2//!
3//! [`startup`] picks what to load from the cache files and the record
4//! copies, following the failure table of ADR 0018 (which supersedes ADR
5//! 0012) and the startup steps of ADR 0007 (still `proposed`):
6//!
7//! | State | [`Load`] | Events |
8//! |---|---|---|
9//! | `active` passes and is at or above the floor | `Active` | none, or `policy.signature.invalid` if loaded unsigned under `audit`/`warn` |
10//! | `active` fails; `previous` verifies and is the recorded fallback, or at or above the floor | `Previous` | the rejection of `active`, `policy.lkg_loaded` |
11//! | both fail, after a policy was ever accepted | `Emergency` | each rejection, `policy.emergency_loaded`, and `tamper.policy.rollback_suspected` if the floor rejected a bundle |
12//! | no policy was ever accepted | `FirstInstall` | none |
13//!
14//! Record copies that are missing, corrupt or disagree raise
15//! `tamper.policy.rollback_suspected`, except for the disagreement an
16//! interrupted activation leaves ([`RecordCopies::interrupted_activation`]).
17//! Staleness never changes what loads ([`super::stale`]).
18//!
19//! [`pending`] decides whether a downloaded bundle is activated, and
20//! returns the record values to write before and after the rename (ADR 0007
21//! activation steps 1 to 5).
22//!
23//! "A policy was ever accepted" (ADR 0018) is: the floor is above zero, an
24//! `active` or `previous` file exists, or a record copy exists but does not
25//! unseal. The last is stricter than the ADR: a corrupt copy is evidence
26//! that a record was written, so it never opens the first-install path.
27
28use super::record::{CopyState, Record, RecordCopies};

30use crate::events::EventType;

32/// A cache file.
33#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
34pub enum CacheFile {
35    /// The current bundle.
36    Active,
37    /// The bundle `active` replaced (N-1).
38    Previous,
39    /// A downloaded bundle not yet activated.
40    Pending,
41}
42
43/// What the agent runs.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum Load {
46    /// The `active` bundle. `signed` is false only when the signature mode
47    /// tolerated an invalid signature.
48    Active {
49        /// The bundle.
50        bundle: BundleMeta,
51        /// Whether its signature verified.
52        signed: bool,
53    },
54    /// The `previous` bundle, under the one-step exception.
55    Previous(BundleMeta),
56    /// The built-in emergency policy, in enforce mode (#225, ADR 0018).
57    Emergency,
58    /// No policy was ever accepted: the built-in first-install defaults
59    /// (audit mode, mandatory blocklists enforced).
60    FirstInstall,
61}

63/// Why the anti-rollback record is suspect. Several may hold at once; they
64/// are reported as one `tamper.policy.rollback_suspected` event.
65#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
66pub struct Suspicion {
67    /// A copy is missing although a policy was accepted.
68    pub copy_missing: bool,
69    /// A copy exists but does not unseal.
70    pub copy_corrupt: bool,
71    /// Intact copies disagree, and not as an interrupted activation would.
72    pub copies_disagree: bool,
73    /// A bundle on disk was below the floor and nothing else could load.
74    pub floor_rejected_all: bool,
75}
76
77impl Suspicion {
78    fn any(self) -> bool {
79        self.copy_missing || self.copy_corrupt || self.copies_disagree || self.floor_rejected_all
80    }
81}

83/// Something the agent reports. Each maps to exactly one event type.
84#[derive(Clone, Copy, Debug, PartialEq, Eq)]
85pub enum Finding {
86    /// A cache file was not loaded.
87    Rejected {
88        /// Which file.
89        file: CacheFile,
90        /// Why.
91        reason: Rejection,
92    },
93    /// A bundle with an invalid signature was loaded because the mode is
94    /// `audit` or `warn`.
95    LoadedUnsigned,
96    /// `previous` was loaded because `active` failed.
97    LkgLoaded,
98    /// The emergency policy was loaded.
99    EmergencyLoaded,
100    /// The anti-rollback record is suspect.
101    RollbackSuspected(Suspicion),
102}
103
104impl Finding {
105    /// The event this finding raises.
106    #[must_use]
107    pub const fn event_type(self) -> EventType {
108        match self {
109            Self::Rejected { reason, .. } => reason.event_type(),
110            Self::LoadedUnsigned => EventType::PolicySignatureInvalid,
111            Self::LkgLoaded => EventType::PolicyLkgLoaded,
112            Self::EmergencyLoaded => EventType::PolicyEmergencyLoaded,
113            Self::RollbackSuspected(_) => EventType::TamperPolicyRollbackSuspected,
114        }
115    }
116}

118/// What the agent reads at startup. `None` means the file does not exist.
119#[derive(Clone, Copy, Debug)]
120pub struct Startup<'a> {
121    /// This device's tenant and target.
122    pub target: Target,
123    /// Every copy of the anti-rollback record.
124    pub record: &'a RecordCopies,
125    /// The `active` file. Delete any leftover `pending` before startup: it
126    /// was never activated.
127    pub active: Option<Examined>,
128    /// The `previous` file.
129    pub previous: Option<Examined>,
130    /// The signature mode in force.
131    pub mode: EffectiveMode,
132}

134/// The startup decision.
135#[derive(Clone, Debug, PartialEq, Eq)]
136pub struct Selection {
137    /// What to run.
138    pub load: Load,
139    /// The record every copy should hold now. Its floor is never below the
140    /// highest floor read.
141    pub record: Record,
142    /// Whether any copy differs from [`Self::record`] and should be
143    /// rewritten.
144    pub rewrite_record: bool,
145    /// What to report, in order.
146    pub findings: Vec<Finding>,
147    /// Whether to check in at once to fetch a fresh bundle.
148    pub rapid_check_in: bool,
149}
150
151/// Picks what to run at startup.
152#[must_use]
153pub fn startup(input: &Startup<'_>) -> Selection {
154    let copies = input.record;
155    let floor = copies.floor();
156    let fallback = copies.fallback();
157    let mut findings = Vec::new();
158    let mut floor_rejected = false;
159    let mut reject = |findings: &mut Vec<Finding>, file, reason| {
160        floor_rejected |= reason == Rejection::Rollback;
161        findings.push(Finding::Rejected { file, reason });
162    };
163
164    // `active`: every check, and at or above the floor.
165    let active = input.active.map(|e| {
166        check(e, input.target, input.mode).and_then(|p| {
167            if p.meta().version < floor {
168                Err(Rejection::Rollback)
169            } else {
170                Ok(p)
171            }
172        })
173    });
174    let mut load = None;
175    match active {
176        Some(Ok(p)) => {
177            if let Passed::Unsigned(_) = p {
178                findings.push(Finding::LoadedUnsigned);
179            }
180            load = Some(Load::Active {
181                bundle: p.meta(),
182                signed: matches!(p, Passed::Verified(_)),
183            });
184        }
185        Some(Err(reason)) => reject(&mut findings, CacheFile::Active, reason),
186        None => {}
187    }
188
189    // `previous`: only if `active` failed, only verified, and only the
190    // recorded fallback or a version the floor accepts.
191    if load.is_none()
192        && let Some(e) = input.previous
193    {
194        match check(e, input.target, EffectiveMode::ENFORCE) {
195            Ok(p) => {
196                let v = p.meta().version;
197                if v >= floor || Some(v) == fallback {
198                    load = Some(Load::Previous(p.meta()));
199                    findings.push(Finding::LkgLoaded);
200                } else {
201                    reject(&mut findings, CacheFile::Previous, Rejection::Rollback);
202                }
203            }
204            Err(reason) => reject(&mut findings, CacheFile::Previous, reason),
205        }
206    }
207
208    let ever_accepted =
209        floor > 0 || input.active.is_some() || input.previous.is_some() || copies.any_corrupt();
210    let load = match load {
211        Some(l) => l,
212        None if ever_accepted => {
213            findings.push(Finding::EmergencyLoaded);
214            Load::Emergency
215        }
216        None => Load::FirstInstall,
217    };
218
219    let active_signed = match load {
220        Load::Active {
221            bundle,
222            signed: true,
223        } => Some(bundle.version),
224        _ => None,
225    };
226    let suspicion = Suspicion {
227        copy_missing: ever_accepted && copies.any_missing(),
228        copy_corrupt: copies.any_corrupt(),
229        copies_disagree: copies.disagree() && !copies.interrupted_activation(active_signed),
230        floor_rejected_all: floor_rejected && load == Load::Emergency,
231    };
232    if suspicion.any() {
233        findings.push(Finding::RollbackSuspected(suspicion));
234    }
235
236    // The record to keep: never a lower floor. A verified `active` above the
237    // floor completes an interrupted activation (step 5).
238    let record = match active_signed {
239        Some(v) if v > floor => {
240            Record::new(input.target, v, None).unwrap_or_else(|_| copies.record())
241        }
242        _ => copies.record(),
243    };
244    // On the first-install path there is nothing to protect yet: `pending`
245    // writes every copy before the first `active` exists.
246    let rewrite_record = load != Load::FirstInstall
247        && copies
248            .states()
249            .iter()
250            .any(|c| *c != CopyState::Intact(record));
251    let rapid_check_in = matches!(load, Load::Previous(_) | Load::Emergency) || suspicion.any();
252    Selection {
253        load,
254        record,
255        rewrite_record,
256        findings,
257        rapid_check_in,
258    }
259}
260
261impl Selection {
262    /// The `active` bundle as [`pending`] needs it: `Some` only when the
263    /// agent runs `active` with a verified signature. `None` while running
264    /// `previous`, the emergency policy, the first-install defaults or an
265    /// unsigned bundle, so a re-fetched copy of the failed version is
266    /// accepted (ADR 0007, step 1).
267    #[must_use]
268    pub const fn loadable_active(&self) -> Option<BundleMeta> {
269        match self.load {
270            Load::Active {
271                bundle,
272                signed: true,
273            } => Some(bundle),
274            _ => None,
275        }
276    }
277}

279/// The plan for activating a pending bundle (ADR 0007 steps 2 to 5).
280#[derive(Clone, Copy, Debug, PartialEq, Eq)]
281pub struct Activation {
282    /// The bundle to activate.
283    pub bundle: BundleMeta,
284    /// Whether its signature verified. An unsigned bundle (tolerated by
285    /// `audit` or `warn`) never raises the floor or becomes a fallback.
286    pub signed: bool,
287    /// Step 2: copy the current `active` to `previous` first.
288    pub copy_active_to_previous: bool,
289    /// Step 3: write this to every copy before the rename (the fallback
290    /// cleared). Write it even on the first activation, so that every copy
291    /// exists before an `active` file does.
292    pub record_before_rename: Record,
293    /// Step 5: write this to every copy after the rename is durable.
294    pub record_after_rename: Record,
295}

297/// What to do with a pending bundle.
298#[derive(Clone, Debug, PartialEq, Eq)]
299pub enum PendingOutcome {
300    /// Activate it.
301    Activate(Box<Activation>),
302    /// It is the current bundle: an idempotent no-op (`policy.applied`).
303    AlreadyCurrent,
304}
305
306/// Decides whether a pending bundle is activated (#225 check order, ADR
307/// 0007 step 1). `record` is the record in force and `active` is
308/// [`Selection::loadable_active`], updated after each activation. On `Err`,
309/// keep what is loaded and report the rejection's event (ADR 0018, "new
310/// bundle fails verification").
311///
312/// # Errors
313///
314/// The [`Rejection`] that refused the bundle.
315pub fn pending(
316    examined: Examined,
317    target: Target,
318    record: Record,
319    active: Option<BundleMeta>,
320    mode: EffectiveMode,
321) -> Result<PendingOutcome, Rejection> {
322    let passed = check(examined, target, mode)?;
323    let meta = passed.meta();
324    if meta.version < record.floor() {
325        return Err(Rejection::Rollback);
326    }
327    if let Some(a) = active {
328        if meta.version < a.version {
329            return Err(Rejection::Rollback);
330        }
331        if meta.version == a.version {
332            return if meta.sha256 == a.sha256 {
333                Ok(PendingOutcome::AlreadyCurrent)
334            } else {
335                Err(Rejection::VersionReused)
336            };
337        }
338    }
339    let signed = matches!(passed, Passed::Verified(_));
340    let before = record.without_fallback();
341    let after = match (signed, active) {
342        (true, Some(a)) => Record::new(target, meta.version, Some(a.version)),
343        (true, None) => Record::new(target, meta.version, None),
344        (false, _) => Ok(before),
345    }
346    .unwrap_or(before);
347    Ok(PendingOutcome::Activate(Box::new(Activation {
348        bundle: meta,
349        signed,
350        copy_active_to_previous: active.is_some(),
351        record_before_rename: before,
352        record_after_rename: after,
353    })))
354}
