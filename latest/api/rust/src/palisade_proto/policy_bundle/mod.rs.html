1//! Policy bundle integrity decisions (part of #225): the anti-rollback
2//! record, the signature mode, and which bundle the agent runs.
3//!
4//! This module is pure: no storage, no OS calls, no clock reads. The agent
5//! (#61 policy cache, #220 tamper protection) supplies the bytes it read and
6//! the time, and carries out what the functions here decide. The decisions
7//! see only [`BundleMeta`] (tenant, target, `policy_version` and content
8//! SHA-256) and the result of the envelope checks ([`Examined`]), which
9//! [`envelope::verify`] produces from the bundle header (#43) and the
10//! agent's trust set.
11//!
12//! - [`record`]: the anti-rollback record. It holds the **floor** (the
13//!   highest version made `active` for a target) and the **fallback** (the
14//!   version in `previous` that `active` directly replaced). Each copy is
15//!   sealed with HMAC-SHA-256, and [`record::RecordCopies`] applies the
16//!   copy-mismatch rule.
17//! - [`SignatureMode`]: `audit`, `warn` or `enforce`, and the rule that
18//!   relaxed modes need a dev build or an explicit group opt-in.
19//! - [`select`]: what to run at startup (`active`, then `previous` under the
20//!   one-step exception, then the emergency policy, then first-install
21//!   audit) and whether a new `pending` bundle is accepted, following the
22//!   failure table of ADR 0018 (which supersedes ADR 0012) and the activation
23//!   order of ADR 0007.
24//! - [`stale`]: staleness, which is reported but never changes what runs.
25//! - [`envelope`]: the signed bundle envelope (#325): signing, and the
26//!   verification that produces [`Examined`].
27//!
28//! Every rejection maps to exactly one event type ([`Rejection::event_type`],
29//! [`select::Finding::event_type`]). None of these functions can return an
30//! allow-all outcome: the only non-enforcing result is
31//! [`select::Load::FirstInstall`], and it is returned only when the device has
32//! never accepted a policy.

52/// What a bundle is for: a group, or one device, within a tenant.
53#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
54pub struct Target {

56    pub tenant: TenantId,
57    /// The group or device.

70/// The facts about a bundle that the decisions need, which
71/// [`envelope::verify`] reads from the bundle header.
72#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
73pub struct BundleMeta {
74    /// The tenant and target named in the envelope.
75    pub target: Target,
76    /// `policy_version`: monotonic per target, starting at 1.
77    pub version: u64,
78    /// SHA-256 of the canonical bundle bytes ([`content_hash`]).
79    pub sha256: Sha256Hex,
80}
81
82/// SHA-256 of the canonical bundle bytes, for [`BundleMeta::sha256`] and for
83/// the periodic in-memory re-check (#225).
84#[must_use]
85pub fn content_hash(bundle: &[u8]) -> Sha256Hex {
86    Sha256Hex::from_bytes(crypto::sha256(bundle))
87}

89/// The result of the envelope checks on one bundle file, before this
90/// module decides anything (#225 check order, steps 1, 2, 5 and 6), as
91/// [`envelope::verify`] returns it. Steps 3 (target) and 4 (version) are
92/// applied here.
93#[derive(Clone, Copy, Debug, PartialEq, Eq)]
94pub enum Examined {
95    /// Step 1 failed: the envelope is over the size limit, does not parse,
96    /// or its bytes do not match the recorded hash. Never loadable, in any
97    /// signature mode.
98    Unreadable,
99    /// The envelope parsed.
100    Parsed {
101        /// The envelope's metadata. Only trustworthy when `signature_valid`.
102        meta: BundleMeta,
103        /// Step 2: at least one signature from an active trusted
104        /// `policy-signing` key verifies.
105        signature_valid: bool,
106        /// Step 5: `min_agent_version` is at or below this agent's version.
107        agent_supported: bool,
108        /// Step 6: the bundle schema validates (#43). Validate the body only
109        /// after the signature verifies, or when the effective mode
110        /// tolerates an invalid signature; otherwise pass `false`, because
111        /// the signature check rejects it first.
112        schema_supported: bool,
113    },
114}

116/// Why a bundle was refused. Each maps to exactly one event type.
117#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
118pub enum Rejection {
119    /// The envelope could not be read (step 1). Reported as
120    /// `policy.signature.invalid`, because the bundle cannot be
121    /// authenticated.
122    Unreadable,
123    /// No valid trusted signature (step 2), in `enforce` mode.
124    SignatureInvalid,
125    /// The tenant or target is not this device's (step 3).
126    TargetMismatch,
127    /// The version is below the anti-rollback floor or the current bundle,
128    /// or is 0 (step 4).
129    Rollback,
130    /// The version equals the current bundle's but the content differs.
131    /// The server never reuses a version, so this is treated as a replay.
132    VersionReused,
133    /// `min_agent_version` is above this agent's (step 5). Reported as
134    /// `policy.schema_unsupported`, which prompts the same agent upgrade.
135    AgentTooOld,
136    /// The bundle schema is unknown or newer (step 6).
137    SchemaUnsupported,
138}
139
140impl Rejection {
141    /// The event this rejection raises.
142    #[must_use]
143    pub const fn event_type(self) -> EventType {
144        match self {
145            Self::Unreadable | Self::SignatureInvalid => EventType::PolicySignatureInvalid,
146            Self::TargetMismatch => EventType::PolicyTargetMismatch,
147            Self::Rollback | Self::VersionReused => EventType::PolicyRollbackRejected,
148            Self::AgentTooOld | Self::SchemaUnsupported => EventType::PolicySchemaUnsupported,
149        }
150    }
151}
152
153impl fmt::Display for Rejection {
154    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
155        f.write_str(match self {
156            Self::Unreadable => "the bundle envelope cannot be read",
157            Self::SignatureInvalid => "no valid signature from a trusted policy-signing key",
158            Self::TargetMismatch => "the bundle is for another tenant or target",
159            Self::Rollback => "the bundle version is below the accepted version",
160            Self::VersionReused => "the bundle reuses the current version with other content",
161            Self::AgentTooOld => "the bundle needs a newer agent",
162            Self::SchemaUnsupported => "the bundle schema is not supported",
163        })
164    }
165}

169/// `policy.signature_mode` (#225).
170#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
171pub enum SignatureMode {
172    /// Load bundles whose signature is invalid, and raise
173    /// `policy.signature.invalid`.
174    Audit,
175    /// As `audit`, plus a tray notification.
176    Warn,
177    /// Refuse them and keep the last known good policy. The default.
178    #[default]
179    Enforce,
180}

182/// Where a relaxed signature mode may come from.
183#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
184pub struct ModeGate {
185    /// This is a dev build of the agent.
186    pub dev_build: bool,
187    /// The device's group explicitly allows a relaxed mode. Take this only
188    /// from a bundle whose signature verified, so that an unsigned bundle
189    /// can never relax the check that would refuse it.
190    pub group_opt_in: bool,
191}
192
193/// The signature mode in force, after the gate.
194#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
195pub struct EffectiveMode {
196    mode: SignatureMode,
197}
198
199impl EffectiveMode {
200    /// `enforce`.
201    pub const ENFORCE: Self = Self {
202        mode: SignatureMode::Enforce,
203    };
204
205    /// The mode in force: `requested` if it is `enforce` or the gate allows
206    /// a relaxed mode, otherwise `enforce`.
207    #[must_use]
208    pub const fn resolve(requested: SignatureMode, gate: ModeGate) -> Self {
209        let allowed = gate.dev_build || gate.group_opt_in;
210        Self {
211            mode: if allowed {
212                requested
213            } else {
214                SignatureMode::Enforce
215            },
216        }
217    }
218
219    /// The mode.
220    #[must_use]
221    pub const fn mode(self) -> SignatureMode {
222        self.mode
223    }
224
225    /// Whether a bundle with an invalid signature may load.
226    #[must_use]
227    pub const fn tolerates_invalid_signature(self) -> bool {
228        !matches!(self.mode, SignatureMode::Enforce)
229    }
230
231    /// Whether the console shows its warning banner for this device.
232    #[must_use]
233    pub const fn console_banner(self) -> bool {
234        self.tolerates_invalid_signature()
235    }
236
237    /// Whether loading a bundle with an invalid signature also notifies the
238    /// user in the tray.
239    #[must_use]
240    pub const fn tray_notification(self) -> bool {
241        matches!(self.mode, SignatureMode::Warn)
242    }
243}
244
245impl Default for EffectiveMode {
246    fn default() -> Self {
247        Self::ENFORCE
248    }
249}

251/// How a bundle passed the checks.
252#[derive(Clone, Copy, Debug, PartialEq, Eq)]
253pub(crate) enum Passed {
254    /// Every check passed.
255    Verified(BundleMeta),
256    /// Every check except the signature passed, and the mode tolerates that.
257    /// Such a bundle never moves the floor, never becomes a fallback, and is
258    /// never copied to `previous`.
259    Unsigned(BundleMeta),
260}
261
262impl Passed {
263    pub(crate) const fn meta(self) -> BundleMeta {
264        match self {
265            Self::Verified(m) | Self::Unsigned(m) => m,
266        }
267    }
268}

270/// Steps 1, 2, 3, 5 and 6 in #225's order. The version (step 4) depends on
271/// the caller's state, so callers check it.
272pub(crate) fn check(
273    examined: Examined,
274    expected: Target,
275    mode: EffectiveMode,
276) -> Result<Passed, Rejection> {
277    let Examined::Parsed {
278        meta,
279        signature_valid,
280        agent_supported,
281        schema_supported,
282    } = examined
283    else {
284        return Err(Rejection::Unreadable);
285    };
286    if !signature_valid && !mode.tolerates_invalid_signature() {
287        return Err(Rejection::SignatureInvalid);
288    }
289    if meta.target != expected {
290        return Err(Rejection::TargetMismatch);
291    }
292    if meta.version == 0 {
293        return Err(Rejection::Rollback);
294    }
295    if !agent_supported {
296        return Err(Rejection::AgentTooOld);
297    }
298    if !schema_supported {
299        return Err(Rejection::SchemaUnsupported);
300    }
301    Ok(if signature_valid {
302        Passed::Verified(meta)
303    } else {
304        Passed::Unsigned(meta)
305    })
