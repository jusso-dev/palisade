5use std::sync::atomic::{AtomicU64, Ordering};
6use std::time::Duration;

8/// Counters updated by the hash cache and [`super::cache::hash_through`].
9/// Every counter only goes up; read them with [`Self::snapshot`].
10#[derive(Debug, Default)]
11pub struct HashMetrics {
12    bytes_hashed: AtomicU64,
13    hashes: AtomicU64,
14    hash_nanos: AtomicU64,
15    hits: AtomicU64,
16    misses: AtomicU64,
17    evictions: AtomicU64,
18    tamper_events: AtomicU64,
19    unstable: AtomicU64,
20}
21
22/// The counters at one moment.
23#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
24pub struct HashMetricsSnapshot {
25    /// Bytes read and hashed.
26    pub bytes_hashed: u64,
27    /// Files hashed (each attempt counts, including ones thrown away
28    /// because the file changed).
29    pub hashes: u64,
30    /// Total time spent hashing.
31    pub hash_time: Duration,
32    /// Cache lookups that returned a value.
33    pub hits: u64,
34    /// Cache lookups that did not.
35    pub misses: u64,
36    /// Entries evicted to stay within the cache's bounds.
37    pub evictions: u64,
38    /// Persisted records rejected because their MAC did not verify.
39    pub tamper_events: u64,
40    /// Files that kept changing while they were hashed.
41    pub unstable: u64,
42}
43
44impl HashMetricsSnapshot {
45    /// Hits over lookups, from 0.0 to 1.0. `None` before any lookup.
46    #[must_use]
47    pub fn hit_ratio(&self) -> Option<f64> {
48        let lookups = self.hits.saturating_add(self.misses);
49        (lookups > 0).then(|| to_f64(self.hits) / to_f64(lookups))
50    }
51
52    /// The mean time per hash. `None` before any hash.
53    #[must_use]
54    pub fn average_latency(&self) -> Option<Duration> {
55        (self.hashes > 0).then(|| {
56            let nanos = self.hash_time.as_nanos() / u128::from(self.hashes);
57            Duration::from_nanos(u64::try_from(nanos).unwrap_or(u64::MAX))
58        })
59    }
60}
61
62#[allow(clippy::cast_precision_loss)] // A ratio needs no exact counts.
63const fn to_f64(n: u64) -> f64 {
64    n as f64
65}

67impl HashMetrics {
68    /// The counters now.
69    #[must_use]
70    pub fn snapshot(&self) -> HashMetricsSnapshot {
71        let get = |c: &AtomicU64| c.load(Ordering::Relaxed);
72        HashMetricsSnapshot {
73            bytes_hashed: get(&self.bytes_hashed),
74            hashes: get(&self.hashes),
75            hash_time: Duration::from_nanos(get(&self.hash_nanos)),
76            hits: get(&self.hits),
77            misses: get(&self.misses),
78            evictions: get(&self.evictions),
79            tamper_events: get(&self.tamper_events),
80            unstable: get(&self.unstable),
81        }
82    }
83
84    /// Records one hash of `bytes` bytes that took `elapsed`.
85    pub fn record_hash(&self, bytes: u64, elapsed: Duration) {
86        add(&self.bytes_hashed, bytes);
87        add(&self.hashes, 1);
88        add(
89            &self.hash_nanos,
90            u64::try_from(elapsed.as_nanos()).unwrap_or(u64::MAX),
91        );
92    }
93
94    pub(crate) fn record_hit(&self) {
95        add(&self.hits, 1);
96    }
97
98    pub(crate) fn record_miss(&self) {
99        add(&self.misses, 1);
100    }
101
102    pub(crate) fn record_eviction(&self) {
103        add(&self.evictions, 1);
104    }
105
106    pub(crate) fn record_tamper(&self) {
107        add(&self.tamper_events, 1);
108    }
109
110    pub(crate) fn record_unstable(&self) {
111        add(&self.unstable, 1);
112    }
113}

115/// Adds without wrapping: a counter stops at `u64::MAX`.
116fn add(counter: &AtomicU64, n: u64) {
117    // A compare-exchange loop: `fetch_update` is deprecated on the pinned
118    // toolchain, and this works on every supported one.
119    let mut current = counter.load(Ordering::Relaxed);
120    loop {
121        match counter.compare_exchange_weak(
122            current,
123            current.saturating_add(n),
124            Ordering::Relaxed,
125            Ordering::Relaxed,
126        ) {
127            Ok(_) => return,
128            Err(actual) => current = actual,
129        }
130    }
