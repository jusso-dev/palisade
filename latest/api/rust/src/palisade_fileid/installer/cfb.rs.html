1//! A bounded reader for compound files (MS-CFB, "structured storage"), the
2//! container Windows Installer packages use.
3//!
4//! The whole file is in memory. [`CompoundFile::parse`] checks the header,
5//! loads the FAT, the mini FAT and the directory, walks the directory tree
6//! from the root and builds an [`Entry`] tree of every reachable stream
7//! and storage. Stream data is read on demand by [`CompoundFile::stream`].
8//!
9//! # Fail closed
10//!
11//! A package's identity and its signature both rest on what this reader
12//! returns, so anything ambiguous is an error, never a guess:
13//!
14//! - a sector chain that loops, runs off the file or the FAT, or is shorter
15//!   or longer than its stream (every chain of a known length must end in
16//!   `ENDOFCHAIN` exactly where its data does), and a DIFAT chain longer
17//!   than the FAT;
18//! - a sector (or mini sector) claimed by two chains, or by a chain and the
19//!   FAT, DIFAT, directory or mini FAT, so no two streams can share bytes
20//!   and no data is read twice;
21//! - a directory entry reached twice, a child of the wrong type, two
22//!   siblings with the same name (names compare case-insensitively, as the
23//!   format requires), or a name that is empty or has a character the
24//!   format forbids (`/`, `\`, `:`, `!`);
25//! - a header that is not version 3 with 512-byte sectors or version 4 with
26//!   4096-byte sectors, a mini sector size other than 64, or a mini stream
27//!   cutoff other than 4096.
28//!
29//! Unreachable directory entries are ignored, as the OS ignores them.
30//!
31//! For version 3 files, the upper 32 bits of a stream size are ignored, as
32//! the format specification recommends (older writers left them
33//! uninitialised).
34
35#![forbid(unsafe_code)]

40/// The compound file signature.
41pub(crate) const SIGNATURE: [u8; 8] = [0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1];

64/// Why a compound file could not be read.
65#[derive(Debug, Clone, PartialEq, Eq, Hash)]
66#[non_exhaustive]
67pub enum CfbError {
68    /// Shorter than the header, or not a compound file.
69    NotCompoundFile,
70    /// A header field has a value the format does not allow.
71    BadHeader(&'static str),
72    /// A sector number is past the end of the file or of the FAT.
73    SectorOutOfRange(u32),
74    /// A sector chain ends before its stream does.
75    TruncatedChain,
76    /// A sector chain does not end (with `ENDOFCHAIN`) where its stream
77    /// does.
78    ChainTooLong,
79    /// A sector or mini sector is claimed twice: a loop, or two chains that
80    /// share data.
81    SharedSector(u32),
82    /// A directory entry is malformed (bad name length, unknown type, a
83    /// child or sibling ID out of range).
84    BadDirectoryEntry(u32),
85    /// A directory entry is reached twice while walking the tree.
86    DirectoryCycle(u32),
87    /// Two entries in one storage have the same name.
88    DuplicateName,
89    /// More than [`MAX_DIRECTORY_ENTRIES`] entries, or storages nested
90    /// deeper than [`MAX_STORAGE_DEPTH`].
91    TooComplex,
92}
93
94impl fmt::Display for CfbError {
95    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
96        match self {
97            Self::NotCompoundFile => f.write_str("not a compound file"),
98            Self::BadHeader(what) => write!(f, "bad compound file header: {what}"),
99            Self::SectorOutOfRange(s) => write!(f, "sector {s} is out of range"),
100            Self::TruncatedChain => f.write_str("a sector chain ends before its stream"),
101            Self::ChainTooLong => f.write_str("a sector chain runs past its stream"),
102            Self::SharedSector(s) => write!(f, "sector {s} is claimed twice"),
103            Self::BadDirectoryEntry(i) => write!(f, "directory entry {i} is malformed"),
104            Self::DirectoryCycle(i) => write!(f, "directory entry {i} is reached twice"),
105            Self::DuplicateName => f.write_str("two entries in one storage share a name"),
106            Self::TooComplex => f.write_str("too many directory entries or storage levels"),
107        }
108    }
109}

115/// One reachable directory entry: a stream or a storage.
116#[derive(Debug, Clone, PartialEq, Eq)]
117pub struct Entry {
118    /// The name as UTF-16 code units, without the terminator.
119    pub name: Vec<u16>,
120    /// Its directory entry number.
121    pub id: u32,
122    /// The CLSID (16 bytes, as stored). All zero for streams.
123    pub clsid: [u8; 16],
124    /// The state bits.
125    pub state_bits: [u8; 4],
126    /// Creation and modification times (FILETIME, as stored).
127    pub times: [u8; 16],
128    /// The stream size, or 0 for a storage.
129    pub size: u64,
130    /// The stream's size field exactly as stored (all 8 bytes).
131    pub raw_size: [u8; 8],
132    /// The children, for a storage. Empty for a stream.
133    pub children: Vec<Entry>,
134    /// Whether this is a storage (the root counts as one).
135    pub is_storage: bool,
136    /// Where the stream's data is: a list of `(offset, len)` byte ranges
137    /// in the file, in order, summing to `size`.
138    extents: Vec<(usize, usize)>,
139}
140
141impl Entry {
142    /// The name as text, lossily (MSI stream names use private-use code
143    /// points that decode fine; unpaired surrogates become U+FFFD).
144    #[must_use]
145    pub fn name_lossy(&self) -> String {
146        String::from_utf16_lossy(&self.name)
147    }
148
149    /// The child with this name, compared case-insensitively as the
150    /// format (and the OS) compares names. Names are unique under that
151    /// comparison, so there is at most one.
152    #[must_use]
153    pub fn child(&self, name: &[u16]) -> Option<&Entry> {
154        let want = fold(name);
155        self.children.iter().find(|c| fold(&c.name) == want)
156    }
157
158    /// Whether this entry's name equals `name` case-insensitively.
159    #[must_use]
160    pub fn is_named(&self, name: &str) -> bool {
161        let units: Vec<u16> = name.encode_utf16().collect();
162        fold(&self.name) == fold(&units)
163    }
164}

166/// A parsed compound file.
167#[derive(Debug, Clone)]
168pub struct CompoundFile<'a> {
169    data: &'a [u8],
170    /// The root storage.
171    pub root: Entry,
172    /// The major version (3 or 4).
173    pub version: u16,
174    /// Bytes after the last sector any structure or stream uses. Nothing
175    /// covers them, so they can hide data in a signed package.
176    pub trailing_bytes: u64,
177}
178
179impl<'a> CompoundFile<'a> {
180    /// Parses `data`.
181    pub fn parse(data: &'a [u8]) -> Result<Self> {
182        Parser::new(data)?.run()
183    }
184
185    /// The content of `entry`, a stream of this file. Its extents were
186    /// checked during parsing, so this cannot fail; a storage reads as
187    /// empty.
188    #[must_use]
189    pub fn stream(&self, entry: &Entry) -> Vec<u8> {
190        let mut out = Vec::with_capacity(usize::try_from(entry.size).unwrap_or(0));
191        for &(offset, len) in &entry.extents {
192            if let Some(chunk) = self.data.get(offset..offset.saturating_add(len)) {
193                out.extend_from_slice(chunk);
194            }
195        }
196        out
197    }
198
199    /// Feeds the content of `entry` to `sink` chunk by chunk.
200    pub fn for_each_chunk(&self, entry: &Entry, mut sink: impl FnMut(&[u8])) {
201        for &(offset, len) in &entry.extents {
202            if let Some(chunk) = self.data.get(offset..offset.saturating_add(len)) {
203                sink(chunk);
204            }
205        }
206    }
207}

209/// Ownership of sectors, so that no sector is used twice.
210struct Claims(Vec<bool>);
211
212impl Claims {
213    fn claim(&mut self, sector: u32) -> Result<()> {
214        let slot = usize::try_from(sector)
215            .ok()
216            .and_then(|i| self.0.get_mut(i))
217            .ok_or(CfbError::SectorOutOfRange(sector))?;
218        if *slot {
219            return Err(CfbError::SharedSector(sector));
220        }
221        *slot = true;
222        Ok(())
223    }
224}
225
226fn u16_at(data: &[u8], at: usize) -> u16 {
227    let mut b = [0_u8; 2];
228    if let Some(s) = data.get(at..at.saturating_add(2)) {
229        b.copy_from_slice(s);
230    }
231    u16::from_le_bytes(b)
232}
233
234fn u32_at(data: &[u8], at: usize) -> u32 {
235    let mut b = [0_u8; 4];
236    if let Some(s) = data.get(at..at.saturating_add(4)) {
237        b.copy_from_slice(s);
238    }
239    u32::from_le_bytes(b)
240}
241
242fn array_at<const N: usize>(data: &[u8], at: usize) -> [u8; N] {
243    let mut b = [0_u8; N];
244    if let Some(s) = data.get(at..at.saturating_add(N)) {
245        b.copy_from_slice(s);
246    }
247    b
248}

250struct Parser<'a> {
251    data: &'a [u8],
252    version: u16,
253    sector_len: usize,
254    /// Whole sectors after the header.
255    sector_count: u32,
256    fat: Vec<u32>,
257    claims: Claims,
258    minifat: Vec<u32>,
259    mini_claims: Claims,
260    /// The mini stream's sectors, in order.
261    mini_stream: Vec<u32>,
262    mini_stream_len: u64,
263    dir: Vec<u8>,
264}
265
266impl<'a> Parser<'a> {
267    fn new(data: &'a [u8]) -> Result<Self> {
268        if data.len() < HEADER_LEN || data.get(..8) != Some(&SIGNATURE[..]) {
269            return Err(CfbError::NotCompoundFile);
270        }
271        let version = u16_at(data, 0x1a);
272        let byte_order = u16_at(data, 0x1c);
273        let sector_shift = u16_at(data, 0x1e);
274        let mini_shift = u16_at(data, 0x20);
275        if byte_order != 0xfffe {
276            return Err(CfbError::BadHeader("byte order"));
277        }
278        let sector_len = match (version, sector_shift) {
279            (3, 9) => 512,
280            (4, 12) => 4096,
281            _ => return Err(CfbError::BadHeader("version and sector size")),
282        };
283        if mini_shift != 6 {
284            return Err(CfbError::BadHeader("mini sector size"));
285        }
286        if u64::from(u32_at(data, 0x38)) != MINI_STREAM_CUTOFF {
287            return Err(CfbError::BadHeader("mini stream cutoff"));
288        }
289        if version == 3 && u32_at(data, 0x28) != 0 {
290            return Err(CfbError::BadHeader("directory sector count"));
291        }
292        if data.len() < sector_len {
293            return Err(CfbError::BadHeader("file shorter than one sector"));
294        }
295        // A final sector may be cut short; stream data in it is checked
296        // against the end of the file when read.
297        let body = data.len().saturating_sub(sector_len);
298        let sector_count = u32::try_from(body.div_ceil(sector_len))
299            .ok()
300            .filter(|&n| n <= MAXREGSECT)
301            .ok_or(CfbError::BadHeader("too many sectors"))?;
302        let claims = Claims(vec![false; usize::try_from(sector_count).unwrap_or(0)]);
303        Ok(Self {
304            data,
305            version,
306            sector_len,
307            sector_count,
308            fat: Vec::new(),
309            claims,
310            minifat: Vec::new(),
311            mini_claims: Claims(Vec::new()),
312            mini_stream: Vec::new(),
313            mini_stream_len: 0,
314            dir: Vec::new(),
315        })
316    }
317
318    /// The byte offset of sector `s`. The sector starts inside the file;
319    /// whether it is long enough is checked where it is read.
320    fn sector(&self, s: u32) -> Result<usize> {
321        if s >= self.sector_count {
322            return Err(CfbError::SectorOutOfRange(s));
323        }
324        usize::try_from(s)
325            .ok()
326            .and_then(|n| n.checked_add(1))
327            .and_then(|n| n.checked_mul(self.sector_len))
328            .filter(|&o| o < self.data.len())
329            .ok_or(CfbError::SectorOutOfRange(s))
330    }
331
332    /// A whole sector (FAT, DIFAT, mini FAT and directory sectors must be
333    /// complete).
334    fn sector_bytes(&self, s: u32) -> Result<&'a [u8]> {
335        let at = self.sector(s)?;
336        self.data
337            .get(at..at.saturating_add(self.sector_len))
338            .ok_or(CfbError::SectorOutOfRange(s))
339    }
340
341    /// `len` bytes of stream data at `at`, which must be inside the file.
342    fn extent(&self, at: usize, len: usize, s: u32) -> Result<(usize, usize)> {
343        if at
344            .checked_add(len)
345            .is_some_and(|end| end <= self.data.len())
346        {
347            Ok((at, len))
348        } else {
349            Err(CfbError::SectorOutOfRange(s))
350        }
351    }

353    fn run(mut self) -> Result<CompoundFile<'a>> {
354        self.load_fat()?;
355        self.load_directory()?;
356        self.load_minifat()?;
357        let root = self.walk()?;
358        // The end of the last claimed sector (or of its stream data, if
359        // that sector is the short final one).
360        let last = self
361            .claims
362            .0
363            .iter()
364            .rposition(|&c| c)
365            .map_or(1, |i| i.saturating_add(2));
366        let used = last.saturating_mul(self.sector_len).min(self.data.len());
367        let trailing_bytes =
368            u64::try_from(self.data.len().saturating_sub(used)).unwrap_or(u64::MAX);
369        Ok(CompoundFile {
370            data: self.data,
371            root,
372            version: self.version,
373            trailing_bytes,
374        })
375    }

377    fn load_fat(&mut self) -> Result<()> {
378        let fat_count = u32_at(self.data, 0x2c);
379        if fat_count > self.sector_count {
380            return Err(CfbError::BadHeader("FAT sector count"));
381        }
382        let fat_count = usize::try_from(fat_count).unwrap_or(usize::MAX);
383        let mut fat_sectors: Vec<u32> = Vec::with_capacity(fat_count);
384        for i in 0..HEADER_DIFAT_ENTRIES.min(fat_count) {
385            fat_sectors.push(u32_at(self.data, 0x4c + 4 * i));
386        }
387        // The DIFAT chain lists the rest.
388        let per_difat = self.sector_len / 4 - 1;
389        let mut next = u32_at(self.data, 0x44);
390        let mut difat_left = u32_at(self.data, 0x48);
391        while fat_sectors.len() < fat_count {
392            if difat_left == 0 || next > MAXREGSECT {
393                return Err(CfbError::BadHeader("DIFAT ends before the FAT"));
394            }
395            self.claims.claim(next)?;
396            let sector = self.sector_bytes(next)?;
397            for i in 0..per_difat {
398                if fat_sectors.len() == fat_count {
399                    break;
400                }
401                fat_sectors.push(u32_at(sector, 4 * i));
402            }
403            next = u32_at(sector, 4 * per_difat);
404            difat_left -= 1;
405        }
406        // The DIFAT chain ends where the FAT does.
407        let read_any = fat_count > HEADER_DIFAT_ENTRIES;
408        if difat_left != 0 || (read_any && next != ENDOFCHAIN && next != FREESECT) {
409            return Err(CfbError::BadHeader("DIFAT longer than the FAT"));
410        }
411        let per_sector = self.sector_len / 4;
412        let mut fat = Vec::with_capacity(fat_count.saturating_mul(per_sector));
413        for &s in &fat_sectors {
414            self.claims.claim(s)?;
415            let sector = self.sector_bytes(s)?;
416            for i in 0..per_sector {
417                fat.push(u32_at(sector, 4 * i));
418            }
419        }
420        self.fat = fat;
421        Ok(())
422    }

424    /// The FAT chain from `start`, `count` sectors long (or to its end when
425    /// `count` is `None`), claiming each sector.
426    fn chain(&mut self, start: u32, count: Option<usize>) -> Result<Vec<u32>> {
427        let mut out = Vec::new();
428        let mut s = start;
429        loop {
430            if count.is_some_and(|c| out.len() == c) {
431                // A chain of a known length must end exactly there: a
432                // longer one would share sectors, loop or hide data.
433                return if s == ENDOFCHAIN {
434                    Ok(out)
435                } else {
436                    Err(CfbError::ChainTooLong)
437                };
438            }
439            if s == ENDOFCHAIN {
440                return match count {
441                    Some(_) => Err(CfbError::TruncatedChain),
442                    None => Ok(out),
443                };
444            }
445            if s > MAXREGSECT {
446                return Err(CfbError::SectorOutOfRange(s));
447            }
448            self.sector(s)?;
449            self.claims.claim(s)?;
450            out.push(s);
451            s = usize::try_from(s)
452                .ok()
453                .and_then(|i| self.fat.get(i))
454                .copied()
455                .ok_or(CfbError::SectorOutOfRange(s))?;
456        }
457    }

459    fn load_directory(&mut self) -> Result<()> {
460        let start = u32_at(self.data, 0x30);
461        // Version 4 states how many directory sectors there are; the chain
462        // must be exactly that long (version 3 states 0, checked earlier).
463        let count = if self.version == 4 {
464            let n = u32_at(self.data, 0x28);
465            if n == 0 || n > self.sector_count {
466                return Err(CfbError::BadHeader("directory sector count"));
467            }
468            Some(usize::try_from(n).map_err(|_| CfbError::BadHeader("directory sector count"))?)
469        } else {
470            None
471        };
472        let sectors = self.chain(start, count)?;
473        let per = self.sector_len / DIR_ENTRY_LEN;
474        if sectors.len().saturating_mul(per) > MAX_DIRECTORY_ENTRIES {
475            return Err(CfbError::TooComplex);
476        }
477        let mut dir = Vec::with_capacity(sectors.len().saturating_mul(self.sector_len));
478        for s in sectors {
479            dir.extend_from_slice(self.sector_bytes(s)?);
480        }
481        if dir.is_empty() {
482            return Err(CfbError::BadHeader("empty directory"));
483        }
484        self.dir = dir;
485        Ok(())
486    }

488    fn load_minifat(&mut self) -> Result<()> {
489        let start = u32_at(self.data, 0x3c);
490        let count = u32_at(self.data, 0x40);
491        if count > self.sector_count {
492            return Err(CfbError::BadHeader("mini FAT sector count"));
493        }
494        let sectors = if count == 0 {
495            Vec::new()
496        } else {
497            self.chain(start, Some(usize::try_from(count).unwrap_or(usize::MAX)))?
498        };
499        let per_sector = self.sector_len / 4;
500        let mut minifat = Vec::with_capacity(sectors.len().saturating_mul(per_sector));
501        for s in sectors {
502            let sector = self.sector_bytes(s)?;
503            for i in 0..per_sector {
504                minifat.push(u32_at(sector, 4 * i));
505            }
506        }
507        self.minifat = minifat;
508
509        // The root entry's stream is the mini stream.
510        let root = self.raw_entry(0)?;
511        if root.kind != TYPE_ROOT {
512            return Err(CfbError::BadDirectoryEntry(0));
513        }
514        let len = root.size(self.version);
515        let needed = self.sectors_for(len, self.sector_len)?;
516        self.mini_stream = if needed == 0 {
517            Vec::new()
518        } else {
519            self.chain(root.start, Some(needed))?
520        };
521        self.mini_stream_len = len;
522        let mini_sectors = len.div_ceil(MINI_SECTOR_LEN as u64);
523        self.mini_claims = Claims(vec![false; usize::try_from(mini_sectors).unwrap_or(0)]);
524        Ok(())
525    }
526
527    fn sectors_for(&self, len: u64, unit: usize) -> Result<usize> {
528        let n = len.div_ceil(unit as u64);
529        // A chain can never be longer than the file has sectors.
530        let limit = u64::from(self.sector_count).saturating_mul(self.sector_len as u64);
531        if len > limit {
532            return Err(CfbError::TruncatedChain);
533        }
534        usize::try_from(n).map_err(|_| CfbError::TruncatedChain)
535    }

537    fn raw_entry(&self, id: u32) -> Result<RawEntry> {
538        let at = usize::try_from(id)
539            .ok()
540            .and_then(|i| i.checked_mul(DIR_ENTRY_LEN))
541            .ok_or(CfbError::BadDirectoryEntry(id))?;
542        let e = self
543            .dir
544            .get(at..at + DIR_ENTRY_LEN)
545            .ok_or(CfbError::BadDirectoryEntry(id))?;
546        let name_len = usize::from(u16_at(e, 0x40));
547        let kind = e.get(0x42).copied().unwrap_or(0);
548        if kind == TYPE_UNUSED {
549            return Ok(RawEntry::unused());
550        }
551        if !matches!(kind, TYPE_STORAGE | TYPE_STREAM | TYPE_ROOT)
552            // At least one character: MS-CFB does not allow an empty name.
553            || !(4..=64).contains(&name_len)
554            || name_len % 2 != 0
555        {
556            return Err(CfbError::BadDirectoryEntry(id));
557        }
558        let units = name_len / 2 - 1;
559        let name: Vec<u16> = (0..units).map(|i| u16_at(e, 2 * i)).collect();
560        // MS-CFB 2.6.1: these characters are not allowed in a name.
561        let illegal = [
562            0,
563            u16::from(b'/'),
564            u16::from(b'\\'),
565            u16::from(b':'),
566            u16::from(b'!'),
567        ];
568        if u16_at(e, 2 * units) != 0 || name.iter().any(|u| illegal.contains(u)) {
569            return Err(CfbError::BadDirectoryEntry(id));
570        }
571        Ok(RawEntry {
572            kind,
573            name,
574            left: u32_at(e, 0x44),
575            right: u32_at(e, 0x48),
576            child: u32_at(e, 0x4c),
577            clsid: array_at(e, 0x50),
578            state_bits: array_at(e, 0x60),
579            times: array_at(e, 0x64),
580            start: u32_at(e, 0x74),
581            raw_size: array_at(e, 0x78),
582        })
583    }

585    fn entry_count(&self) -> u32 {
586        u32::try_from(self.dir.len() / DIR_ENTRY_LEN).unwrap_or(u32::MAX)

589    /// Walks the tree from the root, depth first, with every entry visited
590    /// at most once.
591    fn walk(&mut self) -> Result<Entry> {
592        let count = self.entry_count();
593        let mut visited = vec![false; usize::try_from(count).unwrap_or(0)];
594        let mut mark = |id: u32| -> Result<()> {
595            let slot = usize::try_from(id)
596                .ok()
597                .and_then(|i| visited.get_mut(i))
598                .ok_or(CfbError::BadDirectoryEntry(id))?;
599            if *slot {
600                return Err(CfbError::DirectoryCycle(id));
601            }
602            *slot = true;
603            Ok(())
604        };
605        mark(0)?;
606        let root_raw = self.raw_entry(0)?;
607        let mut root = self.make_entry(0, &root_raw)?;
608        root.children = self.build_children(&root_raw, 0, &mut mark)?;
609        Ok(root)
610    }
611
612    /// The IDs in the sibling tree rooted at `top`, in tree order, each
613    /// marked as visited.
614    fn siblings(&self, top: u32, mark: &mut impl FnMut(u32) -> Result<()>) -> Result<Vec<u32>> {
615        let mut out = Vec::new();
616        if top == NOSTREAM {
617            return Ok(out);
618        }
619        let count = self.entry_count();
620        let mut pending = vec![top];
621        while let Some(id) = pending.pop() {
622            if id >= count {
623                return Err(CfbError::BadDirectoryEntry(id));
624            }
625            mark(id)?;
626            if out.len() >= MAX_DIRECTORY_ENTRIES {
627                return Err(CfbError::TooComplex);
628            }
629            let raw = self.raw_entry(id)?;
630            if !matches!(raw.kind, TYPE_STORAGE | TYPE_STREAM) {
631                return Err(CfbError::BadDirectoryEntry(id));
632            }
633            out.push(id);
634            for next in [raw.left, raw.right] {
635                if next != NOSTREAM {
636                    pending.push(next);
637                }
638            }
639        }
640        Ok(out)
641    }
642
643    fn build_children(
644        &mut self,
645        storage: &RawEntry,
646        depth: usize,
647        mark: &mut impl FnMut(u32) -> Result<()>,
648    ) -> Result<Vec<Entry>> {
649        if depth >= MAX_STORAGE_DEPTH {
650            return Err(CfbError::TooComplex);
651        }
652        let ids = self.siblings(storage.child, mark)?;
653        let mut children = Vec::with_capacity(ids.len());
654        for id in ids {
655            let raw = self.raw_entry(id)?;
656            let mut entry = self.make_entry(id, &raw)?;
657            if raw.kind == TYPE_STORAGE {
658                entry.children = self.build_children(&raw, depth + 1, mark)?;
659            }
660            children.push(entry);
661        }
662        // Names are unique within a storage, compared case-insensitively.
663        let mut keys: Vec<Vec<u16>> = children.iter().map(|c| fold(&c.name)).collect();
664        keys.sort_unstable();
665        if keys.windows(2).any(|w| w.first() == w.get(1)) {
666            return Err(CfbError::DuplicateName);
667        }
668        children.sort_by(|a, b| tree_order(&a.name, &b.name));
669        Ok(children)
670    }
671
672    fn make_entry(&mut self, id: u32, raw: &RawEntry) -> Result<Entry> {
673        let is_storage = raw.kind != TYPE_STREAM;
674        let size = if is_storage {
675            0
676        } else {
677            raw.size(self.version)
678        };
679        let extents = if is_storage || size == 0 {
680            Vec::new()
681        } else if size < MINI_STREAM_CUTOFF {
682            self.mini_extents(raw.start, size)?
683        } else {
684            let n = self.sectors_for(size, self.sector_len)?;
685            let sectors = self.chain(raw.start, Some(n))?;
686            let mut left = size;
687            let mut out = Vec::with_capacity(sectors.len());
688            for s in sectors {
689                let at = self.sector(s)?;
690                let len = usize::try_from(left.min(self.sector_len as u64)).unwrap_or(0);
691                out.push(self.extent(at, len, s)?);
692                left -= len as u64;
693            }
694            out
695        };
696        Ok(Entry {
697            name: raw.name.clone(),
698            id,
699            clsid: raw.clsid,
700            state_bits: raw.state_bits,
701            times: raw.times,
702            size,
703            raw_size: raw.raw_size,
704            children: Vec::new(),
705            is_storage,
706            extents,
707        })
708    }
709
710    fn mini_extents(&mut self, start: u32, size: u64) -> Result<Vec<(usize, usize)>> {
711        let n = usize::try_from(size.div_ceil(MINI_SECTOR_LEN as u64)).unwrap_or(usize::MAX);
712        let mut out = Vec::with_capacity(n);
713        let mut s = start;
714        let mut left = size;
715        for _ in 0..n {
716            if s > MAXREGSECT {
717                return Err(CfbError::TruncatedChain);
718            }
719            self.mini_claims.claim(s)?;
720            let len = left.min(MINI_SECTOR_LEN as u64);
721            let begin = u64::from(s) * MINI_SECTOR_LEN as u64;
722            if begin.saturating_add(len) > self.mini_stream_len {
723                return Err(CfbError::SectorOutOfRange(s));
724            }
725            // Map the mini stream offset to the file.
726            let sector_index = usize::try_from(begin / self.sector_len as u64)
727                .map_err(|_| CfbError::SectorOutOfRange(s))?;
728            let within = usize::try_from(begin % self.sector_len as u64).unwrap_or(0);
729            let host = *self
730                .mini_stream
731                .get(sector_index)
732                .ok_or(CfbError::SectorOutOfRange(s))?;
733            let at = self.sector(host)?.saturating_add(within);
734            let len = usize::try_from(len).unwrap_or(0);
735            out.push(self.extent(at, len, s)?);
736            left -= len as u64;
737            s = usize::try_from(s)
738                .ok()
739                .and_then(|i| self.minifat.get(i))
740                .copied()
741                .ok_or(CfbError::SectorOutOfRange(s))?;
742        }
743        if s != ENDOFCHAIN {
744            return Err(CfbError::ChainTooLong);
745        }
746        Ok(out)
747    }
748}

750struct RawEntry {
751    kind: u8,
752    name: Vec<u16>,
753    left: u32,
754    right: u32,
755    child: u32,
756    clsid: [u8; 16],

761}

764    fn unused() -> Self {
765        Self {
766            kind: TYPE_UNUSED,
767            name: Vec::new(),
768            left: NOSTREAM,
769            right: NOSTREAM,
770            child: NOSTREAM,
771            clsid: [0; 16],

776        }

779    fn size(&self, version: u16) -> u64 {
780        let full = u64::from_le_bytes(self.raw_size);
781        if version == 3 {
782            full & 0xffff_ffff
783        } else {
784            full
785        }
786    }
787}

789/// The case-folded form of a name, for duplicate detection: each code unit
790/// that is a character with a single-unit uppercase form is uppercased.
791fn fold(name: &[u16]) -> Vec<u16> {
792    name.iter()
793        .map(|&u| {
794            char::from_u32(u32::from(u))
795                .and_then(|c| {
796                    let mut up = c.to_uppercase();
797                    match (up.next(), up.next()) {
798                        (Some(x), None) => u16::try_from(u32::from(x)).ok(),
799                        _ => None,
800                    }
801                })
802                .unwrap_or(u)
803        })
804        .collect()
805}
806
807/// The format's directory order: shorter names first, then by uppercased
808/// code unit. Used only to give children a stable order.
809fn tree_order(a: &[u16], b: &[u16]) -> Ordering {
810    a.len().cmp(&b.len()).then_with(|| fold(a).cmp(&fold(b)))
811}
