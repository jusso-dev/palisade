1//! Alternate data streams as file-system context (#54): each named stream
2//! with its size, a tag for the streams the OS and common software write
3//! for their own bookkeeping, and whether its content looks executable.
4//!
5//! A benign tag says who usually writes a stream of that name, not that
6//! its content is safe: anyone who can write the file can write a stream
7//! with any name. So a tagged stream is still listed, and still sniffed.
8
9#![forbid(unsafe_code)]
10

17/// Why a stream name is known.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
19pub enum BenignStream {
20    /// `Zone.Identifier`: the mark-of-the-web.
21    ZoneIdentifier,
22    /// The OS download-reputation check's stream. Its exact name is an OS
23    /// identifier, matched here as the policy compilers match OS literals
24    /// (docs/NAMING.md).
25    DownloadReputation,
26    /// `encryptable`: the OS thumbnail cache marker.
27    Encryptable,
28    /// `favicon`: a browser shortcut icon.
29    Favicon,
30    /// `{4c8cc155-6c1e-11d1-8e41-00c04fb9386d}`: the OS summary-information
31    /// property stream (thumbnails and document properties).
32    SummaryInformation,
33    /// File-sync client metadata, named `com.<name>.attrs`.
34    SyncMetadata,
35    /// `Palisade.Origin`: Palisade's own provenance tag.
36    PalisadeOrigin,
37}
38
39impl BenignStream {
40    /// The tag for a stream name, compared ignoring ASCII case as NTFS
41    /// compares stream names. `None` if the name is not a known one.
42    #[must_use]
43    pub fn from_name(name: &str) -> Option<Self> {
44        let n = name.to_ascii_lowercase();
45        Some(match n.as_str() {
46            "zone.identifier" => Self::ZoneIdentifier,
47            "smartscreen" => Self::DownloadReputation,
48            "encryptable" => Self::Encryptable,
49            "favicon" => Self::Favicon,
50            "{4c8cc155-6c1e-11d1-8e41-00c04fb9386d}" | "\u{5}summaryinformation" => {
51                Self::SummaryInformation
52            }
53            "palisade.origin" => Self::PalisadeOrigin,
54            _ if n
55                .strip_prefix("com.")
56                .and_then(|r| r.strip_suffix(".attrs"))
57                .is_some_and(|mid| !mid.is_empty()) =>
58            {
59                Self::SyncMetadata
60            }
61            _ => return None,
62        })
63    }
64}

66/// One alternate data stream, as [`FsStreams`] reports it.
67#[derive(Debug, Clone, PartialEq, Eq)]
68pub struct StreamInfo {
69    /// The stream name, without the leading `:` or the `:$DATA` type.
70    pub name: String,
71    /// The size in bytes.
72    pub size: u64,
73    /// The known name, if it is one.
74    pub benign: Option<BenignStream>,
75    /// Whether its content is executable, or may be: [`sniff_stream`].
76    /// `Some(true)` also when the classifier could not establish what the
77    /// content is (a parse error), so a damaged or oversized container is
78    /// never reported as plain data. `None` if it was not read.
79    pub is_executable_content: Option<bool>,
80}

82/// The alternate data streams of one file: `streams` in #54.
83#[derive(Debug, Clone, PartialEq, Eq, Default)]
84pub struct FsStreams {
85    /// Every named data stream, in the order the OS lists them.
86    pub streams: Vec<StreamInfo>,
87    /// The OS listed more streams than were read.
88    pub truncated: bool,
89}
90
91impl FsStreams {
92    /// The named data streams of `list`, tagged, with no content read.
93    #[must_use]
94    pub fn from_list(list: &StreamList) -> Self {
95        Self {
96            streams: list
97                .alternate()
98                .map(|s| StreamInfo {
99                    name: s.name.clone(),
100                    size: s.size,
101                    benign: BenignStream::from_name(&s.name),
102                    is_executable_content: None,
103                })
104                .collect(),
105            truncated: list.truncated,
106        }
107    }
108
109    /// Whether any stream is known to hold executable-looking content.
110    #[must_use]
111    pub fn any_executable(&self) -> bool {
112        self.streams
113            .iter()
114            .any(|s| s.is_executable_content == Some(true))
115    }
116
117    /// Whether the streams may hold executable content: one is known to,
118    /// one was not sniffed (whatever its name), or the list was cut short.
119    /// A missing fact never reads as "no executable content" (ADR 0018).
120    #[must_use]
121    pub fn may_hide_executable(&self) -> bool {
122        self.truncated
123            || self
124                .streams
125                .iter()
126                .any(|s| s.is_executable_content != Some(false))
127    }
128
129    /// The streams that are not known names.
130    pub fn unknown(&self) -> impl Iterator<Item = &StreamInfo> {
131        self.streams.iter().filter(|s| s.benign.is_none())
132    }
133}
134
135/// Whether a classification of a stream's content counts as executable
136/// for [`StreamInfo::is_executable_content`]: executable content, or any
137/// classification with a parse error. A parse error means the kind was not
138/// established (a compound file whose directory could not be read is
139/// `OtherCfb`, though it may be an installer), and a missing fact must not
140/// make the stream look like data (ADR 0018). `Some(true)` rather than
141/// `None` is used for it, so [`FsStreams::any_executable`] fails closed as
142/// well as [`FsStreams::may_hide_executable`].
143#[must_use]
144pub fn counts_as_executable(c: &Classification) -> bool {
145    c.is_executable_content() || c.parse_error.is_some()
146}
147
148/// Classifies the content of the stream named `name` from `reader`, the
149/// open stream, and says whether it [`counts_as_executable`].
150///
151/// The whole stream is available to the classifier, which bounds its own
152/// reads (the first 64 KiB, then a few targeted reads), so a structure past
153/// the first 64 KiB, such as an installer's directory, is still found. The
154/// stream's own name is the name hint: the classifier uses the extension to
155/// tell batch, PowerShell and HTML application content from plain text,
156/// and never lets a harmless extension hide content it recognises.
157pub fn sniff_stream<R: Read + Seek + ?Sized>(reader: &mut R, name: &str) -> bool {
158    counts_as_executable(&classify(reader, Some(OsStr::new(name))))
159}

198    #[test]
199    fn streams_from_a_list() {
200        let list = StreamList {
201            streams: vec![
202                entry("", 10),
203                entry("Zone.Identifier", 26),
204                entry("payload", 4096),
205                StreamEntry {
206                    kind: "$INDEX_ALLOCATION".to_owned(),
207                    ..entry("$I30", 0)
208                },
209            ],
210            truncated: false,
211        };
212        let mut s = FsStreams::from_list(&list);
213        // Only named data streams, in order.
214        let names: Vec<_> = s.streams.iter().map(|x| x.name.as_str()).collect();
215        assert_eq!(names, ["Zone.Identifier", "payload"]);
216        assert_eq!(s.unknown().count(), 1);
217        // Nothing sniffed yet: may hide code, though none is known.
218        assert!(!s.any_executable());
219        assert!(s.may_hide_executable());
220        for x in &mut s.streams {
221            x.is_executable_content = Some(false);
222        }
223        assert!(!s.may_hide_executable());
224        s.streams[1].is_executable_content = Some(true);
225        assert!(s.any_executable() && s.may_hide_executable());
226        // A cut-short list may hide code past the cut.
227        let cut = FsStreams {
228            truncated: true,
229            ..FsStreams::default()
230        };
231        assert!(cut.may_hide_executable());
232        assert!(!FsStreams::default().may_hide_executable());
233    }

250    /// A version 3 compound file whose directory starts at `sector`, with
251    /// `clsid` on its root entry: just the header and the root entry, all
252    /// the classifier reads.
253    fn compound_file(sector: u32, clsid: [u8; 16]) -> Vec<u8> {
254        let offset = (usize::try_from(sector).unwrap() + 1) << 9;
255        let mut b = vec![0_u8; offset + 128];
256        b[..8].copy_from_slice(&[0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1]);
257        b[0x18..0x1a].copy_from_slice(&0x3e_u16.to_le_bytes());
258        b[0x1a..0x1c].copy_from_slice(&3_u16.to_le_bytes());
259        b[0x1c..0x1e].copy_from_slice(&0xfffe_u16.to_le_bytes());
260        b[0x1e..0x20].copy_from_slice(&9_u16.to_le_bytes());
261        b[0x30..0x34].copy_from_slice(&sector.to_le_bytes());
262        // The root entry: "Root Entry", type 5, the CLSID.
263        let name: Vec<u8> = "Root Entry"
264            .encode_utf16()
265            .chain([0])
266            .flat_map(u16::to_le_bytes)
267            .collect();
268        b[offset..offset + name.len()].copy_from_slice(&name);
269        b[offset + 0x40..offset + 0x42]
270            .copy_from_slice(&u16::try_from(name.len()).unwrap().to_le_bytes());
271        b[offset + 0x42] = 5;
272        b[offset + 0x50..offset + 0x60].copy_from_slice(&clsid);
273        b
274    }

281    #[test]
282    fn an_installer_whose_directory_is_past_64_kib_is_executable() {
283        // The reviewer's case: the directory at sector 200, 100 KiB in.
284        let msi = compound_file(200, MSI);
285        assert!(msi.len() > 64 * 1024);
286        let full = classify(&mut std::io::Cursor::new(&msi[..]), None);
287        assert_eq!(
288            full.content_kind,
289            crate::filetype::FileKind::Msi,
290            "{full:?}"
291        );
292        assert!(sniff(&msi, "payload"));
293        // Only the first 64 KiB, as the stream reader used to pass: the
294        // directory cannot be read, the kind is `OtherCfb` with a parse
295        // error, and that still counts as executable, never as data.
296        let head = classify(&mut std::io::Cursor::new(&msi[..64 * 1024]), None);
297        assert!(
298            head.parse_error.is_some() && !head.is_executable_content(),
299            "{head:?}"
300        );
301        assert!(counts_as_executable(&head));
302        assert!(sniff(&msi[..64 * 1024], "payload"));
303        // A plain compound file with a readable directory is data.
304        let other = compound_file(200, [7; 16]);
305        assert!(!sniff(&other, "payload"));
306    }

308    #[test]
309    fn the_stream_name_is_the_name_hint() {
310        for (name, content) in [
311            ("x.bat", &b"@echo off\r\ncopy a b\r\n"[..]),
312            ("x.cmd", &b"@echo off\r\ncopy a b\r\n"[..]),
313            ("x.ps1", &b"Get-ChildItem | Remove-Item\r\n"[..]),
314            (
315                "x.hta",
316                &b"<html><head><title>t</title></head><body>hi</body></html>"[..],
317            ),
318        ] {
319            assert!(sniff(content, name), "{name}");
320        }
321        // A harmless name does not hide content the classifier knows.
322        let vbs = b"Set sh = CreateObject(\"WScript.Shell\")\r\nsh.Run \"cmd.exe\"\r\n";
323        assert!(sniff(vbs, "notes.txt"));
324        assert!(sniff(vbs, "payload"));
325    }
