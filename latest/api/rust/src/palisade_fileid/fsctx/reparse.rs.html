1//! Reparse points (#54): what a link, junction, mount point, app execution
2//! alias, cloud placeholder or deduplicated file says it is.
3//!
4//! The Windows backend reads the reparse data buffer of a file opened
5//! without following it (`FSCTL_GET_REPARSE_POINT`) and hands the bytes to
6//! [`parse_reparse_point`]. The parser is pure, bounded and fuzzed: the
7//! bytes come from the file system, but a standard user can create
8//! symbolic links (with developer mode), junctions and app execution
9//! aliases in folders they own, so every offset and length is checked and
10//! nothing panics.
11//!
12//! A reparse point is reported, never followed here. Decisions about a
13//! file use the final path of the opened handle (threat model T-agent-13);
14//! [`ReparsePoint`] says why the name and the file differ. An app execution
15//! alias is parsed into its package family name, app user model ID and
16//! target executable, so installed-package identity (#53) can use it.
17
18#![forbid(unsafe_code)]
19

22/// The longest reparse data buffer the OS allows
23/// (`MAXIMUM_REPARSE_DATA_BUFFER_SIZE`, 16 KiB).
24pub const MAX_REPARSE_BYTES: usize = 16 * 1024;
25
26/// `IO_REPARSE_TAG_MOUNT_POINT`: a junction or a volume mount point.
27pub const TAG_MOUNT_POINT: u32 = 0xA000_0003;
28/// `IO_REPARSE_TAG_SYMLINK`.
29pub const TAG_SYMLINK: u32 = 0xA000_000C;
30/// `IO_REPARSE_TAG_DEDUP`: a file whose data is in the deduplication store.
31pub const TAG_DEDUP: u32 = 0x8000_0013;
32/// `IO_REPARSE_TAG_APPEXECLINK`: an app execution alias.
33pub const TAG_APPEXECLINK: u32 = 0x8000_001B;
34/// `IO_REPARSE_TAG_CLOUD`. The sixteen cloud tags `IO_REPARSE_TAG_CLOUD_1`
35/// to `_F` differ from it only in bits 12 to 15.
36pub const TAG_CLOUD: u32 = 0x9000_001A;
37/// The bits that distinguish the sixteen cloud tags.
38const CLOUD_MASK: u32 = 0x0000_F000;
39/// The name-surrogate bit: the reparse point stands for another named
40/// entity (a link, junction or mount point).
41const NAME_SURROGATE: u32 = 0x2000_0000;
42/// `SYMLINK_FLAG_RELATIVE`.
43const SYMLINK_FLAG_RELATIVE: u32 = 1;
44/// The longest string in an app execution alias, in UTF-16 code units.
45const MAX_ALIAS_UNITS: usize = 32_767;
46
47/// What kind of reparse point a tag names.
48#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
49pub enum ReparseKind {
50    /// A symbolic link to a file or folder.
51    Symlink,
52    /// A junction: a folder that redirects to another folder.
53    Junction,
54    /// A volume mount point: a folder where another volume is mounted.
55    MountPoint,
56    /// An app execution alias, which starts a packaged app's executable.
57    AppExecLink,
58    /// A cloud-files placeholder (any of the cloud tags).
59    CloudPlaceholder,
60    /// A deduplicated file.
61    Dedup,
62    /// Any other tag, as reported.
63    Other(u32),
64}
65
66impl ReparseKind {
67    /// Classifies a tag. A mount-point tag is [`Self::Junction`] until its
68    /// target shows it is a volume ([`parse_reparse_point`] decides).
69    #[must_use]
70    pub const fn from_tag(tag: u32) -> Self {
71        match tag {
72            TAG_SYMLINK => Self::Symlink,
73            TAG_MOUNT_POINT => Self::Junction,
74            TAG_APPEXECLINK => Self::AppExecLink,
75            TAG_DEDUP => Self::Dedup,
76            t if t & !CLOUD_MASK == TAG_CLOUD => Self::CloudPlaceholder,
77            t => Self::Other(t),
78        }
79    }
80}

82/// The target of a symbolic link, junction or mount point.
83#[derive(Debug, Clone, PartialEq, Eq, Hash)]
84pub struct LinkTarget {
85    /// The substitute name: the path the OS follows, in NT form
86    /// (`\??\C:\target`), or relative to the link's folder for a relative
87    /// symbolic link.
88    pub substitute_name: String,
89    /// The print name: the target as shown to users. Can be anything the
90    /// link's creator chose; never decide on it.
91    pub print_name: String,
92    /// A symbolic link whose target is relative to its own folder.
93    pub relative: bool,
94}

96/// The fields of an app execution alias (`IO_REPARSE_TAG_APPEXECLINK`).
97///
98/// **Every field is a claim by whoever created the alias.** A standard
99/// user can create an alias in a folder they can write to and put any
100/// package family name, app ID and target path in it. None of these may
101/// decide anything on its own: a caller must first check the alias against
102/// the installed package identity (#53), that a package with this family
103/// name is installed for the user and that the target is that package's
104/// executable. Until then they say only what the alias claims to start.
105#[derive(Debug, Clone, PartialEq, Eq, Hash)]
106pub struct AppExecLink {
107    /// The structure version the alias reports (3 on current systems).
108    pub version: u32,
109    /// The package family name the alias claims to start. Unverified.
110    pub package_family_name: String,
111    /// The app user model ID (`PackageFamilyName!AppId`) it claims.
112    /// Unverified.
113    pub app_user_model_id: String,
114    /// The full path of the executable it claims to start. Unverified.
115    pub target_path: String,
116    /// Any further strings (version 3 adds the app type), as given.
117    pub extra: Vec<String>,
118}

121#[derive(Debug, Clone, PartialEq, Eq, Hash)]
122pub enum ReparseTarget {

124    Link(LinkTarget),

129/// A reparse point: `reparse { tag, target }` in #54.
130#[derive(Debug, Clone, PartialEq, Eq, Hash)]
131pub struct ReparsePoint {
132    /// The raw tag.
133    pub tag: u32,
134    /// What the tag means.
135    pub kind: ReparseKind,
136    /// The target, for the kinds that name one.
137    pub target: Option<ReparseTarget>,
138}
139
140impl ReparsePoint {
141    /// Whether the reparse point stands for another named entity (the
142    /// name-surrogate bit): opening the name reaches some other object.
143    #[must_use]
144    pub const fn is_name_surrogate(&self) -> bool {
145        self.tag & NAME_SURROGATE != 0
146    }
147
148    /// The app execution alias, if this is one.
149    #[must_use]
150    pub fn app_exec_link(&self) -> Option<&AppExecLink> {
151        match &self.target {
152            Some(ReparseTarget::AppExecLink(a)) => Some(a),
153            _ => None,
154        }
155    }
156
157    /// The link target, if this is a symbolic link, junction or mount point.
158    #[must_use]
159    pub fn link_target(&self) -> Option<&LinkTarget> {
160        match &self.target {
161            Some(ReparseTarget::Link(l)) => Some(l),
162            _ => None,
163        }
164    }
165}

167/// Why a reparse data buffer could not be parsed.
168#[derive(Debug, Clone, Copy, PartialEq, Eq)]
169#[non_exhaustive]
170pub enum ReparseError {
171    /// Shorter than its header or its stated length.
172    Truncated,
173    /// Longer than [`MAX_REPARSE_BYTES`].
174    TooLong,
175    /// A name offset or length is odd or points outside the buffer.
176    BadName,
177    /// A string is not valid UTF-16, or holds a NUL where none may be.
178    BadString,
179}
180
181impl fmt::Display for ReparseError {
182    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
183        f.write_str(match self {
184            Self::Truncated => "reparse data is truncated",
185            Self::TooLong => "reparse data is too long",
186            Self::BadName => "reparse data has a bad name offset or length",
187            Self::BadString => "reparse data holds a malformed string",
188        })
189    }
190}

192impl std::error::Error for ReparseError {}

194fn u16_at(b: &[u8], at: usize) -> Option<u16> {
195    let bytes = b.get(at..at.checked_add(2)?)?;
196    Some(u16::from_le_bytes(bytes.try_into().ok()?))
197}
198
199fn u32_at(b: &[u8], at: usize) -> Option<u32> {
200    let bytes = b.get(at..at.checked_add(4)?)?;
201    Some(u32::from_le_bytes(bytes.try_into().ok()?))
202}

204fn utf16(bytes: &[u8]) -> Result<String, ReparseError> {
205    if bytes.len() % 2 != 0 {
206        return Err(ReparseError::BadName);
207    }
208    let units: Vec<u16> = bytes
209        .chunks_exact(2)
210        .map(|c| {
211            u16::from_le_bytes([
212                c.first().copied().unwrap_or(0),
213                c.get(1).copied().unwrap_or(0),
214            ])
215        })
216        .collect();
217    let s = String::from_utf16(&units).map_err(|_| ReparseError::BadString)?;
218    if s.contains('\0') {
219        return Err(ReparseError::BadString);
220    }
221    Ok(s)
222}

224/// A name in the path buffer that starts at `base`, at `offset` for
225/// `len` bytes.
226fn name_at(data: &[u8], base: usize, offset: u16, len: u16) -> Result<String, ReparseError> {
227    let start = base
228        .checked_add(usize::from(offset))
229        .ok_or(ReparseError::BadName)?;
230    let end = start
231        .checked_add(usize::from(len))
232        .ok_or(ReparseError::BadName)?;
233    utf16(data.get(start..end).ok_or(ReparseError::BadName)?)
234}

236/// Whether an NT substitute name names a volume root rather than a folder:
237/// `\??\Volume{GUID}\`, as a volume mount point's target is.
238fn is_volume_target(substitute: &str) -> bool {
239    let rest = substitute
240        .strip_prefix(r"\??\")
241        .unwrap_or(substitute)
242        .trim_end_matches('\\');
243    rest.len() > 7
244        && rest
245            .get(..7)
246            .is_some_and(|p| p.eq_ignore_ascii_case("Volume{"))
247        && rest.ends_with('}')
248        && !rest.contains('\\')
249}

251/// The string list of an app execution alias: UTF-16 strings, each ended
252/// by a NUL. Bytes after the last NUL are ignored.
253fn alias_strings(data: &[u8]) -> Result<Vec<String>, ReparseError> {
254    let mut out = Vec::new();
255    let mut current = Vec::new();
256    for c in data.chunks_exact(2) {
257        let u = u16::from_le_bytes([
258            c.first().copied().unwrap_or(0),
259            c.get(1).copied().unwrap_or(0),
260        ]);
261        if u == 0 {
262            out.push(String::from_utf16(&current).map_err(|_| ReparseError::BadString)?);
263            current.clear();
264        } else {
265            if current.len() >= MAX_ALIAS_UNITS {
266                return Err(ReparseError::BadString);
267            }
268            current.push(u);
269        }
270    }
271    Ok(out)
272}

274/// Parses a reparse data buffer (`REPARSE_DATA_BUFFER`, or the GUID form
275/// for third-party tags) as `FSCTL_GET_REPARSE_POINT` returns it.
276///
277/// Symbolic links, junctions, mount points and app execution aliases are
278/// parsed for their targets. Other tags are reported by kind only, with
279/// their data ignored. An app execution alias with fewer than three
280/// strings has no target.
281pub fn parse_reparse_point(bytes: &[u8]) -> Result<ReparsePoint, ReparseError> {
282    if bytes.len() > MAX_REPARSE_BYTES {
283        return Err(ReparseError::TooLong);
284    }
285    let tag = u32_at(bytes, 0).ok_or(ReparseError::Truncated)?;
286    let len = usize::from(u16_at(bytes, 4).ok_or(ReparseError::Truncated)?);
287    let data = bytes
288        .get(8..8_usize.saturating_add(len))
289        .ok_or(ReparseError::Truncated)?;
290    let mut kind = ReparseKind::from_tag(tag);
291    let target = match tag {
292        TAG_SYMLINK | TAG_MOUNT_POINT => {
293            let field = |at: usize| u16_at(data, at).ok_or(ReparseError::Truncated);
294            let (sub_off, sub_len, print_off, print_len) =
295                (field(0)?, field(2)?, field(4)?, field(6)?);
296            let (base, relative) = if tag == TAG_SYMLINK {
297                let flags = u32_at(data, 8).ok_or(ReparseError::Truncated)?;
298                (12, flags & SYMLINK_FLAG_RELATIVE != 0)
299            } else {
300                (8, false)
301            };
302            let substitute_name = name_at(data, base, sub_off, sub_len)?;
303            let print_name = name_at(data, base, print_off, print_len)?;
304            if tag == TAG_MOUNT_POINT && is_volume_target(&substitute_name) {
305                kind = ReparseKind::MountPoint;
306            }
307            Some(ReparseTarget::Link(LinkTarget {
308                substitute_name,
309                print_name,
310                relative,
311            }))
312        }
313        TAG_APPEXECLINK => {
314            let version = u32_at(data, 0).ok_or(ReparseError::Truncated)?;
315            let strings = alias_strings(data.get(4..).unwrap_or_default())?;
316            let mut it = strings.into_iter();
317            match (it.next(), it.next(), it.next()) {
318                (Some(package_family_name), Some(app_user_model_id), Some(target_path)) => {
319                    Some(ReparseTarget::AppExecLink(AppExecLink {
320                        version,
321                        package_family_name,
322                        app_user_model_id,
323                        target_path,
324                        extra: it.collect(),
325                    }))
326                }
327                _ => None,
328            }
329        }
330        _ => None,
331    };
332    Ok(ReparsePoint { tag, kind, target })
333}

339    fn units(s: &str) -> Vec<u8> {
340        s.encode_utf16().flat_map(u16::to_le_bytes).collect()
341    }

343    /// A symbolic-link or mount-point buffer with the two names.
344    pub(super) fn link_buffer(tag: u32, sub: &str, print: &str, flags: Option<u32>) -> Vec<u8> {
345        let s = units(sub);
346        let p = units(print);
347        let mut data = Vec::new();
348        data.extend_from_slice(&0_u16.to_le_bytes());
349        data.extend_from_slice(&u16::try_from(s.len()).unwrap().to_le_bytes());
350        data.extend_from_slice(&u16::try_from(s.len() + 2).unwrap().to_le_bytes());
351        data.extend_from_slice(&u16::try_from(p.len()).unwrap().to_le_bytes());
352        if let Some(f) = flags {
353            data.extend_from_slice(&f.to_le_bytes());
354        }
355        data.extend_from_slice(&s);
356        data.extend_from_slice(&[0, 0]);
357        data.extend_from_slice(&p);
358        data.extend_from_slice(&[0, 0]);
359        let mut out = Vec::new();
360        out.extend_from_slice(&tag.to_le_bytes());
361        out.extend_from_slice(&u16::try_from(data.len()).unwrap().to_le_bytes());
362        out.extend_from_slice(&0_u16.to_le_bytes());
363        out.extend(data);
364        out
365    }

367    fn alias_buffer(strings: &[&str]) -> Vec<u8> {
368        let mut data = 3_u32.to_le_bytes().to_vec();
369        for s in strings {
370            data.extend(units(s));
371            data.extend_from_slice(&[0, 0]);
372        }
373        let mut out = TAG_APPEXECLINK.to_le_bytes().to_vec();
374        out.extend_from_slice(&u16::try_from(data.len()).unwrap().to_le_bytes());
375        out.extend_from_slice(&0_u16.to_le_bytes());
376        out.extend(data);
377        out
378    }

473    #[test]
474    fn malformed_buffers_are_errors() {
475        assert_eq!(parse_reparse_point(&[]), Err(ReparseError::Truncated));
476        assert_eq!(parse_reparse_point(&[0; 7]), Err(ReparseError::Truncated));
477        assert_eq!(
478            parse_reparse_point(&vec![0; MAX_REPARSE_BYTES + 1]),
479            Err(ReparseError::TooLong)
480        );
481        // Stated length past the end.
482        let mut b = link_buffer(TAG_SYMLINK, "a", "b", Some(0));
483        b[4] = 0xFF;
484        assert_eq!(parse_reparse_point(&b), Err(ReparseError::Truncated));
485        // Name offset past the buffer.
486        let mut b = link_buffer(TAG_SYMLINK, "a", "b", Some(0));
487        b[8] = 0xF0;
488        assert_eq!(parse_reparse_point(&b), Err(ReparseError::BadName));
489        // Odd name length.
490        let mut b = link_buffer(TAG_SYMLINK, "ab", "b", Some(0));
491        b[10] = 3;
492        assert_eq!(parse_reparse_point(&b), Err(ReparseError::BadName));
493        // A NUL inside a name.
494        let b = link_buffer(TAG_MOUNT_POINT, "a\0b", "", None);
495        assert_eq!(parse_reparse_point(&b), Err(ReparseError::BadString));
496        // An unpaired surrogate in an alias.
497        let mut b = alias_buffer(&["a", "b", "c"]);
498        let at = b.len() - 4;
499        b[at..at + 2].copy_from_slice(&0xD800_u16.to_le_bytes());
500        assert_eq!(parse_reparse_point(&b), Err(ReparseError::BadString));
501        // Every prefix of every valid buffer kind parses or errors, never
502        // panics, and a prefix shorter than the stated length is an error.
503        for full in [
504            link_buffer(TAG_SYMLINK, r"\??\C:\x", r"C:\x", Some(0)),
505            link_buffer(TAG_SYMLINK, r"..\x", r"..\x", Some(1)),
506            link_buffer(TAG_MOUNT_POINT, r"\??\C:\Program Files\App", "", None),
507            link_buffer(
508                TAG_MOUNT_POINT,
509                r"\??\Volume{01234567-89ab-cdef-0123-456789abcdef}\",
510                "",
511                None,
512            ),
513            alias_buffer(&[
514                "Example.App_8wekyb3d8bbwe",
515                "Example.App_8wekyb3d8bbwe!App",
516                r"C:\Program Files\WindowsApps\Example\app.exe",
517                "0",
518            ]),
519        ] {
520            assert!(parse_reparse_point(&full).is_ok());
521            for n in 0..full.len() {
522                assert!(
523                    parse_reparse_point(&full[..n]).is_err(),
524                    "{n} of {}",
525                    full.len()
526                );
527            }
528        }
529    }
