1//! Parsers for the variable-length records that the Windows file
2//! information calls return (#46, #54): the alternate data stream list
3//! (`FILE_STREAM_INFO`) and the file's change-journal record (`USN_RECORD_V2`
4//! or `USN_RECORD_V3`).
5//!
6//! The Windows backend (`palisade_fileid::win`) fills a byte buffer from the
7//! OS and hands it here, so the parsing is pure, bounded and tested on every
8//! OS. The bytes come from the file system, not from file content, but a
9//! filter driver or a damaged volume can still return anything, so every
10//! offset and length is checked and nothing panics.
11

16/// The most streams listed for one file. A file can have far more; the rest
17/// are reported as [`StreamList::truncated`].
18pub const MAX_STREAMS: usize = 4096;
19
20/// The longest stream name, in UTF-16 code units, as NTFS allows.
21pub const MAX_STREAM_NAME_UNITS: usize = 255;
22
23/// The longest stream type accepted, in UTF-16 code units. The longest the
24/// OS defines is `$LOGGED_UTILITY_STREAM` (22).
25pub const MAX_STREAM_TYPE_UNITS: usize = 64;
26
27/// The longest `:name:type` field accepted, in UTF-16 code units.
28const MAX_FIELD_UNITS: usize = 2 + MAX_STREAM_NAME_UNITS + MAX_STREAM_TYPE_UNITS;
29
30/// One data stream of a file.
31#[derive(Debug, Clone, PartialEq, Eq, Hash)]
32pub struct StreamEntry {
33    /// The stream name without the leading `:` or the `:$DATA` type, so the
34    /// default stream is `""` and `file.txt:Zone.Identifier` is
35    /// `Zone.Identifier`.
36    pub name: String,
37    /// The stream type as the OS reports it (`$DATA` for every stream that
38    /// holds data).
39    pub kind: String,
40    /// The stream size in bytes.
41    pub size: u64,
42    /// The bytes allocated on disk for the stream.
43    pub allocation_size: u64,
44}
45
46impl StreamEntry {
47    /// Whether this is a data stream (`$DATA`, compared without regard to
48    /// ASCII case, as the OS compares stream types).
49    #[must_use]
50    pub fn is_data(&self) -> bool {
51        self.kind.eq_ignore_ascii_case("$DATA")
52    }
53
54    /// Whether this is the file's default data stream: unnamed and `$DATA`.
55    #[must_use]
56    pub fn is_default(&self) -> bool {
57        self.name.is_empty() && self.is_data()
58    }
59}

61/// The streams of one file, in the order the OS lists them.
62#[derive(Debug, Clone, PartialEq, Eq, Default)]
63pub struct StreamList {
64    /// The streams read.
65    pub streams: Vec<StreamEntry>,
66    /// More than [`MAX_STREAMS`] were listed; the rest were not read.
67    pub truncated: bool,
68}
69
70impl StreamList {
71    /// The named data streams (alternate data streams): every named
72    /// `$DATA` stream. Unnamed and non-data streams are left out.
73    pub fn alternate(&self) -> impl Iterator<Item = &StreamEntry> {
74        self.streams
75            .iter()
76            .filter(|s| !s.name.is_empty() && s.is_data())
77    }
78}

80/// Why a record could not be parsed.
81#[derive(Debug, Clone, Copy, PartialEq, Eq)]
82#[non_exhaustive]
83pub enum RecordError {
84    /// A record or one of its fields runs past the buffer.
85    Truncated {
86        /// The byte offset of the record.
87        offset: usize,
88    },
89    /// A stream name is not of the form `:name:type`, its name is longer
90    /// than NTFS allows or its type longer than [`MAX_STREAM_TYPE_UNITS`],
91    /// it holds a path separator or NUL, or it is not valid UTF-16.
92    BadStreamName {
93        /// The byte offset of the record.
94        offset: usize,
95    },
96    /// A `NextEntryOffset` points inside its own record (its header or its
97    /// name), which would overlap or loop.
98    BadNextOffset {
99        /// The byte offset of the record.
100        offset: usize,
101    },
102    /// The record is not a version-2 or version-3 change-journal record.
103    UnsupportedVersion(u16),
104}
105
106impl fmt::Display for RecordError {
107    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
108        match self {
109            Self::Truncated { offset } => write!(f, "record at offset {offset} is truncated"),
110            Self::BadStreamName { offset } => {
111                write!(f, "record at offset {offset} has a malformed stream name")
112            }
113            Self::BadNextOffset { offset } => {
114                write!(f, "record at offset {offset} has a bad next-entry offset")
115            }
116            Self::UnsupportedVersion(v) => {
117                write!(f, "unsupported change-journal record version {v}")
118            }
119        }
120    }
121}

123impl std::error::Error for RecordError {}

125fn u16_at(b: &[u8], at: usize) -> Option<u16> {
126    let bytes = b.get(at..at.checked_add(2)?)?;
127    Some(u16::from_le_bytes(bytes.try_into().ok()?))
128}
129
130fn u32_at(b: &[u8], at: usize) -> Option<u32> {
131    let bytes = b.get(at..at.checked_add(4)?)?;
132    Some(u32::from_le_bytes(bytes.try_into().ok()?))
133}
134
135fn i64_at(b: &[u8], at: usize) -> Option<i64> {
136    let bytes = b.get(at..at.checked_add(8)?)?;
137    Some(i64::from_le_bytes(bytes.try_into().ok()?))
138}

140/// The fixed part of `FILE_STREAM_INFO`: `NextEntryOffset`,
141/// `StreamNameLength`, `StreamSize` and `StreamAllocationSize`.
142const STREAM_HEADER: usize = 24;
143
144/// Parses the `FILE_STREAM_INFO` list that `GetFileInformationByHandleEx`
145/// returns for `FileStreamInfo`. An empty buffer is a file with no streams
146/// (a directory without named streams).
147pub fn parse_stream_info(bytes: &[u8]) -> Result<StreamList, RecordError> {
148    let mut list = StreamList::default();
149    if bytes.is_empty() {
150        return Ok(list);
151    }
152    let mut offset = 0_usize;
153    loop {
154        if list.streams.len() >= MAX_STREAMS {
155            list.truncated = true;
156            break;
157        }
158        let rec = bytes
159            .get(offset..)
160            .ok_or(RecordError::Truncated { offset })?;
161        let (Some(next), Some(name_len), Some(size), Some(alloc)) = (
162            u32_at(rec, 0),
163            u32_at(rec, 4),
164            i64_at(rec, 8),
165            i64_at(rec, 16),
166        ) else {
167            return Err(RecordError::Truncated { offset });
168        };
169        let name_len = usize::try_from(name_len).map_err(|_| RecordError::Truncated { offset })?;
170        if name_len % 2 != 0 || name_len / 2 > MAX_FIELD_UNITS {
171            return Err(RecordError::BadStreamName { offset });
172        }
173        let raw = STREAM_HEADER
174            .checked_add(name_len)
175            .and_then(|end| rec.get(STREAM_HEADER..end))
176            .ok_or(RecordError::Truncated { offset })?;
177        let units: Vec<u16> = raw
178            .as_chunks::<2>()
179            .0
180            .iter()
181            .map(|c| u16::from_le_bytes(*c))
182            .collect();
183        let full = String::from_utf16(&units).map_err(|_| RecordError::BadStreamName { offset })?;
184        // `:name:type`, where the default stream is `::$DATA`.
185        // A name or type holding a separator or NUL could change the path a
186        // caller builds from it, so it is malformed.
187        let plain = |t: &str| !t.contains(['\\', '/', '\0', ':']);
188        let (name, kind) = full
189            .strip_prefix(':')
190            .and_then(|rest| rest.rsplit_once(':'))
191            .filter(|(name, kind)| {
192                !kind.is_empty()
193                    && plain(name)
194                    && plain(kind)
195                    && name.encode_utf16().count() <= MAX_STREAM_NAME_UNITS
196                    && kind.encode_utf16().count() <= MAX_STREAM_TYPE_UNITS
197            })
198            .ok_or(RecordError::BadStreamName { offset })?;
199        list.streams.push(StreamEntry {
200            name: name.to_owned(),
201            kind: kind.to_owned(),
202            size: u64::try_from(size).unwrap_or(0),
203            allocation_size: u64::try_from(alloc).unwrap_or(0),
204        });
205        if next == 0 {
206            break;
207        }
208        let next = usize::try_from(next).map_err(|_| RecordError::BadNextOffset { offset })?;
209        if next < STREAM_HEADER.saturating_add(name_len) {
210            return Err(RecordError::BadNextOffset { offset });
211        }
212        offset = offset
213            .checked_add(next)
214            .ok_or(RecordError::BadNextOffset { offset })?;
215    }
216    Ok(list)
217}

219/// The change-journal facts in a file's USN record.
220#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
221pub struct UsnFacts {
222    /// The record version (2 or 3).
223    pub major_version: u16,
224    /// The file's last update sequence number.
225    pub usn: i64,
226    /// The 128-bit file reference number as the record gives it. Version 2
227    /// records carry 64 bits, zero-extended here.
228    pub file_reference: [u8; 16],
229}
230
231/// Parses the `USN_RECORD_V2` or `USN_RECORD_V3` that
232/// `FSCTL_READ_FILE_USN_DATA` returns.
233pub fn parse_usn_record(bytes: &[u8]) -> Result<UsnFacts, RecordError> {
234    let truncated = RecordError::Truncated { offset: 0 };
235    let length = u32_at(bytes, 0).ok_or(truncated)?;
236    let major = u16_at(bytes, 4).ok_or(truncated)?;
237    let length = usize::try_from(length).map_err(|_| truncated)?;
238    let record = bytes.get(..length).ok_or(truncated)?;
239    let mut file_reference = [0_u8; 16];
240    let usn = match major {
241        2 => {
242            // RecordLength, versions, FileReferenceNumber (8),
243            // ParentFileReferenceNumber (8), then Usn.
244            let id = record.get(8..16).ok_or(truncated)?;
245            if let Some(dst) = file_reference.get_mut(..8) {
246                dst.copy_from_slice(id);
247            }
248            i64_at(record, 24).ok_or(truncated)?
249        }
250        3 => {
251            // As version 2, with 16-byte reference numbers.
252            let id = record.get(8..24).ok_or(truncated)?;
253            file_reference.copy_from_slice(id);
254            i64_at(record, 40).ok_or(truncated)?
255        }
256        other => return Err(RecordError::UnsupportedVersion(other)),
257    };
258    Ok(UsnFacts {
259        major_version: major,
260        usn,
261        file_reference,
262    })
263}

269    fn stream_record(next: u32, name: &str, size: i64) -> Vec<u8> {
270        let units: Vec<u16> = name.encode_utf16().collect();
271        let mut out = Vec::new();
272        out.extend_from_slice(&next.to_le_bytes());
273        out.extend_from_slice(&u32::try_from(units.len() * 2).unwrap().to_le_bytes());
274        out.extend_from_slice(&size.to_le_bytes());
275        out.extend_from_slice(&(size + 7).to_le_bytes());
276        for u in units {
277            out.extend_from_slice(&u.to_le_bytes());
278        }
279        out
280    }

282    /// Two records with the first's `NextEntryOffset` set, padded to 8 bytes
283    /// as the OS does.
284    fn two(a: &str, b: &str) -> Vec<u8> {
285        let mut first = stream_record(0, a, 10);
286        while !first.len().is_multiple_of(8) {
287            first.push(0);
288        }
289        let next = u32::try_from(first.len()).unwrap();
290        first[..4].copy_from_slice(&next.to_le_bytes());
291        first.extend(stream_record(0, b, 3));
292        first
293    }

296    fn streams() {
297        assert_eq!(parse_stream_info(&[]).unwrap(), StreamList::default());
298        let list = parse_stream_info(&two("::$DATA", ":Zone.Identifier:$DATA")).unwrap();
299        assert_eq!(list.streams.len(), 2);
300        assert!(list.streams[0].is_default());
301        assert_eq!(list.streams[0].size, 10);
302        assert_eq!(list.streams[0].allocation_size, 17);
303        assert_eq!(list.streams[1].name, "Zone.Identifier");
304        assert_eq!(list.streams[1].kind, "$DATA");
305        assert_eq!(
306            list.alternate()
307                .map(|s| s.name.as_str())
308                .collect::<Vec<_>>(),
309            ["Zone.Identifier"]
310        );
311        assert!(!list.truncated);
312        // Unnamed or non-data streams are neither default nor alternate.
313        let mut odd = stream_record(0, "::$INDEX_ALLOCATION", 0);
314        while !odd.len().is_multiple_of(8) {
315            odd.push(0);
316        }
317        let next = u32::try_from(odd.len()).unwrap();

339    #[test]
340    fn malformed_streams_are_errors() {
341        // Truncated header and name.
342        assert!(matches!(
343            parse_stream_info(&[0; 10]),
344            Err(RecordError::Truncated { .. })
345        ));
346        let mut rec = stream_record(0, "::$DATA", 1);
347        rec.truncate(rec.len() - 2);
348        assert!(matches!(
349            parse_stream_info(&rec),
350            Err(RecordError::Truncated { .. })
351        ));
352        // Names that are not `:name:type`.
353        for bad in ["Zone", ":Zone", ":a:b:$DATA", ":Zone:"] {
354            assert!(
355                matches!(
356                    parse_stream_info(&stream_record(0, bad, 1)),
357                    Err(RecordError::BadStreamName { .. })
358                ),
359                "{bad}"
360            );
361        }
362        // Separators and NUL in the name or type.
363        for bad in [":a\\..\\b:$DATA", ":a/b:$DATA", ":a\0:$DATA", ":a:$DA/TA"] {
364            assert!(
365                matches!(
366                    parse_stream_info(&stream_record(0, bad, 1)),
367                    Err(RecordError::BadStreamName { .. })
368                ),
369                "{bad:?}"
370            );
371        }
372        // A 255-unit name with the longest OS type is fine; 256 is not.
373        let max = format!(":{}:$LOGGED_UTILITY_STREAM", "n".repeat(255));
374        assert_eq!(
375            parse_stream_info(&stream_record(0, &max, 1))
376                .unwrap()
377                .streams[0]
378                .kind,
379            "$LOGGED_UTILITY_STREAM"
380        );
381        let over = format!(":{}:x", "n".repeat(256));
382        assert!(matches!(
383            parse_stream_info(&stream_record(0, &over, 1)),
384            Err(RecordError::BadStreamName { .. })
385        ));
386        // An odd name length, and one longer than NTFS allows.
387        let mut odd = stream_record(0, "::$DATA", 1);
388        odd[4] = 3;
389        assert!(matches!(
390            parse_stream_info(&odd),
391            Err(RecordError::BadStreamName { .. })
392        ));
393        let long = format!(":{}:$DATA", "a".repeat(300));
394        assert!(matches!(
395            parse_stream_info(&stream_record(0, &long, 1)),
396            Err(RecordError::BadStreamName { .. })
397        ));
398        // Unpaired surrogate.
399        let mut surrogate = stream_record(0, "::$DATA", 1);
400        surrogate[24] = 0x00;
401        surrogate[25] = 0xd8;
402        assert!(matches!(
403            parse_stream_info(&surrogate),
404            Err(RecordError::BadStreamName { .. })
405        ));
406        // A next offset inside the header, which would loop, or inside the
407        // record's own name, which would overlap.
408        for next in [8, 24, 24 + 12] {
409            let rec = stream_record(next, "::$DATA", 1);
410            assert!(
411                matches!(
412                    parse_stream_info(&rec),
413                    Err(RecordError::BadNextOffset { .. })
414                ),
415                "{next}"
416            );
417        }
418        // A next offset past the end.
419        let rec = stream_record(4096, "::$DATA", 1);
420        assert!(matches!(
421            parse_stream_info(&rec),
422            Err(RecordError::Truncated { .. })
423        ));
424        // A negative size reads as zero, never wraps.
425        let list = parse_stream_info(&stream_record(0, "::$DATA", -5)).unwrap();
426        assert_eq!(list.streams[0].size, 0);
427    }

430    fn stream_count_is_bounded() {
431        let mut one = stream_record(0, ":s:$DATA", 1);
432        while !one.len().is_multiple_of(8) {

455    #[test]
456    fn usn_records() {
457        let f = parse_usn_record(&usn_v2(12345)).unwrap();
458        assert_eq!((f.major_version, f.usn), (2, 12345));
459        assert_eq!(
460            f.file_reference[..8],
461            0x1122_3344_5566_7788_u64.to_le_bytes()
462        );
463        assert_eq!(f.file_reference[8..], [0; 8]);
464
465        let mut v3 = vec![0_u8; 80];
466        v3[..4].copy_from_slice(&80_u32.to_le_bytes());
467        v3[4..6].copy_from_slice(&3_u16.to_le_bytes());
468        v3[8..24].copy_from_slice(&[9; 16]);
469        v3[40..48].copy_from_slice(&77_i64.to_le_bytes());
470        let f = parse_usn_record(&v3).unwrap();
471        assert_eq!((f.major_version, f.usn, f.file_reference), (3, 77, [9; 16]));
472
473        // Version 4 has no USN in this place; other versions are refused.
474        let mut v4 = usn_v2(1);
475        v4[4..6].copy_from_slice(&4_u16.to_le_bytes());
476        assert_eq!(
477            parse_usn_record(&v4),
478            Err(RecordError::UnsupportedVersion(4))
479        );
480        // RecordLength past the buffer, or too short for the USN.
481        let mut long = usn_v2(1);
482        long[..4].copy_from_slice(&1000_u32.to_le_bytes());
483        assert!(parse_usn_record(&long).is_err());
484        let mut short = usn_v2(1);
485        short[..4].copy_from_slice(&20_u32.to_le_bytes());
486        assert!(parse_usn_record(&short).is_err());
487        assert!(parse_usn_record(&[]).is_err());
488    }
