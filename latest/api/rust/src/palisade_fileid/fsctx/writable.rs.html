1//! Cached and streaming writability assessment (#54).
2//!
3//! [`super::acl::assess_path`] needs every ancestor's descriptor up front.
4//! This module adds what a scanner needs on top:
5//!
6//! - [`WritabilityCache`]: the result of [`super::acl::evaluate`] for each
7//!   distinct security descriptor, keyed by the SHA-256 of its
8//!   self-relative bytes and the object kind. Folders that inherit the same
9//!   ACL (most of a volume) share one entry, so each distinct descriptor is
10//!   evaluated once. A security-change journal record (#60) evicts by
11//!   descriptor hash ([`WritabilityCache::evict`]) or clears the cache.
12//! - [`assess_ancestors`]: [`super::acl::assess_path`] over an iterator of
13//!   ancestors, with the cache, so the chain is not built first, and with
14//!   [`AssessOptions::stop_at_first_writable`] it stops at the first
15//!   user-writable folder.
16//! - [`enumerate_user_writable_dirs`]: the folders under some roots that a
17//!   principal set may write to, for the E8 self-test and path scanner
18//!   (#17), and the path-rule guardrails (#212). It walks through a
19//!   [`DirectorySource`], so it is portable and tested here; the Windows
20//!   backend provides the source that reads real folders.
21//!
22//! Everything here is pure: no file system access happens in this module.
23

35/// The default bound on cached descriptors.
36pub const DEFAULT_CACHE_ENTRIES: usize = 4096;
37
38/// The SHA-256 of a descriptor's self-relative bytes: the cache key.
39pub type DescriptorHash = [u8; 32];
40
41/// The cache key of `bytes`, a self-relative security descriptor as the OS
42/// returned it.
43#[must_use]
44pub fn descriptor_hash(bytes: &[u8]) -> DescriptorHash {
45    Sha256::digest(bytes).into()
46}

48/// A security descriptor as read from the OS, with its hash.
49#[derive(Debug, Clone, PartialEq, Eq)]
50pub struct ReadDescriptor {
51    /// The parsed descriptor.
52    pub descriptor: SecurityDescriptor,
53    /// [`descriptor_hash`] of the bytes it was parsed from.
54    pub hash: DescriptorHash,
55    /// The self-relative bytes it was parsed from, for the OS access check
56    /// ([`OsAccessCheck`]).
57    pub bytes: Vec<u8>,
58}
59
60impl ReadDescriptor {
61    /// Parses `bytes`. `None` if they do not parse, which a caller must
62    /// treat as unreadable (possibly writable).
63    #[must_use]
64    pub fn parse(bytes: &[u8]) -> Option<Self> {
65        let descriptor = SecurityDescriptor::from_self_relative(bytes).ok()?;
66        Some(Self {
67            descriptor,
68            hash: descriptor_hash(bytes),
69            bytes: bytes.to_vec(),
70        })
71    }
72}

74/// The OS access check for the synthetic standard user: the write-type
75/// rights it grants on the self-relative descriptor `bytes` for an object
76/// of the given kind, or `None` if it could not run. On Windows this is
77/// `palisade_fileid::win::access::os_standard_user_rights`; tests pass a
78/// stub.
79pub type OsAccessCheck = fn(&[u8], ObjectKind, &EvaluationOptions) -> Option<WriteRights>;
80
81/// Evaluation results by descriptor hash and object kind.
82///
83/// The results depend on the [`EvaluationOptions`] and on the OS access
84/// check too, so a cache belongs to one set of options and one check:
85/// [`Self::evaluate`] clears it when it is called with different options,
86/// and [`Self::set_os_check`] when the check changes.
87///
88/// With an OS check set, each result is the portable evaluation with the
89/// OS verdict merged in ([`ObjectWritability::merge_os_check`]): the union
90/// of the two, so the OS check can add rights but never remove them.
91#[derive(Debug, Clone)]
92pub struct WritabilityCache {
93    entries: HashMap<(DescriptorHash, ObjectKind), ObjectWritability>,
94    order: VecDeque<(DescriptorHash, ObjectKind)>,
95    max_entries: usize,
96    options: Option<EvaluationOptions>,
97    os_check: Option<OsAccessCheck>,
98    hits: u64,
99    misses: u64,
100}
101
102impl Default for WritabilityCache {
103    fn default() -> Self {
104        Self::new(DEFAULT_CACHE_ENTRIES)
105    }
106}
107
108impl WritabilityCache {
109    /// A cache holding at most `max_entries` results (at least 1). The
110    /// oldest entry is evicted first.
111    #[must_use]
112    pub fn new(max_entries: usize) -> Self {
113        Self {
114            entries: HashMap::new(),
115            order: VecDeque::new(),
116            max_entries: max_entries.max(1),
117            options: None,
118            os_check: None,
119            hits: 0,
120            misses: 0,
121        }
122    }
123
124    /// Sets the OS access check merged into every result (`None` for the
125    /// portable evaluator alone). Changing it clears the cache.
126    pub fn set_os_check(&mut self, check: Option<OsAccessCheck>) {
127        if self.os_check.map(|f| f as usize) != check.map(|f| f as usize) {
128            self.clear();
129            self.os_check = check;
130        }
131    }
132
133    /// Whether an OS access check is set.
134    #[must_use]
135    pub const fn has_os_check(&self) -> bool {
136        self.os_check.is_some()
137    }
138
139    /// [`evaluate`] of `sd` for `kind`, with the OS check merged in if one
140    /// is set, from the cache when possible.
141    pub fn evaluate(
142        &mut self,
143        sd: &ReadDescriptor,
144        kind: ObjectKind,
145        options: &EvaluationOptions,
146    ) -> ObjectWritability {
147        if self.options.as_ref() != Some(options) {
148            self.clear();
149            self.options = Some(options.clone());
150        }
151        let key = (sd.hash, kind);
152        if let Some(w) = self.entries.get(&key) {
153            self.hits = self.hits.saturating_add(1);
154            return w.clone();
155        }
156        self.misses = self.misses.saturating_add(1);
157        let mut w = evaluate(&sd.descriptor, kind, options);
158        if let Some(check) = self.os_check {
159            w.merge_os_check(check(&sd.bytes, kind, options));
160        }
161        if self.entries.len() >= self.max_entries
162            && let Some(old) = self.order.pop_front()
163        {
164            self.entries.remove(&old);
165        }
166        self.entries.insert(key, w.clone());
167        self.order.push_back(key);
168        w
169    }
170
171    /// Drops every result for the descriptor with this hash, as a
172    /// security-change journal record for a file using it requires (#60).
173    pub fn evict(&mut self, hash: &DescriptorHash) {
174        self.entries.retain(|(h, _), _| h != hash);
175        self.order.retain(|(h, _)| h != hash);
176    }
177
178    /// Drops everything.
179    pub fn clear(&mut self) {
180        self.entries.clear();
181        self.order.clear();
182    }
183
184    /// The number of cached results.
185    #[must_use]
186    pub fn len(&self) -> usize {
187        self.entries.len()
188    }
189
190    /// Whether the cache is empty.
191    #[must_use]
192    pub fn is_empty(&self) -> bool {
193        self.entries.is_empty()
194    }
195
196    /// Lookups answered from the cache, and lookups that evaluated.
197    #[must_use]
198    pub const fn stats(&self) -> (u64, u64) {
199        (self.hits, self.misses)
200    }
201
202    fn result(
203        &mut self,
204        sd: Option<&ReadDescriptor>,
205        kind: ObjectKind,
206        options: &EvaluationOptions,
207    ) -> ObjectWritability {
208        match sd {
209            Some(sd) => self.evaluate(sd, kind, options),
210            None => ObjectWritability::unreadable(kind),
211        }
212    }
213}

215/// Options for [`assess_ancestors`].
216#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
217pub struct AssessOptions {
218    /// Stop reading ancestors once one is user-writable. The result then
219    /// lists only the ancestors read, and [`Writability::principals_with_write`]
220    /// and [`Writability::change_perms_by_non_admin`] cover only those; the
221    /// answer to "is a path rule here safe?" is already no.
222    pub stop_at_first_writable: bool,
223}

227pub struct AncestorRead {

244/// [`super::acl::assess_path`] over ancestors read one at a time (immediate
245/// parent first, the volume root last), evaluated through `cache`. The
246/// result is the same as `assess_path` over the same ancestors, unless
247/// [`AssessOptions::stop_at_first_writable`] stopped early.
248pub fn assess_ancestors<I>(
249    target: Option<&ReadDescriptor>,
250    kind: ObjectKind,
251    ancestors: I,
252    options: &EvaluationOptions,
253    assess: AssessOptions,
254    cache: &mut WritabilityCache,
255) -> Writability
256where
257    I: IntoIterator<Item = AncestorRead>,
258{
259    let target_w = cache.result(target, kind, options);
260    let mut ancestor_w = Vec::new();
261    let mut first_user_writable_ancestor = None;
262    for a in ancestors {
263        let w = cache.result(a.descriptor.as_ref(), ObjectKind::Directory, options);
264        let writable = w.user_writable();
265        ancestor_w.push(w);
266        if writable && first_user_writable_ancestor.is_none() {
267            first_user_writable_ancestor = Some(a.path);
268            if assess.stop_at_first_writable {
269                break;
270            }
271        }
272    }
273    let change_perms_by_non_admin = std::iter::once(&target_w)
274        .chain(&ancestor_w)
275        .any(|w| w.rights.intersects(WriteRights::CHANGE_PERMISSIONS));
276    let principals_with_write: Vec<Principal> = std::iter::once(&target_w)
277        .chain(&ancestor_w)
278        .flat_map(|w| w.principals.iter().map(|p| p.principal))
279        .collect::<BTreeSet<_>>()
280        .into_iter()
281        .collect();
282    Writability {
283        file_user_writable: target_w.user_writable(),
284        owner_trust_blockers: target_w
285            .rights
286            .intersection(WriteRights::OWNER_TRUST_BLOCKING),
287        target: target_w,
288        ancestors: ancestor_w,
289        first_user_writable_ancestor,
290        change_perms_by_non_admin,
291        principals_with_write,
292    }
293}

295/// The subfolders of one folder, as a [`DirectorySource`] lists them.
296#[derive(Debug, Clone, PartialEq, Eq, Default)]
297pub struct Listing {
298    /// The subfolders, as full paths.
299    pub dirs: Vec<String>,
300    /// Every entry was read and is either listed or deliberately left out
301    /// (a link, junction or mount point). `false` if the folder could not
302    /// be listed, or an entry could not be read or named; the walk then
303    /// reports [`WritableDirs::truncated`], since writable folders may lie
304    /// in the part not seen.
305    pub complete: bool,
306}
307
308/// Where [`enumerate_user_writable_dirs`] reads folders from.
309pub trait DirectorySource {
310    /// The error the source reports.
311    type Error;
312
313    /// The descriptor of the folder at `path`. `Ok(None)` if it cannot be
314    /// read (counted as writable); `Err` stops the walk.
315    fn descriptor(&mut self, path: &str) -> Result<Option<ReadDescriptor>, Self::Error>;
316
317    /// The subfolders of `path`, as full paths. Links, junctions and mount
318    /// points must not be listed: following them could leave the root or
319    /// loop. A folder that cannot be listed (fully) is an incomplete
320    /// [`Listing`], never an empty complete one.
321    fn subdirectories(&mut self, path: &str) -> Result<Listing, Self::Error>;
322}

324/// Bounds for [`enumerate_user_writable_dirs`].
325#[derive(Debug, Clone, Copy, PartialEq, Eq)]
326pub struct EnumerateOptions {
327    /// The deepest level below a root that is read (the root is 0).
328    pub max_depth: usize,
329    /// The most folders read in total.
330    pub max_dirs: usize,
331    /// Do not descend into a folder found writable: everything below it is
332    /// unsafe for a path rule anyway, so one entry covers it.
333    pub prune_writable: bool,
334}
335
336impl Default for EnumerateOptions {
337    fn default() -> Self {
338        Self {
339            max_depth: 32,
340            max_dirs: 100_000,
341            prune_writable: true,
342        }
343    }
344}

346/// One folder some principal in the set may write to.
347#[derive(Debug, Clone, PartialEq, Eq)]
348pub struct WritableDir {
349    /// The folder's path.
350    pub path: String,
351    /// The rights the principal set may hold on it.
352    pub rights: WriteRights,
353    /// The subset of `rights` that is uncertain.
354    pub uncertain: WriteRights,
355    /// The descriptor could not be read, so writability is assumed.
356    pub unreadable: bool,
357}

359/// The result of [`enumerate_user_writable_dirs`].
360#[derive(Debug, Clone, PartialEq, Eq, Default)]
361pub struct WritableDirs {
362    /// The writable folders, in the order found (breadth first).
363    pub dirs: Vec<WritableDir>,
364    /// Folders read.
365    pub visited: usize,
366    /// A bound in [`EnumerateOptions`] stopped the walk before every
367    /// folder was read, or some folder could not be listed in full.
368    pub truncated: bool,
369    /// The folders that could not be listed in full.
370    pub incomplete: Vec<String>,
371}
372
373/// The folders under `roots` (each a folder path, read first) that the
374/// principal set in `options` may write to, as the synthetic standard user
375/// of [`EvaluationOptions::standard_user`] holds them. Pass options whose
376/// `standard_user` is a single user's SIDs for per-user evaluation.
377///
378/// A folder counts when [`ObjectWritability::standard_user`] holds any
379/// write-type right, including uncertain ones. The walk is breadth first
380/// and bounded by [`EnumerateOptions`]; descriptors are evaluated through
381/// `cache`.
382pub fn enumerate_user_writable_dirs<S: DirectorySource>(
383    roots: &[String],
384    source: &mut S,
385    options: &EvaluationOptions,
386    bounds: &EnumerateOptions,
387    cache: &mut WritabilityCache,
388) -> Result<WritableDirs, S::Error> {
389    let mut out = WritableDirs::default();
390    let mut queue: VecDeque<(String, usize)> = roots.iter().map(|r| (r.clone(), 0)).collect();
391    while let Some((path, depth)) = queue.pop_front() {
392        if out.visited >= bounds.max_dirs {
393            out.truncated = true;
394            break;
395        }
396        out.visited += 1;
397        let sd = source.descriptor(&path)?;
398        let w = cache.result(sd.as_ref(), ObjectKind::Directory, options);
399        let writable = !w.standard_user.is_empty();
400        if writable {
401            out.dirs.push(WritableDir {
402                path: path.clone(),
403                rights: w.standard_user,
404                uncertain: w.standard_user_uncertain,
405                unreadable: w.unreadable,
406            });
407            if bounds.prune_writable {
408                continue;
409            }
410        }
411        let listing = source.subdirectories(&path)?;
412        if !listing.complete {
413            out.truncated = true;
414            out.incomplete.push(path.clone());
415        }
416        if depth >= bounds.max_depth {
417            if !listing.dirs.is_empty() {
418                out.truncated = true;
419            }
420            continue;
421        }
422        for child in listing.dirs {
423            queue.push_back((child, depth + 1));
424        }
425    }
426    if !queue.is_empty() {
427        out.truncated = true;
428    }
429    Ok(out)
430}

475    fn the_os_check_adds_rights_but_never_removes_them() {
476        use crate::fsctx::acl::{OsCheck, Principal};
477        let opts = EvaluationOptions::default();
478        let mut cache = WritabilityCache::default();
479        assert!(!cache.has_os_check());
480
481        // The OS grants a right the evaluator missed: it is added, certain,
482        // for an unknown principal, and the locked folder becomes writable.
483        cache.set_os_check(Some(os_grants_write_data));
484        assert!(cache.has_os_check());
485        let w = cache.evaluate(&read(&locked()), ObjectKind::Directory, &opts);
486        assert!(w.user_writable());

488        assert!(w.uncertain.is_empty());
489        assert_eq!(

493                missed: WriteRights::WRITE_DATA,
494            }
495        );
496        assert_eq!(w.principals.last().unwrap().principal, Principal::Unknown);
497
498        // The OS grants nothing where the evaluator reports Users modify:
499        // the evaluator's result stands (the union).
500        cache.set_os_check(Some(os_grants_nothing));

502        let w = cache.evaluate(&read(&open()), ObjectKind::Directory, &opts);

504        assert_eq!(w.standard_user, portable.standard_user);

514        cache.set_os_check(Some(os_fails));

645        );
646        assert_eq!(got, expected);

649        // Stopping early reads nothing past the first writable folder.
650        let mut read_count = 0;

673    /// An in-memory folder tree. Folders in `unlistable` cannot be listed.
674    struct Tree(
675        Vec<(&'static str, Option<SecurityDescriptor>)>,
676        Vec<&'static str>,
677    );
678
679    impl DirectorySource for Tree {
680        type Error = ();
681        fn descriptor(&mut self, path: &str) -> Result<Option<ReadDescriptor>, ()> {
682            Ok(self
683                .0
684                .iter()
685                .find(|(p, _)| *p == path)
686                .and_then(|(_, sd)| sd.as_ref().map(read)))
687        }
688        fn subdirectories(&mut self, path: &str) -> Result<Listing, ()> {
689            if self.1.contains(&path) {
690                return Ok(Listing::default());
691            }
692            let prefix = format!("{path}\\");
693            let dirs = self
694                .0
695                .iter()
696                .filter(|(p, _)| {
697                    p.strip_prefix(&prefix)
698                        .is_some_and(|rest| !rest.is_empty() && !rest.contains('\\'))
699                })
700                .map(|(p, _)| (*p).to_owned())
701                .collect();
702            Ok(Listing {
703                dirs,
704                complete: true,
705            })
706        }
707    }
