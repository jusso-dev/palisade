1//! Volume and drive classification.
2//!
3//! The OS reports a volume's file system name (`GetVolumeInformationW`), its
4//! drive type (`GetDriveTypeW`) and the bus of the disk behind it
5//! (`IOCTL_STORAGE_QUERY_PROPERTY`, `STORAGE_BUS_TYPE`). The functions here
6//! turn those raw values into enums, and the enums into a
7//! [`LocationClass`] that rules and guardrails can condition on. Reading
8//! the values needs Windows FFI and is left to the OS backend (#54).
9//!
10//! The classification is conservative: when the inputs disagree, the less
11//! trusted class wins. A fixed drive on a USB bus is [`LocationClass::Removable`],
12//! and a disk on a virtual bus is [`LocationClass::VirtualDisk`] unless it
13//! is the OS volume. When an input is missing, the result is
14//! [`LocationClass::Unknown`], never a trusted class: the OS volume is only
15//! [`LocationClass::OsVolume`] when it is known not to be hot-pluggable, and
16//! another fixed disk is only [`LocationClass::OtherFixed`] when its file
17//! system, bus and hot-plug state are all known.
18//!
19//! The location class and owner support are separate questions. A VHD the
20//! OS boots from is [`LocationClass::OsVolume`] for location rules, but its
21//! owners can still have been written on another machine, so
22//! [`VolumeInfo::supports_owner`] also requires a known, non-virtual bus.
23
24#![forbid(unsafe_code)]

26/// The file system of a volume.
27#[derive(Debug, Clone, PartialEq, Eq, Hash)]
28pub enum FsType {
29    /// NTFS.
30    Ntfs,
31    /// ReFS.
32    Refs,
33    /// FAT12, FAT16 or FAT32.
34    Fat,
35    /// exFAT.
36    ExFat,
37    /// UDF (optical and some removable media).
38    Udf,
39    /// CDFS (ISO 9660).
40    Cdfs,
41    /// A network redirector. The OS reports the remote file system's name
42    /// for a share, so the backend sets this from the drive type instead.
43    Network,
44    /// Any other name, as reported.
45    Other(String),
46}
47
48impl FsType {
49    /// Classifies the name `GetVolumeInformationW` reports (`NTFS`, `FAT32`
50    /// and so on), ignoring ASCII case.
51    #[must_use]
52    pub fn from_name(name: &str) -> Self {
53        let n = name.trim().to_ascii_uppercase();
54        match n.as_str() {
55            "NTFS" => Self::Ntfs,
56            "REFS" => Self::Refs,
57            "FAT" | "FAT12" | "FAT16" | "FAT32" => Self::Fat,
58            "EXFAT" => Self::ExFat,
59            "UDF" => Self::Udf,
60            "CDFS" => Self::Cdfs,
61            _ => Self::Other(name.trim().to_owned()),
62        }
63    }
64
65    /// True if the file system records a reliable owner SID for each file.
66    /// FAT, exFAT, UDF and CDFS do not, and a network share's owner is
67    /// whatever the server reports, so owner-based conditions do not apply.
68    #[must_use]
69    pub fn supports_owner(&self) -> bool {
70        matches!(self, Self::Ntfs | Self::Refs)
71    }
72
73    /// True if the file system can hold alternate data streams, and so a
74    /// mark-of-the-web. ReFS gained streams later and only small ones; it is
75    /// counted as supporting them.
76    #[must_use]
77    pub fn supports_streams(&self) -> bool {
78        matches!(self, Self::Ntfs | Self::Refs)
79    }
80}

82/// The drive type `GetDriveTypeW` reports.
83#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
84pub enum DriveType {
85    /// `DRIVE_UNKNOWN` (0) or `DRIVE_NO_ROOT_DIR` (1), or any other value.
86    Unknown,
87    /// `DRIVE_REMOVABLE` (2).
88    Removable,
89    /// `DRIVE_FIXED` (3).
90    Fixed,
91    /// `DRIVE_REMOTE` (4).
92    Network,
93    /// `DRIVE_CDROM` (5).
94    CdRom,
95    /// `DRIVE_RAMDISK` (6).
96    RamDisk,
97}
98
99impl DriveType {
100    /// Maps a `GetDriveTypeW` return value.
101    #[must_use]
102    pub const fn from_win32(value: u32) -> Self {
103        match value {
104            2 => Self::Removable,
105            3 => Self::Fixed,
106            4 => Self::Network,
107            5 => Self::CdRom,
108            6 => Self::RamDisk,
109            _ => Self::Unknown,
110        }
111    }

167            2 => Self::Atapi,

182            17 => Self::Nvme,

184            19 => Self::Ufs,
185            20 => Self::NvmeOf,
186            n => Self::Unknown(n),
187        }
188    }

190    /// True for buses whose devices are usually plugged in by a user: USB,
191    /// IEEE 1394, SD and MMC.
192    #[must_use]
193    pub const fn is_user_pluggable(self) -> bool {
194        matches!(self, Self::Usb | Self::Ieee1394 | Self::Sd | Self::Mmc)
195    }
196
197    /// True for a disk backed by a file or a network target rather than
198    /// local hardware: a VHD/VHDX or an iSCSI or NVMe-over-fabrics target.
199    #[must_use]
200    pub const fn is_virtual(self) -> bool {
201        matches!(
202            self,
203            Self::FileBackedVirtual | Self::Virtual | Self::Iscsi | Self::NvmeOf
204        )
205    }

208/// Where a file lives, for location conditions and guardrails.
209#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
210pub enum LocationClass {
211    /// The volume the OS runs from.
212    OsVolume,
213    /// Another fixed local disk.
214    OtherFixed,
215    /// Removable media, or a disk on a user-pluggable bus.
216    Removable,
217    /// An optical disc.
218    Optical,
219    /// A network share.
220    Network,
221    /// A RAM disk.
222    RamDisk,
223    /// A virtual disk (VHD/VHDX, iSCSI and similar) that is not the OS
224    /// volume.
225    VirtualDisk,
226    /// Not enough information to classify.
227    Unknown,
228}
229
230impl LocationClass {
231    /// True if files here should be treated as coming from outside the
232    /// managed system: everything except the OS volume and other fixed
233    /// local disks.
234    #[must_use]
235    pub const fn is_untrusted_location(self) -> bool {
236        !matches!(self, Self::OsVolume | Self::OtherFixed)
237    }
238}

240/// What is known about a volume. Every field is optional because each comes
241/// from a separate OS call that can fail.
242#[derive(Debug, Clone, PartialEq, Eq, Default)]
243pub struct VolumeInfo {
244    /// The volume serial number.
245    pub serial: Option<u32>,
246    /// The file system.
247    pub fs_type: Option<FsType>,
248    /// The drive type.
249    pub drive_type: Option<DriveType>,
250    /// The bus of the disk.
251    pub bus_type: Option<BusType>,
252    /// The device can be removed while the system runs (hot-plug).
253    pub is_hotplug: Option<bool>,
254    /// The volume is the one the OS runs from.
255    pub is_os_volume: Option<bool>,
256    /// The file system keeps a change journal (USN).
257    pub supports_usn: Option<bool>,
258}
259
260impl VolumeInfo {
261    /// True if the disk is a VHD or VHDX file.
262    #[must_use]
263    pub fn is_vhd(&self) -> bool {
264        self.bus_type == Some(BusType::FileBackedVirtual)
265    }
266
267    /// True if the owner recorded on this volume can be relied on: the file
268    /// system records owners, the volume is a local fixed disk
269    /// ([`LocationClass::OsVolume`] or [`LocationClass::OtherFixed`]), and
270    /// its bus is known and not virtual.
271    ///
272    /// Removable media, virtual disks and network shares are excluded: their
273    /// contents can be prepared on another machine, with any owner SID
274    /// written into them, or the owner is whatever a server reports. That
275    /// holds for a VHD even when the OS boots from it, so the bus is checked
276    /// separately from the location class.
277    #[must_use]
278    pub fn supports_owner(&self) -> bool {
279        let physical_bus = matches!(
280            self.bus_type,
281            Some(b) if !b.is_virtual() && !matches!(b, BusType::Unknown(_))
282        );
283        self.fs_type.as_ref().is_some_and(FsType::supports_owner)
284            && physical_bus
285            && matches!(
286                self.location_class(),
287                LocationClass::OsVolume | LocationClass::OtherFixed
288            )
289    }
290
291    /// The location class. See the module documentation for how
292    /// disagreements are settled.
293    #[must_use]
294    pub fn location_class(&self) -> LocationClass {
295        match self.drive_type {
296            Some(DriveType::Network) => return LocationClass::Network,
297            Some(DriveType::CdRom) => return LocationClass::Optical,
298            Some(DriveType::RamDisk) => return LocationClass::RamDisk,
299            Some(DriveType::Removable) => return LocationClass::Removable,
300            Some(DriveType::Fixed) => {}
301            Some(DriveType::Unknown) | None => return LocationClass::Unknown,
302        }
303        if matches!(self.fs_type, Some(FsType::Network)) {
304            return LocationClass::Network;
305        }
306        // Pluggable and virtual disks are checked before the file system:
307        // a USB disk or a VHD can be formatted with UDF, and is still a
308        // USB disk or a VHD.
309        if self.is_hotplug == Some(true) || self.bus_type.is_some_and(BusType::is_user_pluggable) {
310            return LocationClass::Removable;
311        }
312        if self.is_os_volume == Some(true) {
313            // An OS volume whose hot-plug state is unknown might be on a
314            // pluggable disk, which a known state would make removable.
315            return if self.is_hotplug == Some(false) {
316                LocationClass::OsVolume
317            } else {
318                LocationClass::Unknown
319            };
320        }
321        if self.bus_type.is_some_and(BusType::is_virtual) {
322            return LocationClass::VirtualDisk;
323        }
324        if matches!(self.fs_type, Some(FsType::Cdfs | FsType::Udf)) {
325            return LocationClass::Optical;
326        }
327        // A disk that is neither the OS volume nor known to be a local,
328        // non-pluggable disk stays unknown. External USB disks and attached
329        // VHDs also report `DRIVE_FIXED`, so a failed bus or hot-plug query
330        // must not make a volume look like a trusted fixed disk. Nor must a
331        // failed file-system query: a UDF disk is optical, checked above.
332        let bus_known = matches!(self.bus_type, Some(b) if !matches!(b, BusType::Unknown(_)));
333        if self.is_os_volume == Some(false)
334            && bus_known
335            && self.is_hotplug == Some(false)
336            && self.fs_type.is_some()
337        {
338            LocationClass::OtherFixed
339        } else {
340            LocationClass::Unknown
341        }
342    }
343}

360    #[test]
361    fn raw_values_map() {
362        assert_eq!(FsType::from_name("ntfs"), FsType::Ntfs);
363        assert_eq!(FsType::from_name("FAT32"), FsType::Fat);
364        assert_eq!(FsType::from_name("exFAT"), FsType::ExFat);
365        assert_eq!(FsType::from_name("ReFS"), FsType::Refs);
366        assert_eq!(FsType::from_name(" Btrfs "), FsType::Other("Btrfs".into()));
367        assert_eq!(DriveType::from_win32(3), DriveType::Fixed);
368        assert_eq!(DriveType::from_win32(1), DriveType::Unknown);
369        assert_eq!(DriveType::from_win32(99), DriveType::Unknown);
370        assert_eq!(BusType::from_win32(7), BusType::Usb);
371        assert_eq!(BusType::from_win32(17), BusType::Nvme);
372        assert_eq!(BusType::from_win32(20), BusType::NvmeOf);
373        assert_eq!(BusType::from_win32(0), BusType::Unknown(0));
374        assert_eq!(BusType::from_win32(0x7F), BusType::Unknown(0x7F));

377    #[test]
378    fn classification() {
379        assert_eq!(
380            fixed(BusType::Nvme, true).location_class(),
381            LocationClass::OsVolume
382        );
383        assert_eq!(
384            fixed(BusType::Sata, false).location_class(),
385            LocationClass::OtherFixed
386        );
387        // A fixed drive on USB is removable, even if it claims to be the OS
388        // volume (a portable OS install).
389        assert_eq!(
390            fixed(BusType::Usb, true).location_class(),
391            LocationClass::Removable
392        );
393        assert_eq!(
394            fixed(BusType::FileBackedVirtual, false).location_class(),
395            LocationClass::VirtualDisk
396        );
397        assert!(fixed(BusType::FileBackedVirtual, false).is_vhd());
398        // Booting from a VHD makes it the OS volume.
399        assert_eq!(
400            fixed(BusType::FileBackedVirtual, true).location_class(),
401            LocationClass::OsVolume
402        );
403        // UDF on a USB disk or a VHD is still removable or virtual.
404        let mut udf_usb = fixed(BusType::Usb, false);
405        udf_usb.fs_type = Some(FsType::Udf);
406        assert_eq!(udf_usb.location_class(), LocationClass::Removable);
407        let mut udf_vhd = fixed(BusType::FileBackedVirtual, false);
408        udf_vhd.fs_type = Some(FsType::Udf);
409        assert_eq!(udf_vhd.location_class(), LocationClass::VirtualDisk);
410        let mut udf_sata = fixed(BusType::Sata, false);
411        udf_sata.fs_type = Some(FsType::Udf);
412        assert_eq!(udf_sata.location_class(), LocationClass::Optical);
413        let mut hot = fixed(BusType::Sata, false);
414        hot.is_hotplug = Some(true);
415        assert_eq!(hot.location_class(), LocationClass::Removable);
416
417        for (drive, class) in [
418            (DriveType::Network, LocationClass::Network),
419            (DriveType::CdRom, LocationClass::Optical),
420            (DriveType::RamDisk, LocationClass::RamDisk),
421            (DriveType::Removable, LocationClass::Removable),
422            (DriveType::Unknown, LocationClass::Unknown),
423        ] {
424            let v = VolumeInfo {
425                drive_type: Some(drive),
426                ..VolumeInfo::default()
427            };
428            assert_eq!(v.location_class(), class);
429            assert!(class.is_untrusted_location());
430        }
431        assert_eq!(
432            VolumeInfo::default().location_class(),
433            LocationClass::Unknown
434        );
435        let unknown_os = VolumeInfo {
436            drive_type: Some(DriveType::Fixed),
437            ..VolumeInfo::default()
438        };
439        assert_eq!(unknown_os.location_class(), LocationClass::Unknown);
440        // A non-OS fixed disk whose bus or hot-plug state is unknown is not
441        // promoted to a trusted fixed disk.
442        let mut v = fixed(BusType::Sata, false);
443        v.bus_type = None;
444        assert_eq!(v.location_class(), LocationClass::Unknown);
445        let mut v = fixed(BusType::Unknown(0), false);
446        assert_eq!(v.location_class(), LocationClass::Unknown);
447        v.bus_type = Some(BusType::Sata);
448        v.is_hotplug = None;
449        assert_eq!(v.location_class(), LocationClass::Unknown);
450        // A failed file-system query does not turn a possibly optical disk
451        // into a trusted fixed disk.
452        let mut v = fixed(BusType::Sata, false);
453        v.fs_type = None;
454        assert_eq!(v.location_class(), LocationClass::Unknown);
455        // Nor does an unknown hot-plug state make an OS volume trusted.
456        let mut v = fixed(BusType::Sata, true);
457        v.is_hotplug = None;
458        assert_eq!(v.location_class(), LocationClass::Unknown);
459    }

461    #[test]
462    fn owner_support() {
463        assert!(fixed(BusType::Sata, false).supports_owner());
464        let fat = VolumeInfo {
465            fs_type: Some(FsType::Fat),
466            drive_type: Some(DriveType::Removable),
467            ..VolumeInfo::default()
468        };
469        assert!(!fat.supports_owner());
470        let share = VolumeInfo {
471            fs_type: Some(FsType::Ntfs),
472            drive_type: Some(DriveType::Network),
473            ..VolumeInfo::default()
474        };
475        assert!(!share.supports_owner());
476        assert!(!VolumeInfo::default().supports_owner());
477        // NTFS on removable or virtual media: the owner may have been set
478        // on another machine.
479        assert!(!fixed(BusType::Usb, false).supports_owner());
480        assert!(!fixed(BusType::FileBackedVirtual, false).supports_owner());
481        let ntfs_stick = VolumeInfo {
482            fs_type: Some(FsType::Ntfs),
483            drive_type: Some(DriveType::Removable),
484            ..VolumeInfo::default()
485        };
486        assert!(!ntfs_stick.supports_owner());
487        let mut no_drive = fixed(BusType::Sata, false);
488        no_drive.drive_type = None;
489        assert!(!no_drive.supports_owner());
490        // A boot VHD is the OS volume for location rules, but its owners may
491        // have been written elsewhere.
492        for bus in [
493            BusType::FileBackedVirtual,
494            BusType::Virtual,
495            BusType::Iscsi,
496            BusType::NvmeOf,
497        ] {
498            let boot = fixed(bus, true);
499            assert_eq!(boot.location_class(), LocationClass::OsVolume);
500            assert!(!boot.supports_owner(), "{bus:?}");
501        }
502        // A physical OS volume keeps owner support; an unknown bus or
503        // hot-plug state does not.
504        assert!(fixed(BusType::Nvme, true).supports_owner());
505        let mut os = fixed(BusType::Nvme, true);
506        os.bus_type = None;
507        assert!(!os.supports_owner());
508        os.bus_type = Some(BusType::Unknown(0));
509        assert!(!os.supports_owner());
510        let mut os = fixed(BusType::Sata, true);
511        os.is_hotplug = None;
512        assert!(!os.supports_owner());
513    }
