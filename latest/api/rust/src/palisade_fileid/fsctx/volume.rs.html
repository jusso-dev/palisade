1//! Volume and drive classification.
2//!
3//! The OS reports a volume's file system name (`GetVolumeInformationW`), its
4//! drive type (`GetDriveTypeW`) and the bus of the disk behind it
5//! (`IOCTL_STORAGE_QUERY_PROPERTY`, `STORAGE_BUS_TYPE`). The functions here
6//! turn those raw values into enums, and the enums into a
7//! [`LocationClass`] that rules and guardrails can condition on. The
8//! storage-query results are parsed here too
9//! ([`parse_storage_device_descriptor`], [`parse_hotplug_info`]); the
10//! Windows backend (`palisade_fileid::win::volume`) makes the calls.
11//!
12//! The classification is conservative: when the inputs disagree, the less
13//! trusted class wins. A fixed drive on a USB bus is [`LocationClass::Removable`],
14//! and a disk on a virtual bus is [`LocationClass::VirtualDisk`] unless it
15//! is the OS volume. When an input is missing, the result is
16//! [`LocationClass::Unknown`], never a trusted class: the OS volume is only
17//! [`LocationClass::OsVolume`] when it is known not to be hot-pluggable, and
18//! another fixed disk is only [`LocationClass::OtherFixed`] when its file
19//! system, bus and hot-plug state are all known.
20//!
21//! The location class and owner support are separate questions. A VHD the
22//! OS boots from is [`LocationClass::OsVolume`] for location rules, but its
23//! owners can still have been written on another machine, so
24//! [`VolumeInfo::supports_owner`] also requires a known, non-virtual bus.
25
26#![forbid(unsafe_code)]

28/// The file system of a volume.
29#[derive(Debug, Clone, PartialEq, Eq, Hash)]
30pub enum FsType {
31    /// NTFS.
32    Ntfs,
33    /// ReFS.
34    Refs,
35    /// FAT12, FAT16 or FAT32.
36    Fat,
37    /// exFAT.
38    ExFat,
39    /// UDF (optical and some removable media).
40    Udf,
41    /// CDFS (ISO 9660).
42    Cdfs,
43    /// A network redirector. The OS reports the remote file system's name
44    /// for a share, so the backend sets this from the drive type instead.
45    Network,
46    /// Any other name, as reported.
47    Other(String),
48}
49
50impl FsType {
51    /// Classifies the name `GetVolumeInformationW` reports (`NTFS`, `FAT32`
52    /// and so on), ignoring ASCII case.
53    #[must_use]
54    pub fn from_name(name: &str) -> Self {
55        let n = name.trim().to_ascii_uppercase();
56        match n.as_str() {
57            "NTFS" => Self::Ntfs,
58            "REFS" => Self::Refs,
59            "FAT" | "FAT12" | "FAT16" | "FAT32" => Self::Fat,
60            "EXFAT" => Self::ExFat,
61            "UDF" => Self::Udf,
62            "CDFS" => Self::Cdfs,
63            _ => Self::Other(name.trim().to_owned()),
64        }
65    }
66
67    /// True if the file system records a reliable owner SID for each file.
68    /// FAT, exFAT, UDF and CDFS do not, and a network share's owner is
69    /// whatever the server reports, so owner-based conditions do not apply.
70    #[must_use]
71    pub fn supports_owner(&self) -> bool {
72        matches!(self, Self::Ntfs | Self::Refs)
73    }
74
75    /// True if the file system can hold alternate data streams, and so a
76    /// mark-of-the-web. ReFS gained streams later and only small ones; it is
77    /// counted as supporting them.
78    #[must_use]
79    pub fn supports_streams(&self) -> bool {
80        matches!(self, Self::Ntfs | Self::Refs)
81    }
82}

210/// Where a file lives, for location conditions and guardrails.
211#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
212pub enum LocationClass {
213    /// The volume the OS runs from.
214    OsVolume,
215    /// Another fixed local disk.
216    OtherFixed,
217    /// Removable media, or a disk on a user-pluggable bus.
218    Removable,
219    /// An optical disc.
220    Optical,
221    /// A network share.
222    Network,
223    /// A RAM disk.
224    RamDisk,
225    /// A virtual disk (VHD/VHDX, iSCSI and similar) that is not the OS
226    /// volume.
227    VirtualDisk,
228    /// Not enough information to classify.
229    Unknown,
230}
231
232impl LocationClass {
233    /// True if files here should be treated as coming from outside the
234    /// managed system: everything except the OS volume and other fixed
235    /// local disks.
236    #[must_use]
237    pub const fn is_untrusted_location(self) -> bool {
238        !matches!(self, Self::OsVolume | Self::OtherFixed)
239    }
240}

242/// What is known about a volume. Every field is optional because each comes
243/// from a separate OS call that can fail.
244#[derive(Debug, Clone, PartialEq, Eq, Default)]
245pub struct VolumeInfo {
246    /// The volume serial number.
247    pub serial: Option<u32>,
248    /// The file system.
249    pub fs_type: Option<FsType>,
250    /// The drive type.
251    pub drive_type: Option<DriveType>,
252    /// The bus of the disk.
253    pub bus_type: Option<BusType>,
254    /// The device can be removed while the system runs (hot-plug).
255    pub is_hotplug: Option<bool>,
256    /// The volume is the one the OS runs from.
257    pub is_os_volume: Option<bool>,
258    /// The file system keeps a change journal (USN).
259    pub supports_usn: Option<bool>,
260}
261
262impl VolumeInfo {
263    /// True if the disk is a VHD or VHDX file.
264    #[must_use]
265    pub fn is_vhd(&self) -> bool {
266        self.bus_type == Some(BusType::FileBackedVirtual)
267    }
268
269    /// True if the owner recorded on this volume can be relied on: the file
270    /// system records owners, the volume is a local fixed disk
271    /// ([`LocationClass::OsVolume`] or [`LocationClass::OtherFixed`]), and
272    /// its bus is known and not virtual.
273    ///
274    /// Removable media, virtual disks and network shares are excluded: their
275    /// contents can be prepared on another machine, with any owner SID
276    /// written into them, or the owner is whatever a server reports. That
277    /// holds for a VHD even when the OS boots from it, so the bus is checked
278    /// separately from the location class.
279    #[must_use]
280    pub fn supports_owner(&self) -> bool {
281        let physical_bus = matches!(
282            self.bus_type,
283            Some(b) if !b.is_virtual() && !matches!(b, BusType::Unknown(_))
284        );
285        self.fs_type.as_ref().is_some_and(FsType::supports_owner)
286            && physical_bus
287            && matches!(
288                self.location_class(),
289                LocationClass::OsVolume | LocationClass::OtherFixed
290            )
291    }
292
293    /// The location class. See the module documentation for how
294    /// disagreements are settled.
295    #[must_use]
296    pub fn location_class(&self) -> LocationClass {
297        match self.drive_type {
298            Some(DriveType::Network) => return LocationClass::Network,
299            Some(DriveType::CdRom) => return LocationClass::Optical,
300            Some(DriveType::RamDisk) => return LocationClass::RamDisk,
301            Some(DriveType::Removable) => return LocationClass::Removable,
302            Some(DriveType::Fixed) => {}
303            Some(DriveType::Unknown) | None => return LocationClass::Unknown,
304        }
305        if matches!(self.fs_type, Some(FsType::Network)) {
306            return LocationClass::Network;
307        }
308        // Pluggable and virtual disks are checked before the file system:
309        // a USB disk or a VHD can be formatted with UDF, and is still a
310        // USB disk or a VHD.
311        if self.is_hotplug == Some(true) || self.bus_type.is_some_and(BusType::is_user_pluggable) {
312            return LocationClass::Removable;
313        }
314        if self.is_os_volume == Some(true) {
315            // An OS volume whose hot-plug state is unknown might be on a
316            // pluggable disk, which a known state would make removable.
317            return if self.is_hotplug == Some(false) {
318                LocationClass::OsVolume
319            } else {
320                LocationClass::Unknown
321            };
322        }
323        if self.bus_type.is_some_and(BusType::is_virtual) {
324            return LocationClass::VirtualDisk;
325        }
326        if matches!(self.fs_type, Some(FsType::Cdfs | FsType::Udf)) {
327            return LocationClass::Optical;
328        }
329        // A disk that is neither the OS volume nor known to be a local,
330        // non-pluggable disk stays unknown. External USB disks and attached
331        // VHDs also report `DRIVE_FIXED`, so a failed bus or hot-plug query
332        // must not make a volume look like a trusted fixed disk. Nor must a
333        // failed file-system query: a UDF disk is optical, checked above.
334        let bus_known = matches!(self.bus_type, Some(b) if !matches!(b, BusType::Unknown(_)));
335        if self.is_os_volume == Some(false)
336            && bus_known
337            && self.is_hotplug == Some(false)
338            && self.fs_type.is_some()
339        {
340            LocationClass::OtherFixed
341        } else {
342            LocationClass::Unknown
343        }
344    }
345}

347/// Independent signs that a file was reached through a network redirector.
348/// Each is `None` when it could not be read.
349///
350/// No single OS answer is relied on. `FileRemoteProtocolInfo` is the
351/// documented test, but it has been seen to fail on a real redirector
352/// handle (#54, the Windows 11 Arm64 runner), and a share's server reports
353/// its own file system (`NTFS`) and serial. Any one sign saying "network"
354/// makes the volume [`VolumeInfo::network`]; see [`VolumeInfo::settle`].
355#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
356pub struct NetworkEvidence {
357    /// `FileRemoteProtocolInfo` answered for the handle, which only a
358    /// redirector does.
359    pub remote_protocol: Option<bool>,
360    /// The handle's NT path is under a network provider
361    /// (`\Device\Mup\…`), from [`nt_path_on_network`].
362    pub nt_path_on_network: Option<bool>,
363    /// The name the file was opened by is a UNC path, or a drive letter
364    /// the OS reports as remote.
365    pub opened_as_network: Option<bool>,
366}
367
368impl NetworkEvidence {
369    /// True if any sign says the file is on the network.
370    #[must_use]
371    pub const fn is_network(&self) -> bool {
372        matches!(self.remote_protocol, Some(true))
373            || matches!(self.nt_path_on_network, Some(true))
374            || matches!(self.opened_as_network, Some(true))
375    }
376}

378/// NT devices of network redirectors and providers: the multiple UNC
379/// provider, SMB, WebDAV, remote-desktop drive redirection and the
380/// Linux-subsystem file server.
381const REDIRECTOR_DEVICES: [&str; 5] = [
382    "Mup",
383    "LanmanRedirector",
384    "WebDavRedirector",
385    "RdpDr",
386    "P9Rdr",
387];
388
389/// What an NT device name (`Mup`, `HarddiskVolume3`) says: `Some(true)` for
390/// a redirector, `Some(false)` for a local disk volume or shadow copy,
391/// `None` for anything else.
392fn nt_device_on_network(device: &str) -> Option<bool> {
393    if REDIRECTOR_DEVICES
394        .iter()
395        .any(|r| device.eq_ignore_ascii_case(r))
396    {
397        return Some(true);
398    }
399    let local = device
400        .get(..14)
401        .is_some_and(|p| p.eq_ignore_ascii_case("HarddiskVolume"))
402        && device.len() > 14;
403    local.then_some(false)
404}
405
406/// Whether an NT path, as `GetFinalPathNameByHandleW` reports it with
407/// `VOLUME_NAME_NT`, is under a network provider. `Some(true)` for
408/// `\Device\Mup\…`, `\Device\LanmanRedirector\…` and the other redirector
409/// devices; `Some(false)` for a local disk volume
410/// (`\Device\HarddiskVolumeN\…`); `None` for any other device or a path
411/// that is not under `\Device\`, which never counts as either.
412///
413/// Only the device name is read, not the rest of the path, so a
414/// redirector path the path normaliser would reject (a provider prefix
415/// such as `;:1`) still counts as the network.
416#[must_use]
417pub fn nt_path_on_network(nt_path: &str) -> Option<bool> {
418    let mut parts = nt_path.split('\\');
419    let (Some(""), Some(dev), Some(device)) = (parts.next(), parts.next(), parts.next()) else {
420        return None;
421    };
422    if !dev.eq_ignore_ascii_case("Device") {
423        return None;
424    }
425    nt_device_on_network(device)
426}

428/// What the name a file was opened by says about the network, without the
429/// OS: `Some(true)` for a UNC path (also `\\?\UNC\…` and
430/// `\\?\GLOBALROOT\Device\Mup\…`), `Some(false)` for a volume GUID or local
431/// disk path, and `None` for a drive letter, which only the OS can map
432/// (it may be a mapped network drive), or anything else.
433#[must_use]
434pub fn opened_name_on_network(path: &super::path::NormalizedPath) -> Option<bool> {
435    match path.root() {
436        super::path::Root::Unc { .. } => Some(true),
437        super::path::Root::VolumeGuid(_) => Some(false),
438        super::path::Root::NtDevice(d) => nt_device_on_network(d),
439        _ => None,
440    }
441}

443impl VolumeInfo {
444    /// A volume reached through a network redirector: [`FsType::Network`],
445    /// [`DriveType::Network`] and no change journal, whatever the server
446    /// reports. The serial is kept as reported; it is the server's.
447    #[must_use]
448    pub fn network(serial: Option<u32>) -> Self {
449        Self {
450            serial,
451            fs_type: Some(FsType::Network),
452            drive_type: Some(DriveType::Network),
453            supports_usn: Some(false),
454            ..Self::default()
455        }
456    }
457
458    /// Settles what the OS `reported` for a handle into the value the
459    /// backend returns, so a missing or contradicted fact can never make
460    /// the volume look like a trusted local disk (ADR 0018):
461    ///
462    /// - any sign of the network in `evidence` gives [`VolumeInfo::network`],
463    ///   even if the redirector query said "local";
464    /// - otherwise, if the handle's local volume root could not be resolved
465    ///   (`root_resolved` is false), only the serial is kept. The file
466    ///   system, journal and drive facts may then be a server's, so they are
467    ///   dropped and the class is [`LocationClass::Unknown`];
468    /// - otherwise `reported` is returned unchanged.
469    #[must_use]
470    pub fn settle(reported: Self, evidence: NetworkEvidence, root_resolved: bool) -> Self {
471        if evidence.is_network() {
472            Self::network(reported.serial)
473        } else if root_resolved {
474            reported
475        } else {
476            Self {
477                serial: reported.serial,
478                ..Self::default()
479            }
480        }
481    }
482}

484/// What the disk behind a volume says about itself: the header of the
485/// `STORAGE_DEVICE_DESCRIPTOR` that `IOCTL_STORAGE_QUERY_PROPERTY` returns
486/// for `StorageDeviceProperty`.
487#[derive(Debug, Clone, Copy, PartialEq, Eq)]
488pub struct StorageDevice {
489    /// The bus.
490    pub bus_type: BusType,
491    /// The device reports removable media.
492    pub removable_media: bool,
493}
494
495/// The fixed part of `STORAGE_DEVICE_DESCRIPTOR`, up to and including
496/// `BusType`.
497const STORAGE_DEVICE_HEADER: usize = 32;
498
499fn u32_le(b: &[u8], at: usize) -> Option<u32> {
500    let bytes = b.get(at..at.checked_add(4)?)?;
501    Some(u32::from_le_bytes(bytes.try_into().ok()?))
502}
503
504/// Parses the `STORAGE_DEVICE_DESCRIPTOR` bytes the OS returned. `None` if
505/// they are shorter than the header, or the descriptor's own `Size` says
506/// it is.
507#[must_use]
508pub fn parse_storage_device_descriptor(bytes: &[u8]) -> Option<StorageDevice> {
509    let size = usize::try_from(u32_le(bytes, 4)?).ok()?;
510    if bytes.len() < STORAGE_DEVICE_HEADER || size < STORAGE_DEVICE_HEADER {
511        return None;
512    }
513    Some(StorageDevice {
514        bus_type: BusType::from_win32(u32_le(bytes, 28)?),
515        removable_media: *bytes.get(10)? != 0,
516    })
517}

519/// What `IOCTL_STORAGE_GET_HOTPLUG_INFO` reports (`STORAGE_HOTPLUG_INFO`).
520#[derive(Debug, Clone, Copy, PartialEq, Eq)]
521pub struct HotplugInfo {
522    /// The media can be removed from the device.
523    pub media_removable: bool,
524    /// The media can be removed while the system runs.
525    pub media_hotplug: bool,
526    /// The device itself can be removed while the system runs.
527    pub device_hotplug: bool,
528}
529
530impl HotplugInfo {
531    /// Whether the device or its media can be removed while the system
532    /// runs: the value of [`VolumeInfo::is_hotplug`].
533    #[must_use]
534    pub const fn is_hotplug(self) -> bool {
535        self.media_hotplug || self.device_hotplug
536    }
537}
538
539/// The value of [`VolumeInfo::is_hotplug`] from the two storage queries,
540/// either of which may have failed. Removable media counts as
541/// hot-pluggable: it can be swapped while the system runs, whatever the
542/// hot-plug flags say. Either query reporting removable media is enough.
543/// `None` only when neither query answered and so nothing is known.
544#[must_use]
545pub fn combined_hotplug(
546    storage: Option<StorageDevice>,
547    hotplug: Option<HotplugInfo>,
548) -> Option<bool> {
549    let removable =
550        storage.is_some_and(|s| s.removable_media) || hotplug.is_some_and(|h| h.media_removable);
551    match hotplug {
552        Some(h) => Some(h.is_hotplug() || removable),
553        None if removable => Some(true),
554        None => None,
555    }
556}
557
558/// Parses the `STORAGE_HOTPLUG_INFO` bytes the OS returned: a `u32` size
559/// and four one-byte flags. `None` if shorter.
560#[must_use]
561pub fn parse_hotplug_info(bytes: &[u8]) -> Option<HotplugInfo> {
562    let size = usize::try_from(u32_le(bytes, 0)?).ok()?;
563    let flags = bytes.get(4..8)?;
564    if size < 8 {
565        return None;
566    }
567    Some(HotplugInfo {
568        media_removable: flags.first().is_some_and(|&b| b != 0),
569        media_hotplug: flags.get(1).is_some_and(|&b| b != 0),
570        device_hotplug: flags.get(2).is_some_and(|&b| b != 0),
571    })
572}

619            removable_media,
620        };
621        // The storage query failed; the hot-plug reply says removable
622        // media with both hot-plug flags clear.
623        let only_removable = flags(true, false, false);
624        assert!(!only_removable.is_hotplug());
625        assert_eq!(combined_hotplug(None, Some(only_removable)), Some(true));
626        assert_eq!(
627            combined_hotplug(Some(disk(false)), Some(only_removable)),
628            Some(true)
629        );
630        // Removable media from the storage query alone.
631        assert_eq!(combined_hotplug(Some(disk(true)), None), Some(true));
632        assert_eq!(
633            combined_hotplug(Some(disk(true)), Some(flags(false, false, false))),
634            Some(true)
635        );
636        // Either hot-plug flag.
637        assert_eq!(
638            combined_hotplug(None, Some(flags(false, false, true))),
639            Some(true)
640        );
641        assert_eq!(
642            combined_hotplug(None, Some(flags(false, true, false))),
643            Some(true)
644        );
645        // A fixed, non-pluggable disk, and nothing known.
646        assert_eq!(
647            combined_hotplug(Some(disk(false)), Some(flags(false, false, false))),
648            Some(false)
649        );
650        assert_eq!(combined_hotplug(Some(disk(false)), None), None);
651        assert_eq!(combined_hotplug(None, None), None);
652    }

682    #[test]
683    fn classification() {
684        assert_eq!(
685            fixed(BusType::Nvme, true).location_class(),
686            LocationClass::OsVolume
687        );
688        assert_eq!(
689            fixed(BusType::Sata, false).location_class(),
690            LocationClass::OtherFixed
691        );
692        // A fixed drive on USB is removable, even if it claims to be the OS
693        // volume (a portable OS install).
694        assert_eq!(
695            fixed(BusType::Usb, true).location_class(),
696            LocationClass::Removable
697        );
698        assert_eq!(
699            fixed(BusType::FileBackedVirtual, false).location_class(),
700            LocationClass::VirtualDisk
701        );
702        assert!(fixed(BusType::FileBackedVirtual, false).is_vhd());
703        // Booting from a VHD makes it the OS volume.
704        assert_eq!(
705            fixed(BusType::FileBackedVirtual, true).location_class(),
706            LocationClass::OsVolume
707        );
708        // UDF on a USB disk or a VHD is still removable or virtual.
709        let mut udf_usb = fixed(BusType::Usb, false);
710        udf_usb.fs_type = Some(FsType::Udf);
711        assert_eq!(udf_usb.location_class(), LocationClass::Removable);
712        let mut udf_vhd = fixed(BusType::FileBackedVirtual, false);
713        udf_vhd.fs_type = Some(FsType::Udf);
714        assert_eq!(udf_vhd.location_class(), LocationClass::VirtualDisk);
715        let mut udf_sata = fixed(BusType::Sata, false);
716        udf_sata.fs_type = Some(FsType::Udf);
717        assert_eq!(udf_sata.location_class(), LocationClass::Optical);
718        let mut hot = fixed(BusType::Sata, false);
719        hot.is_hotplug = Some(true);
720        assert_eq!(hot.location_class(), LocationClass::Removable);
721
722        for (drive, class) in [
723            (DriveType::Network, LocationClass::Network),
724            (DriveType::CdRom, LocationClass::Optical),
725            (DriveType::RamDisk, LocationClass::RamDisk),
726            (DriveType::Removable, LocationClass::Removable),
727            (DriveType::Unknown, LocationClass::Unknown),
728        ] {
729            let v = VolumeInfo {
730                drive_type: Some(drive),
731                ..VolumeInfo::default()
732            };
733            assert_eq!(v.location_class(), class);
734            assert!(class.is_untrusted_location());
735        }
736        assert_eq!(
737            VolumeInfo::default().location_class(),
738            LocationClass::Unknown
739        );
740        let unknown_os = VolumeInfo {
741            drive_type: Some(DriveType::Fixed),
742            ..VolumeInfo::default()
743        };
744        assert_eq!(unknown_os.location_class(), LocationClass::Unknown);
745        // A non-OS fixed disk whose bus or hot-plug state is unknown is not
746        // promoted to a trusted fixed disk.
747        let mut v = fixed(BusType::Sata, false);
748        v.bus_type = None;
749        assert_eq!(v.location_class(), LocationClass::Unknown);
750        let mut v = fixed(BusType::Unknown(0), false);
751        assert_eq!(v.location_class(), LocationClass::Unknown);
752        v.bus_type = Some(BusType::Sata);
753        v.is_hotplug = None;
754        assert_eq!(v.location_class(), LocationClass::Unknown);
755        // A failed file-system query does not turn a possibly optical disk
756        // into a trusted fixed disk.
757        let mut v = fixed(BusType::Sata, false);
758        v.fs_type = None;
759        assert_eq!(v.location_class(), LocationClass::Unknown);
760        // Nor does an unknown hot-plug state make an OS volume trusted.
761        let mut v = fixed(BusType::Sata, true);
762        v.is_hotplug = None;
763        assert_eq!(v.location_class(), LocationClass::Unknown);
764    }

766    #[test]
767    fn owner_support() {
768        assert!(fixed(BusType::Sata, false).supports_owner());
769        let fat = VolumeInfo {
770            fs_type: Some(FsType::Fat),
771            drive_type: Some(DriveType::Removable),
772            ..VolumeInfo::default()
773        };
774        assert!(!fat.supports_owner());
775        let share = VolumeInfo {
776            fs_type: Some(FsType::Ntfs),
777            drive_type: Some(DriveType::Network),
778            ..VolumeInfo::default()
779        };
780        assert!(!share.supports_owner());
781        assert!(!VolumeInfo::default().supports_owner());
782        // NTFS on removable or virtual media: the owner may have been set
783        // on another machine.
784        assert!(!fixed(BusType::Usb, false).supports_owner());
785        assert!(!fixed(BusType::FileBackedVirtual, false).supports_owner());
786        let ntfs_stick = VolumeInfo {
787            fs_type: Some(FsType::Ntfs),
788            drive_type: Some(DriveType::Removable),
789            ..VolumeInfo::default()
790        };
791        assert!(!ntfs_stick.supports_owner());
792        let mut no_drive = fixed(BusType::Sata, false);
793        no_drive.drive_type = None;
794        assert!(!no_drive.supports_owner());
795        // A boot VHD is the OS volume for location rules, but its owners may
796        // have been written elsewhere.
797        for bus in [
798            BusType::FileBackedVirtual,
799            BusType::Virtual,
800            BusType::Iscsi,
801            BusType::NvmeOf,
802        ] {
803            let boot = fixed(bus, true);
804            assert_eq!(boot.location_class(), LocationClass::OsVolume);
805            assert!(!boot.supports_owner(), "{bus:?}");
806        }
807        // A physical OS volume keeps owner support; an unknown bus or
808        // hot-plug state does not.
809        assert!(fixed(BusType::Nvme, true).supports_owner());
810        let mut os = fixed(BusType::Nvme, true);
811        os.bus_type = None;
812        assert!(!os.supports_owner());
813        os.bus_type = Some(BusType::Unknown(0));
814        assert!(!os.supports_owner());
815        let mut os = fixed(BusType::Sata, true);
816        os.is_hotplug = None;
817        assert!(!os.supports_owner());
818    }
