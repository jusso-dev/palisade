1//! The version resource (`VS_VERSIONINFO`, resource type `RT_VERSION`).
2//!
3//! [`parse_version_info`] reads the fixed file info (numeric file and product
4//! versions, flags, OS and type) and every `StringFileInfo` language block.
5//! [`VersionInfo`] then selects one block, gives typed access to the standard
6//! keys, and provides the file-name fallback chain that OS-native policy
7//! rules use.
8//!
9//! Only the 32-bit (UTF-16) layout is read; the 16-bit layout used by NE
10//! images is rejected with [`VersionIssueKind::BadRootKey`].
11//!
12//! The resource is untrusted. Its size is at most 64 KiB (`wLength` is a
13//! `u16`), every block length is checked against its parent, and the number
14//! of language blocks and strings kept is capped.
15
16#![forbid(unsafe_code)]
17

23/// Language blocks (`StringTable`s) kept at most in
24/// [`VersionInfo::all_langs`]. The selected block is always among them.
25pub const MAX_STRING_TABLES: usize = 64;
26/// Strings kept at most per language block.
27pub const MAX_STRINGS_PER_TABLE: usize = 256;
28/// `Translation` entries kept at most.
29pub const MAX_TRANSLATIONS: usize = 64;
30/// Child blocks examined at most per parent, so a block full of tiny
31/// children cannot cause unbounded work.
32const MAX_CHILDREN: usize = 4096;

35const FIXED_INFO_SIZE: usize = 52;

44/// A problem in the version resource. `offset` is relative to the start of
45/// the resource data.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct VersionInfoIssue {
48    /// Offset from the start of the resource data.
49    pub offset: u64,
50    /// What went wrong.
51    pub kind: VersionIssueKind,
52}
53
54/// The kind of a [`VersionInfoIssue`].
55#[derive(Debug, Clone, PartialEq, Eq)]
56#[non_exhaustive]
57pub enum VersionIssueKind {
58    /// A block header or key is cut off.
59    Truncated,
60    /// The root key is not `VS_VERSION_INFO`.
61    BadRootKey(String),
62    /// The fixed file info has a wrong signature or size; it was ignored.
63    BadFixedInfo,
64    /// A block's `wLength` runs past its parent; it was cut to fit.
65    BlockOverrun,
66    /// A `StringTable` key is not eight hex digits (language + code page).
67    BadStringTableKey(String),
68    /// More than [`MAX_STRING_TABLES`] language blocks; the rest were
69    /// skipped.
70    TooManyStringTables,
71    /// More than [`MAX_STRINGS_PER_TABLE`] strings in one block; the rest
72    /// were skipped.
73    TooManyStrings,
74    /// More child blocks than the walker examines in one parent.
75    TooManyChildren,
76    /// A second `StringFileInfo`, `VarFileInfo` or `Translation` block. The
77    /// OS version API only reaches the first, so this one was ignored.
78    DuplicateBlock(String),
79}

91/// A four-part numeric version, `major.minor.build.revision`.
92///
93/// Ordering compares the parts numerically from left to right, so
94/// `10.0.2.0 > 9.9.9.9`. Version comparisons must use this type, never the
95/// version strings.
96#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Default)]
97pub struct FileVersionQuad(pub u16, pub u16, pub u16, pub u16);
98
99impl FileVersionQuad {
100    /// Builds a version from the fixed file info's most and least
101    /// significant 32-bit halves.
102    #[must_use]
103    pub const fn from_ms_ls(ms: u32, ls: u32) -> Self {
104        // Each half holds two 16-bit parts, so the shifts and masks cannot
105        // lose information.
106        #[allow(clippy::cast_possible_truncation)]
107        Self(
108            (ms >> 16) as u16,
109            (ms & 0xffff) as u16,
110            (ls >> 16) as u16,
111            (ls & 0xffff) as u16,
112        )
113    }
114
115    /// Parses the leading numeric part of a version string, such as
116    /// `"10.0.19041.1 (build.160101-0800)"` or `"1, 2, 3, 4"`.
117    ///
118    /// One to four parts separated by `.` or `,` are accepted (spaces around
119    /// separators are ignored); missing parts are zero. Text after the
120    /// numeric part is ignored. Returns `None` if there is no leading number
121    /// or a part does not fit in 16 bits.
122    #[must_use]
123    pub fn parse_prefix(text: &str) -> Option<Self> {
124        let mut parts = [0_u16; 4];
125        let mut count = 0_usize;
126        let mut rest = text.trim_start();
127        loop {
128            let digits = rest.bytes().take_while(u8::is_ascii_digit).count();
129            if digits == 0 {
130                break;
131            }
132            let (number, tail) = rest.split_at(digits);
133            let slot = parts.get_mut(count)?;
134            *slot = number.parse().ok()?;
135            count += 1;
136            if count == 4 {
137                break;
138            }
139            let trimmed = tail.trim_start();
140            match trimmed.strip_prefix(['.', ',']) {
141                Some(after) if after.trim_start().starts_with(|c: char| c.is_ascii_digit()) => {
142                    rest = after.trim_start();
143                }
144                _ => break,
145            }
146        }
147        (count > 0).then(|| {
148            let [a, b, c, d] = parts;
149            Self(a, b, c, d)
150        })
151    }
152}

154impl fmt::Display for FileVersionQuad {

156        write!(f, "{}.{}.{}.{}", self.0, self.1, self.2, self.3)
157    }
158}

209impl StringTable {

214        self.strings
215            .iter()
216            .find(|(k, _)| k.eq_ignore_ascii_case(key))
217            .map(|(_, v)| v.as_str())

329/// The parsed version resource.
330#[derive(Debug, Clone, PartialEq, Eq)]
331pub struct VersionInfo {
332    /// `VS_FIXEDFILEINFO`, if present and valid.
333    pub fixed: Option<FixedFileInfo>,
334    /// The language blocks of the first `StringFileInfo`, in file order, at
335    /// most [`MAX_STRING_TABLES`]. These are the first blocks, except that
336    /// when the selected block lies past the cap it takes the last slot.
337    /// (Later `StringFileInfo` blocks are unreachable through the OS version
338    /// API and are reported as [`VersionIssueKind::DuplicateBlock`].)
339    pub all_langs: Vec<StringTable>,
340    /// The first [`MAX_TRANSLATIONS`] `VarFileInfo\Translation` entries, in
341    /// file order. Language selection considers every entry, not only
342    /// these.
343    pub translations: Vec<Translation>,
344    /// Index into [`all_langs`](Self::all_langs) of the selected block.
345    pub selected: Option<usize>,
346    /// How the block was selected.
347    pub selection: Option<LanguageSelection>,
348    /// Non-fatal problems.
349    pub issues: Vec<VersionInfoIssue>,
350}

352impl VersionInfo {
353    /// The selected language block.
354    #[must_use]
355    pub fn selected_table(&self) -> Option<&StringTable> {
356        self.all_langs.get(self.selected?)
357    }

369        self.fixed.map(|f| f.file_version).or_else(|| {
370            self.get(VersionField::FileVersion)

373    }

376    /// [`file_version`](Self::file_version).
377    #[must_use]
378    pub fn product_version(&self) -> Option<FileVersionQuad> {
379        self.fixed.map(|f| f.product_version).or_else(|| {
380            self.get(VersionField::ProductVersion)
381                .and_then(FileVersionQuad::parse_prefix)
382        })
383    }

385    /// The `FileVersion` string parses to a different version than the fixed
386    /// info. `false` when either is missing or the string does not parse.
387    #[must_use]
388    pub fn file_version_mismatch(&self) -> bool {
389        mismatch(
390            self.fixed.map(|f| f.file_version),
391            self.get(VersionField::FileVersion),
392        )
393    }
394
395    /// The `ProductVersion` string parses to a different version than the
396    /// fixed info.
397    #[must_use]
398    pub fn product_version_mismatch(&self) -> bool {
399        mismatch(
400            self.fixed.map(|f| f.product_version),
401            self.get(VersionField::ProductVersion),
402        )
403    }
404
405    /// The file name OS-native policy would match on.
406    ///
407    /// Walks [`VersionField::FILE_NAME_CHAIN`] (`OriginalFileName`, then
408    /// `InternalName`, `FileDescription`, `ProductName`) starting at
409    /// `level`, and returns the first field whose value is not empty or
410    /// whitespace. Pass [`VersionField::OriginalFileName`] for the full
411    /// chain. A `level` outside the chain returns `None`.
412    #[must_use]
413    pub fn effective_file_name(&self, level: VersionField) -> Option<EffectiveFileName<'_>> {
414        let start = VersionField::FILE_NAME_CHAIN
415            .iter()
416            .position(|f| *f == level)?;
417        VersionField::FILE_NAME_CHAIN
418            .iter()
419            .skip(start)
420            .find_map(|field| {
421                let value = self.get(*field)?;
422                (!value.trim().is_empty()).then_some(EffectiveFileName {
423                    field: *field,
424                    value,
425                })
426            })
427    }
428}

430fn mismatch(fixed: Option<FileVersionQuad>, text: Option<&str>) -> bool {
431    match (fixed, text.and_then(FileVersionQuad::parse_prefix)) {
432        (Some(fixed), Some(parsed)) => fixed != parsed,
433        _ => false,
434    }

437/// One block header: `wLength`, `wValueLength`, `wType`, `szKey`.
438#[derive(Clone)]
439struct Block {
440    start: usize,
441    /// End of the block, cut to the parent's end.
442    end: usize,
443    /// `wValueLength`: in bytes for binary values (`wType` 0), in UTF-16
444    /// code units for text values (`wType` 1).
445    value_len: u16,
446    /// `wType`: 0 for binary data, 1 for text.
447    w_type: u16,
448    key: String,
449    /// Start of the value, after the key and padding.
450    value_start: usize,
451}
452
453impl Block {
454    /// The value bytes, `byte_len` long, cut to the block end.
455    fn value<'a>(&self, data: &'a [u8], byte_len: usize) -> &'a [u8] {
456        let end = self.value_start.saturating_add(byte_len).min(self.end);
457        data.get(self.value_start..end).unwrap_or_default()
458    }
459
460    /// The value's length in bytes, as the OS version API computes it:
461    /// `wValueLength`, doubled for text values.
462    fn value_bytes(&self) -> usize {
463        let len = usize::from(self.value_len);
464        if self.w_type == 0 {
465            len
466        } else {
467            len.saturating_mul(2)
468        }
469    }
470
471    /// Start of the children: after the value, whatever block this is, and
472    /// 4-byte aligned. Skipping the value the same way for every block, as
473    /// the OS does, means bytes hidden inside a value are never read as
474    /// child blocks.
475    fn children_start(&self) -> usize {
476        align4(self.value_start.saturating_add(self.value_bytes()))
477            .unwrap_or(usize::MAX)
478            .min(self.end)
479    }
480}

482/// Reads the block at `start`, which must end by `limit`. Returns `None` for
483/// a zero `wLength` (end-of-children padding), and an issue when the header
484/// or key is unreadable.
485fn read_block(
486    data: &[u8],
487    start: usize,
488    limit: usize,
489    issues: &mut Vec<VersionInfoIssue>,
490) -> Result<Option<Block>, VersionInfoIssue> {
491    let truncated = VersionInfoIssue {
492        offset: to_u64(start),
493        kind: VersionIssueKind::Truncated,
494    };
495    let header =
496        bytes(data, start, BLOCK_HEADER).filter(|_| start.saturating_add(BLOCK_HEADER) <= limit);
497    let Some(header) = header else {
498        return Err(truncated);
499    };
500    let len = usize::from(u16_at(header, 0).unwrap_or(0));
501    if len == 0 {
502        return Ok(None);
503    }
504    if len < BLOCK_HEADER {
505        return Err(truncated);
506    }
507    let mut end = start.saturating_add(len);
508    if end > limit {
509        issues.push(VersionInfoIssue {
510            offset: to_u64(start),
511            kind: VersionIssueKind::BlockOverrun,
512        });
513        end = limit;
514    }
515    let key_start = start.saturating_add(BLOCK_HEADER);
516    let key_area = data.get(key_start..end).unwrap_or_default();
517    let key = utf16_until_nul(key_area);
518    if !key.terminated {
519        return Err(truncated);
520    }
521    let value_start = align4(key_start.saturating_add(key.consumed))
522        .unwrap_or(usize::MAX)
523        .min(end);
524    Ok(Some(Block {
525        start,
526        end,
527        value_len: u16_at(header, 2).unwrap_or(0),
528        w_type: u16_at(header, 4).unwrap_or(0),
529        key: key.text,
530        value_start,
531    }))
532}

534/// Iterates the child blocks of `parent` starting at `first`, calling `each`
535/// for every readable child. Stops at the first unreadable child.
536fn for_each_child(
537    data: &[u8],
538    first: usize,
539    parent_end: usize,
540    issues: &mut Vec<VersionInfoIssue>,
541    mut each: impl FnMut(&Block, &mut Vec<VersionInfoIssue>),
542) {
543    let mut pos = first;
544    for _ in 0..MAX_CHILDREN {
545        if pos.saturating_add(BLOCK_HEADER) > parent_end {
546            return;
547        }
548        match read_block(data, pos, parent_end, issues) {
549            Ok(Some(child)) => {
550                each(&child, issues);
551                // `read_block` guarantees `end > start`, so this advances.
552                pos = match align4(child.end) {
553                    Some(next) if next > pos => next,
554                    _ => return,
555                };
556            }
557            Ok(None) => return,
558            Err(issue) => {
559                issues.push(issue);
560                return;
561            }
562        }
563    }
564    issues.push(VersionInfoIssue {
565        offset: to_u64(pos),
566        kind: VersionIssueKind::TooManyChildren,
567    });
568}

570/// Parses a `VS_VERSIONINFO` resource.
571///
572/// Fails only when the root block is unreadable or its key is not
573/// `VS_VERSION_INFO`. Other problems are listed in
574/// [`VersionInfo::issues`], and whatever could be read is returned.
575pub fn parse_version_info(data: &[u8]) -> Result<VersionInfo, VersionInfoIssue> {
576    let mut issues = Vec::new();
577    let root = read_block(data, 0, data.len(), &mut issues)?.ok_or(VersionInfoIssue {
578        offset: 0,
579        kind: VersionIssueKind::Truncated,
580    })?;
581    // Block keys compare ASCII case-insensitively, as the OS version API
582    // does.
583    if !root.key.eq_ignore_ascii_case(ROOT_KEY) {
584        return Err(VersionInfoIssue {
585            offset: 0,
586            kind: VersionIssueKind::BadRootKey(root.key),
587        });
588    }
589
590    let value_bytes = root.value_bytes();
591    let fixed = if value_bytes == 0 {
592        None
593    } else {
594        let parsed = parse_fixed(root.value(data, value_bytes), root.value_start);
595        if parsed.is_none() {
596            issues.push(VersionInfoIssue {
597                offset: to_u64(root.value_start),
598                kind: VersionIssueKind::BadFixedInfo,
599            });
600        }
601        parsed
602    };
603
604    // Pass 1: find the first `StringFileInfo` and `VarFileInfo`. The OS
605    // version API resolves a path one segment at a time and takes the first
606    // child with a matching key, so later ones are unreachable; they are
607    // reported and ignored rather than merged.
608    let mut string_file_info = None;
609    let mut var_file_info = None;
610    for_each_child(
611        data,
612        root.children_start(),
613        root.end,
614        &mut issues,
615        |child, issues| {
616            let slot = if child.key.eq_ignore_ascii_case("StringFileInfo") {
617                &mut string_file_info
618            } else if child.key.eq_ignore_ascii_case("VarFileInfo") {
619                &mut var_file_info
620            } else {
621                return;
622            };
623            if slot.is_some() {
624                issues.push(duplicate(child));
625            } else {
626                *slot = Some(child.clone());
627            }
628        },
629    );
630
631    // Every `Translation` pair and every table key, so that selection is not
632    // affected by the caps on what is reported.
633    let all_translations = var_file_info
634        .map(|b| translation_pairs(data, &b, &mut issues))
635        .unwrap_or_default();
636    let tables = string_file_info
637        .map(|b| string_tables(data, &b, &mut issues))
638        .unwrap_or_default();
639    let keys: Vec<Option<Translation>> = tables.iter().map(|t| parse_table_key(&t.key)).collect();
640    let (winner, selection) = select_language(&keys, &all_translations);
641
642    // Pass 2: decode the kept tables, at most `MAX_STRING_TABLES`. If the
643    // winner lies past the cap it takes the last slot, so the selected table
644    // is always kept and the bound always holds.
645    let winner_past_cap = winner.is_some_and(|w| w >= MAX_STRING_TABLES);
646    let leading = if winner_past_cap {
647        MAX_STRING_TABLES - 1
648    } else {
649        MAX_STRING_TABLES
650    };
651    let keep = |index: usize| index < leading || winner == Some(index);
652    let mut all_langs = Vec::new();
653    let mut selected = None;
654    for (index, (block, translation)) in tables.iter().zip(&keys).enumerate() {
655        if !keep(index) {
656            continue;
657        }
658        if winner == Some(index) {
659            selected = Some(all_langs.len());
660        }
661        all_langs.push(decode_table(data, block, *translation, &mut issues));
662    }
663    // Report the first table actually dropped, if any.
664    if let Some(first_dropped) = (0..tables.len())
665        .find(|i| !keep(*i))
666        .and_then(|i| tables.get(i))
667    {
668        issues.push(VersionInfoIssue {
669            offset: to_u64(first_dropped.start),
670            kind: VersionIssueKind::TooManyStringTables,
671        });
672    }
673    let translations = all_translations
674        .into_iter()
675        .take(MAX_TRANSLATIONS)
676        .collect();
677
678    Ok(VersionInfo {
679        fixed,
680        all_langs,
681        translations,
682        selected,
683        selection,
684        issues,
685    })
686}

688fn parse_fixed(value: &[u8], offset: usize) -> Option<FixedFileInfo> {
689    if value.len() < FIXED_INFO_SIZE || u32_at(value, 0)? != FIXED_SIGNATURE {
690        return None;
691    }
692    let f = |i: usize| u32_at(value, i.saturating_mul(4)).unwrap_or(0);
693    Some(FixedFileInfo {
694        offset: to_u64(offset),
695        struct_version: f(1),
696        file_version: FileVersionQuad::from_ms_ls(f(2), f(3)),
697        product_version: FileVersionQuad::from_ms_ls(f(4), f(5)),
698        file_flags_mask: f(6),
699        file_flags: f(7),
700        file_os: f(8),
701        file_type: f(9),
702        file_subtype: f(10),
703        file_date: (u64::from(f(11)) << 32) | u64::from(f(12)),
704    })
705}

707/// The `StringTable` blocks of a `StringFileInfo`, in file order. Bounded
708/// by the block size (at most 64 KiB) and the child limit.
709fn string_tables(data: &[u8], block: &Block, issues: &mut Vec<VersionInfoIssue>) -> Vec<Block> {
710    let mut tables = Vec::new();
711    for_each_child(
712        data,
713        block.children_start(),
714        block.end,
715        issues,
716        |table, _| tables.push(table.clone()),
717    );
718    tables
719}
720
721/// Decodes one `StringTable`.
722fn decode_table(
723    data: &[u8],
724    table: &Block,
725    translation: Option<Translation>,
726    issues: &mut Vec<VersionInfoIssue>,
727) -> StringTable {
728    if translation.is_none() {
729        issues.push(VersionInfoIssue {
730            offset: to_u64(table.start),
731            kind: VersionIssueKind::BadStringTableKey(table.key.clone()),
732        });
733    }
734    let mut strings = Vec::new();
735    let mut overflow = false;
736    for_each_child(data, table.children_start(), table.end, issues, |s, _| {
737        if strings.len() >= MAX_STRINGS_PER_TABLE {
738            overflow = true;
739            return;
740        }
741        // A `String` value's length is in UTF-16 code units, whatever
742        // `wType` says. The value is read within that length, cut to the
743        // block end (which also covers linkers that store a byte count), and
744        // then up to the first NUL.
745        let declared = usize::from(s.value_len).saturating_mul(2);
746        let value = utf16_until_nul(s.value(data, declared)).text;
747        strings.push((s.key.clone(), value));
748    });
749    if overflow {
750        issues.push(VersionInfoIssue {
751            offset: to_u64(table.start),
752            kind: VersionIssueKind::TooManyStrings,
753        });
754    }
755    StringTable {
756        offset: to_u64(table.start),
757        key: table.key.clone(),
758        translation,
759        strings,
760    }
761}

763/// Every pair of the first `Translation` in a `VarFileInfo`, in file order.
764/// Bounded by the block size (at most 64 KiB, so 16,384 pairs).
765fn translation_pairs(
766    data: &[u8],
767    block: &Block,
768    issues: &mut Vec<VersionInfoIssue>,
769) -> Vec<Translation> {
770    let mut pairs = Vec::new();
771    let mut seen = false;
772    for_each_child(
773        data,
774        block.children_start(),
775        block.end,
776        issues,
777        |var, issues| {
778            if !var.key.eq_ignore_ascii_case("Translation") {
779                return;
780            }
781            // Only the first `Translation` is reachable, as for the blocks
782            // above.
783            if seen {
784                issues.push(duplicate(var));
785                return;
786            }
787            seen = true;
788            let value = var.value(data, var.value_bytes());
789            for pair in value.as_chunks::<4>().0 {
790                let (Some(language), Some(code_page)) = (u16_at(pair, 0), u16_at(pair, 2)) else {
791                    break;
792                };
793                pairs.push(Translation {
794                    language,
795                    code_page,
796                });
797            }
798        },
799    );
800    pairs
801}
802
803fn duplicate(block: &Block) -> VersionInfoIssue {
804    VersionInfoIssue {
805        offset: to_u64(block.start),
806        kind: VersionIssueKind::DuplicateBlock(block.key.clone()),
807    }
808}

810/// Decodes a `StringTable` key such as `"040904B0"`.
811fn parse_table_key(key: &str) -> Option<Translation> {
812    if key.len() != 8 || !key.bytes().all(|b| b.is_ascii_hexdigit()) {
813        return None;
814    }
815    let language = u16::from_str_radix(key.get(0..4)?, 16).ok()?;
816    let code_page = u16::from_str_radix(key.get(4..8)?, 16).ok()?;
817    Some(Translation {
818        language,
819        code_page,
820    })
821}

823/// Picks the language block: the first `Translation` pair, in order, that
824/// has a matching block (the first such block), then US English Unicode,
825/// then the first block. `keys` holds each block's decoded key.
826fn select_language(
827    keys: &[Option<Translation>],
828    translations: &[Translation],
829) -> (Option<usize>, Option<LanguageSelection>) {
830    // The first block for each key, so each lookup is logarithmic. Both
831    // lists are bounded by the 64 KiB resource, and a linear search per
832    // pair would make selection quadratic.
833    let mut first_index = BTreeMap::new();
834    for (index, key) in keys.iter().enumerate() {
835        if let Some(key) = key {
836            first_index.entry(*key).or_insert(index);
837        }
838    }
839    let find = |want: Translation| first_index.get(&want).copied();
840    if let Some(i) = translations.iter().find_map(|t| find(*t)) {
841        return (Some(i), Some(LanguageSelection::Translation));
842    }
843    let english = Translation {
844        language: LANG_EN_US,
845        code_page: CODE_PAGE_UNICODE,
846    };
847    if let Some(i) = find(english) {
848        return (Some(i), Some(LanguageSelection::EnglishUnicode));
849    }
850    if keys.is_empty() {
851        (None, None)
852    } else {
853        (Some(0), Some(LanguageSelection::First))
854    }
855}
