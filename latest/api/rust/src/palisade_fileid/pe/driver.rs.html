1//! Kernel driver detection (#45).
2//!
3//! No single field marks a kernel-mode driver, so [`classify`] combines the
4//! signals the loader and driver tooling leave. An image is a kernel
5//! driver when its subsystem is `NATIVE` **and** it has a kernel signal:
6//! it imports a kernel-mode module, it has an `INIT` section, or it sets
7//! `IMAGE_FILE_SYSTEM`. Native user-mode programs (session and boot-time
8//! tools) are also `NATIVE`, but they import only `ntdll.dll`, so the
9//! subsystem alone is not enough. A non-native image is never classed as
10//! a driver, whatever it imports.
11//!
12//! The reasons are returned too, so a report can say why.
13
14#![forbid(unsafe_code)]
15
16use super::Section;
17use super::imports::ImportInfo;
18use super::types::{FileCharacteristics, Subsystem};
19
20/// Modules only kernel-mode code imports. Compared ASCII
21/// case-insensitively.
22pub const KERNEL_MODULES: [&str; 7] = [
23    "ntoskrnl.exe",
24    "hal.dll",
25    "fltmgr.sys",
26    "wdfldr.sys",
27    "ndis.sys",
28    "ksecdd.sys",
29    "cng.sys",
30];
31
32/// One reason an image is (or looks like) a kernel driver.
33#[derive(Debug, Clone, PartialEq, Eq)]
34#[non_exhaustive]
35pub enum DriverSignal {
36    /// The subsystem is `NATIVE`.
37    NativeSubsystem,
38    /// It imports this kernel-mode module (lowercase).
39    ImportsKernelModule(String),
40    /// It has an `INIT` section (discarded after driver initialisation).
41    InitSection,
42    /// `IMAGE_FILE_SYSTEM` is set.
43    SystemFile,
44}
45
46/// The kernel driver verdict ([`super::PeInfo::driver`]).
47#[derive(Debug, Clone, Default, PartialEq, Eq)]
48pub struct DriverClass {
49    /// Whether the image is a kernel-mode driver.
50    pub is_kernel_driver: bool,
51    /// Every signal found, in a fixed order.
52    pub signals: Vec<DriverSignal>,
53}
54
55/// Classifies an image from its subsystem, characteristics, sections and
56/// imports. `imports` is `None` when the image has no import table.
57#[must_use]
58pub fn classify(
59    subsystem: Subsystem,
60    characteristics: FileCharacteristics,
61    sections: &[Section],
62    imports: Option<&ImportInfo>,
63) -> DriverClass {
64    let mut signals = Vec::new();
65    let native = subsystem == Subsystem::Native;
66    if native {
67        signals.push(DriverSignal::NativeSubsystem);
68    }
69    if let Some(imports) = imports {
70        for module in KERNEL_MODULES {
71            if imports.imports_module(module) {
72                signals.push(DriverSignal::ImportsKernelModule(module.to_owned()));
73            }
74        }
75    }
76    if sections.iter().any(|s| s.raw_name.starts_with(b"INIT\0")) {
77        signals.push(DriverSignal::InitSection);
78    }
79    if characteristics.is_system() {
80        signals.push(DriverSignal::SystemFile);
81    }
82    let kernel_signal = signals
83        .iter()
84        .any(|s| !matches!(s, DriverSignal::NativeSubsystem));
85    DriverClass {
86        is_kernel_driver: native && kernel_signal,
87        signals,
88    }
89}
