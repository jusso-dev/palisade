1//! The application manifest (`RT_MANIFEST`, #45): requested execution
2//! level, UI access, auto-elevation, DPI awareness, assembly identity and
3//! supported OS GUIDs.
4//!
5//! The manifest is untrusted XML. [`parse_manifest`] uses `quick-xml`, a
6//! streaming parser that never expands entities defined in a DTD: only the
7//! five predefined entities and character references are resolved, and
8//! anything else is an error. A document with a `<!DOCTYPE>` is rejected
9//! outright, so no entity definition is ever read. Input size, nesting
10//! depth, element count and string lengths are bounded
11//! ([`MAX_MANIFEST_SIZE`], [`MAX_DEPTH`], [`MAX_ELEMENTS`],
12//! [`MAX_VALUE_LEN`]).
13//!
14//! Each setting is read only at its place in the manifest schema, by local
15//! name (which namespace a name is in is not checked, but every prefix used
16//! must be declared, so an undeclared prefix makes the manifest malformed):
17//!
18//! | Setting | Path |
19//! |---|---|
20//! | `requestedExecutionLevel` | `assembly/trustInfo/security/requestedPrivileges/requestedExecutionLevel` |
21//! | `assemblyIdentity` (the manifest's own) | `assembly/assemblyIdentity` |
22//! | `supportedOS` | `assembly/compatibility/application/supportedOS` |
23//! | `autoElevate`, `dpiAware`, `dpiAwareness` | `assembly/application/windowsSettings/<name>` |
24//!
25//! So a dependency's `assemblyIdentity`, or a decoy setting placed anywhere
26//! else, is never taken for the real one. The first occurrence at the right
27//! place wins, whether it is written `<x/>` or `<x></x>`. Element and
28//! attribute names are case-sensitive, as in XML; only unprefixed
29//! attributes are read, as the manifest schema has no prefixed ones. Values
30//! are trimmed, then compared ASCII case-insensitively where the OS does
31//! (`requireAdministrator`, `true`). Text may be split by references,
32//! comments or CDATA sections; it is joined before it is read, and leading
33//! whitespace is dropped as it arrives, so padding cannot crowd a value
34//! out.
35//!
36//! Only XML 1.0 in UTF-8 is read (with at most one byte order mark); XML
37//! 1.1 and UTF-16 manifests are errors. The whole document must be well formed: one root element, every element
38//! closed, every element and attribute name a valid XML name, every
39//! attribute well formed, unique and free of undefined entities, even on
40//! elements whose settings are not read.
41//!
42//! [`requires_elevation`] is the #45 heuristic for whether starting the
43//! file will ask for administrator rights.
44
45#![forbid(unsafe_code)]

54/// Manifests larger than this are not parsed.
55pub const MAX_MANIFEST_SIZE: usize = 256 * 1024;
56/// The deepest element nesting accepted.
57pub const MAX_DEPTH: usize = 32;
58/// At most this many elements are read.
59pub const MAX_ELEMENTS: usize = 4096;
60/// Attribute values and text longer than this many bytes are cut off.
61pub const MAX_VALUE_LEN: usize = 1024;
62/// At most this many `supportedOS` GUIDs are kept.
63pub const MAX_SUPPORTED_OS: usize = 64;

65/// The `requestedExecutionLevel` `level`.
66#[derive(Debug, Clone, PartialEq, Eq, Hash)]
67#[non_exhaustive]
68pub enum ExecutionLevel {
69    /// `asInvoker`: runs with the caller's token.
70    AsInvoker,
71    /// `highestAvailable`: elevates if the user can.
72    HighestAvailable,
73    /// `requireAdministrator`: always asks for administrator rights.
74    RequireAdministrator,
75    /// Any other value, as written.
76    Other(String),
77}
78
79impl ExecutionLevel {
80    fn parse(value: &str) -> Self {
81        if value.eq_ignore_ascii_case("asInvoker") {
82            Self::AsInvoker
83        } else if value.eq_ignore_ascii_case("highestAvailable") {
84            Self::HighestAvailable
85        } else if value.eq_ignore_ascii_case("requireAdministrator") {
86            Self::RequireAdministrator
87        } else {
88            Self::Other(value.to_owned())
89        }
90    }
91}

137/// Why a manifest could not be parsed.
138#[derive(Debug, Clone, PartialEq, Eq)]
139#[non_exhaustive]
140pub enum ManifestError {
141    /// Larger than [`MAX_MANIFEST_SIZE`].
142    TooLarge(usize),
143    /// Not valid UTF-8, or UTF-16, which is not supported.
144    NotUtf8,
145    /// It has a `<!DOCTYPE>`, which manifests never need and which could
146    /// define entities.
147    DocType,
148    /// Nested deeper than [`MAX_DEPTH`].
149    TooDeep,
150    /// More than [`MAX_ELEMENTS`] elements. A manifest's settings can come
151    /// after any number of others, so a cut-off manifest is not read at
152    /// all rather than read in part.
153    TooManyElements,
154    /// The XML is malformed, or uses an undefined entity.
155    Xml(String),
156}
157
158impl fmt::Display for ManifestError {
159    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
160        match self {
161            Self::TooLarge(n) => write!(f, "manifest of {n} bytes is over the limit"),
162            Self::NotUtf8 => f.write_str("manifest is not UTF-8"),
163            Self::DocType => f.write_str("manifest has a DOCTYPE"),
164            Self::TooDeep => f.write_str("manifest is nested too deeply"),
165            Self::TooManyElements => f.write_str("manifest has too many elements"),
166            Self::Xml(e) => write!(f, "manifest is not well-formed XML: {e}"),
167        }
168    }
169}

173/// XML 1.0 `S`: space, tab, carriage return and line feed only. Unicode
174/// whitespace such as U+00A0 is ordinary text in XML.
175const fn is_xml_space(c: char) -> bool {
176    matches!(c, ' ' | '\t' | '\r' | '\n')
177}
178
179/// `s` without leading and trailing XML whitespace.
180fn xml_trim(s: &str) -> &str {
181    s.trim_matches(is_xml_space)
182}

184/// `s` cut to at most [`MAX_VALUE_LEN`] bytes on a character boundary, and
185/// whether anything was cut off. Not trimmed again afterwards: a cut value
186/// is never read as a keyword, whatever its prefix looks like.
187fn cut(s: &str) -> (String, bool) {
188    if s.len() <= MAX_VALUE_LEN {
189        return (s.to_owned(), false);
190    }
191    let mut end = MAX_VALUE_LEN;
192    while !s.is_char_boundary(end) {
193        end -= 1;
194    }
195    (s.get(..end).unwrap_or_default().to_owned(), true)
196}

198/// A trimmed attribute or text value, and whether it was cut to
199/// [`MAX_VALUE_LEN`].
200struct Value {
201    text: String,
202    cut: bool,
203}

205impl Value {
206    fn new(s: &str) -> Self {
207        let (text, cut) = cut(xml_trim(s));
208        Self { text, cut }
209    }

211    /// The value as a boolean; a cut value is never one.
212    fn as_bool(&self) -> Option<bool> {
213        if self.cut {
214            None
215        } else {
216            parse_bool(&self.text)
217        }
218    }

221/// Whether `name` is a qualified name as XML Namespaces defines it: an
222/// `NCName`, or `NCName:NCName`. So no leading, trailing or second colon.
223fn is_xml_name(name: &[u8]) -> bool {
224    let Ok(name) = std::str::from_utf8(name) else {
225        return false;
226    };
227    let mut parts = name.split(':');
228    match (parts.next(), parts.next(), parts.next()) {
229        (Some(local), None, _) => is_ncname(local),
230        (Some(prefix), Some(local), None) => is_ncname(prefix) && is_ncname(local),
231        _ => false,
232    }
233}

235/// Whether `name` is an XML `NCName` (a name without colons), by the XML
236/// 1.0 fifth edition character classes.
237fn is_ncname(name: &str) -> bool {
238    let start = |c: char| {
239        c.is_ascii_alphabetic()
240            || c == '_'
241            || matches!(c,
242                '\u{C0}'..='\u{D6}' | '\u{D8}'..='\u{F6}' | '\u{F8}'..='\u{2FF}'
243                | '\u{370}'..='\u{37D}' | '\u{37F}'..='\u{1FFF}' | '\u{200C}'..='\u{200D}'
244                | '\u{2070}'..='\u{218F}' | '\u{2C00}'..='\u{2FEF}' | '\u{3001}'..='\u{D7FF}'
245                | '\u{F900}'..='\u{FDCF}' | '\u{FDF0}'..='\u{FFFD}' | '\u{10000}'..='\u{EFFFF}')
246    };
247    let rest = |c: char| {
248        start(c)
249            || c.is_ascii_digit()
250            || c == '-'
251            || c == '.'
252            || matches!(c, '\u{B7}' | '\u{300}'..='\u{36F}' | '\u{203F}'..='\u{2040}')
253    };
254    let mut chars = name.chars();
255    chars.next().is_some_and(start) && chars.all(rest)
256}

258/// Checks the inside of a start tag (`name attr="v" ...`, without `<`,
259/// `>` or a trailing `/`) against the XML 1.0 grammar: a valid name, then
260/// attributes each preceded by whitespace, as `Name Eq AttValue`, with no
261/// `<` in a value. The attribute parser in use is lenient about these, so
262/// they are checked here first.
263fn check_tag(raw: &[u8]) -> Result<(), ManifestError> {
264    let tag = std::str::from_utf8(raw).map_err(|_| ManifestError::NotUtf8)?;
265    let bad = |what: &str| Err(ManifestError::Xml(format!("malformed start tag: {what}")));
266    let name_end = tag.find(is_xml_space).unwrap_or(tag.len());
267    let (name, mut rest) = tag.split_at(name_end);
268    if !is_xml_name(name.as_bytes()) {
269        return bad("invalid element name");
270    }
271    loop {
272        let trimmed = rest.trim_start_matches(is_xml_space);
273        if trimmed.is_empty() {
274            return Ok(());
275        }
276        if trimmed.len() == rest.len() {
277            return bad("no whitespace before an attribute");
278        }
279        let eq = trimmed.find('=').unwrap_or(trimmed.len());
280        let attr = trimmed
281            .get(..eq)
282            .unwrap_or_default()
283            .trim_end_matches(is_xml_space);
284        if !is_xml_name(attr.as_bytes()) {
285            return bad("invalid attribute name");
286        }
287        let after = trimmed.get(eq..).unwrap_or_default();
288        let Some(after) = after.strip_prefix('=') else {
289            return bad("attribute without a value");
290        };
291        let after = after.trim_start_matches(is_xml_space);
292        let Some(quote) = after.chars().next().filter(|q| matches!(q, '"' | '\'')) else {
293            return bad("unquoted attribute value");
294        };
295        let body = after.get(1..).unwrap_or_default();
296        let Some(close) = body.find(quote) else {
297            return bad("unterminated attribute value");
298        };
299        let value = body.get(..close).unwrap_or_default();
300        if value.contains('<') {
301            return bad("`<` in an attribute value");
302        }
303        check_chars(value)?;
304        rest = body.get(close + 1..).unwrap_or_default();
305    }
306}

308/// Checks an XML declaration's content (`xml version="1.0" ...`) against
309/// the grammar: `version`, then optionally `encoding`, then optionally
310/// `standalone`, in that order, each once, with valid values.
311///
312/// Only XML 1.0 is accepted. XML 1.1 reads some characters differently
313/// (NEL and LS are line ends, so `"\u{85}asInvoker"` would be `asInvoker`),
314/// and reading a manifest differently from the OS could make a file look
315/// less likely to elevate than it is, so an XML 1.1 manifest is an error.
316fn check_decl(raw: &[u8]) -> Result<(), ManifestError> {
317    let decl = std::str::from_utf8(raw).map_err(|_| ManifestError::NotUtf8)?;
318    let bad = || Err(ManifestError::Xml("malformed XML declaration".into()));
319    let Some(body) = decl.strip_prefix("xml") else {
320        return bad();
321    };
322    check_tag(format!("x{body}").as_bytes())?;
323    let mut order = Vec::new();
324    let mut rest = body;
325    loop {
326        rest = rest.trim_start_matches(is_xml_space);
327        if rest.is_empty() {
328            break;
329        }
330        let eq = rest.find('=').unwrap_or(rest.len());
331        let name = rest
332            .get(..eq)
333            .unwrap_or_default()
334            .trim_end_matches(is_xml_space);
335        let after = rest
336            .get(eq + 1..)
337            .unwrap_or_default()
338            .trim_start_matches(is_xml_space);
339        let quote = after.chars().next().unwrap_or('"');
340        let body = after.get(1..).unwrap_or_default();
341        let close = body.find(quote).unwrap_or(body.len());
342        let value = body.get(..close).unwrap_or_default();
343        let ok = match name {
344            "version" => {
345                if order.is_empty() && value == "1.1" {
346                    return Err(ManifestError::Xml("XML 1.1 is not supported".into()));
347                }
348                order.is_empty() && value == "1.0"
349            }
350            "encoding" => {
351                order == ["version"]
352                    && (value.eq_ignore_ascii_case("UTF-8") || value.eq_ignore_ascii_case("UTF8"))
353            }
354            "standalone" => {
355                matches!(order.as_slice(), ["version"] | ["version", "encoding"])
356                    && matches!(value, "yes" | "no")
357            }
358            _ => false,
359        };
360        if !ok {
361            if name == "encoding" && order == ["version"] {
362                return Err(ManifestError::NotUtf8);
363            }
364            return bad();
365        }
366        order.push(name);
367        rest = body.get(close + 1..).unwrap_or_default();
368    }
369    if order.first() == Some(&"version") {
370        Ok(())
371    } else {
372        bad()
373    }
374}

376/// Whether `c` is an XML 1.0 `Char`.
377const fn is_xml_char(c: char) -> bool {
378    matches!(c, '\t' | '\n' | '\r' | '\u{20}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..='\u{10FFFF}')
379}

381/// Fails if `text` has a character XML 1.0 does not allow.
382fn check_chars(text: &str) -> Result<(), ManifestError> {
383    if text.chars().all(is_xml_char) {
384        Ok(())
385    } else {
386        Err(ManifestError::Xml("character not allowed in XML".into()))
387    }
388}

390/// Every attribute of `e`, checked: each must have a valid name and be
391/// well formed, unique and free of undefined entities, whether or not it
392/// is read. Returns the values of those in `wanted`, matched by exact name
393/// and only when unprefixed, trimmed and bounded, in `wanted` order.
394fn attrs<const N: usize>(
395    e: &BytesStart<'_>,
396    wanted: [&str; N],
397) -> Result<[Option<Value>; N], ManifestError> {
398    check_tag(e.as_bytes())?;
399    let mut out: [Option<Value>; N] = std::array::from_fn(|_| None);
400    for a in e.attributes() {
401        let a = a.map_err(|err| ManifestError::Xml(err.to_string()))?;
402        if !is_xml_name(a.key.as_ref().as_bytes()) {
403            return Err(ManifestError::Xml("invalid attribute name".into()));
404        }
405        // Only the five predefined entities resolve; anything else fails.
406        let value = a
407            .normalized_value(XmlVersion::Implicit1_0)
408            .map_err(|err| ManifestError::Xml(err.to_string()))?;
409        check_chars(&value)?;
410        // A prefixed attribute (`x:level`) is another attribute: never the
411        // schema's `level`.
412        if a.key.prefix().is_some() {
413            continue;
414        }
415        if let Some(slot) = wanted
416            .iter()
417            .position(|w| a.key.as_ref() == *w)
418            .and_then(|i| out.get_mut(i))
419        {
420            *slot = Some(Value::new(&value));
421        }
422    }
423    Ok(out)
424}

426fn parse_bool(value: &str) -> Option<bool> {
427    let v = xml_trim(value);
428    if v.eq_ignore_ascii_case("true") {
429        Some(true)
430    } else if v.eq_ignore_ascii_case("false") {
431        Some(false)
432    } else {
433        None
434    }
435}

445/// The open element path, by local name. Bounded by [`MAX_DEPTH`].
446struct Path(Vec<Vec<u8>>);
447
448impl Path {
449    /// Whether the open elements, from the root, are exactly `names`.
450    fn is(&self, names: &[&str]) -> bool {
451        self.0.len() == names.len() && self.0.iter().zip(names).all(|(a, b)| a == b.as_bytes())
452    }

456    "assembly",
457    "trustInfo",
458    "security",
459    "requestedPrivileges",
460    "requestedExecutionLevel",
461];
462const IDENTITY: [&str; 2] = ["assembly", "assemblyIdentity"];
463const SUPPORTED_OS: [&str; 4] = ["assembly", "compatibility", "application", "supportedOS"];
464const WINDOWS_SETTINGS: [&str; 3] = ["assembly", "application", "windowsSettings"];

466/// A setting's text, gathered across text, references and CDATA.
467///
468/// Leading XML whitespace is dropped. Whitespace after the value is held
469/// back and kept only if more text follows it, so trailing padding is never
470/// stored and never counts towards the cap. Anything that does not fit in
471/// [`MAX_VALUE_LEN`] bytes is dropped and marks the text cut, so a cut is
472/// always detected, wherever a character boundary falls.
473#[derive(Default)]
474struct Text {
475    text: String,
476    /// Whitespace seen after the last non-space character, not yet kept.
477    pending: usize,
478    pending_text: String,
479    cut: bool,
480}
481
482impl Text {
483    fn clear(&mut self) {
484        *self = Self::default();
485    }
486
487    fn push(&mut self, piece: &str) {
488        for c in piece.chars() {
489            if self.cut {
490                return;
491            }
492            if is_xml_space(c) {
493                if !self.text.is_empty() {
494                    self.pending = self.pending.saturating_add(1);
495                    // Held whitespace is bounded like the text: past the
496                    // cap it can only end up cut off anyway.
497                    if self.pending_text.len() <= MAX_VALUE_LEN {
498                        self.pending_text.push(c);
499                    }
500                }
501                continue;
502            }
503            let needed = self
504                .text
505                .len()
506                .saturating_add(self.pending)
507                .saturating_add(c.len_utf8());
508            if needed > MAX_VALUE_LEN {
509                self.cut = true;
510                return;
511            }
512            self.text.push_str(&self.pending_text);
513            self.pending_text.clear();
514            self.pending = 0;
515            self.text.push(c);
516        }
517    }
518
519    fn value(&self) -> Value {
520        Value {
521            text: self.text.clone(),
522            cut: self.cut,
523        }
524    }
525}

527/// Parses a manifest resource.
528pub fn parse_manifest(data: &[u8]) -> Result<Manifest, ManifestError> {
529    if data.len() > MAX_MANIFEST_SIZE {
530        return Err(ManifestError::TooLarge(data.len()));
531    }
532    if data.starts_with(&[0xff, 0xfe]) || data.starts_with(&[0xfe, 0xff]) {
533        return Err(ManifestError::NotUtf8);
534    }
535    // One UTF-8 byte order mark is skipped by the reader itself. A second
536    // one is then text outside the root element, and an error.
537    let text = std::str::from_utf8(data).map_err(|_| ManifestError::NotUtf8)?;
538    let mut reader = Reader::from_str(text);
539    // Text is gathered as written (see `Text`), so whitespace next to a
540    // reference survives.
541    let config = reader.config_mut();
542    config.check_end_names = true;
543    config.check_comments = true;
544
545    let mut out = Manifest::default();
546    let mut path = Path(Vec::new());
547    // The namespace prefixes in scope, in step with `path`.
548    let mut scopes = Scopes::default();
549    let mut elements = 0_usize;
550    let mut roots = 0_usize;
551    // Anything read yet (for the declaration, which must come first).
552    let mut seen_anything = false;
553    let mut seen_execution_level = false;
554    // Which text settings were met, so the first wins however it is
555    // written (`<x/>`, `<x></x>`, or a value that does not parse).
556    let mut seen_text: Vec<TextField> = Vec::with_capacity(3);
557    let mut text_field: Option<TextField> = None;
558    let mut text = Text::default();
559    // Only whitespace may sit outside the root element.
560    let outside = |path: &Path, t: &str| path.0.is_empty() && !t.chars().all(is_xml_space);
561    loop {
562        let event = reader
563            .read_event()
564            .map_err(|e| ManifestError::Xml(e.to_string()))?;
565        match event {
566            Event::DocType(_) => return Err(ManifestError::DocType),
567            Event::Start(ref e) | Event::Empty(ref e) => {
568                let empty = matches!(event, Event::Empty(_));
569                elements += 1;
570                if elements > MAX_ELEMENTS {
571                    return Err(ManifestError::TooManyElements);
572                }
573                if path.0.is_empty() {
574                    roots += 1;
575                    if roots > 1 {
576                        return Err(ManifestError::Xml("more than one root element".into()));
577                    }
578                }
579                if path.0.len() >= MAX_DEPTH {
580                    return Err(ManifestError::TooDeep);
581                }
582                bind_prefixes(e, &mut scopes)?;
583                path.0.push(e.local_name().as_ref().as_bytes().to_vec());
584                element(e, &path, &mut out, &mut seen_execution_level)?;
585                // Text belongs to a setting only directly inside it.
586                text_field = None;
587                if !empty {
588                    let name = path.0.last().map(Vec::as_slice).unwrap_or_default();
589                    let in_settings = path.0.len() == WINDOWS_SETTINGS.len() + 1
590                        && path
591                            .0
592                            .iter()
593                            .zip(WINDOWS_SETTINGS)
594                            .all(|(a, b)| a == b.as_bytes());
595                    text_field = match name {
596                        _ if !in_settings => None,
597                        b"autoElevate" => Some(TextField::AutoElevate),
598                        b"dpiAware" => Some(TextField::DpiAware),
599                        b"dpiAwareness" => Some(TextField::DpiAwareness),
600                        _ => None,
601                    }
602                    .filter(|f| !seen_text.contains(f));
603                    if let Some(f) = text_field {
604                        seen_text.push(f);
605                    }
606                    text.clear();
607                } else {
608                    let name = path.0.last().map(Vec::as_slice).unwrap_or_default();
609                    let in_settings = path.0.len() == WINDOWS_SETTINGS.len() + 1
610                        && path
611                            .0
612                            .iter()
613                            .zip(WINDOWS_SETTINGS)
614                            .all(|(a, b)| a == b.as_bytes());
615                    let field = match name {
616                        _ if !in_settings => None,
617                        b"autoElevate" => Some(TextField::AutoElevate),
618                        b"dpiAware" => Some(TextField::DpiAware),
619                        b"dpiAwareness" => Some(TextField::DpiAwareness),
620                        _ => None,
621                    };
622                    // `<x/>` is the same as `<x></x>`: an empty value.
623                    if let Some(f) = field.filter(|f| !seen_text.contains(f)) {
624                        seen_text.push(f);
625                        match f {
626                            TextField::AutoElevate => out.auto_elevate = None,
627                            TextField::DpiAware => out.dpi_aware = Some(String::new()),
628                            TextField::DpiAwareness => out.dpi_awareness = Some(String::new()),
629                        }
630                    }
631                    path.0.pop();
632                    scopes.pop();
633                }
634            }
635            Event::Text(t) => {
636                let raw = t.xml10_content();
637                check_chars(&raw)?;
638                if raw.contains("]]>") {
639                    return Err(ManifestError::Xml("`]]>` in character data".into()));
640                }
641                if outside(&path, &raw) {
642                    return Err(ManifestError::Xml("text outside the root element".into()));
643                }
644                if text_field.is_some() {
645                    // References arrive as their own events, so the text is
646                    // already plain.
647                    text.push(&raw);
648                }
649            }
650            Event::CData(t) => {
651                let raw = t.xml10_content();
652                check_chars(&raw)?;
653                if path.0.is_empty() {
654                    return Err(ManifestError::Xml("CDATA outside the root element".into()));
655                }
656                if text_field.is_some() {
657                    text.push(&raw);
658                }
659            }
660            Event::GeneralRef(r) => {
661                // Only character references and the five predefined
662                // entities are accepted; anything else is undefined.
663                let resolved = match r.resolve_char_ref() {
664                    Ok(Some(c)) if is_xml_char(c) => c.to_string(),
665                    Ok(Some(_)) => {
666                        return Err(ManifestError::Xml(
667                            "character reference to a character XML does not allow".into(),
668                        ));
669                    }
670                    Ok(None) => {
671                        let name: &str = &r;
672                        quick_xml::escape::resolve_predefined_entity(name)
673                            .ok_or_else(|| {
674                                ManifestError::Xml(format!("undefined entity &{name};"))
675                            })?
676                            .to_owned()
677                    }
678                    Err(e) => return Err(ManifestError::Xml(e.to_string())),
679                };
680                if path.0.is_empty() {
681                    return Err(ManifestError::Xml(
682                        "reference outside the root element".into(),
683                    ));
684                }
685                if text_field.is_some() {
686                    text.push(&resolved);
687                }
688            }
689            Event::End(_) => {
690                path.0.pop();
691                scopes.pop();
692                if let Some(field) = text_field.take() {
693                    let value = text.value();
694                    match field {
695                        TextField::AutoElevate => out.auto_elevate = value.as_bool(),
696                        TextField::DpiAware => out.dpi_aware = Some(value.text),
697                        TextField::DpiAwareness => out.dpi_awareness = Some(value.text),
698                    }
699                }
700            }
701            Event::Eof => {
702                if !path.0.is_empty() {
703                    return Err(ManifestError::Xml("an element is not closed".into()));
704                }
705                if roots == 0 {
706                    return Err(ManifestError::Xml("no root element".into()));
707                }
708                break;
709            }
710            Event::Decl(d) => {
711                // Only as the very first thing, with a valid version.
712                if seen_anything {
713                    return Err(ManifestError::Xml("misplaced XML declaration".into()));
714                }
715                check_decl(d.as_bytes())?;
716            }
717            Event::PI(pi) => {
718                let target = pi.target();
719                if target.eq_ignore_ascii_case("xml")
720                    || !is_xml_name(target.as_bytes())
721                    || target.contains(':')
722                {
723                    return Err(ManifestError::Xml("bad processing instruction".into()));
724                }
725                check_chars(&pi)?;
726            }
727            Event::Comment(c) => {
728                check_chars(&c)?;
729            }
730        }
731        seen_anything = true;
732    }
733    Ok(out)
734}

736/// The namespace prefixes in scope: how many open elements declare each
737/// one, and what each open element declared, so closing it undoes exactly
738/// that. Lookups are logarithmic, however many prefixes are declared.
739#[derive(Default)]
740struct Scopes {
741    bound: BTreeMap<Vec<u8>, usize>,
742    frames: Vec<Vec<Vec<u8>>>,
743}
744
745impl Scopes {
746    fn push(&mut self, declared: Vec<Vec<u8>>) {
747        for prefix in &declared {
748            let count = self.bound.entry(prefix.clone()).or_insert(0);
749            *count = count.saturating_add(1);
750        }
751        self.frames.push(declared);
752    }
753
754    fn pop(&mut self) {
755        for prefix in self.frames.pop().unwrap_or_default() {
756            if let Some(count) = self.bound.get_mut(&prefix) {
757                *count = count.saturating_sub(1);
758                if *count == 0 {
759                    self.bound.remove(&prefix);
760                }
761            }
762        }
763    }
764
765    fn is_bound(&self, prefix: &[u8]) -> bool {
766        self.bound.contains_key(prefix)
767    }
768}
769
770/// Records the namespace prefixes `e` declares (`xmlns:p="..."`) as a new
771/// scope, then checks that its own prefix and its attributes' prefixes are
772/// declared, as XML Namespaces requires. `xml` is always bound and `xmlns`
773/// only declares. A name with an undeclared prefix is not namespace
774/// well formed, so the manifest is an error rather than read by its local
775/// name.
776fn bind_prefixes(e: &BytesStart<'_>, scopes: &mut Scopes) -> Result<(), ManifestError> {
777    let bad = |what: &str| Err(ManifestError::Xml(format!("namespace prefix: {what}")));
778    let mut declared = Vec::new();
779    let mut used = Vec::new();
780    for a in e.attributes() {
781        let a = a.map_err(|err| ManifestError::Xml(err.to_string()))?;
782        let key = a.key.as_ref();
783        if let Some(prefix) = key.strip_prefix("xmlns:") {
784            // Undeclaring a prefix (`xmlns:p=""`) is not allowed in XML
785            // Namespaces 1.0, and `xmlns` itself cannot be declared.
786            if a.value.is_empty() || prefix == "xmlns" {
787                return bad("bad declaration");
788            }
789            declared.push(prefix.as_bytes().to_vec());
790        } else if let Some(prefix) = a.key.prefix() {
791            used.push(prefix.as_ref().as_bytes().to_vec());
792        }
793    }
794    scopes.push(declared);
795    if let Some(prefix) = e.name().prefix() {
796        used.push(prefix.as_ref().as_bytes().to_vec());
797    }
798    for prefix in used {
799        if prefix == b"xmlns" {
800            return bad("`xmlns` used as a prefix");
801        }
802        if prefix != b"xml" && !scopes.is_bound(&prefix) {
803            return bad("undeclared");
804        }
805    }
806    Ok(())
807}

809fn element(
810    e: &BytesStart<'_>,
811    path: &Path,
812    out: &mut Manifest,
813    seen_execution_level: &mut bool,
814) -> Result<(), ManifestError> {
815    if path.is(&EXECUTION_LEVEL) {
816        let [level, ui_access] = attrs(e, ["level", "uiAccess"])?;
817        // The first element decides, even if it lacks one of them.
818        if !*seen_execution_level {
819            *seen_execution_level = true;
820            // A cut level is kept as `Other`, never matched as a keyword.
821            out.execution_level = level.map(|v| {
822                if v.cut {
823                    ExecutionLevel::Other(v.text)
824                } else {
825                    ExecutionLevel::parse(&v.text)
826                }
827            });
828            out.ui_access = ui_access.and_then(|v| v.as_bool());
829        }
830    } else if path.is(&IDENTITY) {
831        let [name, version, arch, token, kind] = attrs(
832            e,
833            [
834                "name",
835                "version",
836                "processorArchitecture",
837                "publicKeyToken",
838                "type",
839            ],
840        )?;
841        if out.identity.is_none() {
842            let text = |v: Option<Value>| v.map(|v| v.text);
843            out.identity = Some(AssemblyIdentity {
844                name: text(name),
845                version: text(version),
846                processor_architecture: text(arch),
847                public_key_token: text(token),
848                kind: text(kind),
849            });
850        }
851    } else if path.is(&SUPPORTED_OS) {
852        let [id] = attrs(e, ["Id"])?;
853        if let Some(id) = id {
854            if out.supported_os.len() < MAX_SUPPORTED_OS {
855                out.supported_os.push(id.text);
856            } else {
857                out.truncated = true;
858            }
859        }
860    } else {
861        // Not read, but still checked.
862        let _ = attrs(e, [])?;
863    }
864    Ok(())
865}

867/// Why [`requires_elevation`] decided as it did.
868#[derive(Debug, Clone, PartialEq, Eq)]
869#[non_exhaustive]
870pub enum ElevationReason {
871    /// The manifest requests `requireAdministrator`.
872    RequireAdministrator,
873    /// The manifest requests `highestAvailable`, which elevates for an
874    /// administrator. Reported, but not counted as requiring elevation.
875    HighestAvailable,
876    /// The manifest requests `asInvoker`, so no installer heuristic
877    /// applies.
878    ManifestAsInvoker,
879    /// The manifest requests a level this parser does not recognise. The
880    /// OS refuses to start such a file, so it is not treated as `asInvoker`;
881    /// the installer heuristics still apply.
882    UnknownLevel(String),
883    /// A manifest is present but requests no execution level. The
884    /// installer heuristics apply only when there is no manifest, so they
885    /// are not used.
886    ManifestWithoutLevel,
887    /// The process manifest (ID 1) is present but could not be read or
888    /// parsed, so what it requests is unknown. Treated as requiring
889    /// elevation, and never decided from another manifest or from the
890    /// installer heuristics.
891    ManifestUnreadable,
892    /// There is no manifest, and the file name contains this installer
893    /// word.
894    InstallerFileName(&'static str),
895    /// There is no manifest, and this version field mentions installing.
896    InstallerVersionInfo(&'static str),
897    /// Nothing suggests elevation.
898    None,
899}
900
901/// The #45 elevation heuristic.
902#[derive(Debug, Clone, PartialEq, Eq)]
903pub struct Elevation {
904    /// Whether starting the file is expected to ask for administrator
905    /// rights.
906    pub requires_elevation: bool,
907    /// Why.
908    pub reason: ElevationReason,
909}
910
911/// The words in a file name that the OS installer detection looks for.
912const INSTALLER_WORDS: [&str; 4] = ["setup", "install", "update", "patch"];
913
914/// Whether starting a file is expected to ask for administrator rights:
915/// true if its manifest requests `requireAdministrator`, or if it has no
916/// manifest (or one with an unrecognised execution level) and the
917/// installer heuristics match (its file name contains setup, install,
918/// update or patch, or its version info mentions installing). A manifest
919/// without an execution level is [`ElevationReason::ManifestWithoutLevel`]
920/// and is not counted. `manifest` is the parsed process manifest, or
921/// `None` when there is none; a process manifest that is present but
922/// cannot be read is decided by [`super::PeInfo::requires_elevation`]
923/// ([`ElevationReason::ManifestUnreadable`]), never passed as `None`. The
924/// OS applies the installer heuristics only to 32-bit images; the caller
925/// decides whether that applies.
926///
927/// `version_fields` are `(field name, value)` pairs, such as
928/// `("FileDescription", "Foo Installer")`.
929#[must_use]
930pub fn requires_elevation(
931    manifest: Option<&Manifest>,
932    file_name: Option<&str>,
933    version_fields: &[(&'static str, &str)],
934) -> Elevation {
935    let Some(manifest) = manifest else {
936        return match installer_heuristic(file_name, version_fields) {
937            Some(reason) => Elevation {
938                requires_elevation: true,
939                reason,
940            },
941            None => Elevation {
942                requires_elevation: false,
943                reason: ElevationReason::None,
944            },
945        };
946    };
947    match &manifest.execution_level {
948        Some(ExecutionLevel::RequireAdministrator) => Elevation {
949            requires_elevation: true,
950            reason: ElevationReason::RequireAdministrator,
951        },
952        Some(ExecutionLevel::HighestAvailable) => Elevation {
953            requires_elevation: false,
954            reason: ElevationReason::HighestAvailable,
955        },
956        Some(ExecutionLevel::AsInvoker) => Elevation {
957            requires_elevation: false,
958            reason: ElevationReason::ManifestAsInvoker,
959        },
960        Some(ExecutionLevel::Other(level)) => {
961            let heuristic = installer_heuristic(file_name, version_fields);
962            Elevation {
963                requires_elevation: heuristic.is_some(),
964                reason: heuristic.unwrap_or(ElevationReason::UnknownLevel(level.clone())),
965            }
966        }
967        None => Elevation {
968            requires_elevation: false,
969            reason: ElevationReason::ManifestWithoutLevel,
970        },
971    }
972}
973
974/// The installer heuristics: a file name with an installer word, or a
975/// version field mentioning installing.
976fn installer_heuristic(
977    file_name: Option<&str>,
978    version_fields: &[(&'static str, &str)],
979) -> Option<ElevationReason> {
980    if let Some(name) = file_name {
981        let lower = name.to_lowercase();
982        if let Some(word) = INSTALLER_WORDS.iter().find(|w| lower.contains(*w)) {
983            return Some(ElevationReason::InstallerFileName(word));
984        }
985    }
986    version_fields
987        .iter()
988        .find(|(_, value)| value.to_lowercase().contains("install"))
989        .map(|(field, _)| ElevationReason::InstallerVersionInfo(field))
990}
