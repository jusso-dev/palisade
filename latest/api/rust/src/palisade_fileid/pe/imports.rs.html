1//! The import and export tables (#45).
2//!
3//! Both are read the way the OS loader reads them. The import descriptor
4//! table ends at the first descriptor with no name or no import address
5//! table, whatever the directory size says, and an export is looked up by
6//! binary search over the name table, as `GetProcAddress` does. Tables and
7//! names are read from mapped memory: they may run on into the next
8//! section when the two are adjacent, a section's zero-filled tail past
9//! its raw data reads as zeros, and unmapped memory, the end of a
10//! truncated file, or a read spread over more than [`MAX_READ_EXTENTS`]
11//! extents ends the read as unreadable. No name read looks at more
12//! than a name's cap, so the work stays linear in the number of entries.
13//! The lists kept here are capped
14//! ([`MAX_IMPORT_MODULES`], [`MAX_IMPORTED_FUNCTIONS`], [`MAX_EXPORTS`]).
15//! Questions that must not be dodged by padding a table (which modules are
16//! imported at all, does an export exist) are answered over the whole
17//! table, within [`MAX_IMPORT_DESCRIPTORS`], not over the capped lists.
18//!
19//! Whenever a read is ambiguous, cut or unreadable, the result is marked
20//! (`truncated`, and a warning for a malformed table) rather than read as
21//! an absence, so it never makes a file look more benign than it is.
22
23#![forbid(unsafe_code)]

31/// At most this many imported modules are listed.
32pub const MAX_IMPORT_MODULES: usize = 4096;
33/// At most this many imported functions are listed, across all modules.
34pub const MAX_IMPORTED_FUNCTIONS: usize = 10_000;
35/// At most this many import descriptors are read. Module names past the
36/// listed ones still count for [`ImportInfo::imports_module`].
37pub const MAX_IMPORT_DESCRIPTORS: usize = 65_536;
38/// At most this many exports are listed.
39pub const MAX_EXPORTS: usize = 10_000;
40/// At most this many export address table slots are scanned for unnamed
41/// exports. Export ordinals are 16-bit, so real tables are smaller.
42pub const MAX_EXPORT_SLOTS: u32 = 65_536;
43/// At most this many ordinal table entries are scanned to learn which
44/// export slots have a name. With more names than this, which slots are
45/// unnamed is unknown, so none is listed as unnamed and the result is
46/// [`ExportInfo::truncated`].
47pub const MAX_ORDINAL_SCAN: u32 = 131_072;
48/// Module, function and forwarder names are kept up to this many bytes;
49/// longer ones are cut (and reported as cut). Comparisons, such as the
50/// export lookup, always use the whole name.
51pub const MAX_NAME_LEN: usize = 512;
52/// At most this many bytes of module names are kept in
53/// [`ImportInfo::all_module_names`]. Kernel modules are always recorded,
54/// in [`ImportInfo::kernel_modules`], however many names come first.
55pub const MAX_MODULE_NAME_BYTES: usize = 1024 * 1024;
56
57/// A single read of a name, descriptor or thunk crosses at most this many
58/// mapped extents (runs of file-backed bytes or zero fill), so each read
59/// costs at most this many section lookups. Real images need two (into a
60/// section's zero fill, or into the next section). A read that needs more
61/// is unreadable, and is reported as such, never taken as complete.
62pub const MAX_READ_EXTENTS: usize = 4;

64const DESCRIPTOR_SIZE: usize = 20;
65const EXPORT_DIRECTORY_SIZE: usize = 40;

81/// One imported module.
82#[derive(Debug, Clone, PartialEq, Eq)]
83pub struct ImportedModule {
84    /// The module name as written (`KERNEL32.dll`).
85    pub name: String,
86    /// The name was longer than [`MAX_NAME_LEN`], so [`Self::name`] is only
87    /// its start. A cut name never matches in [`ImportInfo::imports_module`].
88    pub name_cut: bool,
89    /// The name is not valid UTF-8, so [`Self::name`] is a lossy rendering
90    /// for display (with U+FFFD in place of bad bytes). Which module the
91    /// loader resolves it to depends on the system code page, so a lossy
92    /// name never matches in [`ImportInfo::imports_module`] and sets
93    /// [`ImportInfo::truncated`].
94    pub name_lossy: bool,
95    /// The functions imported from it that were listed.
96    pub functions: Vec<ImportedFunction>,
97    /// How many functions were counted for it. Once the table-wide
98    /// [`MAX_IMPORTED_FUNCTIONS`] is reached nothing more is read, so this
99    /// is then a lower bound ([`ImportInfo::truncated`]).
100    pub function_count: usize,
101}

103/// The import table.
104#[derive(Debug, Clone, Default, PartialEq, Eq)]
105pub struct ImportInfo {
106    /// The listed modules, in table order.
107    pub modules: Vec<ImportedModule>,
108    /// Every complete module name in the table (within
109    /// [`MAX_IMPORT_DESCRIPTORS`] and [`MAX_MODULE_NAME_BYTES`]),
110    /// ASCII-lowercased, in table order with duplicates kept once. A name cut
111    /// at [`MAX_NAME_LEN`], or not valid UTF-8, is not kept. Past either
112    /// limit [`Self::truncated`] is set.
113    pub all_module_names: Vec<String>,
114    /// The [`super::KERNEL_MODULES`] imported anywhere in the table (within
115    /// [`MAX_IMPORT_DESCRIPTORS`]), lowercase, in that list's order. This is
116    /// never cut short by the caps, so padding a table cannot hide a kernel
117    /// import.
118    pub kernel_modules: Vec<&'static str>,
119    /// Imported functions across all listed modules.
120    pub function_count: usize,
121    /// A limit was reached, or part of the table was unreadable, so the
122    /// lists are incomplete.
123    pub truncated: bool,
124}
125
126impl ImportInfo {
127    /// Whether an import descriptor names `module` (ASCII case-insensitive,
128    /// as the loader compares module names), matching whole names only.
129    /// A kernel module is found anywhere in the table. Any other module is
130    /// found among the listed modules and the names kept in
131    /// [`Self::all_module_names`], so `false` for one is reliable only when
132    /// [`Self::truncated`] is not set: past [`MAX_IMPORT_DESCRIPTORS`], the
133    /// [`MAX_MODULE_NAME_BYTES`] budget, for a name cut at
134    /// [`MAX_NAME_LEN`], for a name that is not valid UTF-8, or for a table
135    /// that could not be read to its end, a module can be imported and
136    /// still not be found.
137    #[must_use]
138    pub fn imports_module(&self, module: &str) -> bool {
139        let wanted = module.to_ascii_lowercase();
140        self.kernel_modules.contains(&wanted.as_str())
141            || self.all_module_names.contains(&wanted)
142            || self
143                .modules
144                .iter()
145                .any(|m| !m.name_cut && !m.name_lossy && m.name.eq_ignore_ascii_case(&wanted))
146    }
147}

219/// A NUL-terminated name at `rva`, as the loader reads it. The text kept is
220/// at most [`MAX_NAME_LEN`] bytes; a longer name is marked cut, and no more
221/// of it is read. `None` if the RVA is zero or unmapped, or the name runs
222/// into unmapped memory or the end of the file. An empty name is valid.
223fn name_at(data: &[u8], layout: Layout<'_>, rva: u32) -> Option<Name> {
224    let (raw, terminated) = c_string(data, layout, rva, MAX_NAME_LEN)?;
225    let (text, lossy) = match String::from_utf8_lossy(&raw) {
226        Cow::Borrowed(text) => (text.to_owned(), false),
227        Cow::Owned(text) => (text, true),
228    };
229    Some(Name {
230        text,
231        cut: !terminated,
232        lossy,
233    })
234}

236/// Up to `max` bytes of the NUL-terminated string at `rva`, and whether its
237/// NUL was found within them. At most `max + 1` bytes are looked at. The
238/// string is read from mapped memory: past the end of one section's
239/// file-backed bytes it goes on into the next section if that is mapped
240/// right after, and a section's zero-filled tail (mapped, but past its
241/// declared raw data) reads as NUL, as it does in memory, including at the
242/// start (an empty string). `None` if the RVA is zero or unmapped, or the
243/// string runs into unmapped memory, or into the end of the input inside
244/// the declared raw data (a truncated file), before `max` bytes or a NUL.
245fn c_string<'a>(
246    data: &'a [u8],
247    layout: Layout<'_>,
248    rva: u32,
249    max: usize,
250) -> Option<(Cow<'a, [u8]>, bool)> {
251    if rva == 0 {
252        return None;
253    }
254    let limit = max.saturating_add(1);
255    let mut out: Cow<'a, [u8]> = Cow::Borrowed(&[]);
256    let mut at = rva;
257    // Each pass reads at least one byte, and at most `limit` in all, over
258    // at most `MAX_READ_EXTENTS` extents.
259    for _ in 0..MAX_READ_EXTENTS {
260        let (offset, available) = match layout.extent(at) {
261            Extent::Backed(offset, available) => (offset, available),
262            Extent::Zero(_) => return Some((out, true)),
263            Extent::Unmapped => return None,
264        };
265        let len = to_usize(available)?.min(limit.saturating_sub(out.len()));
266        let raw = bytes(data, offset, len)?;
267        let end = raw.iter().position(|b| *b == 0);
268        let piece = raw.get(..end.unwrap_or(raw.len()))?;
269        if out.is_empty() {
270            out = Cow::Borrowed(piece);
271        } else {
272            out.to_mut().extend_from_slice(piece);
273        }
274        if end.is_some() {
275            return Some((out, true));
276        }
277        if out.len() > max {
278            match &mut out {
279                Cow::Borrowed(b) => *b = b.get(..max)?,
280                Cow::Owned(v) => v.truncate(max),
281            }
282            return Some((out, false));
283        }
284        if len == 0 {
285            return None;
286        }
287        at = at.checked_add(u32::try_from(len).ok()?)?;
288    }
289    // Spread over more extents than a real image uses: unreadable.
290    None
291}

302/// One warning per table, however many entries are bad.
303struct Once {
304    at: usize,
305    kind: WarningKind,
306    done: bool,
307}
308
309impl Once {
310    const fn new(at: usize, kind: WarningKind) -> Self {
311        Self {
312            at,
313            kind,
314            done: false,
315        }
316    }
317
318    fn report(&mut self, warnings: &mut Warnings) {
319        if !self.done {
320            self.done = true;
321            warnings.push(self.at, self.kind.clone());
322        }
323    }
324}

326/// `N` bytes of mapped memory at `rva`, as the loader lays it out:
327/// file-backed bytes, and zeros where they are in a section's zero-filled
328/// tail. `None` if any byte is unmapped, or past the end of a truncated
329/// input inside the declared raw data (which is never taken for zero fill),
330/// or if the bytes are spread over more than [`MAX_READ_EXTENTS`] extents.
331fn mapped<const N: usize>(data: &[u8], layout: Layout<'_>, rva: u32) -> Option<[u8; N]> {
332    let mut out = [0_u8; N];
333    let mut done = 0;
334    for _ in 0..MAX_READ_EXTENTS {
335        if done >= N {
336            break;
337        }
338        let at = rva.checked_add(u32::try_from(done).ok()?)?;
339        match layout.extent(at) {
340            Extent::Backed(offset, available) => {
341                let n = to_usize(available)?.min(N - done);
342                out.get_mut(done..done + n)?
343                    .copy_from_slice(bytes(data, offset, n)?);
344                done += n;
345            }
346            // `out` is already zero there.
347            Extent::Zero(len) => done += to_usize(len)?.min(N - done),
348            Extent::Unmapped => return None,
349        }
350    }
351    (done >= N).then_some(out)
352}

354/// `len` file-backed bytes at `rva`.
355fn at_rva<'a>(data: &'a [u8], layout: Layout<'_>, rva: u32, len: usize) -> Option<&'a [u8]> {
356    if rva == 0 {
357        return None;
358    }
359    let offset = layout.rva_to_offset(rva, u32::try_from(len).ok()?)?;
360    bytes(data, offset, len)
361}

363pub(super) fn parse_imports(
364    data: &[u8],
365    layout: Layout<'_>,
366    dir: DataDirectory,
367    format: PeFormat,
368    warnings: &mut Warnings,
369) -> Option<ImportInfo> {
370    let entry_offset = usize::try_from(dir.entry_offset).unwrap_or(usize::MAX);
371    let Some((start, _)) = (dir.virtual_address != 0)
372        .then(|| layout.backed_extent(dir.virtual_address))
373        .flatten()
374    else {
375        warnings.push(
376            entry_offset,
377            WarningKind::DirectoryNotMapped(DataDirectoryKind::Import),
378        );
379        return None;
380    };
381    let mut info = ImportInfo::default();
382    let mut bad = Once::new(start, WarningKind::ImportTableMalformed);
383    let mut seen = BTreeSet::new();
384    let mut name_bytes = 0_usize;
385    let mut kernel = [false; super::KERNEL_MODULES.len()];
386    let thunk_size = if format == PeFormat::Pe32Plus { 8 } else { 4 };
387    let mut terminated = false;
388    for i in 0..MAX_IMPORT_DESCRIPTORS {
389        // The table is read from mapped memory, as the loader reads it: it
390        // may go on into an adjacent section, and a section's zero-filled
391        // tail reads as zeros, so a terminator there ends the table. An
392        // unmapped descriptor, or the end of a truncated input, ends the
393        // read without a terminator.
394        let d = u32::try_from(i.saturating_mul(DESCRIPTOR_SIZE))
395            .ok()
396            .and_then(|at| dir.virtual_address.checked_add(at))
397            .and_then(|rva| mapped::<DESCRIPTOR_SIZE>(data, layout, rva));
398        let Some(d) = d else {
399            break;
400        };
401        let field = |o| u32_at(&d, o).unwrap_or(0);
402        // The loader stops at a descriptor with no name or no import
403        // address table.
404        if field(12) == 0 || field(16) == 0 {
405            terminated = true;
406            break;
407        }
408        let Some(name) = name_at(data, layout, field(12)) else {
409            bad.report(warnings);
410            info.truncated = true;
411            continue;
412        };
413        if name.cut {
414            bad.report(warnings);
415            info.truncated = true;
416        }
417        // A lossy name is kept for display only: the module it names is
418        // not known exactly.
419        if name.lossy {
420            info.truncated = true;
421        }
422        let exact = !name.cut && !name.lossy;
423        let lower = name.text.to_ascii_lowercase();
424        for (found, module) in kernel.iter_mut().zip(super::KERNEL_MODULES) {
425            *found |= exact && lower == module;
426        }
427        // A cut name is only a prefix: kept for display, never matched.
428        if exact && !seen.contains(&lower) {
429            if name_bytes.saturating_add(lower.len()) > MAX_MODULE_NAME_BYTES {
430                info.truncated = true;
431            } else {
432                name_bytes = name_bytes.saturating_add(lower.len());
433                seen.insert(lower.clone());
434                info.all_module_names.push(lower);
435            }
436        }
437        if info.modules.len() >= MAX_IMPORT_MODULES {
438            info.truncated = true;
439            continue;
440        }
441        let thunks = match field(0) {
442            0 => field(16),
443            rva => rva,
444        };
445        let mut module = ImportedModule {
446            name: name.text,
447            name_cut: name.cut,
448            name_lossy: name.lossy,
449            functions: Vec::new(),
450            function_count: 0,
451        };
452        if !read_thunks(data, layout, thunks, thunk_size, &mut module, &mut info) {
453            bad.report(warnings);
454        }
455        info.modules.push(module);
456    }
457    if !terminated {
458        warnings.push(start, WarningKind::ImportTableTruncated);
459        info.truncated = true;
460    }
461    info.kernel_modules = super::KERNEL_MODULES
462        .iter()
463        .zip(kernel)
464        .filter_map(|(m, found)| found.then_some(*m))
465        .collect();
466    Some(info)
467}

469/// Reads one module's thunk array. Returns `false` when part of it was
470/// malformed (as opposed to merely over the cap).
471fn read_thunks(
472    data: &[u8],
473    layout: Layout<'_>,
474    rva: u32,
475    thunk_size: usize,
476    module: &mut ImportedModule,
477    info: &mut ImportInfo,
478) -> bool {
479    if rva == 0 {
480        info.truncated = true;
481        return false;
482    }
483    // The array is read from mapped memory, as the loader reads it: it may
484    // go on into an adjacent section, and a section's zero-filled tail
485    // reads as a null thunk. An unmapped slot, or the end of a truncated
486    // input, ends it without one. Every slot read either ends the list or
487    // counts towards [`MAX_IMPORTED_FUNCTIONS`], so the loop is bounded.
488    for i in 0_usize.. {
489        let at = u32::try_from(i.saturating_mul(thunk_size))
490            .ok()
491            .and_then(|at| rva.checked_add(at));
492        let raw = if thunk_size == 8 {
493            at.and_then(|at| mapped::<8>(data, layout, at))
494                .map(u64::from_le_bytes)
495        } else {
496            at.and_then(|at| mapped::<4>(data, layout, at))
497                .map(|b| u64::from(u32::from_le_bytes(b)))
498        };
499        let Some(v) = raw else {
500            break;
501        };
502        let (value, by_ordinal) = if thunk_size == 8 {
503            (v & 0x7fff_ffff_ffff_ffff, v >> 63 == 1)
504        } else {
505            (v & 0x7fff_ffff, v >> 31 == 1)
506        };
507        if value == 0 && !by_ordinal {
508            return true;
509        }
510        // Checked after the terminator, so exactly the cap is not "over".
511        if info.function_count >= MAX_IMPORTED_FUNCTIONS {
512            info.truncated = true;
513            return true;
514        }
515        let function = if by_ordinal {
516            ImportedFunction::Ordinal(u16::try_from(value & 0xffff).unwrap_or(0))
517        } else {
518            // An unreadable entry ends the module's list: continuing past
519            // it would let a crafted table make every entry cost a read
520            // without ever counting towards the cap.
521            let hint = u32::try_from(value)
522                .ok()
523                .and_then(|r| Some((r, u16_at(at_rva(data, layout, r, 2)?, 0)?)));
524            let name = hint
525                .and_then(|(r, _)| r.checked_add(2))
526                .and_then(|r| name_at(data, layout, r));
527            match (hint, name) {
528                (Some((_, hint)), Some(name)) => {
529                    if name.cut {
530                        info.truncated = true;
531                    }
532                    ImportedFunction::Name {
533                        name: name.text,
534                        hint,
535                    }
536                }
537                _ => {
538                    info.truncated = true;
539                    return false;
540                }
541            }
542        };
543        module.functions.push(function);
544        module.function_count = module.function_count.saturating_add(1);
545        info.function_count = info.function_count.saturating_add(1);
546    }
547    // No terminator within the mapped data.
548    info.truncated = true;
549    false
550}

552pub(super) fn parse_exports(
553    data: &[u8],
554    layout: Layout<'_>,
555    dir: DataDirectory,
556    warnings: &mut Warnings,
557) -> Option<ExportInfo> {
558    let entry_offset = usize::try_from(dir.entry_offset).unwrap_or(usize::MAX);
559    let Some(header) = at_rva(data, layout, dir.virtual_address, EXPORT_DIRECTORY_SIZE) else {
560        warnings.push(
561            entry_offset,
562            WarningKind::DirectoryNotMapped(DataDirectoryKind::Export),
563        );
564        return None;
565    };
566    let field = |o| u32_at(header, o).unwrap_or(0);
567    let base = field(16);
568    let mut info = ExportInfo {
569        dll_name: None,
570        number_of_functions: field(20),
571        number_of_names: field(24),
572        ..ExportInfo::default()
573    };
574    let functions = field(28);
575    let names = field(32);
576    let ordinals = field(36);
577    let dir_start = u64::from(dir.virtual_address);
578    let dir_end = dir_start.saturating_add(u64::from(dir.size));
579    let mut bad = Once::new(
580        layout
581            .rva_to_offset(dir.virtual_address, 0)
582            .unwrap_or(entry_offset),
583        WarningKind::ExportTableMalformed,
584    );
585    let number_of_functions = info.number_of_functions;
586    match (field(12), name_at(data, layout, field(12))) {
587        (0, _) => {}
588        (_, Some(name)) => {
589            if name.cut {
590                info.truncated = true;
591            }
592            info.dll_name = Some(name.text);
593        }
594        (_, None) => {
595            bad.report(warnings);
596            info.truncated = true;
597        }
598    }
599
600    // Like the loader, an index past `NumberOfFunctions`, or a slot holding
601    // RVA 0, is not an export.
602    // A table at RVA 0 is absent: adding an index to it would read the
603    // headers as the table.
604    let slot_at = |index: u32| -> Slot {
605        if index >= number_of_functions || functions == 0 {
606            return Slot::Bad;
607        }
608        let Some(rva) = index
609            .checked_mul(4)
610            .and_then(|o| functions.checked_add(o))
611            .and_then(|at| at_rva(data, layout, at, 4))
612            .and_then(|b| u32_at(b, 0))
613        else {
614            return Slot::Bad;
615        };
616        if rva == 0 {
617            Slot::Unused
618        } else if (dir_start..dir_end).contains(&u64::from(rva)) {
619            // The loader takes any RVA inside the directory as a forwarder
620            // and reads its string wherever it ends, so one whose string
621            // runs past the directory is still a forwarder, but the table
622            // is malformed.
623            let inside = dir_end.saturating_sub(u64::from(rva));
624            match c_string(data, layout, rva, MAX_NAME_LEN) {
625                Some((raw, terminated)) => {
626                    // The string and its NUL, which comes after at least
627                    // the bytes kept (and one more if it was cut).
628                    let needed = u64::try_from(raw.len())
629                        .unwrap_or(u64::MAX)
630                        .saturating_add(if terminated { 1 } else { 2 });
631                    let flaw = if needed > inside {
632                        Flaw::PastDirectory
633                    } else if terminated {
634                        Flaw::None
635                    } else {
636                        Flaw::Cut
637                    };
638                    let text = String::from_utf8_lossy(&raw).into_owned();
639                    Slot::Export(ExportTarget::Forwarder(text), flaw)
640                }
641                None => Slot::Bad,
642            }
643        } else {
644            Slot::Export(ExportTarget::Rva(rva), Flaw::None)
645        }
646    };
647    let name_rva_at = |i: u32| -> Option<u32> {
648        let names = (names != 0).then_some(names)?;
649        u32_at(
650            at_rva(data, layout, names.checked_add(i.checked_mul(4)?)?, 4)?,
651            0,
652        )
653    };
654    let ordinal_at = |i: u32| -> Option<u16> {
655        let ordinals = (ordinals != 0).then_some(ordinals)?;
656        u16_at(
657            at_rva(data, layout, ordinals.checked_add(i.checked_mul(2)?)?, 2)?,
658            0,
659        )
660    };
661
662    // Named exports, in name-table order.
663    // Not more slots than the export address table has file-backed bytes for.
664    let backed_slots = (functions != 0)
665        .then(|| layout.backed_extent(functions))
666        .flatten()
667        .map_or(0, |(_, n)| n / 4);
668    let slots = info
669        .number_of_functions
670        .min(MAX_EXPORT_SLOTS)
671        .min(backed_slots);
672    // A table the file does not hold in full is malformed, whether or not
673    // any entry points into the missing part, and however far past the
674    // scan caps it runs: the export address table (4 bytes per function),
675    // and the name pointer and ordinal tables (4 and 2 bytes per name).
676    let backed = |rva: u32, count: u32, size: u32| {
677        count == 0
678            || (rva != 0
679                && layout
680                    .backed_extent(rva)
681                    .is_some_and(|(_, n)| u64::from(n) >= u64::from(count) * u64::from(size)))
682    };
683    if !backed(functions, info.number_of_functions, 4)
684        || !backed(names, info.number_of_names, 4)
685        || !backed(ordinals, info.number_of_names, 2)
686    {
687        bad.report(warnings);
688        info.truncated = true;
689    }
690    // Which slots some name points to, from the whole ordinal table, not
691    // only the listed names: a slot named past the listing cap, or by a
692    // name that cannot be read, is not an unnamed export. If the file does
693    // not hold the whole ordinal table, which slots are unnamed is unknown,
694    // so none are listed as unnamed.
695    let mut named = vec![false; to_usize(slots).unwrap_or(0)];
696    let ordinal_table = (ordinals != 0)
697        .then(|| layout.backed_extent(ordinals))
698        .flatten();
699    let known_names = ordinal_table
700        .map_or(0, |(_, n)| n / 2)
701        .min(info.number_of_names);
702    // The scan is bounded: with more names than [`MAX_ORDINAL_SCAN`], which
703    // slots are unnamed is not worked out, so none is listed as unnamed and
704    // the listing is marked incomplete.
705    let list_unnamed = known_names == info.number_of_names && known_names <= MAX_ORDINAL_SCAN;
706    if known_names > MAX_ORDINAL_SCAN {
707        info.truncated = true;
708    }
709    // Only needed when the unnamed slots are listed at all.
710    if let Some((offset, _)) = ordinal_table.filter(|_| list_unnamed) {
711        for i in 0..known_names {
712            let index = to_usize(i)
713                .and_then(|i| offset.checked_add(i.checked_mul(2)?))
714                .and_then(|at| u16_at(data, at));
715            if let Some(slot) = index.and_then(|index| named.get_mut(usize::from(index))) {
716                *slot = true;
717            }
718        }
719    }
720    let listed_names = info
721        .number_of_names
722        .min(u32::try_from(MAX_EXPORTS).unwrap_or(u32::MAX));
723    for i in 0..listed_names {
724        // A bad entry is skipped, not the end of the listing: one crafted
725        // name must not hide the rest.
726        let (Some(name), Some(index)) = (
727            name_rva_at(i).and_then(|r| name_at(data, layout, r)),
728            ordinal_at(i),
729        ) else {
730            bad.report(warnings);
731            info.truncated = true;
732            continue;
733        };
734        if name.cut {
735            info.truncated = true;
736        }
737        let Slot::Export(target, flaw) = slot_at(u32::from(index)) else {
738            bad.report(warnings);
739            info.truncated = true;
740            continue;
741        };
742        // `Base + index` must fit: a wrapped ordinal would name another
743        // export.
744        let Some(ordinal) = base.checked_add(u32::from(index)) else {
745            bad.report(warnings);
746            info.truncated = true;
747            continue;
748        };
749        flaw.record(&mut info.truncated, &mut bad, warnings);
750        info.exports.push(Export {
751            name: Some(name.text),
752            ordinal,
753            target,
754        });
755    }
756    if info.number_of_names > listed_names {
757        info.truncated = true;
758    }
759    if info.number_of_functions > slots {
760        info.truncated = true;
761    }
762    // Unnamed exports, by ordinal.
763    let unnamed_slots = if list_unnamed { slots } else { 0 };
764    for index in 0..unnamed_slots {
765        if to_usize(index)
766            .and_then(|i| named.get(i))
767            .copied()
768            .unwrap_or(false)
769        {
770            continue;
771        }
772        match slot_at(index) {
773            Slot::Unused => {}
774            Slot::Export(target, flaw) => {
775                // Checked only when there is one to add, so exactly the cap
776                // is not "over".
777                if info.exports.len() >= MAX_EXPORTS {
778                    info.truncated = true;
779                    break;
780                }
781                let Some(ordinal) = base.checked_add(index) else {
782                    bad.report(warnings);
783                    info.truncated = true;
784                    continue;
785                };
786                flaw.record(&mut info.truncated, &mut bad, warnings);
787                info.exports.push(Export {
788                    name: None,
789                    ordinal,
790                    target,
791                });
792            }
793            // One bad slot does not hide the rest.
794            Slot::Bad => {
795                bad.report(warnings);
796                info.truncated = true;
797            }
798        }
799    }
800
801    let number_of_names = info.number_of_names;
802    let find = |wanted: &str| -> Lookup {
803        // The loader's binary search over the whole name table: inclusive
804        // bounds and midpoint `(low + high) / 2`, comparing whole
805        // NUL-terminated names byte by byte (`strcmp`). The name found must
806        // also map to a real export.
807        let Some(mut high) = number_of_names.checked_sub(1) else {
808            return Lookup::Absent;
809        };
810        let mut low = 0_u32;
811        while low <= high {
812            let mid = low + (high - low) / 2;
813            // `strcmp` against `wanted` is decided within `wanted.len() + 1`
814            // bytes, so no more are read. A name that cannot be read leaves
815            // the answer unknown, never "absent".
816            let Some((name, terminated)) =
817                name_rva_at(mid).and_then(|r| c_string(data, layout, r, wanted.len()))
818            else {
819                return Lookup::Unknown;
820            };
821            // Unterminated within `wanted.len()` bytes means the name is
822            // longer: equal so far, so it sorts after `wanted`.
823            let order = match name.as_ref().cmp(wanted.as_bytes()) {
824                std::cmp::Ordering::Equal if !terminated => std::cmp::Ordering::Greater,
825                other => other,
826            };
827            match order {
828                std::cmp::Ordering::Equal => {
829                    return match ordinal_at(mid).map(|i| slot_at(u32::from(i))) {
830                        Some(Slot::Export(_, flaw)) => Lookup::Found(flaw),
831                        Some(Slot::Unused) => Lookup::Absent,
832                        Some(Slot::Bad) | None => Lookup::Unknown,
833                    };
834                }
835                std::cmp::Ordering::Less => low = mid + 1,
836                std::cmp::Ordering::Greater => match mid.checked_sub(1) {
837                    Some(h) => high = h,
838                    None => return Lookup::Absent,
839                },
840            }
841        }
842        Lookup::Absent
843    };
844    let lookups = [
845        find("CPlApplet"),
846        find("DllRegisterServer"),
847        find("DllGetClassObject"),
848        find("ServiceMain"),
849    ];
850    // A flaw met by the search is reported as the listing reports it, even
851    // when the export is past the listing cap.
852    for lookup in lookups {
853        match lookup {
854            Lookup::Found(flaw) => flaw.record(&mut info.truncated, &mut bad, warnings),
855            Lookup::Unknown => {
856                bad.report(warnings);
857                info.truncated = true;
858            }
859            Lookup::Absent => {}
860        }
861    }
862    let [
863        cpl_applet,
864        dll_register_server,
865        dll_get_class_object,
866        service_main,
867    ] = lookups.map(|l| matches!(l, Lookup::Found(_)));
868    info.notable = NotableExports {
869        cpl_applet,
870        dll_register_server,
871        dll_get_class_object,
872        service_main,
873    };
874    Some(info)
875}
876
877/// What the notable-export search found.
878#[derive(Clone, Copy)]
879enum Lookup {
880    /// The name maps to an export, with this forwarder flaw.
881    Found(Flaw),
882    /// The name is not in the table, or its slot holds RVA 0.
883    Absent,
884    /// A name or entry on the search path cannot be read, so whether the
885    /// export exists is not known.
886    Unknown,
887}
888
889/// One slot of the export address table.
890enum Slot {
891    /// RVA 0: no export at this ordinal.
892    Unused,
893    /// An export, and what is wrong with its forwarder string, if anything.
894    Export(ExportTarget, Flaw),
895    /// Past the table (or a table at RVA 0), unreadable, or a forwarder
896    /// that cannot be read.
897    Bad,
898}
899
900/// What is wrong with an export's forwarder string.
901#[derive(Clone, Copy)]
902enum Flaw {
903    None,
904    /// Longer than [`MAX_NAME_LEN`]; the text kept is its start.
905    Cut,
906    /// It ends past the export directory, where the format says it must
907    /// not.
908    PastDirectory,
909}
910
911impl Flaw {
912    fn record(self, truncated: &mut bool, bad: &mut Once, warnings: &mut Warnings) {
913        match self {
914            Self::None => {}
915            Self::Cut => *truncated = true,
916            Self::PastDirectory => {
917                *truncated = true;
918                bad.report(warnings);
919            }
920        }
921    }
922}
