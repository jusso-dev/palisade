1//! Bounds-checked little-endian reads from untrusted bytes.
2//!
3//! Every helper returns `None` instead of panicking when the requested range
4//! does not fit, and all offset arithmetic is checked.
5
6#![forbid(unsafe_code)]
7
8/// `len` bytes at `offset`, or `None` if the range does not fit in `data`.
9pub(crate) fn bytes(data: &[u8], offset: usize, len: usize) -> Option<&[u8]> {
10    data.get(offset..offset.checked_add(len)?)
11}
12
13fn array<const N: usize>(data: &[u8], offset: usize) -> Option<[u8; N]> {
14    bytes(data, offset, N)?.try_into().ok()
15}
16
17pub(crate) fn u8_at(data: &[u8], offset: usize) -> Option<u8> {
18    data.get(offset).copied()
19}
20
21pub(crate) fn u16_at(data: &[u8], offset: usize) -> Option<u16> {
22    array(data, offset).map(u16::from_le_bytes)
23}
24
25pub(crate) fn u32_at(data: &[u8], offset: usize) -> Option<u32> {
26    array(data, offset).map(u32::from_le_bytes)
27}
28
29pub(crate) fn u64_at(data: &[u8], offset: usize) -> Option<u64> {
30    array(data, offset).map(u64::from_le_bytes)
31}
32
33/// A `u32` from the file as a `usize`. Always succeeds on the 32- and 64-bit
34/// targets Palisade supports.
35pub(crate) fn to_usize(value: u32) -> Option<usize> {
36    usize::try_from(value).ok()
37}
38
39/// A `usize` offset as a `u64`, for reporting.
40pub(crate) fn to_u64(value: usize) -> u64 {
41    u64::try_from(value).unwrap_or(u64::MAX)
42}
43
44/// Rounds `value` up to a multiple of four.
45pub(crate) fn align4(value: usize) -> Option<usize> {
46    Some(value.checked_add(3)? & !3)
47}

49/// UTF-16LE text read up to a NUL unit.
50pub(crate) struct Utf16Text {
51    /// The text. Unpaired surrogates are replaced with U+FFFD.
52    pub text: String,
53    /// Bytes consumed, including the NUL unit if one was found.
54    pub consumed: usize,
55    /// Whether a NUL terminator was found before the end of the input.
56    pub terminated: bool,
57}
58
59/// Reads UTF-16LE code units from `data` up to the first NUL unit or the end
60/// of `data`. An odd trailing byte is ignored. Allocates only for the units
61/// before the terminator.
62pub(crate) fn utf16_until_nul(data: &[u8]) -> Utf16Text {
63    let pairs = data.as_chunks::<2>().0;
64    let available = pairs.len();
65    let len = pairs
66        .iter()
67        .position(|p| u16::from_le_bytes(*p) == 0)
68        .unwrap_or(available);
69    let terminated = len < available;
70    let units: Vec<u16> = pairs
71        .iter()
72        .take(len)
73        .map(|p| u16::from_le_bytes(*p))
74        .collect();
75    Utf16Text {
76        text: String::from_utf16_lossy(&units),
77        consumed: len
78            .saturating_add(usize::from(terminated))
79            .saturating_mul(2),
80        terminated,
81    }
82}

84#[cfg(test)]
85mod tests {
86    use super::*;
87
88    #[test]
89    fn reads_are_bounded() {
90        let data = [1_u8, 2, 3, 4, 5, 6, 7, 8];
91        assert_eq!(u8_at(&data, 7), Some(8));
92        assert_eq!(u8_at(&data, 8), None);
93        assert_eq!(u16_at(&data, 0), Some(0x0201));
94        assert_eq!(u32_at(&data, 4), Some(0x0807_0605));
95        assert_eq!(u64_at(&data, 0), Some(0x0807_0605_0403_0201));
96        assert_eq!(u16_at(&data, 7), None);
97        assert_eq!(u32_at(&data, usize::MAX), None);
98        assert_eq!(bytes(&data, usize::MAX, 2), None);
99        assert_eq!(bytes(&data, 8, 0), Some(&[][..]));
100    }
101
102    #[test]
103    fn alignment() {
104        assert_eq!(align4(0), Some(0));
105        assert_eq!(align4(1), Some(4));
106        assert_eq!(align4(4), Some(4));
107        assert_eq!(align4(usize::MAX), None);
108    }
109
110    #[test]
111    fn utf16_reading() {
112        let text: Vec<u8> = "ab\0c".encode_utf16().flat_map(u16::to_le_bytes).collect();
113        let full = utf16_until_nul(&text);
114        assert_eq!(
115            (full.text.as_str(), full.consumed, full.terminated),
116            ("ab", 6, true)
117        );
118        let cut = utf16_until_nul(text.get(..4).unwrap());
119        assert_eq!(
120            (cut.text.as_str(), cut.consumed, cut.terminated),
121            ("ab", 4, false)
122        );
123        // An odd trailing byte is ignored, a lone surrogate is replaced.
124        let odd = utf16_until_nul(&[0x00, 0xD8, 0x41]);
125        assert_eq!((odd.text.as_str(), odd.consumed), ("\u{FFFD}", 2));
126    }
