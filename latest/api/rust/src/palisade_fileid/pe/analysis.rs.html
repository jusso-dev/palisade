5/// Computes the PE image checksum with the standard algorithm.
6///
7/// The file is summed as little-endian 16-bit words with end-around carry,
8/// with the 4-byte `CheckSum` field at `checksum_offset` read as zeros. An
9/// odd final byte is treated as if followed by a zero byte. The 16-bit sum
10/// is then added to the file length, wrapping at 32 bits as the OS's
11/// `DWORD` arithmetic does.
12///
13/// Returns `None` when the file length does not fit in 32 bits, because the
14/// algorithm is undefined there.
15#[must_use]
16pub fn compute_checksum(data: &[u8], checksum_offset: usize) -> Option<u32> {
17    let file_len = u32::try_from(data.len()).ok()?;
18    let skip = checksum_offset..checksum_offset.saturating_add(4);
19    let byte = |offset: usize, value: u8| {
20        if skip.contains(&offset) {
21            0
22        } else {
23            u16::from(value)
24        }
25    };
26    let mut sum: u32 = 0;
27    for (index, pair) in data.chunks(2).enumerate() {
28        let offset = index.saturating_mul(2);
29        let word = match pair {
30            [lo, hi] => byte(offset, *lo) | (byte(offset.saturating_add(1), *hi) << 8),
31            [lo] => byte(offset, *lo),
32            _ => 0,
33        };
34        // Fold the carry back in so the sum never exceeds 17 bits.
35        sum += u32::from(word);
36        sum = (sum & 0xffff) + (sum >> 16);
37    }
38    sum = (sum & 0xffff) + (sum >> 16);
39    Some((sum & 0xffff).wrapping_add(file_len))
40}

42/// Shannon entropy of `data` in bits per byte, from 0.0 (one repeated value)
43/// to 8.0 (uniformly random). Returns 0.0 for empty input.
44#[must_use]
45pub fn shannon_entropy(data: &[u8]) -> f64 {
46    if data.is_empty() {
47        return 0.0;
48    }
49    let mut counts = [0_u64; 256];
50    for byte in data {
51        if let Some(slot) = counts.get_mut(usize::from(*byte)) {
52            *slot = slot.saturating_add(1);
53        }
54    }
55    // Counts above 2^53 would lose precision as f64, which is irrelevant for
56    // an advisory entropy figure.
57    let total = data.len() as f64;
58    counts
59        .iter()
60        .filter(|c| **c != 0)
61        .map(|c| {
62            let p = *c as f64 / total;
63            -p * p.log2()
64        })
65        .sum()
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn checksum_of_simple_inputs() {
74        // Words 0x0201 + 0x0403 = 0x0604, plus length 4.
75        assert_eq!(compute_checksum(&[1, 2, 3, 4], 100), Some(0x0604 + 4));
76        // The checksum field is skipped.
77        assert_eq!(
78            compute_checksum(&[1, 2, 3, 4, 9, 9, 9, 9], 4),
79            Some(0x0604 + 8)
80        );
81        // A checksum field at an odd offset zeroes the bytes it covers.
82        assert_eq!(
83            compute_checksum(&[1, 2, 3, 4, 5, 6], 1),
84            Some(0x0001 + 0x0600 + 6)
85        );
86        // An odd trailing byte counts as a low byte.
87        assert_eq!(compute_checksum(&[1, 2, 7], 100), Some(0x0201 + 7 + 3));
88        // Carries fold back in: 0xffff + 0x0001 = 0x1_0000 -> 0x0001.
89        assert_eq!(compute_checksum(&[0xff, 0xff, 1, 0], 100), Some(1 + 4));
90        assert_eq!(compute_checksum(&[], 0), Some(0));
91    }
92
93    #[test]
94    fn entropy_bounds() {
95        assert!(shannon_entropy(&[]).abs() < f64::EPSILON);
96        assert!(shannon_entropy(&[7; 100]).abs() < f64::EPSILON);
97        let all: Vec<u8> = (0..=255).collect();
98        assert!((shannon_entropy(&all) - 8.0).abs() < 1e-9);
99        assert!((shannon_entropy(&[0, 1]) - 1.0).abs() < 1e-9);
100    }
