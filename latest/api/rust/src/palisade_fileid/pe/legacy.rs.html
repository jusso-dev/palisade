1//! Identity of 16-bit NE and 32-bit LE / LX executables (#45).
2//!
3//! These older `MZ` formats have no version resource that the OS reads, but
4//! they name themselves. [`parse`] reads:
5//!
6//! - the **module name**, the first entry of the resident name table;
7//! - the **module description**, the first entry of the non-resident name
8//!   table;
9//! - the format and target operating system, which are the minimal identity
10//!   returned when the names cannot be read.
11//!
12//! Each name table entry is a length byte, then that many bytes of
13//! single-byte text, then an ordinal. A length byte of zero ends the table,
14//! so an empty table means the name is absent.
15//!
16//! The input is untrusted, and the parser fails conservative:
17//!
18//! - Every read is bounds-checked, and all offset arithmetic is checked. A
19//!   name is at most 255 bytes, because its length is one byte, so nothing
20//!   is allocated from a size the file declares.
21//! - A name is reported only if it lies wholly inside the file and, where
22//!   the header gives the table's length, inside the table. Otherwise it is
23//!   absent, with a [`LegacyWarning`].
24//! - The text is in a code page that the file does not name. Printable
25//!   ASCII is kept as it is. Any other byte becomes U+FFFD and
26//!   [`LegacyName::non_ascii`] is set, so a rule can never match a guessed
27//!   decoding. The exact bytes are kept in [`LegacyName::raw`].
28//! - An LE or LX header that declares big-endian byte or word order is
29//!   identified, but its names are not read.
30//!
31//! [`parse`] fails with a [`LegacyError`] only when the file is not one of
32//! these formats.
33

39/// Where `e_lfanew` is in the DOS header.
40const E_LFANEW_AT: usize = 0x3c;
41/// The DOS header is this long.
42const DOS_HEADER_LEN: usize = 0x40;
43
44/// NE header fields, relative to the header.
45mod ne {
46    /// The fields read end here.
47    pub const LEN: usize = 0x40;
48    pub const NON_RESIDENT_SIZE: usize = 0x20;
49    pub const RESIDENT_NAMES: usize = 0x26;
50    /// An absolute file offset.
51    pub const NON_RESIDENT_NAMES: usize = 0x2c;
52    pub const TARGET_OS: usize = 0x36;
53}
54
55/// LE and LX header fields, relative to the header.
56mod lx {
57    /// The fields read end here.
58    pub const LEN: usize = 0x90;
59    pub const BYTE_ORDER: usize = 0x02;
60    pub const WORD_ORDER: usize = 0x03;
61    pub const TARGET_OS: usize = 0x0a;
62    pub const RESIDENT_NAMES: usize = 0x58;
63    /// An absolute file offset.
64    pub const NON_RESIDENT_NAMES: usize = 0x88;
65    pub const NON_RESIDENT_SIZE: usize = 0x8c;
66}

68/// The format of the file.
69#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
70pub enum LegacyFormat {
71    /// A 16-bit new executable (`NE`).
72    Ne,
73    /// A linear executable (`LE`), such as a virtual device driver.
74    Le,
75    /// A linear executable (`LX`).
76    Lx,
77}

79/// The operating system a module targets, from its header.
80#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
81pub enum TargetOs {
82    /// The header does not say (`0`).
83    Unspecified,
84    /// OS/2 (`1`).
85    Os2,
86    /// Windows (`2`).
87    Windows,
88    /// European multitasking DOS 4 (`3`).
89    Dos4,
90    /// Windows 386 virtual device drivers (`4`).
91    Win386,
92    /// Any other value.
93    Other(u16),
94}
95
96impl TargetOs {
97    const fn of(value: u16) -> Self {
98        match value {
99            0 => Self::Unspecified,
100            1 => Self::Os2,
101            2 => Self::Windows,
102            3 => Self::Dos4,
103            4 => Self::Win386,
104            other => Self::Other(other),
105        }
106    }
107}

109/// A name from a name table.
110#[derive(Debug, Clone, PartialEq, Eq, Hash)]
111pub struct LegacyName {
112    /// The file offset of the name's length byte.
113    pub offset: u64,
114    /// The bytes exactly as the file has them.
115    pub raw: Vec<u8>,
116    /// The text: printable ASCII kept, any other byte U+FFFD.
117    pub text: String,
118    /// Some byte was not printable ASCII and was replaced.
119    pub non_ascii: bool,
120    /// The match forms of [`text`](Self::text).
121    pub normalized: NormalizedName,
122}
123
124impl LegacyName {
125    fn new(offset: usize, raw: &[u8]) -> Self {
126        let mut non_ascii = false;
127        let text: String = raw
128            .iter()
129            .map(|&b| {
130                if b.is_ascii_graphic() || b == b' ' {
131                    char::from(b)
132                } else {
133                    non_ascii = true;
134                    char::REPLACEMENT_CHARACTER
135                }
136            })
137            .collect();
138        Self {
139            offset: to_u64(offset),
140            raw: raw.to_vec(),
141            normalized: NormalizedName::new(&text),
142            text,
143            non_ascii,
144        }
145    }
146}

157/// Why something was not read. The result is still returned.
158#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
159pub enum LegacyWarning {
160    /// The file ends inside the header: only the format is known.
161    HeaderTruncated,
162    /// An LE or LX header declares big-endian byte or word order. Its names
163    /// are not read.
164    BigEndian,
165    /// The name table starts outside the file.
166    TableOutOfBounds(NameKind),
167    /// The name runs past the end of the file.
168    NameTruncated(NameKind),
169    /// The name runs past the table length that the header declares.
170    NameOutsideTable(NameKind),
171}

173/// The identity of an NE, LE or LX file.
174#[derive(Debug, Clone, PartialEq, Eq, Hash)]
175pub struct LegacyInfo {
176    /// The format.
177    pub format: LegacyFormat,
178    /// The header's file offset (`e_lfanew`).
179    pub header_offset: u32,
180    /// The target operating system, if the header was read.
181    pub target_os: Option<TargetOs>,
182    /// The module name, if present and readable.
183    pub module_name: Option<LegacyName>,

185    pub description: Option<LegacyName>,
186    /// What could not be read, in the order found.
187    pub warnings: Vec<LegacyWarning>,
188}

190/// Why the file is not an NE, LE or LX file.
191#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
192pub enum LegacyError {
193    /// The input is shorter than a DOS header (64 bytes).
194    TooSmall,
195    /// The input does not start with `MZ`.
196    NoDosSignature,
197    /// `e_lfanew` is above [`MAX_E_LFANEW`] or leaves no room for a
198    /// signature.
199    LfanewOutOfRange(u32),
200    /// There is no `NE`, `LE` or `LX` signature at `e_lfanew`.
201    NoSignature,
202}
203
204impl core::fmt::Display for LegacyError {
205    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
206        match self {
207            Self::TooSmall => f.write_str("input is smaller than a DOS header"),
208            Self::NoDosSignature => f.write_str("missing MZ signature"),
209            Self::LfanewOutOfRange(v) => write!(f, "e_lfanew {v:#x} is out of range"),
210            Self::NoSignature => f.write_str("no NE, LE or LX signature"),
211        }
212    }
213}

217/// Reads the identity of an NE, LE or LX file. See the module
218/// documentation.
219///
220/// # Errors
221///
222/// [`LegacyError`] if `data` is not an NE, LE or LX file.
223pub fn parse(data: &[u8]) -> Result<LegacyInfo, LegacyError> {
224    if data.len() < DOS_HEADER_LEN {
225        return Err(LegacyError::TooSmall);
226    }
227    if bytes(data, 0, 2) != Some(b"MZ") {
228        return Err(LegacyError::NoDosSignature);
229    }
230    let e_lfanew = u32_at(data, E_LFANEW_AT).ok_or(LegacyError::TooSmall)?;
231    let at = to_usize(e_lfanew)
232        .filter(|_| e_lfanew <= MAX_E_LFANEW)
233        .ok_or(LegacyError::LfanewOutOfRange(e_lfanew))?;
234    let format = match bytes(data, at, 2) {
235        Some(b"NE") => LegacyFormat::Ne,
236        Some(b"LE") => LegacyFormat::Le,
237        Some(b"LX") => LegacyFormat::Lx,
238        Some(_) => return Err(LegacyError::NoSignature),
239        None => return Err(LegacyError::LfanewOutOfRange(e_lfanew)),
240    };
241    let mut info = LegacyInfo {
242        format,
243        header_offset: e_lfanew,
244        target_os: None,
245        module_name: None,
246        description: None,
247        warnings: Vec::new(),
248    };
249    let len = match format {
250        LegacyFormat::Ne => ne::LEN,
251        LegacyFormat::Le | LegacyFormat::Lx => lx::LEN,
252    };
253    let Some(header) = bytes(data, at, len) else {
254        info.warnings.push(LegacyWarning::HeaderTruncated);
255        return Ok(info);
256    };
257    match format {
258        LegacyFormat::Ne => read_ne(data, at, header, &mut info),
259        LegacyFormat::Le | LegacyFormat::Lx => read_lx(data, at, header, &mut info),
260    }
261    Ok(info)
262}

264/// The header fields used. `header` holds at least [`ne::LEN`] bytes, so the
265/// field reads cannot fail; a failed one reads as zero (absent).
266fn read_ne(data: &[u8], at: usize, header: &[u8], info: &mut LegacyInfo) {
267    info.target_os = Some(TargetOs::of(u16::from(
268        u8_at(header, ne::TARGET_OS).unwrap_or(0),
269    )));
270    let resident = u16_at(header, ne::RESIDENT_NAMES).unwrap_or(0);
271    // A resident table offset of zero points at the header itself, which
272    // is not a name table.
273    if resident != 0 {
274        let start = at.checked_add(usize::from(resident));
275        info.module_name = name(data, start, None, NameKind::ModuleName, info);
276    }
277    let non_resident = u32_at(header, ne::NON_RESIDENT_NAMES).unwrap_or(0);
278    let size = u16_at(header, ne::NON_RESIDENT_SIZE).unwrap_or(0);
279    if non_resident != 0 && size != 0 {
280        info.description = name(
281            data,
282            to_usize(non_resident),
283            Some(usize::from(size)),
284            NameKind::Description,
285            info,
286        );
287    }
288}

290/// As [`read_ne`], for LE and LX: `header` holds at least [`lx::LEN`] bytes.
291fn read_lx(data: &[u8], at: usize, header: &[u8], info: &mut LegacyInfo) {
292    info.target_os = Some(TargetOs::of(u16_at(header, lx::TARGET_OS).unwrap_or(0)));
293    let byte_order = u8_at(header, lx::BYTE_ORDER).unwrap_or(1);
294    let word_order = u8_at(header, lx::WORD_ORDER).unwrap_or(1);
295    if byte_order != 0 || word_order != 0 {
296        info.warnings.push(LegacyWarning::BigEndian);
297        return;
298    }
299    let resident = u32_at(header, lx::RESIDENT_NAMES).unwrap_or(0);
300    if resident != 0 {
301        let start = to_usize(resident).and_then(|r| at.checked_add(r));
302        info.module_name = name(data, start, None, NameKind::ModuleName, info);
303    }
304    let non_resident = u32_at(header, lx::NON_RESIDENT_NAMES).unwrap_or(0);
305    let size = u32_at(header, lx::NON_RESIDENT_SIZE).unwrap_or(0);
306    if non_resident != 0 && size != 0 {
307        info.description = name(
308            data,
309            to_usize(non_resident),
310            to_usize(size),
311            NameKind::Description,
312            info,
313        );
314    }
315}

317/// The first entry of the name table at `start`, which is `table_len` bytes
318/// long if the header says. `None`, with a warning, if it cannot be read;
319/// `None` alone if the table is empty.
320fn name(
321    data: &[u8],
322    start: Option<usize>,
323    table_len: Option<usize>,
324    kind: NameKind,
325    info: &mut LegacyInfo,
326) -> Option<LegacyName> {
327    let mut warn = |w| {
328        info.warnings.push(w);
329        None
330    };
331    let Some((start, len)) = start.and_then(|s| Some((s, u8_at(data, s)?))) else {
332        return warn(LegacyWarning::TableOutOfBounds(kind));
333    };
334    if len == 0 {
335        return None;
336    }
337    let len = usize::from(len);
338    if table_len.is_some_and(|t| len.saturating_add(1) > t) {
339        return warn(LegacyWarning::NameOutsideTable(kind));
340    }
341    let Some(raw) = start.checked_add(1).and_then(|s| bytes(data, s, len)) else {
342        return warn(LegacyWarning::NameTruncated(kind));
343    };
344    Some(LegacyName::new(start, raw))
