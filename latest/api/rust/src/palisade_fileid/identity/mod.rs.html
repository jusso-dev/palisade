1//! The unified file identity record (#55).
2//!
3//! [`identify`] and [`identify_bytes`] run the other modules of this crate
4//! over one file and fold their outputs into one [`FileIdentity`]:
5//!
6//! | Group | From | Tier |
7//! |---|---|---|
8//! | `size`, `timestamps`, `fs.path` | the opened handle, [`crate::fsctx`] (#54) | stat (T0) |
9//! | `kind` | [`crate::filetype`] (#44) | classify (T1) |
10//! | `pe` | [`crate::pe`] (#45) | classify (T1) |
11//! | `hashes` | [`crate::hash`] (#46), [`crate::authenticode`] image and page hashes (#47) | hash (T2) |
12//! | `signing_state`, `signatures` | [`crate::authenticode`] signatures, chains, validity (#48, #49, #50) | trust (T3) |
13//! | `catalog_matches` | [`crate::catalog`] (#51) | trust (T3) |
14//! | `fs.motw`, `fs.path_variables` | [`crate::fsctx`] (#54) | context (T4) |
15//! | `derived`, `masquerade`, `strengths` | this module | always |
16//!
17//! Groups other issues still deliver are present in the schema now, so the
18//! record's shape does not change when they land, and hold
19//! [`Field::NotYetSupported`] with the issue numbers: `script` and the
20//! normalised script hash (#52), `installer`, `package`
21//! and the normalised MSI hash (#53), and the owner, writability, streams,
22//! reparse points, file ID, volume and provenance in `fs` (#54).
23//!
24//! # Signing state
25//!
26//! The record folds embedded signatures and catalogs into one verdict,
27//! [`SigningStateKind`]. It is `trusted` only when a trusted signer vouches
28//! for the file: an embedded signature that passed every #48, #49 and #50
29//! check, or a catalog whose signer is trusted and whose match does not
30//! rely on SHA-1 (ADR 0008). It is `unsigned` only when the file has no
31//! embedded signature and a catalog lookup found nothing. A file whose
32//! format's embedded signatures are not read yet (anything but a PE image,
33//! until #52 and #53), or that was not looked up in any catalog, is
34//! `unverified`, never `unsigned`.
35//!
36//! # Fail-conservative
37//!
38//! Nothing in the record is guessed. A tier that fails leaves its fields
39//! [`Field::Unavailable`] and an entry in [`FileIdentity::errors`]; it
40//! never turns into "unsigned", "not a driver" or "clean". A masquerade
41//! check that could not run is listed in [`Masquerade::unassessed`]. A path
42//! counts as user-writable until #54 reads its ACLs.
43//!
44//! # Serialised form and evolution
45//!
46//! [`FileIdentity`] serialises with serde: `snake_case` field and variant
47//! names, digests as lowercase hex, times as seconds since the Unix epoch.
48//! Each [`Field`] is `{"status": "computed", "value": ...}` or a status
49//! without a value (`not_requested`, `not_applicable`, `unavailable` with a
50//! reason, `not_yet_supported` with the issue numbers). The schema is
51//! `fileid.v1` ([`SCHEMA_VERSION`]). Within `fileid.v1`:
52//!
53//! - fields and enum variants are only ever added, never renamed, removed
54//!   or given a new meaning;
55//! - a consumer must ignore fields and variants it does not know;
56//! - a breaking change is a new schema version, `fileid.v2`.
57//!
58//! The protobuf form for the agent-server protocol is `palisade-proto`'s
59//! (#55, follow-up), with the same field names; protobuf field numbers,
60//! once assigned, are never reused or renumbered.

89impl FileIdentity {
90    /// Folds `other`, a record of the same file computed for other tiers,
91    /// into this one: the agent computes T0 to T3 at execution and T4 and
92    /// T5 later. A field takes `other`'s value when it says more (a value
93    /// beats a reason, a reason beats "not requested"). The derived
94    /// attributes, masquerade assessment and strengths are recomputed with
95    /// `ctx`.
96    ///
97    /// Returns `false`, and changes nothing, when the records are shown to
98    /// describe different files or versions: a different path or name hint,
99    /// a different final path, size, last-write or creation time, or a
100    /// different content digest; and when both records hold attributes
101    /// derived from the content, unless both carry the same content digest.
102    /// Otherwise an attribute only one record has is not compared. Until #54 adds the file ID and change journal stamp,
103    /// that is the strongest check available; a file rewritten in place
104    /// with the same size and restored times is not detected.
105    pub fn merge(&mut self, other: Self, ctx: &IdentityContext<'_>) -> bool {
106        fn differ<T: PartialEq>(a: Option<T>, b: Option<T>) -> bool {
107            matches!((a, b), (Some(a), Some(b)) if a != b)
108        }
109        let final_path = |id: &Self| id.fs.path.computed().and_then(|p| p.final_path.clone());
110        let times = |id: &Self| id.timestamps.computed().copied();
111        // Attributes derived from the content (classify, hash, trust) are
112        // joined only when both records carry the digest of the content
113        // they read, and it is the same: two records of the same size and
114        // name may still be of different content.
115        let reads_content = |id: &Self| {
116            [Tier::Classify, Tier::Hash, Tier::Trust]
117                .iter()
118                .any(|t| id.computed_tiers.contains(*t))
119        };
120        let content_unproven = reads_content(self)
121            && reads_content(&other)
122            && match (&self.content_sha256, &other.content_sha256) {
123                (Some(a), Some(b)) => a != b,
124                _ => true,
125            };
126        let disproven = self.path != other.path
127            || self.name_hint != other.name_hint
128            || differ(self.size.computed(), other.size.computed())
129            || differ(self.content_sha256.as_ref(), other.content_sha256.as_ref())
130            || content_unproven
131            || differ(final_path(self), final_path(&other))
132            || differ(
133                times(self).and_then(|t| t.modified),
134                times(&other).and_then(|t| t.modified),
135            )
136            || differ(
137                times(self).and_then(|t| t.created),
138                times(&other).and_then(|t| t.created),
139            );
140        if disproven {
141            return false;
142        }
143        self.requested_tiers = self.requested_tiers.union(other.requested_tiers);
144        self.computed_tiers = self.computed_tiers.union(other.computed_tiers);
145        self.size.absorb(other.size);
146        if self.content_sha256.is_none() {
147            self.content_sha256 = other.content_sha256;
148        }
149        self.timestamps.absorb(other.timestamps);
150        self.kind.absorb(other.kind);
151        self.pe.absorb(other.pe);
152        self.hashes.absorb(other.hashes);
153        self.signing_state.absorb(other.signing_state);
154        self.signatures.absorb(other.signatures);
155        self.catalog_matches.absorb(other.catalog_matches);
156        self.installer.absorb(other.installer);
157        self.package.absorb(other.package);
158        self.script.absorb(other.script);
159        self.fs.absorb(other.fs);
160        for e in other.errors {
161            if !self.errors.contains(&e) {
162                self.errors.push(e);
163            }
164        }
165        self.derived = compose::derived_for(self);
166        self.masquerade =
167            masquerade::assess(self, ctx.protected, ctx.code_trust, ctx.known_folders);
168        self.strengths = strength::classify(self);
169        true
170    }
171}

173/// One difference between two identities, from [`compare`].
174#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
175pub struct Difference {
176    /// The attribute.
177    pub attribute: &'static str,
178    /// Its value in the first record.
179    pub a: Option<String>,
180    /// Its value in the second record.
181    pub b: Option<String>,
182}
183
184/// The rule-relevant attributes in which `a` and `b` differ: content
185/// hashes, signing state, publisher, product, signer certificates, file
186/// names and versions. An update that keeps its publisher and product
187/// shows differences only in hashes and versions.
188#[must_use]
189pub fn compare(a: &FileIdentity, b: &FileIdentity) -> Vec<Difference> {
190    fn attrs(id: &FileIdentity) -> Vec<(&'static str, Option<String>)> {
191        let h = id.hashes.computed();
192        let v = id.version();
193        let s = id.trusted_signer();
194        vec![
195            ("sha256", h.and_then(|h| h.sha256.clone())),
196            (
197                "authenticode_sha256",
198                h.and_then(|h| h.authenticode.computed())
199                    .and_then(|a| a.sha256.clone()),
200            ),
201            (
202                "signing_state",
203                id.signing_state
204                    .computed()
205                    .map(|s| format!("{:?}", s.state).to_lowercase()),
206            ),
207            ("publisher", s.and_then(|s| s.publisher_name.clone())),
208            ("publisher_dn", s.and_then(|s| s.publisher_dn.clone())),
209            (
210                "signer_leaf_sha256_thumbprint",
211                s.and_then(|s| s.leaf.as_ref())
212                    .map(|c| c.sha256_thumbprint.clone()),
213            ),
214            (
215                "signer_pca_tbs_hash",
216                s.and_then(|s| s.pca.as_ref())
217                    .and_then(|c| c.tbs_hash.clone()),
218            ),
219            ("product_name", v.and_then(|v| v.product_name.clone())),
220            ("company_name", v.and_then(|v| v.company_name.clone())),
221            (
222                "original_file_name",
223                v.and_then(|v| v.original_file_name.clone()),
224            ),
225            ("file_version", v.and_then(|v| v.file_version_quad.clone())),
226            (
227                "product_version",
228                v.and_then(|v| v.product_version_quad.clone()),
229            ),
230            ("file_name", id.name.file_name.clone()),
231            ("size", id.size.computed().map(ToString::to_string)),
232        ]
233    }
234    attrs(a)
235        .into_iter()
236        .zip(attrs(b))
237        .filter(|((_, x), (_, y))| x != y)
238        .map(|((attribute, x), (_, y))| Difference {
239            attribute,
240            a: x,
241            b: y,
242        })
243        .collect()
244}
