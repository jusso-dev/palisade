46/// Everything [`identify`] needs besides the file: trust anchors, the
47/// catalog index, revocation data, the protected-binary list, the clock
48/// and options. The library keeps no state of its own.
49#[derive(Debug, Clone)]
50pub struct IdentityContext<'a> {
51    /// Anchors for code signers.
52    pub code_trust: &'a TrustConfig,
53    /// Anchors for timestamping authorities.
54    pub tsa_trust: &'a TrustConfig,
55    /// The catalogs to look the file up in. `None` skips the lookup.
56    pub catalogs: Option<&'a CatalogStore>,
57    /// Revocation data.
58    pub revocation: &'a RevocationSet,
59    /// The validity settings.
60    pub validity: &'a ValidityPolicy,
61    /// The current time, in seconds since the Unix epoch.
62    pub now: i64,
63    /// A lower bound on real time (see [`ValidityContext::clock_floor`]).
64    pub clock_floor: Option<i64>,
65    /// The protected-binary and protected-vendor lists.
66    pub protected: &'a ProtectedList,
67    /// The machine's known folders, for path variables and the protected
68    /// binaries' expected locations. `None` leaves those unassessed.
69    pub known_folders: Option<&'a KnownFolders>,
70    /// Which flat hashes to compute. SHA-256 is always added.
71    pub flat_algorithms: AlgorithmSet,
72    /// Hashing limits.
73    pub hash_options: HashOptions,
74    /// Files larger than this are not read into memory, so their PE
75    /// metadata and signatures are not read; their hashes are still
76    /// streamed. Default [`MAX_IN_MEMORY`].
77    pub max_in_memory: u64,
78}

130/// Computes the identity of the file at `path`, for the requested tiers.
131///
132/// The file is opened once, without following it as a FIFO, and every
133/// tier reads that one handle, so all attributes describe the same file.
134/// The trust tier needs the hash tier's digests and computes them too.
135/// Failures within a tier are recorded in [`FileIdentity::errors`] and
136/// leave that tier's fields [`Field::Unavailable`]; nothing in the record
137/// is ever guessed.
138pub fn identify(
139    path: &Path,
140    tiers: Tiers,
141    ctx: &IdentityContext<'_>,
142) -> Result<FileIdentity, IdentifyError> {
143    if !std::fs::metadata(path)
144        .map_err(IdentifyError::Open)?
145        .is_file()
146    {
147        return Err(IdentifyError::NotAFile);
148    }
149    let mut file = crate::hash::open_without_blocking(path).map_err(IdentifyError::Open)?;
150    let meta = file.metadata().map_err(IdentifyError::Open)?;
151    if !meta.is_file() {
152        return Err(IdentifyError::NotAFile);
153    }
154    let text = path.to_string_lossy().into_owned();
155    let mut id = FileIdentity::empty(Some(text.clone()), None, tiers);
156    let len = meta.len();
157
158    // T0 always runs: it costs one stat, and every other tier needs the
159    // size.
160    id.size = Field::Computed(len);
161    id.timestamps = Field::Computed(timestamps(&meta));
162    // The final path is the open handle's, so it describes the file whose
163    // content is read below even if the path is replaced meanwhile.
164    id.fs.path = Field::Computed(path_info(&text, false, Some(&file)));
165    id.computed_tiers.insert(Tier::Stat);
166
167    let content = Content::read(&mut file, len, tiers, ctx, &mut id);
168    compose(&mut id, content, tiers, ctx);
169    context_tier(&mut id, &file, tiers, ctx);
170    finish(&mut id, ctx);
171    Ok(id)
172}

174/// Computes the identity of an uploaded file held in memory. `name_hint`
175/// is the name the uploader gave, which is untrusted: it is used for the
176/// name checks only, never as a location. There is no file system context.
177#[must_use]
178pub fn identify_bytes(
179    bytes: &[u8],
180    name_hint: Option<&str>,
181    tiers: Tiers,
182    ctx: &IdentityContext<'_>,
183) -> FileIdentity {
184    let mut id = FileIdentity::empty(None, name_hint.map(str::to_owned), tiers);
185    let len = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
186    id.size = Field::Computed(len);
187    id.timestamps = Field::NotApplicable;
188    id.fs.path = name_hint.map_or(Field::NotApplicable, |n| {
189        Field::Computed(path_info(n, true, None))
190    });
191    id.computed_tiers.insert(Tier::Stat);
192    let mut cursor = io::Cursor::new(bytes);
193    let content = Content::read(&mut cursor, len, tiers, ctx, &mut id);
194    compose(&mut id, content, tiers, ctx);
195    if tiers.contains(Tier::Context) {
196        id.fs.not_applicable_context();
197    }
198    finish(&mut id, ctx);
199    id
200}

202/// What was read from the file's content.
203struct Content {
204    /// The whole file, when a tier needs it and it fits
205    /// [`IdentityContext::max_in_memory`].
206    bytes: Option<Vec<u8>>,
207    /// The hashes, from one streaming pass.
208    hashes: Option<ImageHashes>,
209    /// The content classification (#44). It reads only the head of the
210    /// file and a few bounded targeted reads, so it runs on large files
211    /// too.
212    kind: Option<Classification>,
213}
214
215impl Content {
216    /// Reads the content once. When a tier needs the bytes and the file
217    /// fits [`IdentityContext::max_in_memory`], it is read into memory and hashed from
218    /// there, so the PE metadata, the digests and the signatures all
219    /// describe exactly the same bytes. A larger file is only streamed
220    /// through the hashes.
221    fn read<R: Read + Seek>(
222        r: &mut R,
223        len: u64,
224        tiers: Tiers,
225        ctx: &IdentityContext<'_>,
226        id: &mut FileIdentity,
227    ) -> Self {
228        let need_bytes = tiers.contains(Tier::Classify)
229            || tiers.contains(Tier::Trust)
230            || tiers.contains(Tier::Deep);
231        let need_hashes = tiers.contains(Tier::Hash) || tiers.contains(Tier::Trust);
232        let mut out = Self {
233            bytes: None,
234            hashes: None,
235            kind: None,
236        };
237        if need_bytes {
238            if len > ctx.max_in_memory {
239                let msg = format!(
240                    "{len} bytes is over the in-memory limit of {}",
241                    ctx.max_in_memory
242                );
243                for tier in [Tier::Classify, Tier::Trust, Tier::Deep] {
244                    if tiers.contains(tier) {
245                        id.error(tier, "too_large", msg.clone());
246                    }
247                }
248            } else {
249                let mut buf = Vec::with_capacity(usize::try_from(len).unwrap_or(0));
250                let read = r
251                    .seek(SeekFrom::Start(0))
252                    .and_then(|_| r.take(len.saturating_add(1)).read_to_end(&mut buf));
253                let failure = match read {
254                    Ok(_) if u64::try_from(buf.len()).ok() == Some(len) => {
255                        id.content_sha256 = Some(sha256_hex(&buf));
256                        out.bytes = Some(buf);
257                        None
258                    }
259                    Ok(_) => Some((
260                        "changed_while_read",
261                        "the file changed size while it was read".to_owned(),
262                    )),
263                    Err(e) => Some(("read_failed", e.to_string())),
264                };
265                if let Some((code, msg)) = failure {
266                    // Charged to every requested tier that needed the
267                    // bytes or would have hashed them; nothing describes a
268                    // file that changed under the read.
269                    for tier in [Tier::Classify, Tier::Hash, Tier::Trust, Tier::Deep] {
270                        if tiers.contains(tier) {
271                            id.error(tier, code, msg.clone());
272                        }
273                    }
274                    if need_hashes {
275                        id.hashes = Field::Unavailable(msg);
276                    }
277                    return out;
278                }
279            }
280        }
281        // The trust and deep tiers need the content kind too: which
282        // signature reader runs depends on it.
283        if tiers.contains(Tier::Classify)
284            || tiers.contains(Tier::Trust)
285            || tiers.contains(Tier::Deep)
286        {
287            let name = id.name.file_name.as_deref().map(OsStr::new);
288            out.kind = match &out.bytes {
289                Some(bytes) => Some(classify_bytes(bytes, name)),
290                None => match r.seek(SeekFrom::Start(0)) {
291                    Ok(_) => Some(classify(&mut *r, name)),
292                    Err(e) => {
293                        id.error(Tier::Classify, "read_failed", e.to_string());
294                        None
295                    }
296                },
297            };
298        }
299        if need_hashes {
300            // Every algorithm a revoked file hash is listed in, so a hash
301            // revoked by, say, its SHA-384 is found whatever the caller
302            // asked to compute; both the flat and the Authenticode digests.
303            let revoked_algs = ctx
304                .revocation
305                .file_hashes()
306                .iter()
307                .fold(AlgorithmSet::EMPTY, |set, d| {
308                    set | AlgorithmSet::from(d.algorithm())
309                });
310            let options = ImageHashOptions {
311                hash: HashOptions {
312                    algorithms: ctx.flat_algorithms | AlgorithmSet::SHA256 | revoked_algs,
313                    ..ctx.hash_options
314                },
315                authenticode_extra: revoked_algs,
316                parallel: false,
317            };
318            let result = match &out.bytes {
319                Some(bytes) => hash_image_reader(bytes.as_slice(), len, &options),
320                None => r
321                    .seek(SeekFrom::Start(0))
322                    .map_err(HashError::Io)
323                    .and_then(|_| hash_image_reader(&mut *r, len, &options)),
324            };
325            match result {
326                Ok(ImageHashStatus::Complete(h)) => {
327                    if let Some(d) = h.flat.get(HashAlgorithm::Sha256) {
328                        id.content_sha256 = Some(d.to_string());
329                    }
330                    out.hashes = Some(h);
331                }
332                Ok(ImageHashStatus::TooLarge { limit }) => {
333                    let msg = format!("{len} bytes is over the hash limit of {limit}");
334                    id.hashes = Field::Unavailable(msg.clone());
335                    id.error(Tier::Hash, "too_large", msg);
336                }
337                Err(e) => {
338                    id.hashes = Field::Unavailable(e.to_string());
339                    id.error(Tier::Hash, "hash_failed", e.to_string());
340                }
341            }
342        }
343        out
344    }
345}

347/// The SHA-256 of `bytes`, lowercase hex.
348fn sha256_hex(bytes: &[u8]) -> String {
349    let mut h = MultiHasher::new(AlgorithmSet::SHA256);
350    h.update(bytes);
351    h.finalize()
352        .get(HashAlgorithm::Sha256)
353        .map(ToString::to_string)
354        .unwrap_or_default()
355}

357/// The classify, hash and trust tiers.
358fn compose(id: &mut FileIdentity, content: Content, tiers: Tiers, ctx: &IdentityContext<'_>) {
359    let info = content.bytes.as_deref().map(pe::parse);
360    let pe_info = match &info {
361        Some(Ok(p)) => Some(p),
362        _ => None,
363    };
364    // PE content, by the parse when the bytes were read, otherwise by the
365    // classifier's look at the headers (a large file, or the hash tier
366    // alone), so a missing Authenticode digest of a PE is `unavailable`,
367    // not `not_applicable`.
368    let is_pe = match &info {
369        Some(r) => r.is_ok(),
370        None => match &content.kind {
371            Some(k) => k.content_kind == FileKind::Pe,
372            None => content
373                .hashes
374                .as_ref()
375                .is_some_and(|h| h.authenticode.is_some()),
376        },
377    };
378    // Scripts, installer packages and app packages, chosen by the content
379    // the classifier saw, never by the name alone: a PE named `.ps1` is
380    // judged as a PE, so no script or package signature can vouch for it.
381    let non_pe = if is_pe || !(tiers.contains(Tier::Trust) || tiers.contains(Tier::Deep)) {
382        None
383    } else {
384        content
385            .kind
386            .as_ref()
387            .and_then(|k| NonPe::read(k, content.bytes.as_deref()))
388    };
389
390    if tiers.contains(Tier::Classify) {
391        match (&info, content.bytes.is_some()) {
392            (Some(Ok(p)), _) => {
393                id.pe = Field::Computed(pe_summary(p, id.name.file_name.as_deref()));
394            }
395            (Some(Err(_)), _) => id.pe = Field::NotApplicable,
396            (None, _) => id.pe = Field::Unavailable("the content could not be read".to_owned()),
397        }
398        id.kind = match &content.kind {
399            Some(c) => Field::Computed(kind_summary(c)),
400            None => Field::Unavailable("the content could not be read".to_owned()),
401        };
402        if content.bytes.is_some() && content.kind.is_some() {
403            id.computed_tiers.insert(Tier::Classify);
404        }
405    }
406
407    // The trust tier computes the digests too; they are reported whenever
408    // they exist.
409    if let Some(h) = &content.hashes {
410        let mut summary = hashes(h, is_pe);
411        // The normalised hashes need the content parsed, which the hash
412        // tier alone does not read: they are filled in where the reader
413        // ran, and are not applicable to other kinds.
414        let kind = content.kind.as_ref().map(|k| k.content_kind);
415        summary.msi_normalised_sha256 = match (&non_pe, kind) {
416            (Some(NonPe::Installer(Ok(i))), _) => Field::Computed(i.normalised_sha256.to_string()),
417            (Some(NonPe::Installer(Err(e))), _) => Field::Unavailable(e.message.clone()),
418            (_, Some(k)) if is_installer_kind(k) => Field::Unavailable(
419                "the installer was not read (request the trust or deep tier)".to_owned(),
420            ),
421            // The content kind was not determined (the hash tier alone):
422            // it may be an installer, unless it is a PE image.
423            (_, None) if !is_pe => Field::Unavailable(
424                "the file type was not determined (request the classify, trust or deep tier)"
425                    .to_owned(),
426            ),
427            _ => Field::NotApplicable,
428        };
429        summary.script_normalised_sha256 = match (&non_pe, kind) {
430            (Some(NonPe::Script(r)), _) => r.normalised_hash.as_ref().map_or_else(
431                || Field::Unavailable(script_unhashed_reason(r).to_owned()),
432                |d| Field::Computed(d.to_string()),
433            ),
434            (_, Some(FileKind::Script(s))) if script_kind_of(s).is_some() => Field::Unavailable(
435                "the script was not read (request the trust or deep tier)".to_owned(),
436            ),
437            (_, None) if !is_pe => Field::Unavailable(
438                "the file type was not determined (request the classify, trust or deep tier)"
439                    .to_owned(),
440            ),
441            _ => Field::NotApplicable,
442        };
443        id.hashes = Field::Computed(summary);
444        id.computed_tiers.insert(Tier::Hash);
445    }
446
447    if tiers.contains(Tier::Trust) {
448        trust_tier(id, &content, pe_info, non_pe.as_ref(), ctx);
449    }
450
451    if tiers.contains(Tier::Deep) {
452        deep_tier(id, &content, is_pe, non_pe.as_ref(), ctx);
453    }
454}

457enum NonPe {

459    Script(Box<ScriptReport>),
460    /// A Windows Installer package (#53), or why it could not be read.

463    Package(Result<Box<PackageIdentity>, Unread>),

466/// Why a container could not be read at all.
467#[derive(Debug, Clone)]
468struct Unread {
469    /// It is over the reader's size limit: not read, so not judged.
470    too_large: bool,
471    /// What went wrong, for reading.
472    message: String,
473}
474
475impl NonPe {
476    /// Reads `bytes` as the format `kind` names. `None` for any other
477    /// content, and when the bytes were not read (a reason is recorded:
478    /// the format's signatures are then not checked).
479    fn read(kind: &Classification, bytes: Option<&[u8]>) -> Option<Self> {
480        let content = kind.content_kind;
481        let script = match content {
482            FileKind::Script(s) => script_kind_of(s).map(|k| {
483                // `.vbe`/`.jse` and components share their content kind
484                // with a plain script; the extension tells them apart, and
485                // the script reader reports those unread.
486                match (k, kind.extension.as_deref()) {
487                    (ScriptKind::VbScript, Some("vbe")) => ScriptKind::EncodedVbScript,
488                    (ScriptKind::JScript, Some("jse")) => ScriptKind::EncodedJScript,
489                    (ScriptKind::PowerShell, Some("ps1xml" | "psc1" | "cdxml")) => {
490                        ScriptKind::PowerShellXml
491                    }
492                    (k, _) => k,
493                }
494            }),
495            _ => None,
496        };
497        let wanted = script.is_some()
498            || is_installer_kind(content)
499            || matches!(
500                content,
501                FileKind::Msix | FileKind::Appx | FileKind::MsixBundle | FileKind::AppxBundle
502            );
503        if !wanted {
504            return None;
505        }
506        // Not read into memory (too large): the trust tier records why.
507        let bytes = bytes?;
508        Some(if let Some(k) = script {
509            Self::Script(Box::new(verify_script(bytes, k, &ScriptOptions::default())))
510        } else if is_installer_kind(content) {
511            Self::Installer(parse_installer(bytes).map(Box::new).map_err(|e| Unread {
512                too_large: matches!(e, crate::installer::InstallerError::TooLarge(_)),
513                message: e.to_string(),
514            }))
515        } else {
516            Self::Package(parse_package(bytes).map(Box::new).map_err(|e| Unread {
517                too_large: matches!(e, crate::package::PackageError::TooLarge(_)),
518                message: e.to_string(),
519            }))
520        })
521    }
522}

531/// The script signature format of a classified script kind, for the kinds
532/// that have one.
533const fn script_kind_of(k: crate::filetype::ScriptKind) -> Option<ScriptKind> {
534    use crate::filetype::ScriptKind as F;

537        F::VbScript => Some(ScriptKind::VbScript),
538        F::JScript => Some(ScriptKind::JScript),

541        _ => None,
542    }

545fn script_unhashed_reason(r: &ScriptReport) -> &'static str {
546    if r.empty_text {
547        "the script's text is empty, so its digest identifies nothing"
548    } else if matches!(
549        r.signing_state,
550        ScriptSigningState::TooLargeForSignatureCheck { .. }
551    ) {
552        "the script is over the size limit, so it was not read"
553    } else if r.text_issue.is_some() {
554        "the script's text cannot be decoded without ambiguity"
555    } else if r.kind.signature_supported() {
556        "the script's signature block cannot be read"
557    } else {
558        "this script format is not read"
559    }
560}

562fn trust_tier(
563    id: &mut FileIdentity,
564    content: &Content,
565    pe_info: Option<&PeInfo>,
566    non_pe: Option<&NonPe>,
567    ctx: &IdentityContext<'_>,
568) {
569    let Some(hashes) = &content.hashes else {
570        let why = "the trust tier needs the file's digests, which were not computed";
571        id.signing_state = Field::Unavailable(why.to_owned());
572        id.signatures = Field::Unavailable(why.to_owned());
573        id.catalog_matches = Field::Unavailable(why.to_owned());
574        id.error(Tier::Trust, "no_digests", why);
575        return;
576    };
577    let version = pe_info.and_then(|p| p.version_info.as_ref());
578
579    let mut digests = file_digests(hashes);
580    // A script or installer is also revoked, and catalog-listed, by the
581    // digest its signature signs: the script package's digest of the text,
582    // or the installer's content hash.
583    match non_pe {
584        Some(NonPe::Script(r)) => {
585            if let Some(sip) = &r.sip_digests {
586                digests.extend(sip.digests.iter().cloned());
587            }
588        }
589        Some(NonPe::Installer(Ok(i))) => digests.push(i.content_sha256.clone()),
590        _ => {}
591    }
592    // Embedded signatures (#48), chains (#49) and validity (#50).
593    let mut embedded = EmbeddedState::NotChecked;
594    let mut embedded_invalid_reason = None;
595    let mut sigs: Option<(SignatureReport, TrustReport, ValidityReport)> = None;
596    // A PE image whose Authenticode digests could not be computed has its
597    // signatures left unchecked, so its state stays unverified.
598    if let (Some(info), Some(bytes), Some(image)) = (pe_info, &content.bytes, &hashes.authenticode)
599    {
600        {
601            let report = verify_certificate_table(bytes, info, image, &VerifyOptions::default());
602            let trust = evaluate_trust(&report, ctx.code_trust, version);
603            let vctx = ValidityContext {
604                now: ctx.now,
605                clock_floor: ctx.clock_floor,
606                policy: ctx.validity,
607                revocation: ctx.revocation,
608                tsa_trust: ctx.tsa_trust,
609                file_digests: &digests,
610                version,
611            };
612            let validity = evaluate_validity(&report, &trust, &vctx);
613            embedded = match &report.signing_state {
614                EmbeddedSigning::Unsigned => EmbeddedState::Unsigned,
615                EmbeddedSigning::Valid => EmbeddedState::Valid,
616                EmbeddedSigning::Invalid(reason) => {
617                    embedded_invalid_reason = Some(invalid_reason(reason));
618                    EmbeddedState::Invalid
619                }
620            };
621            sigs = Some((report, trust, validity));
622        }
623    }
624    // Script, installer and package signatures (#52, #53): the same trust
625    // and validity evaluation as a PE file's, over the format's own report.
626    // A format that is not read, a file over its size limit, and a package
627    // the strict reader refuses never read as unsigned.
628    let mut package_publisher = None;
629    if let Some(non_pe) = non_pe {
630        let report = match non_pe {
631            NonPe::Script(r) => match &r.signing_state {
632                ScriptSigningState::TooLargeForSignatureCheck { size, limit } => {
633                    id.error(
634                        Tier::Trust,
635                        "signatures_unread",
636                        format!("{size} bytes is over the script signature limit of {limit}"),
637                    );
638                    None
639                }
640                _ if !r.kind.signature_supported() => None,
641                _ => Some(Ok(r.signatures.clone())),
642            },
643            NonPe::Installer(Ok(i)) => {
644                let mut report = i.signature.clone().unwrap_or_else(unsigned_report);
645                // Bytes after the compound file's last sector are outside
646                // the content hash: a signed package with data appended
647                // (a second file glued on, the CVE-2020-1464 pattern) is
648                // not what its signer signed, so no signature is valid.
649                if installer_trailing_bytes(i).is_some() && i.signature.is_some() {
650                    demote_unsigned_tail(&mut report);
651                }
652                Some(Ok(report))
653            }
654            NonPe::Package(Ok(p)) => {
655                package_publisher = Some(p);
656                Some(Ok(p
657                    .signature
658                    .as_ref()
659                    .map_or_else(unsigned_report, |s| s.report.clone())))
660            }
661            NonPe::Installer(Err(e)) | NonPe::Package(Err(e)) => Some(Err(e.clone())),
662        };
663        match report {
664            // Not read: the state stays unverified.
665            None => {}
666            Some(Err(Unread {
667                too_large: true,
668                message,
669            })) => {
670                id.error(Tier::Trust, "signatures_unread", message);
671            }
672            Some(Err(Unread { message, .. })) => {
673                // Any container the strict reader refuses is an error,
674                // never an unsigned file.
675                embedded = EmbeddedState::Invalid;
676                embedded_invalid_reason = Some(format!("unreadable: {message}"));
677            }
678            Some(Ok(report)) => {
679                let mut trust = match non_pe {
680                    NonPe::Script(r) => {
681                        evaluate_script_trust(
682                            r,
683                            &ScriptTrustAnchors {
684                                policy: ctx.code_trust,
685                                os_root_store: None,
686                            },
687                            ScriptAnchorPolicy::PolicyAnchors,
688                        )
689                        .trust
690                    }
691                    _ => evaluate_trust(&report, ctx.code_trust, None),
692                };
693                let vctx = ValidityContext {
694                    now: ctx.now,
695                    clock_floor: ctx.clock_floor,
696                    policy: ctx.validity,
697                    revocation: ctx.revocation,
698                    tsa_trust: ctx.tsa_trust,
699                    file_digests: &digests,
700                    version: None,
701                };
702                let mut validity = evaluate_validity(&report, &trust, &vctx);
703                // A packaged app's display name, only once its manifest's
704                // publisher is shown to be the signer (#53).
705                if let (Some(p), Some(identity)) = (package_publisher, validity.identity.as_mut()) {
706                    if let Some(name) = p.publisher_display_name.as_deref() {
707                        let _ = identity.attach_package_publisher(&p.publisher_dn, name);
708                    }
709                    if let Some(t) = trust.identity.as_mut() {
710                        t.package_display_name
711                            .clone_from(&identity.package_display_name);
712                    }
713                }
714                embedded = match &report.signing_state {
715                    EmbeddedSigning::Unsigned => EmbeddedState::Unsigned,
716                    EmbeddedSigning::Valid => EmbeddedState::Valid,
717                    EmbeddedSigning::Invalid(reason) => {
718                        embedded_invalid_reason = Some(match non_pe {
719                            NonPe::Script(r) => script_invalid_reason(r),
720                            NonPe::Installer(Ok(i)) => match installer_trailing_bytes(i) {
721                                Some(n) => format!(
722                                    "trailing data is not signed: {n} bytes follow the signed package"
723                                ),
724                                None => invalid_reason(reason),
725                            },
726                            NonPe::Package(Ok(p)) => p
727                                .signature
728                                .as_ref()
729                                .and_then(|s| s.issue.as_ref())
730                                .map_or_else(|| invalid_reason(reason), package_issue_name),
731                            _ => invalid_reason(reason),
732                        });
733                        EmbeddedState::Invalid
734                    }
735                };
736                sigs = Some((report, trust, validity));
737            }
738        }
739    }
740
741    // Catalogs (#51). One lookup serves the summary and the signer. A
742    // script is listed by its script package digest and an installer by
743    // its content hash; whether those match catalogs made by OS tooling is
744    // not yet checked (#51), so for them a miss proves nothing.
745    let (catalog_digests, miss_proves_nothing) = match non_pe {
746        Some(NonPe::Script(r)) => (r.catalog_digests(), true),
747        Some(NonPe::Installer(Ok(i))) => (
748            FileDigests {
749                flat_sha256: Some(i.content_sha256.clone()),
750                ..FileDigests::default()
751            },
752            true,
753        ),
754        _ => (
755            FileDigests::from_hashes(hashes.authenticode.as_ref(), Some(&hashes.flat)),
756            false,
757        ),
758    };
759    // A file revoked by hash is never vouched for, by any signer.
760    let file_revoked = ctx.revocation.file_revoked(&digests).is_some();
761    let found = ctx.catalogs.map(|store| store.lookup(&catalog_digests));
762    // The time revocation is judged at: the later of the clock and the
763    // floor, as for embedded signatures (ADR 0012).
764    let effective_now = ctx.clock_floor.map_or(ctx.now, |f| f.max(ctx.now));
765    // One budget of CRL signature checks for every catalog signer.
766    let mut checker = ctx.revocation.checker();
767    let mut matches = Vec::new();
768    // Which matches a revoked catalog signer disqualifies.
769    let mut usable = Vec::new();
770    let catalog = match (ctx.catalogs, &found) {
771        (Some(store), Some(found)) => {
772            for m in &found.matches {
773                let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
774                let signer_revoked =
775                    chain.is_some_and(|c| chain_revoked(c, ctx, &mut checker, effective_now));
776                // A catalog vouches only when its signer is trusted and its
777                // signature's validity (expiry, revocation of its own hash,
778                // key sizes, timestamps) was judged and passed: a catalog
779                // whose validity was never judged does not vouch (#50).
780                let validity_ok = m.usable();
781                usable.push(validity_ok && !m.relies_on_sha1() && !signer_revoked && !file_revoked);
782                matches.push(CatalogMatchSummary {
783                    catalog_id: m.record.id.as_ref().map(Digest::to_string),
784                    catalog_path: m.record.path.as_ref().map(|p| p.display().to_string()),
785                    source: match m.record.source {
786                        CatalogSource::System => "system",
787                        CatalogSource::DriverStore => "driver_store",
788                        CatalogSource::Policy => "policy",
789                    },
790                    member: m.member,
791                    matched_by: match m.matched_by {
792                        DigestKind::AuthenticodeSha256 => "authenticode_sha256",
793                        DigestKind::AuthenticodeSha1 => "authenticode_sha1",
794                        DigestKind::FlatSha256 => "sha256",
795                        DigestKind::FlatSha1 => "sha1",
796                    },
797                    trusted: m.trusted(),
798                    validity: m.record.validity.as_ref().map(|v| v.error),
799                    relies_on_sha1: m.relies_on_sha1(),
800                    signer_revoked,
801                    publisher_name: m
802                        .signer
803                        .and_then(|s| s.identity.as_ref())
804                        .and_then(|i| i.publisher_name.as_ref())
805                        .map(|p| p.value.clone()),
806                    signer: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
807                });
808            }
809            if found.truncated {
810                id.error(
811                    Tier::Trust,
812                    "catalog_matches_truncated",
813                    "more catalogs list the file than are reported",
814                );
815            }
816            // A store whose index was cut short cannot show that a file is
817            // not listed.
818            let index_truncated = store.records().iter().any(|r| r.index_truncated);
819            if usable.iter().any(|u| *u) {
820                CatalogState::ListedTrusted
821            } else if found.matches.iter().zip(&matches).any(|(m, summary)| {
822                m.usable() && m.relies_on_sha1() && !summary.signer_revoked && !file_revoked
823            }) {
824                CatalogState::ListedTrustedSha1Only
825            } else if !found.matches.is_empty() {
826                CatalogState::ListedUntrusted
827            } else if index_truncated {
828                id.error(
829                    Tier::Trust,
830                    "catalog_index_truncated",
831                    "the catalog index is incomplete, so a missing match proves nothing",
832                );
833                CatalogState::NotChecked
834            } else if miss_proves_nothing {
835                CatalogState::NotChecked
836            } else {
837                CatalogState::NotListed
838            }
839        }
840        _ => CatalogState::NotChecked,
841    };
842
843    // The signer that vouches for the file: a fully valid embedded
844    // signature first, then the best trusted catalog that does not rely on
845    // SHA-1.
846    let mut signer = None;
847    let mut embedded_sha1_only = false;
848    let mut summaries = Vec::new();
849    let mut verification_error = if file_revoked {
850        VerificationError::RevokedByHash
851    } else {
852        VerificationError::NotSigned
853    };
854    let mut clock_suspect = ctx.clock_floor.is_some_and(|f| ctx.now < f);
855    if let Some((report, trust, validity)) = &sigs {
856        verification_error = validity.error;
857        clock_suspect |= validity.clock_suspect;
858        for sig in &report.signatures {
859            let t = trust.signers.iter().find(|s| s.signature == sig.index);
860            let v = validity
861                .signatures
862                .iter()
863                .find(|v| v.signature == sig.index);
864            summaries.push(SignatureSummary {
865                index: sig.index,
866                kind: match sig.kind {
867                    SignatureKind::Nested => "nested",
868                    SignatureKind::Catalog => "catalog",
869                    SignatureKind::Script => "script",
870                    SignatureKind::Msi => "msi",
871                    SignatureKind::Package => "package",
872                    _ => "embedded",
873                },
874                depth: sig.depth,
875                parent: sig.parent,
876                status: status_name(&sig.status),
877                status_detail: format!("{:?}", sig.status),
878                digest_alg: sig.digest_alg,
879                signer_digest_alg: sig.signer_digest_alg,
880                sig_alg: sig.sig_alg.clone(),
881                key_alg: sig.key_alg.clone(),
882                key_bits: sig.key_bits,
883                digest_match: sig.digest_match,
884                program_name: sig.signer.as_ref().and_then(|s| s.program_name.clone()),
885                more_info_url: sig.signer.as_ref().and_then(|s| s.more_info_url.clone()),
886                claimed_signing_time: sig.signer.as_ref().and_then(|s| s.signing_time),
887                has_page_hashes: sig.page_hashes.is_some(),
888                // The signature and, when a chain was built, every
889                // certificate on it (#49).
890                os_ci_compatible: t.map_or_else(|| sig.os_ci_compatible(), |t| t.os_ci_compatible),
891                trusted: t.is_some_and(|t| t.trusted),
892                untrusted: t.map(|t| t.untrusted.clone()).unwrap_or_default(),
893                validity_error: v.map_or(VerificationError::Generic, |v| v.error),
894                trusted_signing_time: v.and_then(|v| v.signing_time),
895                timestamps: sig.timestamp_tokens.len(),
896                chain: t.and_then(|t| t.chain.as_ref()).map(chain_summary),
897            });
898        }
899        // A primary signature that rests on SHA-1 (its content digest or
900        // its signer digest) is valid, but no allow decision may rest on it
901        // (ADR 0008, ADR 0017): it is not a trusted signer, as a catalog
902        // match that relies on SHA-1 is not.
903        let sha2_primary = validity
904            .primary
905            .and_then(|p| report.signatures.get(p))
906            .is_some_and(|sig| {
907                sig.digest_alg.is_some_and(DigestAlgorithm::is_sha2)
908                    && sig.signer_digest_alg.is_some_and(DigestAlgorithm::is_sha2)
909            });
910        embedded_sha1_only = validity.primary.is_some() && !sha2_primary;
911        if let (Some(p), Some(identity), false, true) = (
912            validity.primary,
913            &validity.identity,
914            file_revoked,
915            sha2_primary,
916        ) {
917            let chain = trust
918                .signers
919                .iter()
920                .find(|s| s.signature == p)
921                .and_then(|s| s.chain.as_ref());
922            signer = Some(TrustedSigner {
923                source: TrustSource::Embedded,
924                signature: Some(p),
925                catalog_match: None,
926                publisher_name: identity.publisher_name.as_ref().map(|n| n.value.clone()),
927                publisher_dn: identity.publisher_dn.clone(),
928                product_name: identity.product_name.clone(),
929                leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
930                pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
931                root: chain.and_then(full_root).map(cert_brief),
932                well_known_root_id: chain.and_then(|c| c.well_known_root_id),
933                anchored_to: chain
934                    .and_then(|c| c.anchored_to.as_ref())
935                    .map(|a| a.label.clone()),
936                package_display_name: identity.package_display_name.clone(),
937            });
938        }
939    }
940    // The product of a non-PE file comes from what its signature covers:
941    // a script's signed metadata, an installer's `ProductName` (the
942    // signature covers every stream). A package has none: its family name
943    // is its product identity.
944    if let Some(s) = signer.as_mut() {
945        let product = match non_pe {
946            Some(NonPe::Script(r)) => r
947                .signature()
948                .and_then(|sig| sig.signer.as_ref())
949                .and_then(|a| a.signed_metadata.product.clone()),
950            Some(NonPe::Installer(Ok(i))) => i.properties.product_name.clone(),
951            _ => None,
952        };
953        if non_pe.is_some() {
954            s.product_name =
955                product.filter(|p| !p.trim().is_empty() && !p.chars().any(char::is_control));
956        }
957    }
958    if signer.is_none()
959        && let Some(found) = &found
960    {
961        let best = found
962            .matches
963            .iter()
964            .enumerate()
965            .find(|(i, _)| usable.get(*i).copied().unwrap_or(false));
966        if let Some((i, m)) = best {
967            let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
968            let identity = m.signer.and_then(|s| s.identity.as_ref());
969            signer = Some(TrustedSigner {
970                source: TrustSource::Catalog,
971                signature: None,
972                catalog_match: Some(i),
973                publisher_name: identity
974                    .and_then(|i| i.publisher_name.as_ref())
975                    .map(|n| n.value.clone()),
976                publisher_dn: identity.and_then(|i| i.publisher_dn.clone()),
977                product_name: version
978                    .and_then(|v| v.get(VersionField::ProductName))
979                    .filter(|p| !p.trim().is_empty())
980                    .map(str::to_owned),
981                leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
982                pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
983                root: chain.and_then(full_root).map(cert_brief),
984                well_known_root_id: chain.and_then(|c| c.well_known_root_id),
985                anchored_to: chain
986                    .and_then(|c| c.anchored_to.as_ref())
987                    .map(|a| a.label.clone()),
988                package_display_name: None,
989            });
990        }
991    }
992
993    // A valid embedded signature whose signer was revoked (a certificate in
994    // its chain, or its own hash) is no weaker evidence of tampering than a
995    // broken one: it is invalid, never merely untrusted.
996    let revoked = matches!(
997        verification_error,
998        VerificationError::Revoked | VerificationError::RevokedByHash
999    );
1000    let state = signing_state_kind(
1001        signer.is_some(),
1002        file_revoked || (embedded == EmbeddedState::Valid && signer.is_none() && revoked),
1003        embedded,
1004        catalog,
1005    );
1006
1007    id.signing_state = Field::Computed(SigningState {
1008        state,
1009        embedded,
1010        embedded_invalid_reason,
1011        catalog,
1012        verification_error,
1013        clock_suspect,
1014        embedded_sha1_only,
1015        signer,
1016    });
1017    if sigs.is_some() {
1018        id.signatures = Field::Computed(summaries);
1019    } else if content.bytes.is_none() {
1020        let why = "the file was not read into memory, so its signatures were not read";
1021        id.signatures = Field::Unavailable(why.to_owned());
1022        id.error(Tier::Trust, "signatures_unread", why);
1023    } else if pe_info.is_some() {
1024        let why = "the image's Authenticode digests could not be computed";
1025        id.signatures = Field::Unavailable(why.to_owned());
1026        id.error(Tier::Trust, "signatures_unread", why);
1027    } else {
1028        // A format with no signature reader here (a cabinet, a Java
1029        // archive, an encoded script): its signatures, if it has any, are
1030        // not read, so it stays unverified.
1031        id.signatures = Field::Unavailable("this file type's signatures are not read".to_owned());
1032    }
1033    id.catalog_matches = match ctx.catalogs {
1034        Some(_) => Field::Computed(matches),
1035        None => Field::NotRequested,
1036    };
1037    if id.signatures.is_computed() || (content.bytes.is_some() && pe_info.is_none()) {
1038        id.computed_tiers.insert(Tier::Trust);
1039    }
1040}

1042/// Whether a catalog signer's chain is revoked, or its revocation state is
1043/// unknown where policy says that fails. Catalog signatures carry no
1044/// trusted signing time here (catalog timestamps are #50), so every entry
1045/// counts, whatever its date.
1046fn chain_revoked(
1047    chain: &ChainReport,
1048    ctx: &IdentityContext<'_>,
1049    checker: &mut RevocationChecker<'_>,
1050    now: i64,
1051) -> bool {
1052    let certs: Vec<&CertSummary> = chain.full_chain_certs().collect();
1053    (0..certs.len()).any(|i| match checker.state_of(&certs, i, now) {
1054        CertRevocation::Good => false,
1055        CertRevocation::Unknown => ctx.revocation.failure == RevocationFailure::FailClosed,
1056        // Revoked, or a CRL that was not checked: both fail closed.
1057        _ => true,
1058    })
1059}

1061/// The context tier: mark-of-the-web and path variables. Owner,
1062/// writability, streams and reparse points are #54.
1063fn context_tier(
1064    id: &mut FileIdentity,
1065    file: &std::fs::File,
1066    tiers: Tiers,
1067    ctx: &IdentityContext<'_>,
1068) {
1069    if !tiers.contains(Tier::Context) {
1070        return;
1071    }
1072    // The Palisade provenance tag has no reader yet (#54).
1073    id.fs.provenance = Field::pending(&[54]);
1074    read_fs_details(id, file);
1075    // The stream is read for the opened file itself: by the handle's final
1076    // path, and kept only if that path still names the same file, so the
1077    // record never pairs this file's content with another file's mark.
1078    // A named stream has no mark of its own: `Zone.Identifier` is a stream
1079    // of the host file, and the host's mark would credit the stream with a
1080    // download origin it may not share.
1081    id.fs.motw = if id.name.stream.is_some() {
1082        Field::NotApplicable
1083    } else {
1084        read_motw(file)
1085    };
1086    if let Field::Unavailable(why) = &id.fs.motw
1087        && cfg!(all(windows, feature = "os-trust"))
1088    {
1089        let why = why.clone();
1090        id.error(Tier::Context, "motw_unreadable", why);
1091    }
1092    id.fs.path_variables = match (ctx.known_folders, id.fs.path.computed()) {
1093        (None, _) => Field::Unavailable("no known folders in the context".to_owned()),
1094        (Some(folders), Some(info)) => match info.rule_path() {
1095            Some(p) => match path_variables_matched(&p, folders, None) {
1096                Ok(vars) => Field::Computed(vars.iter().map(|v| v.token().to_owned()).collect()),
1097                Err(e) => Field::Unavailable(e.to_string()),
1098            },
1099            None => Field::Unavailable("the path is not a full Windows path".to_owned()),
1100        },
1101        (Some(_), None) => Field::Unavailable("no path".to_owned()),
1102    };
1103    // The tier counts as computed only when it computed something: with
1104    // every field pending or unavailable it did not.
1105    let fs = &id.fs;
1106    if fs.motw.is_computed()
1107        || fs.path_variables.is_computed()
1108        || fs.file_id.is_computed()
1109        || fs.volume.is_computed()
1110        || fs.owner.is_computed()
1111        || fs.user_writable.is_computed()
1112        || fs.streams.is_computed()
1113        || fs.reparse.is_computed()
1114    {
1115        id.computed_tiers.insert(Tier::Context);
1116    }
1117}

1120fn read_motw(file: &std::fs::File) -> Field<Option<MotwSummary>> {
1121    // The handle may name a stream though the path given did not (a link
1122    // to one): its own final path decides.
1123    if crate::win::links::final_path_of(file).is_ok_and(|p| p.stream().is_some()) {
1124        return Field::NotApplicable;
1125    }
1126    match crate::win::links::read_motw_of(file) {
1127        Ok(m) => Field::Computed(m.as_ref().map(motw_summary)),
1128        Err(e) => Field::Unavailable(e.to_string()),
1129    }

1141/// The file ID, volume, owner, writability, streams and reparse point of
1142/// the opened file, from the OS (#54). Every read is of the open handle, or
1143/// of the path the handle's own final path names, so nothing describes
1144/// another file. A read that fails leaves its field unavailable.
1145#[cfg(all(windows, feature = "os-trust"))]
1146fn read_fs_details(id: &mut FileIdentity, file: &std::fs::File) {
1147    use crate::fsctx::writable::{AssessOptions, WritabilityCache};
1148    use crate::win::{access, links, security::NameCache, volume};
1149    let unavailable = |e: &io::Error| e.to_string();
1150
1151    // The real path of the opened file: every path-based read below uses
1152    // it, never the path as given (a link could lead elsewhere).
1153    let final_path = links::final_path_of(file).map(|p| p.to_win32_string());
1154
1155    id.fs.file_id = match links::read_meta(file) {
1156        Ok(meta) => {
1157            let (count, names) = match &final_path {
1158                Ok(p) => links::hard_links(Path::new(p), file)
1159                    .map_or((None, Vec::new()), |h| (Some(h.count), h.names.clone())),
1160                Err(_) => (None, Vec::new()),
1161            };
1162            Field::Computed(super::record::FileIdSummary {
1163                volume_serial: meta.volume_serial,
1164                file_id: meta
1165                    .file_id
1166                    .map(|b| b.iter().map(|x| format!("{x:02x}")).collect()),
1167                usn_journal_id: meta.usn.map(|(j, _)| j),
1168                usn: meta.usn.map(|(_, u)| u),
1169                link_count: count,
1170                link_names: names,
1171            })
1172        }
1173        Err(e) => Field::Unavailable(unavailable(&e)),
1174    };
1175
1176    let vol = volume::read_volume(file);
1177    id.fs.volume = Field::Computed(volume_summary(&vol));
1178
1179    let mut names = NameCache::default();
1180    let local = matches!(
1181        vol.location_class(),
1182        crate::fsctx::LocationClass::OsVolume | crate::fsctx::LocationClass::OtherFixed
1183    );
1184    id.fs.owner = match access::read_owner(file, vol.fs_type.as_ref(), local, &mut names) {
1185        Ok(o) => Field::Computed(super::record::OwnerSummary {
1186            sid: o.owner_sid.map(|s| s.to_string()),
1187            name: o.owner_name.clone(),
1188            trust_class: snake(&format!("{:?}", o.owner_trust_class)),
1189            supported: o.owner_supported,
1190        }),
1191        Err(e) => Field::Unavailable(unavailable(&e)),
1192    };
1193
1194    // The access check reopens by path: its verdict is kept only if that
1195    // path still names the opened file before and after it ran, so the
1196    // record never pairs this file's ID and owner with another file's
1197    // writability. Otherwise it is unavailable, and the path stays Weak.
1198    let still_this_file = |p: &str| {
1199        let opened = links::read_meta(file)
1200            .ok()
1201            .and_then(|m| m.file_id.zip(m.volume_serial));
1202        let at_path = links::open_link_itself(Path::new(p))
1203            .and_then(|f| links::read_meta(&f))
1204            .ok()
1205            .and_then(|m| m.file_id.zip(m.volume_serial));
1206        opened.is_some() && opened == at_path
1207    };
1208    id.fs.user_writable = match &final_path {
1209        Ok(p) if !still_this_file(p) => {
1210            Field::Unavailable("the file's final path no longer names the opened file".to_owned())
1211        }
1212        Ok(p) => match access::assess_file(
1213            Path::new(p),
1214            &crate::fsctx::EvaluationOptions::default(),
1215            AssessOptions::default(),
1216            &mut WritabilityCache::default(),
1217        ) {
1218            Ok(_) if !still_this_file(p) => Field::Unavailable(
1219                "the file's final path changed while its writability was read".to_owned(),
1220            ),
1221            Ok(w) => Field::Computed(super::record::WritableSummary {
1222                path_user_writable: w.path_user_writable(),
1223                file_user_writable: w.file_user_writable,
1224                first_user_writable_ancestor: w.first_user_writable_ancestor.clone(),
1225                change_perms_by_non_admin: w.change_perms_by_non_admin,
1226                principals_with_write: w
1227                    .principals_with_write
1228                    .iter()
1229                    .map(ToString::to_string)
1230                    .collect(),
1231            }),
1232            Err(e) => Field::Unavailable(unavailable(&e)),
1233        },
1234        Err(e) => Field::Unavailable(unavailable(e)),
1235    };
1236
1237    id.fs.streams = match links::read_streams(file) {
1238        Ok(s) => Field::Computed(super::record::StreamsSummary {
1239            streams: s
1240                .streams
1241                .iter()
1242                .map(|st| super::record::StreamSummary {
1243                    name: st.name.clone(),
1244                    size: st.size,
1245                    benign: st.benign.map(|b| snake(&format!("{b:?}"))),
1246                    executable_content: st.is_executable_content,
1247                })
1248                .collect(),
1249            truncated: s.truncated,
1250        }),
1251        Err(e) => Field::Unavailable(unavailable(&e)),
1252    };
1253
1254    // The reparse point of the path as given: the opened handle followed
1255    // it, so the link itself is opened without following.
1256    id.fs.reparse = match id.path.as_deref() {
1257        Some(p) => {
1258            match links::open_link_itself(Path::new(p)).and_then(|f| links::read_reparse(&f)) {
1259                Ok(r) => Field::Computed(r.map(|r| reparse_summary(&r))),
1260                Err(e) => Field::Unavailable(unavailable(&e)),
1261            }
1262        }
1263        None => Field::NotApplicable,
1264    };
1265
1266    // Every field that could not be read is an error of the tier, as the
1267    // `Field::Unavailable` contract says.
1268    let failures: Vec<(&'static str, String)> = [
1269        ("file_id_unreadable", unavailable_reason(&id.fs.file_id)),
1270        ("owner_unreadable", unavailable_reason(&id.fs.owner)),
1271        (
1272            "writability_unreadable",
1273            unavailable_reason(&id.fs.user_writable),
1274        ),
1275        ("streams_unreadable", unavailable_reason(&id.fs.streams)),
1276        ("reparse_unreadable", unavailable_reason(&id.fs.reparse)),
1277    ]
1278    .into_iter()
1279    .filter_map(|(code, why)| why.map(|w| (code, w)))
1280    .collect();
1281    for (code, why) in failures {
1282        id.error(Tier::Context, code, why);
1283    }
1284}

1558            out.push(c);
1559        }
1560    }

1777/// The deep tier: the installer, package and script records (#52, #53).
1778fn deep_tier(
1779    id: &mut FileIdentity,
1780    content: &Content,
1781    is_pe: bool,
1782    non_pe: Option<&NonPe>,
1783    ctx: &IdentityContext<'_>,
1784) {
1785    let _ = ctx;
1786    const NOT_READ: &str = "the content could not be read";
1787    id.installer = Field::NotApplicable;
1788    id.package = Field::NotApplicable;
1789    id.script = Field::NotApplicable;
1790    if content.bytes.is_none() {
1791        // Too large to read, or a read that failed: what it is may still
1792        // be one of these, so nothing is "not applicable".
1793        let kind = content.kind.as_ref().map(|k| k.content_kind);
1794        match kind {
1795            Some(k) if is_installer_kind(k) => id.installer = Field::Unavailable(NOT_READ.into()),
1796            Some(FileKind::Msix | FileKind::Appx | FileKind::MsixBundle | FileKind::AppxBundle) => {
1797                id.package = Field::Unavailable(NOT_READ.into());
1798            }
1799            Some(FileKind::Script(s)) if script_kind_of(s).is_some() => {
1800                id.script = Field::Unavailable(NOT_READ.into());
1801            }
1802            Some(_) => {}
1803            None => {
1804                id.installer = Field::Unavailable(NOT_READ.into());
1805                id.package = Field::Unavailable(NOT_READ.into());
1806                id.script = Field::Unavailable(NOT_READ.into());
1807            }
1808        }
1809        return;
1810    }
1811    // A PE image has no record here; its deep-tier content tags are not
1812    // run yet, so the tier is not marked computed for it.
1813    if is_pe {
1814        return;
1815    }
1816    match non_pe {
1817        Some(NonPe::Script(r)) => id.script = Field::Computed(script_summary(r)),
1818        Some(NonPe::Installer(Ok(i))) => id.installer = Field::Computed(installer_summary(i)),
1819        Some(NonPe::Package(Ok(p))) => id.package = Field::Computed(package_summary(p)),
1820        Some(NonPe::Installer(Err(e))) => {
1821            id.installer = Field::Unavailable(e.message.clone());
1822            id.error(Tier::Deep, "installer_unreadable", e.message.clone());
1823        }
1824        Some(NonPe::Package(Err(e))) => {
1825            id.package = Field::Unavailable(e.message.clone());
1826            id.error(Tier::Deep, "package_unreadable", e.message.clone());
1827        }
1828        None => {}
1829    }
1830    if id.installer.is_computed() || id.package.is_computed() || id.script.is_computed() {
1831        id.computed_tiers.insert(Tier::Deep);
1832    }
1833}

1989/// The signing verdict from its evidence. `revoked` is a revoked file hash,
1990/// or a revoked signer behind a valid embedded signature. `trusted_signer`
1991/// is whether a
1992/// trusted signer vouches for the file: a valid embedded signature that
1993/// passed every check, or a catalog match counted usable (trusted, and not
1994/// relying on SHA-1). "Unsigned" needs both halves of the evidence: no
1995/// embedded signature, and a catalog lookup that found nothing. Without the
1996/// lookup (no store, or an index cut short) the file may be catalog-signed,
1997/// so it is unverified.
1998fn signing_state_kind(
1999    trusted_signer: bool,
2000    revoked: bool,
2001    embedded: EmbeddedState,
2002    catalog: CatalogState,
2003) -> SigningStateKind {
2004    if trusted_signer {
2005        SigningStateKind::Trusted
2006    } else if revoked || embedded == EmbeddedState::Invalid {
2007        // A broken embedded signature, or a revoked file or signer, is
2008        // invalid whatever a catalog says: an untrusted catalog listing
2009        // does not make it merely "untrusted".
2010        SigningStateKind::Invalid
2011    } else if embedded == EmbeddedState::NotChecked
2012        || (embedded == EmbeddedState::Unsigned && catalog == CatalogState::NotChecked)
2013    {
2014        SigningStateKind::Unverified
2015    } else if embedded == EmbeddedState::Valid
2016        || matches!(
2017            catalog,
2018            CatalogState::ListedUntrusted | CatalogState::ListedTrustedSha1Only
2019        )
2020    {
2021        SigningStateKind::Untrusted
2022    } else {
2023        SigningStateKind::Unsigned
2024    }
2025}
