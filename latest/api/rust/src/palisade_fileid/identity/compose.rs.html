48/// catalog index, revocation data, the protected-binary list, the clock

51pub struct IdentityContext<'a> {
52    /// Anchors for code signers.
53    pub code_trust: &'a TrustConfig,
54    /// Anchors for timestamping authorities.
55    pub tsa_trust: &'a TrustConfig,
56    /// The catalogs to look the file up in. `None` skips the lookup.
57    pub catalogs: Option<&'a CatalogStore>,
58    /// Revocation data.
59    pub revocation: &'a RevocationSet,
60    /// The validity settings.
61    pub validity: &'a ValidityPolicy,
62    /// The current time, in seconds since the Unix epoch.
63    pub now: i64,
64    /// A lower bound on real time (see [`ValidityContext::clock_floor`]).
65    pub clock_floor: Option<i64>,
66    /// The protected-binary and protected-vendor lists.
67    pub protected: &'a ProtectedList,
68    /// The machine's known folders, for path variables and the protected
69    /// binaries' expected locations. `None` leaves those unassessed.
70    pub known_folders: Option<&'a KnownFolders>,
71    /// Which flat hashes to compute. SHA-256 is always added.
72    pub flat_algorithms: AlgorithmSet,
73    /// Hashing limits.
74    pub hash_options: HashOptions,

79    /// The content tagging rules the deep tier scans with (#44): the

166/// Computes the identity of the file at `path`, for the requested tiers.
167///
168/// The file is opened once, without following it as a FIFO, and every
169/// tier reads that one handle, so all attributes describe the same file.
170/// The trust tier needs the hash tier's digests and computes them too.
171/// Failures within a tier are recorded in [`FileIdentity::errors`] and
172/// leave that tier's fields [`Field::Unavailable`]; nothing in the record
173/// is ever guessed.
174pub fn identify(
175    path: &Path,
176    tiers: Tiers,
177    ctx: &IdentityContext<'_>,
178) -> Result<FileIdentity, IdentifyError> {
179    if !std::fs::metadata(path)
180        .map_err(IdentifyError::Open)?
181        .is_file()
182    {
183        return Err(IdentifyError::NotAFile);
184    }
185    let mut file = crate::hash::open_without_blocking(path).map_err(IdentifyError::Open)?;
186    let meta = file.metadata().map_err(IdentifyError::Open)?;
187    if !meta.is_file() {
188        return Err(IdentifyError::NotAFile);
189    }
190    let text = path.to_string_lossy().into_owned();
191    let mut id = FileIdentity::empty(Some(text.clone()), None, tiers);
192    let len = meta.len();
193
194    // T0 always runs: it costs one stat, and every other tier needs the
195    // size.
196    id.size = Field::Computed(len);
197    id.timestamps = Field::Computed(timestamps(&meta));
198    // The final path is the open handle's, so it describes the file whose
199    // content is read below even if the path is replaced meanwhile.
200    id.fs.path = Field::Computed(path_info(&text, false, Some(&file)));
201    id.computed_tiers.insert(Tier::Stat);
202
203    let content = Content::read(&mut file, len, tiers, ctx, &mut id);
204    let pe_report = compose(&mut id, content, tiers, ctx);
205    if tiers.contains(Tier::Trust) {
206        id.os_cross_check = os_cross_check(&file, path, pe_report.as_ref());
207    }
208    context_tier(&mut id, &file, tiers, ctx);
209    finish(&mut id, ctx);
210    Ok(id)
211}

213/// Computes the identity of an uploaded file held in memory. `name_hint`
214/// is the name the uploader gave, which is untrusted: it is used for the
215/// name checks only, never as a location. There is no file system context.
216#[must_use]
217pub fn identify_bytes(
218    bytes: &[u8],
219    name_hint: Option<&str>,
220    tiers: Tiers,
221    ctx: &IdentityContext<'_>,
222) -> FileIdentity {
223    let mut id = FileIdentity::empty(None, name_hint.map(str::to_owned), tiers);
224    let len = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
225    id.size = Field::Computed(len);
226    id.timestamps = Field::NotApplicable;
227    id.fs.path = name_hint.map_or(Field::NotApplicable, |n| {
228        Field::Computed(path_info(n, true, None))
229    });
230    id.computed_tiers.insert(Tier::Stat);
231    let mut cursor = io::Cursor::new(bytes);
232    let content = Content::read(&mut cursor, len, tiers, ctx, &mut id);
233    let _ = compose(&mut id, content, tiers, ctx);
234    if tiers.contains(Tier::Trust) {
235        // The OS verifies files on disk, not uploads.
236        id.os_cross_check = Field::NotApplicable;
237    }
238    if tiers.contains(Tier::Context) {
239        id.fs.not_applicable_context();
240    }
241    finish(&mut id, ctx);
242    id
243}

245/// What was read from the file's content.
246struct Content {
247    /// The whole file, when a tier needs it and it fits
248    /// [`IdentityContext::max_in_memory`].
249    bytes: Option<Vec<u8>>,
250    /// The hashes, from one streaming pass.
251    hashes: Option<ImageHashes>,
252    /// The content classification (#44). It reads only the head of the
253    /// file and a few bounded targeted reads, so it runs on large files
254    /// too.
255    kind: Option<Classification>,
256}
257
258impl Content {
259    /// Reads the content once. When a tier needs the bytes and the file
260    /// fits [`IdentityContext::max_in_memory`], it is read into memory and hashed from
261    /// there, so the PE metadata, the digests and the signatures all
262    /// describe exactly the same bytes. A larger file is only streamed
263    /// through the hashes.
264    fn read<R: Read + Seek>(
265        r: &mut R,
266        len: u64,
267        tiers: Tiers,
268        ctx: &IdentityContext<'_>,
269        id: &mut FileIdentity,
270    ) -> Self {
271        let need_bytes = tiers.contains(Tier::Classify)
272            || tiers.contains(Tier::Trust)
273            || tiers.contains(Tier::Deep);
274        let need_hashes = tiers.contains(Tier::Hash) || tiers.contains(Tier::Trust);
275        let mut out = Self {
276            bytes: None,
277            hashes: None,
278            kind: None,
279        };
280        if need_bytes {
281            if len > ctx.max_in_memory {
282                let msg = format!(
283                    "{len} bytes is over the in-memory limit of {}",
284                    ctx.max_in_memory
285                );
286                for tier in [Tier::Classify, Tier::Trust, Tier::Deep] {
287                    if tiers.contains(tier) {
288                        id.error(tier, "too_large", msg.clone());
289                    }
290                }
291            } else {
292                let mut buf = Vec::with_capacity(usize::try_from(len).unwrap_or(0));
293                let read = r
294                    .seek(SeekFrom::Start(0))
295                    .and_then(|_| r.take(len.saturating_add(1)).read_to_end(&mut buf));
296                let failure = match read {
297                    Ok(_) if u64::try_from(buf.len()).ok() == Some(len) => {
298                        id.content_sha256 = Some(sha256_hex(&buf));
299                        out.bytes = Some(buf);
300                        None
301                    }
302                    Ok(_) => Some((
303                        "changed_while_read",
304                        "the file changed size while it was read".to_owned(),
305                    )),
306                    Err(e) => Some(("read_failed", e.to_string())),
307                };
308                if let Some((code, msg)) = failure {
309                    // Charged to every requested tier that needed the
310                    // bytes or would have hashed them; nothing describes a
311                    // file that changed under the read.
312                    for tier in [Tier::Classify, Tier::Hash, Tier::Trust, Tier::Deep] {
313                        if tiers.contains(tier) {
314                            id.error(tier, code, msg.clone());
315                        }
316                    }
317                    if need_hashes {
318                        id.hashes = Field::Unavailable(msg);
319                    }
320                    return out;
321                }
322            }
323        }
324        // The trust and deep tiers need the content kind too: which
325        // signature reader runs depends on it.
326        if tiers.contains(Tier::Classify)
327            || tiers.contains(Tier::Trust)
328            || tiers.contains(Tier::Deep)
329        {
330            let name = id.name.file_name.as_deref().map(OsStr::new);
331            out.kind = match &out.bytes {
332                Some(bytes) => Some(classify_bytes(bytes, name)),
333                None => match r.seek(SeekFrom::Start(0)) {
334                    Ok(_) => Some(classify(&mut *r, name)),
335                    Err(e) => {
336                        id.error(Tier::Classify, "read_failed", e.to_string());
337                        None
338                    }
339                },
340            };
341        }
342        if need_hashes {
343            // Every algorithm a revoked file hash is listed in, so a hash
344            // revoked by, say, its SHA-384 is found whatever the caller
345            // asked to compute; both the flat and the Authenticode digests.
346            let revoked_algs = ctx
347                .revocation
348                .file_hashes()
349                .iter()
350                .fold(AlgorithmSet::EMPTY, |set, d| {
351                    set | AlgorithmSet::from(d.algorithm())
352                });
353            let options = ImageHashOptions {
354                hash: HashOptions {
355                    algorithms: ctx.flat_algorithms | AlgorithmSet::SHA256 | revoked_algs,
356                    ..ctx.hash_options
357                },
358                authenticode_extra: revoked_algs,
359                parallel: false,
360            };
361            let result = match &out.bytes {
362                Some(bytes) => hash_image_reader(bytes.as_slice(), len, &options),
363                None => r
364                    .seek(SeekFrom::Start(0))
365                    .map_err(HashError::Io)
366                    .and_then(|_| hash_image_reader(&mut *r, len, &options)),
367            };
368            match result {
369                Ok(ImageHashStatus::Complete(h)) => {
370                    if let Some(d) = h.flat.get(HashAlgorithm::Sha256) {
371                        id.content_sha256 = Some(d.to_string());
372                    }
373                    out.hashes = Some(h);
374                }
375                Ok(ImageHashStatus::TooLarge { limit }) => {
376                    let msg = format!("{len} bytes is over the hash limit of {limit}");
377                    id.hashes = Field::Unavailable(msg.clone());
378                    id.error(Tier::Hash, "too_large", msg);
379                }
380                Err(e) => {
381                    id.hashes = Field::Unavailable(e.to_string());
382                    id.error(Tier::Hash, "hash_failed", e.to_string());
383                }
384            }
385        }
386        out
387    }
388}

390/// The SHA-256 of `bytes`, lowercase hex.
391fn sha256_hex(bytes: &[u8]) -> String {
392    let mut h = MultiHasher::new(AlgorithmSet::SHA256);
393    h.update(bytes);
394    h.finalize()
395        .get(HashAlgorithm::Sha256)
396        .map(ToString::to_string)
397        .unwrap_or_default()
398}

400/// The classify, hash and trust tiers.
401/// Returns the PE image's signature report, when the trust tier read one,
402/// for the OS cross-check.
403fn compose(
404    id: &mut FileIdentity,
405    content: Content,
406    tiers: Tiers,
407    ctx: &IdentityContext<'_>,
408) -> Option<SignatureReport> {
409    let info = content.bytes.as_deref().map(pe::parse);
410    let pe_info = match &info {
411        Some(Ok(p)) => Some(p),
412        _ => None,
413    };
414    // PE content, by the parse when the bytes were read, otherwise by the
415    // classifier's look at the headers (a large file, or the hash tier
416    // alone), so a missing Authenticode digest of a PE is `unavailable`,
417    // not `not_applicable`.
418    let is_pe = match &info {
419        Some(r) => r.is_ok(),
420        None => match &content.kind {
421            Some(k) => k.content_kind == FileKind::Pe,
422            None => content
423                .hashes
424                .as_ref()
425                .is_some_and(|h| h.authenticode.is_some()),
426        },
427    };
428    // Scripts, installer packages and app packages, chosen by the content
429    // the classifier saw, never by the name alone: a PE named `.ps1` is
430    // judged as a PE, so no script or package signature can vouch for it.
431    let non_pe = if is_pe || !(tiers.contains(Tier::Trust) || tiers.contains(Tier::Deep)) {
432        None
433    } else {
434        content
435            .kind
436            .as_ref()
437            .and_then(|k| NonPe::read(k, content.bytes.as_deref()))
438    };
439
440    if tiers.contains(Tier::Classify) {
441        match (&info, content.bytes.is_some()) {
442            (Some(Ok(p)), _) => {
443                id.pe = Field::Computed(pe_summary(p, id.name.file_name.as_deref()));
444            }
445            (Some(Err(_)), _) => id.pe = Field::NotApplicable,
446            (None, _) => id.pe = Field::Unavailable("the content could not be read".to_owned()),
447        }
448        id.kind = match &content.kind {
449            Some(c) => Field::Computed(kind_summary(c)),
450            None => Field::Unavailable("the content could not be read".to_owned()),
451        };
452        if content.bytes.is_some() && content.kind.is_some() {
453            id.computed_tiers.insert(Tier::Classify);
454        }
455    }
456
457    // The trust tier computes the digests too; they are reported whenever
458    // they exist.
459    if let Some(h) = &content.hashes {
460        let mut summary = hashes(h, is_pe);
461        // The normalised hashes need the content parsed, which the hash
462        // tier alone does not read: they are filled in where the reader
463        // ran, and are not applicable to other kinds.
464        let kind = content.kind.as_ref().map(|k| k.content_kind);
465        summary.msi_normalised_sha256 = match (&non_pe, kind) {
466            (Some(NonPe::Installer(Ok(i))), _) => Field::Computed(i.normalised_sha256.to_string()),
467            (Some(NonPe::Installer(Err(e))), _) => Field::Unavailable(e.message.clone()),
468            (_, Some(k)) if is_installer_kind(k) => Field::Unavailable(
469                "the installer was not read (request the trust or deep tier)".to_owned(),
470            ),
471            // The content kind was not determined (the hash tier alone):
472            // it may be an installer, unless it is a PE image.
473            (_, None) if !is_pe => Field::Unavailable(
474                "the file type was not determined (request the classify, trust or deep tier)"
475                    .to_owned(),
476            ),
477            _ => Field::NotApplicable,
478        };
479        summary.script_normalised_sha256 = match (&non_pe, kind) {
480            (Some(NonPe::Script(r)), _) => r.normalised_hash.as_ref().map_or_else(
481                || Field::Unavailable(script_unhashed_reason(r).to_owned()),
482                |d| Field::Computed(d.to_string()),
483            ),
484            (_, Some(FileKind::Script(s))) if script_kind_of(s).is_some() => Field::Unavailable(
485                "the script was not read (request the trust or deep tier)".to_owned(),
486            ),
487            (_, None) if !is_pe => Field::Unavailable(
488                "the file type was not determined (request the classify, trust or deep tier)"
489                    .to_owned(),
490            ),
491            _ => Field::NotApplicable,
492        };
493        id.hashes = Field::Computed(summary);
494        id.computed_tiers.insert(Tier::Hash);
495    }
496
497    let pe_report = if tiers.contains(Tier::Trust) {
498        trust_tier(id, &content, pe_info, non_pe.as_ref(), ctx)
499    } else {
500        None
501    };
502
503    if tiers.contains(Tier::Deep) {
504        deep_tier(id, &content, is_pe, non_pe.as_ref(), ctx);
505    }
506    pe_report
507}

509/// A non-PE format this crate reads signatures and identity from.
510enum NonPe {

512    Script(Box<ScriptReport>),
513    /// A Windows Installer package (#53), or why it could not be read.

516    Package(Result<Box<PackageIdentity>, Unread>),

519/// Why a container could not be read at all.
520#[derive(Debug, Clone)]
521struct Unread {
522    /// It is over the reader's size limit: not read, so not judged.
523    too_large: bool,
524    /// What went wrong, for reading.
525    message: String,
526}
527
528impl NonPe {
529    /// Reads `bytes` as the format `kind` names. `None` for any other
530    /// content, and when the bytes were not read (a reason is recorded:
531    /// the format's signatures are then not checked).
532    fn read(kind: &Classification, bytes: Option<&[u8]>) -> Option<Self> {
533        let content = kind.content_kind;
534        let script = match content {
535            FileKind::Script(s) => script_kind_of(s).map(|k| {
536                // `.vbe`/`.jse` and components share their content kind
537                // with a plain script; the extension tells them apart, and
538                // the script reader reports those unread.
539                match (k, kind.extension.as_deref()) {
540                    (ScriptKind::VbScript, Some("vbe")) => ScriptKind::EncodedVbScript,
541                    (ScriptKind::JScript, Some("jse")) => ScriptKind::EncodedJScript,
542                    (ScriptKind::PowerShell, Some("ps1xml" | "psc1" | "cdxml")) => {
543                        ScriptKind::PowerShellXml
544                    }
545                    (k, _) => k,
546                }
547            }),
548            _ => None,
549        };
550        let wanted = script.is_some()
551            || is_installer_kind(content)
552            || matches!(
553                content,
554                FileKind::Msix | FileKind::Appx | FileKind::MsixBundle | FileKind::AppxBundle
555            );
556        if !wanted {
557            return None;
558        }
559        // Not read into memory (too large): the trust tier records why.
560        let bytes = bytes?;
561        Some(if let Some(k) = script {
562            Self::Script(Box::new(verify_script(bytes, k, &ScriptOptions::default())))
563        } else if is_installer_kind(content) {
564            Self::Installer(parse_installer(bytes).map(Box::new).map_err(|e| Unread {
565                too_large: matches!(e, crate::installer::InstallerError::TooLarge(_)),
566                message: e.to_string(),
567            }))
568        } else {
569            Self::Package(parse_package(bytes).map(Box::new).map_err(|e| Unread {
570                too_large: matches!(e, crate::package::PackageError::TooLarge(_)),
571                message: e.to_string(),
572            }))
573        })
574    }
575}

586const fn script_kind_of(k: crate::filetype::ScriptKind) -> Option<ScriptKind> {
587    use crate::filetype::ScriptKind as F;
588    match k {
589        F::PowerShell => Some(ScriptKind::PowerShell),
590        F::VbScript => Some(ScriptKind::VbScript),
591        F::JScript => Some(ScriptKind::JScript),
592        F::Wsf => Some(ScriptKind::WindowsScriptFile),
593        F::Wsc => Some(ScriptKind::ScriptComponent),
594        _ => None,
595    }
596}
597
598fn script_unhashed_reason(r: &ScriptReport) -> &'static str {
599    if r.empty_text {
600        "the script's text is empty, so its digest identifies nothing"
601    } else if matches!(
602        r.signing_state,
603        ScriptSigningState::TooLargeForSignatureCheck { .. }
604    ) {
605        "the script is over the size limit, so it was not read"
606    } else if r.text_issue.is_some() {
607        "the script's text cannot be decoded without ambiguity"
608    } else if r.kind.signature_supported() {
609        "the script's signature block cannot be read"
610    } else {
611        "this script format is not read"
612    }
613}
614
615fn trust_tier(
616    id: &mut FileIdentity,
617    content: &Content,
618    pe_info: Option<&PeInfo>,
619    non_pe: Option<&NonPe>,
620    ctx: &IdentityContext<'_>,
621) -> Option<SignatureReport> {
622    let Some(hashes) = &content.hashes else {
623        let why = "the trust tier needs the file's digests, which were not computed";
624        id.signing_state = Field::Unavailable(why.to_owned());
625        id.signatures = Field::Unavailable(why.to_owned());
626        id.catalog_matches = Field::Unavailable(why.to_owned());
627        id.error(Tier::Trust, "no_digests", why);
628        return None;
629    };
630    let version = pe_info.and_then(|p| p.version_info.as_ref());
631
632    let mut digests = file_digests(hashes);
633    // A script or installer is also revoked, and catalog-listed, by the
634    // digest its signature signs: the script package's digest of the text,
635    // or the installer's content hash.
636    match non_pe {
637        Some(NonPe::Script(r)) => {
638            if let Some(sip) = &r.sip_digests {
639                digests.extend(sip.digests.iter().cloned());
640            }
641        }
642        Some(NonPe::Installer(Ok(i))) => digests.push(i.content_sha256.clone()),
643        _ => {}
644    }
645    // Embedded signatures (#48), chains (#49) and validity (#50).
646    let mut embedded = EmbeddedState::NotChecked;
647    let mut embedded_invalid_reason = None;
648    let mut sigs: Option<(SignatureReport, TrustReport, ValidityReport)> = None;
649    // A PE image whose Authenticode digests could not be computed has its
650    // signatures left unchecked, so its state stays unverified.
651    if let (Some(info), Some(bytes), Some(image)) = (pe_info, &content.bytes, &hashes.authenticode)
652    {
653        {
654            let report = verify_certificate_table(bytes, info, image, &VerifyOptions::default());
655            let trust = evaluate_trust(&report, ctx.code_trust, version);
656            let vctx = ValidityContext {
657                now: ctx.now,
658                clock_floor: ctx.clock_floor,
659                policy: ctx.validity,
660                revocation: ctx.revocation,
661                tsa_trust: ctx.tsa_trust,
662                file_digests: &digests,
663                version,
664            };
665            let validity = evaluate_validity(&report, &trust, &vctx);
666            embedded = match &report.signing_state {
667                EmbeddedSigning::Unsigned => EmbeddedState::Unsigned,
668                EmbeddedSigning::Valid => EmbeddedState::Valid,
669                EmbeddedSigning::Invalid(reason) => {
670                    embedded_invalid_reason = Some(invalid_reason(reason));
671                    EmbeddedState::Invalid
672                }
673            };
674            sigs = Some((report, trust, validity));
675        }
676    }
677    // Script, installer and package signatures (#52, #53): the same trust
678    // and validity evaluation as a PE file's, over the format's own report.
679    // A format that is not read, a file over its size limit, and a package
680    // the strict reader refuses never read as unsigned.
681    let mut package_publisher = None;
682    if let Some(non_pe) = non_pe {
683        let report = match non_pe {
684            NonPe::Script(r) => match &r.signing_state {
685                ScriptSigningState::TooLargeForSignatureCheck { size, limit } => {
686                    id.error(
687                        Tier::Trust,
688                        "signatures_unread",
689                        format!("{size} bytes is over the script signature limit of {limit}"),
690                    );
691                    None
692                }
693                _ if !r.kind.signature_supported() => None,
694                _ => Some(Ok(r.signatures.clone())),
695            },
696            NonPe::Installer(Ok(i)) => {
697                let mut report = i.signature.clone().unwrap_or_else(unsigned_report);
698                // Bytes after the compound file's last sector are outside
699                // the content hash: a signed package with data appended
700                // (a second file glued on, the CVE-2020-1464 pattern) is
701                // not what its signer signed, so no signature is valid.
702                if installer_trailing_bytes(i).is_some() && i.signature.is_some() {
703                    demote_unsigned_tail(&mut report);
704                }
705                Some(Ok(report))
706            }
707            NonPe::Package(Ok(p)) => {
708                package_publisher = Some(p);
709                Some(Ok(p
710                    .signature
711                    .as_ref()
712                    .map_or_else(unsigned_report, |s| s.report.clone())))
713            }
714            NonPe::Installer(Err(e)) | NonPe::Package(Err(e)) => Some(Err(e.clone())),
715        };
716        match report {
717            // Not read: the state stays unverified.
718            None => {}
719            Some(Err(Unread {
720                too_large: true,
721                message,
722            })) => {
723                id.error(Tier::Trust, "signatures_unread", message);
724            }
725            Some(Err(Unread { message, .. })) => {
726                // Any container the strict reader refuses is an error,
727                // never an unsigned file.
728                embedded = EmbeddedState::Invalid;
729                embedded_invalid_reason = Some(format!("unreadable: {message}"));
730            }
731            Some(Ok(report)) => {
732                let mut trust = match non_pe {
733                    NonPe::Script(r) => {
734                        evaluate_script_trust(
735                            r,
736                            &ScriptTrustAnchors {
737                                policy: ctx.code_trust,
738                                os_root_store: None,
739                            },
740                            ScriptAnchorPolicy::PolicyAnchors,
741                        )
742                        .trust
743                    }
744                    _ => evaluate_trust(&report, ctx.code_trust, None),
745                };
746                let vctx = ValidityContext {
747                    now: ctx.now,
748                    clock_floor: ctx.clock_floor,
749                    policy: ctx.validity,
750                    revocation: ctx.revocation,
751                    tsa_trust: ctx.tsa_trust,
752                    file_digests: &digests,
753                    version: None,
754                };
755                let mut validity = evaluate_validity(&report, &trust, &vctx);
756                // A packaged app's display name, only once its manifest's
757                // publisher is shown to be the signer (#53).
758                if let (Some(p), Some(identity)) = (package_publisher, validity.identity.as_mut()) {
759                    if let Some(name) = p.publisher_display_name.as_deref() {
760                        let _ = identity.attach_package_publisher(&p.publisher_dn, name);
761                    }
762                    if let Some(t) = trust.identity.as_mut() {
763                        t.package_display_name
764                            .clone_from(&identity.package_display_name);
765                    }
766                }
767                embedded = match &report.signing_state {
768                    EmbeddedSigning::Unsigned => EmbeddedState::Unsigned,
769                    EmbeddedSigning::Valid => EmbeddedState::Valid,
770                    EmbeddedSigning::Invalid(reason) => {
771                        embedded_invalid_reason = Some(match non_pe {
772                            NonPe::Script(r) => script_invalid_reason(r),
773                            NonPe::Installer(Ok(i)) => match installer_trailing_bytes(i) {
774                                Some(n) => format!(
775                                    "trailing data is not signed: {n} bytes follow the signed package"
776                                ),
777                                None => invalid_reason(reason),
778                            },
779                            NonPe::Package(Ok(p)) => p
780                                .signature
781                                .as_ref()
782                                .and_then(|s| s.issue.as_ref())
783                                .map_or_else(|| invalid_reason(reason), package_issue_name),
784                            _ => invalid_reason(reason),
785                        });
786                        EmbeddedState::Invalid
787                    }
788                };
789                sigs = Some((report, trust, validity));
790            }
791        }
792    }
793
794    // Catalogs (#51). One lookup serves the summary and the signer. A
795    // script is listed by its script package digest and an installer by
796    // its content hash; whether those match catalogs made by OS tooling is
797    // not yet checked (#51), so for them a miss proves nothing.
798    let (catalog_digests, miss_proves_nothing) = match non_pe {
799        Some(NonPe::Script(r)) => (r.catalog_digests(), true),
800        Some(NonPe::Installer(Ok(i))) => (
801            FileDigests {
802                flat_sha256: Some(i.content_sha256.clone()),
803                ..FileDigests::default()
804            },
805            true,
806        ),
807        _ => (
808            FileDigests::from_hashes(hashes.authenticode.as_ref(), Some(&hashes.flat)),
809            false,
810        ),
811    };
812    // A file revoked by hash is never vouched for, by any signer.
813    let file_revoked = ctx.revocation.file_revoked(&digests).is_some();
814    let found = ctx.catalogs.map(|store| store.lookup(&catalog_digests));
815    // The time revocation is judged at: the later of the clock and the
816    // floor, as for embedded signatures (ADR 0012).
817    let effective_now = ctx.clock_floor.map_or(ctx.now, |f| f.max(ctx.now));
818    // One budget of CRL signature checks for every catalog signer.
819    let mut checker = ctx.revocation.checker();
820    let mut matches = Vec::new();
821    // Which matches a revoked catalog signer disqualifies.
822    let mut usable = Vec::new();
823    let catalog = match (ctx.catalogs, &found) {
824        (Some(store), Some(found)) => {
825            for m in &found.matches {
826                let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
827                let signer_revoked =
828                    chain.is_some_and(|c| chain_revoked(c, ctx, &mut checker, effective_now));
829                // A catalog vouches only when its signer is trusted and its
830                // signature's validity (expiry, revocation of its own hash,
831                // key sizes, timestamps) was judged and passed: a catalog
832                // whose validity was never judged does not vouch (#50).
833                let validity_ok = m.usable();
834                usable.push(validity_ok && !m.relies_on_sha1() && !signer_revoked && !file_revoked);
835                matches.push(CatalogMatchSummary {
836                    catalog_id: m.record.id.as_ref().map(Digest::to_string),
837                    catalog_path: m.record.path.as_ref().map(|p| p.display().to_string()),
838                    source: match m.record.source {
839                        CatalogSource::System => "system",
840                        CatalogSource::DriverStore => "driver_store",
841                        CatalogSource::Policy => "policy",
842                    },
843                    member: m.member,
844                    matched_by: match m.matched_by {
845                        DigestKind::AuthenticodeSha256 => "authenticode_sha256",
846                        DigestKind::AuthenticodeSha1 => "authenticode_sha1",
847                        DigestKind::FlatSha256 => "sha256",
848                        DigestKind::FlatSha1 => "sha1",
849                    },
850                    trusted: m.trusted(),
851                    validity: m.record.validity.as_ref().map(|v| v.error),
852                    relies_on_sha1: m.relies_on_sha1(),
853                    signer_revoked,
854                    publisher_name: m
855                        .signer
856                        .and_then(|s| s.identity.as_ref())
857                        .and_then(|i| i.publisher_name.as_ref())
858                        .map(|p| p.value.clone()),
859                    signer: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
860                    signer_chain: chain.map(chain_briefs).unwrap_or_default(),
861                    signer_chain_complete: chain.is_some_and(|c| c.chain_complete),
862                });
863            }
864            if found.truncated {
865                id.error(
866                    Tier::Trust,
867                    "catalog_matches_truncated",
868                    "more catalogs list the file than are reported",
869                );
870            }
871            // A store whose index was cut short cannot show that a file is
872            // not listed.
873            let index_truncated = store.records().iter().any(|r| r.index_truncated);
874            if usable.iter().any(|u| *u) {
875                CatalogState::ListedTrusted
876            } else if found.matches.iter().zip(&matches).any(|(m, summary)| {
877                m.usable() && m.relies_on_sha1() && !summary.signer_revoked && !file_revoked
878            }) {
879                CatalogState::ListedTrustedSha1Only
880            } else if !found.matches.is_empty() {
881                CatalogState::ListedUntrusted
882            } else if index_truncated {
883                id.error(
884                    Tier::Trust,
885                    "catalog_index_truncated",
886                    "the catalog index is incomplete, so a missing match proves nothing",
887                );
888                CatalogState::NotChecked
889            } else if miss_proves_nothing {
890                CatalogState::NotChecked
891            } else {
892                CatalogState::NotListed
893            }
894        }
895        _ => CatalogState::NotChecked,
896    };
897
898    // The signer that vouches for the file: a fully valid embedded
899    // signature first, then the best trusted catalog that does not rely on
900    // SHA-1.
901    let mut signer = None;
902    let mut embedded_sha1_only = false;
903    let mut summaries = Vec::new();
904    let mut verification_error = if file_revoked {
905        VerificationError::RevokedByHash
906    } else {
907        VerificationError::NotSigned
908    };
909    let mut clock_suspect = ctx.clock_floor.is_some_and(|f| ctx.now < f);
910    if let Some((report, trust, validity)) = &sigs {
911        verification_error = validity.error;
912        clock_suspect |= validity.clock_suspect;
913        for sig in &report.signatures {
914            let t = trust.signers.iter().find(|s| s.signature == sig.index);
915            let v = validity
916                .signatures
917                .iter()
918                .find(|v| v.signature == sig.index);
919            summaries.push(SignatureSummary {
920                index: sig.index,
921                kind: match sig.kind {
922                    SignatureKind::Nested => "nested",
923                    SignatureKind::Catalog => "catalog",
924                    SignatureKind::Script => "script",
925                    SignatureKind::Msi => "msi",
926                    SignatureKind::Package => "package",
927                    _ => "embedded",
928                },
929                depth: sig.depth,
930                parent: sig.parent,
931                status: status_name(&sig.status),
932                status_detail: format!("{:?}", sig.status),
933                digest_alg: sig.digest_alg,
934                signer_digest_alg: sig.signer_digest_alg,
935                sig_alg: sig.sig_alg.clone(),
936                key_alg: sig.key_alg.clone(),
937                key_bits: sig.key_bits,
938                digest_match: sig.digest_match,
939                program_name: sig.signer.as_ref().and_then(|s| s.program_name.clone()),
940                more_info_url: sig.signer.as_ref().and_then(|s| s.more_info_url.clone()),
941                claimed_signing_time: sig.signer.as_ref().and_then(|s| s.signing_time),
942                has_page_hashes: sig.page_hashes.is_some(),
943                // The signature and, when a chain was built, every
944                // certificate on it (#49).
945                os_ci_compatible: t.map_or_else(|| sig.os_ci_compatible(), |t| t.os_ci_compatible),
946                trusted: t.is_some_and(|t| t.trusted),
947                untrusted: t.map(|t| t.untrusted.clone()).unwrap_or_default(),
948                validity_error: v.map_or(VerificationError::Generic, |v| v.error),
949                trusted_signing_time: v.and_then(|v| v.signing_time),
950                timestamps: sig.timestamp_tokens.len(),
951                rfc3161_timestamp: v.and_then(|v| rfc3161_verified(&v.timestamps)),
952                expired_now: t
953                    .and_then(|t| t.chain.as_ref())
954                    .map(|c| chain_expired_at(c, validity.effective_now)),
955                chain: t.and_then(|t| t.chain.as_ref()).map(chain_summary),
956            });
957        }
958        // A primary signature that rests on SHA-1 (its content digest or
959        // its signer digest) is valid, but no allow decision may rest on it
960        // (ADR 0008, ADR 0017): it is not a trusted signer, as a catalog
961        // match that relies on SHA-1 is not.
962        let sha2_primary = validity
963            .primary
964            .and_then(|p| report.signatures.get(p))
965            .is_some_and(|sig| {
966                sig.digest_alg.is_some_and(DigestAlgorithm::is_sha2)
967                    && sig.signer_digest_alg.is_some_and(DigestAlgorithm::is_sha2)
968            });
969        embedded_sha1_only = validity.primary.is_some() && !sha2_primary;
970        if let (Some(p), Some(identity), false, true) = (
971            validity.primary,
972            &validity.identity,
973            file_revoked,
974            sha2_primary,
975        ) {
976            let chain = trust
977                .signers
978                .iter()
979                .find(|s| s.signature == p)
980                .and_then(|s| s.chain.as_ref());
981            signer = Some(TrustedSigner {
982                source: TrustSource::Embedded,
983                signature: Some(p),
984                catalog_match: None,
985                publisher_name: identity.publisher_name.as_ref().map(|n| n.value.clone()),
986                publisher_dn: identity.publisher_dn.clone(),
987                product_name: identity.product_name.clone(),
988                leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
989                pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
990                root: chain.and_then(full_root).map(cert_brief),
991                well_known_root_id: chain.and_then(|c| c.well_known_root_id),
992                anchored_to: chain
993                    .and_then(|c| c.anchored_to.as_ref())
994                    .map(|a| a.label.clone()),
995                package_display_name: identity.package_display_name.clone(),
996            });
997        }
998    }
999    // The product of a non-PE file comes from what its signature covers:
1000    // a script's signed metadata, an installer's `ProductName` (the
1001    // signature covers every stream). A package has none: its family name
1002    // is its product identity.
1003    if let Some(s) = signer.as_mut() {
1004        let product = match non_pe {
1005            Some(NonPe::Script(r)) => r
1006                .signature()
1007                .and_then(|sig| sig.signer.as_ref())
1008                .and_then(|a| a.signed_metadata.product.clone()),
1009            Some(NonPe::Installer(Ok(i))) => i.properties.product_name.clone(),
1010            _ => None,
1011        };
1012        if non_pe.is_some() {
1013            s.product_name =
1014                product.filter(|p| !p.trim().is_empty() && !p.chars().any(char::is_control));
1015        }
1016    }
1017    if signer.is_none()
1018        && let Some(found) = &found
1019    {
1020        let best = found
1021            .matches
1022            .iter()
1023            .enumerate()
1024            .find(|(i, _)| usable.get(*i).copied().unwrap_or(false));
1025        if let Some((i, m)) = best {
1026            let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
1027            let identity = m.signer.and_then(|s| s.identity.as_ref());
1028            signer = Some(TrustedSigner {
1029                source: TrustSource::Catalog,
1030                signature: None,
1031                catalog_match: Some(i),
1032                publisher_name: identity
1033                    .and_then(|i| i.publisher_name.as_ref())
1034                    .map(|n| n.value.clone()),
1035                publisher_dn: identity.and_then(|i| i.publisher_dn.clone()),
1036                product_name: version
1037                    .and_then(|v| v.get(VersionField::ProductName))
1038                    .filter(|p| !p.trim().is_empty())
1039                    .map(str::to_owned),
1040                leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
1041                pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
1042                root: chain.and_then(full_root).map(cert_brief),
1043                well_known_root_id: chain.and_then(|c| c.well_known_root_id),
1044                anchored_to: chain
1045                    .and_then(|c| c.anchored_to.as_ref())
1046                    .map(|a| a.label.clone()),
1047                package_display_name: None,
1048            });
1049        }
1050    }
1051
1052    // A valid embedded signature whose signer was revoked (a certificate in
1053    // its chain, or its own hash) is no weaker evidence of tampering than a
1054    // broken one: it is invalid, never merely untrusted.
1055    let revoked = matches!(
1056        verification_error,
1057        VerificationError::Revoked | VerificationError::RevokedByHash
1058    );
1059    let state = signing_state_kind(
1060        signer.is_some(),
1061        file_revoked || (embedded == EmbeddedState::Valid && signer.is_none() && revoked),
1062        embedded,
1063        catalog,
1064    );
1065
1066    id.signing_state = Field::Computed(SigningState {
1067        state,
1068        embedded,
1069        embedded_invalid_reason,
1070        catalog,
1071        verification_error,
1072        clock_suspect,
1073        embedded_sha1_only,
1074        signer,
1075    });
1076    if sigs.is_some() {
1077        id.signatures = Field::Computed(summaries);
1078    } else if content.bytes.is_none() {
1079        let why = "the file was not read into memory, so its signatures were not read";
1080        id.signatures = Field::Unavailable(why.to_owned());
1081        id.error(Tier::Trust, "signatures_unread", why);
1082    } else if pe_info.is_some() {
1083        let why = "the image's Authenticode digests could not be computed";
1084        id.signatures = Field::Unavailable(why.to_owned());
1085        id.error(Tier::Trust, "signatures_unread", why);
1086    } else {
1087        // A format with no signature reader here (a cabinet, a Java
1088        // archive, an encoded script): its signatures, if it has any, are
1089        // not read, so it stays unverified.
1090        id.signatures = Field::Unavailable("this file type's signatures are not read".to_owned());
1091    }
1092    id.catalog_matches = match ctx.catalogs {
1093        Some(_) => Field::Computed(matches),
1094        None => Field::NotRequested,
1095    };
1096    if id.signatures.is_computed() || (content.bytes.is_some() && pe_info.is_none()) {
1097        id.computed_tiers.insert(Tier::Trust);
1098    }
1099    sigs.filter(|_| pe_info.is_some())
1100        .map(|(report, _, _)| report)
1101}

1103/// Whether a catalog signer's chain is revoked, or its revocation state is
1104/// unknown where policy says that fails. Catalog signatures carry no
1105/// trusted signing time here (catalog timestamps are #50), so every entry
1106/// counts, whatever its date.
1107fn chain_revoked(
1108    chain: &ChainReport,
1109    ctx: &IdentityContext<'_>,
1110    checker: &mut RevocationChecker<'_>,
1111    now: i64,
1112) -> bool {
1113    let certs: Vec<&CertSummary> = chain.full_chain_certs().collect();
1114    (0..certs.len()).any(|i| match checker.state_of(&certs, i, now) {
1115        CertRevocation::Good => false,
1116        CertRevocation::Unknown => ctx.revocation.failure == RevocationFailure::FailClosed,
1117        // Revoked, or a CRL that was not checked: both fail closed.
1118        _ => true,
1119    })
1120}

1122/// The context tier: mark-of-the-web and path variables. Owner,
1123/// writability, streams and reparse points are #54.
1124fn context_tier(
1125    id: &mut FileIdentity,
1126    file: &std::fs::File,
1127    tiers: Tiers,
1128    ctx: &IdentityContext<'_>,
1129) {
1130    if !tiers.contains(Tier::Context) {
1131        return;
1132    }
1133    // The Palisade provenance tag has no reader yet (#54).
1134    id.fs.provenance = Field::pending(&[54]);
1135    read_fs_details(id, file);
1136    // The stream is read for the opened file itself: by the handle's final
1137    // path, and kept only if that path still names the same file, so the
1138    // record never pairs this file's content with another file's mark.
1139    // A named stream has no mark of its own: `Zone.Identifier` is a stream
1140    // of the host file, and the host's mark would credit the stream with a
1141    // download origin it may not share.
1142    id.fs.motw = if id.name.stream.is_some() {
1143        Field::NotApplicable
1144    } else {
1145        read_motw(file)
1146    };
1147    if let Field::Unavailable(why) = &id.fs.motw
1148        && cfg!(all(windows, feature = "os-trust"))
1149    {
1150        let why = why.clone();
1151        id.error(Tier::Context, "motw_unreadable", why);
1152    }
1153    id.fs.path_variables = match (ctx.known_folders, id.fs.path.computed()) {
1154        (None, _) => Field::Unavailable("no known folders in the context".to_owned()),
1155        (Some(folders), Some(info)) => match info.rule_path() {
1156            Some(p) => match path_variables_matched(&p, folders, None) {
1157                Ok(vars) => Field::Computed(vars.iter().map(|v| v.token().to_owned()).collect()),
1158                Err(e) => Field::Unavailable(e.to_string()),
1159            },
1160            None => Field::Unavailable("the path is not a full Windows path".to_owned()),
1161        },
1162        (Some(_), None) => Field::Unavailable("no path".to_owned()),
1163    };
1164    // The tier counts as computed only when it computed something: with
1165    // every field pending or unavailable it did not.
1166    let fs = &id.fs;
1167    if fs.motw.is_computed()
1168        || fs.path_variables.is_computed()
1169        || fs.file_id.is_computed()
1170        || fs.volume.is_computed()
1171        || fs.owner.is_computed()
1172        || fs.user_writable.is_computed()
1173        || fs.streams.is_computed()
1174        || fs.reparse.is_computed()
1175    {
1176        id.computed_tiers.insert(Tier::Context);
1177    }
1178}

1202/// The file ID, volume, owner, writability, streams and reparse point of
1203/// the opened file, from the OS (#54). Every read is of the open handle, or
1204/// of the path the handle's own final path names, so nothing describes
1205/// another file. A read that fails leaves its field unavailable.
1206#[cfg(all(windows, feature = "os-trust"))]
1207fn read_fs_details(id: &mut FileIdentity, file: &std::fs::File) {
1208    use crate::fsctx::writable::{AssessOptions, WritabilityCache};
1209    use crate::win::{access, links, security::NameCache, volume};
1210    let unavailable = |e: &io::Error| e.to_string();
1211
1212    // The real path of the opened file: every path-based read below uses
1213    // it, never the path as given (a link could lead elsewhere).
1214    let final_path = links::final_path_of(file).map(|p| p.to_win32_string());
1215
1216    id.fs.file_id = match links::read_meta(file) {
1217        Ok(meta) => {
1218            let (count, names) = match &final_path {
1219                Ok(p) => links::hard_links(Path::new(p), file)
1220                    .map_or((None, Vec::new()), |h| (Some(h.count), h.names.clone())),
1221                Err(_) => (None, Vec::new()),
1222            };
1223            Field::Computed(super::record::FileIdSummary {
1224                volume_serial: meta.volume_serial,
1225                file_id: meta
1226                    .file_id
1227                    .map(|b| b.iter().map(|x| format!("{x:02x}")).collect()),
1228                usn_journal_id: meta.usn.map(|(j, _)| j),
1229                usn: meta.usn.map(|(_, u)| u),
1230                link_count: count,
1231                link_names: names,
1232            })
1233        }
1234        Err(e) => Field::Unavailable(unavailable(&e)),
1235    };
1236
1237    let vol = volume::read_volume(file);
1238    id.fs.volume = Field::Computed(volume_summary(&vol));
1239
1240    let mut names = NameCache::default();
1241    let local = matches!(
1242        vol.location_class(),
1243        crate::fsctx::LocationClass::OsVolume | crate::fsctx::LocationClass::OtherFixed
1244    );
1245    id.fs.owner = match access::read_owner(file, vol.fs_type.as_ref(), local, &mut names) {
1246        Ok(o) => Field::Computed(super::record::OwnerSummary {
1247            sid: o.owner_sid.map(|s| s.to_string()),
1248            name: o.owner_name.clone(),
1249            trust_class: snake(&format!("{:?}", o.owner_trust_class)),
1250            supported: o.owner_supported,
1251        }),
1252        Err(e) => Field::Unavailable(unavailable(&e)),
1253    };
1254
1255    // The access check reads the target's descriptor through the open
1256    // handle, and its folders from the handle's own final path, so the
1257    // verdict is about this file even if its path is swapped meanwhile.
1258    id.fs.user_writable = match access::assess_open_file(
1259        file,
1260        &crate::fsctx::EvaluationOptions::default(),
1261        AssessOptions::default(),
1262        &mut WritabilityCache::default(),
1263    ) {
1264        Ok(w) => Field::Computed(super::record::WritableSummary {
1265            path_user_writable: w.path_user_writable(),
1266            file_user_writable: w.file_user_writable,
1267            first_user_writable_ancestor: w.first_user_writable_ancestor.clone(),
1268            change_perms_by_non_admin: w.change_perms_by_non_admin,
1269            principals_with_write: w
1270                .principals_with_write
1271                .iter()
1272                .map(ToString::to_string)
1273                .collect(),
1274        }),
1275        Err(e) => Field::Unavailable(unavailable(&e)),
1276    };
1277
1278    id.fs.streams = match links::read_streams(file) {
1279        Ok(s) => Field::Computed(super::record::StreamsSummary {
1280            streams: s
1281                .streams
1282                .iter()
1283                .map(|st| super::record::StreamSummary {
1284                    name: st.name.clone(),
1285                    size: st.size,
1286                    benign: st.benign.map(|b| snake(&format!("{b:?}"))),
1287                    executable_content: st.is_executable_content,
1288                })
1289                .collect(),
1290            truncated: s.truncated,
1291        }),
1292        Err(e) => Field::Unavailable(unavailable(&e)),
1293    };
1294
1295    // The reparse point of the path as given: the opened handle followed
1296    // it, so the link itself is opened without following.
1297    id.fs.reparse = match id.path.as_deref() {
1298        Some(p) => {
1299            match links::open_link_itself(Path::new(p)).and_then(|f| links::read_reparse(&f)) {
1300                Ok(r) => Field::Computed(r.map(|r| reparse_summary(&r))),
1301                Err(e) => Field::Unavailable(unavailable(&e)),
1302            }
1303        }
1304        None => Field::NotApplicable,
1305    };
1306
1307    // Every field that could not be read is an error of the tier, as the
1308    // `Field::Unavailable` contract says.
1309    let failures: Vec<(&'static str, String)> = [
1310        ("file_id_unreadable", unavailable_reason(&id.fs.file_id)),
1311        ("owner_unreadable", unavailable_reason(&id.fs.owner)),
1312        (
1313            "writability_unreadable",
1314            unavailable_reason(&id.fs.user_writable),
1315        ),
1316        ("streams_unreadable", unavailable_reason(&id.fs.streams)),
1317        ("reparse_unreadable", unavailable_reason(&id.fs.reparse)),
1318    ]
1319    .into_iter()
1320    .filter_map(|(code, why)| why.map(|w| (code, w)))
1321    .collect();
1322    for (code, why) in failures {
1323        id.error(Tier::Context, code, why);
1324    }
1325}

1598/// `CamelCase` to `snake_case`, for the `Debug` names of #44's enums.
1599fn snake(debug: &str) -> String {
1600    let mut out = String::with_capacity(debug.len() + 4);
1601    for (i, c) in debug.chars().enumerate() {
1602        if c.is_ascii_uppercase() {
1603            if i > 0 {
1604                out.push('_');
1605            }
1606            out.push(c.to_ascii_lowercase());
1607        } else {
1608            out.push(c);
1609        }
1610    }
1611    out
1612}

1829/// The deep tier: the installer, package and script records (#52, #53).
1830fn deep_tier(
1831    id: &mut FileIdentity,
1832    content: &Content,
1833    is_pe: bool,
1834    non_pe: Option<&NonPe>,
1835    ctx: &IdentityContext<'_>,
1836) {
1837    const NOT_READ: &str = "the content could not be read";
1838    id.installer = Field::NotApplicable;
1839    id.package = Field::NotApplicable;
1840    id.script = Field::NotApplicable;
1841    id.content_tags = content_tags(content.bytes.as_deref(), ctx);
1842    if let Field::Unavailable(why) = &id.content_tags
1843        && content.bytes.is_some()
1844    {
1845        let why = why.clone();
1846        id.error(Tier::Deep, "content_tags_unavailable", why);
1847    }
1848    if content.bytes.is_none() {
1849        // Too large to read, or a read that failed: what it is may still
1850        // be one of these, so nothing is "not applicable".
1851        let kind = content.kind.as_ref().map(|k| k.content_kind);
1852        match kind {
1853            Some(k) if is_installer_kind(k) => id.installer = Field::Unavailable(NOT_READ.into()),
1854            Some(FileKind::Msix | FileKind::Appx | FileKind::MsixBundle | FileKind::AppxBundle) => {
1855                id.package = Field::Unavailable(NOT_READ.into());
1856            }
1857            Some(FileKind::Script(s)) if script_kind_of(s).is_some() => {
1858                id.script = Field::Unavailable(NOT_READ.into());
1859            }
1860            Some(_) => {}
1861            None => {
1862                id.installer = Field::Unavailable(NOT_READ.into());
1863                id.package = Field::Unavailable(NOT_READ.into());
1864                id.script = Field::Unavailable(NOT_READ.into());
1865            }
1866        }
1867        return;
1868    }
1869    // A PE image has no installer, package or script record; its deep
1870    // tier is its content tags.
1871    if is_pe {
1872        if id.content_tags.is_computed() {
1873            id.computed_tiers.insert(Tier::Deep);
1874        }
1875        return;
1876    }
1877    match non_pe {
1878        Some(NonPe::Script(r)) => id.script = Field::Computed(script_summary(r)),
1879        Some(NonPe::Installer(Ok(i))) => id.installer = Field::Computed(installer_summary(i)),
1880        Some(NonPe::Package(Ok(p))) => id.package = Field::Computed(package_summary(p)),
1881        Some(NonPe::Installer(Err(e))) => {
1882            id.installer = Field::Unavailable(e.message.clone());
1883            id.error(Tier::Deep, "installer_unreadable", e.message.clone());
1884        }
1885        Some(NonPe::Package(Err(e))) => {
1886            id.package = Field::Unavailable(e.message.clone());
1887            id.error(Tier::Deep, "package_unreadable", e.message.clone());
1888        }
1889        None => {}
1890    }
1891    if id.installer.is_computed()
1892        || id.package.is_computed()
1893        || id.script.is_computed()
1894        || id.content_tags.is_computed()
1895    {
1896        id.computed_tiers.insert(Tier::Deep);
1897    }
1898}

2222) -> SigningStateKind {
2223    if trusted_signer {
2224        SigningStateKind::Trusted
2225    } else if revoked || embedded == EmbeddedState::Invalid {
2226        // A broken embedded signature, or a revoked file or signer, is
2227        // invalid whatever a catalog says: an untrusted catalog listing
2228        // does not make it merely "untrusted".
2229        SigningStateKind::Invalid
2230    } else if embedded == EmbeddedState::NotChecked
2231        || (embedded == EmbeddedState::Unsigned && catalog == CatalogState::NotChecked)
2232    {
2233        SigningStateKind::Unverified
2234    } else if embedded == EmbeddedState::Valid
2235        || matches!(
2236            catalog,
2237            CatalogState::ListedUntrusted | CatalogState::ListedTrustedSha1Only
2238        )
2239    {
2240        SigningStateKind::Untrusted
2241    } else {
2242        SigningStateKind::Unsigned
2243    }
2244}
