40/// Everything [`identify`] needs besides the file: trust anchors, the
41/// catalog index, revocation data, the protected-binary list, the clock
42/// and options. The library keeps no state of its own.
43#[derive(Debug, Clone)]
44pub struct IdentityContext<'a> {
45    /// Anchors for code signers.
46    pub code_trust: &'a TrustConfig,
47    /// Anchors for timestamping authorities.
48    pub tsa_trust: &'a TrustConfig,
49    /// The catalogs to look the file up in. `None` skips the lookup.
50    pub catalogs: Option<&'a CatalogStore>,
51    /// Revocation data.
52    pub revocation: &'a RevocationSet,
53    /// The validity settings.
54    pub validity: &'a ValidityPolicy,
55    /// The current time, in seconds since the Unix epoch.
56    pub now: i64,
57    /// A lower bound on real time (see [`ValidityContext::clock_floor`]).
58    pub clock_floor: Option<i64>,
59    /// The protected-binary and protected-vendor lists.
60    pub protected: &'a ProtectedList,
61    /// The machine's known folders, for path variables and the protected
62    /// binaries' expected locations. `None` leaves those unassessed.
63    pub known_folders: Option<&'a KnownFolders>,
64    /// Which flat hashes to compute. SHA-256 is always added.
65    pub flat_algorithms: AlgorithmSet,
66    /// Hashing limits.
67    pub hash_options: HashOptions,
68    /// Files larger than this are not read into memory, so their PE
69    /// metadata and signatures are not read; their hashes are still
70    /// streamed. Default [`MAX_IN_MEMORY`].
71    pub max_in_memory: u64,
72}
73
74impl IdentityContext<'_> {
75    /// A context that trusts nothing, looks nothing up and uses the built-in
76    /// protected list, at time `now`. For tools and tests; the agent builds
77    /// its context from policy.
78    #[must_use]
79    pub fn untrusted(now: i64) -> IdentityContext<'static> {
80        static EMPTY_TRUST: std::sync::OnceLock<TrustConfig> = std::sync::OnceLock::new();
81        static EMPTY_REVOCATION: std::sync::OnceLock<RevocationSet> = std::sync::OnceLock::new();
82        static DEFAULT_VALIDITY: std::sync::OnceLock<ValidityPolicy> = std::sync::OnceLock::new();
83        static DEFAULT_LIST: std::sync::OnceLock<ProtectedList> = std::sync::OnceLock::new();
84        let trust = EMPTY_TRUST.get_or_init(TrustConfig::new);
85        IdentityContext {
86            code_trust: trust,
87            tsa_trust: trust,
88            catalogs: None,
89            revocation: EMPTY_REVOCATION.get_or_init(RevocationSet::new),
90            validity: DEFAULT_VALIDITY.get_or_init(ValidityPolicy::default),
91            now,
92            clock_floor: None,
93            protected: DEFAULT_LIST.get_or_init(ProtectedList::defaults),
94            known_folders: None,
95            flat_algorithms: AlgorithmSet::DEFAULT,
96            hash_options: HashOptions::default(),
97            max_in_memory: MAX_IN_MEMORY,
98        }
99    }
100}

102/// Why [`identify`] produced no record at all. Anything after the file is
103/// opened is recorded in [`FileIdentity::errors`] instead.
104#[derive(Debug)]
105#[non_exhaustive]
106pub enum IdentifyError {
107    /// The file could not be opened or its metadata read.
108    Open(io::Error),
109    /// The path is not a regular file.
110    NotAFile,
111}
112
113impl std::fmt::Display for IdentifyError {
114    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
115        match self {
116            Self::Open(e) => write!(f, "cannot open the file: {e}"),
117            Self::NotAFile => f.write_str("not a regular file"),
118        }
119    }
120}

124/// Computes the identity of the file at `path`, for the requested tiers.
125///
126/// The file is opened once, without following it as a FIFO, and every
127/// tier reads that one handle, so all attributes describe the same file.
128/// The trust tier needs the hash tier's digests and computes them too.
129/// Failures within a tier are recorded in [`FileIdentity::errors`] and
130/// leave that tier's fields [`Field::Unavailable`]; nothing in the record
131/// is ever guessed.
132pub fn identify(
133    path: &Path,
134    tiers: Tiers,
135    ctx: &IdentityContext<'_>,
136) -> Result<FileIdentity, IdentifyError> {
137    if !std::fs::metadata(path)
138        .map_err(IdentifyError::Open)?
139        .is_file()
140    {
141        return Err(IdentifyError::NotAFile);
142    }
143    let mut file = crate::hash::open_without_blocking(path).map_err(IdentifyError::Open)?;
144    let meta = file.metadata().map_err(IdentifyError::Open)?;
145    if !meta.is_file() {
146        return Err(IdentifyError::NotAFile);
147    }
148    let text = path.to_string_lossy().into_owned();
149    let mut id = FileIdentity::empty(Some(text.clone()), None, tiers);
150    let len = meta.len();
151
152    // T0 always runs: it costs one stat, and every other tier needs the
153    // size.
154    id.size = Field::Computed(len);
155    id.timestamps = Field::Computed(timestamps(&meta));
156    // The final path is the open handle's, so it describes the file whose
157    // content is read below even if the path is replaced meanwhile.
158    id.fs.path = Field::Computed(path_info(&text, false, Some(&file)));
159    id.computed_tiers.insert(Tier::Stat);
160
161    let content = Content::read(&mut file, len, tiers, ctx, &mut id);
162    compose(&mut id, content, tiers, ctx);
163    context_tier(&mut id, &file, tiers, ctx);
164    finish(&mut id, ctx);
165    Ok(id)
166}

168/// Computes the identity of an uploaded file held in memory. `name_hint`
169/// is the name the uploader gave, which is untrusted: it is used for the
170/// name checks only, never as a location. There is no file system context.
171#[must_use]
172pub fn identify_bytes(
173    bytes: &[u8],
174    name_hint: Option<&str>,
175    tiers: Tiers,
176    ctx: &IdentityContext<'_>,
177) -> FileIdentity {
178    let mut id = FileIdentity::empty(None, name_hint.map(str::to_owned), tiers);
179    let len = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
180    id.size = Field::Computed(len);
181    id.timestamps = Field::NotApplicable;
182    id.fs.path = name_hint.map_or(Field::NotApplicable, |n| {
183        Field::Computed(path_info(n, true, None))
184    });
185    id.computed_tiers.insert(Tier::Stat);
186    let mut cursor = io::Cursor::new(bytes);
187    let content = Content::read(&mut cursor, len, tiers, ctx, &mut id);
188    compose(&mut id, content, tiers, ctx);
189    if tiers.contains(Tier::Context) {
190        id.fs.not_applicable_context();
191    }
192    finish(&mut id, ctx);
193    id
194}

196/// What was read from the file's content.
197struct Content {
198    /// The whole file, when a tier needs it and it fits
199    /// [`IdentityContext::max_in_memory`].
200    bytes: Option<Vec<u8>>,
201    /// The hashes, from one streaming pass.
202    hashes: Option<ImageHashes>,
203    /// The content classification (#44). It reads only the head of the
204    /// file and a few bounded targeted reads, so it runs on large files
205    /// too.
206    kind: Option<Classification>,
207}
208
209impl Content {
210    /// Reads the content once. When a tier needs the bytes and the file
211    /// fits [`IdentityContext::max_in_memory`], it is read into memory and hashed from
212    /// there, so the PE metadata, the digests and the signatures all
213    /// describe exactly the same bytes. A larger file is only streamed
214    /// through the hashes.
215    fn read<R: Read + Seek>(
216        r: &mut R,
217        len: u64,
218        tiers: Tiers,
219        ctx: &IdentityContext<'_>,
220        id: &mut FileIdentity,
221    ) -> Self {
222        let need_bytes = tiers.contains(Tier::Classify) || tiers.contains(Tier::Trust);
223        let need_hashes = tiers.contains(Tier::Hash) || tiers.contains(Tier::Trust);
224        let mut out = Self {
225            bytes: None,
226            hashes: None,
227            kind: None,
228        };
229        if need_bytes {
230            if len > ctx.max_in_memory {
231                let msg = format!(
232                    "{len} bytes is over the in-memory limit of {}",
233                    ctx.max_in_memory
234                );
235                for tier in [Tier::Classify, Tier::Trust] {
236                    if tiers.contains(tier) {
237                        id.error(tier, "too_large", msg.clone());
238                    }
239                }
240            } else {
241                let mut buf = Vec::with_capacity(usize::try_from(len).unwrap_or(0));
242                let read = r
243                    .seek(SeekFrom::Start(0))
244                    .and_then(|_| r.take(len.saturating_add(1)).read_to_end(&mut buf));
245                let failure = match read {
246                    Ok(_) if u64::try_from(buf.len()).ok() == Some(len) => {
247                        id.content_sha256 = Some(sha256_hex(&buf));
248                        out.bytes = Some(buf);
249                        None
250                    }
251                    Ok(_) => Some((
252                        "changed_while_read",
253                        "the file changed size while it was read".to_owned(),
254                    )),
255                    Err(e) => Some(("read_failed", e.to_string())),
256                };
257                if let Some((code, msg)) = failure {
258                    // Charged to every requested tier that needed the
259                    // bytes or would have hashed them; nothing describes a
260                    // file that changed under the read.
261                    for tier in [Tier::Classify, Tier::Hash, Tier::Trust] {
262                        if tiers.contains(tier) {
263                            id.error(tier, code, msg.clone());
264                        }
265                    }
266                    if need_hashes {
267                        id.hashes = Field::Unavailable(msg);
268                    }
269                    return out;
270                }
271            }
272        }
273        if tiers.contains(Tier::Classify) {
274            let name = id.name.file_name.as_deref().map(OsStr::new);
275            out.kind = match &out.bytes {
276                Some(bytes) => Some(classify_bytes(bytes, name)),
277                None => match r.seek(SeekFrom::Start(0)) {
278                    Ok(_) => Some(classify(&mut *r, name)),
279                    Err(e) => {
280                        id.error(Tier::Classify, "read_failed", e.to_string());
281                        None
282                    }
283                },
284            };
285        }
286        if need_hashes {
287            // Every algorithm a revoked file hash is listed in, so a hash
288            // revoked by, say, its SHA-384 is found whatever the caller
289            // asked to compute; both the flat and the Authenticode digests.
290            let revoked_algs = ctx
291                .revocation
292                .file_hashes()
293                .iter()
294                .fold(AlgorithmSet::EMPTY, |set, d| {
295                    set | AlgorithmSet::from(d.algorithm())
296                });
297            let options = ImageHashOptions {
298                hash: HashOptions {
299                    algorithms: ctx.flat_algorithms | AlgorithmSet::SHA256 | revoked_algs,
300                    ..ctx.hash_options
301                },
302                authenticode_extra: revoked_algs,
303                parallel: false,
304            };
305            let result = match &out.bytes {
306                Some(bytes) => hash_image_reader(bytes.as_slice(), len, &options),
307                None => r
308                    .seek(SeekFrom::Start(0))
309                    .map_err(HashError::Io)
310                    .and_then(|_| hash_image_reader(&mut *r, len, &options)),
311            };
312            match result {
313                Ok(ImageHashStatus::Complete(h)) => {
314                    if let Some(d) = h.flat.get(HashAlgorithm::Sha256) {
315                        id.content_sha256 = Some(d.to_string());
316                    }
317                    out.hashes = Some(h);
318                }
319                Ok(ImageHashStatus::TooLarge { limit }) => {
320                    let msg = format!("{len} bytes is over the hash limit of {limit}");
321                    id.hashes = Field::Unavailable(msg.clone());
322                    id.error(Tier::Hash, "too_large", msg);
323                }
324                Err(e) => {
325                    id.hashes = Field::Unavailable(e.to_string());
326                    id.error(Tier::Hash, "hash_failed", e.to_string());
327                }
328            }
329        }
330        out
331    }
332}

334/// The SHA-256 of `bytes`, lowercase hex.
335fn sha256_hex(bytes: &[u8]) -> String {
336    let mut h = MultiHasher::new(AlgorithmSet::SHA256);
337    h.update(bytes);
338    h.finalize()
339        .get(HashAlgorithm::Sha256)
340        .map(ToString::to_string)
341        .unwrap_or_default()
342}

344/// The classify, hash and trust tiers.
345fn compose(id: &mut FileIdentity, content: Content, tiers: Tiers, ctx: &IdentityContext<'_>) {
346    let info = content.bytes.as_deref().map(pe::parse);
347    let pe_info = match &info {
348        Some(Ok(p)) => Some(p),
349        _ => None,
350    };
351    // PE content, by the parse when the bytes were read, otherwise by the
352    // classifier's look at the headers (a large file, or the hash tier
353    // alone), so a missing Authenticode digest of a PE is `unavailable`,
354    // not `not_applicable`.
355    let is_pe = match &info {
356        Some(r) => r.is_ok(),
357        None => match &content.kind {
358            Some(k) => k.content_kind == FileKind::Pe,
359            None => content
360                .hashes
361                .as_ref()
362                .is_some_and(|h| h.authenticode.is_some()),
363        },
364    };
365    let deep_pending = Field::pending(&[52, 53]);
366
367    if tiers.contains(Tier::Classify) {
368        match (&info, content.bytes.is_some()) {
369            (Some(Ok(p)), _) => {
370                id.pe = Field::Computed(pe_summary(p, id.name.file_name.as_deref()));
371            }
372            (Some(Err(_)), _) => id.pe = Field::NotApplicable,
373            (None, _) => id.pe = Field::Unavailable("the content could not be read".to_owned()),
374        }
375        id.kind = match &content.kind {
376            Some(c) => Field::Computed(kind_summary(c)),
377            None => Field::Unavailable("the content could not be read".to_owned()),
378        };
379        if content.bytes.is_some() && content.kind.is_some() {
380            id.computed_tiers.insert(Tier::Classify);
381        }
382    }
383
384    // The trust tier computes the digests too; they are reported whenever
385    // they exist.
386    if let Some(h) = &content.hashes {
387        id.hashes = Field::Computed(hashes(h, is_pe));
388        id.computed_tiers.insert(Tier::Hash);
389    }
390
391    if tiers.contains(Tier::Trust) {
392        trust_tier(id, &content, pe_info, ctx);
393    }
394
395    if tiers.contains(Tier::Deep) {
396        id.installer = deep_pending.clone();
397        id.package = deep_pending;
398        id.script = Field::pending(&[52]);
399    }
400}

402fn trust_tier(
403    id: &mut FileIdentity,
404    content: &Content,
405    pe_info: Option<&PeInfo>,
406    ctx: &IdentityContext<'_>,
407) {
408    let Some(hashes) = &content.hashes else {
409        let why = "the trust tier needs the file's digests, which were not computed";
410        id.signing_state = Field::Unavailable(why.to_owned());
411        id.signatures = Field::Unavailable(why.to_owned());
412        id.catalog_matches = Field::Unavailable(why.to_owned());
413        id.error(Tier::Trust, "no_digests", why);
414        return;
415    };
416    let version = pe_info.and_then(|p| p.version_info.as_ref());
417
418    let digests = file_digests(hashes);
419    // Embedded signatures (#48), chains (#49) and validity (#50).
420    let mut embedded = EmbeddedState::NotChecked;
421    let mut embedded_invalid_reason = None;
422    let mut sigs: Option<(SignatureReport, TrustReport, ValidityReport)> = None;
423    // A PE image whose Authenticode digests could not be computed has its
424    // signatures left unchecked, so its state stays unverified.
425    if let (Some(info), Some(bytes), Some(image)) = (pe_info, &content.bytes, &hashes.authenticode)
426    {
427        {
428            let report = verify_certificate_table(bytes, info, image, &VerifyOptions::default());
429            let trust = evaluate_trust(&report, ctx.code_trust, version);
430            let vctx = ValidityContext {
431                now: ctx.now,
432                clock_floor: ctx.clock_floor,
433                policy: ctx.validity,
434                revocation: ctx.revocation,
435                tsa_trust: ctx.tsa_trust,
436                file_digests: &digests,
437                version,
438            };
439            let validity = evaluate_validity(&report, &trust, &vctx);
440            embedded = match &report.signing_state {
441                EmbeddedSigning::Unsigned => EmbeddedState::Unsigned,
442                EmbeddedSigning::Valid => EmbeddedState::Valid,
443                EmbeddedSigning::Invalid(reason) => {
444                    embedded_invalid_reason = Some(invalid_reason(reason));
445                    EmbeddedState::Invalid
446                }
447            };
448            sigs = Some((report, trust, validity));
449        }
450    }
451
452    // Catalogs (#51). One lookup serves the summary and the signer.
453    let catalog_digests =
454        FileDigests::from_hashes(hashes.authenticode.as_ref(), Some(&hashes.flat));
455    // A file revoked by hash is never vouched for, by any signer.
456    let file_revoked = ctx.revocation.file_revoked(&digests).is_some();
457    let found = ctx.catalogs.map(|store| store.lookup(&catalog_digests));
458    // The time revocation is judged at: the later of the clock and the
459    // floor, as for embedded signatures (ADR 0012).
460    let effective_now = ctx.clock_floor.map_or(ctx.now, |f| f.max(ctx.now));
461    // One budget of CRL signature checks for every catalog signer.
462    let mut checker = ctx.revocation.checker();
463    let mut matches = Vec::new();
464    // Which matches a revoked catalog signer disqualifies.
465    let mut usable = Vec::new();
466    let catalog = match (ctx.catalogs, &found) {
467        (Some(store), Some(found)) => {
468            for m in &found.matches {
469                let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
470                let signer_revoked =
471                    chain.is_some_and(|c| chain_revoked(c, ctx, &mut checker, effective_now));
472                // A catalog vouches only when its signer is trusted and its
473                // signature's validity (expiry, revocation of its own hash,
474                // key sizes, timestamps) was judged and passed: a catalog
475                // whose validity was never judged does not vouch (#50).
476                let validity_ok = m.usable();
477                usable.push(validity_ok && !m.relies_on_sha1() && !signer_revoked && !file_revoked);
478                matches.push(CatalogMatchSummary {
479                    catalog_id: m.record.id.as_ref().map(Digest::to_string),
480                    catalog_path: m.record.path.as_ref().map(|p| p.display().to_string()),
481                    source: match m.record.source {
482                        CatalogSource::System => "system",
483                        CatalogSource::DriverStore => "driver_store",
484                        CatalogSource::Policy => "policy",
485                    },
486                    member: m.member,
487                    matched_by: match m.matched_by {
488                        DigestKind::AuthenticodeSha256 => "authenticode_sha256",
489                        DigestKind::AuthenticodeSha1 => "authenticode_sha1",
490                        DigestKind::FlatSha256 => "sha256",
491                        DigestKind::FlatSha1 => "sha1",
492                    },
493                    trusted: m.trusted(),
494                    validity: m.record.validity.as_ref().map(|v| v.error),
495                    relies_on_sha1: m.relies_on_sha1(),
496                    signer_revoked,
497                    publisher_name: m
498                        .signer
499                        .and_then(|s| s.identity.as_ref())
500                        .and_then(|i| i.publisher_name.as_ref())
501                        .map(|p| p.value.clone()),
502                    signer: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
503                });
504            }
505            if found.truncated {
506                id.error(
507                    Tier::Trust,
508                    "catalog_matches_truncated",
509                    "more catalogs list the file than are reported",
510                );
511            }
512            // A store whose index was cut short cannot show that a file is
513            // not listed.
514            let index_truncated = store.records().iter().any(|r| r.index_truncated);
515            if usable.iter().any(|u| *u) {
516                CatalogState::ListedTrusted
517            } else if found.matches.iter().zip(&matches).any(|(m, summary)| {
518                m.usable() && m.relies_on_sha1() && !summary.signer_revoked && !file_revoked
519            }) {
520                CatalogState::ListedTrustedSha1Only
521            } else if !found.matches.is_empty() {
522                CatalogState::ListedUntrusted
523            } else if index_truncated {
524                id.error(
525                    Tier::Trust,
526                    "catalog_index_truncated",
527                    "the catalog index is incomplete, so a missing match proves nothing",
528                );
529                CatalogState::NotChecked
530            } else {
531                CatalogState::NotListed
532            }
533        }
534        _ => CatalogState::NotChecked,
535    };
536
537    // The signer that vouches for the file: a fully valid embedded
538    // signature first, then the best trusted catalog that does not rely on
539    // SHA-1.
540    let mut signer = None;
541    let mut embedded_sha1_only = false;
542    let mut summaries = Vec::new();
543    let mut verification_error = if file_revoked {
544        VerificationError::RevokedByHash
545    } else {
546        VerificationError::NotSigned
547    };
548    let mut clock_suspect = ctx.clock_floor.is_some_and(|f| ctx.now < f);
549    if let Some((report, trust, validity)) = &sigs {
550        verification_error = validity.error;
551        clock_suspect |= validity.clock_suspect;
552        for sig in &report.signatures {
553            let t = trust.signers.iter().find(|s| s.signature == sig.index);
554            let v = validity
555                .signatures
556                .iter()
557                .find(|v| v.signature == sig.index);
558            summaries.push(SignatureSummary {
559                index: sig.index,
560                kind: match sig.kind {
561                    SignatureKind::Nested => "nested",
562                    SignatureKind::Catalog => "catalog",
563                    _ => "embedded",
564                },
565                depth: sig.depth,
566                parent: sig.parent,
567                status: status_name(&sig.status),
568                status_detail: format!("{:?}", sig.status),
569                digest_alg: sig.digest_alg,
570                signer_digest_alg: sig.signer_digest_alg,
571                sig_alg: sig.sig_alg.clone(),
572                key_alg: sig.key_alg.clone(),
573                key_bits: sig.key_bits,
574                digest_match: sig.digest_match,
575                program_name: sig.signer.as_ref().and_then(|s| s.program_name.clone()),
576                more_info_url: sig.signer.as_ref().and_then(|s| s.more_info_url.clone()),
577                claimed_signing_time: sig.signer.as_ref().and_then(|s| s.signing_time),
578                has_page_hashes: sig.page_hashes.is_some(),
579                // The signature and, when a chain was built, every
580                // certificate on it (#49).
581                os_ci_compatible: t.map_or_else(|| sig.os_ci_compatible(), |t| t.os_ci_compatible),
582                trusted: t.is_some_and(|t| t.trusted),
583                untrusted: t.map(|t| t.untrusted.clone()).unwrap_or_default(),
584                validity_error: v.map_or(VerificationError::Generic, |v| v.error),
585                trusted_signing_time: v.and_then(|v| v.signing_time),
586                timestamps: sig.timestamp_tokens.len(),
587                chain: t.and_then(|t| t.chain.as_ref()).map(chain_summary),
588            });
589        }
590        // A primary signature that rests on SHA-1 (its content digest or
591        // its signer digest) is valid, but no allow decision may rest on it
592        // (ADR 0008, ADR 0017): it is not a trusted signer, as a catalog
593        // match that relies on SHA-1 is not.
594        let sha2_primary = validity
595            .primary
596            .and_then(|p| report.signatures.get(p))
597            .is_some_and(|sig| {
598                sig.digest_alg.is_some_and(DigestAlgorithm::is_sha2)
599                    && sig.signer_digest_alg.is_some_and(DigestAlgorithm::is_sha2)
600            });
601        embedded_sha1_only = validity.primary.is_some() && !sha2_primary;
602        if let (Some(p), Some(identity), false, true) = (
603            validity.primary,
604            &validity.identity,
605            file_revoked,
606            sha2_primary,
607        ) {
608            let chain = trust
609                .signers
610                .iter()
611                .find(|s| s.signature == p)
612                .and_then(|s| s.chain.as_ref());
613            signer = Some(TrustedSigner {
614                source: TrustSource::Embedded,
615                signature: Some(p),
616                catalog_match: None,
617                publisher_name: identity.publisher_name.as_ref().map(|n| n.value.clone()),
618                publisher_dn: identity.publisher_dn.clone(),
619                product_name: identity.product_name.clone(),
620                leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
621                pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
622                root: chain.and_then(full_root).map(cert_brief),
623                well_known_root_id: chain.and_then(|c| c.well_known_root_id),
624                anchored_to: chain
625                    .and_then(|c| c.anchored_to.as_ref())
626                    .map(|a| a.label.clone()),
627            });
628        }
629    }
630    if signer.is_none() {
631        if let Some(found) = &found {
632            let best = found
633                .matches
634                .iter()
635                .enumerate()
636                .find(|(i, _)| usable.get(*i).copied().unwrap_or(false));
637            if let Some((i, m)) = best {
638                let chain = m.signer.and_then(|s| s.trust.chain.as_ref());
639                let identity = m.signer.and_then(|s| s.identity.as_ref());
640                signer = Some(TrustedSigner {
641                    source: TrustSource::Catalog,
642                    signature: None,
643                    catalog_match: Some(i),
644                    publisher_name: identity
645                        .and_then(|i| i.publisher_name.as_ref())
646                        .map(|n| n.value.clone()),
647                    publisher_dn: identity.and_then(|i| i.publisher_dn.clone()),
648                    product_name: version
649                        .and_then(|v| v.get(VersionField::ProductName))
650                        .filter(|p| !p.trim().is_empty())
651                        .map(str::to_owned),
652                    leaf: chain.and_then(ChainReport::leaf_cert).map(cert_brief),
653                    pca: chain.and_then(ChainReport::pca_cert).map(cert_brief),
654                    root: chain.and_then(full_root).map(cert_brief),
655                    well_known_root_id: chain.and_then(|c| c.well_known_root_id),
656                    anchored_to: chain
657                        .and_then(|c| c.anchored_to.as_ref())
658                        .map(|a| a.label.clone()),
659                });
660            }
661        }
662    }
663
664    // A valid embedded signature whose signer was revoked (a certificate in
665    // its chain, or its own hash) is no weaker evidence of tampering than a
666    // broken one: it is invalid, never merely untrusted.
667    let revoked = matches!(
668        verification_error,
669        VerificationError::Revoked | VerificationError::RevokedByHash
670    );
671    let state = signing_state_kind(
672        signer.is_some(),
673        file_revoked || (embedded == EmbeddedState::Valid && signer.is_none() && revoked),
674        embedded,
675        catalog,
676    );
677
678    id.signing_state = Field::Computed(SigningState {
679        state,
680        embedded,
681        embedded_invalid_reason,
682        catalog,
683        verification_error,
684        clock_suspect,
685        embedded_sha1_only,
686        signer,
687    });
688    if sigs.is_some() {
689        id.signatures = Field::Computed(summaries);
690    } else if content.bytes.is_none() {
691        let why = "the file was not read into memory, so its signatures were not read";
692        id.signatures = Field::Unavailable(why.to_owned());
693        id.error(Tier::Trust, "signatures_unread", why);
694    } else if pe_info.is_some() {
695        let why = "the image's Authenticode digests could not be computed";
696        id.signatures = Field::Unavailable(why.to_owned());
697        id.error(Tier::Trust, "signatures_unread", why);
698    } else {
699        id.signatures = Field::pending(&[52, 53]);
700    }
701    id.catalog_matches = match ctx.catalogs {
702        Some(_) => Field::Computed(matches),
703        None => Field::NotRequested,
704    };
705    if id.signatures.is_computed() || (content.bytes.is_some() && pe_info.is_none()) {
706        id.computed_tiers.insert(Tier::Trust);
707    }
708}

710/// Whether a catalog signer's chain is revoked, or its revocation state is
711/// unknown where policy says that fails. Catalog signatures carry no
712/// trusted signing time here (catalog timestamps are #50), so every entry
713/// counts, whatever its date.
714fn chain_revoked(
715    chain: &ChainReport,
716    ctx: &IdentityContext<'_>,
717    checker: &mut RevocationChecker<'_>,
718    now: i64,
719) -> bool {
720    let certs: Vec<&CertSummary> = chain.full_chain_certs().collect();
721    (0..certs.len()).any(|i| match checker.state_of(&certs, i, now) {
722        CertRevocation::Good => false,
723        CertRevocation::Unknown => ctx.revocation.failure == RevocationFailure::FailClosed,
724        // Revoked, or a CRL that was not checked: both fail closed.
725        _ => true,
726    })
727}

729/// The context tier: mark-of-the-web and path variables. Owner,
730/// writability, streams and reparse points are #54.
731fn context_tier(
732    id: &mut FileIdentity,
733    file: &std::fs::File,
734    tiers: Tiers,
735    ctx: &IdentityContext<'_>,
736) {
737    if !tiers.contains(Tier::Context) {
738        return;
739    }
740    let pending = Field::pending(&[54]);
741    id.fs.file_id = pending.clone();
742    id.fs.volume = pending.clone();
743    id.fs.owner = pending.clone();
744    id.fs.user_writable = pending.clone();
745    id.fs.streams = pending.clone();
746    id.fs.reparse = pending.clone();
747    id.fs.provenance = pending;
748    // The stream is read for the opened file itself: by the handle's final
749    // path, and kept only if that path still names the same file, so the
750    // record never pairs this file's content with another file's mark.
751    id.fs.motw = read_motw(file);
752    if let Field::Unavailable(why) = &id.fs.motw {
753        if cfg!(all(windows, feature = "os-trust")) {
754            let why = why.clone();
755            id.error(Tier::Context, "motw_unreadable", why);
756        }
757    }
758    id.fs.path_variables = match (ctx.known_folders, id.fs.path.computed()) {
759        (None, _) => Field::Unavailable("no known folders in the context".to_owned()),
760        (Some(folders), Some(info)) => match info.rule_path() {
761            Some(p) => match path_variables_matched(&p, folders, None) {
762                Ok(vars) => Field::Computed(vars.iter().map(|v| v.token().to_owned()).collect()),
763                Err(e) => Field::Unavailable(e.to_string()),
764            },
765            None => Field::Unavailable("the path is not a full Windows path".to_owned()),
766        },
767        (Some(_), None) => Field::Unavailable("no path".to_owned()),
768    };
769    // The tier counts as computed only when it computed something: with
770    // every field pending or unavailable it did not.
771    if id.fs.motw.is_computed() || id.fs.path_variables.is_computed() {
772        id.computed_tiers.insert(Tier::Context);
773    }
774}

989/// `CamelCase` to `snake_case`, for the `Debug` names of #44's enums.
990fn snake(debug: &str) -> String {
991    let mut out = String::with_capacity(debug.len() + 4);
992    for (i, c) in debug.chars().enumerate() {
993        if c.is_ascii_uppercase() {
994            if i > 0 {
995                out.push('_');
996            }
997            out.push(c.to_ascii_lowercase());
998        } else {
999            out.push(c);
1000        }
1001    }
1002    out
1003}

1005fn kind_summary(c: &Classification) -> KindSummary {
1006    let (content_kind, script_kind) = match c.content_kind {
1007        FileKind::Script(s) => ("script".to_owned(), Some(snake(&format!("{s:?}")))),
1008        other => (snake(&format!("{other:?}")), None),
1009    };

1013        }
1014        crate::filetype::ExtensionKind::Kind(k) => snake(&format!("{k:?}")),
1015        other => snake(&format!("{other:?}")),

1017    let ContentFlags {
1018        macro_capable_container,
1019        package_signature_present,
1020        python_zipapp,
1021        script_signature_block,
1022        encoded_script,
1023        name_bidi_control,
1024        name_double_extension,
1025    } = c.flags;
1026    let flags = [
1027        ("macro_capable_container", macro_capable_container),
1028        ("package_signature_present", package_signature_present),
1029        ("python_zipapp", python_zipapp),
1030        ("script_signature_block", script_signature_block),
1031        ("encoded_script", encoded_script),
1032        ("name_bidi_control", name_bidi_control),
1033        ("name_double_extension", name_double_extension),
1034    ]
1035    .into_iter()
1036    .filter_map(|(n, set)| set.then_some(n))
1037    .collect();
1038    KindSummary {
1039        classifier_version: c.classifier_version,
1040        content_kind,
1041        script_kind,
1042        pe_sub_kind: c.pe.as_ref().map(|p| snake(&format!("{:?}", p.sub_kind))),
1043        extension_kind,
1044        executable_content: c.is_executable_content(),
1045        mismatch: c.mismatch,
1046        mismatch_severity: snake(&format!("{:?}", c.mismatch_severity)),
1047        confidence: snake(&format!("{:?}", c.confidence)),
1048        heuristic: c.heuristic,
1049        extension_assisted: c.extension_assisted,
1050        flags,
1051        parse_error: c.parse_error.as_ref().map(ToString::to_string),
1052    }
1053}

1096fn hashes(h: &ImageHashes, is_pe: bool) -> Hashes {
1097    let f = &h.flat;
1098    let authenticode = match &h.authenticode {
1099        Some(a) => Field::Computed(authenticode_summary(a)),
1100        None if is_pe => Field::Unavailable("the image hash was not computed".to_owned()),
1101        None => Field::NotApplicable,
1102    };
1103    let conformant = h
1104        .authenticode
1105        .as_ref()
1106        .is_some_and(|a| a.status.is_conformant());
1107    Hashes {
1108        size: f.size,
1109        empty: f.empty,
1110        sha256: hex(f.get(HashAlgorithm::Sha256)),
1111        sha1: hex(f.get(HashAlgorithm::Sha1)),
1112        md5: hex(f.get(HashAlgorithm::Md5)),
1113        sha384: hex(f.get(HashAlgorithm::Sha384)),
1114        sha512: hex(f.get(HashAlgorithm::Sha512)),
1115        authenticode,
1116        authoritative: if f.empty {
1117            "none"
1118        } else if conformant {
1119            "authenticode_sha256"
1120        } else {
1121            "sha256"
1122        },
1123        msi_normalised_sha256: Field::pending(&[53]),
1124        script_normalised_sha256: Field::pending(&[52]),
1125    }

1133        non_conformance: a.status.reasons().iter().map(ToString::to_string).collect(),
1134        page_sha256: a.page_sha256().map(|d| d.to_string()),
1135        page_sha1: a.page_sha1().map(|d| d.to_string()),
1136        page_hashes_error: match &a.page_hashes {
1137            Some(Err(e)) => Some(e.to_string()),
1138            _ => None,
1139        },
1140    }

1205/// The signing verdict from its evidence. `revoked` is a revoked file hash,
1206/// or a revoked signer behind a valid embedded signature. `trusted_signer`
1207/// is whether a
1208/// trusted signer vouches for the file: a valid embedded signature that
1209/// passed every check, or a catalog match counted usable (trusted, and not
1210/// relying on SHA-1). "Unsigned" needs both halves of the evidence: no
1211/// embedded signature, and a catalog lookup that found nothing. Without the
1212/// lookup (no store, or an index cut short) the file may be catalog-signed,
1213/// so it is unverified.
1214fn signing_state_kind(
1215    trusted_signer: bool,
1216    revoked: bool,
1217    embedded: EmbeddedState,
1218    catalog: CatalogState,
1219) -> SigningStateKind {
1220    if trusted_signer {
1221        SigningStateKind::Trusted
1222    } else if revoked || embedded == EmbeddedState::Invalid {
1223        // A broken embedded signature, or a revoked file or signer, is
1224        // invalid whatever a catalog says: an untrusted catalog listing
1225        // does not make it merely "untrusted".
1226        SigningStateKind::Invalid
1227    } else if embedded == EmbeddedState::NotChecked
1228        || (embedded == EmbeddedState::Unsigned && catalog == CatalogState::NotChecked)
1229    {
1230        SigningStateKind::Unverified
1231    } else if embedded == EmbeddedState::Valid
1232        || matches!(
1233            catalog,
1234            CatalogState::ListedUntrusted | CatalogState::ListedTrustedSha1Only
1235        )
1236    {
1237        SigningStateKind::Untrusted
1238    } else {
1239        SigningStateKind::Unsigned
1240    }
1241}

1243#[cfg(test)]
1244mod tests {
1245    use super::*;
1246
1247    #[test]
1248    fn signing_state_never_trusts_without_a_usable_signer() {
1249        use CatalogState as C;
1250        use EmbeddedState as E;
1251        // A catalog match that relies on SHA-1 is never trust, and never
1252        // "unsigned"; a catalog index cut short is "not checked", so an
1253        // unsigned file is unverified, never unsigned.
1254        for embedded in [E::Unsigned, E::Valid, E::Invalid, E::NotChecked] {
1255            for catalog in [
1256                C::NotChecked,
1257                C::NotListed,
1258                C::ListedUntrusted,
1259                C::ListedTrustedSha1Only,
1260                C::ListedTrusted,
1261            ] {
1262                let k = signing_state_kind(false, false, embedded, catalog);
1263                assert_ne!(k, SigningStateKind::Trusted, "{embedded:?} {catalog:?}");
1264            }
1265            assert_ne!(
1266                signing_state_kind(false, false, embedded, C::ListedTrustedSha1Only),
1267                SigningStateKind::Unsigned,
1268                "{embedded:?}"
1269            );
1270        }
1271        assert_eq!(
1272            signing_state_kind(false, false, E::Unsigned, C::NotChecked),
1273            SigningStateKind::Unverified
1274        );
1275        assert_eq!(
1276            signing_state_kind(false, false, E::Unsigned, C::NotListed),
1277            SigningStateKind::Unsigned
1278        );
1279        // A broken embedded signature is invalid even when an untrusted
1280        // catalog lists the file.
1281        for catalog in [C::ListedUntrusted, C::ListedTrustedSha1Only, C::NotListed] {
1282            assert_eq!(
1283                signing_state_kind(false, false, E::Invalid, catalog),
1284                SigningStateKind::Invalid,
1285                "{catalog:?}"
1286            );
1287        }
1288        // A revoked hash is invalid whatever else is listed, and a trusted
1289        // signer is the only way to `Trusted`.
1290        assert_eq!(
1291            signing_state_kind(false, true, E::Valid, C::ListedTrusted),
1292            SigningStateKind::Invalid
1293        );
1294        assert_eq!(
1295            signing_state_kind(true, false, E::Valid, C::NotListed),
1296            SigningStateKind::Trusted
1297        );
1298    }
