1//! The recommended rule level for a file, with the reasons: what
2//! `pal fileid explain` prints, and what offline discovery (#73) uses to
3//! propose rules.
4//!
5//! The order of preference follows the strength classes
6//! ([`super::strength`]): a signer rule when a trusted signer vouches for
7//! the file (it survives updates), otherwise a hash rule on the
8//! authoritative digest. A path rule is never recommended while the path's
9//! writability is unknown or user-writable, and nothing is recommended for
10//! an empty file or a file that looks like a masquerade.
11
12#![forbid(unsafe_code)]
13

20/// A rule level, narrowest last.
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
22#[serde(rename_all = "snake_case")]
23pub enum RuleLevel {
24    /// The trusted publisher, any product.
25    Publisher,
26    /// The trusted publisher and product.
27    PublisherProduct,
28    /// Publisher, product, file name and minimum version.
29    FilePublisher,
30    /// The Authenticode SHA-256 of a conformant image.
31    AuthenticodeHash,
32    /// The flat SHA-256.
33    Hash,
34    /// The file's path.
35    Path,
36    /// No rule: the file has no usable identity, or should be looked at
37    /// first.
38    None,
39}
40
41/// The recommendation.
42#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
43pub struct Recommendation {
44    /// The level to use.
45    pub level: RuleLevel,
46    /// The value or values the rule would match.
47    pub values: Vec<(String, String)>,
48    /// Why, one line each, for reading.
49    pub rationale: Vec<String>,
50    /// Levels that were considered and refused, with why.
51    pub refused: Vec<(RuleLevel, String)>,
52}
53
54/// Masquerade scores at or above this stop any recommendation.
55pub const MASQUERADE_THRESHOLD: u8 = 50;
56
57/// Masquerade checks that, when they could not be made, stop any
58/// recommendation: each could have found a renamed or impersonating
59/// binary.
60pub const BLOCKING_UNASSESSED: [&str; 3] = [
61    "known_binary_relocated",
62    "signer_impersonation",
63    "name_mismatch",
64];
65
66/// Recommends a rule level for `id`.
67#[must_use]
68pub fn recommend(id: &FileIdentity) -> Recommendation {
69    let mut rec = Recommendation {
70        level: RuleLevel::None,
71        values: Vec::new(),
72        rationale: Vec::new(),
73        refused: Vec::new(),
74    };
75
76    if id.masquerade.score >= MASQUERADE_THRESHOLD {
77        rec.rationale.push(format!(
78            "masquerade score {} (at least {MASQUERADE_THRESHOLD}): investigate before allowing anything",
79            id.masquerade.score
80        ));
81        return rec;
82    }
83
84    // A masquerade check that could not run is not a clean result (ADR
85    // 0018): a file that may be a renamed or impersonating binary gets no
86    // allow rule until the check can be made.
87    if let Some(check) = id
88        .masquerade
89        .unassessed
90        .iter()
91        .find(|c| BLOCKING_UNASSESSED.contains(c))
92    {
93        rec.rationale.push(format!(
94            "the {check} check could not be made (see masquerade.unassessed), so no allow rule is recommended until it can"
95        ));
96        return rec;
97    }
98    // A file whose version resource claims a protected binary's name is
99    // allowed only through that binary's own trusted signer: a hash rule
100    // would allow an impostor or a copy outside its folders.
101    if let Some(original) = id.masquerade.claims_protected.as_deref()
102        && id.trusted_signer().is_none()
103    {
104        rec.rationale.push(format!(
105                "its OriginalFilename {original:?} names a protected binary and no trusted signer vouches for it, so no allow rule is recommended"
106            ));
107        return rec;
108    }
109
110    let signing = id.signing_state.computed();
111    // A file whose signature was broken after signing, or whose hash or
112    // signer is revoked, gets no allow rule of any kind; nor does a signed
113    // file whose signature failed any other validity check (expired, a
114    // weak key), which a hash rule would otherwise wave through. A valid
115    // signature whose signer policy simply does not trust (no trusted
116    // root, or no chain to one) is not a failure: such a file may get a
117    // hash rule.
118    if let Some(s) = signing.filter(|s| s.state == SigningStateKind::Invalid) {
119        rec.rationale.push(format!(
120            "the file is invalid ({}): it was changed after signing, or its hash or signer is revoked, so no allow rule is recommended",
121            s.verification_error.as_str()
122        ));
123        return rec;
124    }
125    if let Some(s) = signing.filter(|s| {
126        s.embedded != EmbeddedState::Unsigned
127            && s.embedded != EmbeddedState::NotChecked
128            && s.signer.is_none()
129            && !matches!(
130                s.verification_error,
131                VerificationError::Ok
132                    | VerificationError::NotSigned
133                    | VerificationError::UntrustedRoot
134                    | VerificationError::ChainIncomplete
135            )
136    }) {
137        rec.rationale.push(format!(
138            "its signature failed validation ({}), so no allow rule is recommended",
139            s.verification_error.as_str()
140        ));
141        return rec;
142    }
143    let signer = id.trusted_signer();
144    match (signing.map(|s| s.state), signer) {
145        (Some(SigningStateKind::Trusted), Some(s)) => {
146            let source = match s.source {
147                super::record::TrustSource::Embedded => "an embedded signature",
148                super::record::TrustSource::Catalog => "a trusted catalog",
149            };
150            let publisher = s.publisher_name.clone();
151            let product = s.product_name.clone();
152            let file_name = id
153                .derived
154                .effective_file_name
155                .as_ref()
156                .filter(|n| n.source != "file_name");
157            let version = id.version().and_then(|v| v.file_version_quad.clone());
158            if let Some(publisher) = publisher {
159                rec.rationale
160                    .push(format!("signed by {publisher:?}, through {source}"));
161                rec.values.push(("publisher".to_owned(), publisher));
162                rec.level = RuleLevel::Publisher;
163                if let Some(product) = product {
164                    rec.rationale.push(format!(
165                        "product {product:?}, so a publisher and product rule covers its updates"
166                    ));
167                    rec.values.push(("product".to_owned(), product));
168                    rec.level = RuleLevel::PublisherProduct;
169                    if let (Some(name), Some(version)) = (file_name, version) {
170                        rec.rationale.push(format!(
171                            "for this file only, narrow it to {:?} ({}) at version {version} or later",
172                            name.value, name.source
173                        ));
174                        rec.values
175                            .push(("file_name".to_owned(), name.value.clone()));
176                        rec.values.push(("min_version".to_owned(), version));
177                        rec.level = RuleLevel::FilePublisher;
178                    }
179                } else {
180                    rec.rationale.push(
181                        "no product name: a publisher rule would allow everything this signer signs"
182                            .to_owned(),
183                    );
184                }
185            } else {
186                rec.refused.push((
187                    RuleLevel::Publisher,
188                    "the trusted signer has no usable name".to_owned(),
189                ));
190            }
191        }
192        (Some(SigningStateKind::Untrusted), _) => rec.refused.push((
193            RuleLevel::Publisher,
194            if signing.is_some_and(|s| s.embedded_sha1_only) {
195                "the signature rests on SHA-1, which vouches for nothing (ADR 0008)".to_owned()
196            } else {
197                "the signature is valid but its signer is not trusted by policy".to_owned()
198            },
199        )),
200        (Some(SigningStateKind::Unverified), _) => rec.refused.push((
201            RuleLevel::Publisher,
202            match signing.map(|s| (s.embedded, s.catalog)) {
203                // A PE whose signatures could not be read (too large, say).
204                _ if matches!(id.signatures, Field::Unavailable(_)) => {
205                    let why = match &id.signatures {
206                        Field::Unavailable(why) => why.as_str(),
207                        _ => "",
208                    };
209                    format!("its signatures could not be read: {why}")
210                }
211                Some((EmbeddedState::NotChecked, _)) => {
212                    "this file type's signatures are not read".to_owned()
213                }
214                Some((_, CatalogState::NotChecked))
215                    if id.errors.iter().any(|e| e.code == "catalog_index_truncated") =>
216                {
217                    "no embedded signature, and the catalog index is incomplete, so it may be catalog-signed"
218                        .to_owned()
219                }
220                // A script or installer whose catalog lookup cannot prove
221                // a miss: the digests its catalogs list are not yet checked
222                // against catalogs made by OS tooling (#51).
223                Some((_, CatalogState::NotChecked))
224                    if ctx_had_catalogs(id) =>
225                {
226                    "no embedded signature, and a catalog lookup of this file type cannot show it is not catalog-signed"
227                        .to_owned()
228                }
229                Some((_, CatalogState::NotChecked)) => {
230                    "no embedded signature, and no catalogs were given to look it up in (use --catroot)"
231                        .to_owned()
232                }
233                _ => "the signer could not be verified".to_owned(),
234            },
235        )),
236        (Some(_), _) => rec.refused.push((
237            RuleLevel::Publisher,
238            "no trusted signer vouches for the file".to_owned(),
239        )),
240        (None, _) => rec.refused.push((
241            RuleLevel::Publisher,
242            "signatures were not examined (request the trust tier)".to_owned(),
243        )),
244    }
245
246    if rec.level == RuleLevel::None {
247        match id.hashes.computed() {
248            Some(h) if h.empty => rec.refused.push((
249                RuleLevel::Hash,
250                "the file is empty: its digest matches every empty file".to_owned(),
251            )),
252            Some(h) => {
253                let image = h
254                    .authenticode
255                    .computed()
256                    .filter(|a| a.conformant)
257                    .and_then(|a| a.sha256.clone());
258                if let Some(digest) = image {
259                    rec.level = RuleLevel::AuthenticodeHash;
260                    rec.rationale.push(
261                        "unsigned or untrusted, so a hash rule; the Authenticode hash survives re-signing"
262                            .to_owned(),
263                    );
264                    rec.values.push(("authenticode_sha256".to_owned(), digest));
265                } else if let Some(digest) = h.sha256.clone() {
266                    rec.level = RuleLevel::Hash;
267                    rec.rationale.push(
268                        "unsigned or untrusted, so a hash rule on the flat SHA-256".to_owned(),
269                    );
270                    if h.authenticode.computed().is_some_and(|a| !a.conformant) {
271                        rec.refused.push((
272                            RuleLevel::AuthenticodeHash,
273                            "the image does not conform to the Authenticode specification"
274                                .to_owned(),
275                        ));
276                    }
277                    rec.values.push(("sha256".to_owned(), digest));
278                }
279            }
280            None => rec.refused.push((
281                RuleLevel::Hash,
282                "the file was not hashed (request the hash tier)".to_owned(),
283            )),
284        }
285    }
286
287    // A path rule is never the recommendation: a content or signer rule
288    // always exists when this is reached. It is refused outright where
289    // non-administrators can write the path, or its writability is not
290    // known (ISM-1392).
291    if id.fs.path.computed().is_some_and(|p| !p.claimed) {
292        let local = id
293            .fs
294            .volume
295            .computed()
296            .is_some_and(|v| matches!(v.location.as_str(), "os_volume" | "other_fixed"));
297        let why = match id.fs.user_writable.computed() {
298            Some(w) if !w.path_user_writable && !w.change_perms_by_non_admin && !local => {
299                "path rule refused: the volume is not a local fixed disk, so its permissions do not protect the path"
300            }
301            Some(w) if !w.path_user_writable && !w.change_perms_by_non_admin => {
302                "path rule not recommended: the path is not writable by non-administrators, but a content or signer rule identifies the file itself"
303            }
304            Some(_) => {
305                "path rule refused: the file or a folder above it is writable by non-administrators"
306            }
307            None => {
308                "path rule refused: the path's writability by non-administrators is not known, so it is treated as user-writable"
309            }
310        };
311        rec.refused.push((RuleLevel::Path, why.to_owned()));
312    }
313    rec
314}

316/// Whether a catalog store was given: the record lists the matches it found
317/// (possibly none) only then.
318fn ctx_had_catalogs(id: &FileIdentity) -> bool {
319    id.catalog_matches.is_computed()
