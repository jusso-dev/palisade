1//! ZIP containers, classified by their entry names.
2//!
3//! Only the end records and the central directory are read; nothing is
4//! decompressed and local file headers are not read. The directory is read
5//! the one way the APPNOTE layout gives: the last end record, whose comment
6//! reaches the end of the file, the ZIP64 records just before it, and a
7//! directory that ends exactly where they start and holds the recorded
8//! number of entries. ZIP readers differ in how they handle anything else
9//! (a second end record, trailing data, fields that disagree, a shifted or
10//! understated directory), so any such structure is reported as
11//! [`ParseError::ZipDirectoryAmbiguous`] or
12//! [`ParseError::ZipDirectoryMalformed`] instead of being guessed at, and
13//! the caller treats it as possibly executable
14//! ([`Classification::is_executable_content`]).
15
16#![forbid(unsafe_code)]

28const LOCAL_HEADER: &[u8; 4] = b"PK\x03\x04";
29const END_OF_DIRECTORY: &[u8; 4] = b"PK\x05\x06";
30const ZIP64_LOCATOR: &[u8; 4] = b"PK\x06\x07";
31const ZIP64_END: &[u8; 4] = b"PK\x06\x06";
32const DIRECTORY_ENTRY: &[u8; 4] = b"PK\x01\x02";
33const SPANNED: &[u8; 4] = b"PK\x07\x08";

35/// The end of central directory record without its comment.
36const END_LEN: usize = 22;
37/// The longest archive comment.
38const MAX_COMMENT: usize = 0xffff;
39const ZIP64_LOCATOR_LEN: usize = 20;
40const ZIP64_END_LEN: usize = 56;
41const DIRECTORY_ENTRY_LEN: usize = 46;

43/// Whether the head starts with a ZIP record: a local file header, an empty
44/// archive's end record, or a spanned-archive marker.
45pub(super) fn starts_like_zip(head: &[u8]) -> bool {
46    [LOCAL_HEADER, END_OF_DIRECTORY, SPANNED]
47        .iter()
48        .any(|sig| head.starts_with(*sig))
49}

51/// The entry names that decide the kind.
52#[derive(Debug, Default, Clone, Copy)]
53pub(super) struct Entries {
54    appx_manifest: bool,
55    appx_signature: bool,
56    bundle_manifest: bool,
57    jar_manifest: bool,
58    python_main: bool,
59    vba_project: bool,
60}
61
62impl Entries {
63    /// Notes an entry under every name a reader could give it: the name
64    /// itself, the name up to its first NUL byte (some readers stop there),
65    /// the name percent-decoded (the package reader decodes part names),
66    /// and the name in any Unicode path extra field (some readers use that
67    /// instead; its checksum is not checked, so a mismatch over-counts).
68    /// `scratch` is reused for decoding, so no entry allocates. The work is
69    /// linear in the entry's length.
70    ///
71    /// Returns whether the extra field is a whole number of subfields, each
72    /// a 4-byte header and the data it declares, as the APPNOTE requires.
73    /// The names seen before a cut subfield still count.
74    fn note_entry(&mut self, name: &[u8], extra: &[u8], scratch: &mut Vec<u8>) -> bool {
75        self.note_name(name, scratch);
76        let mut rest = extra;
77        while !rest.is_empty() {
78            let (Some(id), Some(len)) = (u16_at(rest, 0), u16_at(rest, 2)) else {
79                return false;
80            };
81            let Some(body) = bytes(rest, 4, usize::from(len)) else {
82                return false;
83            };
84            if id == UNICODE_PATH {
85                // A version byte and the checksum, then the name.
86                if let Some(unicode) = body.get(5..) {
87                    self.note_name(unicode, scratch);
88                }
89            }
90            rest = rest.get(4 + usize::from(len)..).unwrap_or_default();
91        }
92        true
93    }
94
95    /// Notes `name`, its part before a NUL byte if it has one, and its
96    /// percent-decoded form if it has an escape. Each form is a different
97    /// name, so no form is compared twice.
98    fn note_name(&mut self, name: &[u8], scratch: &mut Vec<u8>) {
99        self.note_with_nul(name);
100        if percent_decode_into(name, scratch) {
101            let decoded = std::mem::take(scratch);
102            self.note_with_nul(&decoded);
103            *scratch = decoded;
104        }
105    }
106
107    fn note_with_nul(&mut self, name: &[u8]) {
108        self.note(name);
109        if let Some(nul) = name.iter().position(|b| *b == 0) {
110            self.note(name.get(..nul).unwrap_or_default());
111        }
112    }
113
114    /// Matches one name. Each length has at most one candidate name, so a
115    /// name is compared with at most two targets (one name, and the macro
116    /// project suffix). Package and archive names are matched as
117    /// [`same_name`] does; `__main__.py` and `__main__.pyc` exactly, as
118    /// Python's zip importer looks them up.
119    fn note(&mut self, name: &[u8]) {
120        let is = |want: &[u8]| same_name(name, want);
121        match name.len() {
122            11 => self.python_main |= name == b"__main__.py",
123            12 => self.python_main |= name == b"__main__.pyc",
124            16 => self.appx_manifest |= is(b"AppxManifest.xml"),
125            17 => self.appx_signature |= is(b"AppxSignature.p7x"),
126            20 => self.jar_manifest |= is(b"META-INF/MANIFEST.MF"),
127            35 => self.bundle_manifest |= is(b"AppxMetadata/AppxBundleManifest.xml"),
128            _ => {}
129        }
130        // Macro-enabled office suite documents keep their macros here
131        // (`word/vbaProject.bin`, `xl/vbaProject.bin`, ...).
132        if !self.vba_project && matches!(name.last(), Some(b'n' | b'N')) {
133            self.vba_project = name
134                .len()
135                .checked_sub(15)
136                .filter(|start| *start > 0)
137                .and_then(|start| name.get(start..))
138                .is_some_and(|tail| same_name(tail, b"/vbaProject.bin"));
139        }
140    }
141}

143/// Whether `name` is `want`, ignoring ASCII case and taking a backslash as
144/// `/` (some tools write it as the separator; the package reader accepts
145/// it).
146fn same_name(name: &[u8], want: &[u8]) -> bool {
147    let fold = |b: u8| if b == b'\\' { b'/' } else { b };
148    name.len() == want.len()
149        && name
150            .iter()
151            .zip(want)
152            .all(|(a, b)| fold(*a).eq_ignore_ascii_case(b))
153}

155/// Writes `name` with every `%XX` escape decoded into `out`, replacing its
156/// contents; other bytes are kept. Returns whether there was an escape, so
157/// an unchanged name is not matched again.
158fn percent_decode_into(name: &[u8], out: &mut Vec<u8>) -> bool {
159    let hex = |b: Option<&u8>| {
160        b.and_then(|b| char::from(*b).to_digit(16))
161            .and_then(|d| u8::try_from(d).ok())
162    };
163    let escape = |i: usize| hex(name.get(i + 1)).zip(hex(name.get(i + 2)));
164    let Some(first) = name
165        .iter()
166        .enumerate()
167        .position(|(i, b)| *b == b'%' && escape(i).is_some())
168    else {
169        return false;
170    };
171    out.clear();
172    out.extend_from_slice(name.get(..first).unwrap_or_default());
173    let mut i = first;
174    while let Some(&b) = name.get(i) {
175        match (b == b'%').then(|| escape(i)).flatten() {
176            Some((hi, lo)) => {
177                out.push(hi << 4 | lo);
178                i += 3;
179            }
180            _ => {
181                out.push(b);
182                i += 1;
183            }
184        }
185    }
186    true
187}

189/// Classifies a ZIP archive by its entries. `behind_shebang` is set when
190/// the file starts with a `#!` line, as a zip application does; then only a
191/// zip application is reported, and anything else leaves `out` unchanged
192/// except that a failed read is recorded in its `parse_error`.
193pub(super) fn classify<R: Read + Seek + ?Sized>(
194    src: &mut Source<'_, R>,
195    name: &NameInfo,
196    out: &mut Classification,
197    behind_shebang: bool,
198) -> bool {
199    let (entries, error) = match read_entries(src) {
200        Ok(found) => found,
201        Err(e) if behind_shebang => {
202            // A structure that is not a clean archive is the script the
203            // shebang names; a read that failed is not established.
204            if matches!(e, ParseError::Io(_) | ParseError::ReadBudgetExceeded) {
205                out.parse_error = Some(e);
206            }
207            return false;
208        }
209        Err(e) => {
210            out.content_kind = FileKind::Zip;
211            out.confidence = Confidence::High;
212            out.parse_error = Some(e);
213            return true;
214        }
215    };
216    if behind_shebang {
217        // Only a clean zip application is reported; anything else is the
218        // script the shebang names.
219        if !entries.python_main || error.is_some() {
220            return false;
221        }
222        out.flags.python_zipapp = true;
223        out.content_kind = FileKind::Script(ScriptKind::Python);
224        out.confidence = Confidence::High;
225        out.parse_error = error;
226        return true;
227    }
228    out.confidence = Confidence::High;
229    out.parse_error = error;
230    out.flags.macro_capable_container = entries.vba_project;
231    out.content_kind = if entries.bundle_manifest {
232        out.flags.package_signature_present = entries.appx_signature;
233        if name.is_ext(&["appxbundle"]) {
234            out.extension_assisted = true;
235            FileKind::AppxBundle
236        } else {
237            FileKind::MsixBundle
238        }
239    } else if entries.appx_manifest {
240        out.flags.package_signature_present = entries.appx_signature;
241        if name.is_ext(&["appx"]) {
242            out.extension_assisted = true;
243            FileKind::Appx
244        } else {
245            FileKind::Msix
246        }
247    } else if entries.jar_manifest {
248        FileKind::Jar
249    } else if entries.python_main && name.is_ext(&["pyz", "pyzw"]) {
250        out.flags.python_zipapp = true;
251        out.extension_assisted = true;
252        FileKind::Script(ScriptKind::Python)
253    } else {
254        FileKind::Zip
255    };
256    true
257}

259/// Bytes at the end of the file searched for the end records: the longest
260/// archive comment and the 32-bit end record, plus room for a ZIP64 end
261/// record and locator in front of them (some readers look for the ZIP64
262/// record in this whole window).
263const TAIL_LEN: usize = MAX_COMMENT + END_LEN + ZIP64_END_LEN + ZIP64_LOCATOR_LEN;
264
265/// The size of a ZIP64 end record without extensible data, as its "size of
266/// record" field gives it (the record less its first 12 bytes).
267const ZIP64_PLAIN_SIZE: u64 = 44;
268
269/// The Unicode path extra field: a UTF-8 name some readers use instead of
270/// the entry's own.
271const UNICODE_PATH: u16 = 0x7075;

273/// The central directory, as the end records describe it.
274#[derive(Debug, Clone, Copy, PartialEq, Eq)]
275struct Directory {
276    /// File offset of the first entry.
277    offset: u64,
278    /// Size in bytes.
279    size: u64,
280    /// Number of entries.
281    count: u64,
282}

284/// Reads the central directory the end records describe, and only that.
285///
286/// There is one reading: the directory at its recorded offset, of its
287/// recorded size, ending where the next record (the ZIP64 end record, or
288/// the end record) starts, holding exactly the recorded number of entries.
289/// When the structure allows any other reading, so that ZIP readers could
290/// disagree on the entries, the error is
291/// [`ParseError::ZipDirectoryAmbiguous`]; when an entry runs past the
292/// directory, [`ParseError::ZipDirectoryMalformed`]; when the directory is
293/// over the limits, [`ParseError::ZipDirectoryTruncated`]. The entries read
294/// before such a point are returned with the error. The caller treats any of
295/// them as possibly executable, so a misread fails closed.
296fn read_entries<R: Read + Seek + ?Sized>(
297    src: &mut Source<'_, R>,
298) -> Result<(Entries, Option<ParseError>), ParseError> {
299    let dir = find_directory(src)?;
300    walk(src, dir)
301}

303/// Walks the central directory once, entry by entry. The work is linear in
304/// the directory's size, which is capped at [`MAX_ZIP_DIRECTORY`] bytes and
305/// [`MAX_ZIP_ENTRIES`] entries.
306fn walk<R: Read + Seek + ?Sized>(
307    src: &mut Source<'_, R>,
308    dir: Directory,
309) -> Result<(Entries, Option<ParseError>), ParseError> {
310    let cap = to_u64(MAX_ZIP_DIRECTORY);
311    let over = dir.size > cap;
312    let want = usize::try_from(dir.size.min(cap)).unwrap_or(MAX_ZIP_DIRECTORY);
313    // `find_directory` puts the directory before the end records, inside
314    // the file, so the read is never clamped; a reader that ends early is
315    // an error from `read_at`.
316    let data = src.read_at(dir.offset, want)?;
317    let mut entries = Entries::default();
318    let mut scratch = Vec::new();
319    // An entry cut by the size cap is not malformed; it is past the limit.
320    let cut = |e: ParseError| {
321        if over {
322            ParseError::ZipDirectoryTruncated
323        } else {
324            e
325        }
326    };
327    let mut at = 0_usize;
328    let mut seen = 0_usize;
329    let error = loop {
330        if at == data.len() {
331            break if over {
332                Some(ParseError::ZipDirectoryTruncated)
333            } else if to_u64(seen) != dir.count {
334                // Readers that trust the count and readers that trust the
335                // size see different entries.
336                Some(ParseError::ZipDirectoryAmbiguous)
337            } else {
338                None
339            };
340        }
341        if seen >= MAX_ZIP_ENTRIES {
342            break Some(ParseError::ZipDirectoryTruncated);
343        }
344        let rest = data.get(at..).unwrap_or_default();
345        let Some(sig) = rest.get(..4) else {
346            break Some(cut(ParseError::ZipDirectoryMalformed));
347        };
348        if sig != DIRECTORY_ENTRY {
349            // Another record inside the directory, such as a digital
350            // signature record: readers that stop there and readers that
351            // skip it can see different entries.
352            break Some(ParseError::ZipDirectoryAmbiguous);
353        }
354        let Some(header) = rest.get(..DIRECTORY_ENTRY_LEN) else {
355            break Some(cut(ParseError::ZipDirectoryMalformed));
356        };
357        let field = |off| u16_at(header, off).map_or(0, usize::from);
358        let (name_len, extra_len, comment_len) = (field(28), field(30), field(32));
359        let name_at = at.saturating_add(DIRECTORY_ENTRY_LEN);
360        let Some(name) = bytes(&data, name_at, name_len) else {
361            break Some(cut(ParseError::ZipDirectoryMalformed));
362        };
363        let extra = name_at
364            .checked_add(name_len)
365            .and_then(|e| bytes(&data, e, extra_len));
366        // Names that fit count even if the entry's tail does not.
367        let whole_extra = entries.note_entry(name, extra.unwrap_or_default(), &mut scratch);
368        let next = name_at
369            .checked_add(name_len)
370            .and_then(|n| n.checked_add(extra_len))
371            .and_then(|n| n.checked_add(comment_len))
372            .filter(|n| *n <= data.len());
373        let Some(next) = next else {
374            break Some(cut(ParseError::ZipDirectoryMalformed));
375        };
376        // An extra subfield cut short inside an entry that fits: readers
377        // may stop, skip it or read past it into the comment.
378        if !whole_extra {
379            break Some(ParseError::ZipDirectoryMalformed);
380        }
381        seen = seen.saturating_add(1);
382        at = next;
383    };
384    Ok((entries, error))
385}

392/// Finds the one central directory the end records describe.
393///
394/// The end record is the last end record signature in the last
395/// [`TAIL_LEN`] bytes, and its comment must reach exactly to the end of the
396/// file. A ZIP64 locator just before it must point at a ZIP64 end record
397/// just before the locator, without extensible data, and every 32-bit field
398/// must be the ZIP64 value or its sentinel. Finally the directory's recorded
399/// offset plus its size must be where the next record starts. Any other
400/// layout (a second end record, data after the comment, a ZIP64 record
401/// elsewhere or without a locator, fields that disagree, a directory
402/// shifted by data in front of the archive) lets readers find different
403/// directories, and is [`ParseError::ZipDirectoryAmbiguous`].
404fn find_directory<R: Read + Seek + ?Sized>(
405    src: &mut Source<'_, R>,
406) -> Result<Directory, ParseError> {
407    let len = src.len();
408    let tail_start = len.saturating_sub(to_u64(TAIL_LEN));
409    let tail = src.read_at(tail_start, TAIL_LEN)?;
410    let last = |sig: &[u8; 4], from: usize| {
411        tail.get(from..)
412            .and_then(|t| t.windows(4).rposition(|w| w == sig))
413            .map(|i| i + from)
414    };
415    let end_at = last(END_OF_DIRECTORY, 0).ok_or(ParseError::ZipDirectoryMissing)?;
416    // The last signature, so no reader takes a later one; and its comment
417    // ends at the end of the file, so no reader takes an earlier one.
418    let end = bytes(&tail, end_at, END_LEN).ok_or(ParseError::ZipDirectoryAmbiguous)?;
419    let comment = u16_at(end, 20).map_or(0, usize::from);
420    if end_at + END_LEN + comment != tail.len() {
421        return Err(ParseError::ZipDirectoryAmbiguous);
422    }
423    // An earlier end record whose comment holds this one (or that
424    // overlaps it): a reader that rejects this record, or checks only that
425    // a comment reaches the end, could take that one instead. Earlier
426    // signatures that end before this record (inside stored entry data,
427    // say) are not taken by readers that accept this one.
428    let nested = (0..end_at).any(|p| {
429        bytes(&tail, p, 4) == Some(&END_OF_DIRECTORY[..])
430            && u16_at(&tail, p + 20).is_none_or(|c| p + END_LEN + usize::from(c) > end_at)
431    });
432    if nested {
433        return Err(ParseError::ZipDirectoryAmbiguous);
434    }
435    let end_pos = tail_start + to_u64(end_at);
436
437    let f16 = |off| u16_at(end, off).unwrap_or(0);
438    let f32 = |off| u32_at(end, off).unwrap_or(0);
439    let (disk, dir_disk, here, total) = (f16(4), f16(6), f16(8), f16(10));
440    let (size, offset) = (f32(12), f32(16));
441
442    let locator_at = end_at.checked_sub(ZIP64_LOCATOR_LEN);
443    let record_at = locator_at.and_then(|l| l.checked_sub(ZIP64_END_LEN));
444    let locator = locator_at
445        .and_then(|l| bytes(&tail, l, ZIP64_LOCATOR_LEN))
446        .filter(|l| l.get(..4) == Some(&ZIP64_LOCATOR[..]));
447    let record = record_at
448        .and_then(|r| bytes(&tail, r, ZIP64_END_LEN))
449        .filter(|r| r.get(..4) == Some(&ZIP64_END[..]));
450
451    let (dir, next_record) = match (locator, record, record_at) {
452        (Some(locator), Some(record), Some(record_at)) => {
453            let record_pos = tail_start + to_u64(record_at);
454            let disk_of_record = u32_at(locator, 4);
455            let declared = u64_at(locator, 8);
456            let disks = u32_at(locator, 16);
457            // Readers that see more or fewer than one disk do not use
458            // the ZIP64 record.
459            if disk_of_record != Some(0) || disks != Some(1) || declared != Some(record_pos) {
460                return Err(ParseError::ZipDirectoryAmbiguous);
461            }
462            // Extensible data, or a size that does not reach the locator.
463            if u64_at(record, 4) != Some(ZIP64_PLAIN_SIZE) {
464                return Err(ParseError::ZipDirectoryAmbiguous);
465            }
466            // A later ZIP64 signature: a reader that takes the last one
467            // falls back to the 32-bit record.
468            if last(ZIP64_END, record_at + 4).is_some() {
469                return Err(ParseError::ZipDirectoryAmbiguous);
470            }
471            let g32 = |off| u32_at(record, off).unwrap_or(u32::MAX);
472            let g64 = |off| u64_at(record, off).unwrap_or(u64::MAX);
473            let dir = Directory {
474                offset: g64(48),
475                size: g64(40),
476                count: g64(32),
477            };
478            // Spanned archives, or the two entry counts disagree (readers
479            // use either).
480            if g32(16) != 0 || g32(20) != 0 || g64(24) != dir.count {
481                return Err(ParseError::ZipDirectoryAmbiguous);
482            }
483            // Readers use the 32-bit fields unless one is a sentinel, so
484            // each must be the ZIP64 value or the sentinel.
485            let agrees16 = |v: u16, want: u64| v == u16::MAX || u64::from(v) == want;
486            let agrees32 = |v: u32, want: u64| v == u32::MAX || u64::from(v) == want;
487            let agree = agrees16(disk, 0)
488                && agrees16(dir_disk, 0)
489                && agrees16(here, dir.count)
490                && agrees16(total, dir.count)
491                && agrees32(size, dir.size)
492                && agrees32(offset, dir.offset);
493            if !agree {
494                return Err(ParseError::ZipDirectoryAmbiguous);
495            }
496            (dir, record_pos)
497        }
498        // A locator without its record, or a record without a locator
499        // (some readers take a ZIP64 record there without one).
500        (Some(_), _, _) | (None, Some(_), _) => return Err(ParseError::ZipDirectoryAmbiguous),
501        (None, None, _) => {
502            // Spanned archives, or the two entry counts disagree (readers
503            // use either).
504            if disk != 0 || dir_disk != 0 || here != total {
505                return Err(ParseError::ZipDirectoryAmbiguous);
506            }
507            // ZIP64 sentinels with no ZIP64 record.
508            if size == u32::MAX || offset == u32::MAX {
509                return Err(ParseError::ZipDirectoryMalformed);
510            }
511            let dir = Directory {
512                offset: u64::from(offset),
513                size: u64::from(size),
514                count: u64::from(total),
515            };
516            (dir, end_pos)
517        }
518    };
519    // Readers find the directory from its recorded offset or from where it
520    // ends; both must give the same start. Data in front of the archive is
521    // fine only when the offsets already count it.
522    if dir.offset.checked_add(dir.size) != Some(next_record) {
523        return Err(ParseError::ZipDirectoryAmbiguous);
524    }
525    Ok(dir)
526}
