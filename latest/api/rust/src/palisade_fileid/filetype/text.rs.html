1//! Text, scripts, HTML applications and registry files.
2//!
3//! The head is decoded (byte order marks for UTF-8, UTF-16 and UTF-32, and
4//! UTF-16 without one), then checked in order of certainty:
5//!
6//! 1. definite markers: a shebang line, a registry file header, the
7//!    `<HTA:APPLICATION` element, `<?php`, script component and script file
8//!    elements, and so on ([`Confidence::High`]);
9//! 2. a script signature block marker near the end of the file (#52), which
10//!    names its language;
11//! 3. keyword counts per language ([`Confidence::Low`] or
12//!    [`Confidence::Medium`]). These never override what the extension
13//!    claims (see `apply_name_hints`).
14//!
15//! Content that does not look like text is [`FileKind::Unknown`].
16
17#![forbid(unsafe_code)]

27/// How much of the end of the file is searched for a signature block
28/// marker, when the file is longer than the head.
29const TAIL_LEN: usize = 64 * 1024;

31/// The shortest printable start, in characters, taken as a script's text
32/// when binary data follows it.
33const MIN_PRINTABLE_PREFIX: usize = 8;

35/// Classifies text content.
36pub(super) fn classify<R: Read + Seek + ?Sized>(
37    src: &mut Source<'_, R>,
38    name: &NameInfo,
39    out: &mut Classification,
40) {
41    classify_within(src, name, out, HEAD_LEN);

44/// Classifies text content from only the first `limit` bytes of the head.
45/// Used for a disc image's system area that holds a script, so the volume
46/// descriptors after it are not read as part of the script.
47pub(super) fn classify_within<R: Read + Seek + ?Sized>(
48    src: &mut Source<'_, R>,
49    name: &NameInfo,
50    out: &mut Classification,
51    limit: usize,
52) {
53    let head = src.head();
54    let head = head.get(..limit.min(head.len())).unwrap_or_default();
55    let Some((encoding, bom_len)) = detect_encoding(head) else {
56        return;
57    };
58    // A copy, so the tail can be read below. The head is at most 64 KiB.
59    let body = head.get(bom_len..).unwrap_or_default().to_vec();
60    let body = body.as_slice();
61    let text = decode(body, encoding);
62    // Two views of the same bytes: decoded in the detected encoding, and as
63    // eight-bit text with zero bytes dropped. UTF-16 without a byte order
64    // mark is a guess from zero bytes, and eight-bit text padded with zeros
65    // can pass for it and decode to unrelated characters; the eight-bit
66    // view still shows its markers. For genuine text both views agree on
67    // the ASCII markers looked for.
68    let view = eight_bit_view(body);
69    let guessed = matches!(
70        encoding,
71        TextEncoding::Utf16LeNoBom | TextEncoding::Utf16BeNoBom
72    );
73    let is_text = looks_like_text(body, &text, encoding);
74    let whole_head = limit >= HEAD_LEN;
75
76    // 1. Definite markers, in either view, whether or not the whole head is
77    //    text: binary data before or after a script's text (a
78    //    self-extracting script, or bytes added to dodge detection) does
79    //    not hide them, and neither does a keyword guess.
80    if let Some(found) = best_definite(&text, &view, name) {
81        apply(out, encoding, &found);
82        if out.flags.script_signature_block {
83            return;
84        }
85        // A signature block may still be at the end of a long file.
86        if is_text && whole_head {
87            match tail_signature(src, encoding, bom_len, guessed) {
88                Ok(Some(_)) => out.flags.script_signature_block = true,
89                Ok(None) => {}
90                Err(e) => out.parse_error = Some(e),
91            }
92        }
93        return;
94    }
95
96    if !is_text {
97        // 2. Not text: a script recognised by keywords in its printable
98        //    start, or in its content with zeros and a few stray control
99        //    bytes removed.
100        if let Some(found) = script_prefix(&text)
101            .or_else(|| script_prefix(&view))
102            .or_else(|| lenient_script(body))
103        {
104            apply(out, encoding, &found);
105        }
106        return;
107    }
108
109    // 3. Text: a signature block at the end of a long file settles the
110    //    language, then keywords decide, in either view.
111    out.text_encoding = Some(encoding);
112    out.content_kind = FileKind::Text;
113    out.confidence = Confidence::High;
114    // Only for the whole head: a disc image's system area is classified
115    // alone, without the image's contents.
116    // A tail that could not be read is reported: a signature block there
117    // would have named the language, so the result is not established.
118    let tail = if whole_head {
119        tail_signature(src, encoding, bom_len, guessed).unwrap_or_else(|e| {
120            out.parse_error = Some(e);
121            None
122        })
123    } else {
124        None
125    };
126    if let Some(kind) = tail {
127        out.flags.script_signature_block = true;
128        out.content_kind = FileKind::Script(kind);
129        return;
130    }
131    let lower = text.to_ascii_lowercase();
132    let guess = keywords(&lower).or_else(|| {
133        guessed
134            .then(|| keywords(&view.to_ascii_lowercase()))
135            .flatten()
136    });
137    if let Some((kind, confidence)) = guess {
138        out.content_kind = FileKind::Script(kind);
139        out.confidence = confidence;
140        out.heuristic = true;
141    }
142}

144/// A signature block marker in the last [`TAIL_LEN`] bytes of a file longer
145/// than the head, in the detected encoding or, after a UTF-16 guess, in the
146/// eight-bit view. An error if the tail could not be read.
147fn tail_signature<R: Read + Seek + ?Sized>(
148    src: &mut Source<'_, R>,
149    encoding: TextEncoding,
150    bom_len: usize,
151    guessed: bool,
152) -> Result<Option<ScriptKind>, ParseError> {
153    if src.len() <= u64::try_from(HEAD_LEN).unwrap_or(u64::MAX) {
154        return Ok(None);
155    }
156    let raw = read_tail(src, encoding, bom_len)?;
157    Ok(signature_marker(&decode(&raw, encoding)).or_else(|| {
158        guessed
159            .then(|| signature_marker(&eight_bit_view(&raw)))
160            .flatten()
161    }))
162}

164/// Records what [`definite_in_binary`], [`script_prefix`] or
165/// [`lenient_script`] found.
166fn apply(out: &mut Classification, encoding: TextEncoding, found: &BinaryFind) {
167    out.text_encoding = Some(encoding);
168    out.content_kind = found.kind;
169    out.confidence = found.confidence;
170    out.extension_assisted = found.extension_assisted;
171    out.heuristic = found.heuristic;
172    out.flags.script_signature_block |= found.signature_block;
173    out.flags.encoded_script |= found.encoded;
174}

176/// The bytes as eight-bit text with every zero byte removed, each byte
177/// taken as the character of the same value, so legacy code page bytes
178/// stay single characters. For genuine UTF-16 text in the ASCII range this
179/// is the same text; for eight-bit text padded to look like UTF-16 it is
180/// the eight-bit text.
181///
182/// A zero byte at the very start is kept as U+FFFD, so markers that must
183/// start the file (a shebang line, a registry header) are not found after
184/// it: the system would not read them there either. Genuine BOM-less
185/// UTF-16 big-endian text starts with a zero too, and is read through its
186/// decoded view instead.
187fn eight_bit_view(raw: &[u8]) -> String {
188    let lead = (raw.first() == Some(&0)).then_some(char::REPLACEMENT_CHARACTER);
189    lead.into_iter()
190        .chain(raw.iter().filter(|b| **b != 0).map(|b| char::from(*b)))
191        .collect()
192}

194/// Whether a byte is a control byte that text does not use.
195fn is_binary_byte(b: u8) -> bool {
196    let c = char::from(b);
197    (c.is_ascii_control() && !is_text_control(c)) || b == 0x7f
198}

200/// [`lenient_script`] removes at most one stray control byte (other than
201/// zeros) in this many. Binary formats have far more: fonts, images and
202/// compressed data run at about one in three, and their bytes can spell
203/// short keywords by chance.
204const STRAY_RATIO: usize = 20;
205
206/// A script recognised by keywords once zeros and stray control bytes (at
207/// most one in [`STRAY_RATIO`] of the rest) are removed: padding, or bytes
208/// scattered through a script to dodge detection.
209fn lenient_script(raw: &[u8]) -> Option<BinaryFind> {
210    let kept: Vec<u8> = raw.iter().copied().filter(|b| *b != 0).collect();
211    let stray = kept.iter().filter(|b| is_binary_byte(**b)).count();
212    if kept.len() < MIN_PRINTABLE_PREFIX || stray * STRAY_RATIO > kept.len() {
213        return None;
214    }
215    let view: String = kept
216        .iter()
217        .filter(|b| !is_binary_byte(**b))
218        .map(|b| char::from(*b))
219        .collect();
220    let (kind, confidence) = keywords(&view.to_ascii_lowercase())?;
221    Some(BinaryFind {
222        kind: FileKind::Script(kind),
223        confidence,
224        extension_assisted: false,
225        heuristic: true,
226        signature_block: false,
227        encoded: false,
228    })
229}

231/// The printable start of decoded content: everything before the first
232/// NUL or ASCII control character that text does not use. Legacy code page
233/// bytes (read as U+0080 to U+00FF, or as U+FFFD) do not end it.
234fn printable_prefix(text: &str) -> &str {
235    let end = text
236        .char_indices()
237        .find(|(_, c)| c.is_ascii_control() && !is_text_control(*c))
238        .map_or(text.len(), |(i, _)| i);
239    text.get(..end).unwrap_or_default()
240}

242/// A script recognised by keywords in the printable start of content that
243/// is followed by binary data, as in a self-extracting script or a script
244/// padded with zeros.
245fn script_prefix(text: &str) -> Option<BinaryFind> {
246    let prefix = printable_prefix(text);
247    if prefix.trim().chars().count() < MIN_PRINTABLE_PREFIX {
248        return None;
249    }
250    let (kind, confidence) = keywords(&prefix.to_ascii_lowercase())?;
251    Some(BinaryFind {
252        kind: FileKind::Script(kind),
253        confidence,
254        extension_assisted: false,
255        heuristic: true,
256        signature_block: false,
257        encoded: false,
258    })
259}

261/// The more confident of the definite markers in the decoded and the
262/// eight-bit views, preferring the decoded view on a tie, so a weak marker
263/// in one view (such as the encoded script marker) cannot shadow a strong
264/// one in the other.
265fn best_definite(text: &str, view: &str, name: &NameInfo) -> Option<BinaryFind> {
266    match (
267        definite_in_binary(text, name),
268        definite_in_binary(view, name),
269    ) {
270        (Some(a), Some(b)) if b.confidence > a.confidence => Some(b),
271        (Some(a), _) => Some(a),
272        (None, b) => b,
273    }
274}

276/// What [`definite_in_binary`] found.
277struct BinaryFind {
278    kind: FileKind,
279    confidence: Confidence,
280    extension_assisted: bool,
281    heuristic: bool,
282    signature_block: bool,
283    encoded: bool,
284}

286/// A marker in content that is not text as a whole. The same markers as
287/// for text are searched across the whole decoded head, so NUL or control
288/// bytes before or after them do not hide them; those that must start the
289/// file (a shebang line, a registry header) are still anchored there.
290/// Signature block markers and the encoded script marker are searched too.
291fn definite_in_binary(text: &str, name: &NameInfo) -> Option<BinaryFind> {
292    let lower = text.to_ascii_lowercase();
293    let signed = signature_marker(text);
294    let encoded = lower.contains("#@~^");
295    let found = |kind, confidence, extension_assisted, heuristic| BinaryFind {
296        kind,
297        confidence,
298        extension_assisted,
299        heuristic,
300        signature_block: signed.is_some(),
301        encoded,
302    };
303    if let Some((kind, confidence)) = markers(text, &lower, name) {
304        // HTML is an HTML application only because of its `.hta` name.
305        let assisted = kind == FileKind::Hta && confidence < Confidence::High;
306        return Some(found(kind, confidence, assisted, false));
307    }
308    if let Some(kind) = signed {
309        return Some(found(
310            FileKind::Script(kind),
311            Confidence::High,
312            false,
313            false,
314        ));
315    }
316    encoded.then(|| {
317        let kind = match name.extension_kind() {
318            ExtensionKind::Kind(FileKind::Script(ScriptKind::JScript)) => ScriptKind::JScript,
319            _ => ScriptKind::VbScript,
320        };
321        found(FileKind::Script(kind), Confidence::Medium, false, true)
322    })
323}

325/// Whether the system area in front of a disc image's volume descriptors
326/// holds a script rather than boot code or zeros. The system area is
327/// free-form, so a script can carry a planted descriptor; such a file
328/// stays a script. It does when the file is named as a script or HTML
329/// application (the host runs it whatever else is in it), or when the
330/// system area is text, has a definite marker in either view, or is a
331/// script by keywords as for binary content. A real image's system area is
332/// zeros or boot code.
333pub(super) fn system_area_is_script(area: &[u8], name: &NameInfo) -> bool {
334    if matches!(
335        name.extension_kind(),
336        ExtensionKind::Kind(FileKind::Script(_) | FileKind::Hta)
337    ) {
338        return true;
339    }
340    let Some((encoding, bom_len)) = detect_encoding(area) else {
341        return false;
342    };
343    let body = area.get(bom_len..).unwrap_or_default();
344    let text = decode(body, encoding);
345    looks_like_text(body, &text, encoding)
346        || definite_in_binary(&text, name).is_some()
347        || definite_in_binary(&eight_bit_view(body), name).is_some()
348        || script_prefix(&text).is_some()
349        || script_prefix(&eight_bit_view(body)).is_some()
350        || lenient_script(body).is_some()
351}

353/// The longest shebang line looked past for a ZIP local file header.
354const MAX_SHEBANG_LINE: usize = 512;
355
356/// The script kind a `#!` line names, when that line is followed directly
357/// by a ZIP local file header. This one byte-level test decides the case:
358/// such a file is a zip application or the script the line names, and
359/// never depends on how the rest decodes as text (where `markers` takes
360/// `#![` for a Rust attribute). A Rust source file has no ZIP header on
361/// its second line.
362pub(super) fn shebang_zip(head: &[u8]) -> Option<ScriptKind> {
363    let nl = head
364        .iter()
365        .take(MAX_SHEBANG_LINE)
366        .position(|b| *b == b'\n')?;
367    let interpreter = head.get(..nl)?.strip_prefix(b"#!")?;
368    head.get(nl + 1..)?
369        .starts_with(b"PK\x03\x04")
370        .then(|| shebang(&String::from_utf8_lossy(interpreter)))
371}

373/// The encoding, and the length of its byte order mark. `None` if the head
374/// is empty.
375fn detect_encoding(head: &[u8]) -> Option<(TextEncoding, usize)> {
376    if head.is_empty() {
377        return None;
378    }
379    // UTF-32 first: its little-endian mark starts with UTF-16's.
380    let boms: [(&[u8], TextEncoding); 5] = [
381        (&[0xff, 0xfe, 0, 0], TextEncoding::Utf32Le),
382        (&[0, 0, 0xfe, 0xff], TextEncoding::Utf32Be),
383        (&[0xef, 0xbb, 0xbf], TextEncoding::Utf8Bom),
384        (&[0xff, 0xfe], TextEncoding::Utf16Le),
385        (&[0xfe, 0xff], TextEncoding::Utf16Be),
386    ];
387    if let Some((bom, enc)) = boms.iter().find(|(bom, _)| head.starts_with(bom)) {
388        return Some((*enc, bom.len()));
389    }
390    // The whole head is sampled, so a short run of zero-padded bytes at the
391    // start cannot decide the encoding for the rest.
392    let sample = head;
393    let pairs = sample.len() / 2;
394    if pairs >= 4 {
395        let zeros = |parity: usize| {
396            sample
397                .iter()
398                .skip(parity)
399                .step_by(2)
400                .take(pairs)
401                .filter(|b| **b == 0)
402                .count()
403        };
404        let (even, odd) = (zeros(0), zeros(1));
405        // Mostly-ASCII UTF-16: nearly every high byte is zero, and almost
406        // no low byte is.
407        if odd * 10 >= pairs * 9 && even * 10 <= pairs {
408            return Some((TextEncoding::Utf16LeNoBom, 0));
409        }
410        if even * 10 >= pairs * 9 && odd * 10 <= pairs {
411            return Some((TextEncoding::Utf16BeNoBom, 0));
412        }
413    }
414    Some((TextEncoding::EightBit, 0))
415}

417/// Decodes bytes in `encoding`, replacing anything invalid with U+FFFD. A
418/// trailing partial code unit is dropped.
419fn decode(data: &[u8], encoding: TextEncoding) -> String {
420    match encoding {
421        TextEncoding::EightBit | TextEncoding::Utf8Bom => {
422            String::from_utf8_lossy(data).into_owned()
423        }
424        TextEncoding::Utf16Le | TextEncoding::Utf16LeNoBom => utf16(data, u16::from_le_bytes),
425        TextEncoding::Utf16Be | TextEncoding::Utf16BeNoBom => utf16(data, u16::from_be_bytes),
426        TextEncoding::Utf32Le => utf32(data, u32::from_le_bytes),
427        TextEncoding::Utf32Be => utf32(data, u32::from_be_bytes),
428    }
429}

431fn utf16(data: &[u8], unit: fn([u8; 2]) -> u16) -> String {
432    let units = data.chunks_exact(2).map(|c| {
433        unit([
434            c.first().copied().unwrap_or(0),
435            c.get(1).copied().unwrap_or(0),
436        ])
437    });
438    char::decode_utf16(units)
439        .map(|c| c.unwrap_or(char::REPLACEMENT_CHARACTER))
440        .collect()
441}
442
443fn utf32(data: &[u8], unit: fn([u8; 4]) -> u32) -> String {
444    data.chunks_exact(4)
445        .map(|c| {
446            let mut word = [0_u8; 4];
447            word.copy_from_slice(c);
448            char::from_u32(unit(word)).unwrap_or(char::REPLACEMENT_CHARACTER)
449        })
450        .collect()
451}

453/// Whether content is text. Eight-bit content is judged on its bytes, so
454/// legacy code pages pass and binary data does not: no NUL, and at most one
455/// control byte in a hundred. Wider encodings are judged on the decoded
456/// characters, counting invalid code units as control characters.
457fn looks_like_text(raw: &[u8], text: &str, encoding: TextEncoding) -> bool {
458    let allowed = is_text_control;
459    let (total, control) = if matches!(encoding, TextEncoding::EightBit | TextEncoding::Utf8Bom) {
460        if raw.contains(&0) {
461            return false;
462        }
463        let control = raw
464            .iter()
465            .filter(|b| {
466                let c = char::from(**b);
467                (c.is_ascii_control() && !allowed(c)) || **b == 0x7f
468            })
469            .count();
470        (raw.len(), control)
471    } else {
472        let mut total = 0_usize;
473        let mut control = 0_usize;
474        for c in text.chars() {
475            total += 1;
476            if (c.is_control() && !allowed(c)) || c == char::REPLACEMENT_CHARACTER {
477                control += 1;
478            }
479        }
480        (total, control)
481    };
482    total > 0 && control * 100 <= total
483}

485/// Control characters that are normal in text: tab, line feed, form feed,
486/// carriage return, escape and the DOS end-of-file mark.
487const fn is_text_control(c: char) -> bool {
488    matches!(c, '\t' | '\n' | '\x0c' | '\r' | '\x1b' | '\x1a')
489}

491/// Reads the last [`TAIL_LEN`] bytes, aligned to the encoding's code unit.
492fn read_tail<R: Read + Seek + ?Sized>(
493    src: &mut Source<'_, R>,
494    encoding: TextEncoding,
495    bom_len: usize,
496) -> Result<Vec<u8>, ParseError> {
497    let unit: u64 = match encoding {
498        TextEncoding::EightBit | TextEncoding::Utf8Bom => 1,
499        TextEncoding::Utf32Le | TextEncoding::Utf32Be => 4,
500        _ => 2,
501    };
502    // Both fit in a `u64` on every supported target.
503    let bom = u64::try_from(bom_len).unwrap_or(u64::MAX);
504    let tail = u64::try_from(TAIL_LEN).unwrap_or(u64::MAX);
505    let start = src.len().saturating_sub(tail).max(bom);
506    // Align to a code unit boundary counted from the end of the mark.
507    let start = start + (unit - (start - bom) % unit) % unit;
508    Ok(src.read_at(start, TAIL_LEN)?)
509}

511/// Script signature block markers, and the language each one is for.
512const SIGNATURE_MARKERS: [(&str, ScriptKind); 4] = [
513    ("# SIG # Begin signature block", ScriptKind::PowerShell),
514    ("'' SIG '' Begin signature block", ScriptKind::VbScript),
515    ("// SIG // Begin signature block", ScriptKind::JScript),
516    ("** SIG ** Begin signature block", ScriptKind::Wsf),
517];
518
519fn signature_marker(text: &str) -> Option<ScriptKind> {
520    SIGNATURE_MARKERS
521        .iter()
522        .find(|(marker, _)| text.contains(marker))
523        .map(|(_, kind)| *kind)
524}

526/// Definite markers: shebang lines, file headers and defining elements.
527fn markers(text: &str, lower: &str, name: &NameInfo) -> Option<(FileKind, Confidence)> {
528    let high = |kind| Some((kind, Confidence::High));
529    let first_line = text
530        .lines()
531        .next()
532        .unwrap_or_default()
533        .trim_start_matches('\u{feff}');
534    // `#![` starts a Rust inner attribute, which the Rust reference does
535    // not take as a shebang line. As a shebang it would name a relative
536    // interpreter path starting with `[`, which no real script does. Such
537    // a file still goes through the keyword and name rules below.
538    if let Some(interpreter) = first_line.strip_prefix("#!")
539        && !interpreter.trim_start().starts_with('[')
540    {
541        return high(FileKind::Script(shebang(interpreter)));
542    }
543    let start = lower.trim_start();
544    if start.starts_with("windows registry editor version 5.00") || start.starts_with("regedit4") {
545        return high(FileKind::Reg);
546    }
547    if lower.contains("<hta:application") {
548        return high(FileKind::Hta);
549    }
550    if lower.contains("<?php") {
551        return high(FileKind::Script(ScriptKind::Php));
552    }
553    if lower.contains("<scriptlet")
554        || (lower.contains("<component") && lower.contains("<registration"))
555    {
556        return high(FileKind::Script(ScriptKind::Wsc));
557    }
558    // The `language` attribute is optional (it defaults to VBScript), as is
559    // inline code (`<script src="..."/>`).
560    if lower.contains("<job") && lower.contains("<script") {
561        return high(FileKind::Script(ScriptKind::Wsf));
562    }
563    let xsl = lower.contains("<xsl:stylesheet") || lower.contains("<xsl:transform");
564    if xsl && (lower.contains(":script") || lower.contains("<script")) {
565        return high(FileKind::Script(ScriptKind::Xsl));
566    }
567    if lower.contains("<pcsettings") && lower.contains("<deeplink") {
568        return high(FileKind::Script(ScriptKind::SettingContent));
569    }
570    if lower.contains("[version]")
571        && (lower.contains("$windows nt$") || lower.contains("$chicago$"))
572    {
573        return high(FileKind::Script(ScriptKind::Inf));
574    }
575    // Last: HTML is an HTML application only because of its `.hta` name,
576    // so every marker above, which the content defines, comes first.
577    let html = [
578        "<html",
579        "<!doctype html",
580        "<head",
581        "<body",
582        "<script",
583        "<svg",
584        "<iframe",
585        "<object",
586        "<embed",
587        "<img",
588    ]
589    .iter()
590    .any(|tag| lower.contains(tag));
591    (html && name.is_ext(&["hta"])).then_some((FileKind::Hta, Confidence::Medium))
592}

594/// The script kind a shebang line's interpreter names. The interpreter is
595/// the program's file name, or the first argument of `env` (after any `-S`
596/// or other options), with version suffixes ignored.
597fn shebang(line: &str) -> ScriptKind {
598    let mut words = line.split_whitespace();
599    let program = words.next().unwrap_or_default();
600    let file = |path: &'_ str| -> String {
601        let base = path
602            .rsplit(['/', '\\'])
603            .next()
604            .unwrap_or_default()
605            .to_ascii_lowercase();
606        base.strip_suffix(".exe").map(str::to_owned).unwrap_or(base)
607    };
608    let mut base = file(program);
609    if base == "env" {
610        base = file(env_command(words).unwrap_or_default());
611    }
612    let stem = base.trim_end_matches(|c: char| c.is_ascii_digit() || c == '.');
613    match stem {
614        "python" | "pypy" => ScriptKind::Python,
615        "pwsh" | "powershell" => ScriptKind::PowerShell,
616        "perl" => ScriptKind::Perl,
617        "ruby" => ScriptKind::Ruby,
618        "php" => ScriptKind::Php,
619        "node" | "nodejs" | "deno" | "bun" => ScriptKind::JScript,
620        _ => ScriptKind::Shell,
621    }
622}

624/// The command `env` runs, given its arguments: the first word that is not
625/// an option, an option's operand or a `NAME=VALUE` assignment. `-u`, `-C`,
626/// `-P` and `-a` (and `--unset`, `--chdir` and `--argv0`) take an operand,
627/// in the same word or the next. `-S` takes the rest of the line, already
628/// split into words here, so the command follows it as for any option; a
629/// command written in the same word (`-Spython3`, `--split-string=python3`)
630/// is that command.
631fn env_command<'a>(mut words: impl Iterator<Item = &'a str>) -> Option<&'a str> {
632    const SHORT_WITH_OPERAND: [char; 4] = ['u', 'C', 'P', 'a'];
633    const LONG_WITH_OPERAND: [&str; 3] = ["unset", "chdir", "argv0"];
634    while let Some(word) = words.next() {
635        if word == "--" {
636            return words.find(|w| !w.contains('='));
637        }
638        if let Some(long) = word.strip_prefix("--") {
639            if let Some(command) = long.strip_prefix("split-string=") {
640                return Some(command);
641            }
642            if LONG_WITH_OPERAND.contains(&long) {
643                words.next();
644            }
645            continue;
646        }
647        if let Some(cluster) = word.strip_prefix('-') {
648            for (i, c) in cluster.char_indices() {
649                let rest = cluster.get(i + c.len_utf8()..).unwrap_or_default();
650                if c == 'S' && !rest.is_empty() {
651                    return Some(rest);
652                }
653                if SHORT_WITH_OPERAND.contains(&c) {
654                    // The operand is the rest of this word, or the next.
655                    if rest.is_empty() {
656                        words.next();
657                    }
658                    break;
659                }
660            }
661            continue;
662        }
663        if !word.contains('=') {
664            return Some(word);
665        }
666    }
667    None
668}

788/// Hits needed for a low-confidence guess, and for a medium one.
789const LOW_HITS: usize = 2;
790const MEDIUM_HITS: usize = 4;
791
792/// The best keyword match, if one language clearly leads.
793fn keywords(lower: &str) -> Option<(ScriptKind, Confidence)> {
794    let mut scores: Vec<(ScriptKind, usize)> = KEYWORDS
795        .iter()
796        .map(|(kind, words)| (*kind, words.iter().filter(|w| lower.contains(*w)).count()))
797        .collect();
798    scores.sort_by_key(|s| std::cmp::Reverse(s.1));
799    let (kind, best) = *scores.first()?;
800    let runner_up = scores.get(1).map_or(0, |s| s.1);
801    // A tie is no answer.
802    if best < LOW_HITS || best == runner_up {
803        return None;
804    }
805    let confidence = if best >= MEDIUM_HITS && best >= runner_up * 2 {
806        Confidence::Medium
807    } else {
808        Confidence::Low
809    };
810    Some((kind, confidence))
811}
