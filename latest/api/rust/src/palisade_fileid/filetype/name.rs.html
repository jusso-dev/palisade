1//! What a file name claims: the extension, bidirectional control characters
2//! and disguising double extensions.
3//!
4//! The name is never trusted to decide what a file is. It is used to report
5//! the claim, to grade a mismatch, and to break ties where the content has
6//! no magic.
7
8#![forbid(unsafe_code)]
9
10use std::ffi::OsStr;

12use super::{ExtensionKind, FileKind, PeSubKind, ScriptKind};

14/// The longest extension kept. Longer "extensions" are reported as
15/// [`ExtensionKind::Other`] and are not stored.
16const MAX_EXTENSION_LEN: usize = 32;
17
18/// The parts of a file name the classifier uses.
19#[derive(Debug, Clone, Default)]
20pub(super) struct NameInfo {
21    /// The last extension, lowercased.
22    pub(super) extension: Option<String>,
23    /// The name has an extension longer than [`MAX_EXTENSION_LEN`]. It is
24    /// not stored, and claims [`ExtensionKind::Other`].
25    pub(super) long_extension: bool,
26    /// The extension before it, lowercased and trimmed, if any.
27    pub(super) previous_extension: Option<String>,
28    /// The name has a bidirectional control character.
29    pub(super) bidi_control: bool,
30}
31
32/// Bidirectional control characters that can make a name display in a
33/// different order from the one the system reads it in.
34const fn is_bidi_control(c: char) -> bool {
35    matches!(
36        c,
37        '\u{061C}' | '\u{200E}' | '\u{200F}' | '\u{202A}'..='\u{202E}' | '\u{2066}'..='\u{2069}'
38    )
39}
40
41/// Trailing dots and spaces, which the OS drops from names.
42const TRAILING: [char; 2] = ['.', ' '];
43
44/// Splits `stem.ext`. The stem must not be empty: a leading dot
45/// (`.profile`) makes a hidden-file name, not an extension. A stem of
46/// spaces is a stem: the OS strips only trailing dots and spaces.
47fn split_extension(name: &str) -> Option<(&str, &str)> {
48    let (stem, ext) = name.rsplit_once('.')?;
49    (!stem.is_empty()).then_some((stem, ext))
50}
51
52impl NameInfo {
53    /// Parses a file name or path. A leading drive (`C:`) is dropped, both
54    /// `/` and `\` separate components, and an alternate data stream suffix
55    /// (`name.txt:stream`) is dropped, so the name is the file that holds
56    /// the stream. As the OS does, trailing dots and spaces are ignored,
57    /// before the last extension and before the one in front of it.
58    pub(super) fn parse(name: &OsStr) -> Self {
59        let name = name.to_string_lossy();
60        let path = match name.as_bytes() {
61            [drive, b':', ..] if drive.is_ascii_alphabetic() => name.get(2..).unwrap_or_default(),
62            _ => &name,
63        };
64        let component = path.rsplit(['/', '\\']).next().unwrap_or_default();
65        // A bidi control anywhere in the component, the stream name
66        // included, is reported.
67        let bidi_control = component.chars().any(is_bidi_control);
68        let base = component.split(':').next().unwrap_or_default();
69        // Only ASCII case is folded. Spaces after the dot and other
70        // whitespace stay part of the extension, as they do for the OS, so
71        // `app. exe` and `app.exe\u{a0}` are not `.exe`.
72        let clean = |s: &str| s.to_ascii_lowercase();
73        let mut info = Self {
74            bidi_control,
75            ..Self::default()
76        };
77        let Some((stem, ext)) = split_extension(base.trim_end_matches(TRAILING)) else {
78            return info;
79        };
80        let ext = clean(ext);
81        if ext.is_empty() {
82            return info;
83        }
84        if ext.len() > MAX_EXTENSION_LEN {
85            info.long_extension = true;
86            return info;
87        }
88        info.extension = Some(ext);
89        info.previous_extension = split_extension(stem.trim_end_matches(TRAILING))
90            .map(|(_, prev)| clean(prev))
91            .filter(|p| !p.is_empty() && p.len() <= MAX_EXTENSION_LEN);
92        info
93    }
94
95    /// Whether the extension is one of `exts`.
96    pub(super) fn is_ext(&self, exts: &[&str]) -> bool {
97        self.extension.as_deref().is_some_and(|e| exts.contains(&e))
98    }
99
100    /// What the extension claims.
101    pub(super) fn extension_kind(&self) -> ExtensionKind {
102        if self.long_extension {
103            return ExtensionKind::Other;
104        }
105        self.extension.as_deref().map_or(ExtensionKind::None, claim)
106    }
107
108    /// An inert extension, or one of a recognised kind that cannot run code
109    /// (an archive or disc image), before one that claims executable
110    /// content, as in `invoice.pdf.exe` or `photos.zip.exe`.
111    pub(super) fn double_extension(&self) -> bool {
112        let last_runs = matches!(
113            self.extension_kind(),
114            ExtensionKind::Kind(k) if k.is_executable_content()
115        );
116        last_runs
117            && self
118                .previous_extension
119                .as_deref()
120                .is_some_and(|p| match claim(p) {
121                    ExtensionKind::Inert => true,
122                    ExtensionKind::Kind(k) => !k.is_executable_content(),
123                    ExtensionKind::None | ExtensionKind::Other => false,
124                })
125    }
126}

230/// What an extension (lowercased) claims.
231fn claim(ext: &str) -> ExtensionKind {
232    if PE_EXTENSIONS.iter().any(|(e, _)| *e == ext) {
233        return ExtensionKind::Kind(FileKind::Pe);
234    }
235    if let Some((_, s)) = SCRIPT_EXTENSIONS.iter().find(|(e, _)| *e == ext) {
236        return ExtensionKind::Kind(FileKind::Script(*s));
237    }
238    if let Some((_, k)) = OTHER_EXTENSIONS.iter().find(|(e, _)| *e == ext) {
239        return ExtensionKind::Kind(*k);
240    }
241    if INERT_EXTENSIONS.contains(&ext) {
242        return ExtensionKind::Inert;
243    }
244    ExtensionKind::Other
245}

247/// Whether `content` (with `sub_kind` for a PE) is what the extension
248/// claims (`claimed`).
249pub(super) fn claim_matches(
250    name: &NameInfo,
251    claimed: FileKind,
252    content: FileKind,
253    sub_kind: Option<PeSubKind>,
254) -> bool {
255    let ext = name.extension.as_deref().unwrap_or_default();
256    match (claimed, content) {
257        (FileKind::Pe, FileKind::Pe) => PE_EXTENSIONS
258            .iter()
259            .find(|(e, _)| *e == ext)
260            .zip(sub_kind)
261            .is_some_and(|((_, fits), sub)| fits.contains(&sub)),
262        // Executable names hold older executable formats too: DOS and
263        // 16- and 32-bit DOS-extender programs in `.exe`, and 16-bit
264        // libraries and drivers in `.dll`, `.drv` and the like.
265        (FileKind::Pe, FileKind::Mz16 | FileKind::Ne16 | FileKind::Le | FileKind::Lx) => {
266            ext == "exe"
267                || (content == FileKind::Ne16 && matches!(ext, "dll" | "drv" | "cpl" | "scr"))
268        }
269        // A `.com` name can hold a DOS program of any format, or a PE
270        // executable; the system runs all of them.
271        (FileKind::Com, FileKind::Com | FileKind::Mz16) => true,
272        (FileKind::Com, FileKind::Pe) => sub_kind == Some(PeSubKind::Exe),
273        // App packages and bundles are told apart from their `.msix` forms
274        // only by name, so the pairs fit either way.
275        (FileKind::Msix | FileKind::Appx, FileKind::Msix | FileKind::Appx) => true,
276        (
277            FileKind::MsixBundle | FileKind::AppxBundle,
278            FileKind::MsixBundle | FileKind::AppxBundle,
279        ) => true,
280        // A merge module has the installer database CLSID.
281        (FileKind::Msm, FileKind::Msi) => true,
282        // A package or archive renamed to `.zip` is a mismatch: only a
283        // plain ZIP fits, through the comparison below.
284        (claimed, content) => claimed == content,
285    }
286}

288#[cfg(test)]
289mod tests {
290    use super::*;
291
292    fn parse(s: &str) -> NameInfo {
293        NameInfo::parse(OsStr::new(s))
294    }
295
296    #[test]
297    fn extensions() {
298        assert_eq!(parse("a.EXE").extension.as_deref(), Some("exe"));
299        assert_eq!(parse(r"C:\dir.d\file").extension, None);
300        assert_eq!(
301            parse("/x/y/archive.tar.gz").extension.as_deref(),
302            Some("gz")
303        );
304        assert_eq!(
305            parse("/x/y/archive.tar.gz").previous_extension.as_deref(),
306            Some("tar")
307        );
308        assert_eq!(parse(".profile").extension, None);
309        assert_eq!(parse("a.exe. . ").extension.as_deref(), Some("exe"));
310        assert_eq!(parse("a.txt:hidden.exe").extension.as_deref(), Some("txt"));
311        assert_eq!(parse("noext").extension, None);
312        assert_eq!(parse("a.").extension, None);
313        let long = parse(&format!("a.{}", "x".repeat(40)));
314        assert_eq!(long.extension, None);
315        assert_eq!(long.extension_kind(), ExtensionKind::Other);
316        // A one-letter file with a stream is not a drive.
317        assert_eq!(parse(r"C:\temp\a:hidden.exe").extension, None);
318        assert_eq!(
319            parse(r"C:\temp\a.txt:hidden.exe").extension.as_deref(),
320            Some("txt")
321        );
322        assert_eq!(parse("C:app.exe").extension.as_deref(), Some("exe"));
323        assert!(parse("a.txt:x\u{202E}exe").bidi_control);
324        assert_eq!(parse("   .cmd").extension.as_deref(), Some("cmd"));
325        assert_eq!(
326            parse("   .pdf.exe").previous_extension.as_deref(),
327            Some("pdf")
328        );
329        assert_eq!(parse("app. exe").extension_kind(), ExtensionKind::Other);
330        assert_eq!(
331            parse("app.exe\u{a0}").extension_kind(),
332            ExtensionKind::Other
333        );
334    }
335
336    #[test]
337    fn double_extensions() {
338        assert!(parse("invoice.pdf.exe").double_extension());
339        assert!(parse("invoice.pdf          .exe").double_extension());
340        assert!(parse("report.txt.ps1").double_extension());
341        assert!(parse("invoice.pdf..exe").double_extension());
342        assert!(parse("invoice.pdf. .exe").double_extension());
343        assert!(parse("photos.zip.exe").double_extension());
344        assert!(!parse("archive.tar.gz").double_extension());
345        assert!(!parse("setup.x64.exe").double_extension());
346        assert!(!parse(".pdf.exe").double_extension());
347        assert!(!parse("a.exe.pdf").double_extension());
348    }
349
350    #[test]
351    fn bidi_controls() {
352        assert!(parse("invoice\u{202E}fdp.exe").bidi_control);
353        assert!(parse("a\u{2067}b.txt").bidi_control);
354        assert!(!parse("plain.exe").bidi_control);
355    }
356
357    #[test]
358    fn claims() {
359        assert_eq!(claim("dll"), ExtensionKind::Kind(FileKind::Pe));
360        assert_eq!(
361            claim("cmd"),
362            ExtensionKind::Kind(FileKind::Script(ScriptKind::Batch))
363        );
364        assert_eq!(
365            claim("msixbundle"),
366            ExtensionKind::Kind(FileKind::MsixBundle)
367        );
368        assert_eq!(claim("jpg"), ExtensionKind::Inert);
369        assert_eq!(claim("xyz"), ExtensionKind::Other);
370    }
