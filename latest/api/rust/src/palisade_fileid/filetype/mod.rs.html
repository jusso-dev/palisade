1//! Content-based file type detection (#44).
2//!
3//! [`classify`] decides what a file is from its bytes (magic numbers and
4//! structure), never from its name alone. The name, when given, is used only
5//! to report what the extension claims ([`ExtensionKind`]), to flag a
6//! mismatch between the two, and as a tie-breaker in the few places the
7//! format has no magic (DOS `.com` files, and text that could be any of
8//! several script languages).
9//!
10//! The input is untrusted:
11//!
12//! - The classifier reads the first [`HEAD_LEN`] bytes, then makes a few
13//!   targeted reads whose total is capped at [`MAX_TARGETED_READ`]. It never
14//!   loads the whole file.
15//! - No input can make it panic. Every read is bounds-checked, all offset
16//!   arithmetic is checked, and every loop over file data has a fixed limit.
17//! - It never fails. A file it cannot make sense of is
18//!   [`FileKind::Unknown`]; a file whose magic is recognised but whose
19//!   structures are broken keeps the outer kind. Either way the reason is in
20//!   [`Classification::parse_error`].
21//!
22//! Every result carries [`CLASSIFIER_VERSION`], so results cached by a later
23//! stage can be invalidated when the heuristics change.
24

68/// What a file is, judged from its content.
69#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
70#[non_exhaustive]
71pub enum FileKind {
72    /// A PE32 or PE32+ image. [`Classification::pe`] has the sub-kind.
73    Pe,
74    /// A 16-bit "new executable" (`NE` header).
75    Ne16,
76    /// A "linear executable" (`LE` header), as used by 32-bit DOS extenders
77    /// and old virtual device drivers.
78    Le,
79    /// A "linear executable" (`LX` header).
80    Lx,
81    /// A DOS `MZ` program without a newer header, or an `MZ` file whose PE
82    /// headers could not be read.
83    Mz16,
84    /// A DOS `.com` program. The format has no magic, so this is a heuristic
85    /// ([`Classification::heuristic`] is set).
86    Com,
87    /// An installer database: a compound file whose root CLSID is
88    /// `{000C1084-0000-0000-C000-000000000046}`.
89    Msi,
90    /// An installer patch: root CLSID `{000C1086-0000-0000-C000-000000000046}`.
91    Msp,
92    /// An installer transform: root CLSID
93    /// `{000C1082-0000-0000-C000-000000000046}`.
94    Mst,
95    /// An installer merge module. It has the installer database CLSID, so it
96    /// is told apart from [`Msi`](Self::Msi) only by a `.msm` name.
97    Msm,
98    /// Any other OLE compound file.
99    OtherCfb,
100    /// A cabinet archive (`MSCF`).
101    Cab,
102    /// A ZIP archive not recognised as anything more specific.
103    Zip,
104    /// An app package (`AppxManifest.xml` at the root).
105    Msix,
106    /// An app package named `.appx`. The format is the same as
107    /// [`Msix`](Self::Msix); only the name differs.
108    Appx,
109    /// An app bundle (`AppxMetadata/AppxBundleManifest.xml`).
110    MsixBundle,
111    /// An app bundle named `.appxbundle`.
112    AppxBundle,
113    /// A Java archive (`META-INF/MANIFEST.MF`).
114    Jar,
115    /// A compiled HTML help file (`ITSF`).
116    Chm,
117    /// A shell link (`.lnk`): header size `0x4C` and the shell link CLSID.
118    Lnk,
119    /// An HTML application.
120    Hta,
121    /// A script run by an interpreter.
122    Script(ScriptKind),
123    /// A registry import file.
124    Reg,
125    /// An ELF object or executable.
126    Elf,
127    /// A Mach-O object or executable, thin or fat.
128    MachO,
129    /// An ISO 9660 or UDF disc image.
130    Iso,
131    /// A 7z archive.
132    SevenZip,
133    /// A RAR archive (version 4 or 5).
134    Rar,
135    /// A gzip stream.
136    Gzip,
137    /// Text that is not recognised as a script.
138    Text,
139    /// Nothing recognised.
140    Unknown,
141}

403/// Why part of a file could not be read or parsed.
404///
405/// A recognised container with a parse error keeps its outer kind
406/// (`Zip`, `OtherCfb`, `Mz16`), which may be less specific than the file
407/// really is: a package whose directory was cut short, for one, is `Zip`.
408/// So [`Classification::is_executable_content`] is true for a `Zip` with a
409/// parse error, and callers making allow decisions should treat any
410/// classification with a parse error as not established, and fail closed.
411#[derive(Debug, Clone, PartialEq, Eq)]
412#[non_exhaustive]
413pub enum ParseError {
414    /// The reader failed.
415    Io(io::ErrorKind),
416    /// A targeted read would have gone over [`MAX_TARGETED_READ`].
417    ReadBudgetExceeded,
418    /// The file has a PE signature, but its headers could not be read, so
419    /// it is classified as [`FileKind::Mz16`].
420    Pe(PeErrorKind),
421    /// The file ends inside the PE optional header or section table that
422    /// its COFF header declares, so it is classified as
423    /// [`FileKind::Mz16`].
424    PeHeadersTruncated,
425    /// The compound file header is invalid.
426    CfbHeader,
427    /// The compound file's root directory entry could not be read, or the
428    /// file ends inside it. When the CLSID itself was read, it still
429    /// decides the kind.
430    CfbDirectory,
431    /// No ZIP end of central directory record was found.
432    ZipDirectoryMissing,
433    /// The ZIP central directory is malformed: an entry runs past the
434    /// directory, an entry's extra field ends inside a subfield, the
435    /// directory runs past the file, or the end records are broken.
436    ZipDirectoryMalformed,
437    /// The ZIP central directory is larger than [`MAX_ZIP_DIRECTORY`] or
438    /// has more than [`MAX_ZIP_ENTRIES`] entries; only part was examined.
439    ZipDirectoryTruncated,
440    /// The ZIP end records and central directory do not form the single,
441    /// consistent layout the APPNOTE describes, so ZIP readers could
442    /// disagree on the entries: more than one end record could be taken,
443    /// data follows the archive comment, the ZIP64 and 32-bit records
444    /// disagree, the recorded offset and size do not meet the next record,
445    /// the entry count does not match, or another record sits inside the
446    /// directory.
447    ZipDirectoryAmbiguous,
448}
449
450impl std::fmt::Display for ParseError {
451    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
452        match self {
453            Self::Io(kind) => write!(f, "read failed: {kind}"),
454            Self::ReadBudgetExceeded => f.write_str("targeted read budget exceeded"),
455            Self::Pe(kind) => write!(f, "PE headers could not be read: {kind:?}"),
456            Self::PeHeadersTruncated => {
457                f.write_str("file ends inside the PE optional header or section table")
458            }
459            Self::CfbHeader => f.write_str("compound file header is invalid"),
460            Self::CfbDirectory => f.write_str("compound file root entry could not be read"),
461            Self::ZipDirectoryMissing => f.write_str("no ZIP end of central directory record"),
462            Self::ZipDirectoryMalformed => f.write_str("ZIP central directory is malformed"),
463            Self::ZipDirectoryTruncated => {
464                f.write_str("ZIP central directory is over the size or entry limit")
465            }
466            Self::ZipDirectoryAmbiguous => {
467                f.write_str("ZIP end records or central directory are ambiguous")
468            }
469        }
470    }
471}

484/// The result of [`classify`].
485#[derive(Debug, Clone, PartialEq, Eq)]
486pub struct Classification {
487    /// [`CLASSIFIER_VERSION`] at the time of classification.
488    pub classifier_version: u16,
489    /// What the content is.
490    pub content_kind: FileKind,
491    /// What the extension claims.
492    pub extension_kind: ExtensionKind,
493    /// The lowercased extension, if the name has one.
494    pub extension: Option<String>,
495    /// Content and name disagree (`mismatch_severity` is not `None`).
496    pub mismatch: bool,
497    /// How serious the disagreement is.
498    pub mismatch_severity: MismatchSeverity,
499    /// How sure the classifier is of `content_kind`.
500    pub confidence: Confidence,
501    /// `content_kind` rests on a heuristic rather than a defined magic or
502    /// structure: a DOS `.com` file, or a script found by keywords or by
503    /// its extension.
504    pub heuristic: bool,
505    /// The extension decided `content_kind` where content alone could not:
506    /// a script language, `.com`, `.msm`, `.appx`, `.scr`, `.hta` with HTML
507    /// content, or `.pyc`.
508    pub extension_assisted: bool,
509    /// PE details, for [`FileKind::Pe`].
510    pub pe: Option<PeDetails>,
511    /// The text encoding, for text and script content.
512    pub text_encoding: Option<TextEncoding>,
513    /// Informational flags.
514    pub flags: ContentFlags,
515    /// The file length.
516    pub file_size: u64,
517    /// Why part of the file could not be read or parsed, if it could not.
518    pub parse_error: Option<ParseError>,
519}

521impl Classification {
522    /// Whether the content can run code; see
523    /// [`FileKind::is_executable_content`]. It is also true for a
524    /// [`FileKind::Zip`] with a [`parse_error`](Self::parse_error): its
525    /// structure was ambiguous, broken or too large to read in full, so it
526    /// may be an app package, Java archive or zip application that some ZIP
527    /// reader runs, and it fails closed.
528    #[must_use]
529    pub const fn is_executable_content(&self) -> bool {
530        self.content_kind.is_executable_content()
531            || (self.parse_error.is_some() && matches!(self.content_kind, FileKind::Zip))
532    }
533}

535/// Classifies a file from its content.
536///
537/// `name_hint` is the file name or path, if known. It is used to report the
538/// extension's claim and any mismatch, and as a tie-breaker where the
539/// content has no magic. The reader is left at an unspecified position.
540pub fn classify<R: Read + Seek + ?Sized>(
541    reader: &mut R,
542    name_hint: Option<&OsStr>,
543) -> Classification {
544    let name = name_hint.map(NameInfo::parse).unwrap_or_default();
545    let mut out = Classification {
546        classifier_version: CLASSIFIER_VERSION,
547        content_kind: FileKind::Unknown,
548        extension_kind: name.extension_kind(),
549        extension: name.extension.clone(),
550        mismatch: false,
551        mismatch_severity: MismatchSeverity::None,
552        confidence: Confidence::Low,
553        heuristic: false,
554        extension_assisted: false,
555        pe: None,
556        text_encoding: None,
557        flags: ContentFlags {
558            name_bidi_control: name.bidi_control,
559            name_double_extension: name.double_extension(),
560            ..ContentFlags::default()
561        },
562        file_size: 0,
563        parse_error: None,
564    };
565    match Source::open(reader) {
566        Ok(mut src) => {
567            out.file_size = src.len();
568            detect(&mut src, &name, &mut out);
569            // The reader ended before the length it reported: the result
570            // describes only part of the file, so it is not established.
571            // That is the reason reported, over any structure error it
572            // caused (a compound file header cut short, say).
573            if src.head_short() {
574                out.parse_error = Some(ParseError::Io(io::ErrorKind::UnexpectedEof));
575            }
576            apply_name_hints(&name, &mut out);
577        }
578        Err(e) => out.parse_error = Some(e.into()),
579    }
580    let severity = mismatch_severity(&out, &name);
581    out.mismatch_severity = severity;
582    out.mismatch = severity != MismatchSeverity::None;
583    out
584}

598/// Content detection: magic numbers first, then structure, then text.
599fn detect<R: Read + Seek + ?Sized>(
600    src: &mut Source<'_, R>,
601    name: &NameInfo,
602    out: &mut Classification,
603) {
604    let head = src.head();
605    if head.starts_with(b"MZ") {
606        pe_kind::classify(src, name, out);
607        return;
608    }
609    if let Some(kind) = magic::fixed(head) {
610        set(out, kind);
611        return;
612    }
613    if head.starts_with(&cfb::SIGNATURE) {
614        cfb::classify(src, name, out);
615        return;
616    }
617    if zip::starts_like_zip(head) {
618        zip::classify(src, name, out, false);
619        return;
620    }
621    // A zip application: a shebang line, then a ZIP archive. The archive
622    // is binary, so this is checked before the text test. Anything but a
623    // clean zip application is the script the `#!` line names, at high
624    // confidence, and the rest of the file (text, keywords, markers in
625    // the archive's bytes, a disc image descriptor) is not classified:
626    // the `#!` line alone decides. A read failure in the probe is kept,
627    // since the archive could not be ruled out.
628    if let Some(script) = text::shebang_zip(head) {
629        let mut probe = out.clone();
630        if zip::classify(src, name, &mut probe, true) {
631            *out = probe;
632            return;
633        }
634        set(out, FileKind::Script(script));
635        out.parse_error = probe.parse_error;
636        return;
637    }
638    detect_unstructured(src, name, out);
639}

641/// Content with no structure-defining magic at offset 0: Python bytecode,
642/// disc images (and scripts with a planted volume descriptor), then text.
643fn detect_unstructured<R: Read + Seek + ?Sized>(
644    src: &mut Source<'_, R>,
645    name: &NameInfo,
646    out: &mut Classification,
647) {
648    let head = src.head();
649    // Bytecode has a magic at offset 0, so it is checked before the volume
650    // descriptors deeper in.
651    if let Some(named) = magic::python_bytecode(head, name.extension.as_deref()) {
652        set(out, FileKind::Script(ScriptKind::Python));
653        out.extension_assisted = named;
654        return;
655    }
656    if let Some(kind) = magic::disc_image(head) {
657        let area = head.get(..magic::SYSTEM_AREA_LEN).unwrap_or_default();
658        if !text::system_area_is_script(area, name) {
659            set(out, kind);
660            return;
661        }
662        // A script with a planted descriptor. The system area alone is
663        // classified, so the binary descriptors after it do not hide the
664        // script; then the whole head is, and the more telling result is
665        // kept: an executable kind over a non-executable one, then the
666        // more confident.
667        // Both are ranked as they will be reported, with the name's hints
668        // applied. If neither finds text or a script, the image is an image.
669        let mut area_result = out.clone();
670        text::classify_within(src, name, &mut area_result, magic::SYSTEM_AREA_LEN);
671        apply_name_hints(name, &mut area_result);
672        let mut head_result = out.clone();
673        text::classify(src, name, &mut head_result);
674        apply_name_hints(name, &mut head_result);
675        let rank = |c: &Classification| (c.is_executable_content(), c.confidence);
676        let (mut best, other) = if rank(&head_result) > rank(&area_result) {
677            (head_result, area_result)
678        } else {
679            (area_result, head_result)
680        };
681        // A read that failed for either result (the head's tail read, say)
682        // is reported whichever is kept.
683        if best.parse_error.is_none() {
684            best.parse_error = other.parse_error;
685        }
686        if best.is_executable_content() || best.content_kind == FileKind::Text {
687            *out = best;
688        } else {
689            set(out, kind);
690        }
691        return;
692    }
693    text::classify(src, name, out);
694}

696/// Applies the rules where the name decides between kinds that content
697/// alone cannot tell apart.
698fn apply_name_hints(name: &NameInfo, out: &mut Classification) {
699    // A DOS `.com` program has no magic.
700    let weak = matches!(out.content_kind, FileKind::Unknown | FileKind::Text)
701        || (matches!(out.content_kind, FileKind::Script(_)) && out.confidence < Confidence::High);
702    if weak
703        && out.parse_error.is_none()
704        && name.is_ext(&["com"])
705        && (1..=MAX_COM_SIZE).contains(&out.file_size)
706    {
707        out.content_kind = FileKind::Com;
708        out.confidence = Confidence::Low;
709        out.heuristic = true;
710        out.extension_assisted = true;
711        return;
712    }
713    // Content in a script's extension is that script: the host runs it,
714    // whatever else is in the file. A keyword guess never overrides the
715    // extension's language; only a definite marker does. Unrecognised
716    // content (binary, or text with binary data) is taken to be the script
717    // too, so that it fails closed.
718    if let ExtensionKind::Kind(FileKind::Script(claimed)) = out.extension_kind {
719        let readable = !matches!(out.parse_error, Some(ParseError::Io(_)));
720        let overridable = match out.content_kind {
721            FileKind::Text => true,
722            FileKind::Unknown => readable && out.file_size > 0,
723            FileKind::Script(_) => out.confidence < Confidence::High,
724            _ => false,
725        };
726        if overridable && out.content_kind != FileKind::Script(claimed) {
727            out.content_kind = FileKind::Script(claimed);
728            out.confidence = Confidence::Low;
729            out.heuristic = true;
730            out.extension_assisted = true;
731        }
732    }
733    // An HTML application name on content that is not recognised (binary,
734    // or text with binary data) fails closed in the same way. Plain text
735    // without HTML stays text: the host shows it, but runs nothing.
736    if out.extension_kind == ExtensionKind::Kind(FileKind::Hta)
737        && out.content_kind == FileKind::Unknown
738        && !matches!(out.parse_error, Some(ParseError::Io(_)))
739        && out.file_size > 0
740    {
741        out.content_kind = FileKind::Hta;
742        out.confidence = Confidence::Low;
743        out.heuristic = true;
744        out.extension_assisted = true;
745    }
746}

748/// Grades the disagreement between content and name.
749fn mismatch_severity(out: &Classification, name: &NameInfo) -> MismatchSeverity {
750    let content = out.content_kind;
751    let executable = out.is_executable_content();
752    if executable && (name.bidi_control || name.double_extension()) {
753        return MismatchSeverity::Critical;
754    }
755    if content == FileKind::Unknown {
756        return MismatchSeverity::None;
757    }
758    match out.extension_kind {
759        ExtensionKind::None => MismatchSeverity::None,
760        // Executable content behind a document or media name.
761        ExtensionKind::Inert if executable => MismatchSeverity::Suspicious,
762        // Behind an unknown name, such as the `.tmp` files installers
763        // unpack and run.
764        ExtensionKind::Other if executable => MismatchSeverity::Benign,
765        ExtensionKind::Inert | ExtensionKind::Other => MismatchSeverity::None,
766        ExtensionKind::Kind(claimed) => {
767            let sub_kind = out.pe.as_ref().map(|p| p.sub_kind);
768            if name::claim_matches(name, claimed, content, sub_kind) {
769                MismatchSeverity::None
770            } else if executable && !claimed.is_executable_content() {
771                MismatchSeverity::Suspicious
772            } else {
773                MismatchSeverity::Benign
774            }
775        }
776    }
777}
