1//! The one place that calls a signature-verification library.
2//!
3//! ADR 0008 picks `ring` for signature verification and requires call sites
4//! to go through a thin `crypto` module, so swapping the provider later is a
5//! one-module change. Outside this module, only the hash cache's MAC
6//! module (`hash/mac.rs`) names `ring`.
7//!
8//! What `ring` 0.17 verifies, and so what Palisade verifies:
9//!
10//! | Scheme | Digests | Key sizes |
11//! |---|---|---|
12//! | RSA PKCS#1 v1.5 | SHA-1, SHA-256, SHA-384, SHA-512 | 2048 to 8192 bits (1024 to 8192 for SHA-1, SHA-256 and SHA-512) |
13//! | RSA-PSS (MGF1 with the same digest, salt length equal to the digest length) | SHA-256, SHA-384, SHA-512 | 2048 to 8192 bits |
14//! | ECDSA P-256 | SHA-256, SHA-384 | |
15//! | ECDSA P-384 | SHA-256, SHA-384 | |
16//!
17//! Anything else (ECDSA P-521, other digest and curve pairings, PSS with
18//! other parameters, Ed25519, Ed448, MD5 or MD2) is reported as
19//! [`Unsupported`] and is **never** treated as valid.
20

27/// A signer's public key, borrowed from its certificate.
28#[derive(Debug, Clone, Copy)]
29pub(crate) enum PublicKey<'a> {
30    /// An RSA key: the DER `RSAPublicKey` and the modulus size in bits.
31    Rsa { der: &'a [u8], bits: u32 },
32    /// An elliptic-curve key: the curve and the encoded point.
33    Ec { curve: Curve, point: &'a [u8] },
34}
35
36/// The signature scheme a signer used.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub(crate) enum Scheme {
39    /// RSA PKCS#1 v1.5 over the digest.
40    RsaPkcs1(DigestAlgorithm),
41    /// RSA-PSS with MGF1, both over this digest, and a salt as long as it.
42    /// Other parameter sets are rejected before verification.
43    RsaPss(DigestAlgorithm),
44    /// ECDSA over the digest, DER-encoded `(r, s)`.
45    Ecdsa(DigestAlgorithm),
46}
47
48/// The outcome of one verification.
49#[derive(Debug, Clone, PartialEq, Eq)]
50pub(crate) enum Verified {
51    /// The signature is valid for the message and key.
52    Valid,
53    /// The signature is not valid for the message and key.
54    Invalid,
55    /// The scheme, key or digest cannot be verified here.
56    Unsupported(Unsupported),
57}
58
59/// Verifies `signature` over `message` (which the scheme hashes itself).
60pub(crate) fn verify(
61    scheme: Scheme,
62    key: PublicKey<'_>,
63    message: &[u8],
64    signature: &[u8],
65) -> Verified {
66    let (alg, key_bytes): (&'static dyn VerificationAlgorithm, &[u8]) = match (scheme, key) {
67        (Scheme::RsaPkcs1(digest), PublicKey::Rsa { der, bits }) => match rsa_pkcs1(digest, bits) {
68            Ok(alg) => (alg, der),
69            Err(why) => return Verified::Unsupported(why),
70        },
71        (Scheme::RsaPss(digest), PublicKey::Rsa { der, bits }) => match rsa_pss(digest, bits) {
72            Ok(alg) => (alg, der),
73            Err(why) => return Verified::Unsupported(why),
74        },
75        (Scheme::Ecdsa(digest), PublicKey::Ec { curve, point }) => match ecdsa(curve, digest) {
76            Ok(alg) => (alg, point),
77            Err(why) => return Verified::Unsupported(why),
78        },
79        _ => return Verified::Unsupported(Unsupported::KeyAlgorithmMismatch),
80    };
81    match UnparsedPublicKey::new(alg, key_bytes).verify(message, signature) {
82        Ok(()) => Verified::Valid,
83        Err(_) => Verified::Invalid,
84    }
85}

87fn weak_or_unknown(digest: DigestAlgorithm) -> Unsupported {
88    if digest.is_broken() {
89        Unsupported::WeakAlgorithm(digest)
90    } else {
91        Unsupported::Digest(digest)
92    }
93}
94
95fn rsa_pkcs1(
96    digest: DigestAlgorithm,
97    bits: u32,
98) -> Result<&'static dyn VerificationAlgorithm, Unsupported> {
99    if !(1024..=8192).contains(&bits) {
100        return Err(Unsupported::RsaKeySize(bits));
101    }
102    let large = bits >= 2048;
103    Ok(match (digest, large) {
104        (DigestAlgorithm::Sha1, true) => &ring_sig::RSA_PKCS1_2048_8192_SHA1_FOR_LEGACY_USE_ONLY,
105        (DigestAlgorithm::Sha1, false) => &ring_sig::RSA_PKCS1_1024_8192_SHA1_FOR_LEGACY_USE_ONLY,
106        (DigestAlgorithm::Sha256, true) => &ring_sig::RSA_PKCS1_2048_8192_SHA256,
107        (DigestAlgorithm::Sha256, false) => {
108            &ring_sig::RSA_PKCS1_1024_8192_SHA256_FOR_LEGACY_USE_ONLY
109        }
110        (DigestAlgorithm::Sha384, true) => &ring_sig::RSA_PKCS1_2048_8192_SHA384,
111        (DigestAlgorithm::Sha512, true) => &ring_sig::RSA_PKCS1_2048_8192_SHA512,
112        (DigestAlgorithm::Sha512, false) => {
113            &ring_sig::RSA_PKCS1_1024_8192_SHA512_FOR_LEGACY_USE_ONLY
114        }
115        (DigestAlgorithm::Sha384, false) => return Err(Unsupported::RsaKeySize(bits)),
116        (other, _) => return Err(weak_or_unknown(other)),
117    })
118}
119
120fn rsa_pss(
121    digest: DigestAlgorithm,
122    bits: u32,
123) -> Result<&'static dyn VerificationAlgorithm, Unsupported> {
124    if !(2048..=8192).contains(&bits) {
125        return Err(Unsupported::RsaKeySize(bits));
126    }
127    Ok(match digest {
128        DigestAlgorithm::Sha256 => &ring_sig::RSA_PSS_2048_8192_SHA256,
129        DigestAlgorithm::Sha384 => &ring_sig::RSA_PSS_2048_8192_SHA384,
130        DigestAlgorithm::Sha512 => &ring_sig::RSA_PSS_2048_8192_SHA512,
131        other => return Err(weak_or_unknown(other)),
132    })
133}
134
135fn ecdsa(
136    curve: Curve,
137    digest: DigestAlgorithm,
138) -> Result<&'static dyn VerificationAlgorithm, Unsupported> {
139    Ok(match (curve, digest) {
140        (Curve::P256, DigestAlgorithm::Sha256) => &ring_sig::ECDSA_P256_SHA256_ASN1,
141        (Curve::P256, DigestAlgorithm::Sha384) => &ring_sig::ECDSA_P256_SHA384_ASN1,
142        (Curve::P384, DigestAlgorithm::Sha256) => &ring_sig::ECDSA_P384_SHA256_ASN1,
143        (Curve::P384, DigestAlgorithm::Sha384) => &ring_sig::ECDSA_P384_SHA384_ASN1,
144        (Curve::P256 | Curve::P384, other) if other.is_broken() => {
145            return Err(Unsupported::WeakAlgorithm(other));
146        }
147        (curve @ (Curve::P256 | Curve::P384), other) => {
148            return Err(Unsupported::CurveDigest {
149                curve,
150                digest: other,
151            });
152        }
153        (curve, _) => return Err(Unsupported::Curve(curve)),
154    })
155}

158mod tests {
159    use super::*;

161    #[test]
162    fn unsupported_combinations_are_never_valid() {
163        let rsa = PublicKey::Rsa {
164            der: &[],
165            bits: 2048,
166        };
167        let ec = |curve| PublicKey::Ec { curve, point: &[] };
168        let cases = [
169            (
170                Scheme::RsaPkcs1(DigestAlgorithm::Md5),
171                rsa,
172                Unsupported::WeakAlgorithm(DigestAlgorithm::Md5),
173            ),
174            (
175                Scheme::RsaPkcs1(DigestAlgorithm::Sha256),
176                PublicKey::Rsa {
177                    der: &[],
178                    bits: 512,
179                },
180                Unsupported::RsaKeySize(512),
181            ),
182            (
183                Scheme::RsaPkcs1(DigestAlgorithm::Sha256),
184                PublicKey::Rsa {
185                    der: &[],
186                    bits: 16384,
187                },
188                Unsupported::RsaKeySize(16384),
189            ),
190            (
191                Scheme::RsaPkcs1(DigestAlgorithm::Sha384),
192                PublicKey::Rsa {
193                    der: &[],
194                    bits: 1024,
195                },
196                Unsupported::RsaKeySize(1024),
197            ),
198            (
199                Scheme::RsaPss(DigestAlgorithm::Sha1),
200                rsa,
201                Unsupported::Digest(DigestAlgorithm::Sha1),
202            ),
203            (
204                Scheme::RsaPss(DigestAlgorithm::Sha256),
205                PublicKey::Rsa {
206                    der: &[],
207                    bits: 1024,
208                },
209                Unsupported::RsaKeySize(1024),
210            ),
211            (
212                Scheme::Ecdsa(DigestAlgorithm::Sha256),
213                ec(Curve::P521),
214                Unsupported::Curve(Curve::P521),
215            ),
216            (
217                Scheme::Ecdsa(DigestAlgorithm::Sha512),
218                ec(Curve::P256),
219                Unsupported::CurveDigest {
220                    curve: Curve::P256,
221                    digest: DigestAlgorithm::Sha512,
222                },
223            ),
224            (
225                Scheme::Ecdsa(DigestAlgorithm::Sha1),
226                ec(Curve::P384),
227                Unsupported::CurveDigest {
228                    curve: Curve::P384,
229                    digest: DigestAlgorithm::Sha1,
230                },
231            ),
232            (
233                Scheme::Ecdsa(DigestAlgorithm::Sha256),
234                rsa,
235                Unsupported::KeyAlgorithmMismatch,
236            ),
237            (
238                Scheme::RsaPkcs1(DigestAlgorithm::Sha256),
239                ec(Curve::P256),
240                Unsupported::KeyAlgorithmMismatch,
241            ),
242        ];
243        for (scheme, key, why) in cases {
244            assert_eq!(
245                verify(scheme, key, b"m", b"s"),
246                Verified::Unsupported(why),
247                "{scheme:?}"
248            );
249        }
250    }
251
252    #[test]
253    fn garbage_keys_and_signatures_are_invalid() {
254        let rsa = PublicKey::Rsa {
255            der: &[0x30, 0x00],
256            bits: 2048,
257        };
258        assert_eq!(
259            verify(
260                Scheme::RsaPkcs1(DigestAlgorithm::Sha256),
261                rsa,
262                b"m",
263                &[0; 256]
264            ),
265            Verified::Invalid
266        );
267        let ec = PublicKey::Ec {
268            curve: Curve::P256,
269            point: &[0x04; 65],
270        };
271        assert_eq!(
272            verify(
273                Scheme::Ecdsa(DigestAlgorithm::Sha256),
274                ec,
275                b"m",
276                &[0x30, 0x00]
277            ),
278            Verified::Invalid
279        );
280    }
