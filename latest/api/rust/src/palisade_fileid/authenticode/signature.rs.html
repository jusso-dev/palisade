14/// A digest algorithm named by a signature.
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize)]
16#[serde(rename_all = "snake_case")]
17pub enum DigestAlgorithm {
18    /// MD2. Broken; never trusted, and not even computed.
19    Md2,
20    /// MD5. Broken; never trusted.
21    Md5,
22    /// SHA-1. Not collision resistant; accepted for compatibility, and
23    /// always loses primary selection to SHA-2.
24    Sha1,
25    /// SHA-256.
26    Sha256,
27    /// SHA-384.
28    Sha384,
29    /// SHA-512.
30    Sha512,
31}
32
33impl DigestAlgorithm {
34    /// Whether the algorithm is broken (MD2, MD5), so a signature using it
35    /// is reported as [`SignatureStatus::WeakAlgorithm`] and never trusted.
36    #[must_use]
37    pub const fn is_broken(self) -> bool {
38        matches!(self, Self::Md2 | Self::Md5)
39    }
40
41    /// Whether the algorithm is SHA-256 or stronger.
42    #[must_use]
43    pub const fn is_sha2(self) -> bool {
44        matches!(self, Self::Sha256 | Self::Sha384 | Self::Sha512)
45    }
46
47    /// The matching file-hash algorithm, if Palisade computes it (all but
48    /// MD2).
49    #[must_use]
50    pub const fn hash_algorithm(self) -> Option<HashAlgorithm> {
51        match self {
52            Self::Md2 => None,
53            Self::Md5 => Some(HashAlgorithm::Md5),
54            Self::Sha1 => Some(HashAlgorithm::Sha1),
55            Self::Sha256 => Some(HashAlgorithm::Sha256),
56            Self::Sha384 => Some(HashAlgorithm::Sha384),
57            Self::Sha512 => Some(HashAlgorithm::Sha512),
58        }
59    }
60
61    /// The digest length in bytes.
62    #[must_use]
63    pub const fn digest_len(self) -> usize {
64        match self {
65            Self::Md2 | Self::Md5 => 16,
66            Self::Sha1 => 20,
67            Self::Sha256 => 32,
68            Self::Sha384 => 48,
69            Self::Sha512 => 64,
70        }
71    }
72}

207/// The verdict on one signature, from the checks in this module. The
208/// certificate chain is judged by [`super::evaluate_trust`] (#49), and
209/// timestamps, expiry and revocation by [`super::evaluate_validity`] (#50).
210#[derive(Debug, Clone, PartialEq, Eq)]
211#[non_exhaustive]
212pub enum SignatureStatus {
213    /// The signature is cryptographically valid and its digest matches the
214    /// image. For a [`SignatureKind::Catalog`] signature, which signs a
215    /// trust list rather than an image, it means only that the signature
216    /// over the catalog's CTL is cryptographically valid (`digest_match`
217    /// is `None`): a member file is catalog-signed only when a valid
218    /// catalog also lists its digest (see [`crate::catalog`]).
219    Valid,
220    /// The signature is cryptographically valid, but the image digest it
221    /// signs does not match the file: the file was changed after signing.
222    DigestMismatch,
223    /// The signature is cryptographically valid and its digest matches the
224    /// best-effort digest of an image whose layout departs from the
225    /// Authenticode specification (#47): data after the certificate table,
226    /// a misaligned table, overlapping sections and so on. Such a match is
227    /// not authoritative (bytes can be moved across the certificate table
228    /// without changing it), so it is **never** treated as valid; the flat
229    /// hash is the file's identity.
230    NonConformantImage,
231    /// The signature's cryptography failed.
232    BadSignature(BadSignature),
233    /// The signature uses a broken digest (MD2, MD5). Parsed and reported,
234    /// never trusted.
235    WeakAlgorithm(DigestAlgorithm),
236    /// The signature could not be verified.
237    Unsupported(Unsupported),
238    /// The signature could not be parsed.
239    Malformed(SignatureError),
240}
241
242impl SignatureStatus {
243    /// Whether the status is [`SignatureStatus::Valid`].
244    #[must_use]
245    pub const fn is_valid(&self) -> bool {
246        matches!(self, Self::Valid)
247    }
248}

322/// The signed metadata authenticated attributes
323/// (`szOID_SIGNED_ATTRIBUTE_*`, 1.3.6.1.4.1.311.2.7.1 to .10), which give
324/// files without a version resource version-like identity. Signers encode
325/// each as a UTF8String. A field is `Some` only when its attribute is
326/// present and decodes exactly (see
327/// [`SignatureWarning::SignedMetadataUnreadable`]); an attribute that
328/// appears twice makes the signature [`SignatureStatus::Malformed`].
329///
330/// These are the signer's claims, covered by the signature: they can be
331/// trusted only as far as the signer is.
332#[derive(Debug, Clone, Default, PartialEq, Eq, Hash, Serialize)]
333pub struct SignedMetadata {
334    /// .1 `InternalName`.
335    pub internal_name: Option<String>,
336    /// .2 `FileVersion`.
337    pub file_version: Option<String>,
338    /// .3 `FileDescription`.
339    pub file_description: Option<String>,
340    /// .4 `Product`.
341    pub product: Option<String>,
342    /// .5 `ProductVersion`.
343    pub product_version: Option<String>,
344    /// .6 `OriginalFilename`.
345    pub original_filename: Option<String>,
346    /// .7 `Language`.
347    pub language: Option<String>,
348    /// .8 `Author`.
349    pub author: Option<String>,
350    /// .9 `PublishTime`, as the signer wrote it. Not parsed: it is the
351    /// signer's claim, and a trusted time comes from a timestamp (#50).
352    pub publish_time: Option<String>,
353    /// .10 `SourceURL`.
354    pub source_url: Option<String>,
355}

357impl SignedMetadata {
358    /// The field for the attribute whose final OID arc is `arc` (1 to 10).
359    pub(crate) fn field_mut(&mut self, arc: u8) -> Option<&mut Option<String>> {
360        Some(match arc {
361            1 => &mut self.internal_name,
362            2 => &mut self.file_version,
363            3 => &mut self.file_description,
364            4 => &mut self.product,
365            5 => &mut self.product_version,
366            6 => &mut self.original_filename,
367            7 => &mut self.language,
368            8 => &mut self.author,
369            9 => &mut self.publish_time,
370            10 => &mut self.source_url,
371            _ => return None,
372        })
373    }
374
375    /// Whether no field is set.
376    #[must_use]
377    pub fn is_empty(&self) -> bool {
378        *self == Self::default()
379    }

382/// A page hash table: the one an `SpcSerializedObject` moniker in a
383/// signature carries, or one built from a file ([`super::page`]). Check a
384/// signed one with [`super::verify_signature_page_hashes`].
385#[derive(Debug, Clone, PartialEq, Eq)]
386pub struct PageHashes {
387    /// SHA-1 (version 1, 1.3.6.1.4.1.311.2.3.1) or SHA-256 (version 2,
388    /// 1.3.6.1.4.1.311.2.3.2).
389    pub algorithm: HashAlgorithm,
390    /// The table: `(u32 file offset, digest)` records, little-endian.
391    pub table: Vec<u8>,
392}
393
394impl PageHashes {
395    /// The `(file offset, digest)` records of the table.
396    pub fn entries(&self) -> impl Iterator<Item = (u32, &[u8])> {
397        self.table
398            .chunks_exact(4 + self.algorithm.digest_len())
399            .filter_map(|record| {
400                let (offset, digest) = record.split_first_chunk::<4>()?;
401                Some((u32::from_le_bytes(*offset), digest))
402            })
403    }
404}

492    /// An unauthenticated attribute of an unknown type was ignored.

558    /// The cryptographic check.
559    pub signature_valid: SignatureCheck,
560    /// The overall verdict.
561    pub status: SignatureStatus,

566    /// [`Self::untrusted_signer_attrs`] is `true`.
567    pub signer: Option<SignerAttributes>,
568    /// `true` when [`Self::signer`] is reported only because of

572    /// How the SignerInfo names its certificate. Attacker-controlled; it is
573    /// not covered by the signature.
574    pub signer_id: Option<SignerIdentifier>,

578    /// The `SpcPeImageData` flags octet (bit 7 is `includeResources`).

580    /// The page hash table, if the signature carries one (#47).

582    /// Timestamps in the unauthenticated attributes, verified by

584    pub timestamp_tokens: Vec<TimestampToken>,
585    /// The SignerInfo's signature value (`encryptedDigest`): what a

587    pub signature_value: Vec<u8>,
588    /// Why the signer cannot back an OS-native code integrity signer rule;
589    /// empty when it can.
590    pub os_ci_issues: Vec<OsCiIssue>,
591    /// Anything unusual that does not change the status.
592    pub warnings: Vec<SignatureWarning>,
593}

604/// Why a signature could not be parsed.
605#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize)]
606#[serde(rename_all = "snake_case")]
607#[non_exhaustive]
608pub enum SignatureError {
609    /// The DER encoding is invalid.
610    Asn1(Asn1Error),
611    /// The outer ContentInfo is not `signedData`.
612    NotSignedData(ObjectId),
613    /// The signed content type is not `SPC_INDIRECT_DATA_OBJID`.
614    NotIndirectData(ObjectId),
615    /// The signed content type of a catalog is not `szOID_CTL`
616    /// (1.3.6.1.4.1.311.10.1).
617    NotCatalog(ObjectId),
618    /// The indirect data is not `SPC_PE_IMAGE_DATA_OBJID`, so it does not
619    /// sign a PE image.
620    NotPeImageData(ObjectId),
621    /// The indirect data is not `SPC_SIPINFO_OBJID`
622    /// (1.3.6.1.4.1.311.2.1.30), so it does not sign a non-PE subject such
623    /// as an installer package or a script (#52).
624    NotSipInfo(ObjectId),
625    /// The `SPC_SIPINFO` names a subject type other than the one the file
626    /// is (for example a package signature on an installer).
627    SubjectTypeMismatch,
628    /// The `SpcSipInfo` value is malformed: its GUID is not 16 bytes.
629    SipInfo,
630    /// The `SpcSipInfo` names a subject interface package other than the
631    /// one for the script's format (a VBScript signature in a JScript
632    /// file, say), so the signed digest is not of this kind of file.
633    SipMismatch,
634    /// The SignedData has no signed content.
635    MissingContent,
636    /// Authenticode requires exactly one SignerInfo.
637    SignerCount(usize),
638    /// The SignerInfo has no authenticated attributes.
639    MissingAuthenticatedAttributes,
640    /// A required authenticated attribute (`contentType` or
641    /// `messageDigest`) is missing.
642    MissingAttribute(ObjectId),
643    /// An authenticated attribute appears more than once.
644    DuplicateAttribute(ObjectId),
645    /// An authenticated attribute that must have exactly one value has
646    /// another number of values.
647    AttributeValueCount(ObjectId),
648    /// The `contentType` attribute does not name the signed content type
649    /// (`SPC_INDIRECT_DATA_OBJID`, or `szOID_CTL` for a catalog).
650    ContentTypeMismatch(ObjectId),
651    /// The signature algorithm implies a digest other than the SignerInfo's
652    /// digest algorithm.
653    InconsistentDigestAlgorithms,
654    /// The signed image digest is not as long as its algorithm's output.
655    DigestLength,
656    /// An algorithm identifier has parameters where none are allowed, or
657    /// invalid ones.
658    AlgorithmParameters,
659}
660
661impl fmt::Display for SignatureError {
662    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
663        match self {
664            Self::Asn1(e) => write!(f, "invalid DER: {e}"),
665            Self::NotSignedData(o) => write!(f, "content type {o} is not signedData"),
666            Self::NotIndirectData(o) => {
667                write!(f, "signed content type {o} is not SPC_INDIRECT_DATA")
668            }
669            Self::NotCatalog(o) => write!(f, "signed content type {o} is not szOID_CTL"),
670            Self::NotPeImageData(o) => write!(f, "indirect data type {o} is not SPC_PE_IMAGE_DATA"),
671            Self::NotSipInfo(o) => write!(f, "indirect data type {o} is not SPC_SIPINFO"),
672            Self::SubjectTypeMismatch => {
673                f.write_str("the SIP information names another subject type")
674            }
675            Self::SipInfo => f.write_str("the SpcSipInfo GUID is not 16 bytes"),
676            Self::SipMismatch => {
677                f.write_str("the SpcSipInfo names another format's subject interface package")
678            }
679            Self::MissingContent => f.write_str("the SignedData has no content"),
680            Self::SignerCount(n) => write!(f, "{n} SignerInfos; Authenticode requires exactly one"),
681            Self::MissingAuthenticatedAttributes => f.write_str("no authenticated attributes"),
682            Self::MissingAttribute(o) => write!(f, "missing authenticated attribute {o}"),
683            Self::DuplicateAttribute(o) => write!(f, "authenticated attribute {o} appears twice"),
684            Self::AttributeValueCount(o) => write!(f, "attribute {o} must have exactly one value"),
685            Self::ContentTypeMismatch(o) => {
686                write!(
687                    f,
688                    "contentType attribute {o} does not name the signed content"
689                )
690            }
691            Self::InconsistentDigestAlgorithms => {
692                f.write_str("the signature algorithm's digest differs from the signer's")
693            }
694            Self::DigestLength => f.write_str("the signed image digest has the wrong length"),
695            Self::AlgorithmParameters => f.write_str("invalid algorithm parameters"),
696        }
697    }
698}

704        Self::Asn1(e)

708/// Why the certificate table as a whole could not be read.
709#[derive(Debug, Clone, PartialEq, Eq)]
710#[non_exhaustive]
711pub enum CertificateTableError {
712    /// The PE parser did not accept the security directory as a
713    /// certificate table ([`crate::pe::PeInfo::certificate_table_issue`]).
714    Rejected(Option<WarningKind>),
715    /// The table is larger than [`super::MAX_CERTIFICATE_TABLE_SIZE`].
716    TooLarge(u64),
717    /// The first `WIN_CERTIFICATE` entry's `dwLength` is below 8 or runs
718    /// past the table, so no entry could be read.
719    Malformed {
720        /// File offset of the entry.
721        offset: u64,
722    },
723}

725/// Why a signed file is not validly signed.
726#[derive(Debug, Clone, PartialEq, Eq)]
727#[non_exhaustive]
728pub enum InvalidReason {
729    /// The certificate table could not be read.
730    CertificateTable(CertificateTableError),
731    /// No signature is valid. This is the status of the first one.
732    Signature(SignatureStatus),
733    /// A script carries a signature block (#52) that could not be read:
734    /// the block is malformed, or the text cannot be decoded without
735    /// ambiguity. Why is [`crate::script::ScriptReport::signing_state`].
736    ScriptSignatureBlock,
737}

739/// The signing state of a file.
740///
741/// Catalog signing (`CatalogValid`, from [`crate::catalog`] lookups) is
742/// still to come (#51).
743#[derive(Debug, Clone, PartialEq, Eq)]
744#[non_exhaustive]
745pub enum SigningState {
746    /// No embedded signature.
747    Unsigned,
748    /// At least one signature is valid (see [`SignatureReport::primary`]).
749    Valid,
750    /// The file carries a signature, but none is valid.
751    Invalid(InvalidReason),
752}

754/// One `WIN_CERTIFICATE` entry of the certificate table.
755#[derive(Debug, Clone, Copy, PartialEq, Eq)]
756pub struct CertificateEntry {
757    /// File offset of the entry.
758    pub offset: u64,
759    /// `dwLength`, including the 8-byte header.
760    pub length: u32,
761    /// `wRevision`.
762    pub revision: u16,
763    /// `wCertificateType`.
764    pub certificate_type: u16,
765    /// The index of its embedded signature in
766    /// [`SignatureReport::signatures`], if it is a PKCS#7 entry that was
767    /// read.
768    pub signature: Option<usize>,
769}

771/// Something unusual about the certificate table or the image.
772#[derive(Debug, Clone, PartialEq, Eq)]
773#[non_exhaustive]
774pub enum ReportWarning {
775    /// An entry whose revision is not 0x0100 or 0x0200, or whose type is
776    /// not `WIN_CERT_TYPE_PKCS_SIGNED_DATA` (2), was recorded and skipped.
777    UnknownEntry {
778        /// The entry's index in [`SignatureReport::entries`].
779        entry: usize,
780    },
781    /// An entry's `dwLength` is below 8 or runs past the table. Reading
782    /// stopped there.
783    BadEntryLength {
784        /// File offset of the entry.
785        offset: u64,
786        /// Its `dwLength`.
787        length: u32,
788    },
789    /// Non-zero bytes in the padding between or after entries. They are
790    /// not covered by any hash or signature.
791    UnexpectedTableData {
792        /// File offset of the first such byte.
793        offset: u64,
794    },
795    /// More than [`super::MAX_CERTIFICATE_ENTRIES`] entries; the rest were
796    /// not read.
797    TooManyEntries,
798    /// [`super::MAX_SIGNATURES`] signatures were read; any others were not.
799    TooManySignatures,
800    /// The image layout is not one the Authenticode specification
801    /// describes, so its digest is a best effort (#47) and its flat hash is
802    /// the identity to use.
803    NonConformantImage,
804    /// `tolerate_digest_mismatch` exposed the signer attributes of this
805    /// signature, marked untrusted. Recorded for audit.
806    DigestMismatchTolerated {
807        /// The signature's index.
808        signature: usize,
809    },
810}

812/// Options for signature verification.
813#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
814pub struct VerifyOptions {
815    /// When `true`, the signer attributes of a signature whose only problem
816    /// is [`SignatureStatus::DigestMismatch`] are still reported, marked
817    /// [`SignatureInfo::untrusted_signer_attrs`], so rules can fall back to
818    /// hash identity. The file stays [`SigningState::Invalid`]. Default
819    /// `false`. Every use is recorded as
820    /// [`ReportWarning::DigestMismatchTolerated`], and the setting itself is
821    /// echoed in [`SignatureReport::tolerate_digest_mismatch`].
822    pub tolerate_digest_mismatch: bool,
823}

825/// Every embedded and nested signature of a PE image, verified.
826#[derive(Debug, Clone, PartialEq, Eq)]
827pub struct SignatureReport {
828    /// The `WIN_CERTIFICATE` entries read, in table order.
829    pub entries: Vec<CertificateEntry>,
830    /// Every signature, in [`SignatureInfo::index`] order.
831    pub signatures: Vec<SignatureInfo>,
832    /// The primary signature: the first valid one with the strongest digest
833    /// (SHA-256 or better beats SHA-1). `None` if none is valid.
834    pub primary: Option<usize>,
835    /// The file's signing state.
836    pub signing_state: SigningState,
837    /// The [`VerifyOptions::tolerate_digest_mismatch`] setting used, for
838    /// audit.
839    pub tolerate_digest_mismatch: bool,
840    /// The image's Authenticode layout status (#47).
841    pub image_status: AuthenticodeStatus,
842    /// Anything unusual about the table or the image.
843    pub warnings: Vec<ReportWarning>,
844}
845
846impl SignatureReport {
847    /// The primary signature, if any.
848    #[must_use]
849    pub fn primary_signature(&self) -> Option<&SignatureInfo> {
850        self.primary.and_then(|i| self.signatures.get(i))
851    }
852}
