1//! Signer trust and identity (#49): chains for every signature, the
2//! trust-aware primary signature, and the identity strings publisher rules
3//! match on.
4//!
5//! A [`SignatureStatus::Valid`] signature is sound and signs the file, but
6//! anyone can sign a file, and anyone can append a nested signature to a
7//! signed file. So [`evaluate_trust`] builds each valid signature's chain
8//! against a [`TrustConfig`], and chooses the primary signature only among
9//! the **trusted** ones. An untrusted signer, nested or not, never becomes
10//! primary, and only a trusted primary yields a [`SignerIdentity`].
11//!
12//! "Trusted" here means: the signature is valid, its signer certificate
13//! chains to a configured anchor, every certificate in that chain permits
14//! code signing, and the signer's key usage (if restricted) allows digital
15//! signatures. Validity periods, timestamps and revocation are judged
16//! afterwards by [`super::evaluate_validity`] (#50), and the OS trust store
17//! is not consulted (the `os-trust` cross-check in #48).
18//!
19//! Everything here is plain data. Policy (#31) matches on it without
20//! depending on this crate's internals.
21

36/// Why a signature is not trusted.
37#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize)]
38#[serde(rename_all = "snake_case")]
39#[non_exhaustive]
40pub enum UntrustedReason {
41    /// The signature is not [`SignatureStatus::Valid`].
42    NotValid,
43    /// The signer certificate could not be parsed, so no chain was built.
44    SignerCertificate(CertificateError),
45    /// The chain reaches no configured trust anchor.
46    NoTrustAnchor,
47    /// A certificate in the chain restricts its extended key usage to
48    /// purposes other than code signing.
49    CodeSigningNotPermitted,
50    /// The signer certificate's key usage does not include
51    /// `digitalSignature`.
52    SignerKeyUsage,
53}

55/// One signature's chain and trust verdict.
56#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
57pub struct SignerTrust {
58    /// The signature: an index into [`SignatureReport::signatures`].
59    pub signature: usize,
60    /// The chains of its signer certificate. `None` when the signature has
61    /// no usable signer (not valid, or its certificate did not parse).
62    pub chain: Option<ChainReport>,
63    /// Whether the signer is trusted: [`Self::untrusted`] is empty.
64    pub trusted: bool,
65    /// Why it is not trusted.
66    pub untrusted: Vec<UntrustedReason>,
67    /// The signature's image digest algorithm.
68    pub digest_alg: Option<DigestAlgorithm>,
69    /// The SignerInfo's digest algorithm. Primary selection ranks a
70    /// signature by the weaker of the two.
71    pub signer_digest_alg: Option<DigestAlgorithm>,
72    /// Whether an OS-native code integrity signer rule can use this signer:
73    /// the signature itself is compatible
74    /// ([`SignatureInfo::os_ci_compatible`]), and every certificate in the
75    /// full chain has an RSA key of at most 4096 bits. Conservative: a
76    /// `false` only means a rule must fall back to a hash level.
77    pub os_ci_compatible: bool,
78}

158/// Something that limits which publisher rules can be made for a signer.
159#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize)]
160#[serde(rename_all = "snake_case")]
161#[non_exhaustive]
162pub enum IdentityIssue {
163    /// The leaf subject's `CN` (or, with no `CN`, its `O`) is not an exact
164    /// decoding, is empty or only whitespace, or holds a control
165    /// character, so it is not used as a name.
166    UnusableName,
167    /// A leaf subject attribute is not an exact, control-free decoding, so
168    /// no `publisher_dn` is given.
169    UnusableDn,
170    /// The leaf subject has more than one `CN`; the first is used.
171    MultipleCommonNames,
172    /// The leaf subject has neither `CN` nor `O`.
173    NoName,
174    /// The leaf subject has no usable `CN` (the publisher name came from
175    /// `O`), so the OS Publisher and FilePublisher levels, which need the
176    /// leaf `CN`, cannot be made.
177    NoCommonName,
178    /// The file carries no intermediate, so there is no PCA, and the root
179    /// is not a well-known root: Publisher-level OS rules cannot be made.
180    NoPca,
181    /// The PCA's signature digest has no TBS hash form (not SHA-1 or
182    /// SHA-2).
183    NoPcaTbsHash,
184    /// The version resource has no `ProductName`.
185    NoProductName,
186    /// The version resource has no usable file name or version.
187    NoFileNameOrVersion,
188    /// The signer is not OS code integrity compatible
189    /// ([`SignerTrust::os_ci_compatible`] is `false`), so the OS Publisher
190    /// and FilePublisher levels are not given: the OS would not honour
191    /// them, and a rule must fall back to a hash level.
192    NotOsCiCompatible,
193}

229/// Why [`SignerIdentity::attach_package_publisher`] refused a package's
230/// publisher.
231#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize)]
232#[serde(rename_all = "snake_case")]
233#[non_exhaustive]
234pub enum PackagePublisherError {
235    /// The signer has no usable DN ([`IdentityIssue::UnusableDn`]), so
236    /// nothing can be compared.
237    NoSignerDn,
238    /// The manifest's `Identity@Publisher` is not the signer's subject DN.
239    /// A package whose manifest names another publisher than its signer is
240    /// not that publisher's package.
241    PublisherMismatch,
242    /// The display name is empty, only whitespace, or holds a control
243    /// character.
244    UnusableDisplayName,
245}
246
247impl fmt::Display for PackagePublisherError {
248    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
249        f.write_str(match self {
250            Self::NoSignerDn => "the signer has no usable DN to compare",
251            Self::PublisherMismatch => "the manifest's publisher is not the signer",
252            Self::UnusableDisplayName => "the display name is empty or holds control characters",
253        })
254    }
255}

259impl SignerIdentity {
260    /// Carries a packaged app's publisher display name (#53) onto this
261    /// identity. `manifest_publisher` is the manifest's `Identity@Publisher`
262    /// DN string and `display_name` its `Properties/PublisherDisplayName`.
263    ///
264    /// The OS installs a package only when `Identity@Publisher` equals its
265    /// signer's subject, so the name is attached only when
266    /// `manifest_publisher` equals [`Self::publisher_dn`] (both in the
267    /// canonical `CN=…, O=…` form; surrounding whitespace is ignored).
268    /// Otherwise nothing changes and the reason is returned.
269    pub fn attach_package_publisher(
270        &mut self,
271        manifest_publisher: &str,
272        display_name: &str,
273    ) -> Result<(), PackagePublisherError> {
274        let Some(dn) = self.publisher_dn.as_deref() else {
275            return Err(PackagePublisherError::NoSignerDn);
276        };
277        if manifest_publisher.trim() != dn {
278            return Err(PackagePublisherError::PublisherMismatch);
279        }
280        let name = display_name.trim();
281        if name.is_empty() || name.chars().any(char::is_control) {
282            return Err(PackagePublisherError::UnusableDisplayName);
283        }
284        self.package_display_name = Some(name.to_owned());
285        Ok(())
286    }
287}

301/// The trust verdict on a file's signatures.
302#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
303pub struct TrustReport {
304    /// One entry per signature, in [`SignatureReport::signatures`] order.
305    pub signers: Vec<SignerTrust>,
306    /// The primary signature, chosen after trust: the first trusted
307    /// signature with the strongest digest (SHA-2 beats SHA-1). `None` if
308    /// no signature is trusted. A position in [`SignatureReport::signatures`]
309    /// (and in [`Self::signers`]).
310    pub primary: Option<usize>,
311    /// The identity of the primary signer.
312    pub identity: Option<SignerIdentity>,
313    /// When no signature is trusted: the `CompanyName` the file claims,
314    /// marked unverified.
315    pub vendor_hint: Option<VendorHint>,
316}

320    #[must_use]

322        self.primary.and_then(|i| self.signers.get(i))
323    }

326/// Builds the chain of every valid signature in `report`, decides which are
327/// trusted under `config`, and chooses the primary signature among the
328/// trusted ones. `version` is the file's version resource
329/// ([`crate::pe::PeInfo::version_info`]), for the product and file names.
330#[must_use]
331pub fn evaluate_trust(
332    report: &SignatureReport,
333    config: &TrustConfig,
334    version: Option<&VersionInfo>,
335) -> TrustReport {
336    let mut budget = Budget::new(MAX_FILE_SIGNATURE_CHECKS);
337    let signers: Vec<SignerTrust> = report
338        .signatures
339        .iter()
340        .map(|sig| signer_trust(sig, config, &mut budget))
341        .collect();
342    let primary = select_primary(&signers);
343    let identity = primary.and_then(|p| {
344        let signer = signers.get(p)?;
345        Some(identity(
346            signer.signature,
347            signer.chain.as_ref()?,
348            signer.os_ci_compatible,
349            version,
350        ))
351    });
352    let vendor_hint = match primary {
353        Some(_) => None,
354        None => version
355            .and_then(|v| v.get(VersionField::CompanyName))
356            .filter(|name| !name.trim().is_empty())
357            .map(|name| VendorHint {
358                company_name: name.to_owned(),
359                unverified: true,
360            }),
361    };
362    TrustReport {
363        signers,
364        primary,
365        identity,
366        vendor_hint,
367    }
368}

370fn signer_trust(sig: &SignatureInfo, config: &TrustConfig, budget: &mut Budget) -> SignerTrust {
371    let mut untrusted = Vec::new();
372    // Only a valid signature's signer is looked at. `signer` is also set
373    // for a tolerated digest mismatch, which is never trusted.
374    let signer = match (&sig.status, &sig.signer) {
375        (SignatureStatus::Valid, Some(attrs)) if !sig.untrusted_signer_attrs => {
376            Some(attrs.signer_cert)
377        }
378        _ => None,
379    };
380    let chain = match signer {
381        Some(index) => match build_chain_within(&sig.certs_in_blob, index, config, budget) {
382            Ok(chain) => Some(chain),
383            Err(e) => {
384                untrusted.push(UntrustedReason::SignerCertificate(e));
385                None
386            }
387        },
388        None => {
389            untrusted.push(UntrustedReason::NotValid);
390            None
391        }
392    };
393    if let Some(chain) = &chain {
394        if !chain.full_chain.anchored {
395            untrusted.push(UntrustedReason::NoTrustAnchor);
396        }
397        if !chain.permits_code_signing() {
398            untrusted.push(UntrustedReason::CodeSigningNotPermitted);
399        }
400        let leaf_ku = chain.leaf_cert().and_then(|c| c.key_usage);
401        if leaf_ku.is_some_and(|ku| !ku.contains(KeyUsage::DIGITAL_SIGNATURE)) {
402            untrusted.push(UntrustedReason::SignerKeyUsage);
403        }
404    }
405    let os_ci_compatible = sig.os_ci_compatible()
406        && chain.as_ref().is_some_and(|c| {
407            c.full_chain_certs().all(|cert| {
408                matches!(
409                    (&cert.public_key_alg, cert.key_bits),
410                    (KeyAlgorithm::Rsa, Some(bits)) if bits <= 4096
411                )
412            })
413        });
414    SignerTrust {
415        signature: sig.index,
416        trusted: untrusted.is_empty(),
417        untrusted,
418        chain,
419        digest_alg: sig.digest_alg,
420        signer_digest_alg: sig.signer_digest_alg,
421        os_ci_compatible,
422    }
423}

425/// The rank of a digest for primary selection: SHA-2 above SHA-1 above
426/// anything else.
427pub(crate) const fn digest_rank(d: Option<DigestAlgorithm>) -> u8 {
428    match d {
429        Some(a) if a.is_sha2() => 2,
430        Some(DigestAlgorithm::Sha1) => 1,
431        _ => 0,
432    }
433}
434
435/// The first trusted signature with the strongest digest, ranking each by
436/// the weaker of its image and signer digests (as #48 does).
437fn select_primary(signers: &[SignerTrust]) -> Option<usize> {
438    let mut best: Option<(usize, u8)> = None;
439    for (i, s) in signers.iter().enumerate().filter(|(_, s)| s.trusted) {
440        let r = digest_rank(s.digest_alg).min(digest_rank(s.signer_digest_alg));
441        if best.is_none_or(|(_, b)| r > b) {
442            best = Some((i, r));
443        }
444    }
445    best.map(|(i, _)| i)
446}

448pub(crate) fn identity(
449    signature: usize,
450    chain: &ChainReport,
451    os_ci_compatible: bool,
452    version: Option<&VersionInfo>,
453) -> SignerIdentity {
454    let mut issues = Vec::new();
455    let leaf = chain.leaf_cert();
456    let (publisher_name, leaf_cn) = match leaf {
457        Some(cert) => names(cert, &mut issues),
458        None => (None, None),
459    };
460    let publisher_dn = leaf
461        .filter(|c| {
462            !c.subject.rdns.is_empty()
463                && c.subject
464                    .attributes()
465                    .all(|a| !a.lossy && !a.value.chars().any(char::is_control))
466        })
467        .map(|c| c.subject.canonical.clone());
468    if leaf.is_some() && publisher_dn.is_none() {
469        issues.push(IdentityIssue::UnusableDn);
470    }
471
472    let product_name = version
473        .and_then(|v| v.get(VersionField::ProductName))
474        .filter(|p| !p.trim().is_empty())
475        .map(str::to_owned);
476    if product_name.is_none() {
477        issues.push(IdentityIssue::NoProductName);
478    }
479    let publisher_product = publisher_name
480        .as_ref()
481        .zip(product_name.as_ref())
482        .map(|(n, p)| (n.value.clone(), p.clone()));
483
484    let root = match (chain.well_known_root_id, chain.pca_cert()) {
485        (Some(id), _) => Some(OsRuleRoot::WellKnown(id)),
486        (None, Some(pca)) => {
487            let hash = pca.tbs_hash.clone().map(OsRuleRoot::PcaTbsHash);
488            if hash.is_none() {
489                issues.push(IdentityIssue::NoPcaTbsHash);
490            }
491            hash
492        }
493        (None, None) => {
494            issues.push(IdentityIssue::NoPca);
495            None
496        }
497    };
498    if !os_ci_compatible {
499        issues.push(IdentityIssue::NotOsCiCompatible);
500    }
501    let publisher_os_rule = root
502        .zip(leaf_cn)
503        .filter(|_| os_ci_compatible)
504        .map(|(root, leaf_cn)| PublisherOsRule { root, leaf_cn });
505
506    let file = version.and_then(|v| {
507        let name = v.effective_file_name(VersionField::OriginalFileName)?;
508        Some((name.value.to_owned(), name.field, v.file_version()?))
509    });
510    if file.is_none() {
511        issues.push(IdentityIssue::NoFileNameOrVersion);
512    }
513    let file_publisher_os_rule = publisher_os_rule.clone().zip(file).map(
514        |(publisher, (file_name, file_name_field, min_version))| FilePublisherOsRule {
515            publisher,
516            file_name,
517            file_name_field,
518            min_version,
519        },
520    );
521
522    SignerIdentity {
523        signature,
524        publisher_name,
525        publisher_dn,
526        product_name,
527        publisher_product,
528        publisher_os_rule,
529        file_publisher_os_rule,
530        issues,
531        package_display_name: None,
532    }
533}

535/// The publisher name and the leaf `CN` for OS rules. A name that is not
536/// clean is not used, and no fallback replaces it: falling back to `O`
537/// would let a garbled `CN` change the identity.
538fn names(
539    cert: &CertSummary,
540    issues: &mut Vec<IdentityIssue>,
541) -> (Option<PublisherName>, Option<String>) {
542    let subject = &cert.subject;
543    if subject.values(AttributeKind::CommonName).count() > 1 {
544        issues.push(IdentityIssue::MultipleCommonNames);
545    }
546    let (attr, source) = match (
547        subject.first(AttributeKind::CommonName),
548        subject.first(AttributeKind::Organization),
549    ) {
550        (Some(cn), _) => (cn, NameSource::CommonName),
551        (None, Some(o)) => (o, NameSource::Organization),
552        (None, None) => {
553            issues.push(IdentityIssue::NoName);
554            return (None, None);
555        }
556    };
557    if !attr.is_clean() {
558        issues.push(IdentityIssue::UnusableName);
559        return (None, None);
560    }
561    let leaf_cn = (source == NameSource::CommonName).then(|| attr.value.clone());
562    if leaf_cn.is_none() {
563        issues.push(IdentityIssue::NoCommonName);
564    }
565    (
566        Some(PublisherName {
567            value: attr.value.clone(),
568            source,
569        }),
570        leaf_cn,
571    )
