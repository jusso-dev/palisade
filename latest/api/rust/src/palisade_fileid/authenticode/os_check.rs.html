1//! Comparing this crate's signature results with the OS trust-verification
2//! API (#48).
3//!
4//! The Windows backend (`palisade_fileid::win::trust`, behind the `os-trust`
5//! feature) asks the OS to verify a file's embedded signatures, one call
6//! per signature, and records each result code as an [`OsSignatureResult`].
7//! [`cross_check`] then pairs every OS result with the signature in a
8//! [`SignatureReport`] that the OS looked at, and records whether the two
9//! agree. Everything here is pure, so it is tested on every OS.
10//!
11//! # Never an override
12//!
13//! The comparison is a **diagnostic**. [`cross_check`] takes the report by
14//! shared reference and returns a separate [`CrossCheck`]; nothing in this
15//! crate reads a [`CrossCheck`] to change a verdict. Policy decides from the
16//! Rust result, and a disagreement is something to log and investigate.
17//!
18//! # What is compared
19//!
20//! [`SignatureStatus::Valid`] means the signature is cryptographically
21//! sound and signs this file. It does not mean the signer is trusted. The
22//! OS result folds chain trust in, so an OS result is reduced to the same
23//! question before comparing ([`OsVerdict`]):
24//!
25//! - success, or a certificate chain or policy failure that the OS reports
26//!   only after the file digest and the signature have verified (an
27//!   untrusted root, an expired or revoked certificate, a wrong usage), is
28//!   **accepted**: the signature is intact;
29//! - a digest or signature failure, a missing signer certificate, an
30//!   unknown algorithm or a malformed message or certificate is
31//!   **rejected**, and "no signature" is its own verdict;
32//! - any other code is **unclassified**, and its comparison is
33//!   [`Outcome::NotComparable`]. That is not an agreement:
34//!   [`CrossCheck::agrees`] is false while any comparison is not
35//!   comparable, and [`CrossCheck::not_comparable`] lists them.
36//!
37//! The OS reads only the first embedded signature (index 0) and the
38//! signatures nested directly in it (indexes 1 and up, its "secondary"
39//! signatures). Signatures in a later certificate table entry, or nested
40//! more deeply, are not compared.
41//!
42//! This crate is stricter than the OS in places, by design: an image whose
43//! layout departs from the Authenticode specification is
44//! [`SignatureStatus::NonConformantImage`], and MD5 and MD2 are never
45//! valid. Those show up as [`Outcome::OsAcceptsRustRejects`]. The opposite,
46//! [`Outcome::RustAcceptsOsRejects`], is the one that could hide a
47//! verification bug, and [`CrossCheck::rust_accepts_os_rejects`] picks it
48//! out.
49

159/// What an OS result code says about the signature itself.
160#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
161pub enum OsVerdict {
162    /// The signature verified and the OS trusts its chain.
163    Trusted,
164    /// The signature verified, but the OS does not trust its certificate
165    /// chain, rejected it on policy (an untrusted root, an expired or
166    /// revoked certificate, a wrong usage), or could not verify its
167    /// timestamp. The OS checks the file digest and the signature before
168    /// the chain and the countersignatures, and reports their failures
169    /// first, so the signature itself is intact. ([`super::SignatureStatus`]
170    /// does not cover chains or timestamps either.)
171    ChainNotTrusted,
172    /// The file digest, the `messageDigest` attribute or the signature
173    /// value does not verify, the signer certificate is missing, an
174    /// algorithm is unknown, or the message or a certificate is malformed.
175    Broken,
176    /// The OS found no signature.
177    NoSignature,
178    /// A code this module does not classify.
179    Unclassified,
180}
181
182impl OsVerdict {
183    /// Whether the OS found the signature intact (`Some(true)`), not intact
184    /// or absent (`Some(false)`), or the code says neither (`None`).
185    #[must_use]
186    pub const fn accepts(self) -> Option<bool> {
187        match self {
188            Self::Trusted | Self::ChainNotTrusted => Some(true),
189            Self::Broken | Self::NoSignature => Some(false),
190            Self::Unclassified => None,
191        }
192    }
193}
194
195/// Classifies an OS trust-verification result code.
196#[must_use]
197pub const fn classify(code: u32) -> OsVerdict {
198    match code {
199        code::S_OK => OsVerdict::Trusted,
200        code::TRUST_E_NOSIGNATURE => OsVerdict::NoSignature,
201        code::TRUST_E_BAD_DIGEST
202        | code::TRUST_E_NO_SIGNER_CERT
203        | code::TRUST_E_MALFORMED_SIGNATURE
204        | code::CRYPT_E_HASH_VALUE
205        | code::NTE_BAD_SIGNATURE
206        | code::CRYPT_E_BAD_MSG
207        | code::CRYPT_E_BAD_ENCODE
208        | code::CRYPT_E_UNKNOWN_ALGO
209        | code::CRYPT_E_INVALID_MSG_TYPE
210        | code::CRYPT_E_UNEXPECTED_ENCODING
211        | code::CRYPT_E_AUTH_ATTR_MISSING
212        | code::CRYPT_E_SIGNER_NOT_FOUND
213        | code::CRYPT_E_ATTRIBUTES_MISSING
214        | code::CRYPT_E_UNEXPECTED_MSG_TYPE
215        | code::CRYPT_E_NO_SIGNER
216        | code::NTE_BAD_DATA
217        | code::NTE_BAD_ALGID
218        | code::NTE_BAD_PUBLIC_KEY
219        | code::CERT_E_MALFORMED
220        | code::DIGSIG_E_ENCODE
221        | code::DIGSIG_E_DECODE
222        | code::DIGSIG_E_EXTENSIBILITY
223        | code::DIGSIG_E_CRYPTO
224        | code::CRYPT_E_ASN1_FIRST..=code::CRYPT_E_ASN1_LAST => OsVerdict::Broken,
225        code::CERT_E_EXPIRED
226        | code::CERT_E_VALIDITYPERIODNESTING
227        | code::CERT_E_PURPOSE
228        | code::CERT_E_UNTRUSTEDROOT
229        | code::CERT_E_CHAINING
230        | code::CERT_E_REVOKED
231        | code::CERT_E_UNTRUSTEDTESTROOT
232        | code::CERT_E_REVOCATION_FAILURE
233        | code::CERT_E_WRONG_USAGE
234        | code::TRUST_E_EXPLICIT_DISTRUST
235        | code::CERT_E_UNTRUSTEDCA
236        | code::TRUST_E_CERT_SIGNATURE
237        | code::TRUST_E_BASIC_CONSTRAINTS
238        | code::CRYPT_E_REVOKED
239        | code::CRYPT_E_NO_REVOCATION_CHECK
240        | code::CRYPT_E_REVOCATION_OFFLINE
241        | code::TRUST_E_TIME_STAMP
242        | code::TRUST_E_COUNTER_SIGNER => OsVerdict::ChainNotTrusted,
243        _ => OsVerdict::Unclassified,
244    }
245}

271    /// How many secondary (nested) signatures the OS said the primary has.

278/// How one OS result compares with the Rust result for the same signature.
279#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
280pub enum Outcome {
281    /// Both accept the signature, or both reject it, or both find none:
282    /// an unsigned file, or (at index 0) a certificate table this crate
283    /// could not read and the OS rejected.
284    Agree,
285    /// The OS finds the signature intact and this crate does not: expected
286    /// for the stricter checks described in the module documentation, and
287    /// otherwise worth investigating.
288    OsAcceptsRustRejects,
289    /// This crate finds the signature valid and the OS does not. This could
290    /// hide a verification bug.
291    RustAcceptsOsRejects,
292    /// The OS reported a signature (intact or broken) that has no
293    /// counterpart in the report.
294    OnlyOs,
295    /// The report has a signature the OS should have seen, at an index the
296    /// OS did not report.
297    OnlyRust,
298    /// The OS code is unclassified, so nothing can be said. This is not an
299    /// agreement; see [`CrossCheck::not_comparable`].
300    NotComparable,
301}

303/// One compared pair.
304#[derive(Debug, Clone, PartialEq, Eq)]
305pub struct Comparison {
306    /// The OS signature index.
307    pub os_index: u32,
308    /// The index of the matching signature in
309    /// [`SignatureReport::signatures`], if there is one.
310    pub signature: Option<usize>,
311    /// The Rust status of that signature.
312    pub rust_status: Option<SignatureStatus>,
313    /// The OS result, if the OS reported this index.
314    pub os: Option<OsSignatureResult>,
315    /// The outcome.
316    pub outcome: Outcome,
317}

319/// The result of [`cross_check`]. It is a diagnostic only; see the module
320/// documentation.
321#[derive(Debug, Clone, PartialEq, Eq, Default)]
322pub struct CrossCheck {
323    /// One entry per OS signature index that either side has, in index
324    /// order.
325    pub comparisons: Vec<Comparison>,
326    /// The number of secondary signatures in the report (signatures nested
327    /// directly in the primary), to compare with
328    /// [`OsTrustResult::secondary_count`].
329    pub rust_secondary_count: usize,
330    /// The OS and the report disagree on how many secondary signatures
331    /// there are.
332    pub secondary_count_differs: bool,
333}
334
335impl CrossCheck {
336    /// Whether every comparison is [`Outcome::Agree`] and the secondary
337    /// counts match. A comparison that is [`Outcome::NotComparable`] makes
338    /// this false: an OS code nobody classified is not evidence of
339    /// agreement.
340    #[must_use]
341    pub fn agrees(&self) -> bool {
342        !self.secondary_count_differs
343            && self.comparisons.iter().all(|c| c.outcome == Outcome::Agree)
344    }
345
346    /// The comparisons where the two sides reach different answers. Those
347    /// that cannot be compared are in [`Self::not_comparable`] instead.
348    pub fn disagreements(&self) -> impl Iterator<Item = &Comparison> {
349        self.comparisons
350            .iter()
351            .filter(|c| !matches!(c.outcome, Outcome::Agree | Outcome::NotComparable))
352    }
353
354    /// The comparisons whose OS code is unclassified.
355    pub fn not_comparable(&self) -> impl Iterator<Item = &Comparison> {
356        self.comparisons
357            .iter()
358            .filter(|c| c.outcome == Outcome::NotComparable)
359    }
360
361    /// The comparisons where this crate accepts a signature the OS rejects.
362    pub fn rust_accepts_os_rejects(&self) -> impl Iterator<Item = &Comparison> {
363        self.comparisons
364            .iter()
365            .filter(|c| c.outcome == Outcome::RustAcceptsOsRejects)
366    }
367}

369/// The signatures of `report` the OS reads, by OS index: the first
370/// embedded signature, then the signatures nested directly in it, in
371/// order.
372fn os_view(report: &SignatureReport) -> Vec<usize> {
373    let Some(primary) = report
374        .signatures
375        .iter()
376        .position(|s| s.kind == SignatureKind::Embedded)
377    else {
378        return Vec::new();
379    };
380    let mut view = vec![primary];
381    view.extend(
382        report
383            .signatures
384            .iter()
385            .enumerate()
386            .filter(|(_, s)| s.kind == SignatureKind::Nested && s.parent == Some(primary))
387            .map(|(i, _)| i),
388    );
389    view
390}
391
392/// Whether this crate found the certificate table unreadable, with no
393/// signature in it read.
394fn table_unreadable(report: &SignatureReport) -> bool {
395    matches!(
396        report.signing_state,
397        SigningState::Invalid(InvalidReason::CertificateTable(_))
398    )
399}

401/// Compares the OS results for a file with the report this crate produced
402/// for the same file. The report is not changed.
403///
404/// OS indexes 0 to [`MAX_OS_SIGNATURES`] minus one are compared; a result
405/// at a larger index (which the Windows backend never asks for) is
406/// ignored, so a crafted result cannot make the loop long.
407#[must_use]
408pub fn cross_check(report: &SignatureReport, os: &OsTrustResult) -> CrossCheck {
409    let view = os_view(report);
410    let rust_secondary_count = view.len().saturating_sub(1);
411    let mut out = CrossCheck {
412        comparisons: Vec::new(),
413        rust_secondary_count,
414        secondary_count_differs: false,
415    };
416    let os_primary = os.signatures.iter().find(|r| r.index == 0);
417    // The OS reports "no signature" at index 0 for an unsigned file. Its
418    // secondary count is then meaningless.
419    let os_unsigned = os_primary.is_some_and(|r| r.verdict() == OsVerdict::NoSignature);
420    out.secondary_count_differs = !os_unsigned
421        && !os.truncated
422        && usize::try_from(os.secondary_count).ok() != Some(rust_secondary_count);
423
424    let os_max = os.signatures.iter().map(|r| r.index).max();
425    let rust_max = u32::try_from(view.len())
426        .ok()
427        .and_then(|n| n.checked_sub(1));
428    let Some(last) = os_max.max(rust_max) else {
429        return out;
430    };
431    for os_index in 0..=last.min(MAX_OS_SIGNATURES - 1) {
432        let os_result = os.signatures.iter().find(|r| r.index == os_index).copied();
433        let signature = usize::try_from(os_index)
434            .ok()
435            .and_then(|i| view.get(i))
436            .copied();
437        let rust_status = signature
438            .and_then(|i| report.signatures.get(i))
439            .map(|s| s.status.clone());
440        let outcome = match (&rust_status, os_result) {
441            (None, None) => continue,
442            (Some(_), None) => Outcome::OnlyRust,
443            (None, Some(r)) => match r.verdict() {
444                // Neither side has a signature here.
445                OsVerdict::NoSignature => Outcome::Agree,
446                // The OS found and rejected a signature this crate did not
447                // read. Only at index 0, and only when this crate also
448                // rejected the certificate table, do the two agree.
449                OsVerdict::Broken if os_index == 0 && table_unreadable(report) => Outcome::Agree,
450                OsVerdict::Broken | OsVerdict::Trusted | OsVerdict::ChainNotTrusted => {
451                    Outcome::OnlyOs
452                }
453                OsVerdict::Unclassified => Outcome::NotComparable,
454            },
455            (Some(status), Some(r)) => match r.verdict().accepts() {
456                None => Outcome::NotComparable,
457                Some(os_ok) => match (status.is_valid(), os_ok) {
458                    (true, true) | (false, false) => Outcome::Agree,
459                    (false, true) => Outcome::OsAcceptsRustRejects,
460                    (true, false) => Outcome::RustAcceptsOsRejects,
461                },
462            },
463        };
464        out.comparisons.push(Comparison {
465            os_index,
466            signature,
467            rust_status,
468            os: os_result,
469            outcome,
470        });
471    }
472    out
473}

673            crate::authenticode::CertificateTableError::Malformed { offset: 0 },

676        assert_eq!(outcomes(&check), [Outcome::Agree]);

678        let r = report(vec![sig(
