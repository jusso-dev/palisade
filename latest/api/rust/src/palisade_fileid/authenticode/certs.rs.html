1//! Certificate extraction (#49): every certificate a signed file carries,
2//! for building trust bundles offline.
3//!
4//! [`extract_certificates`] takes the bytes of a signed file and returns
5//! the certificates embedded in its signatures, outer and nested:
6//!
7//! - a **PE image**: every PKCS#7 entry of its certificate table;
8//! - a **cabinet** (`MSCF`): the PKCS#7 signature its reserved header area
9//!   points to;
10//! - a bare **PKCS#7 SignedData** (a detached `.p7`, an `.p7b`, a catalog
11//!   `.cat` or a package signature): the SignedData itself.
12//!
13//! [`CertSelection`] picks all of them, only each signature's signer (the
14//! leaf), or only the self-signed roots, and [`CertificateBundle`] collects
15//! certificates from many files without duplicates and writes them as one
16//! PEM bundle or as DER files. The `pal fileid certs` command (#55) is a
17//! thin wrapper over these; walking directories is the caller's job.
18//!
19//! Nothing here verifies a signature or decides trust: a certificate is
20//! listed because the file carries it, which is what a bundle author needs
21//! to review. The same strict DER reader as the rest of the crate is used:
22//! every element of each SignedData is checked as DER (tags and lengths),
23//! nested signatures included, one level at a time. A malformed outer
24//! signature is an error; a malformed nested one, or a certificate that
25//! does not parse as X.509, is skipped with a warning. Every count is
26//! bounded ([`MAX_EXTRACTED_SIGNATURES`], [`super::MAX_CERTIFICATES`] per
27//! signature, [`MAX_EXTRACT_NESTING`]). Only the first SignerInfo of a
28//! SignedData is read for its signer, as Authenticode and catalogs have
29//! exactly one.
30

82/// Why nothing could be extracted.
83#[derive(Debug, Clone, PartialEq, Eq)]
84#[non_exhaustive]
85pub enum ExtractError {
86    /// The input is larger than [`MAX_EXTRACT_INPUT`].
87    TooLarge(usize),
88    /// The input is not a PE image, a cabinet or a PKCS#7 SignedData.
89    UnknownFormat,
90    /// A PE image that could not be parsed.
91    Pe(String),
92    /// A cabinet whose header or signature location is invalid.
93    Cabinet(CabinetError),
94    /// The file is of a known kind but carries no signature.
95    Unsigned,
96    /// A signature (the outer one, or the only one) is not valid DER.
97    Asn1(Asn1Error),
98}

100impl fmt::Display for ExtractError {
101    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
102        match self {
103            Self::TooLarge(n) => write!(f, "{n} bytes is larger than {MAX_EXTRACT_INPUT}"),
104            Self::UnknownFormat => f.write_str("not a PE image, a cabinet or a PKCS#7 SignedData"),
105            Self::Pe(e) => write!(f, "PE image: {e}"),
106            Self::Cabinet(e) => write!(f, "cabinet: {e}"),
107            Self::Unsigned => f.write_str("the file carries no signature"),
108            Self::Asn1(e) => write!(f, "signature: invalid DER: {e}"),
109        }
110    }
111}

115/// Why a cabinet's signature could not be located.
116#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
117#[non_exhaustive]
118pub enum CabinetError {
119    /// The header is shorter than its fixed fields.
120    Truncated,
121    /// The header's `cbCabinet` is longer than the file.
122    LengthMismatch {
123        /// What the header says.
124        declared: u32,
125        /// The file's length.
126        actual: usize,
127    },
128    /// The reserved header area is too short to hold a signature location.
129    ReserveTooShort(u16),
130    /// The signature location lies outside the file, or inside the cabinet
131    /// itself (before `cbCabinet`): a signer appends it after the cabinet.
132    SignatureOutOfRange {
133        /// Its offset.
134        offset: u32,
135        /// Its length.
136        size: u32,
137    },
138    /// `cbCabinet` is smaller than the fixed header.
139    TooShort(u32),
140    /// The reserved area names a signature of zero bytes.
141    EmptySignature,
142}
143
144impl fmt::Display for CabinetError {
145    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
146        match self {
147            Self::Truncated => f.write_str("truncated header"),
148            Self::LengthMismatch { declared, actual } => {
149                write!(
150                    f,
151                    "header says {declared} bytes, the file has only {actual}"
152                )
153            }
154            Self::ReserveTooShort(n) => {
155                write!(f, "reserved header area of {n} bytes holds no signature")
156            }
157            Self::SignatureOutOfRange { offset, size } => write!(
158                f,
159                "signature at {offset} ({size} bytes) is outside the file or inside the cabinet"
160            ),
161            Self::TooShort(n) => write!(f, "header says the cabinet is only {n} bytes"),
162            Self::EmptySignature => f.write_str("the signature location is empty"),
163        }
164    }
165}

222pub struct Extraction {
223    /// The kind of file.
224    pub kind: SignedFileKind,
225    /// How many signatures were read.
226    pub signatures: usize,

235/// Extracts the certificates of a signed file. See the module docs.
236///
237/// At most [`super::MAX_CERTIFICATES`] (64) certificates are read from any
238/// one SignedData, including a certificate bag (`.p7b`) with no signer; any
239/// more are left out and [`ExtractWarning::LimitReached`] is recorded. A
240/// larger bag must be split before it is read.
241pub fn extract_certificates(
242    data: &[u8],
243    selection: CertSelection,
244) -> Result<Extraction, ExtractError> {
245    if data.len() > MAX_EXTRACT_INPUT {
246        return Err(ExtractError::TooLarge(data.len()));
247    }
248    let mut table_warnings = Vec::new();
249    let (kind, blobs) = if data.starts_with(b"MZ") {
250        (
251            SignedFileKind::PeImage,
252            pe_blobs(data, &mut table_warnings)?,
253        )
254    } else if data.starts_with(b"MSCF") {
255        let blob = cabinet_signature(data).map_err(ExtractError::Cabinet)?;
256        let blobs = if blob.is_empty() {
257            Vec::new()
258        } else {
259            vec![blob]
260        };
261        (SignedFileKind::Cabinet, blobs)
262    } else if data.first() == Some(&tag::SEQUENCE) {
263        (SignedFileKind::Pkcs7, vec![data])
264    } else {
265        return Err(ExtractError::UnknownFormat);
266    };
267    if blobs.is_empty() {
268        return Err(ExtractError::Unsigned);
269    }
270
271    let mut walker = Walker::default();
272    for w in table_warnings {
273        walker.warn(w);
274    }
275    for (i, blob) in blobs.iter().enumerate() {
276        if let Err(e) = walker.signature(blob, 0) {
277            // The first signature must read; a later one is a warning.
278            if i == 0 && walker.signatures.is_empty() {
279                return Err(ExtractError::Asn1(e));
280            }
281            walker.warn(ExtractWarning::UnreadableSignature {
282                signature: walker.signatures.len(),
283            });
284        }
285    }
286
287    // Certificate self-signature checks and chain links share one budget
288    // for the file, as `evaluate_trust` does.
289    let mut budget = Budget::new(MAX_FILE_SIGNATURE_CHECKS);
290    let mut certificates = Vec::new();
291    let mut cert_warnings = Vec::new();
292    for (index, sig) in walker.signatures.iter().enumerate() {
293        let mut parsed: Vec<Option<CertSummary>> = Vec::with_capacity(sig.certs.len());
294        for (position, der) in sig.certs.iter().enumerate() {
295            let mut one = Budget::new(budget.left().min(1));
296            let result = x509::parse_within(der, &mut one);
297            budget.absorb(&one);
298            match result {
299                Ok(c) => parsed.push(Some(c)),
300                Err(_) => {
301                    cert_warnings.push(ExtractWarning::UnparseableCertificate {
302                        signature: index,
303                        position,
304                    });
305                    parsed.push(None);
306                }
307            }
308        }
309        // The signer: the first certificate that parsed in full and that
310        // the SignerInfo's identifier names.
311        let signer = sig.sid.as_ref().and_then(|sid| {
312            sig.certs.iter().enumerate().position(|(i, der)| {
313                parsed.get(i).is_some_and(Option::is_some)
314                    && pkcs7::parse_certificate(der).is_ok_and(|c| pkcs7::matches(&c, sid))
315            })
316        });
317        if signer.is_none() {
318            cert_warnings.push(ExtractWarning::SignerNotFound { signature: index });
319        }
320        let signer_thumbprint = signer
321            .and_then(|s| parsed.get(s))
322            .and_then(Option::as_ref)
323            .map(|c| c.sha256_thumbprint.clone());
324        // The root: the self-signed top of the signer's as-provided chain,
325        // built with no trust configuration (nothing is trusted here).
326        let owned: Vec<Vec<u8>> = sig.certs.iter().map(|c| c.to_vec()).collect();
327        let root_thumbprint = signer
328            .and_then(|s| build_chain_within(&owned, s, &TrustConfig::new(), &mut budget).ok())
329            .and_then(|chain| chain.root_cert().map(|r| r.sha256_thumbprint.clone()));
330        let mut seen: HashSet<String> = HashSet::new();
331        for cert in parsed.into_iter().flatten() {
332            if !seen.insert(cert.sha256_thumbprint.clone()) {
333                continue;
334            }
335            let is_signer = signer_thumbprint.as_deref() == Some(cert.sha256_thumbprint.as_str());
336            let is_root = root_thumbprint.as_deref() == Some(cert.sha256_thumbprint.as_str());
337            let keep = match selection {
338                CertSelection::All => true,
339                CertSelection::Leaf => is_signer,
340                CertSelection::Root => is_root,
341            };
342            if keep {
343                certificates.push(ExtractedCert {
344                    signature: index,
345                    is_signer,
346                    is_root,
347                    cert,
348                });
349            }
350        }
351    }
352    if budget.exhausted {
353        walker.warn(ExtractWarning::LimitReached);
354    }
355    let mut warnings = walker.warnings;
356    warnings.extend(cert_warnings);
357    Ok(Extraction {
358        kind,
359        signatures: walker.signatures.len(),
360        certificates,
361        warnings,
362    })
363}

365/// The PKCS#7 entries of a PE image's certificate table, at most
366/// [`super::MAX_CERTIFICATE_ENTRIES`] entries of any type, as the verifier
367/// reads it (#48). A table that is declared but rejected, too large, or
368/// whose first entry is malformed is an error, never "unsigned".
369fn pe_blobs<'a>(
370    data: &'a [u8],
371    warnings: &mut Vec<ExtractWarning>,
372) -> Result<Vec<&'a [u8]>, ExtractError> {
373    let info = pe::parse_with(data, &ParseOptions::metadata_only())
374        .map_err(|e| ExtractError::Pe(e.to_string()))?;
375    let Some(table) = info.certificate_table else {
376        let declared = info
377            .data_directories
378            .get(DataDirectoryKind::Security.index())
379            .is_some_and(|entry| entry.size != 0);
380        return if declared {
381            Err(ExtractError::Pe(
382                "the certificate table is declared but invalid".to_owned(),
383            ))
384        } else {
385            Ok(Vec::new())
386        };
387    };
388    if table.size > super::MAX_CERTIFICATE_TABLE_SIZE {
389        return Err(ExtractError::Pe(format!(
390            "certificate table of {} bytes is too large",
391            table.size
392        )));
393    }
394    let table = usize::try_from(table.offset)
395        .ok()
396        .zip(usize::try_from(table.size).ok())
397        .and_then(|(offset, size)| data.get(offset..offset.checked_add(size)?))
398        .ok_or_else(|| ExtractError::Pe("certificate table outside the file".to_owned()))?;
399    let mut blobs = Vec::new();
400    let mut pos = 0_usize;
401    let mut entries = 0_usize;
402    while pos < table.len() {
403        if entries >= super::MAX_CERTIFICATE_ENTRIES {
404            warnings.push(ExtractWarning::LimitReached);
405            break;
406        }
407        entries += 1;
408        let header = table.get(pos..pos.saturating_add(8));
409        let length = header
410            .and_then(|h| h.get(0..4))
411            .and_then(|b| b.try_into().ok())
412            .map(u32::from_le_bytes)
413            .and_then(|l| usize::try_from(l).ok())
414            .filter(|l| *l >= 8);
415        let body = length.and_then(|l| table.get(pos.saturating_add(8)..pos.checked_add(l)?));
416        let (Some(header), Some(length), Some(body)) = (header, length, body) else {
417            if pos == 0 {
418                return Err(ExtractError::Pe(
419                    "the first certificate table entry is malformed".to_owned(),
420                ));
421            }
422            // Padding or a damaged later entry: the walk stops here.
423            break;
424        };
425        let kind = header
426            .get(6..8)
427            .and_then(|b| b.try_into().ok())
428            .map(u16::from_le_bytes);
429        // WIN_CERT_TYPE_PKCS_SIGNED_DATA.
430        if kind == Some(2) {
431            // The SignedData element only, without the entry's padding.
432            let mut r = Reader::new(body, 0);
433            if let Ok(tlv) = r.read(tag::SEQUENCE) {
434                blobs.push(tlv.raw);
435            } else {
436                blobs.push(body);
437            }
438        }
439        let next = pos.saturating_add(length).saturating_add(7) & !7;
440        if next <= pos {
441            break;
442        }
443        pos = next;
444    }
445    Ok(blobs)
446}

448/// The signature of a cabinet: `CFHEADER` with `cfhdrRESERVE_PRESENT`,
449/// whose reserved area starts with a 4-byte value, then the signature's
450/// offset and length (both 32-bit little-endian). `cbCabinet` counts the
451/// cabinet only; the signer appends the signature after it.
452fn cabinet_signature(data: &[u8]) -> Result<&[u8], CabinetError> {
453    const FIXED: usize = 36;
454    const RESERVE_PRESENT: u16 = 0x0004;
455    let u16_at = |at: usize| {
456        data.get(at..at + 2)
457            .and_then(|b| b.try_into().ok())
458            .map(u16::from_le_bytes)
459    };
460    let u32_at = |at: usize| {
461        data.get(at..at + 4)
462            .and_then(|b| b.try_into().ok())
463            .map(u32::from_le_bytes)
464    };
465    let (Some(declared), Some(flags)) = (u32_at(8), u16_at(30)) else {
466        return Err(CabinetError::Truncated);
467    };
468    if data.len() < FIXED {
469        return Err(CabinetError::Truncated);
470    }
471    if usize::try_from(declared).is_ok_and(|d| d < FIXED) {
472        return Err(CabinetError::TooShort(declared));
473    }
474    let Some(cabinet_end) = usize::try_from(declared).ok().filter(|d| *d <= data.len()) else {
475        return Err(CabinetError::LengthMismatch {
476            declared,
477            actual: data.len(),
478        });
479    };
480    if flags & RESERVE_PRESENT == 0 {
481        return Ok(&[]);
482    }
483    let Some(reserve) = u16_at(FIXED) else {
484        return Err(CabinetError::Truncated);
485    };
486    // cbCFHeader (2), cbCFFolder (1), cbCFData (1), then abReserve.
487    let area = FIXED + 4;
488    if reserve < 12 {
489        return Err(CabinetError::ReserveTooShort(reserve));
490    }
491    let (Some(offset), Some(size)) = (u32_at(area + 4), u32_at(area + 8)) else {
492        return Err(CabinetError::Truncated);
493    };
494    let range = usize::try_from(offset)
495        .ok()
496        .zip(usize::try_from(size).ok())
497        .filter(|(o, _)| *o >= cabinet_end.max(area + usize::from(reserve)))
498        .and_then(|(o, s)| data.get(o..o.checked_add(s)?));
499    if size == 0 {
500        return Err(CabinetError::EmptySignature);
501    }
502    let blob = range.ok_or(CabinetError::SignatureOutOfRange { offset, size })?;
503    // The signer pads the blob; keep the SignedData element only.
504    let mut r = Reader::new(blob, 0);
505    Ok(r.read(tag::SEQUENCE).map_or(blob, |tlv| tlv.raw))
506}

508/// One signature's certificates and signer.
509struct Found<'a> {
510    certs: Vec<&'a [u8]>,
511    sid: Option<SignerIdentifier>,
512}
513
514#[derive(Default)]
515struct Walker<'a> {
516    signatures: Vec<Found<'a>>,
517    /// Signatures tried, read or not: bounded by
518    /// [`MAX_EXTRACTED_SIGNATURES`], so unreadable nested values cannot make
519    /// the walk try without end.
520    attempts: usize,
521    warnings: Vec<ExtractWarning>,
522}
523
524impl<'a> Walker<'a> {
525    fn warn(&mut self, w: ExtractWarning) {
526        if !self.warnings.contains(&w) {
527            self.warnings.push(w);
528        }
529    }
530
531    /// Reads one `ContentInfo` holding a SignedData, whatever its content
532    /// type, then its nested signatures.
533    fn signature(&mut self, der: &'a [u8], depth: usize) -> Result<(), Asn1Error> {
534        if self.attempts >= MAX_EXTRACTED_SIGNATURES || depth > MAX_EXTRACT_NESTING {
535            self.warn(ExtractWarning::LimitReached);
536            return Ok(());
537        }
538        self.attempts += 1;
539        let mut outer = Reader::new(der, 0);
540        // Read with the strict DER reader. Each certificate is validated
541        // whole; nested signatures are read (and so checked) one level at a
542        // time below, so their depth is bounded by MAX_EXTRACT_NESTING, not
543        // by the DER reader's own depth limit.
544        let content_info = outer.read(tag::SEQUENCE)?;
545        let mut ci = content_info.reader();
546        let content_type = read_oid(&mut ci)?;
547        if content_type != oid::SIGNED_DATA {
548            return Err(content_info.error(asn1::Asn1ErrorKind::InvalidValue));
549        }
550        let explicit = ci.read(tag::context_cons(0))?;
551        ci.finish()?;
552        let sd = asn1::single(explicit.content, explicit.content_offset, tag::SEQUENCE)?;
553        let mut r = sd.reader();
554        for t in [tag::INTEGER, tag::SET, tag::SEQUENCE] {
555            asn1::validate(&r.read(t)?)?;
556        }
557        let mut certs = Vec::new();
558        if let Some(set) = r.read_optional(tag::context_cons(0))? {
559            let mut c = set.reader();
560            while !c.is_empty() {
561                let choice = c.read_any()?;
562                asn1::validate(&choice)?;
563                if choice.tag == tag::SEQUENCE {
564                    if certs.len() < super::MAX_CERTIFICATES {
565                        certs.push(choice.raw);
566                    } else {
567                        self.warn(ExtractWarning::LimitReached);
568                    }
569                }
570            }
571        }
572        if let Some(crls) = r.read_optional(tag::context_cons(1))? {
573            asn1::validate(&crls)?;
574        }
575        let infos = r.read(tag::SET)?;
576        r.finish()?;
577        // Authenticode and catalogs have exactly one SignerInfo; only the
578        // first is read for the signer and nested signatures, and any
579        // others are checked as DER.
580        let mut sid = None;
581        let mut nested: Vec<&'a [u8]> = Vec::new();
582        let mut i = infos.reader();
583        let mut first = true;
584        while !i.is_empty() {
585            let info = i.read(tag::SEQUENCE)?;
586            if first {
587                let truncated;
588                (sid, nested, truncated) = signer_info(&info)?;
589                if truncated {
590                    self.warn(ExtractWarning::LimitReached);
591                }
592                first = false;
593            } else {
594                asn1::validate(&info)?;
595            }
596        }
597        self.signatures.push(Found { certs, sid });
598        for child in nested {
599            let index = self.signatures.len();
600            if self.signature(child, depth + 1).is_err() {
601                self.warn(ExtractWarning::UnreadableSignature { signature: index });
602            }
603        }
604        Ok(())
605    }
606}
607
608/// A SignerInfo's identifier, its nested signatures, and whether nested
609/// values past the limit were dropped.
610type SignerParts<'a> = (Option<SignerIdentifier>, Vec<&'a [u8]>, bool);
611
612/// A SignerInfo's identifier and its nested signatures. Every part but the
613/// nested signatures (which the walker reads itself, one level at a time)
614/// is checked as DER.
615///
616/// At most [`MAX_EXTRACTED_SIGNATURES`] nested values are kept; the rest
617/// are still checked as DER, and the third value says some were dropped.
618fn signer_info<'a>(info: &Tlv<'a>) -> Result<SignerParts<'a>, Asn1Error> {
619    let mut r = info.reader();
620    r.read(tag::INTEGER)?;
621    let sid_start = r.clone();
622    let sid = pkcs7::parse_sid(&mut r).ok();
623    if sid.is_none() {
624        // Not a form `parse_sid` knows: still one valid element.
625        r = sid_start;
626        asn1::validate(&r.read_any()?)?;
627    }
628    let mut nested = Vec::new();
629    let mut truncated = false;
630    while !r.is_empty() {
631        let field = r.read_any()?;
632        if field.tag != tag::context_cons(1) {
633            asn1::validate(&field)?;
634            continue;
635        }
636        let mut attrs = field.reader();
637        while !attrs.is_empty() {
638            let attr = attrs.read(tag::SEQUENCE)?;
639            let mut a = attr.reader();
640            let id = read_oid(&mut a)?;
641            let values = a.read(tag::SET)?;
642            a.finish()?;
643            if id == oid::NESTED_SIGNATURE {
644                let mut v = values.reader();
645                while !v.is_empty() {
646                    let value = v.read_any()?;
647                    if nested.len() < MAX_EXTRACTED_SIGNATURES {
648                        nested.push(value.raw);
649                    } else {
650                        truncated = true;
651                    }
652                }
653            } else {
654                asn1::validate(&values)?;
655            }
656        }
657    }
658    Ok((sid, nested, truncated))
659}

661/// Certificates from many files, each kept once (by SHA-256 thumbprint), in
662/// the order first seen, up to [`MAX_BUNDLE_CERTIFICATES`].
663#[derive(Debug, Clone, Default)]
664pub struct CertificateBundle {
665    certificates: Vec<CertSummary>,
666    seen: HashSet<String>,
667    /// Whether certificates past [`MAX_BUNDLE_CERTIFICATES`] were dropped.
668    pub truncated: bool,
669}
670
671impl CertificateBundle {
672    /// An empty bundle.
673    #[must_use]
674    pub fn new() -> Self {
675        Self::default()
676    }
677
678    /// Adds the certificates of `extraction`. Returns how many were new.
679    pub fn add(&mut self, extraction: &Extraction) -> usize {
680        let mut added = 0;
681        for c in &extraction.certificates {
682            if self.insert(&c.cert) {
683                added += 1;
684            }
685        }
686        added
687    }
688
689    /// Adds one certificate. Returns whether it was new.
690    pub fn insert(&mut self, cert: &CertSummary) -> bool {
691        if self.seen.contains(&cert.sha256_thumbprint) {
692            return false;
693        }
694        if self.certificates.len() >= MAX_BUNDLE_CERTIFICATES {
695            self.truncated = true;
696            return false;
697        }
698        self.seen.insert(cert.sha256_thumbprint.clone());
699        self.certificates.push(cert.clone());
700        true
701    }
702
703    /// The certificates, in the order first seen.
704    #[must_use]
705    pub fn certificates(&self) -> &[CertSummary] {
706        &self.certificates
707    }
708
709    /// Every certificate as PEM, one after another.
710    #[must_use]
711    pub fn to_pem(&self) -> String {
712        self.certificates.iter().map(CertSummary::pem).collect()
713    }
714
715    /// Each certificate's DER, with a file name: its SHA-256 thumbprint
716    /// and `.cer`. Thumbprints are hex, so the names are safe on every OS.
717    pub fn der_files(&self) -> impl Iterator<Item = (String, &[u8])> {
718        self.certificates
719            .iter()
720            .map(|c| (format!("{}.cer", c.sha256_thumbprint), c.der.as_slice()))
721    }
722}
