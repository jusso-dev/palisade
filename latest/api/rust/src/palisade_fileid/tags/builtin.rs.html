1//! The built-in, read-only tag set.
2//!
3//! Every rule here matches a structure a format defines: a header, a
4//! signature or a marker its own loader looks for. They are written as hex,
5//! so no format's own name appears as text. Tags use descriptive, neutral
6//! names (docs/NAMING.md). The fixtures in `tests/tags.rs` build a minimal
7//! instance of each structure byte by byte; no third-party installer or
8//! sample is checked in.
9//!
10//! A built-in tag is evidence of what a file contains, never of whether it
11//! is trustworthy: anyone can build a file with these bytes.
12
13#![forbid(unsafe_code)]
14

38/// A PE image that may carry a Windows Installer database: a compound file
39/// header after the start of the file, and the installer database root
40/// CLSID anywhere in it. The two are found independently: the rule does
41/// not check that the CLSID is the root entry of that compound file, so a
42/// PE that holds some other compound file and the same 16 bytes elsewhere
43/// is tagged too. Installer wrappers and bootstrappers that unpack and run
44/// an MSI look like this.
45pub const INSTALLER_MSI_WRAPPER: &str = "installer:msi-wrapper";
46/// A PE image whose appended data holds the first header of a widely used
47/// script-driven installer system: a flags word, the signature
48/// `0xDEADBEEF` and the system's 12-byte marker.
49pub const INSTALLER_SCRIPTED_SETUP: &str = "installer:scripted-setup";
50/// A PE image with a setup loader offset table (a 12-byte ID, then a
51/// version number below 16; the known versions are 1 and 2), as a widely
52/// used Pascal-scripted setup system's loader stores in a data resource to
53/// find its compressed setup program and data.
54pub const INSTALLER_SETUP_LOADER: &str = "installer:setup-loader";
55/// A bundle bootstrapper that chains installer packages: a PE image with
56/// the bootstrapper's 8-byte section name anywhere a section table can be,
57/// and the bootstrapper's section header (magic `0x00F14300`, version 2)
58/// anywhere in the file. The two are found independently: the rule does
59/// not resolve the named section or check that the header is at its raw
60/// data, so it is a two-marker heuristic.
61pub const INSTALLER_BUNDLE_BOOTSTRAPPER: &str = "installer:bundle-bootstrapper";
62/// A commercial setup-authoring engine's setup stream: its plain or
63/// stream-format signature (each a 13-byte name and a NUL) in a PE
64/// image's appended data.
65pub const INSTALLER_SETUP_STREAM: &str = "installer:setup-stream";
66/// A PE image that carries a cabinet (`MSCF` and a zero reserved field)
67/// after its start, as self-extracting cabinet packages do (often in a
68/// data resource).
69pub const ARCHIVE_SELF_EXTRACTING_CAB: &str = "archive:self-extracting-cab";
70/// A ZIP local file header in a PE image's appended data.
71pub const ARCHIVE_SELF_EXTRACTING_ZIP: &str = "archive:self-extracting-zip";
72/// A 7z signature in a PE image's appended data.
73pub const ARCHIVE_SELF_EXTRACTING_7Z: &str = "archive:self-extracting-7z";
74/// A PE image after the start of any file: `MZ`, then an `e_lfanew` of
75/// `0x40` to `0x400` that points at `PE\0\0`. Droppers, installers,
76/// archives stored without compression and some documents carry one.
77pub const ARCHIVE_EMBEDDED_PE: &str = "archive:embedded-pe";
78/// A cabinet, 7z, RAR (4 or 5) or ZIP local header signature after the
79/// start of a file that does not itself start with a ZIP local file header
80/// (`PK\x03\x04`; a ZIP archive has a local header for every entry). A
81/// file that merely starts with `PK` is not excluded.
82pub const ARCHIVE_EMBEDDED_ARCHIVE: &str = "archive:embedded-archive";
83/// A script compiled into an executable by a desktop automation
84/// language's script-to-executable compiler: its 16-byte script marker,
85/// in the appended data or a data resource of a PE image.
86pub const SCRIPT_COMPILED_AUTOMATION: &str = "script:compiled-automation";
87/// A frozen Python application: the 8-byte archive cookie that a widely
88/// used freezer appends to its bootloader, in the appended data of a PE
89/// image or in the last 64 KiB of any file.
90pub const SCRIPT_FROZEN_PYTHON: &str = "script:frozen-python";
91/// The industry-standard anti-malware test file: its 68-byte string at the
92/// start of a file of at most 128 bytes. It is harmless, and detection
93/// products treat it as a test detection.
94pub const TEST_MALWARE_TEST_FILE: &str = "test:malware-test-file";

96/// The compound file header: signature, a zero CLSID, then a minor
97/// version, a major version below 16 (the format defines 3 and 4), and the
98/// little-endian byte order mark.
99const CFB_HEADER: &str =
100    "D0 CF 11 E0 A1 B1 1A E1 00000000 00000000 00000000 00000000 ?? 00 0? 00 FE FF";
101/// One past the last offset a section header can start at: the largest
102/// `e_lfanew` the parser accepts, the signature and COFF header, the
103/// largest optional header, and the largest section table.
104const BUNDLE_NAME_SEARCH_END: u64 = 0x10_0000 + 4 + 20 + 0xffff + 1024 * 40;
105
106/// The installer database root CLSID `{000C1084-0000-0000-C000-000000000046}`.
107const MSI_CLSID: &str = "84 10 0C 00 00 00 00 00 C0 00 00 00 00 00 00 46";
108const CAB: &str = "4D 53 43 46 00 00 00 00";
109const ZIP_LOCAL: &str = "50 4B 03 04";
110const SEVEN_ZIP: &str = "37 7A BC AF 27 1C";
111const RAR4: &str = "52 61 72 21 1A 07 00";
112const RAR5: &str = "52 61 72 21 1A 07 01 00";

114/// The built-in rules.
115pub(super) fn rules() -> Vec<RuleSource> {
116    vec![
117        rule(
118            INSTALLER_MSI_WRAPPER,
119            true,
120            vec![from(CFB_HEADER, 1), anywhere(MSI_CLSID)],
121            Condition::All,
122        ),
123        rule(
124            INSTALLER_SCRIPTED_SETUP,
125            true,
126            vec![overlay(
127                "?? ?? ?? ?? EF BE AD DE 4E 75 6C 6C 73 6F 66 74 49 6E 73 74",
128            )],
129            Condition::Any,
130        ),
131        rule(
132            INSTALLER_SETUP_LOADER,
133            true,
134            vec![anywhere("72 44 6C 50 74 53 CD E6 D7 7B 0B 2A 0? 00 00 00")],
135            Condition::Any,
136        ),
137        rule(
138            INSTALLER_BUNDLE_BOOTSTRAPPER,
139            true,
140            vec![
141                // Anywhere a section header can be: past the largest
142                // `e_lfanew`, optional header and section table.
143                range("2E 77 69 78 62 75 72 6E", 0, BUNDLE_NAME_SEARCH_END),
144                anywhere("00 43 F1 00 02 00 00 00"),
145            ],
146            Condition::All,
147        ),
148        rule(
149            INSTALLER_SETUP_STREAM,
150            true,
151            vec![
152                overlay("49 6E 73 74 61 6C 6C 53 68 69 65 6C 64 00"),
153                overlay("49 53 53 65 74 75 70 53 74 72 65 61 6D 00"),
154            ],
155            Condition::Any,
156        ),
157        rule(
158            ARCHIVE_SELF_EXTRACTING_CAB,
159            true,
160            vec![from(CAB, 1)],
161            Condition::Any,
162        ),
163        rule(
164            ARCHIVE_SELF_EXTRACTING_ZIP,
165            true,
166            vec![overlay(ZIP_LOCAL)],
167            Condition::Any,
168        ),
169        rule(
170            ARCHIVE_SELF_EXTRACTING_7Z,
171            true,
172            vec![overlay(SEVEN_ZIP)],
173            Condition::Any,
174        ),
175        RuleSource {
176            tag: ARCHIVE_EMBEDDED_PE.to_owned(),
177            require_pe: false,
178            max_file_size: None,
179            unless_start: None,
180            strings: vec![StringSource {
181                pattern: PatternSource::PeImage,
182                location: Location::From(1),
183            }],
184            condition: Condition::Any,
185        },
186        RuleSource {
187            unless_start: Some(PatternSource::Hex(ZIP_LOCAL.to_owned())),
188            ..rule(
189                ARCHIVE_EMBEDDED_ARCHIVE,
190                false,
191                vec![
192                    from(CAB, 1),
193                    from(SEVEN_ZIP, 1),
194                    from(RAR4, 1),
195                    from(RAR5, 1),
196                    from(ZIP_LOCAL, 1),
197                ],
198                Condition::Any,
199            )
200        },
201        rule(
202            SCRIPT_COMPILED_AUTOMATION,
203            true,
204            vec![anywhere("A3 48 4B BE 98 6C 4A A9 99 4C 53 0A 86 D6 48 7D")],
205            Condition::Any,
206        ),
207        rule(
208            SCRIPT_FROZEN_PYTHON,
209            false,
210            vec![
211                overlay("4D 45 49 0C 0B 0A 0B 0E"),
212                StringSource {
213                    pattern: PatternSource::Hex("4D 45 49 0C 0B 0A 0B 0E".to_owned()),
214                    location: Location::Tail(64 * 1024),
215                },
216            ],
217            Condition::Any,
218        ),
219        RuleSource {
220            max_file_size: Some(128),
221            ..rule(
222                TEST_MALWARE_TEST_FILE,
223                false,
224                vec![StringSource {
225                    pattern: PatternSource::Hex(MALWARE_TEST_STRING.to_owned()),
226                    location: Location::At(0),
227                }],
228                Condition::Any,
229            )
230        },
231    ]
232}

234/// The 68-byte anti-malware test string, as hex, so the source file is
235/// never itself taken for the test file.
236pub(super) const MALWARE_TEST_STRING: &str = "\
237    58 35 4F 21 50 25 40 41 50 5B 34 5C 50 5A 58 35 34 28 50 5E 29 37 43 43 \
238    29 37 7D 24 45 49 43 41 52 2D 53 54 41 4E 44 41 52 44 2D 41 4E 54 49 56 \
239    49 52 55 53 2D 54 45 53 54 2D 46 49 4C 45 21 24 48 2B 48 2A";

241fn rule(
242    tag: &str,
243    require_pe: bool,
244    strings: Vec<StringSource>,
245    condition: Condition,
246) -> RuleSource {
247    RuleSource {
248        tag: tag.to_owned(),
249        require_pe,
250        max_file_size: None,
251        unless_start: None,
252        strings,
253        condition,
254    }
255}
256
257fn string(hex: &str, location: Location) -> StringSource {
258    StringSource {
259        pattern: PatternSource::Hex(hex.to_owned()),
260        location,
261    }
262}
263
264fn anywhere(hex: &str) -> StringSource {
265    string(hex, Location::Anywhere)
266}
267
268fn from(hex: &str, start: u64) -> StringSource {
269    string(hex, Location::From(start))
270}
271
272fn range(hex: &str, start: u64, end: u64) -> StringSource {
273    string(hex, Location::Range { start, end })
274}
275
276fn overlay(hex: &str) -> StringSource {
277    string(hex, Location::Overlay)
278}
