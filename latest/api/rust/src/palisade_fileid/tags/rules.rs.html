3//! A [`RuleSetSource`] is plain data (`serde`), so the policy bundle can

18#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]

20pub struct RuleSetSource {

28/// One rule: when its condition holds, the file gets its tag.
29#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
30#[serde(deny_unknown_fields)]
31pub struct RuleSource {
32    /// The tag: two to four segments of lowercase ASCII letters, digits and
33    /// `-`, joined by `:` (`installer:in-house`). Each segment is 1 to 32
34    /// bytes, and the whole tag at most [`MAX_TAG_LEN`] bytes. It must not
35    /// be a built-in tag.
36    pub tag: String,
37    /// The rule applies only to a PE image (one that
38    /// [`filetype::classify`](crate::filetype::classify) would call
39    /// [`FileKind::Pe`](crate::filetype::FileKind::Pe)).
40    #[serde(default)]
41    pub require_pe: bool,
42    /// The rule applies only to files of at most this many bytes.
43    #[serde(default, skip_serializing_if = "Option::is_none")]
44    pub max_file_size: Option<u64>,
45    /// The rule does not apply to a file that starts with this pattern.
46    /// It is checked before anything else is scanned. If it cannot be
47    /// checked (the file ended early, or the budget ran out first), the
48    /// rule is not reported.
49    #[serde(default, skip_serializing_if = "Option::is_none")]
50    pub unless_start: Option<PatternSource>,
51    /// The strings, 1 to [`MAX_STRINGS_PER_RULE`].
52    pub strings: Vec<StringSource>,
53    /// How many of the strings must be found.
54    #[serde(default)]
55    pub condition: Condition,
56}

69/// What a string matches. Every pattern has a fixed length of at most
70/// [`MAX_PATTERN_LEN`](super::MAX_PATTERN_LEN) bytes.
71#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
72#[serde(rename_all = "snake_case")]
73#[non_exhaustive]
74pub enum PatternSource {
75    /// Hex bytes, with `?` for a wildcard nibble and optional whitespace:
76    /// `"4D 5A ?? 0?"`. Unless the string's location is
77    /// [`Location::At`], it needs a run of at least [`MIN_ATOM_LEN`](super::MIN_ATOM_LEN) exact
78    /// bytes, which the searcher looks for.
79    Hex(String),
80    /// The UTF-8 bytes of the text, exactly.
81    Text(String),
82    /// ASCII text as UTF-16LE.
83    WideText(String),
84    /// A PE image: `MZ`, then an `e_lfanew` of `0x40` to `0x400` that
85    /// points at `PE\0\0`.
86    PeImage,
87}
88
89/// Where a string's first byte may be, as a file offset.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
91#[serde(rename_all = "snake_case", deny_unknown_fields)]
92#[non_exhaustive]
93pub enum Location {
94    /// Anywhere in the scanned bytes.
95    #[default]
96    Anywhere,
97    /// Exactly at this offset. Such a string needs no exact-byte run.
98    At(u64),
99    /// At this offset or later.
100    From(u64),
101    /// From `start` up to, but not including, `end`.
102    Range {
103        /// The first offset.
104        start: u64,
105        /// One past the last offset.
106        end: u64,
107    },
108    /// In a PE image's appended data: at or after the end of the headers
109    /// and of every section's raw data. A certificate table, if any, is
110    /// not excluded. Never matches in a file that is not a PE image. A
111    /// crafted image can declare section data or headers that cover its
112    /// appended bytes, so that such a rule does not match them: a missing
113    /// tag is never proof of absence.
114    Overlay,
115    /// In the last this many bytes of the file (1 to [`TAIL_SCAN_LEN`]).
116    Tail(u64),
117}

119/// How many of a rule's strings must be found.
120#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
121#[serde(rename_all = "snake_case")]
122#[non_exhaustive]
123pub enum Condition {
124    /// At least one.
125    #[default]
126    Any,
127    /// Every one.
128    All,
129    /// At least this many (1 to the number of strings).
130    AtLeast(u32),
131}

133/// Why a rule set was rejected.
134#[derive(Debug, Clone, PartialEq, Eq)]
135pub struct RuleError {
136    /// The error is in the built-in rules, not the organisation's (a bug).
137    pub builtin: bool,
138    /// The index of the rule, if the error is in one.
139    pub rule: Option<usize>,
140    /// The index of the string in that rule, if the error is in one.
141    pub string: Option<usize>,
142    /// What is wrong.
143    pub kind: RuleErrorKind,
144}

146/// What is wrong with a rule.
147#[derive(Debug, Clone, Copy, PartialEq, Eq)]
148#[non_exhaustive]
149pub enum RuleErrorKind {
150    /// More than [`MAX_RULES`] rules, or more than [`MAX_STRINGS`] strings
151    /// over all rules.
152    TooManyRules,
153    /// A rule has no strings.
154    NoStrings,
155    /// A rule has more than [`MAX_STRINGS_PER_RULE`] strings.
156    TooManyStrings,
157    /// The tag is not two to four `[a-z0-9-]` segments of 1 to 32 bytes
158    /// joined by `:`, or is longer than [`MAX_TAG_LEN`].
159    BadTag,
160    /// The tag is a built-in tag; the built-in set is read-only.
161    ReservedTag,
162    /// Two rules have the same tag.
163    DuplicateTag,
164    /// Hex text that is not pairs of hex digits or `?`.
165    BadHex,
166    /// A pattern of no bytes.
167    EmptyPattern,
168    /// A pattern longer than [`MAX_PATTERN_LEN`](super::MAX_PATTERN_LEN) bytes.
169    PatternTooLong,
170    /// Wide text that is not ASCII.
171    WideNotAscii,
172    /// A string that is not at a fixed offset has no run of
173    /// [`MIN_ATOM_LEN`](super::MIN_ATOM_LEN) exact bytes.
174    NoAtom,
175    /// [`Condition::AtLeast`] of zero or of more than the strings.
176    BadCondition,
177    /// An empty [`Location::Range`], or a [`Location::Tail`] of zero or
178    /// more than [`TAIL_SCAN_LEN`].
179    BadLocation,
180    /// The search automaton could not be built.
181    TooLarge,
182}

184impl fmt::Display for RuleError {
185    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
186        if self.builtin {
187            f.write_str("built-in ")?;
188        }
189        f.write_str("content rule")?;
190        if let Some(rule) = self.rule {
191            write!(f, " {rule}")?;
192        }
193        if let Some(string) = self.string {
194            write!(f, ", string {string}")?;
195        }
196        write!(f, ": {:?}", self.kind)
197    }
198}

202/// What a compiled string matches.
203#[derive(Debug, Clone, PartialEq, Eq)]
204pub(super) enum Matcher {
205    Bytes(Pattern),
206    PeImage,
207}
208
209/// The lowest and highest `e_lfanew` a [`PatternSource::PeImage`] accepts.
210const PE_IMAGE_LFANEW: std::ops::RangeInclusive<usize> = 0x40..=0x400;
211
212impl Matcher {
213    /// The most bytes a match can cover.
214    pub(super) fn span(&self) -> usize {
215        match self {
216            Self::Bytes(p) => p.len(),
217            Self::PeImage => *PE_IMAGE_LFANEW.end() + 4,
218        }
219    }
220
221    /// The most comparisons one check costs.
222    pub(super) fn cost(&self) -> u64 {
223        match self {
224            Self::Bytes(p) => u64::try_from(p.len()).unwrap_or(u64::MAX),
225            Self::PeImage => 8,
226        }
227    }
228
229    /// Whether a match starts at `at`. Bytes past the end of `buf` never
230    /// match.
231    pub(super) fn matches_at(&self, buf: &[u8], at: usize) -> bool {
232        let Some(rest) = buf.get(at..) else {
233            return false;
234        };
235        match self {
236            Self::Bytes(p) => p.matches(rest),
237            Self::PeImage => {
238                let lfanew = rest
239                    .get(0x3c..0x40)
240                    .and_then(|b| b.try_into().ok())
241                    .map(u32::from_le_bytes)
242                    .and_then(|v| usize::try_from(v).ok());
243                rest.starts_with(b"MZ")
244                    && lfanew.is_some_and(|e| {
245                        PE_IMAGE_LFANEW.contains(&e) && rest.get(e..e + 4) == Some(&b"PE\0\0"[..])
246                    })
247            }
248        }
249    }
250
251    /// The exact bytes the searcher looks for, and their offset in the
252    /// pattern.
253    fn atom(&self) -> Option<(usize, Vec<u8>)> {
254        match self {
255            Self::Bytes(p) => p.atom().map(|(at, bytes)| (at, bytes.to_vec())),
256            Self::PeImage => Some((0, b"MZ".to_vec())),
257        }
258    }
259}

310/// Compiles the built-in rules followed by the organisation's, checking
311/// every limit.
312pub(super) fn compile(
313    builtin: &[RuleSource],
314    org: Option<&RuleSetSource>,
315) -> Result<Compiled, RuleError> {
316    let org_rules = org.map_or(&[][..], |o| &o.rules[..]);
317    if org_rules.len() > MAX_RULES {
318        return Err(RuleError {
319            builtin: false,
320            rule: None,
321            string: None,
322            kind: RuleErrorKind::TooManyRules,
323        });
324    }
325    let mut out = Compiled {
326        strings: Vec::new(),
327        rules: Vec::new(),
328        searcher: None,
329        atom_users: Vec::new(),
330        anchored: Vec::new(),
331        window: 1,
332        prefixes: Vec::new(),
333        prefix_len: 0,
334    };
335    let mut atoms: Vec<Vec<u8>> = Vec::new();
336    let mut org_strings = 0_usize;
337    let sources = builtin
338        .iter()
339        .enumerate()
340        .map(|(i, r)| (true, i, r))
341        .chain(org_rules.iter().enumerate().map(|(i, r)| (false, i, r)));
342    for (is_builtin, index, rule) in sources {
343        let err = |string, kind| RuleError {
344            builtin: is_builtin,
345            rule: Some(index),
346            string,
347            kind,
348        };
349        if !is_builtin {
350            org_strings = org_strings.saturating_add(rule.strings.len());
351            if org_strings > MAX_STRINGS {
352                return Err(err(None, RuleErrorKind::TooManyRules));
353            }
354        }
355        check_tag(&rule.tag).map_err(|kind| err(None, kind))?;
356        if out.rules.iter().any(|r| r.tag == rule.tag) {
357            let builtin_tag = builtin.iter().any(|b| b.tag == rule.tag);
358            let kind = if builtin_tag && !is_builtin {
359                RuleErrorKind::ReservedTag
360            } else {
361                RuleErrorKind::DuplicateTag
362            };
363            return Err(err(None, kind));
364        }
365        if !is_builtin && builtin.iter().any(|b| b.tag == rule.tag) {
366            return Err(err(None, RuleErrorKind::ReservedTag));
367        }
368        if rule.strings.is_empty() {
369            return Err(err(None, RuleErrorKind::NoStrings));
370        }
371        if rule.strings.len() > MAX_STRINGS_PER_RULE {
372            return Err(err(None, RuleErrorKind::TooManyStrings));
373        }
374        let need = match rule.condition {
375            Condition::Any => 1,
376            Condition::All => rule.strings.len(),
377            Condition::AtLeast(n) => usize::try_from(n)
378                .ok()
379                .filter(|n| (1..=rule.strings.len()).contains(n))
380                .ok_or_else(|| err(None, RuleErrorKind::BadCondition))?,
381        };
382        let first = out.strings.len();
383        for (si, s) in rule.strings.iter().enumerate() {
384            let matcher = compile_pattern(&s.pattern).map_err(|kind| err(Some(si), kind))?;
385            check_location(s.location).map_err(|kind| err(Some(si), kind))?;
386            add_string(&mut out, &mut atoms, matcher, s.location)
387                .map_err(|kind| err(Some(si), kind))?;
388        }
389        let strings = first..out.strings.len();
390        let unless = match &rule.unless_start {
391            Some(p) => {
392                let matcher = compile_pattern(p).map_err(|kind| err(None, kind))?;
393                out.prefix_len = out.prefix_len.max(matcher.span());
394                out.prefixes.push(matcher);
395                Some(out.prefixes.len() - 1)
396            }
397            None => None,
398        };
399        out.rules.push(CompiledRule {
400            tag: rule.tag.clone(),
401            require_pe: rule.require_pe,
402            max_file_size: rule.max_file_size,
403            unless,
404            strings,
405            need,
406        });
407    }
408    out.anchored.sort_by_key(|a| (a.start, a.string));
409    if !atoms.is_empty() {
410        let searcher = aho_corasick::AhoCorasick::builder()
411            .match_kind(aho_corasick::MatchKind::Standard)
412            .build(&atoms)
413            .map_err(|_| RuleError {
414                builtin: false,
415                rule: None,
416                string: None,
417                kind: RuleErrorKind::TooLarge,
418            })?;
419        out.searcher = Some(searcher);
420    }
421    Ok(out)
422}

424/// Adds one compiled string: to the anchored list if it is at a fixed
425/// offset, otherwise under its atom.
426fn add_string(
427    out: &mut Compiled,
428    atoms: &mut Vec<Vec<u8>>,
429    matcher: Matcher,
430    location: Location,
431) -> Result<(), RuleErrorKind> {
432    let index = out.strings.len();
433    let span = matcher.span();
434    out.window = out.window.max(span);
435    if let Location::At(start) = location {
436        out.anchored.push(Anchored {
437            start,
438            string: index,
439        });
440    } else {
441        let (offset, atom) = matcher.atom().ok_or(RuleErrorKind::NoAtom)?;
442        let id = match atoms.iter().position(|a| *a == atom) {
443            Some(id) => id,
444            None => {
445                atoms.push(atom);
446                out.atom_users.push(Vec::new());
447                atoms.len() - 1
448            }
449        };
450        if let Some(users) = out.atom_users.get_mut(id) {
451            users.push((index, offset));
452        }
453    }
454    out.strings.push(CompiledString { matcher, location });
455    Ok(())
456}

458fn compile_pattern(p: &PatternSource) -> Result<Matcher, RuleErrorKind> {
459    Ok(match p {
460        PatternSource::Hex(text) => Matcher::Bytes(Pattern::from_hex(text)?),
461        PatternSource::Text(text) => Matcher::Bytes(Pattern::from_text(text, false)?),
462        PatternSource::WideText(text) => Matcher::Bytes(Pattern::from_text(text, true)?),
463        PatternSource::PeImage => Matcher::PeImage,
464    })
465}

467fn check_location(location: Location) -> Result<(), RuleErrorKind> {
468    match location {
469        Location::Range { start, end } if start >= end => Err(RuleErrorKind::BadLocation),
470        Location::Tail(n) if n == 0 || n > TAIL_SCAN_LEN => Err(RuleErrorKind::BadLocation),
471        _ => Ok(()),
472    }
473}

475/// Two to four segments of `[a-z0-9-]`, each 1 to 32 bytes, joined by `:`.
476fn check_tag(tag: &str) -> Result<(), RuleErrorKind> {
477    if tag.len() > MAX_TAG_LEN {
478        return Err(RuleErrorKind::BadTag);
479    }
480    let segments: Vec<&str> = tag.split(':').collect();
481    let ok = tag.len() <= MAX_TAG_LEN
482        && (2..=4).contains(&segments.len())
483        && segments.iter().all(|s| {
484            (1..=32).contains(&s.len())
485                && s.bytes()
486                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
487        });
488    if ok {
489        Ok(())
490    } else {
491        Err(RuleErrorKind::BadTag)
492    }
493}

495/// SHA-256 over a canonical encoding of a rule set that compiled: the
496/// same rules give the same digest however their hex is spaced or cased,
497/// and any change to what they match gives a different one.
498pub(super) fn digest(set: &RuleSetSource) -> [u8; 32] {
499    let mut h = Sha256::new();
500    h.update(b"palisade-content-rules-v1\0");
501    h.update(set.version.to_le_bytes());
502    len(&mut h, set.rules.len());
503    for rule in &set.rules {
504        len(&mut h, rule.tag.len());
505        h.update(rule.tag.as_bytes());
506        h.update([u8::from(rule.require_pe)]);
507        opt_u64(&mut h, rule.max_file_size);
508        match &rule.unless_start {
509            Some(p) => {
510                h.update([1]);
511                pattern(&mut h, p);
512            }
513            None => h.update([0]),
514        }
515        len(&mut h, rule.strings.len());
516        for s in &rule.strings {
517            pattern(&mut h, &s.pattern);
518            match s.location {
519                Location::Anywhere => h.update([0]),
520                Location::At(n) => tagged(&mut h, 1, &[n]),
521                Location::From(n) => tagged(&mut h, 2, &[n]),
522                Location::Range { start, end } => tagged(&mut h, 3, &[start, end]),
523                Location::Overlay => h.update([4]),
524                Location::Tail(n) => tagged(&mut h, 5, &[n]),
525            }
526        }
527        match rule.condition {
528            Condition::Any => h.update([0]),
529            Condition::All => h.update([1]),
530            Condition::AtLeast(n) => tagged(&mut h, 2, &[u64::from(n)]),
531        }
532    }
533    h.finalize().into()
534}
535
536fn len(h: &mut Sha256, n: usize) {
537    h.update(u64::try_from(n).unwrap_or(u64::MAX).to_le_bytes());
538}
539
540fn opt_u64(h: &mut Sha256, v: Option<u64>) {
541    match v {
542        Some(v) => tagged(h, 1, &[v]),
543        None => h.update([0]),
544    }
545}
546
547fn tagged(h: &mut Sha256, tag: u8, values: &[u64]) {
548    h.update([tag]);
549    for v in values {
550        h.update(v.to_le_bytes());
551    }
552}
553
554/// A pattern by what it matches: the value and mask bytes.
555fn pattern(h: &mut Sha256, p: &PatternSource) {
556    match compile_pattern(p) {
557        Ok(Matcher::Bytes(p)) => {
558            h.update([0]);
559            p.encode(h);
560        }
561        Ok(Matcher::PeImage) => h.update([1]),
562        // Only rule sets that compiled are digested.
563        Err(_) => h.update([0xff]),
564    }
565}

726    #[test]
727    fn digest_is_canonical() {
728        let a = org(vec![rule("a:b", vec![hex("4d5a 90")])]);
729        let b = org(vec![rule("a:b", vec![hex("4D 5A 90")])]);
730        let text = org(vec![rule(
731            "a:b",
732            vec![StringSource {
733                pattern: PatternSource::Text("MZ\u{90}".to_owned()),
734                location: Location::Anywhere,
735            }],
736        )]);
737        assert_eq!(digest(&a), digest(&b));
738        // "\u{90}" is two bytes in UTF-8, so this is a different pattern.
739        assert_ne!(digest(&a), digest(&text));
740        let mut changed = a.clone();
741        changed.version = 8;
742        assert_ne!(digest(&a), digest(&changed));
743        let mut changed = a.clone();
744        changed.rules[0].strings[0].location = Location::From(0);
745        assert_ne!(digest(&a), digest(&changed));
746        let mut changed = a.clone();
747        changed.rules[0].require_pe = true;
748        assert_ne!(digest(&a), digest(&changed));
749        let mut changed = a.clone();
750        changed.rules[0].condition = Condition::All;
751        assert_ne!(digest(&a), digest(&changed));
752    }
