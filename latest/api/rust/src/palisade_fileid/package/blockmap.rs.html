1//! `AppxBlockMap.xml`: the per-file block hashes every package carries, and
2//! their check against the archive.
3//!
4//! The block map lists every file of the package (except the signature,
5//! the block map itself, `[Content_Types].xml` and the code integrity
6//! catalog) with its uncompressed size, the size of its local file header
7//! and the hash of each 64 KiB block of its uncompressed content, under the
8//! map's `HashMethod`. The signature signs the block map's hash, so the
9//! map is what ties every file's content to the signature.
10//!
11//! A bundle's block map lists only its bundle manifest; [`verify_bundle`]
12//! checks its payload packages against the bundle manifest instead (the
13//! signature still covers their bytes, through the local records).
14//!
15//! [`verify`] checks the map against an [`Archive`] and fails closed: a
16//! file the archive has but the map does not list (or the reverse), a name
17//! listed twice, a size, local-header size or compressed size that does not
18//! match, a deflated file whose blocks do not all give their compressed
19//! size, a block count that does not fit the size, or any block whose hash
20//! differs is a mismatch.
21

69/// Why the block map does not match, or could not be read.
70#[derive(Debug, Clone, PartialEq, Eq, Hash)]
71#[non_exhaustive]
72pub enum BlockMapError {
73    /// The XML is invalid or not a block map.
74    Malformed(String),
75    /// `HashMethod` is not SHA-256, SHA-384 or SHA-512.
76    HashMethod,
77    /// A file is listed twice.
78    Duplicate(String),
79    /// An archive member is not listed, or a listed file is missing.
80    Unlisted(String),
81    /// A size or the local header size does not match.
82    Size(String),
83    /// A block hash does not match.
84    Hash(String),
85}

87impl fmt::Display for BlockMapError {
88    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
89        match self {
90            Self::Malformed(why) => write!(f, "block map: {why}"),
91            Self::HashMethod => f.write_str("block map hash method is not SHA-2"),
92            Self::Duplicate(n) => write!(f, "{n:?} is listed twice in the block map"),
93            Self::Unlisted(n) => write!(f, "{n:?} is in the package or the block map, not both"),
94            Self::Size(n) => write!(f, "{n:?} does not have the size the block map gives"),
95            Self::Hash(n) => write!(f, "{n:?} does not match its block hashes"),
96        }
97    }
98}

106/// Standard base64 with padding, strict.
107fn base64(text: &str) -> Option<Vec<u8>> {
108    let bytes = text.as_bytes();
109    if bytes.is_empty() || !bytes.len().is_multiple_of(4) {
110        return None;
111    }
112    let val = |c: u8| -> Option<u32> {
113        Some(u32::from(match c {
114            b'A'..=b'Z' => c - b'A',
115            b'a'..=b'z' => c - b'a' + 26,
116            b'0'..=b'9' => c - b'0' + 52,
117            b'+' => 62,
118            b'/' => 63,
119            _ => return None,
120        }))
121    };
122    let mut out = Vec::with_capacity(bytes.len() / 4 * 3);
123    let chunks = bytes.chunks(4);
124    let n = chunks.len();
125    for (i, chunk) in chunks.enumerate() {
126        let pad = chunk.iter().rev().take_while(|&&c| c == b'=').count();
127        if pad > 2 || (pad > 0 && i + 1 != n) {
128            return None;
129        }
130        let mut v = 0_u32;
131        for &c in chunk.get(..4 - pad)? {
132            v = (v << 6) | val(c)?;
133        }
134        v <<= 6 * u32::try_from(pad).ok()?;
135        let b = v.to_be_bytes();
136        out.extend_from_slice(b.get(1..4 - pad)?);
137        // Non-canonical padding bits are refused.
138        if pad > 0 {
139            let mask = if pad == 2 { 0xffff } else { 0xff };
140            if v & mask != 0 {
141                return None;
142            }
143        }
144    }
145    Some(out)
146}

148fn hash_method(uri: &str) -> Option<DigestAlgorithm> {
149    Some(match uri {
150        "http://www.w3.org/2001/04/xmlenc#sha256" => DigestAlgorithm::Sha256,
151        "http://www.w3.org/2001/04/xmldsig-more#sha384" => DigestAlgorithm::Sha384,
152        "http://www.w3.org/2001/04/xmlenc#sha512" => DigestAlgorithm::Sha512,
153        _ => return None,
154    })
155}

157fn number(e: &Element, name: &str) -> Result<Option<u64>, BlockMapError> {
158    e.attr(name)
159        .map(|v| {
160            if v.is_empty() || !v.bytes().all(|b| b.is_ascii_digit()) {
161                return Err(bad("a size is not a decimal number"));
162            }
163            v.parse().map_err(|_| bad("a size is too large"))
164        })
165        .transpose()
166}

168/// Parses `AppxBlockMap.xml`.
169pub fn parse(data: &[u8]) -> Result<BlockMap, BlockMapError> {
170    let root = xml::parse(data).map_err(|e| BlockMapError::Malformed(e.0))?;
171    if &*root.ns != NS || root.name != "BlockMap" {
172        return Err(bad("not a block map"));
173    }
174    let hash = root
175        .attr("HashMethod")
176        .and_then(hash_method)
177        .ok_or(BlockMapError::HashMethod)?;
178    let mut files = Vec::new();
179    for f in root.children_named(&[NS], "File") {
180        if files.len() >= MAX_ENTRIES {
181            return Err(bad("too many files"));
182        }
183        let name = f
184            .attr("Name")
185            .filter(|n| !n.is_empty())
186            .ok_or_else(|| bad("a file has no name"))?
187            .replace('\\', "/");
188        if name.starts_with('/')
189            || name.contains('\0')
190            || name
191                .split('/')
192                .any(|seg| seg.is_empty() || seg == "." || seg == "..")
193        {
194            return Err(bad("a file name is not a valid part name"));
195        }
196        let size = number(f, "Size")?.ok_or_else(|| bad("a file has no size"))?;
197        let lfh_size = number(f, "LfhSize")?.ok_or_else(|| bad("a file has no LfhSize"))?;
198        let mut blocks = Vec::new();
199        let mut block_sizes = Vec::new();
200        for b in f.children_named(&[NS], "Block") {
201            let h = b
202                .attr("Hash")
203                .and_then(base64)
204                .filter(|h| h.len() == hash.digest_len())
205                .ok_or_else(|| bad("a block hash is not valid base64 of the right length"))?;
206            blocks.push(h);
207            let size = number(b, "Size")?;
208            if size == Some(0) {
209                return Err(bad("a block size is zero"));
210            }
211            block_sizes.push(size);
212        }
213        if u64::try_from(blocks.len()).ok() != Some(size.div_ceil(BLOCK)) {
214            return Err(BlockMapError::Size(name));
215        }
216        // Either every block gives its compressed size or none does.
217        let with = block_sizes.iter().filter(|s| s.is_some()).count();
218        if with != 0 && with != block_sizes.len() {
219            return Err(BlockMapError::Size(name));
220        }
221        files.push(MappedFile {
222            name,
223            size,
224            lfh_size,
225            blocks,
226            block_sizes,
227        });
228    }
229    // The schema requires at least one `File`: every package lists its
230    // manifest, every bundle its bundle manifest.
231    if files.is_empty() {
232        return Err(bad("no files"));
233    }
234    Ok(BlockMap { hash, files })
235}

258/// Checks a bundle's `map` against `archive`. A bundle's block map lists
259/// its bundle manifest and nothing else: the payload packages are not in
260/// it. Each payload must instead be listed in `payloads` (the bundle
261/// manifest) exactly once, and be a stored member (the OS packaging tool
262/// never compresses one) whose data starts at the manifest's `Offset` and
263/// has its `Size`, so no member escapes both. Their bytes are covered by
264/// the signature through the local records. A payload that is not in the
265/// archive (a flat bundle's) is a mismatch.
266pub fn verify_bundle(
267    map: &BlockMap,
268    archive: &Archive<'_>,
269    payloads: &[Payload<'_>],
270) -> Result<(), BlockMapError> {
271    verify_with(map, archive, Some(payloads))
272}
273
274fn verify_with(
275    map: &BlockMap,
276    archive: &Archive<'_>,
277    payloads: Option<&[Payload<'_>]>,
278) -> Result<(), BlockMapError> {
279    // Only the block map's own hash methods (SHA-256, SHA-384, SHA-512), for
280    // a map built by a caller as for one `parse` read.
281    if !matches!(
282        map.hash,
283        DigestAlgorithm::Sha256 | DigestAlgorithm::Sha384 | DigestAlgorithm::Sha512
284    ) {
285        return Err(BlockMapError::HashMethod);
286    }
287    // A bundle's payloads, by folded part name; each once.
288    let mut by_part: BTreeMap<String, Payload<'_>> = BTreeMap::new();
289    for p in payloads.unwrap_or_default() {
290        if by_part.insert(fold(p.file_name), *p).is_some() {
291            return Err(BlockMapError::Duplicate(p.file_name.to_owned()));
292        }
293    }
294    let mut payloads_seen = BTreeSet::new();
295    // A bundle's block map lists its bundle manifest and nothing else (the
296    // OS packaging tool writes it so, and the OS reader requires it).
297    if payloads.is_some()
298        && !(map.files.len() == 1
299            && map
300                .files
301                .first()
302                .is_some_and(|f| fold(&f.name) == fold(super::BUNDLE_MANIFEST)))
303    {
304        let name = map
305            .files
306            .iter()
307            .find(|f| fold(&f.name) != fold(super::BUNDLE_MANIFEST))
308            .map_or_else(|| super::BUNDLE_MANIFEST.to_owned(), |f| f.name.clone());
309        return Err(BlockMapError::Unlisted(name));
310    }
311    // Listed files by folded name; each name once.
312    let mut listed: BTreeMap<String, &MappedFile> = BTreeMap::new();
313    for f in &map.files {
314        if listed.insert(fold(&f.name), f).is_some() {
315            return Err(BlockMapError::Duplicate(f.name.clone()));
316        }
317    }
318    // A budget first: the whole uncompressed size the block map lists may
319    // not exceed what a package may be at all, whatever the archive
320    // claims, so a small archive cannot make us inflate more.
321    let total = listed
322        .values()
323        .try_fold(0_u64, |a, f| a.checked_add(f.size))
324        .filter(|&t| t <= super::MAX_PACKAGE_SIZE);
325    if total.is_none() {
326        return Err(BlockMapError::Size("(total)".into()));
327    }
328    // Every member is listed (by its part name), or is one of the four
329    // that are not; and every listed file is exactly one member (part
330    // names are unique in the archive).
331    let mut seen = BTreeSet::new();
332    for m in &archive.members {
333        let key = fold(&m.part);
334        if UNMAPPED.iter().any(|u| fold(u) == key) {
335            continue;
336        }
337        if let Some(p) = by_part.get(&key) {
338            // A payload package: stored, and where and how large the
339            // bundle manifest says. The block map must not list it too.
340            if m.method != 0
341                || p.size != Some(m.size)
342                || p.offset != Some(m.data_offset)
343                || listed.contains_key(&key)
344            {
345                return Err(BlockMapError::Size(m.name.clone()));
346            }
347            payloads_seen.insert(key);
348            continue;
349        }
350        let Some(f) = listed.get(&key) else {
351            return Err(BlockMapError::Unlisted(m.name.clone()));
352        };
353        if !seen.insert(key) {
354            return Err(BlockMapError::Duplicate(f.name.clone()));
355        }
356        if f.size != m.size || f.lfh_size != local_header_len(m) {
357            return Err(BlockMapError::Size(f.name.clone()));
358        }
359        // A deflated file's blocks each give their compressed size (an
360        // empty file has no blocks); a stored file's give none. The OS
361        // packaging tool compresses each 64 KiB block with a full flush and
362        // ends the stream with an empty final block, the two bytes `03 00`
363        // that no block counts, so the sizes sum to the compressed size or
364        // to two bytes less: the two cases the OS reader accepts, and
365        // nothing else.
366        let given: Vec<u64> = f.block_sizes.iter().flatten().copied().collect();
367        if m.method == 8 {
368            let sum = given.iter().try_fold(0_u64, |a, &x| a.checked_add(x));
369            let fits = sum.is_some_and(|s| {
370                s == m.compressed_size || s.checked_add(2) == Some(m.compressed_size)
371            });
372            if !f.blocks.is_empty() && (given.len() != f.blocks.len() || !fits) {
373                return Err(BlockMapError::Size(f.name.clone()));
374            }
375        } else if !given.is_empty() {
376            return Err(BlockMapError::Size(f.name.clone()));
377        }
378    }
379    if let Some(missing) = by_part.keys().find(|k| !payloads_seen.contains(*k)) {
380        let name = by_part
381            .get(missing)
382            .map_or_else(String::new, |p| p.file_name.to_owned());
383        return Err(BlockMapError::Unlisted(name));
384    }
385    if let Some(missing) = listed.keys().find(|k| !seen.contains(*k)) {
386        let name = listed
387            .get(missing)
388            .map_or_else(String::new, |f| f.name.clone());
389        return Err(BlockMapError::Unlisted(name));
390    }
391    // Only then the content, one member at a time.
392    for m in &archive.members {
393        if let Some(f) = listed.get(&fold(&m.part))
394            && !UNMAPPED.iter().any(|u| fold(u) == fold(&m.part))
395        {
396            check_blocks(map.hash, f, archive, m)?;
397        }
398    }
399    Ok(())
400}

402fn check_blocks(
403    alg: DigestAlgorithm,
404    f: &MappedFile,
405    archive: &Archive<'_>,
406    m: &Member,
407) -> Result<(), BlockMapError> {
408    let Some(h) = alg.hash_algorithm() else {
409        return Err(BlockMapError::HashMethod);
410    };
411    let block = usize::try_from(BLOCK).unwrap_or(usize::MAX);
412    if m.method == 8 && !f.block_sizes.is_empty() {
413        check_block_sizes(f, archive, m)?;
414    }
415    let mut pending: Vec<u8> = Vec::with_capacity(block);
416    let mut index = 0_usize;
417    let mut ok = true;
418    let mut finish_block = |data: &[u8], index: &mut usize| {
419        let mut hasher = MultiHasher::new(AlgorithmSet::from(h));
420        hasher.update(data);
421        let digest = hasher.finalize();
422        let matches = digest
423            .get(h)
424            .zip(f.blocks.get(*index))
425            .is_some_and(|(d, want)| d.as_bytes() == want.as_slice());
426        ok &= matches;
427        *index += 1;
428    };
429    archive
430        .stream(m, |mut chunk| {
431            while !chunk.is_empty() {
432                let take = (block - pending.len()).min(chunk.len());
433                let (head, tail) = chunk.split_at(take);
434                pending.extend_from_slice(head);
435                chunk = tail;
436                if pending.len() == block {
437                    finish_block(&pending, &mut index);
438                    pending.clear();
439                }
440            }
441        })
442        .map_err(|_| BlockMapError::Hash(f.name.clone()))?;
443    if !pending.is_empty() {
444        finish_block(&pending, &mut index);
445    }
446    if ok && index == f.blocks.len() {
447        Ok(())
448    } else {
449        Err(BlockMapError::Hash(f.name.clone()))
450    }
451}

453/// A deflated file's block sizes, block by block: block `i`'s compressed
454/// bytes (its `Size`, from where block `i - 1` ended) must be a run of
455/// whole DEFLATE blocks, ended by the full flush the OS packaging tool
456/// writes after every block (an empty stored block, `00 00 ff ff`), that
457/// inflates to exactly its uncompressed block (64 KiB, or the rest of the
458/// file for the last). Only the sum is not enough: sizes that add up but
459/// cut the stream elsewhere describe blocks that do not exist.
460///
461/// Where a segment ends is confirmed by the decoder, not by its last
462/// bytes: the segment is inflated on its own with the empty final block
463/// (`03 00`) appended, and must end the stream exactly there. A final
464/// block header is only read at a block boundary, so a segment cut in the
465/// middle of a block (whatever bytes it happens to end with) does not end
466/// the stream there. What follows the last block is the empty final block
467/// or nothing, when the last size counts it; `verify_with` has checked the
468/// sum allows exactly that. Each block's content is hashed separately, in
469/// [`check_blocks`].
470fn check_block_sizes(
471    f: &MappedFile,
472    archive: &Archive<'_>,
473    m: &Member,
474) -> Result<(), BlockMapError> {
475    const FINAL: [u8; 2] = [0x03, 0x00];
476    const FLUSH: [u8; 4] = [0x00, 0x00, 0xff, 0xff];
477    let bad = || BlockMapError::Size(f.name.clone());
478    let raw = archive.raw_data(m);
479    let mut at = 0_usize;
480    let mut left = m.size;
481    let count = f.block_sizes.len();
482    let mut segment_buf = Vec::new();
483    for (i, size) in f.block_sizes.iter().enumerate() {
484        let size = size.and_then(|s| usize::try_from(s).ok()).ok_or_else(bad)?;
485        let end = at.checked_add(size).ok_or_else(bad)?;
486        let segment = raw.get(at..end).ok_or_else(bad)?;
487        let want = left.min(BLOCK);
488        // The last block may also carry the empty final block itself.
489        let closes = i + 1 == count && end == raw.len() && segment.ends_with(&FINAL);
490        let body = if closes {
491            segment.get(..segment.len() - FINAL.len()).ok_or_else(bad)?
492        } else {
493            segment
494        };
495        // Every block the tool writes ends on its full flush.
496        if !body.ends_with(&FLUSH) {
497            return Err(bad());
498        }
499        segment_buf.clear();
500        segment_buf.extend_from_slice(body);
501        segment_buf.extend_from_slice(&FINAL);
502        let mut out = 0_u64;
503        super::zip::inflate_raw(&segment_buf, want, |c| {
504            out = out.saturating_add(c.len() as u64);
505        })
506        .ok_or_else(bad)?;
507        if out != want {
508            return Err(bad());
509        }
510        left -= want;
511        at = end;
512    }
513    // After the last block: nothing, or exactly the empty final block.
514    match raw.get(at..) {
515        Some([]) | Some([0x03, 0x00]) if left == 0 => Ok(()),
516        _ => Err(bad()),
517    }
518}
