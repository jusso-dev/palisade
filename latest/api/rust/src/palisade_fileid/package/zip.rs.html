1//! A strict, bounded ZIP reader for app packages (OPC containers).
2//!
3//! App packages are ZIP files written by packaging tools in one
4//! predictable shape, so this reader accepts only that shape and rejects
5//! everything ZIP readers disagree about. A package that reads here reads
6//! the same in every reader, so its signature and identity cannot depend
7//! on which reader looked at it:
8//!
9//! - exactly one end-of-central-directory record, found at the very end
10//!   (an archive comment is not allowed), and, when the archive needs
11//!   ZIP64, exactly one ZIP64 locator and record right before it, with
12//!   counts, sizes and offsets that agree with the classic record;
13//! - one disk only (with ZIP64, the classic record's two disk numbers may
14//!   both be the `0xffff` marker, as the OS packaging tool writes them; the
15//!   ZIP64 record's must still be 0);
16//! - the central directory runs from its stated offset to the ZIP64
17//!   record or the end record, with no gap and nothing after it;
18//! - local entries tile the file from offset 0 to the central directory
19//!   in central-directory order: each local header, its data and its data
20//!   descriptor end where the next local header starts;
21//! - a data descriptor's sizes are 8 bytes when the entry's local header
22//!   gives ZIP version 4.5 or later, and 4 bytes otherwise (the OS
23//!   packaging tool writes 4.5 and 8-byte sizes; the OS signing tool,
24//!   rewriting an archive, writes 2.0 and 4-byte sizes, in ZIP64 archives
25//!   too); an entry with a ZIP64 extra field must give 4.5;
26//! - each local header agrees with its central entry (name, method,
27//!   flags, CRC and sizes, or zero for those three when a data descriptor
28//!   follows, which must carry its signature and whose values must then
29//!   agree);
30//! - a central entry's extra data, if any, is exactly one ZIP64 block with
31//!   exactly the fields its `0xffffffff` markers call for (the package
32//!   digest writes entries that way, so anything else would be read
33//!   differently from how it is signed);
34//! - names, and their percent-decoded OPC part names, are valid UTF-8, not
35//!   empty, without `\`, a leading `/`, empty, `.` or `..` segments, or
36//!   NUL; and part names are unique compared ASCII-case-insensitively, as
37//!   OPC compares them, so two spellings of one part are a duplicate;
38//! - extra data, in a local header or a central entry, is either empty or
39//!   exactly one ZIP64 block with exactly the fields its `0xffffffff`
40//!   markers call for (none with a data descriptor);
41//! - method 0 (stored, sizes equal) or 8 (deflate); no encryption, no
42//!   patch data, no strong encryption flags;
43//! - every member's data inflates to exactly its declared size and CRC:
44//!   checked when it is read ([`Archive::read`], [`Archive::stream`]), not
45//!   when the archive is opened, so opening an archive does not vouch for
46//!   members nobody reads (the package signature check reads them all).
47//!
48//! Member reads are bounded by [`MAX_MEMBER_SIZE`], and the whole archive
49//! by [`MAX_ENTRIES`]. Inflation is streamed in fixed chunks with a size
50//! cap, so a small member cannot claim gigabytes.
51
52#![forbid(unsafe_code)]
53

75/// Why an archive is not one this reader accepts.
76#[derive(Debug, Clone, PartialEq, Eq, Hash)]
77#[non_exhaustive]
78pub enum ZipError {
79    /// No end-of-central-directory record at the end of the file, or a
80    /// comment after it, or more than one.
81    EndRecord,
82    /// The ZIP64 records are missing where needed, present where not,
83    /// or disagree with the classic record.
84    Zip64,
85    /// More than one disk.
86    MultiDisk,
87    /// The central directory is malformed or not where it says.
88    CentralDirectory,
89    /// Too many entries.
90    TooManyEntries,
91    /// A local header is malformed, disagrees with its central entry, or
92    /// is not where the tiling requires.
93    LocalHeader(String),
94    /// A name is not acceptable (see the module documentation).
95    BadName(String),
96    /// Two entries have the same name.
97    DuplicateName(String),
98    /// An unsupported method or flag (encryption, patch data).
99    Unsupported(String),
100    /// A member's data does not inflate to its declared size and CRC.
101    Corrupt(String),
102    /// A member is larger than [`MAX_MEMBER_SIZE`].
103    TooLarge(String),
104}
105
106impl fmt::Display for ZipError {
107    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
108        match self {
109            Self::EndRecord => f.write_str("no single end-of-central-directory record at the end"),
110            Self::Zip64 => f.write_str("ZIP64 records missing, unexpected or inconsistent"),
111            Self::MultiDisk => f.write_str("multi-disk archives are not supported"),
112            Self::CentralDirectory => f.write_str("malformed central directory"),
113            Self::TooManyEntries => write!(f, "more than {MAX_ENTRIES} entries"),
114            Self::LocalHeader(n) => write!(f, "local header of {n:?} is inconsistent"),
115            Self::BadName(n) => write!(f, "unacceptable member name {n:?}"),
116            Self::DuplicateName(n) => write!(f, "duplicate member name {n:?}"),
117            Self::Unsupported(n) => write!(f, "unsupported method or flags in {n:?}"),
118            Self::Corrupt(n) => write!(f, "member {n:?} does not match its size and CRC"),
119            Self::TooLarge(n) => write!(f, "member {n:?} exceeds {MAX_MEMBER_SIZE} bytes"),
120        }
121    }
122}

174fn u16_at(d: &[u8], at: usize) -> Option<u16> {
175    d.get(at..at.checked_add(2)?)
176        .and_then(|b| <[u8; 2]>::try_from(b).ok())
177        .map(u16::from_le_bytes)
178}
179
180fn u32_at(d: &[u8], at: usize) -> Option<u32> {
181    d.get(at..at.checked_add(4)?)
182        .and_then(|b| <[u8; 4]>::try_from(b).ok())
183        .map(u32::from_le_bytes)
184}

186fn u64_at(d: &[u8], at: usize) -> Option<u64> {
187    d.get(at..at.checked_add(8)?)
188        .and_then(|b| <[u8; 8]>::try_from(b).ok())
189        .map(u64::from_le_bytes)
190}

192fn to_usize(v: u64) -> Option<usize> {
193    usize::try_from(v).ok()

196/// The ZIP64 extended information extra field (ID 1): the 64-bit values
197/// for each 32-bit field that is `0xffffffff`, in order.
198fn zip64_extra(extra: &[u8], want: [bool; 3]) -> Option<[Option<u64>; 3]> {
199    // Empty, or exactly one ZIP64 block (ID 1) and nothing else, and only
200    // when a field calls for it.
201    let mut out = [None; 3];
202    if extra.is_empty() {
203        return want.iter().all(|w| !w).then_some(out);
204    }
205    if !want.iter().any(|&w| w) || u16_at(extra, 0)? != 1 {
206        return None;
207    }
208    let len = usize::from(u16_at(extra, 2)?);
209    if extra.len() != 4 + len {
210        return None;
211    }
212    let body = extra.get(4..)?;
213    let mut pos = 0;
214    for (slot, &w) in out.iter_mut().zip(&want) {
215        if w {
216            *slot = Some(u64_at(body, pos)?);
217            pos += 8;
218        }
219    }
220    // Nothing after the values we need (a disk number would be at the
221    // end, and is not allowed here).
222    (pos == body.len()).then_some(out)
223}

225/// Percent-decodes an OPC part name. `None` for a bad escape or a result
226/// that is not UTF-8.
227pub fn decode_part_name(name: &str) -> Option<String> {
228    let b = name.as_bytes();
229    let mut out = Vec::with_capacity(b.len());
230    let mut i = 0;
231    while let Some(&c) = b.get(i) {
232        if c == b'%' {
233            let hex = b.get(i + 1..i + 3)?;
234            if !hex.iter().all(u8::is_ascii_hexdigit) {
235                return None;
236            }
237            let s = std::str::from_utf8(hex).ok()?;
238            let v = u8::from_str_radix(s, 16).ok()?;
239            // OPC forbids an escaped `/` or `\` in a segment.
240            if v == b'/' || v == b'\\' {
241                return None;
242            }
243            out.push(v);
244            i += 3;
245        } else {
246            out.push(c);
247            i += 1;
248        }
249    }
250    String::from_utf8(out).ok()
251}

253/// The folded form names are compared in: ASCII case-insensitive, as OPC

260fn check_name(name: &str) -> Result<()> {
261    let bad = || ZipError::BadName(name.to_owned());
262    if name.is_empty() || name.starts_with('/') || name.contains('\\') || name.contains('\0') {
263        return Err(bad());
264    }
265    // A directory entry ends with `/`; every other segment must be a real
266    // name.
267    let body = name.strip_suffix('/').unwrap_or(name);
268    if body
269        .split('/')
270        .any(|s| s.is_empty() || s == "." || s == "..")
271    {
272        return Err(bad());
273    }
274    Ok(())
275}

277fn crc32(data: &[u8], mut crc: u32) -> u32 {
278    crc = !crc;
279    for &b in data {
280        crc ^= u32::from(b);
281        for _ in 0..8 {
282            let mask = (crc & 1).wrapping_neg();
283            crc = (crc >> 1) ^ (0xedb8_8320 & mask);
284        }
285    }
286    !crc
287}

289impl<'a> Archive<'a> {
290    /// Parses `data`.
291    pub fn parse(data: &'a [u8]) -> Result<Self> {
292        let len = data.len();
293        // The end record is the last 22 bytes: no comment.
294        let eocd = len.checked_sub(22).ok_or(ZipError::EndRecord)?;
295        if u32_at(data, eocd) != Some(EOCD_SIG) || u16_at(data, eocd + 20) != Some(0) {
296            return Err(ZipError::EndRecord);
297        }
298        let field16 = |o| u16_at(data, eocd + o).ok_or(ZipError::EndRecord);
299        let field32 = |o| u32_at(data, eocd + o).ok_or(ZipError::EndRecord);
300        let (disk, cd_disk) = (field16(4)?, field16(6)?);
301        let (n_disk, n_total) = (field16(8)?, field16(10)?);
302        let (cd_size32, cd_off32) = (field32(12)?, field32(16)?);
303
304        let needs64 =
305            n_total == 0xffff || n_disk == 0xffff || cd_size32 == u32::MAX || cd_off32 == u32::MAX;
306        let has_locator = eocd >= 20 && u32_at(data, eocd - 20) == Some(ZIP64_LOCATOR_SIG);
307        let (entries, cd_size, cd_off, cd_end, zip64) = if has_locator {
308            let loc = eocd - 20;
309            if u32_at(data, loc + 4) != Some(0) || u32_at(data, loc + 16) != Some(1) {
310                return Err(ZipError::MultiDisk);
311            }
312            let rec = u64_at(data, loc + 8)
313                .and_then(to_usize)
314                .ok_or(ZipError::Zip64)?;
315            if u32_at(data, rec) != Some(ZIP64_EOCD_SIG) {
316                return Err(ZipError::Zip64);
317            }
318            // The record is exactly 56 bytes (no extensible data) and ends
319            // at the locator.
320            if u64_at(data, rec + 4) != Some(44) || rec.checked_add(56) != Some(loc) {
321                return Err(ZipError::Zip64);
322            }
323            if u32_at(data, rec + 16) != Some(0) || u32_at(data, rec + 20) != Some(0) {
324                return Err(ZipError::MultiDisk);
325            }
326            let e_disk = u64_at(data, rec + 24).ok_or(ZipError::Zip64)?;
327            let e_total = u64_at(data, rec + 32).ok_or(ZipError::Zip64)?;
328            let size = u64_at(data, rec + 40).ok_or(ZipError::Zip64)?;
329            let off = u64_at(data, rec + 48).ok_or(ZipError::Zip64)?;
330            // Each classic field is either its value or the marker.
331            let agrees16 = |c: u16, v: u64| c == 0xffff || u64::from(c) == v;
332            let agrees32 = |c: u32, v: u64| c == u32::MAX || u64::from(c) == v;
333            if e_disk != e_total
334                || !agrees16(n_disk, e_disk)
335                || !agrees16(n_total, e_total)
336                || !agrees32(cd_size32, size)
337                || !agrees32(cd_off32, off)
338            {
339                return Err(ZipError::Zip64);
340            }
341            (e_total, size, off, rec as u64, true)
342        } else {
343            if needs64 {
344                return Err(ZipError::Zip64);
345            }
346            (
347                u64::from(n_total),
348                u64::from(cd_size32),
349                u64::from(cd_off32),
350                eocd as u64,
351                false,
352            )
353        };
354        // With ZIP64, the classic disk numbers may be the `0xffff` marker
355        // (packaging tools write it); the ZIP64 record, checked above, holds
356        // the real ones, which must be 0.
357        let disk_ok = disk == 0 || (zip64 && disk == 0xffff);
358        if !disk_ok || cd_disk != disk || (!zip64 && n_disk != n_total) {
359            return Err(ZipError::MultiDisk);
360        }
361        let count = to_usize(entries).ok_or(ZipError::TooManyEntries)?;
362        if count > MAX_ENTRIES {
363            return Err(ZipError::TooManyEntries);
364        }
365        // The central directory ends exactly where the end records start.
366        if cd_off.checked_add(cd_size) != Some(cd_end) {
367            return Err(ZipError::CentralDirectory);
368        }
369
370        let mut members: Vec<Member> = Vec::with_capacity(count.min(len / 46));
371        let mut at = to_usize(cd_off).ok_or(ZipError::CentralDirectory)?;
372        let mut folded: Vec<String> = Vec::with_capacity(count.min(len / 46));
373        for _ in 0..count {
374            let m = Self::central_entry(data, at)?;
375            at = m.central_raw.0 + m.central_raw.1;
376            folded.push(fold(&m.part));
377            members.push(m);
378        }
379        if at as u64 != cd_end {
380            return Err(ZipError::CentralDirectory);
381        }
382        folded.sort_unstable();
383        if let Some(w) = folded.windows(2).find(|w| w.first() == w.get(1)) {
384            return Err(ZipError::DuplicateName(
385                w.first().cloned().unwrap_or_default(),
386            ));
387        }
388
389        // Local entries tile [0, cd_off) in order.
390        let mut next = 0_u64;
391        for m in &mut members {
392            if m.local_offset != next {
393                return Err(ZipError::LocalHeader(m.name.clone()));
394            }
395            Self::local_entry(data, m)?;
396            next = m.end_offset;
397        }
398        if next != cd_off {
399            return Err(ZipError::LocalHeader(String::from(
400                "(after the last member)",
401            )));
402        }
403        Ok(Self {
404            data,
405            members,
406            central_offset: cd_off,
407            central_end: cd_end,
408            zip64,
409        })
410    }

412    fn central_entry(data: &[u8], at: usize) -> Result<Member> {
413        let bad = || ZipError::CentralDirectory;
414        if u32_at(data, at) != Some(CENTRAL_SIG) {
415            return Err(bad());
416        }
417        let flags = u16_at(data, at + 8).ok_or_else(bad)?;
418        let method = u16_at(data, at + 10).ok_or_else(bad)?;
419        let crc = u32_at(data, at + 16).ok_or_else(bad)?;
420        let csize32 = u32_at(data, at + 20).ok_or_else(bad)?;
421        let usize32 = u32_at(data, at + 24).ok_or_else(bad)?;
422        let name_len = usize::from(u16_at(data, at + 28).ok_or_else(bad)?);
423        let extra_len = usize::from(u16_at(data, at + 30).ok_or_else(bad)?);
424        let comment_len = usize::from(u16_at(data, at + 32).ok_or_else(bad)?);
425        let disk = u16_at(data, at + 34).ok_or_else(bad)?;
426        let off32 = u32_at(data, at + 42).ok_or_else(bad)?;
427        let name_bytes = data.get(at + 46..at + 46 + name_len).ok_or_else(bad)?;
428        let extra = data
429            .get(at + 46 + name_len..at + 46 + name_len + extra_len)
430            .ok_or_else(bad)?;
431        let total = 46 + name_len + extra_len + comment_len;
432        if data.get(at..at + total).is_none() {
433            return Err(bad());
434        }
435        let name = if flags & FLAG_UTF8 != 0 || name_bytes.is_ascii() {
436            std::str::from_utf8(name_bytes)
437                .map_err(|_| ZipError::BadName(String::from_utf8_lossy(name_bytes).into_owned()))?
438                .to_owned()
439        } else {
440            return Err(ZipError::BadName(
441                String::from_utf8_lossy(name_bytes).into_owned(),
442            ));
443        };
444        // The raw name and its decoded part name must both be clean, so an
445        // escaped `/`, `\`, `..` or NUL cannot hide.
446        check_name(&name)?;
447        let part = decode_part_name(&name).ok_or_else(|| ZipError::BadName(name.clone()))?;
448        check_name(&part)?;
449        if disk != 0 {
450            return Err(ZipError::MultiDisk);
451        }
452        if flags & (FLAG_ENCRYPTED | FLAG_PATCH | FLAG_STRONG | FLAG_MASKED) != 0
453            || !matches!(method, 0 | 8)
454        {
455            return Err(ZipError::Unsupported(name));
456        }
457        let want = [usize32 == u32::MAX, csize32 == u32::MAX, off32 == u32::MAX];
458        // The package digest rewrites each central entry with only its
459        // ZIP64 block as extra data, so any other extra field would make
460        // the bytes read here differ from the bytes signed: refused.
461        if !extra.is_empty() && (u16_at(extra, 0) != Some(1) || want.iter().all(|w| !w)) {
462            return Err(ZipError::Unsupported(
463                String::from_utf8_lossy(name_bytes).into_owned(),
464            ));
465        }
466        let [u64v, c64v, o64v] = zip64_extra(extra, want).ok_or(ZipError::Zip64)?;
467        let size = u64v.unwrap_or(u64::from(usize32));
468        let compressed_size = c64v.unwrap_or(u64::from(csize32));
469        let local_offset = o64v.unwrap_or(u64::from(off32));
470        if method == 0 && size != compressed_size {
471            return Err(ZipError::Corrupt(name));
472        }
473        Ok(Member {
474            name,
475            part,
476            has_zip64_extra: !extra.is_empty(),
477            method,
478            crc32: crc,
479            compressed_size,
480            size,
481            local_offset,
482            data_offset: 0,
483            end_offset: 0,
484            central_raw: (at, total),
485        })
486    }

488    fn local_entry(data: &[u8], m: &mut Member) -> Result<()> {
489        let bad = || ZipError::LocalHeader(m.name.clone());
490        let at = to_usize(m.local_offset).ok_or_else(bad)?;
491        if u32_at(data, at) != Some(LOCAL_SIG) {
492            return Err(bad());
493        }
494        let version = u16_at(data, at + 4).ok_or_else(bad)?;
495        let flags = u16_at(data, at + 6).ok_or_else(bad)?;
496        let method = u16_at(data, at + 8).ok_or_else(bad)?;
497        let crc = u32_at(data, at + 14).ok_or_else(bad)?;
498        let csize32 = u32_at(data, at + 18).ok_or_else(bad)?;
499        let usize32 = u32_at(data, at + 22).ok_or_else(bad)?;
500        let name_len = usize::from(u16_at(data, at + 26).ok_or_else(bad)?);
501        let extra_len = usize::from(u16_at(data, at + 28).ok_or_else(bad)?);
502        let name = data.get(at + 30..at + 30 + name_len).ok_or_else(bad)?;
503        let extra = data
504            .get(at + 30 + name_len..at + 30 + name_len + extra_len)
505            .ok_or_else(bad)?;
506        if name != m.name.as_bytes() || method != m.method {
507            return Err(bad());
508        }
509        // Local extra data is not used, and would make the local header size
510        // the block map records ambiguous: none is allowed beyond a ZIP64
511        // field when the local sizes need one.
512        // With a data descriptor, the local sizes are zero and no ZIP64
513        // field is needed; otherwise only the one `zip64_extra` accepts.
514        if !extra.is_empty() && flags & FLAG_DESCRIPTOR != 0 {
515            return Err(bad());
516        }
517        // ZIP64 data in either header needs ZIP version 4.5, with a data
518        // descriptor or without.
519        if (m.has_zip64_extra || !extra.is_empty()) && version < 45 {
520            return Err(bad());
521        }
522        // Flags must agree except for bits that only describe the central
523        // record (none that matter here): compare the meaningful ones,
524        // including every bit the central entry is refused for, so a bit
525        // set only in the local header cannot slip past.
526        let meaningful =
527            FLAG_ENCRYPTED | FLAG_DESCRIPTOR | FLAG_PATCH | FLAG_STRONG | FLAG_UTF8 | FLAG_MASKED;
528        let central_flags = u16_at(data, m.central_raw.0 + 8).ok_or_else(bad)?;
529        if flags & meaningful != central_flags & meaningful {
530            return Err(bad());
531        }
532        let data_offset = (at + 30 + name_len + extra_len) as u64;
533        let data_end = data_offset
534            .checked_add(m.compressed_size)
535            .filter(|&e| e <= data.len() as u64)
536            .ok_or_else(bad)?;
537        let end = if flags & FLAG_DESCRIPTOR != 0 {
538            // Local CRC and sizes are zero; the descriptor carries them.
539            if crc != 0 || csize32 != 0 || usize32 != 0 {
540                return Err(bad());
541            }
542            // The descriptor must carry its signature: the package digest
543            // writes one, so one without would hash differently.
544            let d = to_usize(data_end).ok_or_else(bad)?;
545            if u32_at(data, d) != Some(DESCRIPTOR_SIG) {
546                return Err(bad());
547            }
548            let p = d + 4;
549            let d_crc = u32_at(data, p).ok_or_else(bad)?;
550            // The size width follows the entry's ZIP version: 4.5 (ZIP64)
551            // or later, 8-byte sizes; earlier, 4-byte sizes. The OS
552            // packaging tool writes version 4.5 and 8-byte sizes; the OS
553            // signing tool, rewriting an archive, writes version 2.0 and
554            // 4-byte sizes, ZIP64 end records or not. Every descriptor in
555            // the open-source SDK's test corpus follows this. A ZIP64 extra
556            // field needs version 4.5; one with an earlier version is
557            // refused rather than read at either width.
558            let wide = version >= 45;
559            let (dc, du, n) = if wide {
560                (
561                    u64_at(data, p + 4).ok_or_else(bad)?,
562                    u64_at(data, p + 12).ok_or_else(bad)?,
563                    20,
564                )
565            } else {
566                (
567                    u64::from(u32_at(data, p + 4).ok_or_else(bad)?),
568                    u64::from(u32_at(data, p + 8).ok_or_else(bad)?),
569                    12,
570                )
571            };
572            if d_crc != m.crc32 || dc != m.compressed_size || du != m.size {
573                return Err(bad());
574            }
575            (p + n) as u64
576        } else {
577            let want = [usize32 == u32::MAX, csize32 == u32::MAX, false];
578            let [u64v, c64v, _] = zip64_extra(extra, want).ok_or_else(bad)?;
579            let lsize = u64v.unwrap_or(u64::from(usize32));
580            let lcsize = c64v.unwrap_or(u64::from(csize32));
581            if crc != m.crc32 || lsize != m.size || lcsize != m.compressed_size {
582                return Err(bad());
583            }
584            data_end
585        };
586        m.data_offset = data_offset;
587        m.end_offset = end;
588        Ok(())
589    }

619    /// The member's content, decompressed and checked against its size and
620    /// CRC.
621    pub fn read(&self, m: &Member) -> Result<Vec<u8>> {
622        if m.size > MAX_MEMBER_SIZE {
623            return Err(ZipError::TooLarge(m.name.clone()));
624        }
625        let mut out = Vec::with_capacity(to_usize(m.size).unwrap_or(0));
626        self.stream(m, |chunk| out.extend_from_slice(chunk))?;
627        Ok(out)
628    }

630    /// Feeds the member's decompressed content to `sink` in chunks, then
631    /// checks its size and CRC. Unlike [`Self::read`], works for members
632    /// of any size; `sink` may have seen data from a member that then
633    /// fails the check, so callers must discard their result on error.
634    pub fn stream(&self, m: &Member, mut sink: impl FnMut(&[u8])) -> Result<()> {
635        let raw = self.raw_data(m);
636        let corrupt = || ZipError::Corrupt(m.name.clone());
637        let mut crc = 0_u32;
638        let mut total = 0_u64;
639        if m.method == 0 {
640            crc = crc32(raw, crc);
641            total = raw.len() as u64;
642            sink(raw);
643        } else {
644            // The deflate stream must end, and use all of the compressed
645            // data.
646            inflate_raw(raw, m.size, |chunk| {
647                crc = crc32(chunk, crc);
648                total = total.saturating_add(chunk.len() as u64);
649                sink(chunk);
650            })
651            .ok_or_else(corrupt)?;
652        }
653        if total != m.size || crc != m.crc32 {
654            return Err(corrupt());
655        }
656        Ok(())
657    }

659    /// The raw central directory entry of `m`.
660    #[must_use]
661    pub fn central_bytes(&self, m: &Member) -> &'a [u8] {
662        let (at, len) = m.central_raw;
663        self.data
664            .get(at..at.saturating_add(len))
665            .unwrap_or_default()
666    }

669/// Inflates raw DEFLATE `input`, which must be one whole stream: its final
670/// block ends exactly at the end of the input. Feeds the output to `sink`
671/// in chunks and stops with `None` once it passes `max` bytes. `None` if
672/// the data is invalid, stops before the final block, or has bytes after
673/// it.
674pub(crate) fn inflate_raw(input: &[u8], max: u64, mut sink: impl FnMut(&[u8])) -> Option<()> {
675    use miniz_oxide::inflate::stream::{InflateState, inflate};
676    use miniz_oxide::{DataFormat, MZError, MZFlush, MZStatus};
677    let mut state = InflateState::new_boxed(DataFormat::Raw);
678    let mut buf = vec![0_u8; 64 * 1024];
679    let mut input = input;
680    let mut total = 0_u64;
681    // Streamed into a fixed buffer. `Finish` would ask for the whole output
682    // at once, so `None` is used and progress is required on every call: a
683    // call that neither reads nor writes means the data is truncated or
684    // invalid.
685    loop {
686        let r = inflate(&mut state, input, &mut buf, MZFlush::None);
687        input = input.get(r.bytes_consumed..).unwrap_or_default();
688        let chunk = buf.get(..r.bytes_written).unwrap_or_default();
689        total = total.saturating_add(chunk.len() as u64);
690        if total > max {
691            return None;
692        }
693        sink(chunk);
694        match r.status {
695            Ok(MZStatus::StreamEnd) => return input.is_empty().then_some(()),
696            Ok(MZStatus::Ok) | Err(MZError::Buf) if r.bytes_written > 0 || r.bytes_consumed > 0 => {
697            }
698            _ => return None,
699        }
700    }
701}

720    #[test]
721    fn zip64_extra_accepts_exactly_the_needed_block() {
722        let all = [true, true, true];
723        let none = [false, false, false];
724        let body: Vec<u8> = [1_u64, 2, 3].iter().flat_map(|v| v.to_le_bytes()).collect();
725        assert_eq!(
726            zip64_extra(&block(1, &body), all),
727            Some([Some(1), Some(2), Some(3)])
728        );
729        // Only the uncompressed size called for: one value.
730        assert_eq!(
731            zip64_extra(&block(1, &7_u64.to_le_bytes()), [true, false, false]),
732            Some([Some(7), None, None])
733        );
734        assert_eq!(zip64_extra(&[], none), Some([None; 3]));
735        // Missing when needed.
736        assert_eq!(zip64_extra(&[], all), None);
737        // Present when not needed, even empty.
738        assert_eq!(zip64_extra(&block(1, &[]), none), None);
739        assert_eq!(zip64_extra(&block(1, &body), none), None);
740        // Another block after it, or before it, or instead of it.
741        let mut trailing = block(1, &body);
742        trailing.extend(block(0xcafe, &[0; 4]));
743        assert_eq!(zip64_extra(&trailing, all), None);
744        let mut leading = block(0xcafe, &[0; 4]);
745        leading.extend(block(1, &body));
746        assert_eq!(zip64_extra(&leading, all), None);
747        assert_eq!(zip64_extra(&block(0xcafe, &body), all), None);
748        // Values left over (a disk number), or too few.
749        let mut extra_body = body.clone();
750        extra_body.extend_from_slice(&0_u32.to_le_bytes());
751        assert_eq!(zip64_extra(&block(1, &extra_body), all), None);
752        assert_eq!(zip64_extra(&block(1, &body[..16]), all), None);
753        // A length running past the field.
754        let mut short = block(1, &body);
755        short.truncate(10);
756        assert_eq!(zip64_extra(&short, all), None);
757    }
