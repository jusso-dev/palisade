1//! The policy cache on disk (ADR 0007): the `active`, `previous` and
2//! `pending` files under the agent's store directory, and the copies of the
3//! anti-rollback record.
4//!
5//! Every write goes to a temporary file that is flushed to disk, then
6//! renamed over its target. On Windows the rename is `MoveFileExW` with
7//! `MOVEFILE_REPLACE_EXISTING | MOVEFILE_WRITE_THROUGH`, so it is durable
8//! before the call returns. Elsewhere it is `rename(2)` followed by a flush
9//! of the directory. A power loss therefore leaves either the old file or
10//! the new one, never a torn file.
11//!
12//! The directory must be SYSTEM-only (#220); the installer creates it. The
13//! files are authenticated by their signatures, not by their permissions.
14//! Encryption at rest is #326.
15

26/// The largest cache file the agent reads: the 64 MiB bundle limit (#225)
27/// plus 1 MiB for the envelope around it. A larger file is not read at all
28/// and counts as unreadable.
29pub const MAX_FILE_BYTES: u64 = DEFAULT_MAX_BYTES as u64 + (1 << 20);
30
31/// The `active`, `previous` and `pending` files in one directory.
32#[derive(Debug, Clone)]
33pub struct PolicyStore {
34    dir: PathBuf,
35}
36
37impl PolicyStore {
38    /// The store in `dir`, which is created if it does not exist.
39    ///
40    /// # Errors
41    ///
42    /// The I/O error if the directory cannot be created.
43    pub fn open(dir: impl Into<PathBuf>) -> io::Result<Self> {
44        let dir = dir.into();
45        fs::create_dir_all(&dir)?;
46        Ok(Self { dir })
47    }
48
49    /// The directory.
50    pub fn dir(&self) -> &Path {
51        &self.dir
52    }
53
54    /// The path of a cache file.
55    pub fn path(&self, file: CacheFile) -> PathBuf {
56        self.dir.join(match file {
57            CacheFile::Active => "active",
58            CacheFile::Previous => "previous",
59            CacheFile::Pending => "pending",
60        })
61    }
62
63    /// Reads a whole cache file into memory, up to [`MAX_FILE_BYTES`].
64    /// `Ok(None)` if it does not exist.
65    ///
66    /// # Errors
67    ///
68    /// The I/O error, or [`io::ErrorKind::InvalidData`] for a file over the
69    /// limit.
70    pub fn read(&self, file: CacheFile) -> io::Result<Option<Vec<u8>>> {
71        read_limited(&self.path(file), MAX_FILE_BYTES)
72    }
73
74    /// Replaces a cache file with `bytes`, durably.
75    ///
76    /// # Errors
77    ///
78    /// The I/O error. The old file, if any, is then unchanged.
79    pub fn write(&self, file: CacheFile, bytes: &[u8]) -> io::Result<()> {
80        write_atomic(&self.dir, &self.path(file), bytes)
81    }
82
83    /// Renames `pending` over `active`, durably (ADR 0007 step 4).
84    ///
85    /// # Errors
86    ///
87    /// The I/O error. `active` is then unchanged.
88    pub fn promote_pending(&self) -> io::Result<()> {
89        replace_durably(
90            &self.path(CacheFile::Pending),
91            &self.path(CacheFile::Active),
92            &self.dir,
93        )
94    }
95
96    /// Deletes a cache file. A file that does not exist is not an error.
97    ///
98    /// # Errors
99    ///
100    /// Any other I/O error.
101    pub fn remove(&self, file: CacheFile) -> io::Result<()> {
102        match fs::remove_file(self.path(file)) {
103            Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(()),
104            other => other,
105        }
106    }
107}

109/// Where the copies of the anti-rollback record are kept.
110///
111/// #225 asks for copies in separate stores (a DPAPI-protected file and a
112/// registry mirror, later a TPM-sealed blob), so that replacing one is
113/// detected by the others. That is #326; [`FileRecordStore`] keeps two
114/// files until it lands.
115pub trait RecordStore: Send + Sync + fmt::Debug {
116    /// Every copy, always in the same order. `None` for a copy that does not
117    /// exist. A copy that exists but cannot be read is returned as empty
118    /// bytes, which never unseal, so it counts as corrupt rather than
119    /// missing.
120    fn read(&self) -> Vec<Option<Vec<u8>>>;
121
122    /// Writes `sealed` to every copy, trying them all.
123    ///
124    /// # Errors
125    ///
126    /// The first I/O error. Some copies may then hold the new value and
127    /// some the old: startup tells that apart from tampering
128    /// ([`RecordCopies::interrupted_activation`](palisade_proto::policy_bundle::record::RecordCopies::interrupted_activation)).
129    fn write(&self, sealed: &[u8]) -> io::Result<()>;
130}

132/// Two record copies as files, `record.1` and `record.2`, in a directory.
133/// A stand-in for #326's separate stores.
134#[derive(Debug, Clone)]
135pub struct FileRecordStore {
136    dir: PathBuf,
137}
138
139impl FileRecordStore {
140    const NAMES: [&'static str; 2] = ["record.1", "record.2"];
141
142    /// The copies in `dir`, which is created if it does not exist.
143    ///
144    /// # Errors
145    ///
146    /// The I/O error if the directory cannot be created.
147    pub fn open(dir: impl Into<PathBuf>) -> io::Result<Self> {
148        let dir = dir.into();
149        fs::create_dir_all(&dir)?;
150        Ok(Self { dir })
151    }
152}
153
154impl RecordStore for FileRecordStore {
155    fn read(&self) -> Vec<Option<Vec<u8>>> {
156        Self::NAMES
157            .iter()
158            .map(|name| {
159                // One byte over the longest copy is enough to tell that a
160                // copy is too long.
161                read_limited(&self.dir.join(name), SEALED_LEN_WITH_FALLBACK as u64)
162                    .unwrap_or_else(|_| Some(Vec::new()))
163            })
164            .collect()
165    }
166
167    fn write(&self, sealed: &[u8]) -> io::Result<()> {
168        let mut first = Ok(());
169        for name in Self::NAMES {
170            let result = write_atomic(&self.dir, &self.dir.join(name), sealed);
171            if first.is_ok() {
172                first = result;
173            }
174        }
175        first
176    }
177}

179/// Reads at most `limit` bytes of `path`. `Ok(None)` if it does not exist.
180fn read_limited(path: &Path, limit: u64) -> io::Result<Option<Vec<u8>>> {
181    let file = match File::open(path) {
182        Ok(file) => file,
183        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
184        Err(e) => return Err(e),
185    };
186    let mut bytes = Vec::new();
187    file.take(limit.saturating_add(1)).read_to_end(&mut bytes)?;
188    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > limit {
189        return Err(io::Error::new(
190            io::ErrorKind::InvalidData,
191            format!("{} is over the {limit}-byte limit", path.display()),
192        ));
193    }
194    Ok(Some(bytes))
195}
196
197/// Writes `bytes` to `<path>.tmp`, flushes it, then renames it over `path`.
198fn write_atomic(dir: &Path, path: &Path, bytes: &[u8]) -> io::Result<()> {
199    let tmp = path.with_extension("tmp");
200    {
201        let mut file = OpenOptions::new()
202            .write(true)
203            .create(true)
204            .truncate(true)
205            .open(&tmp)?;
206        file.write_all(bytes)?;
207        file.sync_all()?;
208    }
209    replace_durably(&tmp, path, dir)
210}

212/// Renames `from` over `to`, and returns once the rename is on disk.
213#[cfg(windows)]
214fn replace_durably(from: &Path, to: &Path, _dir: &Path) -> io::Result<()> {
215    crate::ffi::replace_file_durably(from, to)
216}
217
218/// Renames `from` over `to`, and returns once the rename is on disk.
219#[cfg(not(windows))]
220fn replace_durably(from: &Path, to: &Path, dir: &Path) -> io::Result<()> {
221    fs::rename(from, to)?;
222    File::open(dir)?.sync_all()
223}

226mod tests {

229    #[test]
230    fn read_limited_refuses_a_file_over_the_limit() {
231        let dir = super::super::tests::TempDir::new("limit");
232        let path = dir.path().join("f");
233        fs::write(&path, [0_u8; 5]).unwrap();
234        assert_eq!(read_limited(&path, 5).unwrap().unwrap().len(), 5);
235        let err = read_limited(&path, 4).unwrap_err();
236        assert_eq!(err.kind(), io::ErrorKind::InvalidData);
237        assert!(
238            read_limited(&dir.path().join("missing"), 4)
239                .unwrap()
240                .is_none()
241        );
242    }
243
244    #[test]
245    fn writes_replace_and_leave_no_temporary_file() {
246        let dir = super::super::tests::TempDir::new("write");
247        let store = PolicyStore::open(dir.path().join("policy")).unwrap();
248        store.write(CacheFile::Pending, b"one").unwrap();
249        store.write(CacheFile::Pending, b"two").unwrap();
250        store.promote_pending().unwrap();
251        assert_eq!(store.read(CacheFile::Active).unwrap().unwrap(), b"two");
252        assert!(store.read(CacheFile::Pending).unwrap().is_none());
253        let names: Vec<_> = fs::read_dir(store.dir())
254            .unwrap()
255            .map(|e| e.unwrap().file_name())
256            .collect();
257        assert_eq!(names, ["active"]);
258        store.remove(CacheFile::Active).unwrap();
259        store.remove(CacheFile::Active).unwrap();
260        assert!(store.read(CacheFile::Active).unwrap().is_none());
261    }
262
263    #[test]
264    fn a_record_copy_that_cannot_be_read_counts_as_corrupt() {
265        let dir = super::super::tests::TempDir::new("records");
266        let records = FileRecordStore::open(dir.path()).unwrap();
267        assert_eq!(records.read(), [None, None]);
268        records.write(b"sealed").unwrap();
269        assert_eq!(
270            records.read(),
271            [Some(b"sealed".to_vec()), Some(b"sealed".to_vec())]
272        );
273        // A directory where a copy should be cannot be read as a file.
274        fs::remove_file(dir.path().join("record.2")).unwrap();
275        fs::create_dir(dir.path().join("record.2")).unwrap();
276        assert_eq!(records.read(), [Some(b"sealed".to_vec()), Some(Vec::new())]);
277        // The write still reaches the first copy, and reports the failure.
278        assert!(records.write(b"next").is_err());
279        assert_eq!(records.read()[0], Some(b"next".to_vec()));
280    }
