1//! Envelope verification (#325): what a cache file holds, and whether it is
2//! trusted.
3//!
4//! A cache file holds one signed bundle envelope
5//! ([`palisade_proto::policy_bundle::envelope`]). Before anything past the
6//! bundle header is decoded, an [`EnvelopeVerifier`] checks it in #225's
7//! order and says where the bundle bytes are. The cache then loads only
8//! those bytes, and only when the signature verified or the signature mode
9//! tolerates that it did not.
10//!
11//! [`SignedEnvelopes`] is the verifier the agent runs: it calls
12//! [`envelope::verify`] against the device's current trust set. A cache
13//! file that is not an envelope, or whose signatures are not from a policy
14//! key the trust set lists inside its validity window, never verifies, so
15//! in the default `enforce` mode it never loads.
16

26        envelope::{self, Agent},

30/// What a verifier found in one cache file.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub struct Examination {
33    /// #225 steps 1, 2 and 5 as checked, and step 6 as far as the verifier
34    /// decoded. The cache clears `schema_supported` if the bundle then
35    /// fails to load (validate or compile).
36    pub examined: Examined,
37    /// Where the canonical bundle bytes are in the file. Meaningful only
38    /// when `examined` is [`Examined::Parsed`]; the cache refuses a range
39    /// outside the file.
40    pub bundle: Range<usize>,
41}
42
43impl Examination {
44    /// An unreadable file.
45    pub const UNREADABLE: Self = Self {
46        examined: Examined::Unreadable,
47        bundle: 0..0,
48    };
49}

51/// Checks a cache file's envelope (#325).
52///
53/// An implementation must not decode anything past the bundle header until
54/// a signature verifies, unless `mode` tolerates an invalid signature
55/// (#225). It reports what it found; the decisions (signature mode, target,
56/// anti-rollback) are made by [`palisade_proto::policy_bundle::select`].
57pub trait EnvelopeVerifier: Send + Sync + fmt::Debug {
58    /// Examines one file's bytes at `now` (trusted time, #220), in the
59    /// signature mode in force.
60    fn examine(&self, file: &[u8], now: Timestamp, mode: EffectiveMode) -> Examination;
61}
62
63/// This agent's version, for `min_agent_version` (#225 step 5). A version
64/// that does not parse reads as 0.0.0, which only refuses more bundles.
65pub fn agent_version() -> AgentVersion {
66    let part = |s: &str| s.parse().unwrap_or(0);
67    AgentVersion::new(
68        part(env!("CARGO_PKG_VERSION_MAJOR")),
69        part(env!("CARGO_PKG_VERSION_MINOR")),
70        part(env!("CARGO_PKG_VERSION_PATCH")),
71    )
72}

74/// The device's trust set, shared with whatever keeps it current (the
75/// trust-set update path of #224). A new set is swapped in whole; every
76/// examination reads one set from start to end.
77pub type SharedTrust = Arc<ArcSwap<TrustSet>>;
78
79/// Verifies signed bundle envelopes against the device's trust set with
80/// [`envelope::verify`].
81#[derive(Clone, Debug)]
82pub struct SignedEnvelopes {
83    trust: SharedTrust,
84    /// The agent version that `min_agent_version` is checked against.
85    pub agent: AgentVersion,
86    /// The size and item limits.
87    pub limits: BundleLimits,
88    /// The clock-skew tolerance for key validity windows.
89    pub skew: SkewTolerance,
90}
91
92impl SignedEnvelopes {
93    /// A verifier reading `trust`, with this agent's version, the default
94    /// limits and the default skew tolerance.
95    pub fn new(trust: SharedTrust) -> Self {
96        Self {
97            trust,
98            agent: agent_version(),
99            limits: BundleLimits::DEFAULT,
100            skew: SkewTolerance::DEFAULT,
101        }
102    }
103
104    /// The trust set it reads.
105    pub const fn trust(&self) -> &SharedTrust {
106        &self.trust
107    }
108}
109
110impl EnvelopeVerifier for SignedEnvelopes {
111    fn examine(&self, file: &[u8], now: Timestamp, mode: EffectiveMode) -> Examination {
112        let trust = self.trust.load();
113        let agent = Agent {
114            version: self.agent,
115            limits: self.limits,
116            skew: self.skew,
117            mode,
118        };
119        let verified = envelope::verify(file, &trust, &agent, now);
120        match verified.examined() {
121            Examined::Unreadable => Examination::UNREADABLE,
122            examined @ Examined::Parsed { .. } => match payload_range(file) {
123                Some(bundle) => Examination { examined, bundle },
124                // `verify` decoded the envelope, so this cannot happen;
125                // if it did, nothing loads.
126                None => Examination::UNREADABLE,
127            },
128        }
129    }
130}
131
132/// Where the payload of an envelope that decoded is: after the version and
133/// kind bytes and the 4-byte big-endian length (see the format in
134/// [`envelope`]). `None` if the length runs past the file.
135fn payload_range(file: &[u8]) -> Option<Range<usize>> {
136    const START: usize = 1 + 1 + 4;
137    let len: [u8; 4] = file.get(2..START)?.try_into().ok()?;
138    let end = START.checked_add(usize::try_from(u32::from_be_bytes(len)).ok()?)?;
139    file.get(START..end).map(|_| START..end)
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn payload_range_stays_inside_the_file() {
148        assert_eq!(payload_range(&[1, 1, 0, 0, 0, 2, 9, 9, 0]), Some(6..8));
149        assert_eq!(payload_range(&[1, 1, 0, 0, 0, 4, 9, 9]), None);
150        assert_eq!(payload_range(&[1, 1, 0]), None);
151        assert_eq!(payload_range(&[]), None);
152    }
153
154    #[test]
155    fn the_agent_version_is_this_crate_s() {
156        let v = agent_version();
157        assert_eq!(
158            v,
159            AgentVersion::new(
160                env!("CARGO_PKG_VERSION_MAJOR").parse().unwrap(),
161                env!("CARGO_PKG_VERSION_MINOR").parse().unwrap(),
162                env!("CARGO_PKG_VERSION_PATCH").parse().unwrap(),
163            )
164        );
165    }
