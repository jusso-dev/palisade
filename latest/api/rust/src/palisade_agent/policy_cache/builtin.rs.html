1//! The built-in policies the agent runs when no bundle loads (ADR 0018).
2//!
3//! - **First install**: no policy was ever accepted. Audit mode, so nothing
4//!   is blocked before the first bundle arrives; every would-block decision
5//!   is reported (`PolicyMissing`). The mandatory blocklists of #87 join it
6//!   when the agent ships them.
7//! - **Emergency**: a policy was accepted before, but neither `active` nor
8//!   `previous` loads. Enforce mode, and every unknown object is blocked.
9//!   #326 replaces the content with ADR 0018's emergency policy (allow
10//!   OS-signed binaries and Palisade itself), and makes it audit-only in
11//!   `native_only` mode. Until then it is stricter than that policy, never
12//!   weaker.
13//!
14//! Neither has any rule, so both build without error; the tests check it.
15
16use palisade_core::Uuid7;
17use palisade_policy::{
18    model::{
19        GovernanceSettings, Metadata, Mode, ModeOptions, Policy, PolicyDocument, PolicyId,
20        PolicySettings, PrecedenceModel, PrecedenceOptions, UnknownPolicy,
21    },
22    schema,
23    validate::ValidatedPolicy,
24};
25
26/// Which built-in policy.
27#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
28pub enum Builtin {
29    /// No policy yet: audit mode.
30    FirstInstall,
31    /// Both cache files failed: enforce mode, block unknown.
32    Emergency,
33}
34
35/// The policy IDs of the built-in policies: fixed UUIDv7s, so events name
36/// them the same way on every device.
37const FIRST_INSTALL_ID: [u8; 16] = [
38    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x01,
39];
40const EMERGENCY_ID: [u8; 16] = [
41    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x02,
42];
43
44impl Builtin {
45    /// The policy document.
46    #[expect(
47        clippy::expect_used,
48        reason = "fixed UUIDv7s; `builtins_validate_and_build` tests them"
49    )]
50    pub fn document(self) -> PolicyDocument {
51        let (id, name, mode) = match self {
52            Self::FirstInstall => (
53                FIRST_INSTALL_ID,
54                "Built-in first-install policy",
55                Mode::Audit,
56            ),
57            Self::Emergency => (EMERGENCY_ID, "Built-in emergency policy", Mode::Enforce),
58        };
59        PolicyDocument {
60            schema_version: schema::CURRENT,
61            policy: Policy {
62                id: PolicyId(Uuid7::from_bytes(id).expect("a fixed UUIDv7")),
63                name: name.to_owned(),
64                description: None,
65                revision: 1,
66                precedence_model: PrecedenceModel::default(),
67                precedence_options: PrecedenceOptions::default(),
68                mode,
69                mode_options: ModeOptions::default(),
70                settings: PolicySettings::default(),
71                // Blocks every unknown object, which audit mode reports
72                // and enforce mode applies.
73                unknown: UnknownPolicy::default(),
74                governance: GovernanceSettings::default(),
75                rule_sets: Vec::new(),
76                scope: None,
77                rule_set_scopes: Vec::new(),
78                location_profiles: Vec::new(),
79                // No catch-all allow, so nothing to confirm.
80                confirm_permit_all: None,
81                metadata: Metadata::default(),
82            },
83            rule_sets: Vec::new(),
84        }
85    }
86
87    /// The validated policy.
88    pub fn policy(self) -> ValidatedPolicy {
89        #[expect(
90            clippy::expect_used,
91            reason = "a fixed document with no rules; `builtins_validate_and_build` tests it"
92        )]
93        self.document()
94            .validate()
95            .expect("the built-in policies are valid")
96    }
97}

99#[cfg(test)]
100mod tests {
101    use palisade_policy::engine::{Engine, EngineConfig};
102
103    use super::*;
104
105    #[test]
106    fn builtins_validate_and_build() {
107        assert!(Uuid7::from_bytes(FIRST_INSTALL_ID).is_some());
108        assert!(Uuid7::from_bytes(EMERGENCY_ID).is_some());
109        for (builtin, mode) in [
110            (Builtin::FirstInstall, Mode::Audit),
111            (Builtin::Emergency, Mode::Enforce),
112        ] {
113            let policy = builtin.policy();
114            assert_eq!(policy.document().policy.mode, mode);
115            assert!(policy.document().rule_sets.is_empty());
116            Engine::new(policy, EngineConfig::default()).unwrap();
117        }
118        assert_ne!(
119            Builtin::FirstInstall.document().policy.id,
120            Builtin::Emergency.document().policy.id
121        );
122    }
123}
