1//! The built-in policies the agent runs when no bundle loads (ADR 0018).
2//!
3//! - **First install**: no policy was ever accepted. Audit mode, so nothing
4//!   is blocked before the first bundle arrives; every would-block decision
5//!   is reported (`PolicyMissing`). The mandatory blocklists of #87 join it
6//!   when the agent ships them.
7//! - **Emergency**: a policy was accepted before, but neither `active` nor
8//!   `previous` loads (#225, ADR 0018). It allows only:
9//!   - binaries signed by the operating system's publishers, in the
10//!     protected OS locations, through the agent's `system_baseline`
11//!     ([`palisade_policy::unknown::baseline`]). That is narrower than "every
12//!     OS-signed binary": user-writable folders inside the Windows directory
13//!     and binaries outside the OS locations stay blocked;
14//!   - Palisade's own images, each pinned by SHA-256 **and** publisher
15//!     ([`SelfPin`]), so a copy signed by someone else or a re-signed file
16//!     with other content is not allowed;
17//!
18//!   and blocks everything else. It runs in enforce mode where a Palisade
19//!   enforcer exists, and in audit mode under the `native_only` strategy,
20//!   where nothing in Palisade's layer enforces it ([`Enforcement`]). It is
21//!   only ever applied to Palisade's own decisions: it never touches the
22//!   installed OS-native policies (ADR 0018, "Emergency policy and the
23//!   native layer").
24//!
25//! Every input that does not build (an invalid pin, baseline facts that do
26//! not fit) is left out, which only removes an allow: the emergency policy
27//! is never weaker than "block everything", and with no facts at all it is
28//! exactly that. The tests check every combination builds.
29

42/// Which built-in policy.
43#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
44pub enum Builtin {
45    /// No policy yet: audit mode.
46    FirstInstall,
47    /// Both cache files failed: allow the OS and Palisade, block the rest.
48    Emergency,
49}
50
51/// Who enforces Palisade's decisions on this device (ADR 0002's
52/// `enforcement_strategy`), which decides the emergency policy's mode.
53#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
54pub enum Enforcement {
55    /// `layered` or `driver_only`: the Palisade driver enforces, so the
56    /// emergency policy runs in enforce mode. The default, because it is
57    /// the stricter reading.
58    #[default]
59    Palisade,
60    /// `native_only`: there is no Palisade pre-execution enforcer. The
61    /// emergency policy runs in audit mode, so events show what it would
62    /// block, and the device reports `PolicyEmergencyUnenforced`. The
63    /// installed native policy keeps enforcing (ADR 0018).
64    NativeOnly,
65}

67/// One of Palisade's own images, for the emergency policy's allow.
68#[derive(Clone, Debug, PartialEq, Eq, Hash)]
69pub struct SelfPin {
70    /// SHA-256 of the image, as 64 hex digits.
71    pub sha256: String,
72    /// The publisher whose trusted signature it must carry.
73    pub publisher: String,
74}

76/// What the agent knows about itself and the device, from its own
77/// measurements, never from a bundle: the built-in policies are built from
78/// it, and the system baseline is added to every bundle loaded.
79#[derive(Clone, Debug, Default, PartialEq, Eq)]
80pub struct AgentFacts {
81    /// The OS publishers and flagged directories for the system baseline.
82    pub baseline: Option<SystemBaseline>,
83    /// Palisade's own images.
84    pub self_pins: Vec<SelfPin>,
85    /// Who enforces.
86    pub enforcement: Enforcement,
87}

89/// The policy IDs of the built-in policies: fixed UUIDv7s, so events name
90/// them the same way on every device.
91const FIRST_INSTALL_ID: [u8; 16] = [
92    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x01,
93];
94const EMERGENCY_ID: [u8; 16] = [
95    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x02,
96];
97/// The rule set holding the self pins.
98const SELF_SET_ID: [u8; 16] = [
99    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x00, 0x01, 0x00,
100];
101/// The first self-pin rule; rule `n` has `n` added to the last two bytes.
102const SELF_RULE_BASE: [u8; 16] = [
103    0x01, 0x9a, 0xc0, 0x00, 0x00, 0x00, 0x70, 0x00, 0x80, 0x00, 0x00, 0x00, 0x00, 0x01, 0x00, 0x00,
104];
105
106#[expect(
107    clippy::expect_used,
108    reason = "fixed UUIDv7s; `builtins_validate_and_build` tests them"
109)]
110fn uuid(bytes: [u8; 16]) -> Uuid7 {
111    Uuid7::from_bytes(bytes).expect("a fixed UUIDv7")
112}
113
114fn self_rule_id(n: u16) -> RuleId {
115    let mut bytes = SELF_RULE_BASE;
116    let [hi, lo] = n.to_be_bytes();
117    bytes[14] = hi;
118    bytes[15] = lo;
119    RuleId::from(uuid(bytes))
120}

122impl Builtin {
123    /// The policy document, without the system baseline (which is added
124    /// after validation).
125    pub fn document(self, facts: &AgentFacts) -> PolicyDocument {
126        self.document_with(facts, true)
127    }
128
129    fn document_with(self, facts: &AgentFacts, pins: bool) -> PolicyDocument {
130        let (id, name, mode) = match self {
131            Self::FirstInstall => (
132                FIRST_INSTALL_ID,
133                "Built-in first-install policy",
134                Mode::Audit,
135            ),
136            Self::Emergency => (
137                EMERGENCY_ID,
138                "Built-in emergency policy",
139                match facts.enforcement {
140                    Enforcement::Palisade => Mode::Enforce,
141                    Enforcement::NativeOnly => Mode::Audit,
142                },
143            ),
144        };
145        let rule_sets = match self {
146            Self::Emergency if pins && !facts.self_pins.is_empty() => {
147                vec![self_set(&facts.self_pins)]
148            }
149            _ => Vec::new(),
150        };
151        PolicyDocument {
152            schema_version: schema::CURRENT,
153            policy: Policy {
154                id: PolicyId(uuid(id)),
155                name: name.to_owned(),
156                description: None,
157                revision: 1,
158                precedence_model: PrecedenceModel::default(),
159                precedence_options: PrecedenceOptions::default(),
160                mode,
161                mode_options: ModeOptions::default(),
162                settings: PolicySettings::default(),
163                // Blocks every unknown object, which audit mode reports
164                // and enforce mode applies.
165                unknown: UnknownPolicy::default(),
166                governance: GovernanceSettings::default(),
167                rule_sets: rule_sets.iter().map(|s| s.id).collect(),
168                scope: None,
169                rule_set_scopes: Vec::new(),
170                location_profiles: Vec::new(),
171                // No catch-all allow, so nothing to confirm.
172                confirm_permit_all: None,
173                metadata: Metadata::default(),
174            },
175            rule_sets,
176        }
177    }
178
179    /// The validated policy. Never fails: an input that does not build is
180    /// left out, which only makes the policy stricter (see the module
181    /// docs).
182    pub fn policy(self, facts: &AgentFacts) -> ValidatedPolicy {
183        let policy = self.document(facts).validate().unwrap_or_else(|errors| {
184            tracing::error!(
185                ?errors,
186                "the built-in policy's self pins do not validate; they are left out"
187            );
188            self.bare(facts)
189        });
190        match (self, &facts.baseline) {
191            (Self::Emergency, Some(baseline)) => {
192                let fallback = policy.clone();
193                policy.with_system_baseline(baseline).unwrap_or_else(|e| {
194                    tracing::error!(error = %e, "the system baseline does not build; the emergency policy runs without it");
195                    fallback
196                })
197            }
198            _ => policy,
199        }
200    }
201
202    /// The policy with no rules at all.
203    fn bare(self, facts: &AgentFacts) -> ValidatedPolicy {
204        #[expect(
205            clippy::expect_used,
206            reason = "a fixed document with no rules; `builtins_validate_and_build` tests it"
207        )]
208        self.document_with(facts, false)
209            .validate()
210            .expect("the built-in policies are valid")
211    }
212}

214/// The allowlist of Palisade's own images: one rule per pin, matching the
215/// hash and a trusted signature from the publisher, for every object type.
216fn self_set(pins: &[SelfPin]) -> RuleSet {
217    let rules = pins
218        .iter()
219        .zip(1_u16..)
220        .map(|(pin, n)| Rule {
221            id: self_rule_id(n),
222            display_id: None,
223            name: format!("Palisade image {n}"),
224            description: None,
225            enabled: true,
226            activates_at: None,
227            expires_at: None,
228            on_expire: Default::default(),
229            object_types: Vec::new(),
230            match_: Match::All(vec![
231                Match::Criterion(Criterion::new(
232                    Field::HashSha256,
233                    Operator::Equals,
234                    [pin.sha256.clone()],
235                )),
236                Match::Criterion(Criterion::new(
237                    Field::SignerPublisher,
238                    Operator::Equals,
239                    [pin.publisher.clone()],
240                )),
241            ]),
242            catch_all: false,
243            attribute: false,
244            action: Action::Allow {},
245            effects: Default::default(),
246            order: None,
247            priority: None,
248            enforcement_override: Default::default(),
249            scope: None,
250            metadata: Default::default(),
251        })
252        .collect();
253    RuleSet {
254        id: RuleSetId::from(uuid(SELF_SET_ID)),
255        name: "Palisade agent images".to_owned(),
256        description: None,
257        version: None,
258        kind: RuleSetKind::Allowlist,
259        mandatory: false,
260        os: Default::default(),
261        read_only: true,
262        hidden: false,
263        mode_override: None,
264        allow_child_suppress: None,
265        category_path: None,
266        band: 0,
267        rules,
268        metadata: Metadata::default(),
269    }
270}

292    #[test]
293    fn builtins_validate_and_build() {
294        for bytes in [FIRST_INSTALL_ID, EMERGENCY_ID, SELF_SET_ID, SELF_RULE_BASE] {
295            assert!(Uuid7::from_bytes(bytes).is_some());
296        }
297        let full = AgentFacts {
298            baseline: Some(baseline()),
299            self_pins: vec![pin()],
300            enforcement: Enforcement::Palisade,
301        };
302        for facts in [AgentFacts::default(), full] {
303            for (builtin, mode) in [
304                (Builtin::FirstInstall, Mode::Audit),
305                (Builtin::Emergency, Mode::Enforce),
306            ] {
307                let policy = builtin.policy(&facts);
308                assert_eq!(policy.document().policy.mode, mode);
309                Engine::new(policy, EngineConfig::default()).unwrap();
310            }
311        }
312        assert_ne!(
313            Builtin::FirstInstall
314                .document(&AgentFacts::default())
315                .policy
316                .id,
317            Builtin::Emergency
318                .document(&AgentFacts::default())
319                .policy
320                .id
321        );
322        assert_ne!(self_rule_id(1), self_rule_id(2));
323    }

325    #[test]
326    fn the_emergency_policy_holds_only_the_pins_and_the_baseline() {
327        let facts = AgentFacts {
328            baseline: Some(baseline()),
329            self_pins: vec![pin(), pin()],
330            enforcement: Enforcement::Palisade,
331        };
332        let doc = Builtin::Emergency.policy(&facts).into_document();
333        assert_eq!(doc.rule_sets.len(), 2);
334        assert_eq!(doc.rule_sets[0].rules.len(), 2);
335        assert!(
336            doc.rule_sets[0]
337                .rules
338                .iter()
339                .all(|r| r.action == Action::Allow {})
340        );
341        assert!(palisade_policy::unknown::baseline::is_system_baseline(
342            &doc.rule_sets[1]
343        ));
344        // The first-install policy never takes them.
345        assert!(
346            Builtin::FirstInstall
347                .policy(&facts)
348                .document()
349                .rule_sets
350                .is_empty()
351        );
352    }

354    #[test]
355    fn native_only_runs_the_emergency_policy_in_audit_mode() {
356        let facts = AgentFacts {
357            enforcement: Enforcement::NativeOnly,
358            ..AgentFacts::default()
359        };
360        assert_eq!(
361            Builtin::Emergency.policy(&facts).document().policy.mode,
362            Mode::Audit
363        );
364        assert_eq!(Enforcement::default(), Enforcement::Palisade);
365    }

367    #[test]
368    fn inputs_that_do_not_build_are_left_out() {
369        // A pin that is not a hash fails validation: no pins at all.
370        let facts = AgentFacts {
371            self_pins: vec![SelfPin {
372                sha256: "not a hash".into(),
373                publisher: "Palisade".into(),
374            }],
375            baseline: Some(SystemBaseline::default()),
376            enforcement: Enforcement::Palisade,
377        };
378        let policy = Builtin::Emergency.policy(&facts);
379        assert!(policy.document().rule_sets.is_empty());
380        assert_eq!(policy.document().policy.mode, Mode::Enforce);
381    }
