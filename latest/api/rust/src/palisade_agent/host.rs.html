23/// Budget for a requested stop (service `STOP`, Ctrl-C).
24pub const STOP_BUDGET: Duration = Duration::from_secs(20);
25/// Budget when the machine is shutting down (`SHUTDOWN`, `PRESHUTDOWN`).
26pub const SHUTDOWN_BUDGET: Duration = Duration::from_secs(10);
27
28/// Why the agent is stopping, and how long it has.
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub struct StopRequest {
31    /// Why.
32    pub reason: StopReason,
33    /// The overall shutdown budget.
34    pub budget: Duration,
35}
36
37/// The source of a [`StopRequest`].
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub enum StopReason {
40    /// Ctrl-C or `SIGTERM` in console mode, or `STOP` from the service
41    /// control manager.
42    Requested,
43    /// The machine is shutting down.
44    SystemShutdown,
45    /// Whoever could request a stop has gone away.
46    ChannelClosed,
47}
48
49impl StopRequest {
50    /// A [`StopReason::Requested`] stop with [`STOP_BUDGET`].
51    pub const REQUESTED: Self = Self {
52        reason: StopReason::Requested,
53        budget: STOP_BUDGET,
54    };
55    /// A [`StopReason::SystemShutdown`] stop with [`SHUTDOWN_BUDGET`].
56    pub const SYSTEM_SHUTDOWN: Self = Self {
57        reason: StopReason::SystemShutdown,
58        budget: SHUTDOWN_BUDGET,
59    };
60}

62/// Builds the multi-threaded runtime: `min(4, cores)` workers unless
63/// configured, and a bounded blocking pool for file I/O and hashing.
64///
65/// Sizes that [`AgentConfig::validate`](crate::config::AgentConfig::validate)
66/// rejects are an `InvalidInput` error here too, rather than a panic in the
67/// runtime builder.
68pub fn build_runtime(runtime: &RuntimeConfig) -> io::Result<Runtime> {
69    let check = AgentConfig {
70        runtime: *runtime,
71        ..AgentConfig::default()
72    };
73    check
74        .validate()
75        .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e.to_string()))?;
76    let cores = std::thread::available_parallelism().map_or(1, usize::from);
77    tokio::runtime::Builder::new_multi_thread()
78        .worker_threads(runtime.effective_worker_threads(cores))
79        .max_blocking_threads(runtime.max_blocking_threads)
80        .thread_name("palisade-agent")
81        .enable_all()
82        .build()
83}

85/// Sends `tracing` records at or above `level` to standard error. Rotating
86/// log files and the event-log channel are still to come (#56). Safe to call
87/// more than once; only the first call has an effect.
88pub fn init_logging(level: LogLevel) {
89    let filter = match level {
90        LogLevel::Error => tracing::Level::ERROR,
91        LogLevel::Warn => tracing::Level::WARN,
92        LogLevel::Info => tracing::Level::INFO,
93        LogLevel::Debug => tracing::Level::DEBUG,
94        LogLevel::Trace => tracing::Level::TRACE,
95    };
96    let _ = tracing_subscriber::fmt()
97        .with_max_level(filter)
98        .with_writer(io::stderr)
99        .with_ansi(false)
100        .try_init();
101}

103/// Routes panics to the log, with a backtrace. Panics inside subsystem tasks
104/// are then contained by the supervisor; this only records them.
105pub fn install_panic_hook() {
106    static ONCE: Once = Once::new();
107    ONCE.call_once(|| {
108        std::panic::set_hook(Box::new(|info| {
109            let backtrace = std::backtrace::Backtrace::force_capture();
110            let thread = std::thread::current();
111            tracing::error!(
112                thread = thread.name().unwrap_or("unnamed"),
113                %info,
114                %backtrace,
115                "panic"
116            );
117        }));
118    });
119}

121/// The `config` subsystem: holds the effective configuration that other
122/// subsystems read. Reloading on `PARAMCHANGE` and the policy-delivered and
123/// local-override layers come with the policy cache (#61).
124#[derive(Debug)]
125pub struct ConfigSubsystem {
126    loaded: LoadedConfig,
127}
128
129impl ConfigSubsystem {
130    /// The subsystem's name.
131    pub const NAME: &'static str = "config";
132
133    /// Wraps a loaded configuration.
134    pub fn new(loaded: LoadedConfig) -> Self {
135        Self { loaded }
136    }
137
138    /// The effective configuration.
139    pub fn config(&self) -> &AgentConfig {
140        &self.loaded.config
141    }
142}
143
144impl Subsystem for ConfigSubsystem {
145    fn name(&self) -> &'static str {
146        Self::NAME
147    }
148
149    fn criticality(&self) -> Criticality {
150        Criticality::Critical
151    }
152
153    fn run(self: Arc<Self>, ctx: SubsystemContext) -> BoxFuture<Result<(), SubsystemError>> {
154        Box::pin(async move {
155            let config = &self.loaded.config;
156            tracing::info!(
157                source = %self.loaded.source.as_ref().map_or("built-in defaults".into(), |p| p.display().to_string()),
158                fail_mode = ?config.fail_mode,
159                safe_mode_behaviour = ?config.safe_mode_behaviour,
160                "configuration loaded"
161            );
162            ctx.set_ready();
163            ctx.stopped().await;
164            Ok(())
165        })
166    }
167}

169/// The subsystems the host registers. Subsystems from later issues (policy

176/// Starts `supervisor`, runs until `stop` resolves, then shuts down within the
177/// requested budget. `on_started` runs once supervision has begun.
178pub async fn run_until(
179    supervisor: Supervisor,
180    on_started: impl FnOnce(),
181    stop: impl Future<Output = StopRequest>,
182) -> (StopRequest, ShutdownReport) {
183    tracing::info!(order = ?supervisor.start_order(), "starting subsystems");
184    let handle = supervisor.start();
185    on_started();
186    let request = stop.await;
187    tracing::info!(reason = ?request.reason, budget = ?request.budget, "stopping");
188    let report = handle.shutdown(request.budget).await;
189    for (name, outcome) in &report.outcomes {
190        tracing::info!(subsystem = name, ?outcome, "subsystem stopped");
191    }
192    (request, report)
193}

195/// Grace period for blocking-pool threads after the supervisor has stopped.
196pub const RUNTIME_SHUTDOWN_GRACE: Duration = Duration::from_secs(1);
197
198/// Shuts the runtime down without waiting indefinitely for blocking-pool
199/// work that outlived its subsystem.
200pub fn finish_runtime(runtime: Runtime) {
201    runtime.shutdown_timeout(RUNTIME_SHUTDOWN_GRACE);
202}

204/// The default supervisor for a loaded configuration.
205pub fn supervisor(loaded: &LoadedConfig) -> Result<Supervisor, SupervisorError> {
206    Supervisor::new(builtin_subsystems(loaded), RestartPolicy::default())

209/// Console stop signals: Ctrl-C everywhere, and `SIGTERM` on Unix.
210///
211/// The OS handlers are installed by [`ConsoleSignals::register`], not when
212/// [`ConsoleSignals::recv`] is first polled. Register before starting the
213/// supervisor, so a signal sent as soon as the agent reports `Running` is
214/// already caught rather than killing the process.
215#[derive(Debug)]
216pub struct ConsoleSignals {
217    #[cfg(unix)]
218    interrupt: tokio::signal::unix::Signal,
219    #[cfg(unix)]
220    terminate: tokio::signal::unix::Signal,
221    #[cfg(windows)]
222    ctrl_c: tokio::signal::windows::CtrlC,
223}
224
225impl ConsoleSignals {
226    /// Installs the handlers. Must be called inside a tokio runtime.
227    pub fn register() -> io::Result<Self> {
228        #[cfg(unix)]
229        {
230            use tokio::signal::unix::{SignalKind, signal};
231            Ok(Self {
232                interrupt: signal(SignalKind::interrupt())?,
233                terminate: signal(SignalKind::terminate())?,
234            })
235        }
236        #[cfg(windows)]
237        {
238            Ok(Self {
239                ctrl_c: tokio::signal::windows::ctrl_c()?,
240            })
241        }
242    }
243
244    /// Resolves on the first signal.
245    pub async fn recv(mut self) -> StopRequest {
246        #[cfg(unix)]
247        tokio::select! {
248            _ = self.interrupt.recv() => {}
249            _ = self.terminate.recv() => {}
250        }
251        #[cfg(windows)]
252        self.ctrl_c.recv().await;
253        StopRequest::REQUESTED
254    }
255}

257/// Loads configuration and starts logging. Errors are printed to standard
258/// error, since logging may not be set up yet.
259pub fn prepare(args: &Args) -> Result<LoadedConfig, ExitCode> {
260    match config::load(args.config.as_deref()) {
261        Ok(loaded) => {
262            init_logging(loaded.config.log_level);
263            install_panic_hook();
264            Ok(loaded)
265        }
266        Err(e) => {
267            eprintln!("palisade-agent: {e}");
268            Err(ExitCode::FAILURE)
269        }
270    }
271}

273/// `--console`: runs in the foreground on any OS until Ctrl-C.
274pub fn run_console(args: &Args) -> ExitCode {
275    let loaded = match prepare(args) {
276        Ok(loaded) => loaded,
277        Err(code) => return code,
278    };
279    let supervisor = match supervisor(&loaded) {
280        Ok(supervisor) => supervisor,
281        Err(e) => {
282            tracing::error!(error = %e, "invalid subsystem set");
283            return ExitCode::FAILURE;
284        }
285    };
286    let runtime = match build_runtime(&loaded.config.runtime) {
287        Ok(runtime) => runtime,
288        Err(e) => {
289            tracing::error!(error = %e, "cannot start the async runtime");
290            return ExitCode::FAILURE;
291        }
292    };
293    let outcome = runtime.block_on(async {
294        let signals = ConsoleSignals::register()?;
295        Ok::<_, io::Error>(
296            run_until(
297                supervisor,
298                || tracing::info!("running in console mode; press Ctrl-C to stop"),
299                signals.recv(),
300            )
301            .await,
302        )
303    });
304    let report = match outcome {
305        Ok((_, report)) => report,
306        Err(e) => {
307            tracing::error!(error = %e, "cannot listen for Ctrl-C");
308            finish_runtime(runtime);
309            return ExitCode::FAILURE;
310        }
311    };
312    finish_runtime(runtime);
313    if report.is_clean() {
314        ExitCode::SUCCESS
315    } else {
316        ExitCode::FAILURE
317    }
318}
