1//! OS code integrity and rule-collection event-log ingestion (#58).
2//!
3//! The event log is the source of truth for what OS-native enforcement
4//! allowed, audited and blocked, and for which OS-native policies are
5//! active. Everything that reads the untrusted record text is portable,
6//! has no `unsafe` code, and is tested (and fuzzed) on every OS:
7//!
8//! - [`decode`] turns rendered event XML into a mapping `Record`.
9//! - [`correlate`] groups each decision with its per-signature records,
10//!   behind a token-bucket rate limit.
11//! - [`ingest`] runs the chain: decode, rate limit, correlate, map through
12//!   `palisade_proto::mapping`, add the process from the #59 process tree,
13//!   keep the policy inventory, and emit into a sink.
14//! - [`inventory`] is the per-device table of active OS-native policies
15//!   from policy activation events (3099).
16//!
17//! The Windows subscription (`EvtSubscribe` with persisted bookmarks)
18//! comes with the agent local store (ADR 0007), which keeps the bookmarks.
19
20#![forbid(unsafe_code)]
21
22pub mod correlate;
23pub mod decode;
24pub mod ingest;
25pub mod inventory;
26pub mod os_ids;

28/// The channels the agent reads, by the suffix of their resolved name
29/// (#58). The OS vendor prefix is resolved at run time by enumerating the
30/// channels, so it is not written here.
31pub const CHANNEL_SUFFIXES: &[ChannelSpec] = &[
32    ChannelSpec {
33        suffix: "-Windows-CodeIntegrity/Operational",
34        source: Source::CodeIntegrity,
35        optional: false,
36    },
37    ChannelSpec {
38        suffix: "-Windows-CodeIntegrity/Verbose",
39        source: Source::CodeIntegrity,
40        optional: true,
41    },
42    ChannelSpec {
43        suffix: "/EXE and DLL",
44        source: Source::RuleCollection,
45        optional: false,
46    },
47    ChannelSpec {
48        suffix: "/MSI and Script",
49        source: Source::RuleCollection,
50        // Absent on Server Core.
51        optional: true,
52    },
53    ChannelSpec {
54        suffix: "/Packaged app-Deployment",
55        source: Source::RuleCollection,
56        optional: true,
57    },
58    ChannelSpec {
59        suffix: "/Packaged app-Execution",
60        source: Source::RuleCollection,
61        optional: true,

63    ChannelSpec {
64        suffix: "PowerShellCore/Analytic",
65        source: Source::PowerShell,
66        optional: true,
67    },
68];

70/// Which mapping a channel's records go through.
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72pub enum Source {
73    /// The OS code integrity log.
74    CodeIntegrity,
75    /// The legacy rule-collection logs.
76    RuleCollection,
77    /// The PowerShell logs.
78    PowerShell,
79}
80
81/// One channel the agent reads.
82#[derive(Clone, Copy, Debug, PartialEq, Eq)]
83pub struct ChannelSpec {
84    /// The end of the channel's full name.
85    pub suffix: &'static str,
86    /// Its records' source.
87    pub source: Source,
88    /// Whether the agent starts without it (absent, or off by policy).
89    pub optional: bool,
90}
91
92/// The channel spec whose suffix `channel` ends with, ASCII
93/// case-insensitively.
94#[must_use]
95pub fn spec_for(channel: &str) -> Option<&'static ChannelSpec> {
96    let lower = channel.to_ascii_lowercase();
97    CHANNEL_SUFFIXES
98        .iter()
99        .find(|s| lower.ends_with(&s.suffix.to_ascii_lowercase()))
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    #[test]
107    fn channels_resolve_by_suffix() {
108        assert_eq!(
109            spec_for("X-Windows-CodeIntegrity/Operational").map(|s| s.source),
110            Some(Source::CodeIntegrity)
111        );
112        assert!(
113            spec_for("X-Windows-RuleCollections/MSI AND SCRIPT")
114                .unwrap()
115                .optional
116        );
117        assert!(
118            !spec_for("X-Windows-RuleCollections/EXE and DLL")
119                .unwrap()
120                .optional
121        );
122        assert_eq!(
123            spec_for("PowerShellCore/Analytic").map(|s| s.source),
124            Some(Source::PowerShell)
125        );
126        assert_eq!(spec_for("Security"), None);
127    }
