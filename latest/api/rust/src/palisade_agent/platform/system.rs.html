26/// The operating system's clocks.
27///
28/// The monotonic origin is the first reading in this process, so monotonic
29/// time is "time since the agent first read the clock".
30#[derive(Clone, Copy, Debug, Default)]
31pub struct SystemClock;
32
33fn origin() -> Instant {
34    static ORIGIN: OnceLock<Instant> = OnceLock::new();
35    *ORIGIN.get_or_init(Instant::now)
36}
37
38/// `time` as microseconds since the Unix epoch, saturating.
39fn utc_micros(time: SystemTime) -> i64 {
40    match time.duration_since(UNIX_EPOCH) {
41        Ok(after) => i64::try_from(after.as_micros()).unwrap_or(i64::MAX),
42        Err(before) => {
43            i64::try_from(before.duration().as_micros()).map_or(i64::MIN, i64::saturating_neg)
44        }
45    }
46}
47
48impl Clock for SystemClock {
49    fn now_utc(&self) -> UtcTime {
50        #[expect(
51            clippy::disallowed_methods,
52            reason = "the Clock implementation is the one place that reads the wall clock"
53        )]
54        let now = SystemTime::now();
55        UtcTime(utc_micros(now))
56    }
57
58    fn monotonic(&self) -> MonotonicTime {
59        let elapsed = origin().elapsed().as_micros();
60        MonotonicTime(u64::try_from(elapsed).unwrap_or(u64::MAX))
61    }
62}

68/// File reads through `std::fs`.
69///
70/// `open`, `read_at` and `metadata` (size, directory flag and times) work on
71/// every OS. Owner SIDs, security descriptors, alternate data streams, file
72/// IDs, volume serials and USNs need Windows APIs: here `owner_sid` and
73/// `security_descriptor` return [`ErrorKind::Unsupported`], `streams` does
74/// too on Windows, and the metadata leaves those fields `None`. On Windows,
75/// `windows::WindowsFileSystem` adds them through `palisade_fileid::win`.
76///
77/// This is not yet the hardened reader the privileged agent needs: like
78/// any `std::fs` open, it follows reparse points (symbolic links and
79/// junctions) and accepts UNC and device paths (threat T-agent-13). The
80/// file inventory replaces it with handle-based reads through
81/// `palisade_fileid::win` (#54, #60). Until then it serves development,
82/// tests and the simulator only.
83#[derive(Clone, Copy, Debug, Default)]
84pub struct StdFileSystem;
85
86/// A file opened by [`StdFileSystem`].
87#[derive(Debug)]
88pub struct StdFile {
89    file: File,
90    is_dir: bool,
91}

96        format!("{what} needs the Windows file-system backend (#54, #60)"),
97    )

100/// Whether `path` names a named data stream (`file:name` or
101/// `file:name:$DATA`, on Windows): those are data even on a directory.
102/// `file::$DATA` and other stream types, such as a directory's
103/// `::$INDEX_ALLOCATION`, do not count, nor does a drive letter's colon.
104pub(crate) fn names_a_stream(path: &str) -> bool {
105    if !cfg!(windows) {
106        return false;
107    }
108    let name_start = path.rfind(['\\', '/']).map_or(0, |i| i + 1);
109    let name = path.get(name_start..).unwrap_or_default();
110    let Some(i) = name.find(':').filter(|&i| !(name_start == 0 && i == 1)) else {
111        return false;
112    };
113    let rest = name.get(i + 1..).unwrap_or_default();
114    match rest.split_once(':') {
115        None => !rest.is_empty(),
116        Some((stream, kind)) => !stream.is_empty() && kind.eq_ignore_ascii_case("$DATA"),
117    }
118}

120/// The path must be absolute, so a relative path cannot be resolved against
121/// the agent's working directory.
122pub(crate) fn absolute(path: &str) -> Result<&Path> {
123    let p = Path::new(path);
124    if p.is_absolute() {
125        Ok(p)
126    } else {
127        Err(PlatformError::new(
128            ErrorKind::InvalidInput,
129            format!("not an absolute path: {path}"),
130        ))
131    }
132}

134impl FileSystem for StdFileSystem {
135    type File = StdFile;
136
137    fn open(&self, path: &str) -> Result<StdFile> {
138        let p = absolute(path)?;
139        let file = open_shared(p).map_err(|e| io_error(&e, path))?;
140        // On Windows a handle to a directory's named stream still reports
141        // the directory attribute, but the stream is data: decide from the
142        // path, as `palisade_core::test_util::MockFileSystem` does.
143        let is_dir =
144            !names_a_stream(path) && file.metadata().map_err(|e| io_error(&e, path))?.is_dir();
145        Ok(StdFile { file, is_dir })
146    }
147
148    fn owner_sid(&self, path: &str) -> Result<Option<String>> {
149        absolute(path)?;
150        Err(unsupported("the owner SID"))
151    }
152
153    fn security_descriptor(&self, path: &str) -> Result<Option<Vec<u8>>> {
154        absolute(path)?;
155        Err(unsupported("the security descriptor"))
156    }
157
158    fn streams(&self, path: &str) -> Result<Vec<StreamInfo>> {
159        let p = absolute(path)?;
160        fs::symlink_metadata(p).map_err(|e| io_error(&e, path))?;
161        if cfg!(windows) {
162            Err(unsupported("listing alternate data streams"))
163        } else {
164            // Other file systems have no alternate data streams.
165            Ok(Vec::new())
166        }
167    }
168}

170/// Opens for reading, letting other processes read, write and delete the
171/// file meanwhile, so the agent never blocks a program or an installer.
172#[cfg(windows)]
173pub(crate) fn open_shared(path: &Path) -> std::io::Result<File> {
174    use std::os::windows::fs::OpenOptionsExt;
175    const FILE_SHARE_READ: u32 = 0x1;
176    const FILE_SHARE_WRITE: u32 = 0x2;
177    const FILE_SHARE_DELETE: u32 = 0x4;
178    // Lets a directory be opened, so `metadata` works on directories as it
179    // does on other OSes.
180    const FILE_FLAG_BACKUP_SEMANTICS: u32 = 0x0200_0000;
181    // If the path names a named pipe (`\\.\pipe\...`, or a link to one),
182    // the pipe's server must not be able to impersonate the agent, which
183    // runs as LocalSystem: connect at the anonymous impersonation level.
184    const SECURITY_SQOS_PRESENT: u32 = 0x0010_0000;
185    const SECURITY_ANONYMOUS: u32 = 0;
186    fs::OpenOptions::new()
187        .read(true)
188        .share_mode(FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE)
189        .custom_flags(FILE_FLAG_BACKUP_SEMANTICS)
190        .security_qos_flags(SECURITY_SQOS_PRESENT | SECURITY_ANONYMOUS)
191        .open(path)
192}
193
194/// Opens for reading. The agent ships for Windows; on other OSes this
195/// backend serves development and tests.
196#[cfg(not(windows))]
197pub(crate) fn open_shared(path: &Path) -> std::io::Result<File> {
198    File::open(path)
199}

209impl OpenFile for StdFile {
210    fn metadata(&self) -> Result<FileMetadata> {
211        let m = self.file.metadata().map_err(|e| io_error(&e, "metadata"))?;
212        Ok(FileMetadata {
213            size: if self.is_dir { 0 } else { m.len() },
214            is_dir: self.is_dir,
215            created: m.created().ok().map(|t| UtcTime(utc_micros(t))),
216            modified: m.modified().ok().map(|t| UtcTime(utc_micros(t))),
217            file_id: None,
218            volume_serial: None,
219            usn: None,
220        })
221    }
222
223    fn read_at(&mut self, offset: u64, buf: &mut [u8]) -> Result<usize> {
224        if self.is_dir {
225            return Err(PlatformError::new(
226                ErrorKind::InvalidInput,
227                "cannot read a directory",
228            ));
229        }
230        // The OS cannot seek past `i64::MAX`, and no file is that long:
231        // such an offset is past the end, as it is for the mocks.
232        if i64::try_from(offset).is_err() {
233            return Ok(0);
234        }
235        self.file
236            .seek(SeekFrom::Start(offset))
237            .map_err(|e| io_error(&e, "seek"))?;
238        loop {
239            match self.file.read(buf) {
240                Ok(n) => return Ok(n),
241                Err(e) if e.kind() == std::io::ErrorKind::Interrupted => {}
242                Err(e) => return Err(io_error(&e, "read")),
243            }
244        }
245    }
246}

252/// The device and the agent's own process environment.
253///
254/// Everything except the interface addresses is captured once, when this is
255/// built, so every lookup in one evaluation sees the same values, and a
256/// lookup is a map read. Variable names are case-insensitive, as on Windows.
257///
258/// On Windows the host name and domain come from the registry and the
259/// version from the OS (`platform::windows`). Elsewhere (development only)
260/// the host name comes from the environment or `/etc/hostname`, and the
261/// domain and version are [`ErrorKind::Unsupported`]. The connection type is
262/// always [`ConnectionType::Unknown`]: how a network location is classified
263/// is defined by the network-location scope work (#38).
264#[derive(Clone, Debug)]
265pub struct StdEnvironment {
266    vars: BTreeMap<String, String>,
267    hostname: Option<String>,
268    /// `None`: the domain cannot be read on this OS.
269    domain: Option<Option<String>>,
270    os_version: Option<OsVersion>,
271}
272
273impl Default for StdEnvironment {
274    fn default() -> Self {
275        Self::capture()
276    }
277}
278
279fn fold(s: &str) -> String {
280    s.chars().flat_map(char::to_uppercase).collect()
281}
282
283impl StdEnvironment {
284    /// Captures the device facts and the current process environment.
285    /// Variables whose name or value is not valid Unicode are skipped.
286    #[must_use]
287    pub fn capture() -> Self {
288        let mut env = Self::from_vars(
289            std::env::vars_os()
290                .filter_map(|(k, v)| Some((k.into_string().ok()?, v.into_string().ok()?))),
291        );
292        #[cfg(windows)]
293        {
294            env.hostname = super::windows::hostname().or(env.hostname);
295            env.domain = Some(super::windows::domain());
296            env.os_version = Some(super::windows::os_version());
297        }
298        #[cfg(not(windows))]
299        if env.hostname.is_none() {
300            env.hostname = fs::read_to_string("/etc/hostname")
301                .ok()
302                .map(|s| s.trim().to_owned())
303                .filter(|s| !s.is_empty());
304        }
305        env
306    }
307
308    /// An environment with exactly these variables, and the host name from
309    /// `COMPUTERNAME` or `HOSTNAME` among them. On a duplicate name in
310    /// another letter case, the first wins.
311    pub fn from_vars(vars: impl IntoIterator<Item = (String, String)>) -> Self {
312        let mut map = BTreeMap::new();
313        for (k, v) in vars {
314            map.entry(fold(&k)).or_insert(v);
315        }
316        let hostname = ["COMPUTERNAME", "HOSTNAME"]
317            .iter()
318            .find_map(|n| map.get(*n).filter(|v| !v.is_empty()).cloned());
319        Self {
320            vars: map,
321            hostname,
322            domain: None,
323            os_version: None,
324        }
325    }
326}
327
328impl Environment for StdEnvironment {
329    fn hostname(&self) -> Result<String> {
330        self.hostname
331            .clone()
332            .ok_or_else(|| PlatformError::new(ErrorKind::Unsupported, "the host name is not known"))
333    }
334
335    fn domain(&self) -> Result<Option<String>> {
336        self.domain.clone().ok_or_else(|| {
337            PlatformError::new(ErrorKind::Unsupported, "the domain is only read on Windows")
338        })
339    }
340
341    fn os_version(&self) -> Result<OsVersion> {
342        self.os_version.ok_or_else(|| {
343            PlatformError::new(
344                ErrorKind::Unsupported,
345                "the OS version is only read on Windows",
346            )
347        })
348    }
349
350    fn architecture(&self) -> Architecture {
351        match std::env::consts::ARCH {
352            "x86_64" => Architecture::X64,
353            "aarch64" => Architecture::Arm64,
354            "x86" => Architecture::X86,
355            _ => Architecture::Other,
356        }
357    }
358
359    fn addresses(&self) -> Result<Vec<InterfaceAddress>> {
360        let interfaces = if_addrs::get_if_addrs().map_err(|e| io_error(&e, "interfaces"))?;
361        Ok(interfaces
362            .into_iter()
363            // Up, or a virtual or tunnel interface that reports no state.
364            .filter(|i| {
365                !i.is_loopback()
366                    && matches!(
367                        i.oper_status,
368                        if_addrs::IfOperStatus::Up | if_addrs::IfOperStatus::Unknown
369                    )
370            })
371            .map(|i| {
372                let (addr, prefix_len) = match i.addr {
373                    if_addrs::IfAddr::V4(a) => (IpAddr::V4(a.ip), a.prefixlen),
374                    if_addrs::IfAddr::V6(a) => (IpAddr::V6(a.ip), a.prefixlen),
375                };
376                InterfaceAddress { addr, prefix_len }
377            })
378            .collect())
379    }
380
381    fn connection_type(&self) -> Result<ConnectionType> {
382        Ok(ConnectionType::Unknown)
383    }
384
385    fn var(&self, name: &str) -> Option<String> {
386        self.vars.get(&fold(name)).cloned()
387    }
