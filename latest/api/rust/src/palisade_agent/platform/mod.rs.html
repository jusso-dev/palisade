1//! The agent's implementations of the `palisade_core::platform` traits.
2//!
3//! These use only the standard library and safe wrapper crates, so they
4//! build and run on every OS, and their contract tests
5//! (`palisade_core::test_util::contract`) run on every CI runner, Windows
6//! included. Where Windows needs something else, the difference lives in
7//! `windows` (compiled on Windows only).
8//!
9//! Not here yet, because each needs Windows FFI (ADR 0011) or comes with the
10//! M1 code that consumes it:
11//!
12//! - `FileSystem` owner SIDs, security descriptors, alternate data streams,
13//!   file IDs and USNs on Windows: they return [`ErrorKind::Unsupported`]
14//!   until the agent is wired to `palisade_fileid::win` (#54) by the file
15//!   inventory (#60).
16//! - `IdentityProvider`, `ProcessInfo` and `SignatureVerifier`: with the
17//!   process tracker, which reads tokens and resolves each image's identity
18//!   (#59).
19//! - `Registry`: with registry conditions in the evaluation context (#59).
20//!
21//! Until then, the simulator (`palisade-agent --simulate`, feature
22//! `test-util`) runs the agent's event building on the mocks.
23
24mod system;
25#[cfg(windows)]
26pub mod windows;

32/// Converts an I/O error, keeping its kind.
33pub(crate) fn io_error(error: &std::io::Error, context: &str) -> PlatformError {
34    let kind = match error.kind() {
35        std::io::ErrorKind::NotFound => ErrorKind::NotFound,
36        std::io::ErrorKind::PermissionDenied => ErrorKind::AccessDenied,
37        std::io::ErrorKind::InvalidInput | std::io::ErrorKind::InvalidFilename => {
38            ErrorKind::InvalidInput
39        }
40        std::io::ErrorKind::Unsupported => ErrorKind::Unsupported,
41        _ => ErrorKind::Other,
42    };
43    PlatformError::new(kind, format!("{context}: {error}"))
44}
