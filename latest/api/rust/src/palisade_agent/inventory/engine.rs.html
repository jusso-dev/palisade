1//! The inventory engine: the initial scan, reconciliation and change
2//! tracking of one volume, over a [`VolumeSource`] and an [`Identifier`].
3//!
4//! The source is the OS: on Windows the master file table and the change
5//! journal (`palisade_fileid::win::journal`); in tests, a scripted volume.
6//! The engine itself is portable, so every path through it is tested on any
7//! OS.
8//!
9//! - [`Engine::scan`] lists every file of a volume in batches, keeps the
10//!   candidates ([`super::scope::InventoryScope::candidate`]), and records
11//!   them. It saves a [`ScanCheckpoint`] every [`CHECKPOINT_FILES`] files
12//!   or [`CHECKPOINT_INTERVAL`], so a restart resumes instead of starting
13//!   again, and asks the [`Throttle`] before every batch: the resource
14//!   budgets (#65) plug in there.
15//! - [`Engine::poll`] reads the journal from the saved cursor and applies
16//!   the changes ([`super::track`]). If the journal was replaced, or
17//!   wrapped past the cursor, it reconciles the volume instead (a scan
18//!   with [`Discovery::Rescan`] that marks files no longer listed as
19//!   deleted) and reports that it did.
20

34/// A checkpoint is saved at least every this many files.
35pub const CHECKPOINT_FILES: u64 = 10_000;
36/// A checkpoint is saved at least this often.
37pub const CHECKPOINT_INTERVAL: Duration = Duration::from_secs(30);

39/// One file or folder a volume listing returns.
40#[derive(Debug, Clone, PartialEq, Eq)]
41pub struct Listed {
42    /// Its 128-bit ID.
43    pub file_id: [u8; 16],
44    /// Its full path, if the source could resolve it.
45    pub path: Option<String>,
46    /// Whether it is a folder.
47    pub directory: bool,
48}

50/// A failed read of the volume.
51#[derive(Debug, Clone, PartialEq, Eq)]
52pub enum SourceError {
53    /// The saved cursor can no longer be read: the journal was deleted,
54    /// replaced or wrapped. The volume must be reconciled.
55    CursorLost,
56    /// Any other failure, with its reason.
57    Other(String),
58}
59
60impl std::fmt::Display for SourceError {
61    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
62        match self {
63            Self::CursorLost => f.write_str("the change journal cursor is no longer readable"),
64            Self::Other(m) => f.write_str(m),
65        }
66    }
67}

69/// One volume, as the engine reads it.
70pub trait VolumeSource {
71    /// The volume's serial number.
72    fn serial(&self) -> u64;
73    /// The journal's state, or `None` if the volume keeps no journal (FAT,
74    /// exFAT, network): it is then rescanned on a schedule instead.
75    fn journal(&mut self) -> Result<Option<JournalData>, SourceError>;
76    /// One batch of the volume's files from `start` (0 for the first), or
77    /// `None` when every file has been listed. The cursor returned must be
78    /// passed back to continue.
79    fn list(
80        &mut self,
81        start: u64,
82        high_usn: i64,
83    ) -> Result<Option<(u64, Vec<Listed>)>, SourceError>;
84    /// Changes from `cursor`, and the cursor to continue from.
85    fn changes(&mut self, cursor: JournalCursor) -> Result<(i64, Vec<Change>), SourceError>;
86    /// The full path of the file with `file_id`, from its ID, never a
87    /// path a user could have swapped. `None` if it is gone.
88    fn resolve(&mut self, file_id: [u8; 16]) -> Option<String>;
89}

91/// What identifying a file found.
92#[derive(Debug, Clone, PartialEq, Eq)]
93pub enum Identified {
94    /// The file's identity, and whether its location is user-writable
95    /// (`None` if that could not be assessed).
96    Known {
97        /// Its identity summary.
98        summary: Box<IdentitySummary>,
99        /// Its location's writability.
100        user_writable: Option<bool>,
101    },
102    /// It was not read, on purpose (a cloud placeholder, or a deferral).
103    Deferred,
104    /// It could not be read.
105    Unreadable(String),
106    /// It is gone.
107    Gone,
108}

118/// The seam for the resource budgets (#65).
119pub trait Throttle {
120    /// Waits, if needed, before the next batch of about `files` files.
121    /// Returns `false` to stop the work for now (a stop was requested, or
122    /// the budget says not now); the checkpoint lets it resume.
123    fn admit(&mut self, files: usize) -> bool;
124}
125
126/// A throttle that admits everything.
127#[derive(Debug, Default, Clone, Copy)]
128pub struct Unthrottled;
129
130impl Throttle for Unthrottled {
131    fn admit(&mut self, _files: usize) -> bool {
132        true
133    }
134}

136/// What a scan did.
137#[derive(Debug, Clone, Default, PartialEq, Eq)]
138pub struct ScanReport {
139    /// Files listed (candidates or not).
140    pub listed: u64,
141    /// Entries added.
142    pub added: u64,
143    /// Entries marked deleted, by a reconciliation.
144    pub deleted: u64,
145    /// The scan finished; `false` if the throttle stopped it.
146    pub complete: bool,
147    /// Changes to report (#62).
148    pub deltas: Vec<(EntryKey, DeltaKind)>,
149}

151/// What one poll did.
152#[derive(Debug, Clone, PartialEq, Eq)]
153pub enum PollOutcome {
154    /// Changes were read and applied.
155    Applied {
156        /// Changes to report (#62).
157        deltas: Vec<(EntryKey, DeltaKind)>,
158    },
159    /// The volume keeps no journal: rescan it on the periodic schedule.
160    NoJournal,
161    /// The cursor was lost, so the volume was reconciled.
162    Reconciled(ScanReport),
163    /// The volume has never been scanned, or its scan was interrupted:
164    /// scan it first.
165    NeedsScan,
166}

187    /// Scans `source` into `store`, resuming from a saved checkpoint.
188    /// `discovery` is [`Discovery::InitialScan`] for the first scan and
189    /// [`Discovery::Rescan`] for a reconciliation, which also marks entries
190    /// no longer listed as deleted.
191    #[allow(clippy::too_many_arguments)]
192    pub fn scan(
193        &mut self,
194        source: &mut dyn VolumeSource,
195        store: &mut dyn InventoryStore,
196        identifier: &mut dyn Identifier,
197        throttle: &mut dyn Throttle,
198        clock: &dyn Fn() -> (UtcTime, MonotonicTime),
199        discovery: Discovery,
200    ) -> Result<ScanReport, ScanError> {
201        let serial = source.serial();
202        let (now, mut last_save) = clock();
203        let mut checkpoint = match store.checkpoint(serial)? {
204            Some(c) => c,
205            None => {
206                let start = match source.journal()? {
207                    Some(j) => JournalCursor {
208                        journal_id: j.journal_id,
209                        next_usn: j.next_usn,
210                    },
211                    None => JournalCursor {
212                        journal_id: 0,
213                        next_usn: i64::MAX,
214                    },
215                };
216                let c = ScanCheckpoint {
217                    next_reference: 0,
218                    start,
219                    files_seen: 0,
220                };
221                store.set_checkpoint(serial, Some(c))?;
222                c
223            }
224        };
225        let mut report = ScanReport::default();
226        let mut saved_at_files = checkpoint.files_seen;
227        // A reconciliation that resumes has lost the set of IDs listed
228        // before the restart, so it never marks deletions from a partial
229        // view: it rescans from the start instead.
230        let mut seen = BTreeSet::new();
231        if discovery == Discovery::Rescan && checkpoint.next_reference != 0 {
232            checkpoint.next_reference = 0;
233            checkpoint.files_seen = 0;
234        }
235        loop {
236            if !throttle.admit(1024) {
237                store.set_checkpoint(serial, Some(checkpoint))?;
238                return Ok(report);
239            }
240            let Some((next, batch)) =
241                source.list(checkpoint.next_reference, checkpoint.start.next_usn)?
242            else {
243                break;
244            };
245            for item in batch {
246                report.listed += 1;
247                checkpoint.files_seen += 1;
248                let key = EntryKey {
249                    volume_serial: serial,
250                    file_id: item.file_id,
251                };
252                if item.directory {
253                    continue;
254                }
255                seen.insert(key);
256                self.record_listed(store, identifier, key, item, now, discovery, &mut report)?;
257            }
258            checkpoint.next_reference = next;
259            let (_, mono) = clock();
260            let elapsed = mono.0.saturating_sub(last_save.0);
261            if checkpoint.files_seen - saved_at_files >= CHECKPOINT_FILES
262                || elapsed >= u64::try_from(CHECKPOINT_INTERVAL.as_micros()).unwrap_or(u64::MAX)
263            {
264                store.set_checkpoint(serial, Some(checkpoint))?;
265                saved_at_files = checkpoint.files_seen;
266                last_save = mono;
267            }
268        }
269        if discovery == Discovery::Rescan {
270            for key in store.keys_on(serial)? {
271                if seen.contains(&key) {
272                    continue;
273                }
274                if let Some(mut e) = store.get(&key)?
275                    && e.state != EntryState::Deleted
276                {
277                    e.state = EntryState::Deleted;
278                    e.deleted_at = Some(now);
279                    e.stale = true;
280                    store.put(e)?;
281                    report.deleted += 1;
282                    report.deltas.push((key, DeltaKind::Deleted));
283                }
284            }
285        }
286        // The scan is complete: changes since it started come from the
287        // journal.
288        if checkpoint.start.journal_id != 0 || checkpoint.start.next_usn != i64::MAX {
289            store.set_cursor(serial, checkpoint.start)?;
290        } else {
291            store.clear_cursor(serial)?;
292        }
293        store.set_checkpoint(serial, None)?;
294        report.complete = true;
295        Ok(report)
296    }

298    #[allow(clippy::too_many_arguments)]
299    fn record_listed(
300        &self,
301        store: &mut dyn InventoryStore,
302        identifier: &mut dyn Identifier,
303        key: EntryKey,
304        item: Listed,
305        now: UtcTime,
306        discovery: Discovery,
307        report: &mut ScanReport,
308    ) -> Result<(), StoreError> {
309        let Some(path) = item.path else {
310            return Ok(());
311        };
312        if self.scope.excludes(&path) {
313            return Ok(());
314        }
315        let existing = store.get(&key)?;
316        let user_writable = existing
317            .as_ref()
318            .is_none_or(InvEntry::in_user_writable_location);
319        let name = path.rsplit('\\').next().unwrap_or(&path);
320        if self.scope.candidate(name, user_writable).is_none() {
321            return Ok(());
322        }
323        match existing {
324            Some(mut e) => {
325                // Known: keep its identity unless it moved or was deleted.
326                let moved = e.move_to(path);
327                if e.state == EntryState::Deleted {
328                    e.state = EntryState::Present;
329                    e.deleted_at = None;
330                    e.stale = true;
331                }
332                if moved {
333                    e.user_writable_location = None;
334                    report.deltas.push((key, DeltaKind::Renamed));
335                }
336                store.put(e)
337            }
338            None => {
339                let mut e = InvEntry::new(key, path, now, discovery);
340                identify_into(&mut e, identifier);
341                // Phase 1: a file whose content cannot run code is not kept.
342                if e.identity
343                    .as_ref()
344                    .is_some_and(|i| i.executable_content == Some(false))
345                {
346                    return Ok(());
347                }
348                store.put(e)?;
349                report.added += 1;
350                report.deltas.push((key, DeltaKind::Added));
351                Ok(())
352            }
353        }
354    }

356    /// Reads and applies the changes since the saved cursor.
357    #[allow(clippy::too_many_arguments)]
358    pub fn poll(
359        &mut self,
360        source: &mut dyn VolumeSource,
361        store: &mut dyn InventoryStore,
362        identifier: &mut dyn Identifier,
363        throttle: &mut dyn Throttle,
364        clock: &dyn Fn() -> (UtcTime, MonotonicTime),
365    ) -> Result<PollOutcome, ScanError> {
366        let serial = source.serial();
367        if store.checkpoint(serial)?.is_some() {
368            return Ok(PollOutcome::NeedsScan);
369        }
370        let Some(cursor) = store.cursor(serial)? else {
371            return Ok(match source.journal()? {
372                Some(_) if store.keys_on(serial)?.is_empty() => PollOutcome::NeedsScan,
373                Some(_) => self.reconcile(source, store, identifier, throttle, clock)?,
374                None => PollOutcome::NoJournal,
375            });
376        };
377        let journal = source.journal()?;
378        let resumable = journal.is_some_and(|j| j.can_resume(cursor.journal_id, cursor.next_usn));
379        if !resumable {
380            return self.reconcile(source, store, identifier, throttle, clock);
381        }
382        let (next, changes) = match source.changes(cursor) {
383            Ok(c) => c,
384            Err(SourceError::CursorLost) => {
385                return self.reconcile(source, store, identifier, throttle, clock);
386            }
387            Err(e) => return Err(e.into()),
388        };
389        let (now, mono) = clock();
390        for change in changes {
391            let key = change.key;
392            let invalidation = self.coalescer.push(change, mono);
393            track::invalidate(store, &key, invalidation)?;
394        }
395        // The cursor moves only once every change it covers is held or
396        // applied: the invalidations above are already in the store.
397        store.set_cursor(
398            serial,
399            JournalCursor {
400                journal_id: cursor.journal_id,
401                next_usn: next,
402            },
403        )?;
404        let mut deltas = Vec::new();
405        for mut c in self.coalescer.ready(mono) {
406            if c.path.is_none() && !c.flags.contains(track::ChangeFlags::DELETED) {
407                c.path = source.resolve(c.key.file_id);
408            }
409            let applied = track::apply(store, &c, now)?;
410            if let Some(d) = applied.delta {
411                deltas.push((c.key, d));
412            }
413            self.follow_up(source, store, identifier, &c, applied.action)?;
414        }
415        Ok(PollOutcome::Applied { deltas })
416    }

418    fn follow_up(
419        &self,
420        source: &mut dyn VolumeSource,
421        store: &mut dyn InventoryStore,
422        identifier: &mut dyn Identifier,
423        change: &track::Coalesced,
424        action: Action,
425    ) -> Result<(), StoreError> {
426        match action {
427            Action::Identify | Action::ReassessLocation => {
428                if let Some(mut e) = store.get(&change.key)? {
429                    identify_into(&mut e, identifier);
430                    store.put(e)?;
431                }
432            }
433            Action::Directory => {
434                if let Some(folder) = source.resolve(change.key.file_id) {
435                    track::invalidate_under(store, change.key.volume_serial, &folder)?;
436                }
437            }
438            Action::ResolvePath | Action::None => {}
439        }
440        Ok(())
441    }

443    fn reconcile(
444        &mut self,
445        source: &mut dyn VolumeSource,
446        store: &mut dyn InventoryStore,
447        identifier: &mut dyn Identifier,
448        throttle: &mut dyn Throttle,
449        clock: &dyn Fn() -> (UtcTime, MonotonicTime),
450    ) -> Result<PollOutcome, ScanError> {
451        let serial = source.serial();
452        tracing::warn!(
453            volume = serial,
454            "change journal cursor lost; reconciling the volume"
455        );
456        // Held changes are applied first, so none is lost.
457        let (now, _) = clock();
458        for c in self.coalescer.drain() {
459            let applied = track::apply(store, &c, now)?;
460            self.follow_up(source, store, identifier, &c, applied.action)?;
461        }
462        store.clear_cursor(serial)?;
463        let report = self.scan(
464            source,
465            store,
466            identifier,
467            throttle,
468            clock,
469            Discovery::Rescan,
470        )?;
471        // A reconciliation re-reads known files whose stamps may have
472        // changed while the journal was not read.
473        for key in store.keys_on(serial)? {
474            if let Some(mut e) = store.get(&key)?
475                && e.state == EntryState::Present
476            {
477                identify_into(&mut e, identifier);
478                store.put(e)?;
479            }
480        }
481        Ok(PollOutcome::Reconciled(report))
482    }
483}

485/// Identifies `e` and records the result on it.
486fn identify_into(e: &mut InvEntry, identifier: &mut dyn Identifier) {
487    match identifier.identify(e.key) {
488        Identified::Known {
489            summary,
490            user_writable,
491        } => {
492            e.identity = Some(*summary);
493            e.user_writable_location = user_writable;
494            e.state = EntryState::Present;
495            e.stale = false;
496        }
497        Identified::Deferred => {
498            e.state = EntryState::Deferred;
499            e.stale = true;
500        }
501        Identified::Unreadable(reason) => {
502            tracing::debug!(path = %e.path, %reason, "inventory could not read a file");
503            e.state = EntryState::Unreadable;
504            e.stale = true;
505        }
506        Identified::Gone => {
507            e.state = EntryState::Deleted;
508            e.stale = true;
509        }
510    }
511}

513/// Why a scan or poll failed.
514#[derive(Debug, Clone, PartialEq, Eq)]
515pub enum ScanError {
516    /// The store failed.
517    Store(StoreError),
518    /// The volume could not be read.
519    Source(SourceError),
520}
521
522impl std::fmt::Display for ScanError {
523    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
524        match self {
525            Self::Store(e) => write!(f, "inventory store: {e}"),
526            Self::Source(e) => write!(f, "volume read: {e}"),
527        }
528    }
529}
