1//! What the process monitor emits, and where it goes (#59).
2//!
3//! The event pipeline (#63, #197) owns the canonical envelope (sequence
4//! numbers, the hash chain, the tenant and host), so the monitor emits
5//! [`TelemetryEvent`]s into a [`TelemetrySink`] and the pipeline turns them
6//! into canonical events. [`process_fields`] builds the canonical
7//! `process` field set from the tree, so that translation is shared.
8//!
9//! Until the pipeline exists, [`ChannelSink`] is the seam: a bounded
10//! channel that never blocks the producer. A full channel drops the event
11//! and counts it ([`Counters::dropped`]); a drop is never silent.
12
13#![forbid(unsafe_code)]

26/// The kind of an image load.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum ImageKind {
29    /// A kernel-mode image (a driver).
30    Driver,
31    /// A user-mode image other than the process's main image.
32    Library,
33    /// The process's main image.
34    Executable,
35}

37/// An image load the monitor reports.
38#[derive(Clone, Debug, PartialEq, Eq)]
39pub struct ImageLoad {
40    /// The process the image was mapped into; `None` for drivers and for
41    /// processes not in the tree.
42    pub process: Option<ProcessKey>,
43    /// The kind.
44    pub kind: ImageKind,
45    /// The path as the OS reported it.
46    pub image_path: String,
47    /// The path with the device resolved to a drive, where possible.
48    pub image_path_normalised: Option<String>,
49    /// The base address.
50    pub base: u64,
51    /// The mapped size.
52    pub size: u64,
53    /// The image header checksum, for a quick identity pre-match.
54    pub checksum: u32,
55    /// The image header time stamp, for a quick identity pre-match.
56    pub time_date_stamp: u32,
57    /// When it was loaded.
58    pub time: UtcTime,
59}

61/// An event from the process monitor.
62#[derive(Clone, Debug, PartialEq, Eq)]
63pub enum TelemetryEvent {
64    /// A process started, with its record and ancestry.
65    ProcessStart {
66        /// The process.
67        process: Box<ProcInfo>,
68        /// Its ancestors, parent first.
69        ancestry: Vec<ProcessKey>,
70    },
71    /// A process stopped.
72    ProcessStop {
73        /// The process.
74        process: Box<ProcInfo>,
75    },
76    /// An image was loaded.
77    ImageLoad(ImageLoad),
78    /// A process was created by a process other than its recorded parent:
79    /// re-parenting, or parent process ID spoofing. Advisory only.
80    ParentSpoof {
81        /// The new process.
82        child: ProcessKey,
83        /// The parent ID the OS recorded.
84        parent_pid: u32,
85        /// The process that created it.
86        creator: ProcessKey,
87    },
88}

90/// Counters shared by the monitor and its health report.
91#[derive(Debug, Default)]
92pub struct Counters {
93    /// Events the sink could not accept.
94    pub dropped: AtomicU64,
95    /// Raw records the consumer could not queue for decoding.
96    pub queue_dropped: AtomicU64,
97    /// Raw records that did not decode.
98    pub decode_errors: AtomicU64,
99    /// Events the OS reports as lost from the session.
100    pub os_lost: AtomicU64,
101    /// Events emitted.
102    pub emitted: AtomicU64,
103}
104
105impl Counters {
106    /// Adds one to `counter`.
107    pub fn bump(counter: &AtomicU64) {
108        counter.fetch_add(1, Ordering::Relaxed);
109    }
110
111    /// The value of `counter`.
112    pub fn get(counter: &AtomicU64) -> u64 {
113        counter.load(Ordering::Relaxed)
114    }
115}

117/// Where the monitor sends its events. Implementations must not block.
118pub trait TelemetrySink: Send + Sync {
119    /// Offers an event. Returns `false` if it was not accepted; the caller
120    /// counts the drop.
121    fn offer(&self, event: TelemetryEvent) -> bool;
122}
123
124/// A bounded channel: the seam for the event pipeline (#63).
125#[derive(Debug, Clone)]
126pub struct ChannelSink {
127    tx: SyncSender<TelemetryEvent>,
128}
129
130/// The default capacity of [`ChannelSink::bounded`]: #59's identity queue
131/// size.
132pub const DEFAULT_CHANNEL_CAPACITY: usize = 50_000;
133
134impl ChannelSink {
135    /// A channel holding at most `capacity` events, and its receiver.
136    #[must_use]
137    pub fn bounded(capacity: usize) -> (Self, Receiver<TelemetryEvent>) {
138        let (tx, rx) = sync_channel(capacity);
139        (Self { tx }, rx)
140    }
141}
142
143impl TelemetrySink for ChannelSink {
144    fn offer(&self, event: TelemetryEvent) -> bool {
145        match self.tx.try_send(event) {
146            Ok(()) => true,
147            Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) => false,
148        }
149    }
150}

152/// A sink that accepts nothing: every event counts as dropped. The host
153/// uses it until the event pipeline (#63) consumes the monitor's events,
154/// so the counters show the volume that will flow.
155#[derive(Debug, Clone, Copy, Default)]
156pub struct DiscardSink;
157
158impl TelemetrySink for DiscardSink {
159    fn offer(&self, _event: TelemetryEvent) -> bool {
160        false
161    }
162}
163
164impl<T: TelemetrySink + ?Sized> TelemetrySink for Arc<T> {
165    fn offer(&self, event: TelemetryEvent) -> bool {
166        (**self).offer(event)
167    }
168}

178fn integrity(level: IntegrityLevel) -> events::IntegrityLevel {
179    match level {
180        IntegrityLevel::Untrusted => events::IntegrityLevel::Untrusted,
181        IntegrityLevel::Low => events::IntegrityLevel::Low,
182        IntegrityLevel::Medium => events::IntegrityLevel::Medium,
183        IntegrityLevel::MediumPlus => events::IntegrityLevel::MediumPlus,
184        IntegrityLevel::High => events::IntegrityLevel::High,
185        IntegrityLevel::System => events::IntegrityLevel::System,
186        IntegrityLevel::Protected => events::IntegrityLevel::Protected,
187    }
188}

190fn elevation(e: Elevation) -> TokenElevationType {
191    match e {
192        Elevation::Default => TokenElevationType::Default,
193        Elevation::Full => TokenElevationType::Full,
194        Elevation::Limited => TokenElevationType::Limited,
195    }
196}

202/// The canonical fields of one process, without its parent. A value that
203/// does not fit the schema is left out, never cut silently.
204fn parent_fields(p: &ProcInfo) -> ParentProcess {
205    ParentProcess {
206        pid: Some(p.key.pid),
207        process_key: Name::new(p.process_key()).ok(),
208        image_path: long(&p.image_path),
209        image_path_normalised: p.image_path_normalised.as_deref().and_then(long),
210        cmdline: p.command_line.text().and_then(long),
211        integrity_level: p.token.integrity.map(integrity),
212        token_elevation_type: p.token.elevation.map(elevation),
213        owner_sid: p.token.user_sid.as_deref().and_then(|s| s.parse().ok()),
214        ..ParentProcess::default()
215    }
216}

218/// The canonical `process` field set for process `key`: the process, its
219/// parent and a summary of its grandparent, where the tree knows them.
220#[must_use]
221pub fn process_fields(tree: &ProcessTree, key: ProcessKey) -> Option<Process> {
222    let p = tree.get(key)?;
223    let own = parent_fields(p);
224    let parent = p.parent.and_then(|k| tree.get(k));
225    let grandparent = parent.and_then(|q| q.parent).and_then(|k| tree.get(k));
226    Some(Process {
227        pid: own.pid,
228        process_key: own.process_key,
229        image_path: own.image_path,
230        image_path_normalised: own.image_path_normalised,
231        cmdline: own.cmdline,
232        integrity_level: own.integrity_level,
233        token_elevation_type: own.token_elevation_type,
234        owner_sid: own.owner_sid,
235        parent: parent.map(|q| Box::new(parent_fields(q))),
236        grandparent: grandparent.map(|g| ProcessSummary {
237            pid: Some(g.key.pid),
238            process_key: Name::new(key_text(g.key)).ok(),
239            image_path: long(&g.image_path),
240        }),
241        ..Process::default()
242    })
243}

246mod tests {
247    use std::time::Duration;
248
249    use super::super::tree::CommandLine;

252    fn key(pid: u32) -> ProcessKey {
253        ProcessKey {
254            pid,
255            created: UtcTime(i64::from(pid)),
256        }
257    }
258
259    #[test]
260    fn process_fields_include_parent_and_grandparent() {
261        let mut t = ProcessTree::new(10, Duration::from_secs(1));
262        let mut g = ProcInfo::new(key(1), 0, "C:\\g.exe".into());
263        g.token.integrity = Some(IntegrityLevel::System);
264        t.insert(g);
265        let mut p = ProcInfo::new(key(2), 1, "C:\\p.exe".into());
266        p.parent = Some(key(1));
267        p.token.elevation = Some(Elevation::Full);
268        p.token.user_sid = Some("S-1-5-18".into());
269        t.insert(p);
270        let mut c = ProcInfo::new(key(3), 2, "\\Device\\HarddiskVolume3\\c.exe".into());
271        c.parent = Some(key(2));
272        c.image_path_normalised = Some("C:\\c.exe".into());
273        c.command_line = CommandLine::Captured {
274            text: "c.exe /x".into(),
275            truncated: false,
276        };
277        for (level, want) in [
278            (IntegrityLevel::Untrusted, events::IntegrityLevel::Untrusted),
279            (IntegrityLevel::Low, events::IntegrityLevel::Low),
280            (IntegrityLevel::Medium, events::IntegrityLevel::Medium),
281            (
282                IntegrityLevel::MediumPlus,
283                events::IntegrityLevel::MediumPlus,
284            ),
285            (IntegrityLevel::High, events::IntegrityLevel::High),
286            (IntegrityLevel::Protected, events::IntegrityLevel::Protected),
287        ] {
288            assert_eq!(integrity(level), want);
289        }
290        c.token.elevation = Some(Elevation::Limited);
291        t.insert(c);
292        let f = process_fields(&t, key(3)).unwrap();
293        assert_eq!(f.pid, Some(3));
294        assert_eq!(f.process_key.unwrap().as_str(), "3-3");
295        assert_eq!(f.cmdline.unwrap().as_str(), "c.exe /x");
296        assert_eq!(f.image_path_normalised.unwrap().as_str(), "C:\\c.exe");
297        assert_eq!(f.token_elevation_type, Some(TokenElevationType::Limited));
298        let parent = f.parent.unwrap();
299        assert_eq!(parent.pid, Some(2));
300        assert_eq!(parent.token_elevation_type, Some(TokenElevationType::Full));
301        assert_eq!(parent.owner_sid.unwrap().as_str(), "S-1-5-18");
302        assert_eq!(f.grandparent.unwrap().pid, Some(1));
303        let g = process_fields(&t, key(1)).unwrap();
304        assert_eq!(g.integrity_level, Some(events::IntegrityLevel::System));
305        assert!(g.parent.is_none());
306        assert!(process_fields(&t, key(9)).is_none());
307        // A default-elevation token maps too.
308        assert_eq!(elevation(Elevation::Default), TokenElevationType::Default);
309    }

311    #[test]
312    fn channel_sink_is_bounded_and_never_blocks() {
313        let (sink, rx) = ChannelSink::bounded(1);
314        let ev = || TelemetryEvent::ParentSpoof {
315            child: key(1),
316            parent_pid: 2,
317            creator: key(3),
318        };
319        assert!(sink.offer(ev()));
320        assert!(!sink.offer(ev()));
321        assert_eq!(rx.try_recv().unwrap(), ev());
322        drop(rx);
323        assert!(!sink.offer(ev()));
324        assert!(!DiscardSink.offer(ev()));
325        let shared: Arc<dyn TelemetrySink> = Arc::new(DiscardSink);
326        assert!(!shared.offer(ev()));
327    }
