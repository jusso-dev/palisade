1//! The fake agent: `palisade-agent --simulate` (#29).
2//!
3//! It replays a [`Scenario`] on the mock platform from
4//! `palisade_core::test_util`, on any OS, and emits canonical events
5//! (`palisade_proto::events::Event`): `agent.started`, one `exec.decision`
6//! per step, then `agent.stopped`. The events are built only through the
7//! platform traits, generically, so the same code builds them from the real
8//! platform once the agent has a process tracker (#59).
9//!
10//! The binary writes the events to standard output as JSON lines, in their
11//! canonical form. Delivering them to a server needs the agent endpoints and
12//! enrolment, which do not exist yet (#62, #223, #126); until then, the
13//! output is what a server or console developer feeds to their code. See
14//! `docs/dev/unix.md`.
15//!
16//! The decisions are scripted by the scenario, not evaluated: policy
17//! evaluation in the agent arrives with the policy cache (#61).
18

75/// Builds events from a [`Platform`], numbering and chaining them.
76///
77/// The chain value here is a placeholder: the SHA-256 of the previous
78/// event's canonical JSON. The keyed chain belongs to #220 and #202.
79struct Producer {
80    identity: Identity,
81    seq: u64,
82    prev: Option<events::Sha256Hex>,
83    ids: IdSource,
84}

86/// Deterministic UUIDv7 values: the timestamp and a counter, so a scenario
87/// always produces the same event IDs.
88struct IdSource(u64);
89
90impl IdSource {
91    fn next(&mut self, at: UtcTime) -> Result<Uuid7, SimulateError> {
92        self.0 = self.0.wrapping_add(1);
93        let millis = u64::try_from(at.0.div_euclid(1000)).unwrap_or(0);
94        let mut bytes = [0u8; 16];
95        if let (Some(dst), Some(src)) = (bytes.get_mut(..6), millis.to_be_bytes().get(2..)) {
96            dst.copy_from_slice(src);
97        }
98        if let Some(dst) = bytes.get_mut(8..) {
99            dst.copy_from_slice(&self.0.to_be_bytes());
100        }
101        bytes[6] = 0x70;
102        bytes[8] = 0x80 | (bytes[8] & 0x3f);
103        Uuid7::from_bytes(bytes).ok_or_else(|| SimulateError("cannot build an event ID".into()))
104    }
105}

119impl Producer {
120    fn event(
121        &mut self,
122        clock: &dyn Clock,
123        event_type: EventType,
124        action: &str,
125        outcome: Outcome,
126        description: &str,
127    ) -> Result<Event, SimulateError> {
128        let now = clock.now_utc();
129        let event = Event {
130            schema_version: CURRENT,
131            event_id: self.ids.next(now)?,
132            event_type,
133            category: event_type.category(),
134            action: action.parse::<Token>().map_err(|e| invalid(action, e))?,
135            outcome,
136            severity: event_type.default_severity(),
137            mode: Some(self.identity.mode),
138            description: events::Description::new(description)
139                .map_err(|e| invalid("description", e))?,
140            ts: timestamp(now)?,
141            ts_agent_monotonic: Some(clock.monotonic().0),
142            clock_synced: true,
143            clock_skew_ms: None,
144            tenant_id: self.identity.tenant,
145            host: self.identity.host.clone(),
146            user: None,
147            policy: None,
148            seq: self.seq,
149            prev_hash: self.prev,
150            labels: events::Labels::default(),
151            source: None,
152            process: None,
153            file: None,
154            script: None,
155            native: None,
156            agent: None,
157            network: None,
158            device: None,
159            registry: None,
160            elevation: None,
161            approval: None,
162            admin: None,
163            identity: None,
164            audit_policy: None,
165            restriction: None,
166            probe: None,
167            fim: None,
168        };
169        Ok(event)
170    }
171
172    /// Validates, numbers and chains `event`, then emits it.
173    fn emit(&mut self, event: Event, sink: &mut dyn EventSink<Event>) -> Result<(), SimulateError> {
174        if let Err(errors) = event.validate() {
175            return Err(SimulateError(format!(
176                "built an invalid event: {}",
177                errors
178                    .iter()
179                    .map(|e| format!("{e:?}"))
180                    .collect::<Vec<_>>()
181                    .join(", ")
182            )));
183        }
184        let json = event
185            .to_canonical_json()
186            .map_err(|e| invalid("serialise", e))?;
187        self.prev = Some(events::Sha256Hex::from_bytes(sha256(json.as_bytes())));
188        self.seq = self.seq.saturating_add(1);
189        sink.emit(event)?;
190        Ok(())
191    }
192}

194fn sha256(data: &[u8]) -> [u8; 32] {
195    use sha2::Digest as _;
196    sha2::Sha256::digest(data).into()
197}

199/// The SHA-256 and size of the file at `path`, read through the
200/// [`FileSystem`] trait.
201fn hash_file<F: FileSystem + ?Sized>(fs: &F, path: &str) -> Result<([u8; 32], u64), SimulateError> {
202    use sha2::Digest as _;
203    let mut file = fs.open(path)?;
204    let mut hasher = sha2::Sha256::new();
205    let mut buf = [0u8; 8192];
206    let mut offset = 0u64;
207    loop {
208        let n = file.read_at(offset, &mut buf)?;
209        if n == 0 {
210            break;
211        }
212        hasher.update(buf.get(..n).unwrap_or_default());
213        offset = offset.saturating_add(u64::try_from(n).unwrap_or(u64::MAX));
214    }
215    Ok((hasher.finalize().into(), offset))
216}

218/// The host facts, read through the [`Environment`] trait.
219fn host<E: Environment + ?Sized>(env: &E, host_id: HostId) -> Result<Host, SimulateError> {
220    let hostname = env.hostname()?;
221    let domain = env.domain()?;
222    let version: OsVersion = env.os_version()?;
223    let arch = match env.architecture() {
224        Architecture::X64 => "x86_64",
225        Architecture::Arm64 => "aarch64",
226        Architecture::X86 => "x86",
227        Architecture::Other => "other",
228    };
229    let fqdn = domain.as_ref().map(|d| format!("{hostname}.{d}"));
230    Ok(Host {
231        id: host_id,
232        name: Some(name(&hostname)?),
233        fqdn: fqdn.as_deref().map(name).transpose()?,
234        domain: domain.as_deref().map(name).transpose()?,
235        os: Some(name("Windows")?),
236        os_build: Some(name(&format!("{}.{}", version.build, version.revision))?),
237        arch: Some(name(arch)?),
238        agent_version: Some(name(env!("CARGO_PKG_VERSION"))?),
239        ips: events::BoundedVec::default(),
240        groups: events::BoundedVec::default(),
241    })
242}

244/// The user field set, from a token and the account name.
245fn user(token: &TokenInfo, account: Option<&AccountName>) -> Result<User, SimulateError> {
246    Ok(User {
247        sid: Some(
248            token
249                .user_sid
250                .parse()
251                .map_err(|e| invalid(&token.user_sid, e))?,
252        ),
253        name: account.map(|a| name(&a.name)).transpose()?,
254        domain: account
255            .and_then(|a| a.domain.as_deref())
256            .map(name)
257            .transpose()?,
258        upn: None,
259        session_id: Some(token.session_id),
260        logon_id: None,
261        logon_type: Some(match token.session_kind {
262            SessionKind::RemoteDesktop => events::LogonType::RemoteInteractive,
263            SessionKind::Services => events::LogonType::Service,
264            SessionKind::Console | SessionKind::Unknown => events::LogonType::Interactive,
265        }),
266        is_admin: Some(
267            token
268                .groups
269                .iter()
270                .any(|g| g.sid.eq_ignore_ascii_case(ADMINISTRATORS)),
271        ),
272        elevated: Some(token.elevated),
273    })
274}

301fn signers<S: SignatureVerifier + ?Sized>(
302    sigs: &S,
303    path: &str,
304) -> Result<events::BoundedVec<Signer, { events::MAX_SIGNERS }>, SimulateError> {
305    let mut out = Vec::new();
306    for sig in sigs.embedded(path)?.into_iter().take(events::MAX_SIGNERS) {
307        let leaf = sig.chain.first();
308        out.push(Signer {
309            publisher: leaf.map(|c| name(&c.subject)).transpose()?,
310            issuer: leaf.map(|c| name(&c.issuer)).transpose()?,
311            chain_valid: Some(sig.status == VerificationError::Ok),
312            verification_error: u8::try_from(sig.status.code()).ok(),
313            ..Signer::default()
314        });
315    }
316    events::BoundedVec::new(out).map_err(|e| invalid("signers", e))
317}

319/// The process field set for `record`, with its parent and grandparent
320/// from the PID-reuse-safe ancestry.
321fn process<P: Platform>(platform: &P, record: &ProcessRecord) -> Result<Process, SimulateError> {
322    let chain = ancestry(platform.processes(), record.key)?;
323    let token = platform.identity().token(record.key)?;
324    let parent = chain
325        .first()
326        .map(|p| -> Result<Box<ParentProcess>, SimulateError> {
327            Ok(Box::new(ParentProcess {
328                pid: Some(p.key.pid),
329                process_key: Some(process_key(p.key)?),
330                image_path: Some(long(&p.image_path)?),
331                ..ParentProcess::default()
332            }))
333        })
334        .transpose()?;
335    let grandparent = chain
336        .get(1)
337        .map(|g| -> Result<ProcessSummary, SimulateError> {
338            Ok(ProcessSummary {
339                pid: Some(g.key.pid),
340                process_key: Some(process_key(g.key)?),
341                image_path: Some(long(&g.image_path)?),
342            })
343        })
344        .transpose()?;
345    Ok(Process {
346        pid: Some(record.key.pid),
347        process_key: Some(process_key(record.key)?),
348        image_path: Some(long(&record.image_path)?),
349        image_path_normalised: Some(long(
350            palisade_core::path::normalize(&record.image_path).as_str(),
351        )?),
352        cmdline: record.command_line.as_deref().map(long).transpose()?,
353        signers: signers(platform.signatures(), &record.image_path)?,
354        integrity_level: Some(integrity(token.integrity)),
355        token_elevation_type: Some(elevation(token.elevation)),
356        owner_sid: Some(
357            token
358                .user_sid
359                .parse()
360                .map_err(|e| invalid(&token.user_sid, e))?,
361        ),
362        parent,
363        grandparent,
364        ..Process::default()
365    })
366}

368/// The `exec.decision` event for the process `key`, read entirely through
369/// the platform traits.
370fn exec_decision<P: Platform>(
371    producer: &mut Producer,
372    platform: &P,
373    key: ProcessKey,
374    outcome: Outcome,
375) -> Result<Event, SimulateError> {
376    let record = platform.processes().get(key)?;
377    let token = platform.identity().token(key)?;
378    let account = platform.identity().account_name(&token.user_sid)?;
379    let verb = match outcome {
380        Outcome::Blocked => "blocked",
381        Outcome::Audited => "audited (would have been blocked)",
382        _ => "allowed",
383    };
384    let mut event = producer.event(
385        platform.clock(),
386        EventType::ExecDecision,
387        "execute",
388        outcome,
389        &format!("Simulated: {} was {verb}.", record.image_path),
390    )?;
391    if outcome != Outcome::Allowed {
392        event.severity =
393            Severity::new(event.severity.get().saturating_add(2)).unwrap_or(Severity::MAX);
394    }
395    event.user = Some(user(&token, account.as_ref())?);
396    let process = process(platform, &record)?;
397    let (digest, size) = hash_file(platform.file_system(), &record.image_path)?;
398    let file = events::File {
399        path: process.image_path.clone(),
400        path_normalised: process.image_path_normalised.clone(),
401        object_type: Some(events::ObjectType::Executable),
402        size: Some(size),
403        hashes: events::Hashes {
404            sha256: Some(events::Sha256Hex::from_bytes(digest)),
405            ..events::Hashes::default()
406        },
407        signers: process.signers.clone(),
408        ..events::File::default()
409    };
410    event.process = Some(process);
411    event.file = Some(file);
412    Ok(event)
413}

421/// `BUILTIN\\Administrators`.
422const ADMINISTRATORS: &str = "S-1-5-32-544";
423
424/// The token of a process run by `sid`: elevated users hold the
425/// Administrators group.
426fn token_for(scenario: &Scenario, sid: &str) -> Result<TokenInfo, SimulateError> {
427    let spec = scenario
428        .users
429        .iter()
430        .find(|u| u.sid.eq_ignore_ascii_case(sid))
431        .ok_or_else(|| SimulateError(format!("user {sid} is not declared")))?;
432    let session_kind = match spec.session.as_str() {
433        "remote_desktop" => SessionKind::RemoteDesktop,
434        "services" => SessionKind::Services,
435        _ => SessionKind::Console,
436    };
437    let system = session_kind == SessionKind::Services;
438    Ok(TokenInfo {
439        user_sid: spec.sid.clone(),
440        groups: if spec.elevated {
441            vec![GroupMembership {
442                sid: ADMINISTRATORS.to_owned(),
443                deny_only: false,
444            }]
445        } else {
446            Vec::new()
447        },
448        session_id: if system { 0 } else { 1 },
449        session_kind,
450        integrity: if system {
451            IntegrityLevel::System
452        } else if spec.elevated {
453            IntegrityLevel::High
454        } else {
455            IntegrityLevel::Medium
456        },
457        elevation: if spec.elevated {
458            Elevation::Full
459        } else {
460            Elevation::Default
461        },
462        elevated: spec.elevated,
463    })
464}

466/// Builds the mock platform for `scenario`: the device, the users and the
467/// processes already running.
468fn setup(scenario: &Scenario) -> Result<(MockPlatform, Identity), SimulateError> {
469    let start = UtcTime(scenario.start.unix_micros());
470    let mut platform = MockPlatform {
471        clock: MockClock::new(start, MonotonicTime(0)),
472        ..MockPlatform::default()
473    };
474    let d = &scenario.device;
475    platform.environment.hostname.clone_from(&d.hostname);
476    platform.environment.domain.clone_from(&d.domain);
477    platform.environment.os_version = parse_version(&d.os_version)?;
478    platform.environment.architecture = match d.architecture.as_str() {
479        "x64" => Architecture::X64,
480        "arm64" => Architecture::Arm64,
481        "x86" => Architecture::X86,
482        other => return Err(SimulateError(format!("unknown architecture `{other}`"))),
483    };
484    for u in &scenario.users {
485        platform.identity.set_account_name(
486            &u.sid,
487            AccountName {
488                domain: u.domain.clone(),
489                name: u.name.clone(),
490            },
491        );
492    }
493    // Running processes were created before the agent started, in order.
494    let count = i64::try_from(scenario.running.len()).unwrap_or(i64::MAX);
495    for (i, p) in scenario.running.iter().enumerate() {
496        let i = i64::try_from(i).unwrap_or(i64::MAX);
497        let key = ProcessKey {
498            pid: p.pid,
499            created: start.saturating_add_micros((i - count).saturating_mul(1_000_000)),
500        };
501        add_process(&mut platform, key, p.parent, &p.image, None);
502        platform
503            .identity
504            .set_token(key, token_for(scenario, &p.user)?);
505    }
506    let mode = match scenario.mode.as_str() {
507        "monitor" => EventMode::Monitor,
508        "learning" => EventMode::Learning,
509        "enforce" => EventMode::Enforce,
510        _ => EventMode::Audit,
511    };
512    let tenant = TenantId(d.tenant_id.parse().map_err(|e| invalid("tenant_id", e))?);
513    let host_id = HostId(d.host_id.parse().map_err(|e| invalid("host_id", e))?);
514    let host = host(&platform.environment, host_id)?;
515    Ok((platform, Identity { tenant, host, mode }))
516}

518fn parse_version(text: &str) -> Result<OsVersion, SimulateError> {
519    let parts: Vec<u32> = text
520        .split('.')
521        .map(str::parse)
522        .collect::<Result<_, _>>()
523        .map_err(|e| invalid("os_version", e))?;
524    match parts.as_slice() {
525        [major, minor, build, revision] => Ok(OsVersion {
526            major: *major,
527            minor: *minor,
528            build: *build,
529            revision: *revision,
530        }),
531        _ => Err(SimulateError(format!(
532            "os_version `{text}` is not major.minor.build.revision"
533        ))),
534    }
535}

537fn add_process(
538    platform: &mut MockPlatform,
539    key: ProcessKey,
540    parent: Option<u32>,
541    image: &str,
542    command_line: Option<&str>,
543) {
544    platform.processes.start(ProcessRecord {
545        key,
546        parent_pid: parent,
547        image_path: image.to_owned(),
548        command_line: command_line.map(str::to_owned),

550    platform

555/// Replays `scenario`, emitting every event into `sink`. Returns the number
556/// of events emitted.
557///
558/// # Errors
559///
560/// A scenario value the event schema rejects (an invalid SID or ID, an
561/// over-long name), or a sink that refuses an event.
562pub fn run(scenario: &Scenario, sink: &mut dyn EventSink<Event>) -> Result<u64, SimulateError> {
563    let (mut platform, identity) = setup(scenario)?;
564    let mut producer = Producer {
565        identity,
566        seq: 0,
567        prev: None,
568        ids: IdSource(0),
569    };
570    let started = producer.event(
571        platform.clock(),
572        EventType::AgentStarted,
573        "start",
574        Outcome::Info,
575        "Simulated agent started.",
576    )?;
577    producer.emit(started, sink)?;
578    for step in &scenario.steps {
579        // At least one microsecond passes, so no two processes share a
580        // creation time, and a reused process ID always gets a new key.
581        platform
582            .clock
583            .advance(step.after_ms.saturating_mul(1000).max(1));
584        for pid in &step.exited {
585            platform.processes.exit(*pid);
586        }
587        let key = ProcessKey {
588            pid: step.pid,
589            created: platform.clock.now_utc(),
590        };
591        add_process(
592            &mut platform,
593            key,
594            step.parent,
595            &step.image,
596            step.command_line.as_deref(),
597        );
598        platform
599            .identity
600            .set_token(key, token_for(scenario, &step.user)?);
601        // Each step states the image's signature, so a later unsigned step
602        // with the same image path does not inherit an earlier signature.
603        platform.signatures.set_embedded(&step.image, Vec::new());
604        if let Some(publisher) = &step.signer {
605            platform.signatures.set_embedded(
606                &step.image,
607                vec![SignatureResult {
608                    source: SignatureSource::Embedded,
609                    status: VerificationError::Ok,
610                    chain: vec![CertificateSummary {
611                        subject: publisher.clone(),
612                        issuer: format!("{publisher} Code Signing CA"),
613                        sha256: sha256(publisher.as_bytes()),
614                    }],
615                    signing_time: None,
616                }],
617            );
618        }
619        let outcome = scenario::parse_outcome(&step.outcome)
620            .ok_or_else(|| SimulateError(format!("unknown outcome `{}`", step.outcome)))?;
621        let event = exec_decision(&mut producer, &platform, key, outcome)?;
622        producer.emit(event, sink)?;
623    }
624    platform.clock.advance(1000);
625    let stopped = producer.event(
626        platform.clock(),
627        EventType::AgentStopped,
628        "stop",
629        Outcome::Info,
630        "Simulated agent stopped.",
631    )?;
632    producer.emit(stopped, sink)?;
633    Ok(producer.seq)
634}

636/// Writes each event as one line of canonical JSON.
637pub struct JsonLines<W: Write>(pub W);
638
639impl<W: Write> EventSink<Event> for JsonLines<W> {
640    fn emit(&mut self, event: Event) -> palisade_core::platform::Result<()> {
641        let json = event.to_canonical_json().map_err(|e| {
