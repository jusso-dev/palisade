1//! The running server: listener supervision, connections and shutdown.
2//!
3//! Each configured listener has a supervisor task. The supervisor resolves
4//! the listener's bind target to socket addresses, binds each one it does
5//! not have yet, and runs an accept loop per bound socket. It repeats this
6//! on a timer, so:
7//!
8//! - an address that cannot be bound yet (an overlay interface that is not
9//!   up, an address not assigned yet, a port still in use) is retried with
10//!   exponential backoff, and health shows `listener_unbound:<name>` until
11//!   it succeeds;
12//! - a named interface that gains or loses addresses while the server runs
13//!   is followed: new addresses are bound, vanished ones stop accepting;
14//! - two listeners whose targets resolve to the same address (an interface
15//!   and a literal) are rejected before start by [`prepare`].
16//!
17//! Connections are TLS only. ALPN offers `h2` and `http/1.1`; a connection
18//! that negotiated `h2` is served as HTTP/2 only, and every other connection
19//! (including one with no ALPN) as HTTP/1.1 only. A client must send its
20//! first complete request within the request-head timeout (30 s) of the
21//! handshake, or the connection is closed, so an idle or stalled client
22//! cannot hold a connection slot. A listener serves at most
23//! `max_connections` connections at a time; further clients wait in the
24//! TCP backlog.
25//!
26//! Shutdown has two phases. **Draining**: listeners stop accepting,
27//! readiness fails, and every open connection is asked to finish its
28//! in-flight requests and close (HTTP/1.1 closes after the current
29//! response, HTTP/2 sends GOAWAY). If connections remain when
30//! `shutdown.grace_secs` runs out, **force**: they are dropped.
31

77pub struct Tuning {
78    /// First retry delay for an unbound listener.
79    pub retry_min: Duration,

119/// Runs the pre-start checks. `host` is the current interface list, used
120/// for the placement warnings. Returns every error found.
121pub fn prepare(config: &Config, host: &[HostAddr]) -> Result<Prepared, Vec<String>> {
122    let identity = Identity::load(&config.tls).map_err(|e| vec![e])?;
123    let mut errors = Vec::new();
124    let mut tls_configs = BTreeMap::new();
125    for l in &config.listeners {
126        match tls::server_config(&config.tls, &identity, l.client_ca_file.as_deref(), &ALPN) {
127            Ok(c) => {
128                tls_configs.insert(l.kind, c);
129            }
130            Err(e) => errors.push(format!("listen.{}: {e}", l.kind)),
131        }
132        if let Some(url) = &l.public_url
133            && let Err(e) = tls::leaf_covers(&identity, &url.host)
134        {
135            errors.push(format!(
136                    "listen.{}.public_url: host of {} is not in the subject alternative names of {} ({e})",
137                    l.kind,
138                    url.url,
139                    config.tls.cert_file.display()
140                ));
141        }
142    }
143    errors.extend(resolved_overlaps(config, host));
144    if !errors.is_empty() {
145        return Err(errors);
146    }
147    Ok(Prepared {
148        tls: tls_configs,
149        warnings: placement_warnings(config, host),
150    })
151}

153/// Listeners whose bind targets resolve to the same socket address on this
154/// host, for example `bind = "lo"` and `bind = "127.0.0.1"` on one port. The
155/// static check in the configuration catches literal clashes; this one needs
156/// the interface list.
157fn resolved_overlaps(config: &Config, host: &[HostAddr]) -> Vec<String> {
158    let mut seen: BTreeMap<SocketAddr, ListenerKind> = BTreeMap::new();
159    let mut errors = Vec::new();
160    for l in &config.listeners {
161        for addr in net::resolve(&l.bind, l.port, host) {
162            match seen.get(&addr) {
163                Some(other) if *other != l.kind => errors.push(format!(
164                    "listen.{other} and listen.{} both resolve to {addr}; give each listener its own address or port",
165                    l.kind
166                )),
167                _ => {
168                    seen.insert(addr, l.kind);
169                }
170            }
171        }
172    }
173    errors
174}

176/// Placement warnings for `deployment.profile = "overlay"`: a listener that
177/// binds every interface, binds a public address, or (for metrics) is not
178/// on loopback.
179pub fn placement_warnings(config: &Config, host: &[HostAddr]) -> Vec<String> {
180    if config.profile != Profile::Overlay {
181        return Vec::new();
182    }
183    let mut out = Vec::new();
184    for l in &config.listeners {
185        let name = l.kind;
186        let addrs = net::resolve(&l.bind, l.port, host);
187        if let BindTarget::Ip(ip) = l.bind
188            && ip.is_unspecified()
189        {
190            out.push(format!(
191                    "listen.{name}: binds every interface ({ip}) under deployment.profile = \"overlay\"; bind the overlay address or interface instead"
192                ));
193        }
194        if let BindTarget::Interface(iface) = &l.bind
195            && addrs.is_empty()
196        {
197            out.push(format!(
198                    "listen.{name}: interface {iface} has no address now; the server will keep retrying until it appears"
199                ));
200        }
201        let public = public_addrs(&addrs, host);
202        if !public.is_empty() {
203            out.push(format!(
204                "listen.{name}: reachable on public address(es) {} under deployment.profile = \"overlay\"",
205                join_ips(&public)
206            ));
207        }
208        if name == ListenerKind::Metrics && addrs.iter().any(|a| !a.ip().is_loopback()) {
209            out.push(
210                "listen.metrics: not bound to loopback under deployment.profile = \"overlay\"; bind it to 127.0.0.1 or ::1".to_owned(),
211            );
212        }
213    }
214    out
215}

217fn public_addrs(bound: &BTreeSet<SocketAddr>, host: &[HostAddr]) -> Vec<IpAddr> {
218    let set: BTreeSet<IpAddr> = bound
219        .iter()
220        .flat_map(|a| net::reachable_on(a, host))
221        .filter(|ip| net::is_public(*ip))
222        .collect();
223    set.into_iter().collect()
224}
225
226fn join_ips(ips: &[IpAddr]) -> String {
227    ips.iter()
228        .map(ToString::to_string)
229        .collect::<Vec<_>>()
230        .join(", ")
231}

299    /// Starts every listener. Must be called inside a Tokio runtime.
300    /// Listeners that cannot bind yet keep retrying in the background.
301    pub fn start(mut self) -> RunningServer {
302        let health = Arc::new(Health::new(self.config.listeners.iter().map(|l| l.kind)));
303        if self.database {
304            health.set_database(DatabaseHealth::Down);
305        }
306        let (phase_tx, phase_rx) = watch::channel(Phase::Running);
307        let (drain_tx, drain_rx) = mpsc::channel::<()>(1);
308        let mut supervisors = JoinSet::new();
309        for settings in self.config.listeners.clone() {
310            let Some(tls) = self.prepared.tls.get(&settings.kind).cloned() else {
311                // `prepare` builds one configuration per listener.
312                continue;
313            };
314            let state = http::AppState {
315                health: health.clone(),
316                detailed_health: settings.kind == ListenerKind::Metrics,
317            };
318            let mut router = http::routes(state);
319            if let Some(extra) = self.routes.remove(&settings.kind) {
320                router = router.merge(extra);
321            }
322            let router = http::apply_stack(router, &settings);
323            let (h1, h2) = http_builders(self.tuning.request_head_timeout);
324            let ctx = Arc::new(ListenerCtx {
325                conn_limit: Arc::new(Semaphore::new(settings.max_connections)),
326                settings,
327                acceptor: TlsAcceptor::from(tls),
328                router,
329                h1,
330                h2,
331                request_head_timeout: self.tuning.request_head_timeout,
332            });
333            supervisors.spawn(supervise(Supervisor {
334                ctx,
335                profile: self.config.profile,
336                health: health.clone(),
337                addrs: self.addrs.clone(),
338                tuning: self.tuning,
339                phase: phase_rx.clone(),
340                drain: drain_tx.clone(),
341            }));
342        }
343        RunningServer {
344            health,
345            phase: phase_tx,
346            supervisors,
347            drain_tx,
348            drain_rx,
349            grace: self.config.shutdown_grace,
350        }
351    }
352}

354/// The HTTP/1.1-only and HTTP/2-only connection builders.
355fn http_builders(
356    head_timeout: Duration,
357) -> (auto::Builder<TokioExecutor>, auto::Builder<TokioExecutor>) {
358    let mut h1 = auto::Builder::new(TokioExecutor::new()).http1_only();
359    h1.http1()
360        .timer(TokioTimer::new())
361        .header_read_timeout(head_timeout)
362        .max_buf_size(H1_MAX_BUF);
363    let mut h2 = auto::Builder::new(TokioExecutor::new()).http2_only();
364    h2.http2()
365        .timer(TokioTimer::new())
366        .max_concurrent_streams(H2_MAX_STREAMS)
367        .max_header_list_size(H2_MAX_HEADER_LIST)
368        .keep_alive_interval(H2_KEEPALIVE_INTERVAL)
369        .keep_alive_timeout(H2_KEEPALIVE_TIMEOUT);
370    (h1, h2)
371}

373/// How shutdown ended.
374#[derive(Debug, Clone, Copy, PartialEq, Eq)]
375pub enum ShutdownOutcome {
376    /// Every connection closed within the grace period.
377    Drained,
378    /// The grace period ran out and the remaining connections were dropped.
379    Forced,
380}
381
382/// A started server.
383pub struct RunningServer {
384    health: Arc<Health>,
385    phase: watch::Sender<Phase>,
386    supervisors: JoinSet<()>,
387    drain_tx: mpsc::Sender<()>,
388    drain_rx: mpsc::Receiver<()>,
389    grace: Duration,
390}
391
392impl RunningServer {
393    /// The shared health state.
394    pub fn health(&self) -> &Arc<Health> {
395        &self.health
396    }
397
398    /// Shuts down: stop accepting, drain within the grace period, then
399    /// drop what is left.
400    pub async fn shutdown(self) -> ShutdownOutcome {
401        let Self {
402            health,
403            phase,
404            mut supervisors,
405            drain_tx,
406            mut drain_rx,
407            grace,
408        } = self;
409        tracing::info!(
410            grace_secs = grace.as_secs(),
411            "shutdown started; draining connections"
412        );
413        health.set_shutting_down();
414        phase.send_replace(Phase::Draining);
415        while supervisors.join_next().await.is_some() {}
416        drop(drain_tx);
417        // `recv` returns `None` once every connection has dropped its
418        // sender, that is, once every connection is closed.
419        if tokio::time::timeout(grace, drain_rx.recv()).await.is_ok() {
420            tracing::info!("all connections closed");
421            return ShutdownOutcome::Drained;
422        }
423        tracing::warn!(
424            grace_secs = grace.as_secs(),
425            "grace period ended with connections still open; dropping them"
426        );
427        phase.send_replace(Phase::Force);
428        let _ = tokio::time::timeout(FORCE_SETTLE, drain_rx.recv()).await;
429        ShutdownOutcome::Forced
430    }
431}

453async fn wait_not_running(phase: &mut watch::Receiver<Phase>) {
454    // An error means the sender is gone, which also means stop.
455    let _ = phase.wait_for(|p| *p != Phase::Running).await;
456}

458async fn supervise(mut sv: Supervisor) {
459    let kind = sv.ctx.settings.kind;
460    let bind = sv.ctx.settings.bind.clone();
461    let port = sv.ctx.settings.port;
462    // The interface list is only needed to resolve an interface name, or to
463    // see what a wildcard is reachable on under the overlay profile.
464    let needs_host = match &bind {
465        BindTarget::Interface(_) => true,
466        BindTarget::Ip(ip) => sv.profile == Profile::Overlay && ip.is_unspecified(),
467    };
468    let mut loops: BTreeMap<SocketAddr, AbortHandle> = BTreeMap::new();
469    let mut tasks = JoinSet::new();
470    let mut last_failures: Vec<String> = Vec::new();
471    let mut last_exposed: Vec<IpAddr> = Vec::new();
472    let mut delay = sv.tuning.retry_min;
473
474    loop {
475        // Accept loops that ended on their own (they only do on shutdown)
476        // are reaped, so their address is bound again if still wanted.
477        while let Some(done) = tasks.try_join_next() {
478            let _ = done;
479        }
480        loops.retain(|_, h| !h.is_finished());
481
482        let host = if needs_host {
483            let source = sv.addrs.clone();
484            match tokio::task::spawn_blocking(move || source()).await {
485                Ok(Ok(h)) => h,
486                Ok(Err(e)) => {
487                    tracing::warn!(listener = %kind, error = %e, "cannot list network interfaces");
488                    Vec::new()
489                }
490                Err(_) => Vec::new(),
491            }
492        } else {
493            Vec::new()
494        };
495        let wanted = net::resolve(&bind, port, &host);
496
497        loops.retain(|addr, h| {
498            let keep = wanted.contains(addr);
499            if !keep {
500                tracing::info!(listener = %kind, address = %addr, "address no longer present; stopped accepting on it");
501                h.abort();
502            }
503            keep
504        });
505
506        let mut failures = Vec::new();
507        for addr in &wanted {
508            if loops.contains_key(addr) {
509                continue;
510            }
511            match bind_socket(*addr) {
512                Ok(listener) => {
513                    tracing::info!(listener = %kind, address = %addr, "listening");
514                    let h = tasks.spawn(accept_loop(
515                        listener,
516                        sv.ctx.clone(),
517                        sv.phase.clone(),
518                        sv.drain.clone(),
519                    ));
520                    loops.insert(*addr, h);
521                }
522                Err(e) => failures.push(format!("{addr}: {e}")),
523            }
524        }
525        if wanted.is_empty() {
526            failures.push(format!("{bind} has no usable address"));
527        }
528        if failures != last_failures {
529            if !failures.is_empty() {
530                tracing::warn!(
531                    listener = %kind,
532                    problems = failures.join("; "),
533                    "listener not fully bound; retrying with backoff"
534                );
535            } else if !last_failures.is_empty() {
536                tracing::info!(listener = %kind, "listener fully bound");
537            }
538            last_failures.clone_from(&failures);
539        }
540
541        let bound: BTreeSet<SocketAddr> = loops.keys().copied().collect();
542        let exposed = if sv.profile == Profile::Overlay {
543            public_addrs(&bound, &host)
544        } else {
545            Vec::new()
546        };
547        if exposed != last_exposed {
548            if !exposed.is_empty() {
549                tracing::warn!(
550                    listener = %kind,
551                    addresses = join_ips(&exposed),
552                    "listener is reachable on a public address under deployment.profile = \"overlay\""
553                );
554            }
555            last_exposed.clone_from(&exposed);
556        }
557        sv.health.set_listener(
558            kind,
559            ListenerHealth {
560                bound,
561                unbound: !failures.is_empty(),
562                public_exposure: !exposed.is_empty(),
563            },
564        );
565
566        let wait = if failures.is_empty() {
567            delay = sv.tuning.retry_min;
568            sv.tuning.poll
569        } else {
570            let d = delay;
571            delay = delay.saturating_mul(2).min(sv.tuning.retry_max);
572            d
573        };
574        tokio::select! {
575            () = tokio::time::sleep(wait) => {}
576            () = wait_not_running(&mut sv.phase) => break,
577        }
578    }
579
580    // Accept loops watch the phase too; wait for them to release their
581    // sockets before reporting the listener stopped.
582    while tasks.join_next().await.is_some() {}
583}

585/// Binds a listening socket. IPv6 sockets are IPv6-only, so `::` never
586/// silently covers IPv4 too. On Windows, `SO_REUSEADDR` is left off because
587/// there it lets another process take over the port. Setting
588/// `SO_EXCLUSIVEADDRUSE`, which also stops another process from binding a
589/// more specific address on the same port, is part of the server hardening
590/// in #229.
591fn bind_socket(addr: SocketAddr) -> std::io::Result<TcpListener> {
592    use socket2::{Domain, Protocol, Socket, Type};
593    let socket = Socket::new(Domain::for_address(addr), Type::STREAM, Some(Protocol::TCP))?;
594    if addr.is_ipv6() {
595        socket.set_only_v6(true)?;
596    }
597    #[cfg(not(windows))]
598    socket.set_reuse_address(true)?;
599    socket.set_nonblocking(true)?;
600    socket.bind(&addr.into())?;
601    socket.listen(BACKLOG)?;
602    TcpListener::from_std(socket.into())
603}

605/// Accepts connections on one socket. A connection is accepted first and a
606/// connection slot acquired second, so an idle socket never holds a slot.
607/// While no slot is free the loop holds at most that one accepted
608/// connection, untouched, and stops accepting; other clients wait in the
609/// TCP backlog.
610async fn accept_loop(
611    listener: TcpListener,
612    ctx: Arc<ListenerCtx>,
613    mut phase: watch::Receiver<Phase>,
614    drain: mpsc::Sender<()>,
615) {
616    let kind = ctx.settings.kind;
617    loop {
618        let (stream, peer) = tokio::select! {
619            r = listener.accept() => match r {
620                Ok(v) => v,
621                Err(e) => {
622                    // Usually a client that went away before it was
623                    // accepted, or a descriptor limit; pause briefly so a
624                    // persistent error cannot spin.
625                    tracing::debug!(listener = %kind, error = %e, "accept failed");
626                    tokio::time::sleep(Duration::from_millis(50)).await;
627                    continue;
628                }
629            },
630            () = wait_not_running(&mut phase) => return,
631        };
632        let permit = tokio::select! {
633            p = ctx.conn_limit.clone().acquire_owned() => match p {
634                Ok(p) => p,
635                Err(_) => return,
636            },
637            () = wait_not_running(&mut phase) => return,
638        };
639        let _ = stream.set_nodelay(true);
640        tokio::spawn(serve_connection(
641            stream,
642            peer,
643            ctx.clone(),
644            phase.clone(),
645            permit,
646            drain.clone(),
647        ));
648    }
649}

655    mut phase: watch::Receiver<Phase>,
656    _permit: OwnedSemaphorePermit,
657    _drain: mpsc::Sender<()>,
658) {
659    let kind = ctx.settings.kind;
660    let tls = tokio::select! {
661        r = tokio::time::timeout(HANDSHAKE_TIMEOUT, ctx.acceptor.accept(stream)) => match r {
662            Ok(Ok(s)) => s,
663            Ok(Err(e)) => {
664                tracing::debug!(listener = %kind, peer = %peer, error = %e, "TLS handshake failed");
665                return;
666            }
667            Err(_) => {
668                tracing::debug!(listener = %kind, peer = %peer, "TLS handshake timed out");
669                return;
670            }
671        },
672        () = wait_not_running(&mut phase) => return,
673    };
674    // The protocol follows ALPN (RFC 7301): `h2` means HTTP/2, anything
675    // else HTTP/1.1. The client cannot switch by sending another preface.
676    let builder = if tls.get_ref().1.alpn_protocol() == Some(b"h2") {
677        &ctx.h2
678    } else {
679        &ctx.h1
680    };
681    // Set when the first request reaches the router. Until then the
682    // connection is on a deadline, which also covers the HTTP/2 preface and
683    // settings exchange that hyper does not time.
684    let started = Arc::new(AtomicBool::new(false));
685    let seen = started.clone();
686    let service = ctx.router.clone().map_request(move |mut req: Request<_>| {
687        seen.store(true, Ordering::Relaxed);
688        req.extensions_mut().insert(Peer(peer));
689        req
690    });
691    let conn = builder.serve_connection(TokioIo::new(tls), TowerToHyperService::new(service));
692    tokio::pin!(conn);
693    let first_request = tokio::time::sleep(ctx.request_head_timeout);
694    tokio::pin!(first_request);
695    let mut draining = false;
696    match *phase.borrow_and_update() {
697        Phase::Running => {}
698        Phase::Draining => {
699            draining = true;
700            conn.as_mut().graceful_shutdown();
701        }
702        Phase::Force => return,
703    }
704    loop {
705        tokio::select! {
706            r = conn.as_mut() => {
707                if let Err(e) = r {
708                    tracing::debug!(listener = %kind, peer = %peer, error = %e, "connection ended with an error");
709                }
710                break;
711            }
712            () = &mut first_request, if !started.load(Ordering::Relaxed) => {
713                if !started.load(Ordering::Relaxed) {
714                    tracing::debug!(listener = %kind, peer = %peer, "no request before the request-head timeout; closing");
715                    break;
716                }
717            }
718            changed = phase.changed() => {
719                if changed.is_err() {
720                    break;
721                }
722                match *phase.borrow_and_update() {
723                    Phase::Draining if !draining => {
724                        draining = true;
725                        conn.as_mut().graceful_shutdown();
726                    }
727                    Phase::Force => break,
728                    _ => {}
729                }
730            }
731        }
732    }
733}
