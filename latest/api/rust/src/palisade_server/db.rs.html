1//! The Postgres layer: connection pools, embedded migrations, the
2//! schema-version check and the per-transaction tenant scope.
3//!
4//! ADR 0006 (`docs/adr/0006-server-database.md`) decides the model:
5//! one shared schema, `tenant_id` on every tenant-owned row, and row-level
6//! security (RLS) that the database enforces for every query. This module
7//! supplies the pieces the rest of the server builds on:
8//!
9//! - [`Database::connect_runtime`]: the pool for the **runtime role**, the
10//!   only login the server serves requests with. That role owns nothing and
11//!   cannot bypass RLS. [`start`] refuses to start with a role that could,
12//!   and readiness keeps checking.
13//! - [`migrator`]: forward-only migrations from `migrations/`, embedded in
14//!   the binary. They run as the separate **migration role**, which owns the
15//!   schema ([`migrate_up`]).
16//! - [`schema_state`]: compares the applied migrations with the embedded
17//!   ones, so the server refuses to start against a schema newer than
18//!   itself, and `migrate verify` reports checksum mismatches.
19//! - [`TenantTx`]: the only way to run tenant-scoped queries. It begins a
20//!   transaction and sets `palisade.tenant_ids` with `SET LOCAL` semantics
21//!   before any query can run, so the scope never leaks to the next user of
22//!   a pooled connection.
23//!
24//! Every query is a fixed SQL string with typed bind parameters. Nothing
25//! here builds SQL from values.
26

44/// The schema that holds every Palisade object, the migration history
45/// included.
46pub const SCHEMA: &str = "palisade";
47/// The migration history table, schema-qualified.
48const HISTORY_TABLE: &str = "palisade._sqlx_migrations";
49/// The group role the runtime login must belong to.
50pub const RUNTIME_GROUP: &str = "palisade_runtime";
51/// The setting that holds the tenant scope of the current transaction.
52pub const TENANT_SETTING: &str = "palisade.tenant_ids";

61/// Environment variables that the Postgres client library reads when it
62/// builds connection options. Palisade sets every connection parameter from
63/// its own configuration, so these are refused rather than silently mixed
64/// in: a password or a TLS client key would bypass the `*_file` rule, and
65/// `PGOPTIONS` could change session settings such as `row_security`.
66pub const REFUSED_PG_ENV: &[&str] = &[
67    "PGPASSWORD",
68    "PGPASSFILE",
69    "PGOPTIONS",
70    "PGSSLKEY",
71    "PGSSLCERT",
72    "PGSSLMODE",
73    "PGSSLROOTCERT",
74    "PGHOSTADDR",
75];

77/// The embedded migrations, in version order, with the history table in
78/// the `palisade` schema.
79pub fn migrator() -> &'static Migrator {
80    static MIGRATOR: LazyLock<Migrator> = LazyLock::new(|| {
81        let mut m = sqlx::migrate!("./migrations");
82        m.dangerous_set_table_name(HISTORY_TABLE);
83        m.create_schema(SCHEMA);
84        m
85    });
86    &MIGRATOR
87}

105    }

108impl From<MigrateError> for DbError {
109    fn from(e: MigrateError) -> Self {
110        Self(match e {
111            MigrateError::VersionMismatch(v) => format!(
112                "migration {v} was applied with different contents than this binary embeds (checksum mismatch); the database or the binary has been changed"
113            ),
114            MigrateError::VersionMissing(v) => format!(
115                "migration {v} is applied in the database but not embedded in this binary; the schema is newer than this binary"
116            ),
117            MigrateError::Dirty(v) => {
118                format!("migration {v} failed part-way earlier; repair it by hand before retrying")
119            }
120            MigrateError::ExecuteMigration(inner, v) => {
121                format!("migration {v} failed: {}", describe(&inner))
122            }
123            MigrateError::Execute(inner) => describe(&inner),
124            other => other.to_string(),
125        })
126    }
127}

129/// A short description of a driver error. Database errors carry their
130/// SQLSTATE, which is what operators search for.
131fn describe(e: &sqlx::Error) -> String {
132    match e {
133        sqlx::Error::Database(d) => match d.code() {
134            Some(code) => format!("database error {code}: {}", d.message()),
135            None => format!("database error: {}", d.message()),
136        },
137        sqlx::Error::PoolTimedOut => "timed out waiting for a database connection".to_owned(),
138        other => other.to_string(),
139    }
140}

142/// Names of [`REFUSED_PG_ENV`] variables for which `is_set` is true.
143pub fn refused_env(is_set: impl Fn(&str) -> bool) -> Vec<&'static str> {
144    REFUSED_PG_ENV
145        .iter()
146        .copied()
147        .filter(|n| is_set(n))
148        .collect()
149}

151/// Fails if the process environment sets any [`REFUSED_PG_ENV`] variable.
152pub fn check_pg_env() -> Result<(), DbError> {
153    let found = refused_env(|n| std::env::var_os(n).is_some());
154    if found.is_empty() {
155        Ok(())
156    } else {
157        Err(DbError(format!(
158            "unset {}: palisade-server takes every database setting from its configuration, and these would override it (passwords come only from *_file paths)",
159            found.join(", ")
160        )))
161    }
162}

181/// Connection options for one login. The password is read from its secret
182/// file here and handed to the driver; it is never logged or returned. The
183/// driver keeps its own copy for reconnecting, which is not zeroised (#227).
184pub fn connect_options(
185    settings: &DatabaseSettings,
186    login: &DbLogin,
187) -> Result<ConnectOptions, DbError> {
188    check_pg_env()?;
189    let password = read_secret_file(&login.password_file, MAX_PASSWORD_BYTES).map_err(DbError)?;
190    let password = Zeroizing::new(
191        std::str::from_utf8(&password)
192            .map_err(|_| {
193                DbError(format!(
194                    "password file {} is not valid UTF-8",
195                    login.password_file.display()
196                ))
197            })?
198            // A trailing newline from `echo` or an editor is not part of
199            // the password.
200            .trim_end_matches(['\r', '\n'])
201            .to_owned(),
202    );
203    if password.is_empty() {
204        return Err(DbError(format!(
205            "password file {} is empty",
206            login.password_file.display()
207        )));
208    }
209    let mut options = PgConnectOptions::new_without_pgpass()
210        .host(&settings.host)
211        .port(settings.port)
212        .database(&settings.name)
213        .username(&login.user)
214        .password(&password)
215        .application_name(APPLICATION_NAME)
216        .ssl_mode(match settings.tls {
217            DbTls::Disable => PgSslMode::Disable,
218            DbTls::Require => PgSslMode::Require,
219            DbTls::VerifyCa => PgSslMode::VerifyCa,
220            DbTls::VerifyFull => PgSslMode::VerifyFull,
221        });
222    if let Some(ca) = &settings.ca_file {
223        // The driver accepts a file with no certificates and falls back to
224        // the OS roots alone, so check it first.
225        read_certs(ca).map_err(|e| DbError(format!("database.ca_file: {e}")))?;
226        options = options.ssl_root_cert(ca);
227    }
228    Ok(ConnectOptions(options))
229}

231/// Checks everything about the database configuration that can be checked
232/// without connecting: the refused `PG*` variables, every password file,
233/// and the CA file. `check-config` and `serve` run it before binding, so a
234/// bad file is a configuration error (exit 2).
235pub fn preflight(settings: &DatabaseSettings) -> Vec<String> {
236    if let Err(e) = check_pg_env() {
237        return vec![e.0];
238    }
239    let mut errors = Vec::new();
240    let logins = std::iter::once(("database", &settings.runtime))
241        .chain(settings.migration.iter().map(|m| ("database.migration", m)));
242    for (key, login) in logins {
243        if let Err(e) = connect_options(settings, login) {
244            let e = if e.0.starts_with("database.ca_file") {
245                e.0
246            } else {
247                format!("{key}.password_file: {e}")
248            };
249            if !errors.contains(&e) {
250                errors.push(e);
251            }
252        }
253    }
254    errors
255}

257/// The runtime pool, with the checks that apply to it.
258#[derive(Debug, Clone)]
259pub struct Database {
260    pool: PgPool,
261    check_timeout: Duration,
262}
263
264impl Database {
265    /// A pool for the runtime login. It connects lazily, so the server can
266    /// start (and report the database as down) while Postgres is
267    /// unreachable.
268    pub fn connect_runtime(settings: &DatabaseSettings) -> Result<Self, DbError> {
269        let options = connect_options(settings, &settings.runtime)?.0;
270        let pool = PgPoolOptions::new()
271            .max_connections(settings.max_connections)
272            .acquire_timeout(settings.connect_timeout)
273            .idle_timeout(Duration::from_secs(600))
274            .test_before_acquire(true)
275            .connect_lazy_with(options);
276        Ok(Self {
277            pool,
278            check_timeout: settings.connect_timeout,
279        })
280    }
281
282    /// The pool, for statements that are not tenant-scoped, such as the
283    /// readiness check. Tenant-owned tables return no rows through it.
284    pub fn pool(&self) -> &PgPool {
285        &self.pool
286    }
287
288    /// Begins a tenant-scoped transaction. See [`TenantTx`].
289    pub async fn tenant_tx(&self, scope: &TenantScope) -> Result<TenantTx, DbError> {
290        TenantTx::begin(&self.pool, scope).await
291    }
292
293    /// Closes every connection.
294    pub async fn close(&self) {
295        self.pool.close().await;
296    }
297}

322/// The tenants a transaction may see: the caller's resolved scope (#131).
323#[derive(Debug, Clone, PartialEq, Eq)]
324pub struct TenantScope(Vec<Uuid>);
325
326impl TenantScope {
327    /// A scope of the given tenants, sorted and without duplicates. An empty
328    /// scope is allowed and sees no tenant rows.
329    pub fn new(tenants: impl IntoIterator<Item = Uuid>) -> Result<Self, DbError> {
330        let mut ids: Vec<Uuid> = tenants.into_iter().collect();
331        ids.sort_unstable();
332        ids.dedup();
333        if ids.len() > MAX_SCOPE {
334            return Err(DbError(format!(
335                "a tenant scope holds at most {MAX_SCOPE} tenants, got {}",
336                ids.len()
337            )));
338        }
339        Ok(Self(ids))
340    }
341
342    /// A scope of one tenant.
343    pub fn one(tenant: Uuid) -> Self {
344        Self(vec![tenant])
345    }
346
347    /// The tenants in the scope.
348    pub fn tenants(&self) -> &[Uuid] {
349        &self.0
350    }
351
352    /// The value of [`TENANT_SETTING`]: a Postgres `uuid[]` literal. UUIDs
353    /// format as hex digits and hyphens only, so the literal needs no
354    /// escaping; it is passed as a bound parameter anyway.
355    fn setting_value(&self) -> String {
356        let inner: Vec<String> = self.0.iter().map(Uuid::to_string).collect();
357        format!("{{{}}}", inner.join(","))
358    }
359}

361/// A transaction with the tenant scope set.
362///
363/// `begin` opens the transaction and runs
364/// `set_config('palisade.tenant_ids', <scope>, true)` before handing it out.
365/// The third argument makes the setting local to the transaction (the same
366/// as `SET LOCAL`), so Postgres discards it at commit or rollback and the
367/// pooled connection goes back without a scope. A transaction dropped
368/// without [`commit`](Self::commit) rolls back.
369///
370/// RLS policies read the setting through `palisade.current_tenant_ids()`.
371/// It is still the server that chooses the scope: RLS stops a handler bug
372/// from crossing tenants, not a compromised server process (threat model
373/// AC-08).
374#[derive(Debug)]
375pub struct TenantTx {
376    tx: PgTransaction<'static>,
377}
378
379impl TenantTx {
380    async fn begin(pool: &PgPool, scope: &TenantScope) -> Result<Self, DbError> {
381        let mut tx = pool.begin().await?;
382        sqlx::query("SELECT pg_catalog.set_config($1, $2, true)")
383            .bind(TENANT_SETTING)
384            .bind(scope.setting_value())
385            .execute(&mut *tx)
386            .await?;
387        Ok(Self { tx })
388    }
389
390    /// The connection, for running queries inside the scope.
391    pub fn conn(&mut self) -> &mut PgConnection {
392        &mut self.tx
393    }
394
395    /// Commits. The scope ends with the transaction.
396    pub async fn commit(self) -> Result<(), DbError> {
397        self.tx.commit().await?;
398        Ok(())
399    }
400
401    /// Rolls back. The scope ends with the transaction.
402    pub async fn rollback(self) -> Result<(), DbError> {
403        self.tx.rollback().await?;
404        Ok(())
405    }
406}

408/// What the runtime login is allowed to do, as Postgres reports it.
409#[derive(Debug, Clone, PartialEq, Eq, sqlx::FromRow)]
410pub struct RoleFacts {
411    /// The login name.
412    pub role_name: String,
413    /// The login, or a role it is a member of, is a superuser.
414    pub superuser: bool,
415    /// The login, or a role it is a member of, has `BYPASSRLS`.
416    pub bypass_rls: bool,
417    /// Member of [`RUNTIME_GROUP`].
418    pub in_runtime_group: bool,
419    /// The login, or a role it is a member of, owns the `palisade` schema or
420    /// a table, view, sequence, function or type in it. An owner can
421    /// disable row-level security, change policies, or replace
422    /// `palisade.current_tenant_ids()`.
423    pub owns_objects: bool,
424    /// The login, or a role it is a member of, can create objects in the
425    /// `palisade` schema.
426    pub can_create: bool,
427}
428
429impl RoleFacts {
430    /// Problems that make the login unfit to be the runtime role.
431    pub fn problems(&self) -> Vec<String> {
432        let mut v = Vec::new();
433        let user = &self.role_name;
434        if self.superuser {
435            v.push(format!(
436                "role {user} is, or is a member of, a superuser, so row-level security does not apply to it"
437            ));
438        }
439        if self.bypass_rls {
440            v.push(format!(
441                "role {user} has BYPASSRLS, directly or through a role it is a member of, so row-level security does not apply to it"
442            ));
443        }
444        if self.owns_objects {
445            v.push(format!(
446                "role {user} owns objects in schema {SCHEMA}, directly or through a role it is a member of; an owner can disable row-level security, so serve requests with a separate non-owner role"
447            ));
448        }
449        if self.can_create {
450            v.push(format!("role {user} can create objects in schema {SCHEMA}"));
451        }
452        if !self.in_runtime_group {
453            v.push(format!("role {user} is not a member of {RUNTIME_GROUP}"));
454        }
455        v
456    }
457}

459/// Reads [`RoleFacts`] for the current login.
460pub async fn role_facts(conn: &mut PgConnection) -> Result<RoleFacts, DbError> {
461    let facts = sqlx::query_as::<_, RoleFacts>(
462        // Every check follows role membership: a role the login can
463        // `SET ROLE` to, or inherits from, counts as the login itself.
464        // `pg_has_role(x, x, 'MEMBER')` is true, so the login is included.
465        "WITH me AS (
466             SELECT oid, rolname FROM pg_catalog.pg_roles WHERE rolname = current_user
467         ),
468         reach AS (
469             SELECT o.oid, o.rolsuper, o.rolbypassrls
470             FROM pg_catalog.pg_roles o, me
471             WHERE pg_catalog.pg_has_role(me.oid, o.oid, 'MEMBER')
472         )
473         SELECT me.rolname::text AS role_name,
474                EXISTS (SELECT 1 FROM reach WHERE rolsuper) AS superuser,
475                EXISTS (SELECT 1 FROM reach WHERE rolbypassrls) AS bypass_rls,
476                EXISTS (
477                    SELECT 1 FROM pg_catalog.pg_roles g
478                    WHERE g.rolname = $2 AND pg_catalog.pg_has_role(me.oid, g.oid, 'MEMBER')
479                ) AS in_runtime_group,
480                EXISTS (
481                    SELECT 1 FROM pg_catalog.pg_namespace n
482                    WHERE n.nspname = $1 AND n.nspowner IN (SELECT oid FROM reach)
483                    UNION ALL
484                    SELECT 1 FROM pg_catalog.pg_class c
485                    JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace
486                    WHERE n.nspname = $1 AND c.relowner IN (SELECT oid FROM reach)
487                    UNION ALL
488                    SELECT 1 FROM pg_catalog.pg_proc f
489                    JOIN pg_catalog.pg_namespace n ON n.oid = f.pronamespace
490                    WHERE n.nspname = $1 AND f.proowner IN (SELECT oid FROM reach)
491                    UNION ALL
492                    SELECT 1 FROM pg_catalog.pg_type t
493                    JOIN pg_catalog.pg_namespace n ON n.oid = t.typnamespace
494                    WHERE n.nspname = $1 AND t.typowner IN (SELECT oid FROM reach)
495                ) AS owns_objects,
496                EXISTS (
497                    SELECT 1 FROM pg_catalog.pg_namespace n, reach
498                    WHERE n.nspname = $1
499                      AND pg_catalog.has_schema_privilege(reach.oid, n.oid, 'CREATE')
500                ) AS can_create
501         FROM me",
502    )
503    .bind(SCHEMA)
504    .bind(RUNTIME_GROUP)
505    .fetch_one(&mut *conn)
506    .await?;
507    Ok(facts)
508}

510/// Fails unless the current login is fit to be the runtime role.
511pub async fn check_runtime_role(conn: &mut PgConnection) -> Result<(), DbError> {
512    let problems = role_facts(conn).await?.problems();
513    if problems.is_empty() {
514        Ok(())
515    } else {
516        Err(DbError(problems.join("; ")))
517    }
518}

520/// One migration as embedded and as applied.
521#[derive(Debug, Clone, PartialEq, Eq)]
522pub struct MigrationStatus {
523    /// The version number.
524    pub version: i64,
525    /// The description from the file name.
526    pub description: String,
527    /// Whether the binary embeds it.
528    pub embedded: bool,
529    /// Whether the database records it as applied successfully.
530    pub applied: bool,
531    /// Whether the database records it as failed part-way.
532    pub failed: bool,
533    /// Whether the applied checksum differs from the embedded one.
534    pub checksum_mismatch: bool,
535}

537/// The schema compared with this binary.
538#[derive(Debug, Clone, PartialEq, Eq)]
539pub struct SchemaState {
540    /// Every migration known to the binary or the database, by version.
541    pub migrations: Vec<MigrationStatus>,
542}
543
544/// How the schema relates to the binary.
545#[derive(Debug, Clone, Copy, PartialEq, Eq)]
546pub enum SchemaVerdict {
547    /// Every embedded migration is applied and matches.
548    Current,
549    /// Some embedded migrations are not applied yet.
550    Behind,
551    /// The database has a migration this binary does not embed.
552    Newer,
553    /// An applied migration's checksum differs, or one failed part-way.
554    Damaged,
555}
556
557impl SchemaVerdict {
558    /// The name used in output and readiness.
559    pub fn as_str(self) -> &'static str {
560        match self {
561            Self::Current => "current",
562            Self::Behind => "behind",
563            Self::Newer => "newer",
564            Self::Damaged => "damaged",
565        }
566    }
567}
568
569impl SchemaState {
570    /// The overall verdict. `Damaged` wins over `Newer`, which wins over
571    /// `Behind`.
572    pub fn verdict(&self) -> SchemaVerdict {
573        let m = &self.migrations;
574        if m.iter().any(|s| s.checksum_mismatch || s.failed) {
575            SchemaVerdict::Damaged
576        } else if m.iter().any(|s| s.applied && !s.embedded) {
577            SchemaVerdict::Newer
578        } else if m.iter().any(|s| s.embedded && !s.applied) {
579            SchemaVerdict::Behind
580        } else {
581            SchemaVerdict::Current
582        }
583    }
584
585    /// The highest applied version, if any.
586    pub fn applied_version(&self) -> Option<i64> {
587        self.migrations
588            .iter()
589            .filter(|s| s.applied)
590            .map(|s| s.version)
591            .max()
592    }
593
594    /// The highest embedded version.
595    pub fn embedded_version(&self) -> Option<i64> {
596        self.migrations
597            .iter()
598            .filter(|s| s.embedded)
599            .map(|s| s.version)
600            .max()
601    }
602
603    /// As JSON, for `migrate status` and `migrate verify`.
604    pub fn to_json(&self) -> Value {
605        json!({
606            "verdict": self.verdict().as_str(),
607            "applied_version": self.applied_version(),
608            "embedded_version": self.embedded_version(),
609            "migrations": self.migrations.iter().map(|m| json!({
610                "version": m.version,
611                "description": m.description,
612                "embedded": m.embedded,
613                "applied": m.applied,
614                "failed": m.failed,
615                "checksum_mismatch": m.checksum_mismatch,
616            })).collect::<Vec<_>>(),
617        })
618    }
619}

621/// Compares the applied migrations with [`migrator`]. Works for the
622/// migration role and the runtime role (which may read the history). A
623/// database with no `palisade` schema yet has nothing applied.
624pub async fn schema_state(conn: &mut PgConnection) -> Result<SchemaState, DbError> {
625    let has_history: bool = sqlx::query_scalar(
626        "SELECT EXISTS (
627             SELECT 1 FROM pg_catalog.pg_class c
628             JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace
629             WHERE n.nspname = $1 AND c.relname = '_sqlx_migrations'
630         )",
631    )
632    .bind(SCHEMA)
633    .fetch_one(&mut *conn)
634    .await?;
635    let applied: Vec<(i64, String, Vec<u8>, bool)> = if has_history {
636        sqlx::query_as(
637            "SELECT version, description, checksum, success
638             FROM palisade._sqlx_migrations ORDER BY version",
639        )
640        .fetch_all(&mut *conn)
641        .await?
642    } else {
643        Vec::new()
644    };
645    Ok(compare(migrator(), &applied))
646}

648/// The comparison behind [`schema_state`], without a database.
649fn compare(migrator: &Migrator, applied: &[(i64, String, Vec<u8>, bool)]) -> SchemaState {
650    let mut by_version: BTreeMap<i64, MigrationStatus> = BTreeMap::new();
651    for m in migrator.iter() {
652        if m.migration_type.is_down_migration() {
653            continue;
654        }
655        by_version.insert(
656            m.version,
657            MigrationStatus {
658                version: m.version,
659                description: m.description.to_string(),
660                embedded: true,
661                applied: false,
662                failed: false,
663                checksum_mismatch: false,
664            },
665        );
666    }
667    for (version, description, checksum, success) in applied {
668        let embedded = migrator.iter().find(|m| m.version == *version);
669        let entry = by_version
670            .entry(*version)
671            .or_insert_with(|| MigrationStatus {
672                version: *version,
673                description: description.clone(),
674                embedded: false,
675                applied: false,
676                failed: false,
677                checksum_mismatch: false,
678            });
679        entry.applied = *success;
680        entry.failed = !*success;
681        entry.checksum_mismatch = embedded.is_some_and(|m| *m.checksum != **checksum);
682    }
683    SchemaState {
684        migrations: by_version.into_values().collect(),
685    }
686}

688/// What the migration login is, as Postgres reports it.
689#[derive(Debug, Clone, PartialEq, Eq, sqlx::FromRow)]
690struct MigrationRoleFacts {
691    role_name: String,
692    superuser: bool,
693    runtime_group_exists: bool,
694    /// The login can act as the runtime group, or the runtime group can act
695    /// as the login. Either way the runtime role could own the schema.
696    shares_runtime_group: bool,
697    /// The owner of the `palisade` schema, if it exists.
698    schema_owner: Option<String>,
699    /// The login is the schema owner or a member of it.
700    owns_schema: bool,
701}
702
703/// Fails unless the current login may run migrations: not a superuser,
704/// `palisade_runtime` exists, the login and `palisade_runtime` are not
705/// members of each other, and the login owns the `palisade` schema if it
706/// already exists. It runs before anything is created, so a mistaken run as
707/// another role changes nothing.
708pub async fn check_migration_role(conn: &mut PgConnection) -> Result<(), DbError> {
709    let f = sqlx::query_as::<_, MigrationRoleFacts>(
710        "SELECT me.rolname::text AS role_name,
711                me.rolsuper AS superuser,
712                g.oid IS NOT NULL AS runtime_group_exists,
713                COALESCE(pg_catalog.pg_has_role(me.oid, g.oid, 'MEMBER')
714                      OR pg_catalog.pg_has_role(g.oid, me.oid, 'MEMBER'), false) AS shares_runtime_group,
715                o.rolname::text AS schema_owner,
716                COALESCE(pg_catalog.pg_has_role(me.oid, n.nspowner, 'MEMBER'), false) AS owns_schema
717         FROM pg_catalog.pg_roles me
718         LEFT JOIN pg_catalog.pg_roles g ON g.rolname = $2
719         LEFT JOIN pg_catalog.pg_namespace n ON n.nspname = $1
720         LEFT JOIN pg_catalog.pg_roles o ON o.oid = n.nspowner
721         WHERE me.rolname = current_user",
722    )
723    .bind(SCHEMA)
724    .bind(RUNTIME_GROUP)
725    .fetch_one(&mut *conn)
726    .await?;
727    let user = &f.role_name;
728    if f.superuser {
729        return Err(DbError(format!(
730            "role {user} is a superuser; run migrations as the schema owner role created by sql/bootstrap.sql"
731        )));
732    }
733    if !f.runtime_group_exists {
734        return Err(DbError(format!(
735            "role {RUNTIME_GROUP} does not exist; run sql/bootstrap.sql first"
736        )));
737    }
738    if f.shares_runtime_group {
739        return Err(DbError(format!(
740            "role {user} and {RUNTIME_GROUP} must not be members of each other"
741        )));
742    }
743    if let Some(owner) = &f.schema_owner {
744        if !f.owns_schema {
745            return Err(DbError(format!(
746                "schema {SCHEMA} is owned by {owner}, not {user}; run migrations as its owner"
747            )));
748        }
749    }
750    Ok(())
751}

753/// Applies every pending migration, as the migration role. Refuses a wrong
754/// role, and a schema that is newer than this binary or damaged, before
755/// changing anything.
756pub async fn migrate_up(conn: &mut PgConnection) -> Result<SchemaState, DbError> {
757    check_migration_role(conn).await?;
758    let before = schema_state(conn).await?;
759    match before.verdict() {
760        SchemaVerdict::Newer => {
761            return Err(DbError(
762                "the database schema is newer than this binary; upgrade palisade-server instead"
763                    .to_owned(),
764            ));
765        }
766        SchemaVerdict::Damaged => {
767            return Err(DbError(
768                "the database schema does not match this binary's migrations; run `migrate verify` for details".to_owned(),
769            ));
770        }
771        SchemaVerdict::Behind | SchemaVerdict::Current => {}
772    }
773    migrator().run_direct(None, conn, false).await?;
774    schema_state(conn).await
775}

777/// Structural checks on the applied schema, beyond the checksums: every
778/// table in the `palisade` schema that has a `tenant_id` column, or is the
779/// tenant table, must have row-level security enabled and forced and at
780/// least one policy; every foreign key from a tenant-owned table to another
781/// tenant-owned table must include `tenant_id`; and the runtime group must
782/// not own anything. Returns one message per problem.
783pub async fn verify_structure(conn: &mut PgConnection) -> Result<Vec<String>, DbError> {
784    let rows: Vec<(String, bool, bool, i64, bool)> = sqlx::query_as(
785        "SELECT c.relname::text,
786                c.relrowsecurity,
787                c.relforcerowsecurity,
788                (SELECT count(*) FROM pg_catalog.pg_policy p WHERE p.polrelid = c.oid),
789                COALESCE((
790                    SELECT a.attnotnull AND a.atttypid = 'pg_catalog.uuid'::pg_catalog.regtype
791                    FROM pg_catalog.pg_attribute a
792                    WHERE a.attrelid = c.oid AND a.attname = 'tenant_id' AND NOT a.attisdropped
793                ), true)
794         FROM pg_catalog.pg_class c
795         JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace
796         WHERE n.nspname = $1 AND c.relkind IN ('r', 'p')
797           AND (c.relname = 'tenant' OR EXISTS (
798               SELECT 1 FROM pg_catalog.pg_attribute a
799               WHERE a.attrelid = c.oid AND a.attname = 'tenant_id' AND NOT a.attisdropped
800           ))
801         ORDER BY 1",
802    )
803    .bind(SCHEMA)
804    .fetch_all(&mut *conn)
805    .await?;
806    let mut problems = Vec::new();
807    for (table, enabled, forced, policies, tenant_id_ok) in &rows {
808        if !tenant_id_ok {
809            problems.push(format!(
810                "table {SCHEMA}.{table}: tenant_id must be uuid NOT NULL"
811            ));
812        }
813        if !enabled {
814            problems.push(format!(
815                "table {SCHEMA}.{table}: row-level security is not enabled"
816            ));
817        }
818        if !forced {
819            problems.push(format!(
820                "table {SCHEMA}.{table}: row-level security is not forced"
821            ));
822        }
823        if *policies == 0 {
824            problems.push(format!(
825                "table {SCHEMA}.{table}: has no row-level security policy"
826            ));
827        }
828    }
829
830    // Foreign keys between two tenant-owned tables (both have tenant_id)
831    // must map tenant_id to tenant_id.
832    let fks: Vec<(String, String)> = sqlx::query_as(
833        "SELECT con.conname::text, src.relname::text
834         FROM pg_catalog.pg_constraint con
835         JOIN pg_catalog.pg_class src ON src.oid = con.conrelid
836         JOIN pg_catalog.pg_class dst ON dst.oid = con.confrelid
837         JOIN pg_catalog.pg_namespace n ON n.oid = src.relnamespace
838         JOIN pg_catalog.pg_attribute sa
839           ON sa.attrelid = src.oid AND sa.attname = 'tenant_id' AND NOT sa.attisdropped
840         JOIN pg_catalog.pg_attribute da
841           ON da.attrelid = dst.oid AND da.attname = 'tenant_id' AND NOT da.attisdropped
842         WHERE con.contype = 'f' AND n.nspname = $1
843           AND NOT EXISTS (
844               SELECT 1 FROM unnest(con.conkey, con.confkey) AS k(src_col, dst_col)
845               WHERE k.src_col = sa.attnum AND k.dst_col = da.attnum
846           )
847         ORDER BY 1",
848    )
849    .bind(SCHEMA)
850    .fetch_all(&mut *conn)
851    .await?;
852    for (name, table) in &fks {
853        problems.push(format!(
854            "table {SCHEMA}.{table}: foreign key {name} does not include tenant_id, so a row could reference another tenant's row"
855        ));
856    }
857
858    // Every role a runtime login could act as: the members of the runtime
859    // group (its logins), and every role any of them can reach by
860    // membership, with or without INHERIT. None of those may own anything
861    // in the schema or be a superuser or BYPASSRLS. The walk follows
862    // pg_auth_members rather than pg_has_role, which is always true for a
863    // superuser and would count every superuser as a member.
864    let (exists, owned, privileged): (bool, i64, bool) = sqlx::query_as(
865        "WITH RECURSIVE
866         g AS (SELECT oid FROM pg_catalog.pg_roles WHERE rolname = $2),
867         members(oid) AS (
868             SELECT oid FROM g
869             UNION
870             SELECT am.member FROM pg_catalog.pg_auth_members am
871             JOIN members m ON am.roleid = m.oid
872         ),
873         reach(oid) AS (
874             SELECT oid FROM members
875             UNION
876             SELECT am.roleid FROM pg_catalog.pg_auth_members am
877             JOIN reach r ON am.member = r.oid
878         ),
879         owners AS (
880             SELECT n.nspowner AS owner FROM pg_catalog.pg_namespace n WHERE n.nspname = $1
881             UNION ALL
882             SELECT c.relowner FROM pg_catalog.pg_class c
883             JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace WHERE n.nspname = $1
884             UNION ALL
885             SELECT f.proowner FROM pg_catalog.pg_proc f
886             JOIN pg_catalog.pg_namespace n ON n.oid = f.pronamespace WHERE n.nspname = $1
887             UNION ALL
888             SELECT t.typowner FROM pg_catalog.pg_type t
889             JOIN pg_catalog.pg_namespace n ON n.oid = t.typnamespace WHERE n.nspname = $1
890         )
891         SELECT EXISTS (SELECT 1 FROM g),
892                (SELECT count(*) FROM owners WHERE owner IN (SELECT oid FROM reach)),
893                EXISTS (SELECT 1 FROM pg_catalog.pg_roles o
894                        WHERE (o.rolsuper OR o.rolbypassrls) AND o.oid IN (SELECT oid FROM reach))",
895    )
896    .bind(SCHEMA)
897    .bind(RUNTIME_GROUP)
898    .fetch_one(&mut *conn)
899    .await?;
900    if !exists {
901        problems.push(format!(
902            "role {RUNTIME_GROUP} does not exist; run sql/bootstrap.sql"
903        ));
904    }
905    if owned > 0 {
906        problems.push(format!(
907            "{owned} object(s) in schema {SCHEMA} are owned by a role that a {RUNTIME_GROUP} login can act as"
908        ));
909    }
910    if privileged {
911        problems.push(format!(
912            "a {RUNTIME_GROUP} login can act as a superuser or BYPASSRLS role"
913        ));
914    }
915    Ok(problems)
916}

918/// Start-up database work for `serve`, run before any listener binds.
919///
920/// - With `auto_migrate`, connects as the migration role and applies pending
921///   migrations ([`migrate_up`] refuses a wrong role, and a newer or damaged
922///   schema).
923/// - Builds the runtime pool and, if the database is reachable, refuses a
924///   runtime login that could bypass row-level security, and a schema that
925///   is newer than this binary or damaged.
926///
927/// An unreachable database does not stop the server: it starts not ready
928/// (`database_down`) and [`watch`] keeps checking. Migrations that could
929/// not run because the database was unreachable are not retried until the
930/// next start or `migrate up`.
931pub async fn start(settings: &DatabaseSettings) -> Result<Database, DbError> {
932    check_pg_env()?;
933    if settings.auto_migrate {
934        let options = migration_options(settings)?;
935        match connect(&options, settings.connect_timeout).await {
936            Ok(mut conn) => {
937                let result = migrate_up(&mut conn).await;
938                let _ = conn.close().await;
939                let state = result?;
940                tracing::info!(
941                    schema_version = state.applied_version(),
942                    "database schema is current"
943                );
944            }
945            Err(e) => tracing::warn!(
946                error = %e,
947                "cannot reach the database to migrate; starting not ready"
948            ),
949        }
950    }
951    let database = Database::connect_runtime(settings)?;
952    // The role is checked before the schema: a login outside
953    // palisade_runtime cannot read the schema at all, and that permission
954    // error must not hide the role problem as "not reachable yet".
955    enum Checked {
956        Unsafe(Vec<String>),
957        Schema(SchemaState),
958    }
959    let checked = tokio::time::timeout(settings.connect_timeout, async {
960        let mut conn = database.pool().acquire().await?;
961        let problems = role_facts(&mut conn).await?.problems();
962        if !problems.is_empty() {
963            return Ok(Checked::Unsafe(problems));
964        }
965        Ok::<_, DbError>(Checked::Schema(schema_state(&mut conn).await?))
966    })
967    .await;
968    match checked {
969        Ok(Ok(Checked::Unsafe(problems))) => return Err(DbError(problems.join("; "))),
970        Ok(Ok(Checked::Schema(state))) => match state.verdict() {
971            SchemaVerdict::Newer => {
972                return Err(DbError(
973                    "the database schema is newer than this binary; upgrade palisade-server"
974                        .to_owned(),
975                ));
976            }
977            SchemaVerdict::Damaged => {
978                return Err(DbError(
979                    "the database schema does not match this binary's migrations; run `migrate verify` for details".to_owned(),
980                ));
981            }
982            SchemaVerdict::Behind | SchemaVerdict::Current => {}
983        },
984        Ok(Err(e)) => {
985            tracing::warn!(error = %e, "cannot check the database yet; starting not ready");
986        }
987        Err(_) => tracing::warn!("the database check timed out; starting not ready"),
988    }
989    Ok(database)
990}

992/// The database state for readiness, with the reason when it is not up.
993/// The whole check is bounded by the connect timeout, so a stalled
994/// connection or a lock on the history table reports the database as down
995/// instead of freezing readiness at its last state.
996pub async fn health_check(database: &Database) -> (DatabaseHealth, String) {
997    match tokio::time::timeout(database.check_timeout, health_check_inner(database)).await {
998        Ok(r) => r,
999        Err(_) => (
1000            DatabaseHealth::Down,
1001            "the database check timed out".to_owned(),
1002        ),
1003    }
1004}
1005
1006async fn health_check_inner(database: &Database) -> (DatabaseHealth, String) {
1007    let mut conn = match database.pool().acquire().await {
1008        Ok(c) => c,
1009        Err(e) => return (DatabaseHealth::Down, DbError::from(e).0),
1010    };
1011    let facts = match role_facts(&mut conn).await {
1012        Ok(f) => f,
1013        Err(e) => return (DatabaseHealth::Down, e.0),
1014    };
1015    let problems = facts.problems();
1016    if !problems.is_empty() {
1017        return (DatabaseHealth::RoleUnsafe, problems.join("; "));
1018    }
1019    match schema_state(&mut conn).await {
1020        Ok(s) if s.verdict() == SchemaVerdict::Current => (DatabaseHealth::Up, String::new()),
1021        Ok(s) => (
1022            DatabaseHealth::Schema(s.verdict()),
1023            format!("the schema is {}", s.verdict().as_str()),
1024        ),
1025        Err(e) => (DatabaseHealth::Down, e.0),
1026    }
1027}

1029/// Runs [`health_check`] every `interval` and records the result in
1030/// `health`, logging each change, until the task is aborted.
1031pub async fn watch(database: Database, health: Arc<Health>, interval: Duration) {
1032    let mut last: Option<DatabaseHealth> = None;
1033    loop {
1034        let (state, reason) = health_check(&database).await;
1035        if last != Some(state) {
1036            if state == DatabaseHealth::Up {
1037                tracing::info!("database ready");
1038            } else {
1039                tracing::warn!(reason = %reason, "database not ready");
1040            }
1041        }
1042        health.set_database(state);
1043        last = Some(state);
1044        tokio::time::sleep(interval).await;
1045    }
1046}
