1//! RFC 9457 problem details (`application/problem+json`).
2//!
3//! Every error response the server sends is a problem document. Handlers
4//! return a [`Problem`]; the request middleware in [`crate::http`] renders it
5//! and adds the request ID. Error responses produced elsewhere in the stack
6//! (no matching route, wrong method, body too large, timeout, panic, an
7//! extractor rejection) are replaced by a generic problem for their status,
8//! so their original bodies, which may describe internals, never reach the
9//! client.
10//!
11//! The `type` URIs are stable identifiers under
12//! `https://palisade.dev/problems/<slug>`, following the catalogue in #185.
13
14use axum::http::{HeaderValue, StatusCode, header};
15use axum::response::{IntoResponse, Response};
16use serde_json::{Map, Value, json};
17
18/// The base of every problem `type` URI.
19pub const TYPE_BASE: &str = "https://palisade.dev/problems/";
20/// The media type of a problem document.
21pub const CONTENT_TYPE: &str = "application/problem+json";
22
23/// A problem to report to the client.
24#[derive(Debug, Clone, PartialEq, Eq)]
25pub struct Problem {
26    /// HTTP status.
27    pub status: StatusCode,
28    /// The last segment of the `type` URI.
29    pub slug: &'static str,
30    /// A short, fixed summary of the problem type.
31    pub title: &'static str,
32    /// An explanation of this occurrence. Never contains internals such as
33    /// SQL, file paths or stack traces.
34    pub detail: String,
35    /// Extension members (RFC 9457 section 3.2), added after the standard
36    /// members. Keys that clash with a standard member are ignored.
37    pub extensions: Map<String, Value>,
38}
39
40impl Problem {
41    /// The generic problem for an error `status` that carries no problem of
42    /// its own.
43    pub fn for_status(status: StatusCode) -> Self {
44        let (slug, title, detail) = match status {
45            StatusCode::BAD_REQUEST => (
46                "bad-request",
47                "Bad request",
48                "The request could not be understood.",
49            ),
50            StatusCode::NOT_FOUND => ("not-found", "Not found", "No resource exists at this path."),
51            StatusCode::METHOD_NOT_ALLOWED => (
52                "method-not-allowed",
53                "Method not allowed",
54                "This resource does not support the request method.",
55            ),
56            StatusCode::PAYLOAD_TOO_LARGE => (
57                "payload-too-large",
58                "Payload too large",
59                "The request body exceeds the limit for this endpoint.",
60            ),
61            StatusCode::UNSUPPORTED_MEDIA_TYPE => (
62                "unsupported-media-type",
63                "Unsupported media type",
64                "The request body has a media type this endpoint does not accept.",
65            ),
66            StatusCode::UNPROCESSABLE_ENTITY => (
67                "validation",
68                "Validation failed",
69                "The request body is not valid for this endpoint.",
70            ),
71            StatusCode::REQUEST_TIMEOUT => (
72                "timeout",
73                "Request timeout",
74                "The request was not completed in time. Re-read the resource before retrying a change.",
75            ),
76            StatusCode::SERVICE_UNAVAILABLE => (
77                "unavailable",
78                "Service unavailable",
79                "The server could not complete the request in time. Re-read the resource before retrying a change.",
80            ),
81            s if s.is_server_error() => (
82                "internal",
83                "Internal server error",
84                "The server failed to complete the request. Re-read the resource before retrying a change.",
85            ),
86            _ => (
87                "client-error",
88                "Request rejected",
89                "The request was rejected.",
90            ),
91        };
92        Self {
93            status,
94            slug,
95            title,
96            detail: detail.to_owned(),
97            extensions: Map::new(),
98        }
99    }
100
101    /// Adds an extension member.
102    #[must_use]
103    pub fn with_extension(mut self, key: &str, value: Value) -> Self {
104        self.extensions.insert(key.to_owned(), value);
105        self
106    }
107
108    /// The `type` URI.
109    pub fn type_uri(&self) -> String {
110        format!("{TYPE_BASE}{}", self.slug)
111    }
112
113    /// Renders the problem document for the request `request_id`.
114    pub fn render(&self, request_id: &str) -> Response {
115        let mut body = json!({
116            "type": self.type_uri(),
117            "title": self.title,
118            "status": self.status.as_u16(),
119            "detail": self.detail,
120            "instance": format!("urn:palisade:request:{request_id}"),
121            "request_id": request_id,
122        });
123        if let Value::Object(members) = &mut body {
124            for (k, v) in &self.extensions {
125                members.entry(k.clone()).or_insert_with(|| v.clone());
126            }
127        }
128        let mut res = (self.status, body.to_string()).into_response();
129        let headers = res.headers_mut();
130        headers.insert(header::CONTENT_TYPE, HeaderValue::from_static(CONTENT_TYPE));
131        headers.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
132        res
133    }
134}
135
136/// A handler can return a [`Problem`] directly. The body is filled in by the
137/// request middleware, which knows the request ID.
138impl IntoResponse for Problem {
139    fn into_response(self) -> Response {
140        let mut res = self.status.into_response();
141        res.extensions_mut().insert(self);
142        res
143    }
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149
150    #[test]
151    fn render_has_every_member() {
152        let p = Problem::for_status(StatusCode::PAYLOAD_TOO_LARGE);
153        let res = p.render("abc123");
154        assert_eq!(res.status(), StatusCode::PAYLOAD_TOO_LARGE);
155        assert_eq!(res.headers()[header::CONTENT_TYPE], CONTENT_TYPE);
156        assert_eq!(res.headers()[header::CACHE_CONTROL], "no-store");
157    }
158
159    #[test]
160    fn extensions_never_override_standard_members() {
161        let p = Problem::for_status(StatusCode::NOT_FOUND)
162            .with_extension("status", json!(200))
163            .with_extension("indicators", json!(["x"]));
164        let res = p.render("r1");
165        assert_eq!(res.status(), StatusCode::NOT_FOUND);
166        let body = futures_body(res);
167        assert_eq!(body["status"], 404);
168        assert_eq!(body["indicators"], json!(["x"]));
169        assert_eq!(body["request_id"], "r1");
170        assert_eq!(body["instance"], "urn:palisade:request:r1");
171        assert_eq!(body["type"], "https://palisade.dev/problems/not-found");
172    }
173
174    fn futures_body(res: Response) -> Value {
175        let rt = tokio::runtime::Builder::new_current_thread()
176            .build()
177            .unwrap();
178        let bytes = rt
179            .block_on(axum::body::to_bytes(res.into_body(), 1 << 20))
180            .unwrap();
181        serde_json::from_slice(&bytes).unwrap()
182    }
183
184    #[test]
185    fn unknown_server_errors_are_internal() {
186        let p = Problem::for_status(StatusCode::BAD_GATEWAY);
187        assert_eq!(p.slug, "internal");
188        assert_eq!(p.type_uri(), "https://palisade.dev/problems/internal");
189    }
190
191    #[test]
192    fn unknown_client_errors_are_generic() {
193        let p = Problem::for_status(StatusCode::IM_A_TEAPOT);
194        assert_eq!(p.slug, "client-error");
195    }
