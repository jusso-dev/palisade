1//! The axum router and the tower middleware stack for one listener.
2//!
3//! Layers, outermost first:
4//!
5//! 1. **Request ID and error normalisation.** Accepts a well-formed
6//!    `Palisade-Request-Id` from the client or generates a random one,
7//!    echoes it on the response, and turns every error response into an
8//!    RFC 9457 problem document (see [`crate::problem`]).
9//! 2. **Trace.** One span per request with `request_id`, `listener`,
10//!    `method` and the matched `route` template (never the raw path or
11//!    query, which may carry tokens), and one log line per response with the
12//!    status and latency.
13//! 3. **Panic guard.** A panicking handler becomes a 500 problem; the panic
14//!    message stays in the server log.
15//! 4. **Timeout.** A handler that runs past the listener's
16//!    `request_timeout_secs` is cancelled with a 503 problem.
17//! 5. **Body limit.** A body larger than the listener's `body_limit_bytes`
18//!    gets a 413 problem, whether the size is declared up front or only
19//!    discovered while streaming.
20//!
21//! Tenant, principal, authentication, rate-limit, CORS and compression
22//! layers join this stack with the API work in #185, #130 and #229.
23

46/// The request ID header, accepted from clients and set on every response.
47pub const REQUEST_ID: HeaderName = HeaderName::from_static("palisade-request-id");
48/// Longest client-supplied request ID that is accepted as is.
49const MAX_REQUEST_ID: usize = 64;
50
51/// Headers from an original error response that survive normalisation.
52const KEPT_ERROR_HEADERS: [HeaderName; 3] =
53    [header::ALLOW, header::RETRY_AFTER, header::WWW_AUTHENTICATE];
54
55/// The request ID of the current request, available as a request
56/// extension to every layer and handler inside the first one.
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub struct RequestId(pub String);

60/// The remote address of the connection, inserted by the listener.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub struct Peer(pub SocketAddr);

69    /// Whether `/readyz` includes the indicator list. Only the metrics

72}

84/// Wraps `router` in the middleware stack, configured for `settings`.
85pub fn apply_stack(router: Router, settings: &ListenerSettings) -> Router {
86    let listener = settings.kind.name();
87    let trace = TraceLayer::new_for_http()
88        .make_span_with(move |req: &Request| {
89            let id = req
90                .extensions()
91                .get::<RequestId>()
92                .map_or("", |r| r.0.as_str());
93            let route = req
94                .extensions()
95                .get::<MatchedPath>()
96                .map_or("unmatched", MatchedPath::as_str);
97            let peer = req
98                .extensions()
99                .get::<Peer>()
100                .map(|p| p.0.to_string())
101                .unwrap_or_default();
102            tracing::info_span!(
103                "request",
104                request_id = id,
105                listener,
106                method = %req.method(),
107                route,
108                peer,
109            )
110        })
111        .on_request(())
112        .on_response(|res: &Response, latency: Duration, _span: &Span| {
113            let status = res.status().as_u16();
114            let latency_ms = u64::try_from(latency.as_millis()).unwrap_or(u64::MAX);
115            if res.status().is_server_error() {
116                tracing::warn!(status, latency_ms, "request failed");
117            } else if res.status().is_success() {
118                // Successful probes arrive every few seconds; keep them out
119                // of the default log level.
120                tracing::debug!(status, latency_ms, "request completed");
121            } else {
122                tracing::info!(status, latency_ms, "request completed");
123            }
124        })
125        .on_body_chunk(())
126        .on_eos(())
127        .on_failure(());
128    // `Router::layer` wraps every route and the fallback, innermost first,
129    // and converts each layer's response back into an axum response.
130    router
131        .layer(DefaultBodyLimit::max(settings.body_limit))
132        .layer(RequestBodyLimitLayer::new(settings.body_limit))
133        .layer(TimeoutLayer::with_status_code(
134            StatusCode::SERVICE_UNAVAILABLE,
135            settings.request_timeout,
136        ))
137        .layer(CatchPanicLayer::custom(|_panic| {
138            StatusCode::INTERNAL_SERVER_ERROR.into_response()
139        }))
140        .layer(trace)
141        .layer(middleware::from_fn(request_id_and_problems))
142}

144/// The first layer: assigns the request ID and normalises errors.
145async fn request_id_and_problems(mut req: Request, next: Next) -> Response {
146    let id = req
147        .headers()
148        .get(&REQUEST_ID)
149        .and_then(|v| v.to_str().ok())
150        .filter(|v| valid_request_id(v))
151        .map_or_else(new_request_id, str::to_owned);
152    req.extensions_mut().insert(RequestId(id.clone()));
153
154    let res = next.run(req).await;
155    let mut res = normalise(res, &id);
156    let headers = res.headers_mut();
157    if let Ok(v) = HeaderValue::from_str(&id) {
158        headers.insert(REQUEST_ID, v);
159    }
160    headers.insert(
161        header::X_CONTENT_TYPE_OPTIONS,
162        HeaderValue::from_static("nosniff"),
163    );
164    res
165}

167/// Replaces an error response with its problem document. A response that
168/// carries a [`Problem`] is rendered from it. Any other error response is
169/// replaced by the generic problem for its status, keeping only the headers
170/// in [`KEPT_ERROR_HEADERS`].
171fn normalise(res: Response, request_id: &str) -> Response {
172    if let Some(problem) = res.extensions().get::<Problem>() {
173        let mut out = problem.render(request_id);
174        copy_kept_headers(res.headers(), out.headers_mut());
175        return out;
176    }
177    let status = res.status();
178    if !(status.is_client_error() || status.is_server_error()) {
179        return res;
180    }
181    let mut out = Problem::for_status(status).render(request_id);
182    copy_kept_headers(res.headers(), out.headers_mut());
183    out
184}
185
186fn copy_kept_headers(from: &HeaderMap, to: &mut HeaderMap) {
187    for name in KEPT_ERROR_HEADERS {
188        for v in from.get_all(&name) {
189            to.append(name.clone(), v.clone());
190        }
191    }
192}

194/// A client-supplied request ID is used only if it is 1 to 64 characters of
195/// `A-Z a-z 0-9 . _ : -`, so it cannot inject into logs or headers.
196fn valid_request_id(v: &str) -> bool {
197    !v.is_empty()
198        && v.len() <= MAX_REQUEST_ID
199        && v.bytes()
200            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b':' | b'-'))
201}
202
203/// 128 random bits, hex encoded.
204fn new_request_id() -> String {
205    static FALLBACK: AtomicU64 = AtomicU64::new(0);
206    let mut bytes = [0u8; 16];
207    if crate::tls::fill_random(&mut bytes).is_err() {
208        // The OS random source failed. IDs only correlate log lines, so a
209        // process-local counter is an acceptable stand-in.
210        return format!("local-{}", FALLBACK.fetch_add(1, Ordering::Relaxed));
211    }
212    bytes.iter().map(|b| format!("{b:02x}")).collect()
213}

215async fn healthz() -> Response {
216    (
217        [(header::CACHE_CONTROL, "no-store")],
218        axum::Json(json!({ "status": "alive" })),
219    )
220        .into_response()
221}

223async fn readyz(State(state): State<AppState>) -> Response {
224    let r = state.health.readiness();
225    let status = r.body["status"].clone();
226    let indicators = r.body["indicators"].clone();
227    if r.ready {
228        let body = if state.detailed_health {
229            r.body
230        } else {
231            json!({ "status": status })
232        };
233        return ([(header::CACHE_CONTROL, "no-store")], axum::Json(body)).into_response();
234    }
235    let mut problem = Problem {
236        status: StatusCode::SERVICE_UNAVAILABLE,
237        slug: "not-ready",
238        title: "Not ready",
239        detail: "The server is not ready to serve traffic.".to_owned(),
240        extensions: serde_json::Map::new(),
241    }
242    .with_extension("readiness", status);
243    if state.detailed_health {
244        problem = problem.with_extension("indicators", indicators);
245    }
246    problem.into_response()
247}

271    fn app(kind: ListenerKind, health: Arc<Health>) -> Router {
272        let state = AppState {
273            health,
274            detailed_health: kind == ListenerKind::Metrics,
275        };
276        let extra = Router::new()
277            .route(
278                "/slow",
279                get(|| async {
280                    tokio::time::sleep(Duration::from_secs(5)).await;
281                    "late"
282                }),
283            )
284            .route(
285                "/panic",
286                get(|| async {
287                    if std::hint::black_box(true) {
288                        panic!("secret internal detail");
289                    }
290                    "unreachable"
291                }),
292            )
293            .route(
294                "/echo",
295                axum::routing::post(|body: axum::body::Bytes| async move { body }),
296            )
297            .route(
298                "/raw-error",
299                get(|| async { (StatusCode::BAD_GATEWAY, "upstream at 10.0.0.5 said no") }),
300            );
301        apply_stack(routes(state).merge(extra), &settings(kind))
302    }

310    async fn send(app: Router, req: Request<Body>) -> (StatusCode, HeaderMap, Value) {
311        let res = app.oneshot(req).await.unwrap();
312        let status = res.status();
313        let headers = res.headers().clone();
314        let bytes = axum::body::to_bytes(res.into_body(), 1 << 20)
315            .await
316            .unwrap();
317        let body = serde_json::from_slice(&bytes).unwrap_or(Value::Null);
318        (status, headers, body)
319    }

325    fn assert_problem(status: StatusCode, headers: &HeaderMap, body: &Value, slug: &str) {
326        assert_eq!(headers[header::CONTENT_TYPE], "application/problem+json");
327        assert_eq!(body["status"], status.as_u16());
328        assert_eq!(
329            body["type"],
330            format!("https://palisade.dev/problems/{slug}")
331        );
332        let id = headers[&REQUEST_ID].to_str().unwrap();
333        assert_eq!(body["request_id"], id);
334        assert_eq!(body["instance"], format!("urn:palisade:request:{id}"));
335        assert!(body["title"].is_string() && body["detail"].is_string());
336    }
