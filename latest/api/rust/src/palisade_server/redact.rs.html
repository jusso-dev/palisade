1//! Log redaction (#227): a second line of defence behind "never pass a
2//! secret to a log macro".
3//!
4//! [`RedactingMakeWriter`] wraps the log writer. Every complete line it is
5//! given goes through [`redact_line`] before it is written:
6//!
7//! - **Sensitive field names.** In a JSON line, any object member whose
8//!   name has a segment (split on `_`, `-` and `.`, compared without case)
9//!   in [`SENSITIVE_SEGMENTS`] has its value replaced by `"[REDACTED]"`,
10//!   whatever its type. This covers event fields and the fields of every
11//!   span on the line. A name whose last segment is `file` or `path` is
12//!   exempt: secrets are configured by path (`tls.key_file`,
13//!   `database.password_file`) and the path is what an operator needs to
14//!   fix a problem.
15//! - **Secret-looking values**, in every string of a JSON line and in any
16//!   line that is not JSON: admin API tokens (`pal_<env>_<id>_<secret>`,
17//!   #227) keep their non-secret `pal_<env>_<id>_` prefix and lose the
18//!   rest, an HTTP `Bearer` or `Basic` credential loses its value, and a PEM
19//!   `PRIVATE KEY` block loses its body.
20//!
21//! Redaction never fails open by dropping a line: a line it cannot parse as
22//! JSON is still scanned for secret-looking values and then written.
23
24use std::io::{self, Write};

29/// What a redacted value becomes.
30pub const REDACTED: &str = crate::secret::REDACTED;
31
32/// Field-name segments that mark a value as secret.
33pub const SENSITIVE_SEGMENTS: &[&str] = &[
34    "password",
35    "passwd",
36    "passphrase",
37    "secret",
38    "token",
39    "key",
40    "authorization",
41    "cookie",
42    "code",
43    "credential",
44    "credentials",
45];
46
47/// Last segments that mark a name as a path to a secret, not the secret.
48const PATH_SEGMENTS: &[&str] = &["file", "path"];
49
50/// Whether a field named `name` holds a secret.
51#[must_use]
52pub fn is_sensitive_name(name: &str) -> bool {
53    let segments: Vec<String> = name
54        .split(['_', '-', '.'])
55        .filter(|s| !s.is_empty())
56        .map(str::to_ascii_lowercase)
57        .collect();
58    if segments
59        .last()
60        .is_some_and(|last| PATH_SEGMENTS.contains(&last.as_str()))
61    {
62        return false;
63    }
64    segments
65        .iter()
66        .any(|s| SENSITIVE_SEGMENTS.contains(&s.as_str()))
67}

69/// Redacts one log line (without its line ending).
70#[must_use]
71pub fn redact_line(line: &str) -> String {
72    match serde_json::from_str::<Value>(line) {
73        Ok(mut value @ (Value::Object(_) | Value::Array(_))) => {
74            redact_value(&mut value);
75            serde_json::to_string(&value).unwrap_or_else(|_| redact_text(line))
76        }
77        _ => redact_text(line),
78    }
79}
80
81fn redact_value(value: &mut Value) {
82    match value {
83        Value::Object(map) => {
84            for (name, v) in map.iter_mut() {
85                if is_sensitive_name(name) {
86                    *v = Value::String(REDACTED.to_owned());
87                } else {
88                    redact_value(v);
89                }
90            }
91        }
92        Value::Array(items) => items.iter_mut().for_each(redact_value),
93        Value::String(s) => {
94            let redacted = redact_text(s);
95            if redacted != *s {
96                *s = redacted;
97            }
98        }
99        Value::Null | Value::Bool(_) | Value::Number(_) => {}
100    }
101}

103/// Masks secret-looking values in free text.
104#[must_use]
105pub fn redact_text(text: &str) -> String {
106    let text = redact_pem_private_keys(text);
107    let text = redact_auth_schemes(&text);
108    redact_api_tokens(&text)
109}
110
111/// Replaces the body of every PEM `PRIVATE KEY` block (`PRIVATE KEY`,
112/// `RSA PRIVATE KEY`, `EC PRIVATE KEY`, `ENCRYPTED PRIVATE KEY`): every
113/// character between the begin marker and the next `-----END `. A block
114/// with no end marker loses everything after its begin marker.
115fn redact_pem_private_keys(text: &str) -> String {
116    const MARKER: &str = "PRIVATE KEY-----";
117    let mut out = String::with_capacity(text.len());
118    let mut rest = text;
119    while let Some(at) = rest.find(MARKER) {
120        let header_end = at + MARKER.len();
121        // Only a begin marker: the text before it on the line has `BEGIN`.
122        let line_start = rest[..at].rfind('\n').map_or(0, |i| i + 1);
123        let is_begin = rest[line_start..at].contains("-----BEGIN ");
124        out.push_str(&rest[..header_end]);
125        rest = &rest[header_end..];
126        if !is_begin {
127            continue;
128        }
129        out.push_str(REDACTED);
130        match rest.find("-----END ") {
131            Some(end) => rest = &rest[end..],
132            None => rest = "",
133        }
134    }
135    out.push_str(rest);
136    out
137}

139/// Replaces the credential after `Bearer ` or `Basic ` (any case).
140fn redact_auth_schemes(text: &str) -> String {
141    let mut out = String::with_capacity(text.len());
142    let mut i = 0;
143    let bytes = text.as_bytes();
144    while i < bytes.len() {
145        let matched = ["bearer ", "basic "].iter().find(|scheme| {
146            text.get(i..i + scheme.len())
147                .is_some_and(|w| w.eq_ignore_ascii_case(scheme))
148                && (i == 0 || !bytes[i - 1].is_ascii_alphanumeric())
149        });
150        if let Some(scheme) = matched {
151            let value_start = i + scheme.len();
152            let value_end = text[value_start..]
153                .find(|c: char| c.is_whitespace() || matches!(c, '"' | ',' | ';' | '\''))
154                .map_or(text.len(), |e| value_start + e);
155            if value_end > value_start {
156                out.push_str(&text[i..value_start]);
157                out.push_str(REDACTED);
158                i = value_end;
159                continue;
160            }
161        }
162        let ch = text[i..].chars().next().unwrap_or(' ');
163        out.push(ch);
164        i += ch.len_utf8();
165    }
166    out
167}

169/// Masks the secret part of admin API tokens: `pal_<env>_<id>_<secret>`,
170/// with `env` lowercase letters and digits, `id` 8 letters and digits and
171/// the secret 20 or more letters and digits. The prefix up to the last
172/// `_` stays, so a token can still be identified.
173fn redact_api_tokens(text: &str) -> String {
174    let mut out = String::with_capacity(text.len());
175    let mut i = 0;
176    while i < text.len() {
177        if let Some(prefix_len) = token_at(text, i) {
178            out.push_str(&text[i..i + prefix_len]);
179            out.push_str(REDACTED);
180            let secret_len = text[i + prefix_len..]
181                .find(|c: char| !c.is_ascii_alphanumeric())
182                .unwrap_or(text.len() - i - prefix_len);
183            i += prefix_len + secret_len;
184            continue;
185        }
186        let ch = text[i..].chars().next().unwrap_or(' ');
187        out.push(ch);
188        i += ch.len_utf8();
189    }
190    out
191}

193/// If a token starts at byte `i`, the length of its non-secret prefix.
194fn token_at(text: &str, i: usize) -> Option<usize> {
195    let bytes = text.as_bytes();
196    if i > 0 && bytes.get(i - 1).is_some_and(u8::is_ascii_alphanumeric) {
197        return None;
198    }
199    let rest = text.get(i..)?.strip_prefix("pal_")?;
200    let (env, rest) = rest.split_once('_')?;
201    if env.is_empty()
202        || !env
203            .bytes()
204            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit())
205    {
206        return None;
207    }
208    let id = rest.get(..8)?;
209    if !id.bytes().all(|b| b.is_ascii_alphanumeric()) || rest.as_bytes().get(8) != Some(&b'_') {
210        return None;
211    }
212    let secret = &rest[9..];
213    let secret_len = secret
214        .find(|c: char| !c.is_ascii_alphanumeric())
215        .unwrap_or(secret.len());
216    (secret_len >= 20).then_some("pal_".len() + env.len() + 1 + 9)
217}

219/// A [`MakeWriter`] whose writers redact each line before passing it to
220/// the inner writer.
221#[derive(Debug, Clone)]
222pub struct RedactingMakeWriter<M>(pub M);
223
224impl<'a, M: MakeWriter<'a>> MakeWriter<'a> for RedactingMakeWriter<M> {
225    type Writer = RedactingWriter<M::Writer>;
226
227    fn make_writer(&'a self) -> Self::Writer {
228        RedactingWriter {
229            inner: self.0.make_writer(),
230            pending: Vec::new(),
231        }
232    }
233}
234
235/// Buffers until a line is complete, then writes it redacted. A partial
236/// line left at drop is redacted and written then.
237#[derive(Debug)]
238pub struct RedactingWriter<W: Write> {
239    inner: W,
240    pending: Vec<u8>,
241}
242
243impl<W: Write> RedactingWriter<W> {
244    fn emit(&mut self, line: &[u8], newline: bool) -> io::Result<()> {
245        let text = String::from_utf8_lossy(line);
246        let mut out = redact_line(&text);
247        if newline {
248            out.push('\n');
249        }
250        self.inner.write_all(out.as_bytes())
251    }
252}
253
254impl<W: Write> Write for RedactingWriter<W> {
255    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
256        self.pending.extend_from_slice(buf);
257        while let Some(pos) = self.pending.iter().position(|&b| b == b'\n') {
258            let line: Vec<u8> = self.pending.drain(..=pos).collect();
259            let body = line.strip_suffix(b"\n").unwrap_or(&line);
260            let body = body.strip_suffix(b"\r").unwrap_or(body);
261            self.emit(body, true)?;
262        }
263        Ok(buf.len())
264    }
265
266    fn flush(&mut self) -> io::Result<()> {
267        self.inner.flush()
268    }
269}
270
271impl<W: Write> Drop for RedactingWriter<W> {
272    fn drop(&mut self) {
273        if !self.pending.is_empty() {
274            let rest = std::mem::take(&mut self.pending);
275            // Nowhere to report a failed log write.
276            let _ = self.emit(&rest, false);
277        }
278        let _ = self.inner.flush();
279    }
280}

321    #[test]
322    fn json_fields_are_redacted_at_any_depth() {
323        let line = json!({
324            "timestamp": "t",
325            "level": "INFO",
326            "fields": {"message": "login", "password": "hunter2", "user": "admin", "attempts": 3},
327            "span": {"name": "request", "authorization": "Bearer abc"},
328            "spans": [{"name": "outer", "api_key": 12345, "nested": {"token": {"a": 1}}}],
329            "key_file": "/etc/palisade/server.key"
330        })
331        .to_string();
332        let out: Value = serde_json::from_str(&redact_line(&line)).unwrap();
333        assert_eq!(out["fields"]["password"], REDACTED);
334        assert_eq!(out["fields"]["user"], "admin");
335        assert_eq!(out["fields"]["attempts"], 3);
336        assert_eq!(out["span"]["authorization"], REDACTED);
337        assert_eq!(out["spans"][0]["api_key"], REDACTED);
338        assert_eq!(out["spans"][0]["nested"]["token"], REDACTED);
339        assert_eq!(out["key_file"], "/etc/palisade/server.key");
340        assert!(!redact_line(&line).contains("hunter2"));
341    }

343    #[test]
344    fn values_that_look_secret_are_masked_anywhere() {
345        let msg = format!("rejected {TOKEN}, retry with {TOKEN}.");
346        let out = redact_text(&msg);
347        assert_eq!(
348            out,
349            "rejected pal_prod_AbCd1234_[REDACTED], retry with pal_prod_AbCd1234_[REDACTED]."
350        );
351        let line = json!({"fields": {"message": msg}}).to_string();
352        assert!(!redact_line(&line).contains("Q9w8E7r6"));
353        // Not tokens: too short a secret, a word before `pal_`, bad id.
354        for text in [
355            "pal_prod_AbCd1234_short",
356            "xpal_prod_AbCd1234_Q9w8E7r6T5y4U3i2O1p0",
357            "pal_PROD_AbCd1234_Q9w8E7r6T5y4U3i2O1p0",
358            "pal_prod_AbCd123_Q9w8E7r6T5y4U3i2O1p0AA",
359            "pal_prod_AbCd12-4_Q9w8E7r6T5y4U3i2O1p0",
360            "pal__AbCd1234_Q9w8E7r6T5y4U3i2O1p0",
361            "pal_",
362        ] {
363            assert_eq!(redact_text(text), text);
364        }
365        assert_eq!(
366            redact_text("Authorization: Bearer eyJhbGciOi.x.y; next"),
367            "Authorization: Bearer [REDACTED]; next"
368        );
369        assert_eq!(redact_text("basic dXNlcjpwYXNz"), "basic [REDACTED]");
370        assert_eq!(redact_text("a bearer "), "a bearer ");
371        assert_eq!(redact_text("debearer x"), "debearer x");
372        assert_eq!(redact_text("naïve bearer ü"), "naïve bearer [REDACTED]");
373    }
