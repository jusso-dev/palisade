1//! Observed facts about one object, as criteria see them (#31).
2//!
3//! The agent and the file-identity parser (#3, #4) compute facts. This
4//! module only defines their shape, so evaluation stays pure: nothing here
5//! reads a file or calls the OS. A fact that was not computed is absent,
6//! and every criterion on an absent fact evaluates to
7//! [`Tri::Unknown`](crate::ops::Tri::Unknown), never to a match.
8
9use std::collections::{BTreeMap, BTreeSet};
10
11use palisade_core::path::{NormalizedPath, VolumeMap, normalize_with};
12use schemars::JsonSchema;
13use serde::{Deserialize, Serialize};
14
15use crate::model::{Field, FieldClass, LocationClass, OriginClass};

17/// Every fact about one object that the criteria of this version can test.
18#[derive(Clone, Debug, Default, PartialEq, Eq)]
19pub struct ObjectFacts {
20    /// Hash, version-resource and version facts, by field. Signer and path
21    /// fields are not stored here; see [`Self::signers`] and [`Self::path`].
22    values: BTreeMap<Field, String>,
23    /// The object's path, if known.
24    pub path: Option<PathFact>,
25    /// Every signature on the object, embedded or from a catalog, if the
26    /// signatures were examined. `Some(vec![])` means the object is known
27    /// to be unsigned; `None` means nobody looked.
28    pub signers: Option<Vec<SignerFact>>,
29    /// The location classes the agent computed from the path, the volume
30    /// and the writable-path scanner (#4, #39). `None` (or empty) means the
31    /// engine classifies the path itself
32    /// ([`crate::unknown::location::classify`]).
33    pub location_class: Option<BTreeSet<LocationClass>>,
34    /// Where the object came from (#4, #39). Two different states:
35    ///
36    /// - `None`: the agent's origin collection did not run or did not
37    ///   report. That is a missing fact: the origin is `unknown`, but stage 6
38    ///   cannot rule out any other origin, so every other origin's entry
39    ///   applies too ([`crate::unknown::MissingFact::Origin`]). Under the
40    ///   default `preexisting_policy: baseline_only`, that blocks the object.
41    /// - `Some(set)`: collection ran, and `set` is every origin it found. An
42    ///   empty set means it found no evidence, and is exactly `{unknown}`.
43    ///
44    /// Nothing turns one state into the other.
45    pub origin: Option<BTreeSet<OriginClass>>,
46}
47
48impl ObjectFacts {
49    /// No facts at all.
50    #[must_use]
51    pub fn new() -> Self {
52        Self::default()
53    }
54
55    /// Records the value of a hash, version-resource or version field.
56    /// Hash values are hex digits in either case.
57    ///
58    /// Returns `false`, and records nothing, for a path or signer field:
59    /// those come from [`Self::path`] and [`Self::signers`]. Nor for a
60    /// context field, which comes from the
61    /// [`EvaluationContext`](crate::context::EvaluationContext).
62    pub fn set(&mut self, field: Field, value: impl Into<String>) -> bool {
63        if is_stored(field) {
64            self.values.insert(field, value.into());
65            true
66        } else {
67            false
68        }
69    }
70
71    /// The same as [`Self::set`], for building facts in one expression.
72    /// A path, signer or context field is ignored.
73    #[must_use]
74    pub fn with(mut self, field: Field, value: impl Into<String>) -> Self {
75        self.set(field, value);
76        self
77    }
78
79    /// The same facts, with the path set.
80    #[must_use]
81    pub fn with_path(mut self, path: PathFact) -> Self {
82        self.path = Some(path);
83        self
84    }
85
86    /// The same facts, with the signatures set. An empty list means the
87    /// object is known to be unsigned.
88    #[must_use]
89    pub fn with_signers(mut self, signers: Vec<SignerFact>) -> Self {
90        self.signers = Some(signers);
91        self
92    }
93
94    /// The same facts, with the location classes set.
95    #[must_use]
96    pub fn with_location_class(mut self, classes: impl IntoIterator<Item = LocationClass>) -> Self {
97        self.location_class = Some(classes.into_iter().collect());
98        self
99    }
100
101    /// The same facts, with the origins collection found. An empty list
102    /// means collection ran and found nothing: origin `unknown`.
103    #[must_use]
104    pub fn with_origin(mut self, origins: impl IntoIterator<Item = OriginClass>) -> Self {
105        self.origin = Some(origins.into_iter().collect());
106        self
107    }
108
109    /// The recorded value of a hash, version-resource or version field.
110    #[must_use]
111    pub fn get(&self, field: Field) -> Option<&str> {
112        self.values.get(&field).map(String::as_str)
113    }
114
115    /// The hash, version-resource and version fields that have a value.
116    pub(crate) fn fields(&self) -> impl Iterator<Item = Field> + '_ {
117        self.values.keys().copied()
118    }
119}

121/// Whether a field's fact is stored in [`ObjectFacts::set`]'s map.
122const fn is_stored(field: Field) -> bool {
123    !field.is_context()
124        && !field.is_signer()
125        && !field.is_path_location()
126        && !matches!(field.class(), FieldClass::Path | FieldClass::FileName)
127}

129/// The path of an object, normalised, with the facts that decide whether
130/// the path can be trusted as an identity.
131#[derive(Clone, Debug, PartialEq, Eq)]
132pub struct PathFact {
133    /// The final path, after symbolic links and junctions are resolved, with
134    /// 8.3 short names expanded by the agent (#4).
135    pub path: NormalizedPath,
136    /// The same file under another name that rules may be written against:
137    /// the UNC form of a path on a mapped network drive, or the drive form
138    /// of a UNC path. Path criteria match if either form matches.
139    pub alias: Option<NormalizedPath>,
140    /// The agent could not expand an 8.3 short name in the path. Path
141    /// criteria then evaluate to unknown, because the short name could be
142    /// a spelling that a rule does not recognise.
143    pub short_name_unresolved: bool,
144    /// The volume's case folding is not known to be the standard NTFS
145    /// upcase table ([`palisade_core::upcase`], #399): every network share,
146    /// and a local volume formatted with another table. Two names that the
147    /// standard table folds apart may be one name there (`ſetup.exe` and
148    /// `SETUP.EXE` on a server that folds with Unicode), so a path test
149    /// that misses by its folded key is not proof of a miss. A path test
150    /// is then `Unknown` whenever its result would make the rule less
151    /// strict: a miss of a `deny` or `audit`, and a match of a rule that
152    /// lets the object run, trusts it or excludes it. `false` when the
153    /// agent knows the table is the standard one, or did not say.
154    pub fold_unknown: bool,
155    /// Whether any non-administrative principal can write the file or any
156    /// directory above it, or change their permissions (#3, #4). `None`
157    /// means it was not determined. A path only identifies an object
158    /// when this is `Some(false)`.
159    pub user_writable: Option<bool>,
160    /// The kind of volume the file is on, as the agent reports it (#4,
161    /// #39). `None` means not known: stage 6 then cannot rule out
162    /// `removable`, `optical` or `network` for a drive path, nor rely on the
163    /// classes that need a fixed volume ([`crate::unknown::location`]).
164    pub volume: Option<crate::unknown::VolumeKind>,
165    /// The volume's drive type as the OS reports it, in `snake_case`
166    /// (`fixed`, `removable`, `network`, `cd_rom`, `ram_disk`), for
167    /// `path.drive_type` (#328). `None`, or `unknown`, means not known.
168    pub drive_type: Option<String>,
169    /// The bus of the volume's disk, in `snake_case` (`usb`, `sata`,
170    /// `nvme`, ...), for `path.bus` (#328). `None`, or `unknown`, means not
171    /// known.
172    pub bus: Option<String>,
173    /// Whether some non-administrator may change the permissions or the
174    /// owner of the file or a directory above it, for
175    /// `path.user_can_change_acl` (#328). `None` means not determined.
176    pub user_can_change_acl: Option<bool>,
177}
178
179impl PathFact {
180    /// A path fact for `raw`, normalised with `volumes`. The ACL facts are
181    /// unknown and the short name is taken as resolved.
182    #[must_use]
183    pub fn new(raw: &str, volumes: &VolumeMap) -> Self {
184        Self {
185            path: normalize_with(raw, volumes),
186            alias: None,
187            short_name_unresolved: false,
188            fold_unknown: false,
189            user_writable: None,
190            volume: None,
191            drive_type: None,
192            bus: None,
193            user_can_change_acl: None,
194        }
195    }
196
197    /// The same path fact, with the volume kind set.
198    #[must_use]
199    pub const fn with_volume(mut self, volume: crate::unknown::VolumeKind) -> Self {
200        self.volume = Some(volume);
201        self
202    }
203
204    /// The same fact, on a volume whose case folding is not known
205    /// ([`Self::fold_unknown`]).
206    #[must_use]
207    pub const fn with_fold_unknown(mut self) -> Self {
208        self.fold_unknown = true;
209        self
210    }
211
212    /// Whether path names can be compared by their folded keys: the short
213    /// name was expanded, the path is a drive or UNC path, and the
214    /// volume's case folding is the standard one. A path that is an
215    /// identity anchor (a draft allow, a cached decision key) needs this.
216    #[must_use]
217    pub fn keys_reliable(&self) -> bool {
218        !self.short_name_unresolved && !self.fold_unknown && self.path.is_resolved()
219    }
220
221    /// The same fact, with `user_writable` set.
222    #[must_use]
223    pub const fn with_user_writable(mut self, writable: bool) -> Self {
224        self.user_writable = Some(writable);
225        self
226    }
227
228    /// What a `path.*` location field (#328) reads from this fact.
229    /// [`Field::is_path_location`] fields only; any other is unknown.
230    #[must_use]
231    pub fn location(&self, field: Field) -> PathLocation<'_> {
232        use palisade_core::path::PathKind;
233        fn known(v: Option<&String>) -> Option<&str> {
234            v.map(String::as_str)
235                .filter(|t| !t.is_empty() && *t != "unknown")
236        }
237        let unc = core::iter::once(&self.path)
238            .chain(&self.alias)
239            .find(|p| p.kind() == PathKind::Unc);
240        // On a share: a UNC form, or a network volume. Not on one: a
241        // resolved path on a volume known to be local. Otherwise, such as
242        // a mapped drive with no UNC form recorded, it is not known.
243        let on_share = if unc.is_some() {
244            Some(true)
245        } else if !self.path.is_resolved() {
246            None
247        } else {
248            match self.volume {
249                Some(crate::unknown::VolumeKind::Network) => Some(true),
250                Some(_) => Some(false),
251                None => None,
252            }
253        };
254        match field {
255            Field::PathDriveType => PathLocation::Text(known(self.drive_type.as_ref())),
256            Field::PathBus => PathLocation::Text(known(self.bus.as_ref())),
257            Field::PathIsUnc => PathLocation::Bool(on_share),
258            Field::PathIsAds => PathLocation::Bool(Some(self.path.stream().is_some())),
259            Field::PathUserCanChangeAcl => PathLocation::Bool(self.user_can_change_acl),
260            Field::PathShare => match (unc, on_share) {
261                (Some(p), _) => PathLocation::Text(Some(p.root())),
262                (None, Some(false)) => PathLocation::None,
263                // A network volume whose UNC form is not recorded.
264                (None, _) => PathLocation::Text(None),
265            },
266            _ => PathLocation::Text(None),
267        }
268    }
269
270    /// Whether the path can be trusted as an identity (#31, #328): the
271    /// file is on a fixed local volume, and no non-administrative principal
272    /// can write it or a directory above it. An ACL only protects a path on
273    /// a local fixed disk: a removable disk can be prepared elsewhere with
274    /// any ACL, and a network share's server decides what its folders hold.
275    /// An unknown volume kind or ACL fact is not proof.
276    #[must_use]
277    pub const fn acl_proven(&self) -> bool {
278        matches!(self.user_writable, Some(false))
279            && matches!(self.volume, Some(crate::unknown::VolumeKind::Fixed))
280    }
281}

326/// Facts about one signature beyond its leaf's names and thumbprint
327/// (#31). As everywhere, `None` means not determined, and every criterion
328/// that needs the fact is then unknown for this signature.
329///
330/// Hashes and thumbprints are hex, in either case.
331#[derive(Clone, Debug, Default, PartialEq, Eq)]
332pub struct SignerDetails {
333    /// The organisation (`O`) of the leaf certificate's subject.
334    pub subject_organization: Option<String>,
335    /// The leaf certificate's whole subject, as RFC 4514 writes it.
336    pub subject_dn: Option<String>,
337    /// The SHA-1 thumbprint of the leaf certificate.
338    pub leaf_thumbprint_sha1: Option<String>,
339    /// The subject common name of the highest certificate below the root,
340    /// as the signature provides it (the PCA).
341    pub pca: Option<String>,
342    /// The SHA-256 thumbprint of the chain's root certificate.
343    pub root_thumbprint_sha256: Option<String>,
344    /// The SHA-1 thumbprint of the chain's root certificate.
345    pub root_thumbprint_sha1: Option<String>,
346    /// The TBS hash of the leaf certificate, under the digest of its own
347    /// signature algorithm.
348    pub leaf_tbs_hash: Option<String>,
349    /// The TBS hashes of every certificate between the leaf and the root.
350    /// `Some(vec![])` means the leaf was issued by the root.
351    pub intermediate_tbs_hashes: Option<Vec<String>>,
352    /// The TBS hash of the root certificate.
353    pub root_tbs_hash: Option<String>,
354    /// The SHA-256 of the leaf certificate's DER `SubjectPublicKeyInfo`
355    /// (#318): its public key, the same in every certificate issued for it.
356    pub leaf_spki_sha256: Option<String>,
357    /// The key hashes of every certificate between the leaf and the root.
358    /// `Some(vec![])` means the leaf was issued by the root.
359    pub intermediate_spki_sha256: Option<Vec<String>>,
360    /// The key hash of the root certificate.
361    pub root_spki_sha256: Option<String>,
362    /// The leaf certificate's extended key usages, as dotted OIDs.
363    /// `Some(vec![])` when the extension is absent.
364    pub eku: Option<Vec<String>>,
365    /// When a trusted timestamp countersignature shows the signature was
366    /// made, in seconds since the Unix epoch (UTC). `None` when there is
367    /// no trusted timestamp: the signer's own claimed signing time is never
368    /// used, because whoever holds the key can write any time there.
369    pub timestamp: Option<i64>,
370    /// Whether the signature carries an RFC 3161 timestamp that verifies.
371    pub countersigned: Option<bool>,
372    /// The public key of every certificate in the chain, leaf first.
373    pub keys: Option<Vec<KeyFact>>,
374    /// Every digest algorithm the signature uses: for the file digest, the
375    /// signer information and the signature of each chain certificate.
376    pub algorithms: Option<Vec<DigestAlgorithm>>,
377}

379/// The public key of one chain certificate.
380#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
381pub struct KeyFact {
382    /// The algorithm.
383    pub kind: KeyKind,
384    /// The key size in bits: the RSA modulus, or the curve's size.
385    pub bits: u32,
386}
387
388impl KeyFact {
389    /// The key's strength in RSA-equivalent bits (NIST SP 800-57 part 1,
390    /// table 2), so one `min_key_bits` covers RSA and elliptic-curve keys:
391    /// an RSA key counts its modulus; a curve of 224, 256, 384 or 521 bits
392    /// counts as 2048, 3072, 7680 or 15360; a smaller curve as 1024. `None`
393    /// for another algorithm, which no bound can be checked against.
394    #[must_use]
395    pub const fn rsa_equivalent_bits(self) -> Option<u32> {
396        match self.kind {
397            KeyKind::Rsa => Some(self.bits),
398            KeyKind::Ec => Some(match self.bits {
399                512.. => 15_360,
400                384.. => 7680,
401                256.. => 3072,
402                224.. => 2048,
403                _ => 1024,
404            }),
405            KeyKind::Other => None,
406        }
407    }
408}

453/// The result of verifying one signature (#31, #48). An unsigned object
454/// has no [`SignerFact`] at all.
455#[derive(
456    Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
457)]
458#[serde(rename_all = "snake_case")]
459pub enum SignerStatus {
460    /// The signature and chain verify, and the certificate is in date.
461    Valid,
462    /// The certificate has expired, but a trusted timestamp shows that the
463    /// signature was made while it was valid.
464    ValidExpiredTimestamped,
465    /// The signature verifies through a signed catalog.
466    CatalogSigned,
467    /// The certificate has expired, with no trusted timestamp.
468    Expired,
469    /// The certificate has been revoked.
470    Revoked,
471    /// The chain ends at a root that is not trusted.
472    UntrustedRoot,
473    /// The file's content does not match the signed digest.
474    DigestMismatch,
475}

477impl SignerStatus {
478    /// Every status.
479    pub const ALL: [Self; 7] = [
480        Self::Valid,
481        Self::ValidExpiredTimestamped,
482        Self::CatalogSigned,
483        Self::Expired,
484        Self::Revoked,
485        Self::UntrustedRoot,
486        Self::DigestMismatch,
487    ];
488
489    /// Whether the signer may vouch for an object's identity: the default
490    /// accepted statuses of #31.
491    #[must_use]
492    pub const fn is_trusted(self) -> bool {
493        matches!(
494            self,
495            Self::Valid | Self::ValidExpiredTimestamped | Self::CatalogSigned
496        )
497    }

509/// A path variable that a criterion value may start with, such as
510/// `%PROGRAMFILES%` (#31).
511#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
512pub enum PathVariable {
513    /// `%WINDIR%`: the Windows directory.
514    Windir,
515    /// `%SYSTEM32%`: the native and the 32-bit system directories.
516    System32,
517    /// `%PROGRAMFILES%`: both program-files directories.
518    ProgramFiles,
519    /// `%PROGRAMDATA%`.
520    ProgramData,
521    /// `%OSDRIVE%`: the drive Windows is installed on.
522    OsDrive,
523    /// `%USERPROFILE%`, for the user the process runs as.
524    UserProfile,
525    /// `%APPDATA%`, for the user the process runs as.
526    AppData,
527    /// `%LOCALAPPDATA%`, for the user the process runs as.
528    LocalAppData,
529    /// `%TEMP%` or `%TMP%`, for the user the process runs as.
530    Temp,
531    /// `%PUBLIC%`: the shared user profile.
532    Public,
533    /// `%REMOVABLE%`: the root of every volume on removable media, such
534    /// as an optical disc or a memory card (#31, #328). The agent sets one
535    /// value per such volume that is mounted.
536    Removable,
537    /// `%HOT%`: the root of every volume on a hot-pluggable disk, such as
538    /// a USB or eSATA disk (#31, #328). The agent sets one value per such
539    /// volume that is mounted.
540    Hot,
541}
542
543impl PathVariable {
544    /// Every variable.
545    pub const ALL: [Self; 12] = [
546        Self::Windir,
547        Self::System32,
548        Self::ProgramFiles,
549        Self::ProgramData,
550        Self::OsDrive,
551        Self::UserProfile,
552        Self::AppData,
553        Self::LocalAppData,
554        Self::Temp,
555        Self::Public,
556        Self::Removable,
557        Self::Hot,
558    ];
559
560    /// The variable with this name, without the `%` signs, in any case.
561    /// `TMP` is another name for `TEMP`.
562    #[must_use]
563    pub fn from_name(name: &str) -> Option<Self> {
564        let upper = name.to_ascii_uppercase();
565        Some(match upper.as_str() {
566            "WINDIR" => Self::Windir,
567            "SYSTEM32" => Self::System32,
568            "PROGRAMFILES" => Self::ProgramFiles,
569            "PROGRAMDATA" => Self::ProgramData,
570            "OSDRIVE" => Self::OsDrive,
571            "USERPROFILE" => Self::UserProfile,
572            "APPDATA" => Self::AppData,
573            "LOCALAPPDATA" => Self::LocalAppData,
574            "TEMP" | "TMP" => Self::Temp,
575            "PUBLIC" => Self::Public,
576            "REMOVABLE" => Self::Removable,
577            "HOT" => Self::Hot,
578            _ => return None,
579        })
580    }
581}

583/// What each path variable expands to, for one evaluation. Per-user
584/// variables hold the values for the user the process runs as.
585#[derive(Clone, Debug, Default, PartialEq, Eq)]
586pub struct PathVariables {
587    values: BTreeMap<PathVariable, Vec<NormalizedPath>>,
588}
589
590impl PathVariables {
591    /// No variables: every criterion that uses one evaluates to unknown.
592    #[must_use]
593    pub fn new() -> Self {
594        Self::default()
595    }
596
597    /// Sets what `variable` expands to. A variable may have several values
598    /// (`%PROGRAMFILES%` names both program-files directories), and a
599    /// criterion matches if any of them matches. Values that are not drive
600    /// or UNC paths, or that name a stream, are ignored, and an empty list
601    /// unsets the variable.
602    pub fn set(&mut self, variable: PathVariable, values: &[&str]) {
603        let values: Vec<NormalizedPath> = values
604            .iter()
605            .map(|v| palisade_core::path::normalize(v))
606            .filter(|p| p.is_resolved() && p.stream().is_none())
607            .collect();
608        if values.is_empty() {
609            self.values.remove(&variable);
610        } else {
611            self.values.insert(variable, values);
612        }
613    }
614
615    /// The same, for building variables in one expression.
616    #[must_use]
617    pub fn with(mut self, variable: PathVariable, values: &[&str]) -> Self {
618        self.set(variable, values);
619        self
620    }
