1//! Observed facts about one object, as criteria see them (#31).
2//!
3//! The agent and the file-identity parser (#3, #4) compute facts. This
4//! module only defines their shape, so evaluation stays pure: nothing here
5//! reads a file or calls the OS. A fact that was not computed is absent,
6//! and every criterion on an absent fact evaluates to
7//! [`Tri::Unknown`](crate::ops::Tri::Unknown), never to a match.
8
9use std::collections::{BTreeMap, BTreeSet};
10
11use palisade_core::path::{NormalizedPath, VolumeMap, normalize_with};
12
13use crate::model::{Field, FieldClass, LocationClass, OriginClass};
14
15/// Every fact about one object that the criteria of this version can test.
16#[derive(Clone, Debug, Default, PartialEq, Eq)]
17pub struct ObjectFacts {
18    /// Hash, version-resource and version facts, by field. Signer and path
19    /// fields are not stored here; see [`Self::signers`] and [`Self::path`].
20    values: BTreeMap<Field, String>,
21    /// The object's path, if known.
22    pub path: Option<PathFact>,
23    /// Every signature on the object, embedded or from a catalog, if the
24    /// signatures were examined. `Some(vec![])` means the object is known
25    /// to be unsigned; `None` means nobody looked.
26    pub signers: Option<Vec<SignerFact>>,
27    /// The location classes the agent computed from the path, the volume
28    /// and the writable-path scanner (#4, #39). `None` (or empty) means the
29    /// engine classifies the path itself
30    /// ([`crate::unknown::location::classify`]).
31    pub location_class: Option<BTreeSet<LocationClass>>,
32    /// Where the object came from (#4, #39). Two different states:
33    ///
34    /// - `None`: the agent's origin collection did not run or did not
35    ///   report. That is a missing fact: the origin is `unknown`, but stage 6
36    ///   cannot rule out any other origin, so every other origin's entry
37    ///   applies too ([`crate::unknown::MissingFact::Origin`]). Under the
38    ///   default `preexisting_policy: baseline_only`, that blocks the object.
39    /// - `Some(set)`: collection ran, and `set` is every origin it found. An
40    ///   empty set means it found no evidence, and is exactly `{unknown}`.
41    ///
42    /// Nothing turns one state into the other.
43    pub origin: Option<BTreeSet<OriginClass>>,
44}
45
46impl ObjectFacts {
47    /// No facts at all.
48    #[must_use]
49    pub fn new() -> Self {
50        Self::default()
51    }
52
53    /// Records the value of a hash, version-resource or version field.
54    /// Hash values are hex digits in either case.
55    ///
56    /// Returns `false`, and records nothing, for a path or signer field:
57    /// those come from [`Self::path`] and [`Self::signers`]. Nor for a
58    /// context field, which comes from the
59    /// [`EvaluationContext`](crate::context::EvaluationContext).
60    pub fn set(&mut self, field: Field, value: impl Into<String>) -> bool {
61        if is_stored(field) {
62            self.values.insert(field, value.into());
63            true
64        } else {
65            false
66        }
67    }
68
69    /// The same as [`Self::set`], for building facts in one expression.
70    /// A path, signer or context field is ignored.
71    #[must_use]
72    pub fn with(mut self, field: Field, value: impl Into<String>) -> Self {
73        self.set(field, value);
74        self
75    }
76
77    /// The same facts, with the path set.
78    #[must_use]
79    pub fn with_path(mut self, path: PathFact) -> Self {
80        self.path = Some(path);
81        self
82    }
83
84    /// The same facts, with the signatures set. An empty list means the
85    /// object is known to be unsigned.
86    #[must_use]
87    pub fn with_signers(mut self, signers: Vec<SignerFact>) -> Self {
88        self.signers = Some(signers);
89        self
90    }
91
92    /// The same facts, with the location classes set.
93    #[must_use]
94    pub fn with_location_class(mut self, classes: impl IntoIterator<Item = LocationClass>) -> Self {
95        self.location_class = Some(classes.into_iter().collect());
96        self
97    }
98
99    /// The same facts, with the origins collection found. An empty list
100    /// means collection ran and found nothing: origin `unknown`.
101    #[must_use]
102    pub fn with_origin(mut self, origins: impl IntoIterator<Item = OriginClass>) -> Self {
103        self.origin = Some(origins.into_iter().collect());
104        self
105    }
106
107    /// The recorded value of a hash, version-resource or version field.
108    #[must_use]
109    pub fn get(&self, field: Field) -> Option<&str> {
110        self.values.get(&field).map(String::as_str)
111    }
112}
113
114/// Whether a field's fact is stored in [`ObjectFacts::set`]'s map.
115const fn is_stored(field: Field) -> bool {
116    !field.is_context()
117        && !matches!(
118            field,
119            Field::SignerPublisher | Field::SignerIssuer | Field::SignerLeafThumbprint
120        )
121        && !matches!(field.class(), FieldClass::Path | FieldClass::FileName)
122}

124/// The path of an object, normalised, with the facts that decide whether
125/// the path can be trusted as an identity.
126#[derive(Clone, Debug, PartialEq, Eq)]
127pub struct PathFact {
128    /// The final path, after symbolic links and junctions are resolved, with
129    /// 8.3 short names expanded by the agent (#4).
130    pub path: NormalizedPath,
131    /// The same file under another name that rules may be written against:
132    /// the UNC form of a path on a mapped network drive, or the drive form
133    /// of a UNC path. Path criteria match if either form matches.
134    pub alias: Option<NormalizedPath>,
135    /// The agent could not expand an 8.3 short name in the path. Path
136    /// criteria then evaluate to unknown, because the short name could be
137    /// a spelling that a rule does not recognise.
138    pub short_name_unresolved: bool,
139    /// Whether any non-administrative principal can write the file or any
140    /// directory above it, or change their permissions (#3, #4). `None`
141    /// means it was not determined. A path only identifies an object
142    /// when this is `Some(false)`.
143    pub user_writable: Option<bool>,
144    /// The kind of volume the file is on, as the agent reports it (#4,
145    /// #39). `None` means not known: stage 6 then cannot rule out
146    /// `removable`, `optical` or `network` for a drive path, nor rely on the
147    /// classes that need a fixed volume ([`crate::unknown::location`]).
148    pub volume: Option<crate::unknown::VolumeKind>,
149}
150
151impl PathFact {
152    /// A path fact for `raw`, normalised with `volumes`. The ACL facts are
153    /// unknown and the short name is taken as resolved.
154    #[must_use]
155    pub fn new(raw: &str, volumes: &VolumeMap) -> Self {
156        Self {
157            path: normalize_with(raw, volumes),
158            alias: None,
159            short_name_unresolved: false,
160            user_writable: None,
161            volume: None,
162        }
163    }
164
165    /// The same path fact, with the volume kind set.
166    #[must_use]
167    pub const fn with_volume(mut self, volume: crate::unknown::VolumeKind) -> Self {
168        self.volume = Some(volume);
169        self
170    }
171
172    /// The same fact, with `user_writable` set.
173    #[must_use]
174    pub const fn with_user_writable(mut self, writable: bool) -> Self {
175        self.user_writable = Some(writable);
176        self
177    }
178}

180/// One signature on an object, with the facts about its signer.
181#[derive(Clone, Debug, PartialEq, Eq)]
182pub struct SignerFact {
183    /// The result of verifying the signature and its chain.
184    pub status: SignerStatus,
185    /// Where the signature came from.
186    pub source: SignatureSource,
187    /// The common name of the leaf certificate's subject.
188    pub publisher: Option<String>,
189    /// The common name of the leaf certificate's issuer.
190    pub issuer: Option<String>,
191    /// The SHA-256 thumbprint of the leaf certificate, as hex.
192    pub leaf_thumbprint_sha256: Option<String>,
193}
194
195/// The result of verifying one signature (#31, #48). An unsigned object
196/// has no [`SignerFact`] at all.
197#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
198pub enum SignerStatus {
199    /// The signature and chain verify, and the certificate is in date.
200    Valid,
201    /// The certificate has expired, but a trusted timestamp shows that the
202    /// signature was made while it was valid.
203    ValidExpiredTimestamped,
204    /// The signature verifies through a signed catalog.
205    CatalogSigned,
206    /// The certificate has expired, with no trusted timestamp.
207    Expired,
208    /// The certificate has been revoked.
209    Revoked,
210    /// The chain ends at a root that is not trusted.
211    UntrustedRoot,
212    /// The file's content does not match the signed digest.
213    DigestMismatch,
214}
215
216impl SignerStatus {
217    /// Whether the signer may vouch for an object's identity: the default
218    /// accepted statuses of #31.
219    #[must_use]
220    pub const fn is_trusted(self) -> bool {
221        matches!(
222            self,
223            Self::Valid | Self::ValidExpiredTimestamped | Self::CatalogSigned
224        )
225    }
226}

228/// Where a signature came from.
229#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
230pub enum SignatureSource {
231    /// Embedded in the file.
232    Embedded,
233    /// From a signed catalog that lists the file's hash.
234    Catalog,
235}

237/// A path variable that a criterion value may start with, such as
238/// `%PROGRAMFILES%` (#31).
239#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
240pub enum PathVariable {
241    /// `%WINDIR%`: the Windows directory.
242    Windir,
243    /// `%SYSTEM32%`: the native and the 32-bit system directories.
244    System32,
245    /// `%PROGRAMFILES%`: both program-files directories.
246    ProgramFiles,
247    /// `%PROGRAMDATA%`.
248    ProgramData,
249    /// `%OSDRIVE%`: the drive Windows is installed on.
250    OsDrive,
251    /// `%USERPROFILE%`, for the user the process runs as.
252    UserProfile,
253    /// `%APPDATA%`, for the user the process runs as.
254    AppData,
255    /// `%LOCALAPPDATA%`, for the user the process runs as.
256    LocalAppData,
257    /// `%TEMP%` or `%TMP%`, for the user the process runs as.
258    Temp,
259    /// `%PUBLIC%`: the shared user profile.
260    Public,
261}
262
263impl PathVariable {
264    /// Every variable.
265    pub const ALL: [Self; 10] = [
266        Self::Windir,
267        Self::System32,
268        Self::ProgramFiles,
269        Self::ProgramData,
270        Self::OsDrive,
271        Self::UserProfile,
272        Self::AppData,
273        Self::LocalAppData,
274        Self::Temp,
275        Self::Public,
276    ];
277
278    /// The variable with this name, without the `%` signs, in any case.
279    /// `TMP` is another name for `TEMP`.
280    #[must_use]
281    pub fn from_name(name: &str) -> Option<Self> {
282        let upper = name.to_ascii_uppercase();
283        Some(match upper.as_str() {
284            "WINDIR" => Self::Windir,
285            "SYSTEM32" => Self::System32,
286            "PROGRAMFILES" => Self::ProgramFiles,
287            "PROGRAMDATA" => Self::ProgramData,
288            "OSDRIVE" => Self::OsDrive,
289            "USERPROFILE" => Self::UserProfile,
290            "APPDATA" => Self::AppData,
291            "LOCALAPPDATA" => Self::LocalAppData,
292            "TEMP" | "TMP" => Self::Temp,
293            "PUBLIC" => Self::Public,
294            _ => return None,
295        })
296    }
297}

299/// What each path variable expands to, for one evaluation. Per-user
300/// variables hold the values for the user the process runs as.
301#[derive(Clone, Debug, Default, PartialEq, Eq)]
302pub struct PathVariables {
303    values: BTreeMap<PathVariable, Vec<NormalizedPath>>,
304}
305
306impl PathVariables {
307    /// No variables: every criterion that uses one evaluates to unknown.
308    #[must_use]
309    pub fn new() -> Self {
310        Self::default()
311    }
312
313    /// Sets what `variable` expands to. A variable may have several values
314    /// (`%PROGRAMFILES%` names both program-files directories), and a
315    /// criterion matches if any of them matches. Values that are not drive
316    /// or UNC paths, or that name a stream, are ignored, and an empty list
317    /// unsets the variable.
318    pub fn set(&mut self, variable: PathVariable, values: &[&str]) {
319        let values: Vec<NormalizedPath> = values
320            .iter()
321            .map(|v| palisade_core::path::normalize(v))
322            .filter(|p| p.is_resolved() && p.stream().is_none())
323            .collect();
324        if values.is_empty() {
325            self.values.remove(&variable);
326        } else {
327            self.values.insert(variable, values);
328        }
329    }
330
331    /// The same, for building variables in one expression.
332    #[must_use]
333    pub fn with(mut self, variable: PathVariable, values: &[&str]) -> Self {
334        self.set(variable, values);
335        self
336    }
337
338    /// What `variable` expands to; empty if it is not set.
339    #[must_use]
340    pub fn get(&self, variable: PathVariable) -> &[NormalizedPath] {
341        self.values.get(&variable).map_or(&[], Vec::as_slice)
342    }
