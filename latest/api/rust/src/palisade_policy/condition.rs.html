1//! Evaluating a whole match tree (#33).
2//!
3//! A [`Match`] is compiled once, when the policy is loaded, into a
4//! [`CompiledMatch`]: an arena of nodes with every criterion compiled
5//! ([`CompiledCriterion`]). Compiling checks the limits that validation
6//! checks ([`crate::model::limits`]), so a compiled tree is always within
7//! them, and evaluation needs no recursion limit of its own.
8//!
9//! # Three-valued logic
10//!
11//! A tree evaluates to a [`Tri`] with Kleene logic:
12//!
13//! - `all` is `NoMatch` if any child is `NoMatch`, else `Unknown` if any
14//!   child is `Unknown`, else `Match`.
15//! - `any` is `Match` if any child is `Match`, else `Unknown` if any child
16//!   is `Unknown`, else `NoMatch`.
17//! - `not` swaps `Match` and `NoMatch`, and keeps `Unknown`.
18//!
19//! `all` stops at the first `NoMatch` and `any` at the first `Match`. Both
20//! operations are commutative, so the order of children never changes the
21//! result, and a later version may reorder them by cost (#42). The notes of
22//! an evaluation only come from the criteria that were evaluated, so they
23//! can depend on the order.
24//!
25//! # Rule level
26//!
27//! [`TreeEvaluation::matches`] decides whether the rule matches: only
28//! `Match` does. `Unknown` counts as no match for every action. For an allow
29//! rule that fails closed: an unknown fact never allows. For a deny rule it
30//! means the rule does not apply, so an unknown fact does not block either;
31//! that is the default (`no_match`) that #32 sets for deny rules.
32//!
33//! A deny rule can change both halves (#35). `unknown_as: match` makes it
34//! apply on `Unknown` as well, which the pipeline in [`crate::evaluate`]
35//! checks. `on_unparseable: match` is applied here, by
36//! [`CompiledMatch::eval_with`] and [`Unparseable::FavourMatch`]: a
37//! criterion that is `Unknown` only because its version fact is not a
38//! version becomes `Match` where the tree would match with it, and
39//! `NoMatch` under an odd number of `not` nodes. So it always counts
40//! towards the deny applying, and the rest of the tree is evaluated as
41//! usual.
42//!
43//! A criterion marked `safety_critical` in a mandatory blocklist works the
44//! same way for any unknown fact ([`CompiledMatch::eval_mandatory`]), so the
45//! shipped blocklists block an object whose facts are missing rather than
46//! let it through (#32, #87).
47

53/// A match tree ready to be evaluated. Build one with
54/// [`CompiledMatch::compile`].
55#[derive(Clone, Debug)]
56pub struct CompiledMatch {
57    /// Every node; the root is the last.
58    nodes: Vec<Node>,
59    /// The children of `all` and `any` nodes, as indices into `nodes`.
60    children: Vec<u32>,
61    /// The compiled criteria, in document order.
62    leaves: Vec<CompiledCriterion>,
63}
64
65#[derive(Clone, Copy, Debug)]
66enum Node {
67    /// `children[start..start + len]`.
68    All { start: u32, len: u32 },
69    /// `children[start..start + len]`.
70    Any { start: u32, len: u32 },
71    /// The node at this index.
72    Not(u32),
73    /// The leaf at this index.
74    Leaf(u32),
75}
76
77/// Why a tree could not be compiled.
78#[derive(Clone, Debug, PartialEq, Eq)]
79pub enum TreeError {
80    /// The tree is nested more deeply than [`MAX_TREE_DEPTH`].
81    TooDeep,
82    /// The tree holds more than [`MAX_CRITERIA_PER_RULE`] criteria.
83    TooManyCriteria,
84    /// An `all` or `any` node has no children. `pointer` is relative to the
85    /// tree's root, such as `/all/1/any`.
86    EmptyGroup {
87        /// Where the group is, relative to the root.
88        pointer: String,
89    },
90    /// The tree still holds a library reference, which the hierarchy
91    /// resolves before evaluation (#37). `pointer` is relative to the root.
92    UnresolvedRef {
93        /// Where the reference is, relative to the root.
94        pointer: String,
95    },
96    /// A criterion did not compile. `pointer` is relative to the tree's
97    /// root, such as `/all/0/criterion`.
98    Criterion {
99        /// Where the criterion is, relative to the root.
100        pointer: String,
101        /// What is wrong with it.
102        errors: Vec<CompileError>,
103    },
104}
105
106impl core::fmt::Display for TreeError {
107    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
108        match self {
109            Self::TooDeep => write!(f, "the tree is nested more than {MAX_TREE_DEPTH} deep"),
110            Self::TooManyCriteria => {
111                write!(
112                    f,
113                    "the tree holds more than {MAX_CRITERIA_PER_RULE} criteria"
114                )
115            }
116            Self::EmptyGroup { pointer } => {
117                write!(f, "{pointer}: a group needs at least one condition")
118            }
119            Self::UnresolvedRef { pointer } => {
120                write!(f, "{pointer}: a library reference was not resolved")
121            }
122            Self::Criterion { pointer, errors } => {
123                write!(f, "{pointer}: ")?;
124                for (i, e) in errors.iter().enumerate() {
125                    if i > 0 {
126                        f.write_str("; ")?;
127                    }
128                    match e.value {
129                        Some(v) => write!(f, "value {v}: {}", e.kind)?,
130                        None => write!(f, "{}", e.kind)?,
131                    }
132                }
133                Ok(())
134            }
135        }
136    }
137}

141/// How a criterion whose version fact is not a version is read (#32, #35).
142#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
143pub enum Unparseable {
144    /// It is `Unknown`, like any other fact that cannot be trusted. The
145    /// default, and the only choice for a rule that is not a deny.
146    #[default]
147    Unknown,
148    /// It counts towards the tree matching: `Match`, or `NoMatch` under an
149    /// odd number of `not` nodes. A deny rule's `on_unparseable: match`.
150    FavourMatch,
151}
152
153/// The result of evaluating a tree.
154#[derive(Clone, Copy, Debug, PartialEq, Eq)]
155pub struct TreeEvaluation {
156    /// The three-valued result.
157    pub result: Tri,
158    /// What the evaluated criteria noticed.
159    pub notes: Notes,
160}
161
162impl TreeEvaluation {
163    /// Whether the rule matches: only `Match` does. `Unknown` is no match
164    /// for every action, so an allow rule does not allow and a deny rule
165    /// does not block (see the module docs).
166    #[must_use]
167    pub fn matches(&self) -> bool {
168        self.result == Tri::Match
169    }
170}
171
172impl CompiledMatch {
173    /// Compiles a tree. Validation compiles every rule's tree, so the tree
174    /// of a validated policy always compiles.
175    ///
176    /// # Errors
177    ///
178    /// Returns every problem: the limits first, and if the tree is within
179    /// them, each empty group and each criterion that does not compile.
180    pub fn compile(tree: &Match) -> Result<Self, Vec<TreeError>> {
181        let mut errors = Vec::new();
182        if tree.depth() > MAX_TREE_DEPTH {
183            errors.push(TreeError::TooDeep);
184        }
185        if leaf_count(tree) > MAX_CRITERIA_PER_RULE {
186            errors.push(TreeError::TooManyCriteria);
187        }
188        if !errors.is_empty() {
189            return Err(errors);
190        }
191        let mut out = Self {
192            nodes: Vec::new(),
193            children: Vec::new(),
194            leaves: Vec::new(),
195        };
196        // The depth was checked, so this recursion is bounded.
197        out.add(tree, String::new(), &mut errors);
198        if errors.is_empty() {
199            Ok(out)
200        } else {
201            Err(errors)
202        }
203    }
204
205    /// Adds `node` and everything below it, and returns its index.
206    fn add(&mut self, node: &Match, pointer: String, errors: &mut Vec<TreeError>) -> u32 {
207        let entry = match node {
208            Match::All(items) | Match::Any(items) => {
209                let key = if matches!(node, Match::All(_)) {
210                    "all"
211                } else {
212                    "any"
213                };
214                if items.is_empty() {
215                    errors.push(TreeError::EmptyGroup {
216                        pointer: format!("{pointer}/{key}"),
217                    });
218                }
219                let ids: Vec<u32> = items
220                    .iter()
221                    .enumerate()
222                    .map(|(i, child)| self.add(child, format!("{pointer}/{key}/{i}"), errors))
223                    .collect();
224                let start = index(self.children.len());
225                let len = index(ids.len());
226                self.children.extend(ids);
227                if matches!(node, Match::All(_)) {
228                    Node::All { start, len }
229                } else {
230                    Node::Any { start, len }
231                }
232            }
233            Match::Not(child) => Node::Not(self.add(child, format!("{pointer}/not"), errors)),
234            Match::Ref(_) => {
235                errors.push(TreeError::UnresolvedRef {
236                    pointer: format!("{pointer}/ref"),
237                });
238                Node::Leaf(u32::MAX)
239            }
240            Match::Criterion(c) => match CompiledCriterion::compile(c) {
241                Ok(compiled) => {
242                    self.leaves.push(compiled);
243                    Node::Leaf(index(self.leaves.len() - 1))
244                }
245                Err(e) => {
246                    errors.push(TreeError::Criterion {
247                        pointer: format!("{pointer}/criterion"),
248                        errors: e,
249                    });
250                    Node::Leaf(u32::MAX)
251                }
252            },
253        };
254        self.nodes.push(entry);
255        index(self.nodes.len() - 1)
256    }
257
258    /// The compiled criteria, in document order.
259    #[must_use]
260    pub fn criteria(&self) -> &[CompiledCriterion] {
261        &self.leaves
262    }
263
264    /// Evaluates the tree against one object's facts and the context.
265    #[must_use]
266    pub fn eval(&self, facts: &ObjectFacts, ctx: &EvalContext<'_>) -> TreeEvaluation {
267        self.eval_with(facts, ctx, Unparseable::Unknown)
268    }
269
270    /// Evaluates the tree, reading criteria on unparseable versions as
271    /// `unparseable` says. With [`Unparseable::Unknown`] this is
272    /// [`eval`](Self::eval).
273    #[must_use]
274    pub fn eval_with(
275        &self,
276        facts: &ObjectFacts,
277        ctx: &EvalContext<'_>,
278        unparseable: Unparseable,
279    ) -> TreeEvaluation {
280        self.eval_full(facts, ctx, unparseable, false)
281    }
282
283    /// Evaluates the tree of a mandatory deny: as [`eval_with`](Self::eval_with),
284    /// and each `safety_critical` criterion that is `Unknown` counts towards
285    /// the deny applying (`Match`, or `NoMatch` under an odd number of `not`
286    /// nodes), so the shipped blocklists fail closed (#32, #87).
287    #[must_use]
288    pub fn eval_mandatory(
289        &self,
290        facts: &ObjectFacts,
291        ctx: &EvalContext<'_>,
292        unparseable: Unparseable,
293    ) -> TreeEvaluation {
294        self.eval_full(facts, ctx, unparseable, true)
295    }
296
297    fn eval_full(
298        &self,
299        facts: &ObjectFacts,
300        ctx: &EvalContext<'_>,
301        unparseable: Unparseable,
302        safety_critical: bool,
303    ) -> TreeEvaluation {
304        let mut eval = Eval {
305            facts,
306            ctx,
307            unparseable,
308            safety_critical,
309            notes: Notes::default(),
310        };
311        let root = index(self.nodes.len() - 1);
312        let result = self.eval_node(root, false, &mut eval);
313        TreeEvaluation {
314            result,
315            notes: eval.notes,
316        }
317    }
318
319    /// Evaluates one node. `negated` is whether an odd number of `not`
320    /// nodes are above it.
321    fn eval_node(&self, id: u32, negated: bool, eval: &mut Eval<'_, '_>) -> Tri {
322        match self.nodes[id as usize] {
323            Node::All { start, len } => {
324                let mut result = Tri::Match;
325                for &child in self.group(start, len) {
326                    result = result.and(self.eval_node(child, negated, eval));
327                    if result == Tri::NoMatch {
328                        break;
329                    }
330                }
331                result
332            }
333            Node::Any { start, len } => {
334                let mut result = Tri::NoMatch;
335                for &child in self.group(start, len) {
336                    result = result.or(self.eval_node(child, negated, eval));
337                    if result == Tri::Match {
338                        break;
339                    }
340                }
341                result
342            }
343            Node::Not(child) => self.eval_node(child, !negated, eval).not(),
344            Node::Leaf(leaf) => {
345                let criterion = &self.leaves[leaf as usize];
346                let evaluation = criterion.eval(eval.facts, eval.ctx);
347                eval.notes.extend(evaluation.notes);
348                if evaluation.result == Tri::Unknown
349                    && eval.safety_critical
350                    && criterion.safety_critical()
351                {
352                    // A safety-critical criterion of a mandatory deny fails
353                    // closed: whatever made it unknown, it counts towards
354                    // the deny applying.
355                    Tri::from_bool(!negated)
356                } else if evaluation.result == Tri::Unknown
357                    && eval.unparseable == Unparseable::FavourMatch
358                    && evaluation.notes.contains(Note::VersionUnparseable)
359                {
360                    // Each criterion tests one fact, so this `Unknown` comes
361                    // from the unparseable version and nothing else.
362                    Tri::from_bool(!negated)
363                } else {
364                    evaluation.result
365                }
366            }
367        }
368    }
369
370    fn group(&self, start: u32, len: u32) -> &[u32] {
371        &self.children[start as usize..(start + len) as usize]
372    }
373}

375/// The inputs of one evaluation, and the notes gathered so far.
376struct Eval<'f, 'c> {
377    facts: &'f ObjectFacts,
378    ctx: &'f EvalContext<'c>,
379    unparseable: Unparseable,
380    /// Whether `safety_critical` criteria fail closed: only in a mandatory
381    /// deny.
382    safety_critical: bool,
383    notes: Notes,
384}
385
386/// The number of criteria in a tree, counted without recursion.
387fn leaf_count(tree: &Match) -> usize {
388    let mut count = 0usize;
389    let mut stack = vec![tree];
390    while let Some(node) = stack.pop() {
391        if matches!(node, Match::Criterion(_)) {
392            count += 1;
393        }
394        stack.extend(node.children());
395    }
396    count
397}

399/// An arena index. The limits keep every tree far below `u32::MAX` nodes.
400fn index(i: usize) -> u32 {
401    u32::try_from(i).unwrap_or(u32::MAX)
402}
