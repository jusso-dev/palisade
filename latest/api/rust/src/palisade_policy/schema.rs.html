1//! The versioned JSON Schema (2020-12) of [`PolicyDocument`].
2//!
3//! The schema is generated from the Rust types, so it cannot drift from
4//! them. The generated text is committed at `schemas/policy/v{major}.json`,
5//! and the test `committed_schema_is_current` fails when the committed file
6//! differs from what [`generate`] produces. Regenerate it with:
7//!
8//! ```sh
9//! PALISADE_BLESS=1 cargo test -p palisade-policy --test schema
10//! ```
11//!
12//! # Versioning
13//!
14//! [`CURRENT`] is the version this crate writes and the major version it
15//! reads. A minor bump adds optional fields; a major bump is a breaking
16//! change, gets a new file (`v2.json`) and a migration step
17//! ([`crate::migrate`]). The schema's `$id` names the major version only, because every
18//! minor version of a major is a valid instance of the latest schema for
19//! that major.
20
21use crate::model::{PolicyDocument, SchemaVersion};
22
23/// The schema version this crate implements.
24pub const CURRENT: SchemaVersion = SchemaVersion::new(1, 0, 0);
25
26/// The schema's `$id`.
27pub const SCHEMA_ID: &str = "https://github.com/jusso-dev/palisade/schemas/policy/v1.json";
28
29/// Generates the JSON Schema of a policy document.
30#[must_use]
31pub fn generate() -> serde_json::Value {
32    let generator = schemars::generate::SchemaSettings::draft2020_12().into_generator();
33    let mut schema = generator.into_root_schema_for::<PolicyDocument>();
34    let object = schema.ensure_object();
35    object.insert("$id".to_owned(), SCHEMA_ID.into());
36    object.insert("title".to_owned(), "Palisade policy document".into());
37    object.insert("x-schema-version".to_owned(), CURRENT.to_string().into());
38    sort_keys(schema.to_value())
39}
40
41/// Rebuilds every object with its keys in sorted order, so the committed
42/// text does not depend on whether `serde_json`'s `preserve_order` feature
43/// happens to be enabled elsewhere in the build.
44pub(crate) fn sort_keys(value: serde_json::Value) -> serde_json::Value {
45    match value {
46        serde_json::Value::Object(map) => {
47            let mut entries: Vec<_> = map.into_iter().collect();
48            entries.sort_by(|a, b| a.0.cmp(&b.0));
49            entries
50                .into_iter()
51                .map(|(k, v)| (k, sort_keys(v)))
52                .collect::<serde_json::Map<_, _>>()
53                .into()
54        }
55        serde_json::Value::Array(items) => items.into_iter().map(sort_keys).collect(),
56        other => other,
57    }
58}

60/// The generated schema as the exact text that is committed: pretty-printed
61/// JSON with a trailing newline.
62///
63/// # Errors
64///
65/// Only fails if serialising a JSON value fails, which does not happen.
66pub fn generate_pretty() -> Result<String, serde_json::Error> {
67    let mut text = serde_json::to_string_pretty(&generate())?;
68    text.push('\n');
69    Ok(text)
70}
