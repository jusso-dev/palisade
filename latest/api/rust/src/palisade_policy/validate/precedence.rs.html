1//! Validation of the precedence model's inputs (#35): the options the
2//! selected model uses, and each rule's `order` and `priority`.
3
4use std::collections::{BTreeMap, BTreeSet};
5
6use super::{ErrorCode, Validator};
7use crate::model::{PolicyDocument, PrecedenceModel, PrecedenceOptions};
8use crate::precedence::Verdict;
9
10/// Reserved `order` ranges for the `first_match` model (#35). Only rules of
11/// custom rule sets are checked: curated (read-only) rule sets may use the
12/// curated range, and the system and catch-all ranges are for rules
13/// Palisade adds itself.
14pub mod order_ranges {
15    use core::ops::RangeInclusive;
16
17    /// Built-in system rules.
18    pub const SYSTEM: RangeInclusive<u32> = 0..=99;
19    /// Curated, read-only rule sets.
20    pub const CURATED: RangeInclusive<u32> = 100..=100_000;
21    /// Custom rules.
22    pub const CUSTOM: RangeInclusive<u32> = 100_001..=999_989;
23    /// Defaults and catch-all rules.
24    pub const CATCH_ALL: RangeInclusive<u32> = 999_990..=1_000_000;
25}
26
27impl Validator {
28    pub(super) fn precedence(&mut self, doc: &PolicyDocument) {
29        let policy = &doc.policy;
30        let model = policy.precedence_model;
31        self.unused_options(model, &policy.precedence_options);
32
33        // Only the rules of referenced rule sets are ever evaluated.
34        let referenced: BTreeSet<_> = policy.rule_sets.iter().collect();
35        // Priority -> (path, verdict) of the first rule seen with it.
36        let mut priorities: BTreeMap<i32, (String, Verdict)> = BTreeMap::new();
37        for (i, set) in doc.rule_sets.iter().enumerate() {
38            for (j, rule) in set.rules.iter().enumerate() {
39                let path = format!("/rule_sets/{i}/rules/{j}");
40                // The schema's limit, checked for every rule set.
41                if let Some(order) = rule.order.filter(|o| o > order_ranges::CATCH_ALL.end()) {
42                    self.push(
43                        ErrorCode::OrderOutOfRange,
44                        &format!("{path}/order"),
45                        format!(
46                            "order {order} is above the largest order, {}",
47                            order_ranges::CATCH_ALL.end()
48                        ),
49                    );
50                }
51                if !referenced.contains(&set.id) {
52                    continue;
53                }
54                // Under `first_match`, every rule of a custom rule set that
55                // has an order must use the custom range, whatever its
56                // action: a `defer` or `audit` rule is placed in the same
57                // order. Mandatory rule sets are decided before any order.
58                if model == PrecedenceModel::FirstMatch
59                    && !set.read_only
60                    && !set.mandatory
61                    && let Some(order) = rule.order.filter(|o| {
62                        !order_ranges::CUSTOM.contains(o)
63                            // A catch-all may also use the catch-all range
64                            // (#329).
65                            && !(rule.catch_all && order_ranges::CATCH_ALL.contains(o))
66                            && o <= order_ranges::CATCH_ALL.end()
67                    })
68                {
69                    self.push(
70                        ErrorCode::OrderReserved,
71                        &format!("{path}/order"),
72                        format!(
73                            "order {order} is reserved; rules of custom rule sets use {} to {}, and catch-all rules also {} to {}",
74                            order_ranges::CUSTOM.start(),
75                            order_ranges::CUSTOM.end(),
76                            order_ranges::CATCH_ALL.start(),
77                            order_ranges::CATCH_ALL.end()
78                        ),
79                    );
80                }
81                // Mandatory and exclusion rules are decided before the
82                // model, and the other non-terminal rules never decide.
83                // Attribute rules are stage 5, after the model (#329).
84                let Some(verdict) =
85                    Verdict::of(&rule.action).filter(|_| !set.mandatory && !rule.attribute)
86                else {
87                    continue;
88                };
89                match model {
90                    PrecedenceModel::FirstMatch => {
91                        if rule.order.is_none() {
92                            self.push(
93                                ErrorCode::OrderRequired,
94                                &format!("{path}/action"),
95                                "under `first_match`, every rule that allows or denies needs an `order`",
96                            );
97                        }
98                    }
99                    PrecedenceModel::Priority => match rule.priority {
100                        None => self.push(
101                            ErrorCode::PriorityRequired,
102                            &format!("{path}/action"),
103                            "under `priority`, every rule that allows or denies needs a `priority`",
104                        ),
105                        Some(p) => match priorities.get(&p) {
106                            Some((first, v)) if v.kind() != verdict.kind() => self.push(
107                                ErrorCode::DuplicatePriority,
108                                &format!("{path}/priority"),
109                                format!(
110                                    "priority {p} is also used by {first}, whose action is `{}`; between them the fixed action order decides",
111                                    v.kind()
112                                ),
113                            ),
114                            Some(_) => {}
115                            None => {
116                                priorities.insert(p, (path, verdict));
117                            }
118                        },
119                    },
120                    PrecedenceModel::DenyWins
121                    | PrecedenceModel::AllowWins
122                    | PrecedenceModel::Specificity => {}
123                }
124            }
125        }
126    }
127
128    /// Warns about each option that differs from its default but that the
129    /// selected model does not read.
130    fn unused_options(&mut self, model: PrecedenceModel, o: &PrecedenceOptions) {
131        use PrecedenceModel as M;
132        let d = PrecedenceOptions::default();
133        let options = [
134            (
135                "allow_tiebreak",
136                o.allow_tiebreak != d.allow_tiebreak,
137                matches!(model, M::DenyWins | M::AllowWins | M::Specificity),
138            ),
139            (
140                "granularity_tiebreak",
141                o.granularity_tiebreak != d.granularity_tiebreak,
142                model == M::AllowWins,
143            ),
144            (
145                "curated_first",
146                o.curated_first != d.curated_first,
147                model == M::FirstMatch,
148            ),
149            (
150                "same_tier_tiebreak",
151                o.same_tier_tiebreak != d.same_tier_tiebreak,
152                model == M::Specificity,
153            ),
154            (
155                "priority_direction",
156                o.priority_direction != d.priority_direction,
157                model == M::Priority,
158            ),
159        ];
160        for (name, set, used) in options {
161            if set && !used {
162                self.push(
163                    ErrorCode::PrecedenceOptionUnused,
164                    &format!("/policy/precedence_options/{name}"),
165                    format!(
166                        "`{name}` has no effect under `{}`",
167                        super::serde_name(&model)
168                    ),
169                );
170            }
171        }
172    }
