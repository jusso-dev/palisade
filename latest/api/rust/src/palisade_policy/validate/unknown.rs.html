17impl Validator {
18    pub(super) fn unknown(&mut self, doc: &PolicyDocument) {
19        let u = &doc.policy.unknown;
20        for (t, level) in &u.by_type {
21            let path = format!("{BASE}/by_type/{t}");
22            if *level == Level::Restrict && !restrict_valid_for(*t) {
23                self.push(
24                    ErrorCode::UnknownLevelNotValidForType,
25                    &path,
26                    format!("`{t}` cannot be contained; `restrict` is only for executables, scripts and installers"),
27                );
28            }
29            if *level == Level::Prompt && !prompt_valid_for(*t) {
30                self.push(
31                    ErrorCode::UnknownLevelNotValidForType,
32                    &path,
33                    format!("`{t}` loads without a user action, so it cannot be prompted for"),
34                );
35            }
36        }
37        if u.restrict_profile.is_none() && u.levels().any(|l| l == Level::Restrict) {
38            self.push(
39                ErrorCode::RestrictProfileRequired,
40                &format!("{BASE}/restrict_profile"),
41                "the `restrict` level needs `restrict_profile`, the containment profile to run unknown objects in",
42            );
43        }
44        self.prompt_settings(u);
45        self.effect_flags(&format!("{BASE}/effects"), ActionKind::Deny, &u.effects);
46        if u.allow_request && u.effects.notify_for(ActionKind::Deny) == Notify::None {
47            self.push(
48                ErrorCode::InvalidActionOption,
49                &format!("{BASE}/allow_request"),
50                "the user can only raise a request from a notification; `notify: none` hides it",
51            );
52        }
53        for (path, level) in entries(u) {
54            if level == Level::Allow {
55                self.push(
56                    ErrorCode::UnknownAllow,
57                    &path,
58                    "`allow` lets unknown objects run silently; use `detect` to at least report them",
59                );
60            }
61        }
62        if let Some(preset) = u.preset {
63            self.preset(doc, u, preset);
64        }
65    }

67    fn prompt_settings(&mut self, u: &UnknownPolicy) {
68        let p = &u.prompt;
69        let base = format!("{BASE}/prompt");
70        if p.options.is_empty() {
71            self.push(
72                ErrorCode::InvalidPromptSettings,
73                &format!("{base}/options"),
74                "a prompt needs at least one option",
75            );
76        }
77        let mut seen = BTreeSet::new();
78        for (i, o) in p.options.iter().enumerate() {
79            if !seen.insert(*o) {
80                self.push(
81                    ErrorCode::InvalidPromptSettings,
82                    &format!("{base}/options/{i}"),
83                    format!("`{}` is listed more than once", super::serde_name(o)),
84                );
85            }
86        }
87        if p.fallback == Level::Prompt {
88            self.push(
89                ErrorCode::InvalidPromptSettings,
90                &format!("{base}/fallback"),
91                "the fallback applies when no prompt can be shown, so it cannot be `prompt`",
92            );
93        }
94        if !(PROMPT_TIMEOUT_MIN..=PROMPT_TIMEOUT_MAX).contains(&p.timeout_secs) {
95            self.push(
96                ErrorCode::SettingOutOfRange,
97                &format!("{base}/timeout_secs"),
98                format!(
99                    "{} is outside {PROMPT_TIMEOUT_MIN}..={PROMPT_TIMEOUT_MAX}",
100                    p.timeout_secs
101                ),
102            );
103        }
104        if !(TEMP_APPROVAL_DAYS_MIN..=TEMP_APPROVAL_DAYS_MAX).contains(&p.temp_approval_days) {
105            self.push(
106                ErrorCode::SettingOutOfRange,
107                &format!("{base}/temp_approval_days"),
108                format!(
109                    "{} is outside {TEMP_APPROVAL_DAYS_MIN}..={TEMP_APPROVAL_DAYS_MAX}",
110                    p.temp_approval_days
111                ),
112            );
113        }
114    }

116    /// The checks of a preset-tagged policy.
117    fn preset(
118        &mut self,
119        doc: &PolicyDocument,
120        u: &UnknownPolicy,
121        preset: crate::model::CompliancePreset,
122    ) {
123        // Silent allow in user profiles or temporary folders.
124        for class in [LocationClass::UserProfile, LocationClass::Temp] {
125            if let Some((t, classes, origin)) = weakest(u, class, |l| l == Level::Allow) {
126                self.push(
127                    ErrorCode::UnknownAllowNotPermitted,
128                    BASE,
129                    format!(
130                        "under the `{}` preset an unknown `{t}` in {} with origin `{}` would run silently; Essential Eight requires default deny in `{}` (ISM-1870)",
131                        preset.as_str(),
132                        describe(&classes),
133                        origin.as_str(),
134                        class.as_str()
135                    ),
136                );
137            }
138        }
139        // Weaker than the preset for some object.
140        let reference = UnknownPolicy::preset(preset);
141        if let Some((t, classes, origin, ours, theirs)) =
142            cases().find_map(|(t, classes, origin)| {
143                let ours = floor(u, t, &classes, origin);
144                let theirs = floor(&reference, t, &classes, origin);
145                (ours < theirs).then_some((t, classes, origin, ours, theirs))
146            })
147        {
148            self.push(
149                ErrorCode::ComplianceWeakened,
150                BASE,
151                format!(
152                    "weaker than the `{}` preset: an unknown `{t}` in {} with origin `{}` gets `{ours}`, the preset gives `{theirs}`",
153                    preset.as_str(),
154                    describe(&classes),
155                    origin.as_str()
156                ),
157            );
158        }
159        if u.preexisting_policy != PreexistingPolicy::BaselineOnly {
160            self.push(
161                ErrorCode::ComplianceWeakened,
162                &format!("{BASE}/preexisting_policy"),
163                format!(
164                    "weaker than the `{}` preset, which allows files present at install only through the approved baseline",
165                    preset.as_str()
166                ),
167            );
168        }
169        if preset == crate::model::CompliancePreset::E8Ml3
170            && u.driver_boot_policy == crate::model::DriverBootPolicy::Audit
171        {
172            // Kept by the preset on purpose: blocking an unrecognised
173            // boot-start driver can leave the device unable to boot (ADR
174            // 0018). The exemption is reported, not silent.
175            self.push(
176                ErrorCode::ComplianceWeakened,
177                &format!("{BASE}/driver_boot_policy"),
178                "under the `e8_ml3` preset unknown drivers are blocked, except during early boot: `driver_boot_policy: audit` only reports drivers loaded before the agent has fully started; set `enforce` once the device's boot-start drivers are known",
179            );
180        }
181        if preset == crate::model::CompliancePreset::E8Ml3 {
182            let referenced: BTreeSet<_> = doc.policy.rule_sets.iter().collect();
183            let has_mandatory = doc.rule_sets.iter().any(|s| {
184                s.mandatory
185                    && s.kind == RuleSetKind::Blocklist
186                    && referenced.contains(&s.id)
187                    // Validation has no clock, so only a rule that is in
188                    // force at every time counts: enabled, with no
189                    // activity window. One that has expired, or has not
190                    // started, is no mandatory blocklist.
191                    && s.rules
192                        .iter()
193                        .any(|r| r.enabled && r.activates_at.is_none() && r.expires_at.is_none())
194            });
195            if !has_mandatory {
196                self.push(
197                    ErrorCode::MandatoryBlocklistRequired,
198                    "/policy/rule_sets",
199                    "the `e8_ml3` preset requires a referenced mandatory blocklist with at least one enabled rule with no `activates_at` or `expires_at` (recommended block rules and the vulnerable driver blocklist, ISM-1544, ISM-1659)",
200                );
201            }
202        }
203    }
204}

206/// Every level the policy sets, with its JSON Pointer.
207fn entries(u: &UnknownPolicy) -> Vec<(String, Level)> {
208    let mut out = vec![(format!("{BASE}/default"), u.default)];
209    out.extend(
210        u.by_location
211            .iter()
212            .map(|(k, l)| (format!("{BASE}/by_location/{}", k.as_str()), *l)),
213    );
214    out.extend(
215        u.by_origin
216            .iter()
217            .map(|(k, l)| (format!("{BASE}/by_origin/{}", k.as_str()), *l)),
218    );
219    out.extend(
220        u.by_type
221            .iter()
222            .map(|(k, l)| (format!("{BASE}/by_type/{k}"), *l)),
223    );
224    out.push((format!("{BASE}/prompt/fallback"), u.prompt.fallback));
225    out
226}

228/// The least an unknown object of type `t` in the location classes
229/// `classes`, with the single origin `origin`, can end up with: a prompt
230/// may end in its fallback.
231fn floor(
232    u: &UnknownPolicy,
233    t: ObjectType,
234    classes: &BTreeSet<LocationClass>,
235    origin: OriginClass,
236) -> Level {
237    let fallback = match u.prompt.fallback {
238        Level::Prompt => Level::Block,
239        other => other,
240    };
241    if origin == OriginClass::Preexisting && u.preexisting_policy == PreexistingPolicy::BaselineOnly
242    {
243        return Level::Block;
244    }
245    let (level, _) = resolve_level(u, t, classes, &BTreeSet::from([origin]));
246    level.floor(fallback)
247}

249/// Every object type, with every set of one or two location classes and
250/// every single origin.
251///
252/// These are exactly the cases that can give an object its weakest level.
253/// The level is the most restrictive entry that applies, or `default` when
254/// none does. So a larger set of classes or origins only ever adds entries,
255/// and the most restrictive wins, except that a set where nothing applies
256/// takes `default`, which a second class with a weaker entry can undercut.
257/// One more class is enough for that, and one origin is always present,
258/// so the weakest level over every set that contains a class is reached
259/// with that class and at most one other. Any two classes are taken to be
260/// able to occur together: the agent supplies the classes, and the check
261/// errs towards a warning.
262fn cases() -> impl Iterator<Item = (ObjectType, BTreeSet<LocationClass>, OriginClass)> {
263    ObjectType::ALL.into_iter().flat_map(|t| {
264        LocationClass::ALL
265            .into_iter()
266            .enumerate()
267            .flat_map(move |(i, a)| {
268                LocationClass::ALL[i..].iter().flat_map(move |b| {
269                    OriginClass::ALL
270                        .into_iter()
271                        .map(move |o| (t, BTreeSet::from([a, *b]), o))
272                })
273            })
274    })
275}

277/// A case that contains `class` and ends up at a level `bad` accepts.
278fn weakest(
279    u: &UnknownPolicy,
280    class: LocationClass,
281    bad: impl Fn(Level) -> bool,
282) -> Option<(ObjectType, BTreeSet<LocationClass>, OriginClass)> {
283    cases().find(|(t, classes, o)| classes.contains(&class) && bad(floor(u, *t, classes, *o)))
284}

286/// `` `a` `` or `` `a` and `b` ``.
287fn describe(classes: &BTreeSet<LocationClass>) -> String {
288    classes
289        .iter()
290        .map(|c| format!("`{}`", c.as_str()))
291        .collect::<Vec<_>>()
292        .join(" and ")
