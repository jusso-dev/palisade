3use std::collections::BTreeSet;
4
5use super::{ErrorCode, Validator};
6use crate::lifecycle::{is_exception, needs_review, review_by};
7use crate::model::lifecycle::limits::{
8    COUNTER_REPORT_HOURS, EXPIRING_SOON_DAYS, MAX_EXCEPTION_DAYS, MAX_EXPIRY_NOTICE_DAYS,
9    MAX_EXPIRY_NOTICES, STALE_AFTER_DAYS,
10};
11use crate::model::{ActionKind, FindingSeverity, OnExpire, PolicyDocument, Rule, RuleSet};
12use crate::unknown::baseline::SYSTEM_BASELINE_ID;
13
14impl Validator {
15    pub(super) fn lifecycle(&mut self, doc: &PolicyDocument) {
16        let g = &doc.policy.governance;
17        let base = "/policy/governance";
18        for (key, value, range) in [
19            ("stale_after_days", g.stale_after_days, STALE_AFTER_DAYS),
20            (
21                "expiring_soon_days",
22                g.expiring_soon_days,
23                EXPIRING_SOON_DAYS,
24            ),
25            (
26                "counter_report_hours",
27                g.counter_report_hours,
28                COUNTER_REPORT_HOURS,
29            ),
30        ] {
31            if !range.contains(&value) {
32                self.push(
33                    ErrorCode::InvalidGovernance,
34                    &format!("{base}/{key}"),
35                    format!("{value} is outside {}..={}", range.start(), range.end()),
36                );
37            }
38        }
39        if g.expiry_notice_days.len() > MAX_EXPIRY_NOTICES {
40            self.push(
41                ErrorCode::InvalidGovernance,
42                &format!("{base}/expiry_notice_days"),
43                format!("at most {MAX_EXPIRY_NOTICES} notices"),
44            );
45        }
46        let mut seen = BTreeSet::new();
47        for (i, d) in g.expiry_notice_days.iter().enumerate() {
48            if *d > MAX_EXPIRY_NOTICE_DAYS || !seen.insert(*d) {
49                self.push(
50                    ErrorCode::InvalidGovernance,
51                    &format!("{base}/expiry_notice_days/{i}"),
52                    format!(
53                        "each notice is 0 to {MAX_EXPIRY_NOTICE_DAYS} days before expiry, once"
54                    ),
55                );
56            }
57        }
58        for (key, severity) in &g.severity {
59            let path = format!("{base}/severity/{}", super::pointer_token(key));
60            match crate::analysis::FindingCode::from_name(key) {
61                None => self.push(
62                    ErrorCode::InvalidGovernance,
63                    &path,
64                    format!("`{key}` is not a finding code"),
65                ),
66                // A policy may raise a severity, never lower it: lowering
67                // an `error` would let the policy's own author pass the
68                // publish gate without an administrator's override (#40).
69                Some(code) if *severity < code.default_severity() => self.push(
70                    ErrorCode::InvalidGovernance,
71                    &path,
72                    format!(
73                        "`{key}` is `{}` by default; a policy may raise a finding's severity but not lower it (a tenant administrator overrides findings at publishing)",
74                        severity_name(code.default_severity())
75                    ),
76                ),
77                Some(_) => {}
78            }
79        }
80
81        for (i, set) in doc.rule_sets.iter().enumerate() {
82            if set.id.to_string() == SYSTEM_BASELINE_ID {
83                self.push(
84                    ErrorCode::ReservedRuleSetId,
85                    &format!("/rule_sets/{i}/id"),
86                    format!(
87                        "`{SYSTEM_BASELINE_ID}` is the managed `system_baseline`; the engine adds it after validation (`ValidatedPolicy::with_system_baseline`), so a policy document cannot contain a rule set with this ID"
88                    ),
89                );
90            }
91            for (j, rule) in set.rules.iter().enumerate() {
92                let path = format!("/rule_sets/{i}/rules/{j}");
93                if let (Some(a), Some(e)) = (rule.activates_at, rule.expires_at)
94                    && e <= a
95                {
96                    self.push(
97                        ErrorCode::InvalidActivityWindow,
98                        &format!("{path}/expires_at"),
99                        format!("`expires_at` ({e}) must be after `activates_at` ({a})"),
100                    );
101                }
102                if rule.on_expire != OnExpire::Deactivate && rule.expires_at.is_none() {
103                    self.push(
104                        ErrorCode::OnExpireNotApplicable,
105                        &format!("{path}/on_expire"),
106                        "`on_expire` needs `expires_at`",
107                    );
108                } else if rule.on_expire == OnExpire::ConvertToAudit && set.mandatory {
109                    self.push(
110                        ErrorCode::OnExpireNotApplicable,
111                        &format!("{path}/on_expire"),
112                        "a mandatory rule set holds only denies; an expired mandatory rule is removed, not converted",
113                    );
114                }
115                if let OnExpire::DeleteAfterDays(d) = rule.on_expire
116                    && !(1..=3650).contains(&d)
117                {
118                    self.push(
119                        ErrorCode::OnExpireNotApplicable,
120                        &format!("{path}/on_expire/delete_after_days"),
121                        format!("{d} is outside 1..=3650"),
122                    );
123                }
124                for (code, suffix, message) in governance_problems(doc, set, rule) {
125                    self.push(code, &format!("{path}{suffix}"), message);
126                }
127            }
128        }
129    }
130}

132const fn severity_name(s: FindingSeverity) -> &'static str {
133    match s {
134        FindingSeverity::Info => "info",
135        FindingSeverity::Warning => "warning",
136        FindingSeverity::Error => "error",
137    }

140/// Whether governance mode is on for `doc`: `require_justification`, or
141/// unset, on exactly when the policy has a compliance preset.
142pub(crate) fn governance_mode(doc: &PolicyDocument) -> bool {
143    doc.policy
144        .governance
145        .mode(doc.policy.unknown.preset.is_some())
146}
147
148/// What governance mode finds wrong with `rule` of `set` in `doc`: each
149/// problem's code, its path below the rule (empty for the rule itself)
150/// and a message. Empty when governance mode is off, the rule is disabled,
151/// or the set is not referenced. No rule set is exempt: `read_only` is a
152/// curated set's precedence flag that an author can set, and the managed
153/// `system_baseline` never reaches validation (it is added afterwards,
154/// [`crate::validate::ValidatedPolicy::with_system_baseline`]).
155pub(crate) fn governance_problems(
156    doc: &PolicyDocument,
157    set: &RuleSet,
158    rule: &Rule,
159) -> Vec<(ErrorCode, &'static str, String)> {
160    let mut out = Vec::new();
161    if !governance_mode(doc) || !rule.enabled || !doc.policy.rule_sets.contains(&set.id) {
162        return out;
163    }
164    let kind = rule.action.kind();
165    let m = &rule.metadata;
166    let exception = is_exception(rule);
167    let needs_justification =
168        kind.permits() || matches!(kind, ActionKind::TrustSource | ActionKind::Ignore) || exception;
169    if needs_justification
170        && m.justification
171            .as_deref()
172            .is_none_or(|j| j.trim().is_empty())
173    {
174        out.push((
175            ErrorCode::JustificationRequired,
176            "/metadata/justification",
177            format!("governance mode needs a justification for this `{kind}` rule"),
178        ));
179    }
180    if needs_review(kind) && review_by(m, Some(kind)).is_none() {
181        out.push((
182            ErrorCode::ReviewDateRequired,
183            "/metadata/review_by",
184            "governance mode needs `review_by`, or `created_at` to derive it (a year later)"
185                .to_owned(),
186        ));
187    }
188    if exception {
189        let mut missing = Vec::new();
190        match (rule.expires_at, crate::lifecycle::exception_start(m)) {
191            (None, None) => missing.extend(["`expires_at`", "`approved_at`"]),
192            (None, Some(_)) => missing.push("`expires_at`"),
193            (Some(_), None) => missing.push("`approved_at`, to bound the expiry"),
194            (Some(e), Some(start)) => {
195                if start.plus_days(MAX_EXCEPTION_DAYS).is_none_or(|l| e > l) {
196                    missing.push("`expires_at` at most 365 days after `approved_at`");
197                }
198            }
199        }
200        if m.approved_by.is_none() {
201            missing.push("`approved_by`");
202        }
203        if m.risk_rating.is_none() {
204            missing.push("`risk_rating`");
205        }
206        if !missing.is_empty() {
207            out.push((
208                ErrorCode::ExceptionIncomplete,
209                "",
210                format!(
211                    "this rule is an exception (an exclusion, an allow on a weak or path identity, or `metadata.exception`); governance mode needs {}",
212                    missing.join(", ")
213                ),
214            ));
215        }
216    }
217    out
