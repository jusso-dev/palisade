1//! Validation of catch-all rules (#33, #35, #329).
2//!
3//! A `catch_all` rule is the default of a policy: the only rule that may
4//! have no identity anchor, and a floor at stage 4 that decides only when no
5//! other rule matched ([`crate::evaluate`]). Because it can match every
6//! object, it is held to stricter rules than any other:
7//!
8//! - its name starts with `Default` or contains `Catch-all`, so it is
9//!   recognisable wherever it is listed;
10//! - it allows or denies (`allow`, `allow_contained`, `allow_elevated` or
11//!   `deny`): a default that only audits, trusts, excludes or defers decides
12//!   nothing;
13//! - it is not in a mandatory rule set, whose rules must identify what they
14//!   block;
15//! - a rule set holds at most one;
16//! - one that lets objects run needs the policy's `confirm_permit_all: "I
17//!   UNDERSTAND"`, and is never allowed under a compliance preset, which
18//!   requires default deny (ISM-1870);
19//! - under `first_match`, a policy should end with one, enabled, in a rule
20//!   set it references (`catch_all_missing`). This is a warning, not an
21//!   error: without a catch-all, an object no rule matched goes to the
22//!   unknown policy (#39), whose default blocks, so a missing one never
23//!   fails open.
24
25use std::collections::BTreeSet;
26
27use super::{ErrorCode, Validator};
28use crate::model::{CONFIRM_PERMIT_ALL, PolicyDocument, PrecedenceModel};
29use crate::precedence::Verdict;
30
31impl Validator {
32    pub(super) fn catch_alls(&mut self, doc: &PolicyDocument) {
33        let policy = &doc.policy;
34        let referenced: BTreeSet<_> = policy.rule_sets.iter().collect();
35        let confirmed = policy.confirm_permit_all.as_deref() == Some(CONFIRM_PERMIT_ALL);
36        if let Some(text) = policy
37            .confirm_permit_all
38            .as_deref()
39            .filter(|t| *t != CONFIRM_PERMIT_ALL)
40        {
41            self.push(
42                ErrorCode::PermitAllUnconfirmed,
43                "/policy/confirm_permit_all",
44                format!("`{text}` does not confirm; write exactly `{CONFIRM_PERMIT_ALL}`"),
45            );
46        }
47        let mut found = false;
48        for (i, set) in doc.rule_sets.iter().enumerate() {
49            for (j, rule) in set.rules.iter().enumerate().filter(|(_, r)| r.attribute) {
50                self.attribute_rule(&format!("/rule_sets/{i}/rules/{j}"), set, rule);
51            }
52            let mut first: Option<usize> = None;
53            for (j, rule) in set.rules.iter().enumerate() {
54                if !rule.catch_all {
55                    continue;
56                }
57                let path = format!("/rule_sets/{i}/rules/{j}");
58                if !rule.has_catch_all_name() {
59                    self.push(
60                        ErrorCode::InvalidCatchAll,
61                        &format!("{path}/name"),
62                        "a catch-all rule's name starts with `Default` or contains `Catch-all`",
63                    );
64                }
65                let verdict = Verdict::of(&rule.action);
66                if verdict.is_none() {
67                    self.push(
68                        ErrorCode::InvalidCatchAll,
69                        &format!("{path}/action"),
70                        format!(
71                            "a catch-all rule allows or denies; `{}` decides nothing",
72                            rule.action.kind()
73                        ),
74                    );
75                }
76                if set.mandatory {
77                    self.push(
78                        ErrorCode::InvalidCatchAll,
79                        &format!("{path}/catch_all"),
80                        "a mandatory rule set blocks only what its rules identify; it cannot hold a catch-all",
81                    );
82                }
83                match first {
84                    Some(k) => self.push(
85                        ErrorCode::InvalidCatchAll,
86                        &format!("{path}/catch_all"),
87                        format!(
88                            "rule set {} already has a catch-all, /rule_sets/{i}/rules/{k}",
89                            set.id
90                        ),
91                    ),
92                    None => first = Some(j),
93                }
94                if verdict.is_some_and(Verdict::allows) {
95                    if !confirmed {
96                        self.push(
97                            ErrorCode::PermitAllUnconfirmed,
98                            &format!("{path}/action"),
99                            format!(
100                                "this catch-all lets every object no other rule matched run; set the policy's `confirm_permit_all` to `{CONFIRM_PERMIT_ALL}` if that is intended"
101                            ),
102                        );
103                    }
104                    if let Some(preset) = policy.unknown.preset {
105                        self.push(
106                            ErrorCode::UnknownAllowNotPermitted,
107                            &format!("{path}/action"),
108                            format!(
109                                "under the `{}` preset a catch-all cannot let objects run: Essential Eight requires default deny (ISM-1870)",
110                                preset.as_str()
111                            ),
112                        );
113                    }
114                }
115                found |= rule.enabled
116                    && verdict.is_some()
117                    && !set.mandatory
118                    && referenced.contains(&set.id);
119            }
120        }
121        if policy.precedence_model == PrecedenceModel::FirstMatch && !found {
122            self.push(
123                ErrorCode::CatchAllMissing,
124                "/policy/precedence_model",
125                "a `first_match` policy ends with a catch-all rule: add an enabled rule with `catch_all: true`, named `Default…` or containing `Catch-all`, such as a deny whose match is `all: []`",
126            );
127        }
128    }
129}

131impl Validator {
132    /// A post-allow attribute rule (#329): it only narrows an allow, so it
133    /// denies or audits, is not in a mandatory rule set and is not a
134    /// catch-all.
135    fn attribute_rule(
136        &mut self,
137        path: &str,
138        set: &crate::model::RuleSet,
139        rule: &crate::model::Rule,
140    ) {
141        use crate::model::Action;
142        if !matches!(rule.action, Action::Deny(_) | Action::Audit {}) {
143            self.push(
144                ErrorCode::InvalidAttributeRule,
145                &format!("{path}/action"),
146                format!(
147                    "an attribute rule only narrows an allow: it denies or audits, not `{}`",
148                    rule.action.kind()
149                ),
150            );
151        }
152        if set.mandatory {
153            self.push(
154                ErrorCode::InvalidAttributeRule,
155                &format!("{path}/attribute"),
156                "a mandatory rule set blocks before any allow; it cannot hold an attribute rule",
157            );
158        }
159        if rule.catch_all {
160            self.push(
161                ErrorCode::InvalidAttributeRule,
162                &format!("{path}/attribute"),
163                "a rule is a catch-all or an attribute rule, not both",
164            );
165        }
166    }
167}
