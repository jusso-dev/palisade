1//! The typed output of an evaluation (#34).
2//!
3//! Every consumer gets the same [`Decision`]: the agent acts on it, the
4//! decision trace explains it (#41), and events embed it (#16). Its JSON
5//! form is stable: fields are always present, in declaration order, with
6//! `null` for an absent value, and the snapshot tests in
7//! `tests/snapshots/decision/` fail on any change to it.
8//!
9//! This module builds the decision of one terminal rule
10//! ([`Decision::from_rule`]) and the implicit deny for an object no rule
11//! decides ([`Decision::implicit_deny`]), both as enforce mode sees them.
12//! The pipeline in [`crate::evaluate`] combines the matching rules (#35):
13//! it adds the `audit` annotations, the rules the deciding rule overrode,
14//! and the trust-source grants. It then applies the effective mode (#36,
15//! [`crate::mode`]), which can change [`Decision::outcome`] and keeps the
16//! enforce-mode result in [`Decision::would_have_been`].
17

30/// What happened to the object.
31#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
32#[serde(rename_all = "snake_case")]
33pub enum Outcome {
34    /// It runs.
35    Allowed,
36    /// It does not run.
37    Blocked,
38    /// It runs, but would have been blocked in enforce mode.
39    AllowedAudit,
40    /// It runs, because an override session allows what would have been
41    /// blocked (#9).
42    AllowedOverride,
43    /// It was excluded by an `ignore` rule: no decision is logged.
44    Ignored,
45}
46
47/// Why the decision was made. The `snake_case` name is part of the API.
48#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
49#[serde(rename_all = "snake_case")]
50pub enum ReasonCode {
51    /// An allow rule decided.
52    ExplicitAllow,
53    /// A deny rule decided.
54    ExplicitDeny,
55    /// A rule in a mandatory rule set decided (ADR 0004).
56    MandatoryDeny,
57    /// No rule decided, and unknown objects are denied (#39).
58    ImplicitDenyUnknown,
59    /// A non-enforcing mode allowed what enforce mode would block (#36).
60    AuditModeWouldBlock,
61    /// An override session allowed what would have been blocked (#9).
62    OverrideSession,
63    /// An `ignore` rule excluded the object.
64    Exclusion,
65    /// The deciding allow's per-session instance limit was reached, or
66    /// the instance count was unknown (#38).
67    InstanceLimit,
68}

70/// A rule's human short ID, `R-` and 6 to 12 digits, kept as a number and
71/// its digit count so that [`RuleRef`] stays `Copy`.
72#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
73pub struct DisplayId {
74    number: u64,
75    digits: u8,
76}
77
78impl DisplayId {
79    /// Parses `R-` followed by 6 to 12 ASCII digits, the form validation
80    /// accepts.
81    #[must_use]
82    pub fn parse(text: &str) -> Option<Self> {
83        let digits = text.strip_prefix("R-")?;
84        if !(6..=12).contains(&digits.len()) || !digits.bytes().all(|b| b.is_ascii_digit()) {
85            return None;
86        }
87        Some(Self {
88            number: digits.parse().ok()?,
89            digits: u8::try_from(digits.len()).ok()?,
90        })
91    }
92}
93
94impl fmt::Display for DisplayId {
95    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
96        write!(
97            f,
98            "R-{:0width$}",
99            self.number,
100            width = usize::from(self.digits)
101        )
102    }
103}

105impl Serialize for DisplayId {
106    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
107        serializer.collect_str(self)
108    }
109}

111/// Which rule, in which revision of which policy. Small and `Copy`, so it

234/// The directives a decision carries for the components that act on it.
235/// The engine passes them through; their meaning belongs to the owning
236/// epics. These are separate from the model's directive types, which leave
237/// defaults out when written, so that a decision's JSON always has every
238/// field.

246    /// matching `trust_source` rule, by rule ID. Only an allowed object
247    /// gets grants (#35).
248    pub trust_sources: Vec<TrustGrant>,
249    /// Terminate running instances that match.
250    pub kill_running: bool,
251    /// The user may ask for approval.
252    pub request: Option<RequestDirective>,
253    /// What to do with the running process when the deciding allow's

258/// What the agent and the tray do when the schedule of the allow that let
259/// a process run closes (#38, #4, #11), as a decision carries it.
260#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
261pub struct WindowClose {
262    /// Warn, or warn and then terminate.
263    pub action: OnWindowClose,
264    /// Seconds between the warning and termination.
265    pub grace_secs: u32,
266    /// When the schedule closes, in seconds since 1970-01-01T00:00:00Z, or
267    /// `null` if it is not known yet. The agent re-evaluates on every
268    /// heartbeat.
269    pub closes_at: Option<i64>,
270}
271
272impl Directives {
273    /// The directives of the action of a deciding rule.
274    #[must_use]
275    pub fn of(action: &Action) -> Self {
276        let mut d = Self::default();
277        match action {
278            Action::Deny(o) => {
279                d.kill_running = o.kill_running;
280                d.request = o.allow_request.then_some(RequestDirective {
281                    template: o.request_template,
282                });
283            }
284            Action::AllowContained(c) => d.containment = Some(c.into()),
285            Action::AllowElevated(e) => d.elevation = Some(e.into()),
286            Action::Allow {}
287            | Action::Audit {}
288            | Action::TrustSource(_)
289            | Action::Ignore {}
290            | Action::Defer {} => {}
291        }
292        d
293    }
294}

296/// The typed result of evaluating one object.
297///
298/// The fields are read through accessors, so a decision is always one of
299/// the forms its constructors build. Serialising it gives the stable JSON
300/// form described in the module documentation.
301#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
302pub struct Decision {
303    outcome: Outcome,
304    would_have_been: Outcome,
305    action: ActionKind,
306    matched_rule: Option<RuleRef>,
307    overridden: Vec<RuleRef>,
308    annotations: Vec<RuleRef>,
309    mode: Mode,
310    learning_candidate: bool,
311    directives: Directives,
312    effects: ResolvedEffects,
313    reason_code: ReasonCode,
314    scope_notes: Vec<ScopeNote>,
315    trace_id: TraceId,
316    /// The directives of an allow that its instance limit blocked (#38),
317    /// for a mode that relaxes the block: the object then runs with them.
318    /// Not part of the JSON form.
319    #[serde(skip)]
320    withheld: Option<Box<Directives>>,
321}

323impl Decision {
324    /// The decision of `rule`, held by `set`, which `policy` references, as
325    /// enforce mode sees it. The caller has already found that the rule
326    /// matches and wins (#35).
327    ///
328    /// Returns `None` for a disabled rule, which never decides, and for an
329    /// action that does not decide by itself (`audit`, `defer`,
330    /// `trust_source`; see [`ActionKind::is_terminal`]).
331    #[must_use]
332    pub fn from_rule(
333        policy: &Policy,
334        set: &RuleSet,
335        rule: &Rule,
336        trace_id: TraceId,
337    ) -> Option<Self> {
338        if !rule.enabled {
339            return None;
340        }
341        let kind = rule.action.kind();
342        let (outcome, reason_code) = match kind {
343            ActionKind::Allow | ActionKind::AllowContained | ActionKind::AllowElevated => {
344                (Outcome::Allowed, ReasonCode::ExplicitAllow)
345            }
346            ActionKind::Deny if set.mandatory => (Outcome::Blocked, ReasonCode::MandatoryDeny),
347            ActionKind::Deny => (Outcome::Blocked, ReasonCode::ExplicitDeny),
348            ActionKind::Ignore => (Outcome::Ignored, ReasonCode::Exclusion),
349            ActionKind::Audit | ActionKind::Defer | ActionKind::TrustSource => return None,
350        };
351        Some(Self {
352            outcome,
353            would_have_been: outcome,
354            action: kind,
355            matched_rule: Some(RuleRef::new(policy, set, rule)),
356            overridden: Vec::new(),
357            annotations: Vec::new(),
358            mode: Mode::Enforce,
359            learning_candidate: false,
360            directives: Directives::of(&rule.action),
361            effects: rule.effects.resolve(kind),
362            reason_code,
363            scope_notes: Vec::new(),
364            trace_id,
365            withheld: None,
366        })
367    }

390    }

432    /// Adds what the pipeline gathered besides the deciding rule (#35).
433    /// Each list is sorted by rule ID, so the decision does not depend on
434    /// the order in which rules were evaluated.
435    pub(crate) fn with_context(
436        mut self,
437        mut overridden: Vec<RuleRef>,
438        mut annotations: Vec<RuleRef>,
439        mut trust_sources: Vec<TrustGrant>,
440    ) -> Self {
441        overridden.sort_by_key(|r| r.rule_id);
442        annotations.sort_by_key(|r| r.rule_id);
443        trust_sources.sort_by_key(|g| g.rule.rule_id);
444        self.overridden = overridden;
445        self.annotations = annotations;
446        self.directives.trust_sources = trust_sources;
447        self
448    }

450    /// Adds rules to [`overridden`](Self::overridden), keeping it sorted.
451    pub(crate) fn with_overridden(mut self, more: Vec<RuleRef>) -> Self {
452        self.overridden.extend(more);
453        self.overridden.sort_by_key(|r| r.rule_id);
454        self
455    }

474    /// An allow's decision turned into a block by its per-session instance
475    /// limit (#38): blocked, naming the allow. The block takes a deny's
476    /// default effects, so it is logged and sent at once and the user is
477    /// told, and keeps the allow's tags. Nothing runs, so it carries no
478    /// directives, and the trust sources are listed as overridden. If a
479    /// mode relaxes the block, the object runs with the allow's directives
480    /// ([`relaxed`](Self::relaxed)).
481    pub(crate) fn instance_limit(mut self) -> Self {
482        self.outcome = Outcome::Blocked;
483        self.would_have_been = Outcome::Blocked;
484        self.reason_code = ReasonCode::InstanceLimit;
485        let tags = core::mem::take(&mut self.effects.tags);
486        self.effects = crate::model::Effects::default().resolve(ActionKind::Deny);
487        self.effects.tags = tags;
488        if let Some(rule) = self.matched_rule {
489            self.overridden.extend(
490                self.directives
491                    .trust_sources
492                    .iter()
493                    .map(|g| g.rule)
494                    .filter(|r| *r != rule),
495            );
496            self.overridden.sort_by_key(|r| r.rule_id);
497        }
498        self.withheld = Some(Box::new(core::mem::take(&mut self.directives)));
499        self
500    }
501
502    /// The directives an object gets if a mode lets it run: those of the
503    /// allow that decided, if it did, including one its instance limit
504    /// blocked; otherwise none.
505    pub(crate) fn runnable_directives(&self) -> Directives {
506        match (&self.withheld, self.outcome) {
507            (Some(d), _) => (**d).clone(),
508            (None, Outcome::Allowed) => self.directives.clone(),
509            (None, _) => Directives::default(),
510        }
511    }

519    /// A block that the effective mode does not enforce (#36): the object
520    /// runs, as `allowed_audit`, or `allowed_override` if an override
521    /// session relaxed an enforcing mode, with `directives`.
522    /// `would_have_been` stays `blocked`, and the deciding rule and effects
523    /// stay, so the would-block event is still logged and sent, but
524    /// nothing was blocked, so the user is not notified.
525    pub(crate) fn relaxed(
526        mut self,
527        mode: crate::mode::EffectiveMode,
528        directives: Directives,
529        learning: bool,
530    ) -> Self {
531        let (outcome, reason_code) = if mode.by_session {
532            (Outcome::AllowedOverride, ReasonCode::OverrideSession)
533        } else {
534            (Outcome::AllowedAudit, ReasonCode::AuditModeWouldBlock)
535        };
536        self.outcome = outcome;
537        self.reason_code = reason_code;
538        self.mode = mode.mode;
539        self.learning_candidate = learning;
540        // A trust source whose grant the object now carries did not lose.
541        self.overridden
542            .retain(|r| !directives.trust_sources.iter().any(|g| g.rule == *r));
543        self.directives = directives;
544        self.withheld = None;
545        self.effects.notify = crate::model::Notify::None;
546        self.effects.notification_text = None;
547        self.effects.help_url = None;
548        self
549    }

598// For the decision trace (#41), which bounds its size.
599impl Decision {
600    /// The same decision with `overridden`, `annotations` and
601    /// `directives.trust_sources` cut to their first `n` entries, and the
602    /// number cut from each.
603    pub(crate) fn with_lists_cut(&self, n: usize) -> (Self, [usize; 3]) {
604        let mut d = self.clone();
605        let cut = [
606            d.overridden.len().saturating_sub(n),
607            d.annotations.len().saturating_sub(n),
608            d.directives.trust_sources.len().saturating_sub(n),
609        ];
610        d.overridden.truncate(n);
611        d.annotations.truncate(n);
612        d.directives.trust_sources.truncate(n);
613        (d, cut)
614    }
