1//! The typed output of an evaluation (#34).
2//!
3//! Every consumer gets the same [`Decision`]: the agent acts on it, the
4//! decision trace explains it (#41), and events embed it (#16). Its JSON
5//! form is stable: fields are always present, in declaration order, with
6//! `null` for an absent value, and the snapshot tests in
7//! `tests/snapshots/decision/` fail on any change to it.
8//!
9//! This module builds the decision of one terminal rule
10//! ([`Decision::from_rule`]) and the implicit deny for an object no rule
11//! decides ([`Decision::implicit_deny`]), both as enforce mode sees them.
12//! The pipeline in [`crate::evaluate`] combines the matching rules (#35):
13//! it adds the `audit` annotations, the rules the deciding rule overrode,
14//! and the trust-source grants. It then applies the effective mode (#36,
15//! [`crate::mode`]), which can change [`Decision::outcome`] and keeps the
16//! enforce-mode result in [`Decision::would_have_been`].
17

30/// What happened to the object.
31#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
32#[serde(rename_all = "snake_case")]
33pub enum Outcome {
34    /// It runs.
35    Allowed,
36    /// It does not run.
37    Blocked,
38    /// It runs, but would have been blocked in enforce mode.
39    AllowedAudit,
40    /// It runs, because an override session allows what would have been
41    /// blocked (#9).
42    AllowedOverride,
43    /// It was excluded by an `ignore` rule: no decision is logged.
44    Ignored,
45    /// It waits for the user to answer a prompt (#39, #11). The agent shows
46    /// [`Directives::prompt`] and applies its fallback when the prompt times
47    /// out or cannot be shown. It does not run until then.
48    PendingUser,
49}
50
51impl Outcome {
52    /// Whether the object does not run, at least for now: `blocked` or
53    /// `pending_user`. Only these are relaxed by a mode that does not
54    /// enforce.
55    #[must_use]
56    pub const fn stops(self) -> bool {
57        matches!(self, Self::Blocked | Self::PendingUser)
58    }
59}

61/// Why the decision was made. The `snake_case` name is part of the API.
62#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
63#[serde(rename_all = "snake_case")]
64pub enum ReasonCode {
65    /// An allow rule decided.
66    ExplicitAllow,
67    /// A deny rule decided.
68    ExplicitDeny,
69    /// A rule in a mandatory rule set decided (ADR 0004).
70    MandatoryDeny,
71    /// No rule decided, and unknown objects are denied (#39).
72    ImplicitDenyUnknown,
73    /// A non-enforcing mode allowed what enforce mode would block (#36).
74    AuditModeWouldBlock,
75    /// An override session allowed what would have been blocked (#9).
76    OverrideSession,
77    /// An `ignore` rule excluded the object.
78    Exclusion,
79    /// The deciding allow's per-session instance limit was reached, or
80    /// the instance count was unknown (#38).
81    InstanceLimit,
82    /// No rule decided, and the unknown-application policy runs the object
83    /// inside its containment profile (`restrict`, #39).
84    UnknownRestricted,
85    /// No rule decided, and the unknown-application policy asks the user
86    /// (`prompt`, #39).
87    UnknownPrompt,
88    /// No rule decided, and the unknown-application policy lets the object
89    /// run, reported as unhandled (`detect`, #39).
90    UnknownDetected,
91    /// No rule decided, and the unknown-application policy lets the object
92    /// run (`allow`, #39).
93    UnknownAllowed,
94    /// No rule decided, the unknown-application policy says `restrict`, and
95    /// the containment component is not installed, so the object is blocked
96    /// (#39).
97    RestrictUnavailable,
98    /// The object is one of Palisade's own pinned images, which tenant rules
99    /// cannot block (#39).
100    SelfAllow,
101    /// Evaluation ran out of its time budget, and the fallback decided
102    /// (#42).
103    EvalTimeout,
104}

106/// A rule's human short ID, `R-` and 6 to 12 digits, kept as a number and
107/// its digit count so that [`RuleRef`] stays `Copy`.
108#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
109pub struct DisplayId {
110    number: u64,
111    digits: u8,
112}
113
114impl DisplayId {
115    /// Parses `R-` followed by 6 to 12 ASCII digits, the form validation
116    /// accepts.
117    #[must_use]
118    pub fn parse(text: &str) -> Option<Self> {
119        let digits = text.strip_prefix("R-")?;
120        if !(6..=12).contains(&digits.len()) || !digits.bytes().all(|b| b.is_ascii_digit()) {
121            return None;
122        }
123        Some(Self {
124            number: digits.parse().ok()?,
125            digits: u8::try_from(digits.len()).ok()?,
126        })
127    }
128}
129
130impl fmt::Display for DisplayId {
131    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
132        write!(
133            f,
134            "R-{:0width$}",
135            self.number,
136            width = usize::from(self.digits)
137        )
138    }
139}

142    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
143        serializer.collect_str(self)
144    }

296/// What the agent and the tray do when the schedule of the allow that let

302    /// Seconds between the warning and termination.
303    pub grace_secs: u32,
304    /// When the schedule closes, in seconds since 1970-01-01T00:00:00Z, or

306    /// heartbeat.

310impl Directives {
311    /// The directives of the action of a deciding rule.
312    #[must_use]
313    pub fn of(action: &Action) -> Self {
314        let mut d = Self::default();
315        match action {
316            Action::Deny(o) => {
317                d.kill_running = o.kill_running;
318                d.request = o.allow_request.then_some(RequestDirective {
319                    template: o.request_template,
320                });
321            }
322            Action::AllowContained(c) => d.containment = Some(c.into()),
323            Action::AllowElevated(e) => d.elevation = Some(e.into()),
324            Action::Allow {}
325            | Action::Audit {}
326            | Action::TrustSource(_)
327            | Action::Ignore {}
328            | Action::Defer {} => {}
329        }
330        d
331    }
332}

369impl Decision {
370    /// The decision of `rule`, held by `set`, which `policy` references, as
371    /// enforce mode sees it. The caller has already found that the rule
372    /// matches and wins (#35).
373    ///
374    /// Returns `None` for a disabled rule, which never decides, and for an
375    /// action that does not decide by itself (`audit`, `defer`,
376    /// `trust_source`; see [`ActionKind::is_terminal`]).
377    #[must_use]
378    pub fn from_rule(
379        policy: &Policy,
380        set: &RuleSet,
381        rule: &Rule,
382        trace_id: TraceId,
383    ) -> Option<Self> {
384        if !rule.enabled {
385            return None;
386        }
387        let kind = rule.action.kind();
388        let (outcome, reason_code) = match kind {
389            ActionKind::Allow | ActionKind::AllowContained | ActionKind::AllowElevated => {
390                (Outcome::Allowed, ReasonCode::ExplicitAllow)
391            }
392            ActionKind::Deny if set.mandatory => (Outcome::Blocked, ReasonCode::MandatoryDeny),
393            ActionKind::Deny => (Outcome::Blocked, ReasonCode::ExplicitDeny),
394            ActionKind::Ignore => (Outcome::Ignored, ReasonCode::Exclusion),
395            ActionKind::Audit | ActionKind::Defer | ActionKind::TrustSource => return None,
396        };
397        Some(Self {
398            outcome,
399            would_have_been: outcome,
400            action: kind,
401            matched_rule: Some(RuleRef::new(policy, set, rule)),
402            overridden: RuleRefs::new(),
403            annotations: RuleRefs::new(),
404            mode: Mode::Enforce,
405            learning_candidate: false,
406            directives: Directives::of(&rule.action),
407            effects: rule.effects.resolve(kind),
408            reason_code,
409            scope_notes: Vec::new(),
410            unknown: None,
411            trace_id,
412            withheld: None,
413        })
414    }

533    /// Adds what the pipeline gathered besides the deciding rule (#35).
534    /// Each list is sorted by rule ID, so the decision does not depend on
535    /// the order in which rules were evaluated.

542        overridden.sort_by_key(|r| r.rule_id);

544        trust_sources.sort_by_key(|g| g.rule.rule_id);

579    /// An allow's decision turned into a block by its per-session instance
580    /// limit (#38): blocked, naming the allow. The block takes a deny's
581    /// default effects, so it is logged and sent at once and the user is
582    /// told, and keeps the allow's tags. Nothing runs, so it carries no
583    /// directives, and the trust sources are listed as overridden. If a
584    /// mode relaxes the block, the object runs with the allow's directives
585    /// ([`relaxed`](Self::relaxed)).
586    pub(crate) fn instance_limit(mut self) -> Self {
587        self.outcome = Outcome::Blocked;
588        self.would_have_been = Outcome::Blocked;
589        self.reason_code = ReasonCode::InstanceLimit;
590        let tags = core::mem::take(&mut self.effects.tags);
591        self.effects = crate::model::Effects::default().resolve(ActionKind::Deny);
592        self.effects.tags = tags;
593        if let Some(rule) = self.matched_rule {
594            self.overridden.extend(
595                self.directives
596                    .trust_sources
597                    .iter()
598                    .map(|g| g.rule)
599                    .filter(|r| *r != rule),
600            );
601            self.overridden.sort_by_key(|r| r.rule_id);
602        }
603        self.withheld = Some(Box::new(core::mem::take(&mut self.directives)));
604        self
605    }

611        match (&self.withheld, self.outcome) {

630    /// A block that the effective mode does not enforce (#36): the object
631    /// runs, as `allowed_audit`, or `allowed_override` if an override
632    /// session relaxed an enforcing mode, with `directives`.
633    /// `would_have_been` keeps the enforce outcome (`blocked`, or
634    /// `pending_user` for a prompt), and the deciding rule and effects
635    /// stay, so the would-block event is still logged and sent, but
636    /// nothing was blocked, so the user is not notified.
637    pub(crate) fn relaxed(
638        mut self,
639        mode: crate::mode::EffectiveMode,
640        directives: Directives,
641        learning: bool,
642    ) -> Self {
643        let (outcome, reason_code) = if mode.by_session {
644            (Outcome::AllowedOverride, ReasonCode::OverrideSession)
645        } else {
646            (Outcome::AllowedAudit, ReasonCode::AuditModeWouldBlock)
647        };
648        self.outcome = outcome;
649        self.reason_code = reason_code;
650        self.mode = mode.mode;
651        self.learning_candidate = learning;
652        // A trust source whose grant the object now carries did not lose.
653        self.overridden
654            .retain(|r| !directives.trust_sources.iter().any(|g| g.rule == *r));
655        self.directives = directives;
656        self.withheld = None;
657        self.effects.notify = crate::model::Notify::None;
658        self.effects.notification_text = None;
659        self.effects.help_url = None;
660        self
661    }

747// For the decision trace (#41), which bounds its size.
748impl Decision {
749    /// The same decision with `overridden`, `annotations` and
750    /// `directives.trust_sources` cut to their first `n` entries, and the
751    /// number cut from each.
752    pub(crate) fn with_lists_cut(&self, n: usize) -> (Self, [usize; 3]) {
753        let mut d = self.clone();
754        let cut = [
755            d.overridden.len().saturating_sub(n),
756            d.annotations.len().saturating_sub(n),
757            d.directives.trust_sources.len().saturating_sub(n),
758        ];
759        d.overridden.truncate(n);
760        d.annotations.truncate(n);
761        d.directives.trust_sources.truncate(n);
762        (d, cut)
763    }
