40/// Who, where and when a rule applies (#38).
41///
42/// Every field is optional. An empty scope applies to everyone, everywhere,
43/// always. A scope only ever narrows: a rule applies when its match tree
44/// matches **and** every scope on the way to it matches (the policy's, its
45/// rule set assignment's, and its own).
46#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
47#[serde(deny_unknown_fields)]
48pub struct Scope {
49    /// Users and groups.
50    #[serde(default, skip_serializing_if = "Option::is_none")]
51    pub principals: Option<RefList<PrincipalRef>>,
52    /// Devices.
53    #[serde(default, skip_serializing_if = "Option::is_none")]
54    pub devices: Option<RefList<DeviceRef>>,
55    /// The device's network location.
56    #[serde(default, skip_serializing_if = "Option::is_none")]
57    pub network: Option<RefList<NetworkRef>>,
58    /// The session type and, on a multi-session host, the connecting client.
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub sessions: Option<SessionFilter>,
61    /// When the scope is active, written in place. At most one of
62    /// `schedule` and `schedule_set`.
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub schedule: Option<Schedule>,
65    /// When the scope is active, as a schedule from the library (#37).
66    #[serde(default, skip_serializing_if = "Option::is_none")]
67    pub schedule_set: Option<SetRef>,
68    /// How the dimensions that are set combine: `all` (the default) needs
69    /// every one to match, `any` needs one.
70    #[serde(default, skip_serializing_if = "Combine::is_all")]
71    pub combine: Combine,
72    /// Apply when the scope does **not** match.
73    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
74    pub negate: bool,
75    /// Whose token `principals` is tested against.
76    #[serde(default, skip_serializing_if = "ScopeSubject::is_process_user")]
77    pub subject: ScopeSubject,
78    /// Only on the scope of a rule that lets an object run: at most this
79    /// many instances per session. `0`, the default, means no limit.
80    #[serde(default, skip_serializing_if = "is_zero")]
81    pub max_instances_per_session: u32,
82}

410/// One weekly window.
411#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
412#[serde(deny_unknown_fields)]
413pub struct Window {
414    /// The days the window starts on. At least one, each once.
415    #[schemars(length(min = 1, max = 7))]
416    pub days: Vec<Weekday>,
417    /// When the window starts, `HH:MM`.
418    pub start: TimeOfDay,
419    /// When it ends, `HH:MM`. At or before `start` means the next day.
420    /// Exactly one of `end` and `duration_mins`.
421    #[serde(default, skip_serializing_if = "Option::is_none")]
422    pub end: Option<TimeOfDay>,
423    /// How long it lasts, in minutes, from 1 to 10080.
424    #[serde(default, skip_serializing_if = "Option::is_none")]
425    #[schemars(range(min = 1, max = 10_080))]
426    pub duration_mins: Option<u32>,
427}
428
429impl Window {
430    /// The window's length in minutes, if it is well formed: at least one
431    /// day, each once, a start before `24:00`, exactly one of `end` and
432    /// `duration_mins`, and from 1 minute to 7 days.
433    #[must_use]
434    pub fn minutes(&self) -> Option<u32> {
435        let mut seen = 0u8;
436        for d in &self.days {
437            let bit = 1 << (*d as u8);
438            if seen & bit != 0 {
439                return None;
440            }
441            seen |= bit;
442        }
443        if seen == 0 || self.start.0 >= 1440 {
444            return None;
445        }
446        let minutes = match (self.end, self.duration_mins) {
447            (Some(end), None) => {
448                let (s, e) = (u32::from(self.start.0), u32::from(end.0));
449                if e > s { e - s } else { 1440 - s + e }
450            }
451            (None, Some(d)) => d,
452            _ => return None,
453        };
454        (1..=limits::MAX_WINDOW_MINUTES)
455            .contains(&minutes)
456            .then_some(minutes)
457    }
458}

498/// A wall-clock time, `HH:MM` from `00:00` to `23:59`, held as minutes
499/// since midnight. `24:00` is also accepted, as the end of a day.
500#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
501pub struct TimeOfDay(u16);
502
503impl TimeOfDay {
504    /// The time `minutes` after midnight, up to 1440 (`24:00`).
505    #[must_use]
506    pub const fn from_minutes(minutes: u16) -> Option<Self> {
507        if minutes <= 1440 {
508            Some(Self(minutes))
509        } else {
510            None
511        }
512    }
513
514    /// Minutes since midnight.
515    #[must_use]
516    pub const fn minutes(self) -> u16 {
517        self.0
518    }

521impl core::str::FromStr for TimeOfDay {
522    type Err = &'static str;
523
524    fn from_str(s: &str) -> Result<Self, Self::Err> {
525        const ERR: &str = "expected a time `HH:MM` from 00:00 to 24:00";
526        let b = s.as_bytes();
527        if b.len() != 5 || b[2] != b':' || ![b[0], b[1], b[3], b[4]].iter().all(u8::is_ascii_digit)
528        {
529            return Err(ERR);
530        }
531        let h = u16::from(b[0] - b'0') * 10 + u16::from(b[1] - b'0');
532        let m = u16::from(b[3] - b'0') * 10 + u16::from(b[4] - b'0');
533        if m > 59 || h > 24 || (h == 24 && m != 0) {
534            return Err(ERR);
535        }
536        Ok(Self(h * 60 + m))
537    }
538}

543    }

546/// A calendar date, `YYYY-MM-DD`, from year 0001 to 9999.
547#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
548pub struct Date(pub civil::Date);
549
550impl core::str::FromStr for Date {
551    type Err = &'static str;
552
553    fn from_str(s: &str) -> Result<Self, Self::Err> {
554        const ERR: &str = "expected a date `YYYY-MM-DD`";
555        let b = s.as_bytes();
556        let digits = |r: core::ops::Range<usize>| b[r].iter().all(u8::is_ascii_digit);
557        if b.len() != 10 || b[4] != b'-' || b[7] != b'-' || !digits(0..4) || !digits(5..7) {
558            return Err(ERR);
559        }
560        if !digits(8..10) {
561            return Err(ERR);
562        }
563        let num = |r: core::ops::Range<usize>| s[r].parse::<i16>().map_err(|_| ERR);
564        let year = num(0..4)?;
565        let month = i8::try_from(num(5..7)?).map_err(|_| ERR)?;
566        let day = i8::try_from(num(8..10)?).map_err(|_| ERR)?;
567        if year < 1 {
568            return Err(ERR);
569        }
570        civil::Date::new(year, month, day)
571            .map(Self)
572            .map_err(|_| ERR)
573    }
574}
575
576impl fmt::Display for Date {
577    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
578        write!(
579            f,
580            "{:04}-{:02}-{:02}",
581            self.0.year(),
582            self.0.month(),
583            self.0.day()
584        )
585    }
586}

592                serializer.collect_str(self)
593            }
594        }
595
596        impl<'de> Deserialize<'de> for $ty {
597            fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
598                struct Visitor;
599                impl serde::de::Visitor<'_> for Visitor {
600                    type Value = $ty;
601                    fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
602                        f.write_str($expecting)
603                    }
604                    fn visit_str<E: serde::de::Error>(self, v: &str) -> Result<$ty, E> {
605                        v.parse().map_err(E::custom)
606                    }
607                }
608                deserializer.deserialize_str(Visitor)
609            }
610        }
611
612        impl JsonSchema for $ty {
613            fn schema_name() -> std::borrow::Cow<'static, str> {
614                stringify!($ty).into()
615            }
616
617            fn json_schema(_: &mut schemars::SchemaGenerator) -> schemars::Schema {
618                schemars::json_schema!({
619                    "description": $description,
620                    "type": "string",
621                    "pattern": $pattern
622                })
623            }
624        }
625    };
626}

641/// An inclusive range of dates. `to` unset means the single day `from`.
642#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
643#[serde(deny_unknown_fields)]
644pub struct DateRange {
645    /// The first day.
646    pub from: Date,
647    /// The last day, on or after `from`.
648    #[serde(default, skip_serializing_if = "Option::is_none")]
649    pub to: Option<Date>,
650}
651
652impl DateRange {
653    /// The last day.
654    #[must_use]
655    pub fn last(&self) -> Date {
656        self.to.unwrap_or(self.from)
657    }
658
659    /// Whether the range does not end before it starts.
660    #[must_use]
661    pub fn is_ordered(&self) -> bool {
662        self.from <= self.last()
663    }
664
665    /// Whether `date` is in the range.
666    #[must_use]
667    pub fn contains(&self, date: civil::Date) -> bool {
668        self.from.0 <= date && date <= self.last().0
669    }
670}
