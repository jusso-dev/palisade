1//! Schedules: when a scope is active (#38).
2//!
3//! A [`CompiledSchedule`] is evaluated at an instant (seconds since the
4//! Unix epoch, from the agent's trusted time) in one time zone, chosen by
5//! the schedule's `timezone_basis`:
6//!
7//! - `utc`: Coordinated Universal Time.
8//! - `device_local`: the device's zone ([`ScopeFacts::device_time_zone`]).
9//! - `user_local`: the session user's zone
10//!   ([`ScopeFacts::session_time_zone`]) if the agent could read it, and
11//!   otherwise the device's, with the note `timezone_fallback`.
12//!
13//! Zones are IANA names, resolved with the time zone database bundled
14//! into the binary, so evaluation never reads the system.
15//!
16//! # Windows
17//!
18//! A window on day `D` from `start` to `end` is active from the instant
19//! that is `start` on `D` to the instant that is `end` on `D` (or on the
20//! next day, if `end` is not after `start`), including the first and
21//! excluding the second. `24:00` is midnight at the end of the day. A
22//! window with `duration_mins` lasts that many minutes of elapsed time from
23//! its start, whatever the clocks do.
24//!
25//! Local times are turned into instants with the "compatible" rule of the
26//! ECMAScript Temporal proposal: a time that occurs twice, when clocks go
27//! back, is its first occurrence; a time that is skipped, when clocks go
28//! forward, is moved forward by the length of the gap (02:30 on a day that
29//! jumps from 02:00 to 03:00 is 03:30). So a window `01:00`–`03:00` on the
30//! day clocks go back lasts three hours of elapsed time, and on the day
31//! they go forward one hour.
32//!
33//! # Dates
34//!
35//! `dates` and `except_dates` test the local date of the instant being
36//! decided, not the day a window started on. With `dates`, the schedule is
37//! only active on those dates; on an `except_dates` date it is never
38//! active.
39

52/// A schedule ready to evaluate.
53#[derive(Clone, Debug, PartialEq, Eq)]
54pub struct CompiledSchedule {
55    /// Some part of the schedule was not well formed. Such a schedule is
56    /// never trusted to say when it is active: it reads as an untrusted
57    /// clock does.
58    invalid: bool,
59    basis: TimeZoneBasis,
60    windows: Vec<CWindow>,
61    dates: Vec<(Date, Date)>,
62    except: Vec<(Date, Date)>,
63    on_close: OnWindowClose,
64    grace_secs: u32,
65}
66
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68struct CWindow {
69    /// Bit `n` is set for weekday `n`, from Monday.
70    days: u8,
71    /// Minutes after midnight.
72    start: u16,
73    end: End,
74}
75
76#[derive(Clone, Copy, Debug, PartialEq, Eq)]
77enum End {
78    /// A wall-clock time, minutes after midnight, on the start day or the
79    /// next day.
80    At { minutes: u16, next_day: bool },
81    /// Elapsed minutes after the start.
82    After(u32),
83}
84
85const fn day_bit(day: Weekday) -> u8 {
86    1 << (day as u8)
87}

89impl CompiledSchedule {
90    /// Compiles a schedule. `default_basis` is the policy's
91    /// `settings.time_zone_basis`, used when the schedule sets none.
92    ///
93    /// Validation rejects a schedule that is not well formed, but a
94    /// library schedule (#37) arrives unvalidated. If a window is not well
95    /// formed ([`Window::minutes`](super::Window::minutes) is `None`), a
96    /// date range ends before it starts, or a list is over its limit, the
97    /// schedule is compiled as invalid: it then evaluates as a schedule
98    /// read without trusted time does, keeping permitting rules inactive
99    /// and restricting rules active (`fact_malformed`).
100    #[must_use]
101    pub fn compile(schedule: &Schedule, default_basis: TimeZoneBasis) -> Self {
102        use super::limits::{MAX_DATE_RANGES, MAX_GRACE_SECS, MAX_WINDOWS};
103        let invalid = schedule.windows.len() > MAX_WINDOWS
104            || schedule.dates.len() > MAX_DATE_RANGES
105            || schedule.except_dates.len() > MAX_DATE_RANGES
106            || schedule.windows.iter().any(|w| w.minutes().is_none())
107            || !schedule.dates.iter().all(super::DateRange::is_ordered)
108            || !schedule
109                .except_dates
110                .iter()
111                .all(super::DateRange::is_ordered)
112            || schedule.grace_secs.is_some_and(|g| g > MAX_GRACE_SECS);
113        let windows = schedule
114            .windows
115            .iter()
116            .filter(|w| w.minutes().is_some())
117            .map(|w| CWindow {
118                days: w.days.iter().fold(0, |acc, d| acc | day_bit(*d)),
119                start: w.start.minutes(),
120                end: match (w.end, w.duration_mins) {
121                    (Some(end), _) => End::At {
122                        minutes: end.minutes(),
123                        next_day: end.minutes() <= w.start.minutes(),
124                    },
125                    (None, Some(d)) => End::After(d),
126                    (None, None) => End::After(0),
127                },
128            })
129            .collect();
130        let range = |r: &super::DateRange| (r.from.0, r.last().0);
131        Self {
132            invalid,
133            basis: schedule.timezone_basis.unwrap_or(default_basis),
134            windows,
135            dates: schedule.dates.iter().map(range).collect(),
136            except: schedule.except_dates.iter().map(range).collect(),
137            on_close: schedule.on_window_close,
138            grace_secs: schedule.grace_secs.unwrap_or(DEFAULT_GRACE_SECS),
139        }
140    }
141
142    /// Whether every part of the schedule was well formed.
143    #[must_use]
144    pub const fn is_valid(&self) -> bool {
145        !self.invalid
146    }
147
148    /// What happens to running processes when the schedule closes.
149    #[must_use]
150    pub const fn on_window_close(&self) -> OnWindowClose {
151        self.on_close
152    }

154    /// The grace period, in seconds.
155    #[must_use]
156    pub const fn grace_secs(&self) -> u32 {
157        self.grace_secs
158    }

160    /// The time zone the schedule is read in, from the facts. `Err` with
161    /// the notes if it is unknown.
162    pub fn zone(&self, facts: &ScopeFacts) -> Result<(TimeZone, ScopeNotes), ScopeNotes> {
163        let mut notes = ScopeNotes::default();
164        let name = match self.basis {
165            TimeZoneBasis::Utc => return Ok((TimeZone::UTC, notes)),
166            TimeZoneBasis::DeviceLocal => facts.device_time_zone.as_deref(),
167            TimeZoneBasis::UserLocal => match facts.session_time_zone.as_deref() {
168                Some(zone) => Some(zone),
169                None => {
170                    notes.insert(ScopeNote::TimezoneFallback);
171                    facts.device_time_zone.as_deref()
172                }
173            },
174        };
175        match name.map(TimeZone::get) {
176            Some(Ok(tz)) => Ok((tz, notes)),
177            Some(Err(_)) => {
178                notes.insert(ScopeNote::FactMalformed);
179                Err(notes)
180            }
181            None => {
182                notes.insert(ScopeNote::FactMissing);
183                Err(notes)
184            }
185        }
186    }
187
188    /// Evaluates the schedule for one rule. `positive` is whether the
189    /// schedule being active makes the rule apply (it is not under
190    /// `negate`). Without trusted time or a known zone, the schedule reads
191    /// as whatever keeps a permitting rule inactive and a restricting rule
192    /// active, with `clock_untrusted`.
193    pub(crate) fn eval(&self, facts: &ScopeFacts, positive: bool, bias: Bias) -> (Tri, ScopeNotes) {
194        let restrictive = |mut notes: ScopeNotes| {
195            notes.insert(ScopeNote::ClockUntrusted);
196            (bias.favours(positive), notes)
197        };
198        if self.invalid {
199            let mut notes = ScopeNotes::default();
200            notes.insert(ScopeNote::FactMalformed);
201            return (bias.favours(positive), notes);
202        }
203        let Some(time) = facts.time.filter(|t| t.trusted) else {
204            let mut notes = ScopeNotes::default();
205            if facts.time.is_none() {
206                notes.insert(ScopeNote::FactMissing);
207            }
208            return restrictive(notes);
209        };
210        match self.zone(facts) {
211            Ok((tz, notes)) => match self.is_active(time.unix_secs, &tz) {
212                Some(active) => (Tri::from_bool(active), notes),
213                None => {
214                    let mut notes = notes;
215                    notes.insert(ScopeNote::FactMalformed);
216                    restrictive(notes)
217                }
218            },
219            Err(notes) => restrictive(notes),
220        }
221    }

223    /// Whether the schedule is active at `unix_secs` in `tz`. `None` if the
224    /// schedule is invalid, or the instant is outside the years -9999 to
225    /// 9999.
226    #[must_use]
227    pub fn is_active(&self, unix_secs: i64, tz: &TimeZone) -> Option<bool> {
228        if self.invalid {
229            return None;
230        }
231        let date = local(unix_secs, tz)?.date();
232        if !self.date_ok(date) {
233            return Some(false);
234        }
235        if self.windows.is_empty() {
236            return Some(true);
237        }
238        Some(self.window_end(unix_secs, tz, date).is_some())
239    }
240
241    /// The instant the schedule next stops being active, after
242    /// `unix_secs`, if it is active then and stops within 64 window or
243    /// date boundaries. The agent re-evaluates on every heartbeat, so a
244    /// later close is still acted on.
245    #[must_use]
246    pub fn closes_at(&self, unix_secs: i64, tz: &TimeZone) -> Option<i64> {
247        if self.is_active(unix_secs, tz) != Some(true) {
248            return None;
249        }
250        let mut cur = unix_secs;
251        for _ in 0..64 {
252            let date = local(cur, tz)?.date();
253            let window = if self.windows.is_empty() {
254                None
255            } else {
256                self.window_end(cur, tz, date)
257            };
258            let day = self
259                .date_boundary(date)
260                .and_then(|d| resolve(d.to_datetime(Time::midnight()), tz));
261            let next = match (window, day) {
262                (Some(a), Some(b)) => a.min(b),
263                (a, b) => a.or(b)?,
264            };
265            if next <= cur {
266                return None;
267            }
268            cur = next;
269            if self.is_active(cur, tz)? {
270                continue;
271            }
272            return Some(cur);
273        }
274        None
275    }
276
277    /// Whether `date` passes `dates` and `except_dates`.
278    fn date_ok(&self, date: Date) -> bool {
279        let within = |r: &(Date, Date)| r.0 <= date && date <= r.1;
280        (self.dates.is_empty() || self.dates.iter().any(within)) && !self.except.iter().any(within)
281    }
282
283    /// The first date after `date` on which the date test may change, if
284    /// there is one.
285    fn date_boundary(&self, date: Date) -> Option<Date> {
286        self.dates
287            .iter()
288            .chain(&self.except)
289            .filter_map(|&(from, last)| {
290                if from > date {
291                    Some(from)
292                } else if last >= date {
293                    last.tomorrow().ok()
294                } else {
295                    None
296                }
297            })
298            .min()
299    }
300
301    /// The latest end of the windows that contain `t`, whose local date is
302    /// `date`, or `None` if none does.
303    fn window_end(&self, t: i64, tz: &TimeZone, date: Date) -> Option<i64> {
304        let mut best: Option<i64> = None;
305        for w in &self.windows {
306            // A window starts at most 7 days and a few hours of clock
307            // changes before `t`. A start on the next local date can still
308            // be before `t` when clocks go back over midnight.
309            for back in -1i64..=8 {
310                let day = if back < 0 {
311                    date.tomorrow()
312                } else {
313                    date.checked_sub(jiff::Span::new().days(back))
314                };
315                let Ok(day) = day else { continue };
316                if w.days & day_bit(Weekday::of(day)) == 0 {
317                    continue;
318                }
319                let Some((start, end)) = w.span(day, tz) else {
320                    continue;
321                };
322                if start <= t && t < end {
323                    best = Some(best.map_or(end, |b| b.max(end)));
324                }
325            }
326        }
327        best
328    }
329}

331impl CWindow {
332    /// The instants the window starting on `day` begins and ends.
333    fn span(self, day: Date, tz: &TimeZone) -> Option<(i64, i64)> {
334        let start = at(day, self.start, tz)?;
335        let end = match self.end {
336            End::At { minutes, next_day } => {
337                let end_day = if next_day { day.tomorrow().ok()? } else { day };
338                at(end_day, minutes, tz)?
339            }
340            End::After(mins) => start.checked_add(i64::from(mins) * 60)?,
341        };
342        Some((start, end))
343    }
344}

346/// The local date and time of an instant.
347fn local(unix_secs: i64, tz: &TimeZone) -> Option<DateTime> {
348    Some(tz.to_datetime(Timestamp::from_second(unix_secs).ok()?))
349}
350
351/// The instant that is `minutes` after midnight on `day` (1440 is the next
352/// midnight), by the compatible rule.
353fn at(day: Date, minutes: u16, tz: &TimeZone) -> Option<i64> {
354    let (day, minutes) = if minutes >= 1440 {
355        (day.tomorrow().ok()?, minutes - 1440)
356    } else {
357        (day, minutes)
358    };
359    let h = i8::try_from(minutes / 60).ok()?;
360    let m = i8::try_from(minutes % 60).ok()?;
361    resolve(day.to_datetime(Time::new(h, m, 0, 0).ok()?), tz)
362}
363
364/// The instant of a local time by the compatible rule of the Temporal
365/// proposal, which jiff implements: the earlier of two, and for a skipped
366/// time, the local time read with the offset in force before the gap.
367fn resolve(local: DateTime, tz: &TimeZone) -> Option<i64> {
368    tz.to_ambiguous_timestamp(local)
369        .compatible()
370        .ok()
371        .map(Timestamp::as_second)
372}

374impl Bias {
375    /// The value that makes a rule of this bias apply in a position of
376    /// this polarity: more for a restricting rule, less for a permitting
377    /// one.
378    pub(crate) const fn favours(self, positive: bool) -> Tri {
379        Tri::from_bool(matches!(self, Self::Restrictive) == positive)
380    }
