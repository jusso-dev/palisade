12/// An inclusive range of addresses of one family.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub struct IpRange {
15    first: u128,
16    last: u128,
17    v6: bool,
18}
19
20impl IpRange {
21    /// Parses an address block, `10.0.0.0/8` or `2001:db8::/32`. The
22    /// address must be the block's first address (no host bits set).
23    pub fn cidr(s: &str) -> Result<Self, String> {
24        let (addr, len) = s
25            .split_once('/')
26            .ok_or_else(|| format!("`{s}` is not an address block such as `10.0.0.0/8`"))?;
27        let addr: IpAddr = addr
28            .parse()
29            .map_err(|_| format!("`{addr}` is not an IP address"))?;
30        let (value, bits, v6) = split(addr);
31        let len: u32 = (!len.is_empty() && len.bytes().all(|b| b.is_ascii_digit()))
32            .then(|| len.parse().ok())
33            .flatten()
34            .filter(|l| *l <= bits)
35            .ok_or_else(|| format!("`/{len}` is not a prefix length from 0 to {bits}"))?;
36        let host_bits = bits - len;
37        let mask: u128 = if host_bits == 128 {
38            u128::MAX
39        } else {
40            (1u128 << host_bits) - 1
41        };
42        if value & mask != 0 {
43            return Err(format!(
44                "`{s}` has host bits set; the block starts at its first address"
45            ));
46        }
47        Ok(Self {
48            first: value,
49            last: value | mask,
50            v6,
51        })
52    }
53
54    /// Parses an inclusive range, `10.0.0.10-10.0.0.99`. Both ends are the
55    /// same family, and the first is not after the last.
56    pub fn range(s: &str) -> Result<Self, String> {
57        let (a, b) = s
58            .split_once('-')
59            .ok_or_else(|| format!("`{s}` is not a range such as `10.0.0.1-10.0.0.9`"))?;
60        let parse = |t: &str| {
61            t.parse::<IpAddr>()
62                .map_err(|_| format!("`{t}` is not an IP address"))
63        };
64        let (a, b) = (split(parse(a)?), split(parse(b)?));
65        if a.2 != b.2 {
66            return Err(format!("`{s}` mixes IPv4 and IPv6"));
67        }
68        if a.0 > b.0 {
69            return Err(format!("`{s}` ends before it starts"));
70        }
71        Ok(Self {
72            first: a.0,
73            last: b.0,
74            v6: a.2,
75        })
76    }
77
78    /// A block or a range, whichever `s` is.
79    pub fn block_or_range(s: &str) -> Result<Self, String> {
80        if s.contains('/') {
81            Self::cidr(s)
82        } else {
83            Self::range(s)
84        }
85    }

87    /// Whether `addr` is in the range. An IPv4 address and the
88    /// IPv4-mapped IPv6 address `::ffff:a.b.c.d` are the same address, so
89    /// either form is in an IPv4 range that holds it and in an IPv6 range
90    /// that holds its mapped form (`::ffff:10.0.0.0/104` is `10.0.0.0/8`,
91    /// and `::/0` holds every address).
92    #[must_use]
93    pub fn contains(&self, addr: IpAddr) -> bool {
94        let v4 = match addr {
95            IpAddr::V4(a) => Some(a),
96            IpAddr::V6(a) => a.to_ipv4_mapped(),
97        };
98        let within = |a: IpAddr| {
99            let (value, _, v6) = split(a);
100            v6 == self.v6 && self.first <= value && value <= self.last
101        };
102        match v4 {
103            Some(a) => within(IpAddr::V4(a)) || within(IpAddr::V6(a.to_ipv6_mapped())),
104            None => within(addr),
105        }
106    }
107}

109/// An address as a number, its width in bits, and whether it is IPv6.
110fn split(addr: IpAddr) -> (u128, u32, bool) {
111    match addr {
112        IpAddr::V4(a) => (u128::from(u32::from(a)), 32, false),
113        IpAddr::V6(a) => (u128::from(a), 128, true),
114    }
115}

117/// Parses a hardware address: six pairs of hex digits separated by `-` or
118/// `:`, in either case.
119pub fn mac(s: &str) -> Result<[u8; 6], String> {
120    let err = || format!("`{s}` is not a hardware address such as `00-1a-2b-3c-4d-5e`");
121    let sep = if s.contains(':') { ':' } else { '-' };
122    let mut out = [0u8; 6];
123    let mut parts = s.split(sep);
124    for byte in &mut out {
125        let part = parts.next().ok_or_else(err)?;
126        if part.len() != 2 || !part.bytes().all(|b| b.is_ascii_hexdigit()) {
127            return Err(err());
128        }
129        *byte = u8::from_str_radix(part, 16).map_err(|_| err())?;
130    }
131    if parts.next().is_some() {
132        return Err(err());
133    }
134    Ok(out)
135}

137/// The canonical form of a DNS suffix: lowercase ASCII, without a leading
138/// or trailing dot. Each label is 1 to 63 letters, digits or hyphens, not
139/// starting or ending with a hyphen, and the whole is at most 253 bytes.
140pub fn dns_suffix(s: &str) -> Result<String, String> {
141    let t = s.strip_prefix('.').unwrap_or(s);
142    let t = t.strip_suffix('.').unwrap_or(t);
143    let ok = !t.is_empty()
144        && t.len() <= 253
145        && t.split('.').all(|label| {
146            !label.is_empty()
147                && label.len() <= 63
148                && !label.starts_with('-')
149                && !label.ends_with('-')
150                && label
151                    .bytes()
152                    .all(|b| b.is_ascii_alphanumeric() || b == b'-')
153        });
154    if ok {
155        Ok(t.to_ascii_lowercase())
156    } else {
157        Err(format!(
158            "`{s}` is not a DNS name such as `corp.example.org`"
159        ))
160    }
161}

163/// Whether the fact `name` is `suffix` or a name below it. `suffix` is
164/// canonical ([`dns_suffix`]).
165#[must_use]
166pub fn dns_within(name: &str, suffix: &str) -> bool {
167    let name = name.strip_suffix('.').unwrap_or(name).to_ascii_lowercase();
168    name == suffix
169        || name
170            .strip_suffix(suffix)
171            .is_some_and(|head| head.ends_with('.'))
172}

174/// Checks a free-text scope value: not empty, at most
175/// [`MAX_SCOPE_VALUE_BYTES`] bytes, no control characters, and not
176/// starting or ending with whitespace.
177pub fn plain(s: &str) -> Result<(), String> {
178    if s.is_empty() {
179        Err("must not be empty".to_owned())
180    } else if s.len() > MAX_SCOPE_VALUE_BYTES {
181        Err(format!(
182            "{} bytes; at most {MAX_SCOPE_VALUE_BYTES} are allowed",
183            s.len()
184        ))
185    } else if s.chars().any(|c| c.is_control() || is_bidi_control(c)) {
186        Err("must not contain control or bidirectional formatting characters".to_owned())
187    } else if s.trim() != s {
188        Err("must not start or end with whitespace".to_owned())
189    } else {
190        Ok(())
191    }
192}

194/// Unicode bidirectional formatting characters, which can make a value
195/// read differently from what it contains.
196const fn is_bidi_control(c: char) -> bool {
197    matches!(c, '\u{061c}' | '\u{200e}' | '\u{200f}' | '\u{202a}'..='\u{202e}' | '\u{2066}'..='\u{2069}')
198}

200/// A wildcard: `*` matches any run of characters (including none) and `?`
201/// exactly one. Compared without case (Unicode simple case folding, after
202/// NFC), unless built case-sensitive.
203#[derive(Clone, Debug, PartialEq, Eq)]
204pub struct Wildcard {
205    pattern: Vec<char>,
206    case_sensitive: bool,
207}
208
209impl Wildcard {
210    /// Compiles a pattern. It must pass [`plain`].
211    pub fn new(pattern: &str, case_sensitive: bool) -> Result<Self, String> {
212        plain(pattern)?;
213        Ok(Self {
214            pattern: fold(pattern, case_sensitive).chars().collect(),
215            case_sensitive,
216        })
217    }
218
219    /// Whether `text` matches the whole pattern.
220    ///
221    /// The usual greedy matcher with one backtrack point: on a mismatch it
222    /// retries from the last `*`, consuming one more character. That is
223    /// at most `pattern × text` steps, and both are bounded.
224    #[must_use]
225    pub fn matches(&self, text: &str) -> bool {
226        let text: Vec<char> = fold(text, self.case_sensitive).chars().collect();
227        let p = &self.pattern;
228        let (mut ti, mut pi) = (0usize, 0usize);
229        let mut star: Option<(usize, usize)> = None;
230        while ti < text.len() {
231            match p.get(pi) {
232                Some('*') => {
233                    star = Some((pi, ti));
234                    pi += 1;
235                }
236                Some(&c) if c == '?' || c == text[ti] => {
237                    pi += 1;
238                    ti += 1;
239                }
240                _ => match star {
241                    Some((sp, st)) => {
242                        pi = sp + 1;
243                        ti = st + 1;
244                        star = Some((sp, st + 1));
245                    }
246                    None => return false,
247                },
248            }
249        }
250        p[pi..].iter().all(|&c| c == '*')
251    }
252}

254/// A distinguished name as its relative names, each trimmed and folded to
255/// lowercase, from the object up. `None` if `s` is not one.
256///
257/// Commas escaped with `\` stay inside a name. Directory names compare
258/// without case, so `OU=Finance,DC=Corp` equals `ou=finance, dc=corp`.
259#[must_use]
260pub fn dn(s: &str) -> Option<Vec<String>> {
261    // A directory name fact may be longer than a policy value; the
262    // directory allows far more than 512 bytes.
263    if s.is_empty() || s.len() > MAX_DN_BYTES || s.chars().any(char::is_control) {
264        return None;
265    }
266    let mut rdns = Vec::new();
267    let mut current = String::new();
268    let mut chars = s.chars();
269    while let Some(c) = chars.next() {
270        match c {
271            '\\' => {
272                current.push(c);
273                current.push(chars.next()?);
274            }
275            ',' => rdns.push(core::mem::take(&mut current)),
276            _ => current.push(c),
277        }
278    }
279    rdns.push(current);
280    rdns.into_iter()
281        .map(|rdn| {
282            let (attr, value) = rdn.split_once('=')?;
283            let (attr, value) = (attr.trim(), value.trim());
284            let attr_ok = !attr.is_empty()
285                && attr
286                    .bytes()
287                    .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'.');
288            (attr_ok && !value.is_empty())
289                .then(|| format!("{}={}", attr.to_ascii_lowercase(), fold_lower(value)))
290        })
291        .collect()
292}
293
294/// The longest distinguished name read, in bytes.
295pub const MAX_DN_BYTES: usize = 16 * 1024;
296
297/// Lowercase, after NFC, for directory names.
298fn fold_lower(s: &str) -> String {
299    use unicode_normalization::UnicodeNormalization;
300    s.nfc().flat_map(char::to_lowercase).collect()
301}

303/// Whether the object named `object` is directly in the OU `ou`, or with
304/// `sub` anywhere below it. Both are parsed with [`dn`].
305#[must_use]
306pub fn in_ou(object: &[String], ou: &[String], sub: bool) -> bool {
307    let Some((_, parent)) = object.split_first() else {
308        return false;
309    };
310    if sub {
311        parent.ends_with(ou)
312    } else {
313        parent == ou
314    }
315}

327    #[test]
328    fn cidr_and_ranges() {
329        let block = IpRange::cidr("10.20.0.0/16").unwrap();
330        assert!(block.contains("10.20.255.1".parse().unwrap()));
331        assert!(!block.contains("10.21.0.0".parse().unwrap()));
332        assert!(block.contains("::ffff:10.20.1.1".parse().unwrap()));
333        assert!(!block.contains("2001:db8::1".parse().unwrap()));
334        assert!(IpRange::cidr("10.20.0.1/16").is_err());
335        assert!(IpRange::cidr("10.0.0.0/33").is_err());
336        assert!(IpRange::cidr("10.0.0.0/").is_err());
337        assert!(
338            IpRange::cidr("0.0.0.0/0")
339                .unwrap()
340                .contains("1.2.3.4".parse().unwrap())
341        );
342        let all6 = IpRange::cidr("::/0").unwrap();
343        assert!(all6.contains("2001:db8::1".parse().unwrap()));
344        let v6 = IpRange::cidr("2001:db8::/32").unwrap();
345        assert!(v6.contains("2001:db8:ffff::1".parse().unwrap()));
346        let range = IpRange::range("10.0.0.10-10.0.0.99").unwrap();
347        assert!(range.contains("10.0.0.10".parse().unwrap()));
348        assert!(range.contains("10.0.0.99".parse().unwrap()));
349        assert!(!range.contains("10.0.0.100".parse().unwrap()));
350        assert!(IpRange::range("10.0.0.9-10.0.0.1").is_err());
351        assert!(IpRange::range("10.0.0.1-::1").is_err());
352        // IPv4-mapped forms are the same addresses.
353        let mapped = IpRange::cidr("::ffff:10.20.0.0/112").unwrap();
354        assert!(mapped.contains("10.20.3.4".parse().unwrap()));
355        assert!(mapped.contains("::ffff:10.20.3.4".parse().unwrap()));
356        assert!(!mapped.contains("10.21.0.0".parse().unwrap()));
357        let mapped_range = IpRange::range("::ffff:10.0.0.1-::ffff:10.0.0.9").unwrap();
358        assert!(mapped_range.contains("10.0.0.1".parse().unwrap()));
359        assert!(mapped_range.contains("::ffff:10.0.0.9".parse().unwrap()));
360        assert!(all6.contains("::ffff:10.20.1.1".parse().unwrap()));
361        assert!(all6.contains("10.20.1.1".parse().unwrap()));
362    }

365    fn macs_and_suffixes() {
366        assert_eq!(
367            mac("00-1A-2b-3c-4d-5e").unwrap(),
368            [0, 0x1a, 0x2b, 0x3c, 0x4d, 0x5e]

370        assert_eq!(
371            mac("00:1a:2b:3c:4d:5e").unwrap(),

373        );
374        assert!(mac("00-1a-2b-3c-4d").is_err());
375        assert!(mac("00-1a-2b-3c-4d-5e-6f").is_err());
376        assert!(mac("0-1a-2b-3c-4d-5e").is_err());
377        assert!(mac("+0-1a-2b-3c-4d-5e").is_err());
378        assert!(mac("00-+a-2b-3c-4d-5e").is_err());
379        assert_eq!(
380            dns_suffix(".Corp.Example.org.").unwrap(),
