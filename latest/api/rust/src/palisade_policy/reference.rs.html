1//! The reference policy of the performance budget (#42), for the
2//! benchmarks and the budget test. Not part of the API.
3//!
4//! The reference policy has 10,000 rules, plus a hash set of 100,000
5//! entries (filed as 4,000 rules of 25 SHA-256 values each, the most a
6//! criterion holds), 500 publisher rules, 200 path rules and 50 rules with a
7//! regular expression leaf. [`Size::Million`] has a hash set of 1,000,000
8//! entries instead: 40,000 rules.
9//!
10//! The objects it decides are a hash hit, a publisher hit, a path wildcard
11//! hit, an unknown object (the implicit deny) and a mandatory deny.
12

20/// The size of the hash set.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub enum Size {
23    /// 100,000 hashes.
24    Reference,
25    /// 1,000,000 hashes.
26    Million,
27}
28
29impl Size {
30    const fn hashes(self) -> u32 {
31        match self {
32            Self::Reference => 100_000,
33            Self::Million => 1_000_000,
34        }
35    }
36}
37
38/// The rules of the reference policy besides the hash set.
39pub const OTHER_RULES: u32 = 10_000;
40/// Publisher rules.
41pub const PUBLISHERS: u32 = 500;
42/// Path rules.
43pub const PATHS: u32 = 200;
44/// Rules with a regular expression leaf.
45pub const REGEXES: u32 = 50;
46
47/// A rule ID: a UUIDv7 whose sequence part is `n`.
48fn id(kind: u16, n: u32) -> String {
49    format!(
50        "01936b2e-{:04x}-7{:03x}-9f21-{:012x}",
51        kind,
52        n >> 20,
53        u64::from(n & 0xf_ffff)
54    )
55}
56
57/// The SHA-256 of hash-set entry `n`.
58#[must_use]
59pub fn hash(n: u32) -> String {
60    format!(
61        "{:016x}{:048x}",
62        0x5ef0_u64 << 48 | u64::from(n),
63        u64::from(n)
64    )
65}
66
67fn crit(field: &str, operator: &str, values: &[String]) -> serde_json::Value {
68    serde_json::json!({"criterion": {"field": field, "operator": operator, "values": values}})
69}
70
71fn rule(kind: u16, n: u32, tree: serde_json::Value, action: &str) -> serde_json::Value {
72    serde_json::json!({
73        "id": id(kind, n),
74        "name": format!("R{kind}-{n}"),
75        "object_types": ["executable", "library", "installer"],
76        "match": tree,
77        "action": {"type": action},
78    })
79}

81/// The reference policy document, under `model`.
82#[must_use]
83pub fn document(size: Size, model: PrecedenceModel) -> PolicyDocument {
84    let mut rules = Vec::new();
85    // The hash set, 25 per rule.
86    let per = 25;
87    for r in 0..size.hashes() / per {
88        let values: Vec<String> = (0..per).map(|k| hash(r * per + k)).collect();
89        rules.push(rule(1, r, crit("hash.sha256", "equals", &values), "allow"));
90    }
91    // Publishers: a publisher, some with a product name.
92    for n in 0..PUBLISHERS {
93        let tree = if n % 2 == 0 {
94            crit("signer.publisher", "equals", &[format!("Publisher {n}")])
95        } else {
96            serde_json::json!({"all": [
97                crit("signer.publisher", "equals", &[format!("Publisher {n}")]),
98                crit("meta.product_name", "equals", &[format!("Product {n}")]),
99            ]})
100        };
101        rules.push(rule(2, n, tree, "allow"));
102    }
103    // Paths: folders and wildcards under program files, half denies.
104    for n in 0..PATHS {
105        let tree = if n % 2 == 0 {
106            crit(
107                "path.directory",
108                "starts_with",
109                &[format!(r"C:\Program Files\Vendor {n}")],
110            )
111        } else {
112            crit(
113                "path.full",
114                "wildcard",
115                &[format!(r"C:\Program Files\Vendor {n}\*\*.exe")],
116            )
117        };
118        rules.push(rule(3, n, tree, if n % 4 == 3 { "deny" } else { "allow" }));
119    }
120    // Regular expressions on a publisher's product: anchored by the
121    // publisher, so they are candidates only for that publisher.
122    for n in 0..REGEXES {
123        rules.push(rule(
124            4,
125            n,
126            serde_json::json!({"all": [
127                crit("signer.publisher", "equals", &[format!("Publisher {n}")]),
128                crit("meta.original_filename", "regex", &[format!("tool{n}[0-9]*\\.exe")]),
129            ]}),
130            "audit",
131        ));
132    }
133    // The rest: single hashes, file names with a publisher, and SHA-1
134    // denies, up to the reference count.
135    let rest = OTHER_RULES - PUBLISHERS - PATHS - REGEXES;
136    for n in 0..rest {
137        let tree = match n % 3 {
138            0 => crit(
139                "hash.sha256",
140                "equals",
141                &[format!("{:064x}", 0xabcd_0000_u64 + u64::from(n))],
142            ),
143            1 => serde_json::json!({"all": [
144                crit("signer.publisher", "equals", &[format!("Publisher {}", n % PUBLISHERS)]),
145                crit("path.file_name", "equals", &[format!("app{n}.exe")]),
146            ]}),
147            _ => crit("hash.sha1", "equals", &[format!("{:040x}", u64::from(n))]),
148        };
149        let action = if n % 3 == 2 { "deny" } else { "allow" };
150        rules.push(rule(5, n, tree, action));
151    }
152    for (i, r) in rules.iter_mut().enumerate() {
153        let i = u32::try_from(i).unwrap_or(u32::MAX);
154        if let Some(r) = r.as_object_mut() {
155            r.insert("order".into(), (200_000 + i % 700_000).into());
156            r.insert("priority".into(), i.into());
157        }
158    }
159    let mandatory = vec![rule(
160        6,
161        0,
162        crit("hash.sha256", "equals", &[MANDATORY.to_owned()]),
163        "deny",
164    )];
165    let model_name = match model {
166        PrecedenceModel::DenyWins => "deny_wins",
167        PrecedenceModel::AllowWins => "allow_wins",
168        PrecedenceModel::Specificity => "specificity",
169        PrecedenceModel::FirstMatch => "first_match",
170        PrecedenceModel::Priority => "priority",
171    };
172    let json = serde_json::json!({
173        "schema_version": "1.0.0",
174        "policy": {
175            "id": "01936b2e-8c4a-7b3d-9f21-0a1b2c3d4e5f",
176            "name": "Reference",
177            "revision": 1,
178            "mode": "enforce",
179            "precedence_model": model_name,
180            "rule_sets": [
181                "01936b2e-8c4a-7b3d-9f21-0a1b2c3d0001",
182                "01936b2e-8c4a-7b3d-9f21-0a1b2c3d0002"
183            ],
184        },
185        "rule_sets": [
186            {
187                "id": "01936b2e-8c4a-7b3d-9f21-0a1b2c3d0001",
188                "name": "Reference",
189                "kind": "mixed",
190                "rules": rules,
191            },
192            {
193                "id": "01936b2e-8c4a-7b3d-9f21-0a1b2c3d0002",
194                "name": "Mandatory",
195                "kind": "blocklist",
196                "mandatory": true,
197                "rules": mandatory,
198            }
199        ],
200    });
201    match serde_json::from_value(json) {
202        Ok(doc) => doc,
203        Err(e) => panic!("the reference policy is not a document: {e}"),
204    }
205}

207/// The reference policy, validated.
208///
209/// # Panics
210///
211/// If it does not validate, which a test checks.
212#[must_use]
213pub fn policy(size: Size, model: PrecedenceModel) -> ValidatedPolicy {
214    match document(size, model).validate() {
215        Ok(p) => p,
216        Err(e) => panic!("the reference policy does not validate: {e:?}"),
217    }
218}

220/// The reference policy compiled into bundle bytes (#43), as an agent
221/// receives it, with no library sets.
222///
223/// # Panics
224///
225/// If it does not compile, which a test checks.
226#[must_use]
227pub fn bundle(size: Size, model: PrecedenceModel) -> Vec<u8> {
228    use crate::bundle::{BundleLimits, Header, Source, SourceRevisions, Target, TargetScope};
229    let policy = policy(size, model);
230    let catalogue = match crate::library::Catalogue::new(Vec::new()) {
231        Ok(c) => c,
232        Err(e) => panic!("an empty catalogue: {e:?}"),
233    };
234    let id = |n: u8| {
235        let mut b = [0u8; 16];
236        b[6] = 0x70;
237        b[8] = 0x80;
238        b[15] = n;
239        palisade_core::Uuid7::from_bytes(b)
240    };
241    let (Some(tenant), Some(group)) = (id(1), id(2)) else {
242        panic!("not UUIDv7s");
243    };
244    let p = &policy.document().policy;
245    let source = Source {
246        policy: &policy,
247        catalogue: &catalogue,
248        from: crate::library::Visibility::Global,
249        sources: SourceRevisions {
250            policies: vec![(p.id, p.revision)],
251            libraries: Vec::new(),
252        },
253        facts: crate::analysis::Facts::default(),
254        admin: None,
255    };
256    let header = Header::new(
257        Target {
258            tenant,
259            scope: TargetScope::Group(group),
260        },
261        1,
262        1_764_201_600_000_000,
263    );
264    match crate::bundle::compile(&source, header, &BundleLimits::DEFAULT) {
265        Ok(c) => c.bytes,
266        Err(e) => panic!("the reference policy does not compile into a bundle: {e}"),
267    }
268}

270/// The SHA-256 the mandatory deny blocks.
271pub const MANDATORY: &str = "dead00000000000000000000000000000000000000000000000000000000beef";
272
273/// The objects the budget measures.
274#[derive(Clone, Copy, Debug, PartialEq, Eq)]
275pub enum Case {
276    /// An entry of the hash set.
277    HashHit,
278    /// A publisher rule's publisher and product.
279    PublisherHit,
280    /// A path wildcard's folder.
281    PathWildcard,
282    /// Nothing matches: the implicit deny.
283    UnknownBlock,
284    /// The mandatory deny's hash.
285    MandatoryDeny,
286}
287
288impl Case {
289    /// Every case.
290    pub const ALL: [Self; 5] = [
291        Self::HashHit,
292        Self::PublisherHit,
293        Self::PathWildcard,
294        Self::UnknownBlock,
295        Self::MandatoryDeny,
296    ];
297
298    /// The benchmark name.
299    #[must_use]
300    pub const fn name(self) -> &'static str {
301        match self {
302            Self::HashHit => "eval_hash_hit",
303            Self::PublisherHit => "eval_publisher_hit",
304            Self::PathWildcard => "eval_path_wildcard",
305            Self::UnknownBlock => "eval_unknown_block",
306            Self::MandatoryDeny => "eval_mandatory_deny",
307        }
308    }
309
310    /// The facts of the object, with every class computed. `i` varies the
311    /// object, so a run does not decide one object only.
312    #[must_use]
313    pub fn facts(self, i: u32, size: Size) -> ObjectFacts {
314        let volumes = VolumeMap::new();
315        let (sha, path, signer) = match self {
316            Self::HashHit => (
317                hash(i.wrapping_mul(7919) % size.hashes()),
318                r"C:\Users\u\Downloads\setup.exe".to_owned(),
319                None,
320            ),
321            Self::PublisherHit => {
322                let n = (i % (PUBLISHERS / 2)) * 2 + 1;
323                (
324                    format!("{:064x}", 0xffff_0000_u64 + u64::from(i)),
325                    r"C:\Apps\x\tool.exe".to_owned(),
326                    Some((format!("Publisher {n}"), format!("Product {n}"))),
327                )
328            }
329            Self::PathWildcard => {
330                // A wildcard allow: odd rule numbers are wildcards, and
331                // those that are 3 modulo 4 deny.
332                let n = (i % (PATHS / 4)) * 4 + 1;
333                (
334                    format!("{:064x}", 0xeeee_0000_u64 + u64::from(i)),
335                    format!(r"C:\Program Files\Vendor {n}\bin\app.exe"),
336                    None,
337                )
338            }
339            Self::UnknownBlock => (
340                format!("{:064x}", 0xdddd_0000_u64 + u64::from(i)),
341                format!(r"C:\Users\u\AppData\Local\Temp\{i}\x.exe"),
342                Some(("Unknown Signer".to_owned(), "Unknown".to_owned())),
343            ),
344            Self::MandatoryDeny => (
345                MANDATORY.to_owned(),
346                r"C:\Program Files\Vendor 1\bin\a.exe".to_owned(),
347                None,
348            ),
349        };
350        let mut facts = ObjectFacts::new()
351            .with(Field::HashSha256, sha)
352            // No SHA-1 deny of the policy names this.
353            .with(
354                Field::HashSha1,
355                format!("{:040x}", 0xf000_0000_u64 + u64::from(i)),
356            )
357            .with(Field::MetaOriginalFilename, "tool.exe")
358            .with_path(
359                PathFact::new(&path, &volumes)
360                    .with_user_writable(false)
361                    .with_volume(crate::unknown::VolumeKind::Fixed),
362            );
363        let signers = match signer {
364            Some((publisher, product)) => {
365                facts.set(Field::MetaProductName, product);
366                vec![SignerFact {
367                    status: SignerStatus::Valid,
368                    source: SignatureSource::Embedded,
369                    publisher: Some(publisher),
370                    issuer: Some("Example CA".to_owned()),
371                    leaf_thumbprint_sha256: Some("ab".repeat(32)),
372                    details: Default::default(),
373                }]
374            }
375            None => Vec::new(),
376        };
377        facts.with_signers(signers)
378    }
379
380    /// The outcome the case must have, under `deny_wins`.
381    #[must_use]
382    pub const fn expected(self) -> crate::decision::Outcome {
383        use crate::decision::Outcome;
384        match self {
385            Self::HashHit | Self::PublisherHit | Self::PathWildcard => Outcome::Allowed,
386            Self::UnknownBlock | Self::MandatoryDeny => Outcome::Blocked,
387        }
388    }
389}

397#[cfg(test)]
398mod tests {
399    use super::*;
400    use crate::evaluate::{Evaluator, Subject};
401    use crate::facts::PathVariables;
402    use crate::model::ObjectType;
403
404    #[test]
405    fn the_reference_policy_validates_and_decides_each_case() {
406        let p = policy(Size::Reference, PrecedenceModel::DenyWins);
407        let rules: usize = p.document().rule_sets.iter().map(|s| s.rules.len()).sum();
408        assert_eq!(rules, 4_000 + 10_000 + 1);
409        let e = Evaluator::new(&p).unwrap();
410        let stats = e.index_stats();
411        assert_eq!(stats.always_eval, 0, "every reference rule is indexed");
412        let v = PathVariables::new();
413        let c = context();
414        for case in Case::ALL {
415            for i in [0, 1, 99] {
416                let f = case.facts(i, Size::Reference);
417                let s = Subject {
418                    object_type: ObjectType::Executable,
419                    facts: &f,
420                    variables: &v,
421                    context: &c,
422                };
423                let d = e.evaluate(&s, crate::decision::TraceId([0; 16]));
424                assert_eq!(d.outcome(), case.expected(), "{case:?} {i}");
425                assert_eq!(
426                    d.matched_rule().is_some(),
427                    case != Case::UnknownBlock,
428                    "{case:?} {i}"
429                );
430                // The linear scan agrees.
431                let linear = e.evaluate_linear(&s, crate::decision::TraceId([0; 16]));
432                assert_eq!(d, linear, "{case:?} {i}");
433            }
434        }
435    }
