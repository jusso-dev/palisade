1//! The lenient reader: policy documents of a newer minor version (#30,
2//! #330).
3//!
4//! A minor version only adds optional fields. A reader built for an older
5//! minor does not know them. [`PolicyDocument::from_json`] (strict mode,
6//! the default and the only mode for authoring) rejects them. [`read_json`]
7//! reads the document anyway, but only where that cannot relax a decision
8//! (ADR 0018). An unknown field has no known meaning, so the reader cannot
9//! tell whether it would have restricted something. It therefore only skips
10//! what never decides:
11//!
12//! - **Governance data.** A field of a `metadata` object (the policy's,
13//!   a rule set's or a rule's), or of the policy's `governance` settings,
14//!   that is unknown or whose value cannot be read is set aside. Metadata
15//!   never changes a decision ([`PolicyDocument::decision_relevant`]), and
16//!   governance settings only steer review and publishing.
17//! - **Rules that never let an object run.** A rule that cannot be read,
18//!   for example because its match tree tests a criterion this reader does
19//!   not know, is unreadable. If its action is `audit`, `defer`,
20//!   `trust_source` or `ignore`, the rule is set aside. Without it, nothing
21//!   that was blocked can run: an annotation or a trust grant is lost, or
22//!   an excluded object is decided by the policy instead.
23//!
24//! Every other unreadable part refuses the whole document
25//! ([`LenientError`]), and the device keeps the policy it has (ADR 0018,
26//! "new bundle fails verification or activation"). That covers an unknown
27//! field anywhere else, and an unreadable rule that allows, denies, or has
28//! an action this reader does not know. The reader cannot drop such a rule
29//! without risking a relaxation: without a deny, the object may run.
30//! Without an `allow_contained`, a plain `allow` may win in its place.
31//! Without a plain `allow`, an `allow_elevated` may.
32//!
33//! What is set aside is kept, with its JSON Pointer and value, in
34//! [`Lenient::ignored`], so a caller can report it.
35//!
36//! Documents of an older major version are migrated first
37//! ([`crate::migrate`]).

47/// A document read by [`read_json`].
48#[derive(Clone, Debug, PartialEq)]
49pub struct Lenient {
50    /// The readable document, in the current version.
51    pub document: PolicyDocument,
52    /// The version it was written for.
53    pub from: SchemaVersion,
54    /// What was set aside, in document order.
55    pub ignored: Vec<Ignored>,
56}
57
58/// One part of a document that [`read_json`] set aside.
59#[derive(Clone, Debug, PartialEq)]
60pub struct Ignored {
61    /// JSON Pointer (RFC 6901) of the part, in the document in the current
62    /// version (after any migration).
63    pub pointer: String,
64    /// What it is.
65    pub kind: IgnoredKind,
66    /// The part, unchanged.
67    pub value: Value,
68    /// Why it could not be read.
69    pub reason: String,
70}
71
72/// What kind of part was set aside.
73#[derive(Clone, Copy, Debug, PartialEq, Eq)]
74pub enum IgnoredKind {
75    /// A field of a `metadata` or `governance` object.
76    GovernanceField,
77    /// A rule that cannot be read, and never lets an object run.
78    Rule(ActionKind),
79}
80
81/// A document could not be read, even leniently.
82#[derive(Debug)]
83pub enum LenientError {
84    /// The input is not JSON, has no supported version, cannot be migrated,
85    /// or has an unknown or unreadable part outside the governance data and
86    /// the rules.
87    Read(MigrateError),
88    /// A rule cannot be read, and setting it aside could relax a decision.
89    UnreadableRule {
90        /// JSON Pointer of the rule.
91        pointer: String,
92        /// Its action, if this reader knows it.
93        action: Option<ActionKind>,
94        /// Why it cannot be read.
95        reason: String,
96    },
97}

99impl fmt::Display for LenientError {
100    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
101        match self {
102            Self::Read(e) => e.fmt(f),
103            Self::UnreadableRule {
104                pointer,
105                action,
106                reason,
107            } => {
108                write!(f, "{pointer}: ")?;
109                match action {
110                    Some(a) => write!(f, "an unreadable `{}` rule", a.as_str())?,
111                    None => f.write_str("an unreadable rule with an unknown action")?,
112                }
113                write!(
114                    f,
115                    " could relax a decision, so the document is refused: {reason}"
116                )
117            }
118        }
119    }
120}

122impl std::error::Error for LenientError {
123    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
124        match self {
125            Self::Read(e) => Some(e),
126            Self::UnreadableRule { .. } => None,
127        }
128    }
129}
130
131impl From<MigrateError> for LenientError {
132    fn from(e: MigrateError) -> Self {
133        Self::Read(e)
134    }
135}

137/// Reads a policy document leniently. See [the module docs](self).
138///
139/// # Errors
140///
141/// Returns [`LenientError`] if the document cannot be read without
142/// risking a relaxed decision.
143pub fn read_json(input: &str) -> Result<Lenient, LenientError> {
144    let UniqueValue(value) =
145        serde_json::from_str(input).map_err(|e| LenientError::Read(MigrateError::Json(e)))?;
146    let from = migrate::schema_version(&value)?;
147    let mut value = migrate::migrate(value)?;
148    let mut ignored = Vec::new();
149    set_aside(&mut value, &mut ignored)?;
150    let document = PolicyDocument::deserialize(value).map_err(MigrateError::Json)?;
151    Ok(Lenient {
152        document,
153        from,
154        ignored,
155    })
156}

158/// Sets aside the unreadable governance fields and the unreadable rules
159/// that never let an object run. Leaves everything else for the strict
160/// reader.
161fn set_aside(document: &mut Value, ignored: &mut Vec<Ignored>) -> Result<(), LenientError> {
162    if let Some(policy) = document.get_mut("policy") {
163        clean::<Metadata>(policy, "metadata", "/policy", ignored);
164        clean::<GovernanceSettings>(policy, "governance", "/policy", ignored);
165    }
166    let Some(Value::Array(sets)) = document.get_mut("rule_sets") else {
167        return Ok(());
168    };
169    for (s, set) in sets.iter_mut().enumerate() {
170        let base = format!("/rule_sets/{s}");
171        clean::<Metadata>(set, "metadata", &base, ignored);
172        let Some(Value::Array(rules)) = set.get_mut("rules") else {
173            continue;
174        };
175        let mut kept = Vec::with_capacity(rules.len());
176        for (r, mut rule) in core::mem::take(rules).into_iter().enumerate() {
177            let pointer = format!("{base}/rules/{r}");
178            clean::<Metadata>(&mut rule, "metadata", &pointer, ignored);
179            let Err(e) = crate::model::Rule::deserialize(&rule) else {
180                kept.push(rule);
181                continue;
182            };
183            let action = action_kind(&rule);
184            match action {
185                Some(
186                    kind @ (ActionKind::Audit
187                    | ActionKind::Defer
188                    | ActionKind::TrustSource
189                    | ActionKind::Ignore),
190                ) => ignored.push(Ignored {
191                    pointer,
192                    kind: IgnoredKind::Rule(kind),
193                    value: rule,
194                    reason: e.to_string(),
195                }),
196                _ => {
197                    return Err(LenientError::UnreadableRule {
198                        pointer,
199                        action,
200                        reason: e.to_string(),
201                    });
202                }
203            }
204        }
205        *rules = kept;
206    }
207    Ok(())
208}

210/// The kind of a rule's action, if its `type` is one this reader knows.
211fn action_kind(rule: &Value) -> Option<ActionKind> {
212    let name = rule.get("action")?.get("type")?.as_str()?;
213    ActionKind::ALL.into_iter().find(|k| k.as_str() == name)
214}

216/// Sets aside each field of `parent[key]` that `T` cannot read on its own.
217/// Every field of `T` must be optional. A value that is not an object is
218/// left for the strict reader.
219fn clean<T: for<'de> Deserialize<'de>>(
220    parent: &mut Value,
221    key: &str,
222    base: &str,
223    ignored: &mut Vec<Ignored>,
224) {
225    let Some(Value::Object(fields)) = parent.get_mut(key) else {
226        return;
227    };
228    let mut kept = Map::new();
229    for (name, value) in core::mem::take(fields) {
230        let mut one = Map::new();
231        one.insert(name, value);
232        let one = Value::Object(one);
233        let result = T::deserialize(&one).map(drop);
234        let Value::Object(one) = one else {
235            continue;
236        };
237        for (name, value) in one {
238            match &result {
239                Ok(()) => {
240                    kept.insert(name, value);
241                }
242                Err(e) => ignored.push(Ignored {
243                    pointer: format!("{base}/{key}/{}", pointer_token(&name)),
244                    kind: IgnoredKind::GovernanceField,
245                    value,
246                    reason: e.to_string(),
247                }),
248            }
249        }
250    }
251    *fields = kept;
252}

254/// Escapes one JSON Pointer reference token (RFC 6901 section 3).
255fn pointer_token(token: &str) -> String {
256    token.replace('~', "~0").replace('/', "~1")
257}
