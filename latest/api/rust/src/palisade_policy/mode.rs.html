1//! Enforcement modes: which mode applies to one decision, and what it does
2//! to the decision (#36).
3//!
4//! The evaluator ([`crate::evaluate`]) first computes the enforce-mode
5//! decision. [`resolve`] then finds the effective mode for that decision,
6//! and [`Evaluator::evaluate_in`](crate::evaluate::Evaluator::evaluate_in)
7//! applies it. Everything here is a pure function of the policy and a
8//! [`DeviceState`] that the agent supplies; nothing reads a clock. The
9//! state machine that changes a device's mode over time is in
10//! [`machine`], and which decisions each mode reports is in [`events`].
11//!
12//! # The modes
13//!
14//! | Mode | Effect on a decision |
15//! |---|---|
16//! | `disabled` | The engine is not consulted: there is no decision ([`Evaluator::evaluate_in`](crate::evaluate::Evaluator::evaluate_in) returns `None`). Mandatory denies do not apply, and the device is reported `disabled`. Only a whole device can be disabled. |
17//! | `monitor` | Evaluated fully. What enforce mode would block runs (`allowed_audit`), and `would_have_been` is `blocked`. |
18//! | `audit` | As `monitor`. The two differ only in which decisions are reported ([`events`]). |
19//! | `learning` | As `audit`. An object that only the implicit deny would block is also a learning candidate ([`Decision::learning_candidate`](crate::decision::Decision::learning_candidate)). |
20//! | `enforce` | Decisions take effect. |
21//!
22//! Two kinds of block are never relaxed by a mode, a session or any
23//! override, only by `disabled`:
24//!
25//! - **A mandatory deny** (ADR 0004, #125: the blocklist always wins). It
26//!   blocks in `monitor`, `audit` and `learning`, during every override
27//!   session including `full_bypass`, and whatever a rule set's
28//!   `mode_override` says (validation rejects one on a mandatory rule set).
29//! - **An `always_enforce` rule**, except during a `full_bypass` session.
30//!
31//! # Resolution
32//!
33//! For an object of type `t` decided by `rule` (or by the implicit deny),
34//! in this order:
35//!
36//! 1. **Lockdown.** If [`DeviceState::lockdown`] is set, `enforce`, and
37//!    stop. Sessions, a disabled device mode, type modes, control levels,
38//!    rule set overrides and `never_enforce` are all ignored.
39//! 2. **Device mode.** The device's base mode, [`DeviceState::mode`]
40//!    (the policy's `mode`, or what the [`machine`] moved it to).
41//!    `disabled` stops here: no decision.
42//! 3. **Mandatory deny.** If `rule` is in a mandatory rule set, `enforce`,
43//!    and stop.
44//! 4. **Disconnected.** While the device is disconnected, a
45//!    `mode_disconnected` stricter than the device mode replaces it.
46//! 5. **Object type.** `mode_by_type[t]` if set (raised to a stricter
47//!    `mode_disconnected` while disconnected), otherwise the device mode.
48//!    Then the class's control level in the settings (`script_control`,
49//!    `script_control_by_kind`, `dll_control`, `driver_control`) caps it:
50//!    `audit` caps it at `audit`, and `off` caps it at `monitor`. `off`
51//!    means the agent does not intercept the class (#4, #92); if such an
52//!    object is evaluated anyway, only a mandatory deny, a blocklist with
53//!    `mode_override: enforce` or an `always_enforce` deny blocks it.
54//! 6. **Rule set.** The `mode_override` of `rule`'s rule set, if set,
55//!    replaces the mode. Only non-mandatory blocklists may have one.
56//! 7. **Rule.** For a deny, `always_enforce` makes it `enforce` and
57//!    `never_enforce` caps it at `audit`. Other actions cannot set it.
58//! 8. **Override session** (#9). An active session caps the mode at
59//!    `audit`, unless the rule is `always_enforce` and the session is not
60//!    `full_bypass`.
61//!
62//! A cap lowers a mode to the cap's strictness and never raises it.
63//! Strictness ranks `enforce` above `learning` and `audit` (equal), above
64//! `monitor`, above `disabled` ([`strictness`]).
65//!
66//! Lockdown never changes [`DeviceState::mode`], so clearing it restores
67//! every prior mode.
68

76/// Whether the device has checked in recently (#36).
77#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
78pub enum Connectivity {
79    /// It has checked in within `disconnected_after_secs`.
80    #[default]
81    Connected,
82    /// It has not.
83    Disconnected,
84}
85
86impl Connectivity {
87    /// The connectivity of a device whose last successful check-in was
88    /// `secs_since_check_in` seconds ago, for a policy that counts a device
89    /// as disconnected after `disconnected_after` seconds
90    /// ([`ModeOptions::disconnected_after`](crate::model::ModeOptions::disconnected_after)).
91    /// The agent measures the time on its monotonic clock.
92    #[must_use]
93    pub const fn since_check_in(secs_since_check_in: u64, disconnected_after: u64) -> Self {
94        if secs_since_check_in >= disconnected_after {
95            Self::Disconnected
96        } else {
97            Self::Connected
98        }
99    }
100}

102/// The kind of an active override session (#9). Opening and timing a
103/// session is #9 and #120; the engine only reads that one is active.
104#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
105pub enum SessionKind {
106    /// Relaxes everything except mandatory denies and `always_enforce`
107    /// rules.
108    Audit,
109    /// Also relaxes `always_enforce` rules. Never a mandatory deny.
110    FullBypass,
111}

113/// What the agent knows about the device, besides the policy. The agent
114/// keeps it (#69); the engine only reads it.
115#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
116pub struct DeviceState {
117    /// The device's base mode: the policy's `mode`, or the mode the
118    /// [`machine`] moved the device to.
119    pub mode: Mode,
120    /// Whether the device has checked in recently.
121    pub connectivity: Connectivity,
122    /// The active override session, if any.
123    pub session: Option<SessionKind>,
124    /// Whether emergency lockdown is set. It comes in the policy bundle
125    /// with top precedence (#43); an offline device applies it at its next
126    /// check-in.
127    pub lockdown: bool,
128}
129
130impl DeviceState {
131    /// A connected device in the policy's mode, with no session and no
132    /// lockdown.
133    #[must_use]
134    pub const fn of(policy: &Policy) -> Self {
135        Self {
136            mode: policy.mode,
137            connectivity: Connectivity::Connected,
138            session: None,
139            lockdown: false,
140        }
141    }
142}

144/// The mode that applies to one decision.
145#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
146pub struct EffectiveMode {
147    /// The mode.
148    pub mode: Mode,
149    /// Whether the override session lowered it from `enforce`, so that a
150    /// block it relaxes is reported as `allowed_override`.
151    pub by_session: bool,
152}

154/// How strict a mode is: `disabled` 0, `monitor` 1, `audit` and `learning`
155/// 2, `enforce` 3. Only `enforce` blocks.
156#[must_use]
157pub const fn strictness(mode: Mode) -> u8 {
158    match mode {
159        Mode::Disabled => 0,
160        Mode::Monitor => 1,
161        Mode::Audit | Mode::Learning => 2,
162        Mode::Enforce => 3,
163    }
164}
165
166/// `mode`, lowered to the strictness of `cap` if it is stricter.
167#[must_use]
168pub const fn capped(mode: Mode, cap: Mode) -> Mode {
169    if strictness(mode) > strictness(cap) {
170        cap
171    } else {
172        mode
173    }
174}

176/// The device mode: `enforce` in lockdown, otherwise
177/// [`DeviceState::mode`], raised while disconnected to `mode_disconnected`
178/// if that is stricter. A disabled device stays disabled while
179/// disconnected: it returns to its prior mode when its time box ends
180/// ([`machine`]).
181#[must_use]
182pub fn device_mode(policy: &Policy, state: &DeviceState) -> Mode {
183    if state.lockdown {
184        return Mode::Enforce;
185    }
186    if state.mode == Mode::Disabled {
187        return Mode::Disabled;
188    }
189    raise_offline(policy, state, state.mode)
190}
191
192/// `mode`, raised to `mode_disconnected` while disconnected if that is
193/// stricter.
194fn raise_offline(policy: &Policy, state: &DeviceState, mode: Mode) -> Mode {
195    match (state.connectivity, policy.mode_options.mode_disconnected) {
196        (Connectivity::Disconnected, Some(offline)) if strictness(offline) > strictness(mode) => {
197            offline
198        }
199        _ => mode,
200    }
201}

203/// Only a whole device is ever disabled. Validation rejects `disabled`
204/// below the device; if it is there anyway, it is read as `monitor`, which
205/// still evaluates and reports.
206const fn enabled(mode: Mode) -> Mode {
207    match mode {
208        Mode::Disabled => Mode::Monitor,
209        other => other,
210    }
211}

213/// The control level of an object type in the policy settings. Types with
214/// no setting of their own are `enforce`.
215#[must_use]
216pub fn control_level(policy: &Policy, object_type: ObjectType) -> ControlLevel {
217    let s = &policy.settings;
218    match object_type {
219        ObjectType::Script(kind) => script_level(s, kind),
220        ObjectType::Library => s.dll_control,
221        ObjectType::Driver => s.driver_control,
222        _ => ControlLevel::Enforce,
223    }
224}
225
226fn script_level(s: &crate::model::PolicySettings, kind: ScriptKind) -> ControlLevel {
227    s.script_control_by_kind
228        .get(&kind)
229        .copied()
230        .unwrap_or(s.script_control)
231}

233/// The mode for objects of one type, before any rule: steps 1 to 5 of the
234/// module documentation, without the session. `disabled` only when the
235/// device is disabled.
236#[must_use]
237pub fn object_mode(policy: &Policy, state: &DeviceState, object_type: ObjectType) -> Mode {
238    let device = device_mode(policy, state);
239    if device == Mode::Disabled || state.lockdown {
240        return device;
241    }
242    let mode = policy
243        .mode_options
244        .mode_by_type
245        .get(&object_type)
246        .map_or(device, |&m| raise_offline(policy, state, enabled(m)));
247    match control_level(policy, object_type) {
248        ControlLevel::Enforce => mode,
249        ControlLevel::Audit => capped(mode, Mode::Audit),
250        ControlLevel::Off => capped(mode, Mode::Monitor),
251    }
252}

254/// The effective mode for a decision about an object of `object_type`,
255/// decided by `rule` in `set`, or by the implicit deny when `rule` is
256/// `None`. See the module documentation for the steps.
257#[must_use]
258pub fn resolve(
259    policy: &Policy,
260    object_type: ObjectType,
261    rule: Option<(&RuleSet, &Rule)>,
262    state: &DeviceState,
263) -> EffectiveMode {
264    let fixed = |mode| EffectiveMode {
265        mode,
266        by_session: false,
267    };
268    if state.lockdown {
269        return fixed(Mode::Enforce);
270    }
271    if state.mode == Mode::Disabled {
272        return fixed(Mode::Disabled);
273    }
274    if rule.is_some_and(|(set, _)| set.mandatory) {
275        return fixed(Mode::Enforce);
276    }
277    let mut mode = object_mode(policy, state, object_type);
278    // Only a deny blocks, so only a deny's override means anything;
279    // validation rejects it on any other action.
280    let enforcement = rule
281        .filter(|(_, r)| matches!(r.action, Action::Deny(_)))
282        .map_or(EnforcementOverride::Inherit, |(_, r)| {
283            r.enforcement_override
284        });
285    if let Some(over) = rule.and_then(|(set, _)| set.mode_override) {
286        mode = enabled(over);
287    }
288    mode = match enforcement {
289        EnforcementOverride::Inherit => mode,
290        EnforcementOverride::AlwaysEnforce => Mode::Enforce,
291        EnforcementOverride::NeverEnforce => capped(mode, Mode::Audit),
292    };
293    let Some(session) = state.session else {
294        return fixed(mode);
295    };
296    if enforcement == EnforcementOverride::AlwaysEnforce && session != SessionKind::FullBypass {
297        return fixed(mode);
298    }
299    EffectiveMode {
300        mode: capped(mode, Mode::Audit),
301        by_session: mode == Mode::Enforce,
302    }
303}

305/// The object types that Essential Eight application control requires to
306/// be enforced (ISM-1657, ISM-1658): executables, libraries, drivers,
307/// every script kind, installers, compiled HTML, HTML applications and
308/// control panel applets.
309#[must_use]
310pub fn essential_eight_types() -> Vec<ObjectType> {
311    ObjectType::ALL
312        .into_iter()
313        .filter(|t| {
314            matches!(
315                t,
316                ObjectType::Executable
317                    | ObjectType::Library
318                    | ObjectType::Driver
319                    | ObjectType::Script(_)
320                    | ObjectType::Installer
321                    | ObjectType::CompiledHtml
322                    | ObjectType::HtmlApplication
323                    | ObjectType::ControlPanelApplet
324            )
325        })
326        .collect()
327}

329/// A device's protection status, for device lists, health and evidence
330/// (#10, #17). The snake_case name is part of the API.
331#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, serde::Serialize)]
332#[serde(rename_all = "snake_case")]
333pub enum ProtectionStatus {
334    /// Every Essential Eight type is enforced.
335    ProtectedEnforce,
336    /// The device mode is `enforce`, but at least one Essential Eight type
337    /// is not. It does not count as effective (#17).
338    ProtectedPartial,
339    /// The device is in `audit`.
340    Audit,
341    /// The device is in `learning`.
342    Learning,
343    /// The device is in `monitor`.
344    Monitor,
345    /// The device is disabled: unprotected.
346    Disabled,
347    /// An override session is active.
348    OverrideActive,
349    /// Emergency lockdown is set.
350    Lockdown,
351}
352
353impl ProtectionStatus {
354    /// The status of a device. The first that applies: `lockdown`,
355    /// `disabled`, `override_active`, then `protected_enforce` if every
356    /// Essential Eight type is in `enforce`, then `protected_partial` if
357    /// the device mode is `enforce`, and otherwise the device mode.
358    ///
359    /// `stale_policy` needs the bundle's expiry and is #61.
360    #[must_use]
361    pub fn of(policy: &Policy, state: &DeviceState) -> Self {
362        let device = device_mode(policy, state);
363        if state.lockdown {
364            return Self::Lockdown;
365        }
366        if device == Mode::Disabled {
367            return Self::Disabled;
368        }
369        if state.session.is_some() {
370            return Self::OverrideActive;
371        }
372        let all_enforced = essential_eight_types()
373            .into_iter()
374            .all(|t| object_mode(policy, state, t) == Mode::Enforce);
375        match device {
376            _ if all_enforced => Self::ProtectedEnforce,
377            Mode::Enforce => Self::ProtectedPartial,
378            Mode::Learning => Self::Learning,
379            Mode::Audit => Self::Audit,
380            Mode::Monitor => Self::Monitor,
381            Mode::Disabled => Self::Disabled,
382        }
383    }
384}
