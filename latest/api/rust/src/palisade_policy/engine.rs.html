1//! The engine the agent decides through (#42): a policy snapshot that is
2//! swapped atomically, the decision cache, the lazy fact protocol and the
3//! per-decision watchdog.
4//!
5//! An [`Engine`] holds the current [`Snapshot`]: one validated policy with
6//! its compiled [`Evaluator`]. [`Engine::load`] builds a new snapshot off
7//! the hot path and swaps it in with one atomic store (`arc-swap`). A
8//! decision that started on the old snapshot keeps it until it finishes;
9//! the next one sees the new policy. Reading the snapshot takes no lock.
10//!
11//! [`Engine::evaluate`] decides one object:
12//!
13//! 1. A cache hit returns the cached decision ([`cache`]).
14//! 2. Otherwise the evaluator decides with the facts computed so far, or
15//!    returns [`NeedFacts`] with what to compute next
16//!    ([`crate::evaluate::index`]).
17//! 3. A watchdog bounds the time spent. Mandatory denies are evaluated
18//!    first; the budget is checked after each match tree, and once it is
19//!    spent no further tree is evaluated. A mandatory deny that already
20//!    matched then decides, as always (ADR 0004). If the mandatory tier
21//!    was not finished, the object is blocked. Otherwise the decision
22//!    falls back to [`TimeoutAction`]: `block` (the default when the
23//!    object's type is enforced), or `allow_audit`. The result names the
24//!    slow rule for the `policy.eval_timeout` health event
25//!    ([`EvalTimeout`]).
26//!
27//! The engine is `Send + Sync`. Time comes from the caller ([`Clock`]), so
28//! the engine stays pure and tests can drive time.

62/// One loaded policy, ready to decide.
63#[derive(Debug)]
64pub struct Snapshot {
65    compiled: Compiled,
66    generation: u64,
67    plan: KeyPlan,
68}
69
70impl Snapshot {
71    /// Compiles a validated policy.
72    ///
73    /// # Errors
74    ///
75    /// [`BuildError`] if a match tree does not compile, which validation
76    /// rules out.
77    pub fn new(policy: ValidatedPolicy, generation: u64) -> Result<Self, BuildError> {
78        let compiled = Compiled::try_new(policy, |p| Evaluator::new(p))?;
79        // Only the rules the evaluator uses: enabled rules of the rule sets
80        // the policy references.
81        let plan = {
82            let doc = compiled.borrow_owner().document();
83            KeyPlan::of(
84                doc.rule_sets
85                    .iter()
86                    .filter(|s| doc.policy.rule_sets.contains(&s.id))
87                    .flat_map(|s| &s.rules)
88                    .filter(|r| r.enabled),
89            )
90        };
91        Ok(Self {
92            compiled,
93            generation,
94            plan,
95        })
96    }
97
98    /// The evaluator.
99    #[must_use]
100    pub fn evaluator(&self) -> &Evaluator<'_> {
101        self.compiled.borrow_dependent()
102    }
103
104    /// The validated policy.
105    #[must_use]
106    pub fn policy(&self) -> &ValidatedPolicy {
107        self.compiled.borrow_owner()
108    }
109
110    /// The generation: different for every snapshot an engine loads.
111    #[must_use]
112    pub const fn generation(&self) -> u64 {
113        self.generation
114    }
115}

117/// What a decision does when its watchdog budget runs out (#42):
118/// `on_eval_timeout`.
119#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
120pub enum TimeoutAction {
121    /// Block, with the reason `eval_timeout`.
122    Block,
123    /// Let the object run, reported as `allowed_audit` with
124    /// `would_have_been: blocked` and the reason `eval_timeout`.
125    AllowAudit,
126}
127
128impl TimeoutAction {
129    /// The default when the object's type is decided in `mode`, before any
130    /// override session: `block` when it enforces, `allow_audit` otherwise,
131    /// since a mode that does not enforce would not have blocked either.
132    /// Whichever it is, a mandatory deny always blocks (ADR 0004).
133    #[must_use]
134    pub const fn default_for(mode: Mode) -> Self {
135        match mode {
136            Mode::Enforce => Self::Block,
137            Mode::Disabled | Mode::Monitor | Mode::Audit | Mode::Learning => Self::AllowAudit,
138        }
139    }
140}

142/// How the engine is tuned for the kind of machine it runs on (#42). Only
143/// sizes and heuristics change; decisions never do.
144#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
145pub enum PerfProfile {
146    /// A single-user workstation: the default.
147    #[default]
148    Workstation,
149    /// A server: a smaller cache, and no eager hashing.
150    Server,
151    /// A multi-session host: a larger cache, shared by every session.
152    MultiSession,
153}
154
155impl PerfProfile {
156    /// The cache configuration of the profile.
157    #[must_use]
158    pub const fn cache(self) -> CacheConfig {
159        match self {
160            Self::Workstation => CacheConfig {
161                max_bytes: 32 << 20,
162                shards: 16,
163            },
164            Self::Server => CacheConfig {
165                max_bytes: 8 << 20,
166                shards: 8,
167            },
168            Self::MultiSession => CacheConfig {
169                max_bytes: 64 << 20,
170                shards: 64,
171            },
172        }
173    }
174
175    /// Whether the agent should hash every executable when it is created,
176    /// rather than wait for the engine to ask, given the policy's needs.
177    #[must_use]
178    pub const fn hash_eagerly(self, required: crate::evaluate::RequiredFacts) -> bool {
179        match self {
180            Self::Server => false,
181            Self::Workstation | Self::MultiSession => required.hash_eagerly(),
182        }
183    }
184}

221/// The agent's clocks, as the engine reads them.
222pub trait Clock {
223    /// Monotonic microseconds since an arbitrary origin, for the watchdog
224    /// and cache expiry.
225    fn monotonic_micros(&self) -> u64;
226    /// Whether wall-clock time can be trusted (#220). While it cannot,
227    /// nothing is cached, because cached schedule-dependent decisions
228    /// would rest on it.
229    fn trusted(&self) -> bool {
230        true
231    }
232}

234/// The monotonic clock of the process.
235#[derive(Clone, Copy, Debug)]

238}

240impl SystemClock {
241    /// A clock whose origin is now.
242    #[must_use]
243    #[expect(
244        clippy::disallowed_methods,
245        reason = "the engine's Clock implementation is the one place that reads the monotonic clock"
246    )]
247    pub fn new() -> Self {
248        Self {
249            origin: std::time::Instant::now(),
250        }
251    }
252}
253
254impl Default for SystemClock {
255    fn default() -> Self {
256        Self::new()
257    }
258}
259
260impl Clock for SystemClock {
261    fn monotonic_micros(&self) -> u64 {
262        u64::try_from(self.origin.elapsed().as_micros()).unwrap_or(u64::MAX)
263    }

266/// A decision request: the object, what has been computed, and how the
267/// result may be cached.
268#[derive(Clone, Copy, Debug)]
269pub struct Request<'a> {
270    /// The object and its context.
271    pub subject: Subject<'a>,
272    /// The fact classes computed so far, besides the cheap ones.
273    pub computed: FactSet,
274    /// The device's state.

282    /// A monotonic time, in microseconds, after which a cached decision

303/// A decision, made now or shared from the cache.
304#[derive(Clone, Debug, PartialEq, Eq)]
305// A fresh decision is returned by value, so the decision path does not
306// allocate a box for it.
307#[allow(clippy::large_enum_variant)]
308pub enum Decided {
309    /// Made by this evaluation.
310    Fresh(Decision),
311    /// From the cache. It is the decision of the evaluation that cached
312    /// it, including that evaluation's trace ID, so an event about this
313    /// launch can link back to the full trace (#41).
314    Cached(Arc<Decision>),
315}
316
317impl core::ops::Deref for Decided {
318    type Target = Decision;
319
320    fn deref(&self) -> &Decision {
321        match self {
322            Self::Fresh(d) => d,
323            Self::Cached(d) => d,
324        }
325    }
326}

328/// The result of one [`Engine::evaluate`].
329#[derive(Clone, Debug, PartialEq, Eq)]
330pub enum Outcome {
331    /// Decided. `None` when the device is disabled.
332    Decided(Option<Decided>),
333    /// More facts are needed.
334    NeedFacts(NeedFacts),
335    /// The budget ran out; `decision` is the fallback.
336    TimedOut {
337        /// The fallback decision.

344/// The engine.
345#[derive(Debug)]
346pub struct Engine {
347    current: ArcSwap<Snapshot>,
348    cache: DecisionCache,
349    config: EngineConfig,
350    next_generation: AtomicU64,
351    /// The key of the subject, context and object key hashes, chosen when
352    /// the engine starts.
353    seed: cache::KeySeed,
354}
355
356impl Engine {
357    /// An engine that decides with `policy`.
358    ///
359    /// # Errors
360    ///
361    /// [`BuildError`] if a match tree does not compile, which validation
362    /// rules out.
363    pub fn new(policy: ValidatedPolicy, config: EngineConfig) -> Result<Self, BuildError> {
364        let snapshot = Snapshot::new(policy, 1)?;
365        let seed = cache::KeySeed::new();
366        Ok(Self {
367            current: ArcSwap::from_pointee(snapshot),
368            cache: DecisionCache::new(config.cache),
369            config,
370            next_generation: AtomicU64::new(2),
371            seed,
372        })
373    }
374
375    /// Compiles `policy` and makes it current. Decisions already running
376    /// finish on the old snapshot; the cache is purged, and its keys name
377    /// the generation, so no decision of the old policy is ever returned
378    /// for the new one.
379    ///
380    /// # Errors
381    ///
382    /// [`BuildError`] if a match tree does not compile; the current policy
383    /// stays.
384    pub fn load(&self, policy: ValidatedPolicy) -> Result<(), BuildError> {
385        let generation = self.next_generation.fetch_add(1, Ordering::Relaxed);
386        let snapshot = Snapshot::new(policy, generation)?;
387        self.current.store(Arc::new(snapshot));
388        self.cache.retain_generation(generation);
389        Ok(())
390    }
391
392    /// The current snapshot. Holding it keeps that policy alive.
393    #[must_use]
394    pub fn snapshot(&self) -> Arc<Snapshot> {
395        self.current.load_full()
396    }
397
398    /// The cache's counters.
399    #[must_use]
400    pub fn cache_stats(&self) -> CacheStats {
401        self.cache.stats()
402    }
403
404    /// The engine's settings.
405    #[must_use]
406    pub const fn config(&self) -> &EngineConfig {
407        &self.config
408    }
409
410    /// Forgets every cached decision about a file that was written,
411    /// renamed or deleted.
412    pub fn invalidate_file(&self, volume_serial: u64, file_id: u128) {
413        self.cache.invalidate_file(volume_serial, file_id);
414    }
415
416    /// Forgets every cached decision made in a logon session that ended.
417    pub fn invalidate_session(&self, session: u32) {
418        self.cache.invalidate_session(session);
419    }
420
421    /// Forgets every cached decision.
422    pub fn purge_cache(&self) {
423        self.cache.purge();
424    }
425
426    /// Decides one object. See the module documentation.
427    pub fn evaluate(&self, request: &Request<'_>, clock: &dyn Clock) -> Outcome {
428        let start = clock.monotonic_micros();
429        let guard = self.current.load();
430        let snapshot: &Snapshot = &guard;
431        let evaluator = snapshot.evaluator();
432        let policy = evaluator.policy();
433
434        let key = request.file.map(|file| CacheKey {
435            file,
436            object_type: request.subject.object_type,
437            subject: snapshot
438                .plan
439                .subject_key(request.subject.context, &self.seed),
440            context: snapshot
441                .plan
442                .context_key(request.subject.context, &self.seed)
443                ^ state_key(request.state),
444            object: cache::object_key(request.subject.facts, request.subject.variables, &self.seed),
445            policy: snapshot.generation,
446        });
447        let ttl = u64::from(policy.settings.decision_cache_ttl_secs);
448        let use_cache = ttl > 0 && clock.trusted();
449        if let (Some(key), true) = (key.as_ref(), use_cache)
450            && let Some(d) = self.cache.get(key, start)
451        {
452            return Outcome::Decided(Some(Decided::Cached(d)));
453        }
454
455        let budget = self.config.budget_micros;
456        let mut expired = |_: &RuleRef| clock.monotonic_micros().saturating_sub(start) > budget;
457        match evaluator.evaluate_watched(
458            &request.subject,
459            request.computed,
460            request.state,
461            request.trace_id,
462            &mut expired,
463        ) {
464            Ok(None) => Outcome::Decided(None),
465            Ok(Some((decision, unknown))) => {
466                if let (Some(key), true) = (key, use_cache)
467                    && cache::cacheable(&decision, unknown, request.state.session.is_some())
468                {
469                    let mut expires = start.saturating_add(ttl.min(86_400) * 1_000_000);
470                    if let Some(deadline) = request.expires_before {
471                        expires = expires.min(deadline);
472                    }
473                    // A deadline that has already passed: nothing to keep.
474                    if expires > clock.monotonic_micros() {
475                        // The caller gets the decision as made now: only a
476                        // later hit is `Cached` (#41 records the trace of a
477                        // fresh decision, and links hits back to it).
478                        self.cache.insert(
479                            key,
480                            Arc::new(decision.clone()),
481                            request.session,
482                            expires,
483                        );
484                    }
485                }
486                Outcome::Decided(Some(Decided::Fresh(decision)))
487            }
488            Err(Stop::Need(need)) => Outcome::NeedFacts(need),
489            Err(Stop::Timeout(timeout)) => {
490                let elapsed = clock.monotonic_micros().saturating_sub(start);
491                // The mode the object's type is decided in, before any
492                // override session.
493                let mode = crate::mode::object_mode(
494                    policy,
495                    &DeviceState {
496                        session: None,
497                        ..*request.state
498                    },
499                    request.subject.object_type,
500                );
501                let action = self
502                    .config
503                    .on_timeout
504                    .unwrap_or(TimeoutAction::default_for(mode));
505                let decision = evaluator.timed_out(
506                    &request.subject,
507                    &timeout,
508                    action == TimeoutAction::Block,
509                    request.trace_id,
510                );
511                Outcome::TimedOut {
512                    decision,
513                    event: EvalTimeout {
514                        slow_rule: timeout.slow_rule,
515                        elapsed_micros: elapsed,
516                    },
517                }
518            }
519        }
520    }
521}

523/// A key for the device state, folded into the context key: a decision is
524/// only reused in the state it was made in.
525fn state_key(state: &DeviceState) -> u64 {
526    let mode = match state.mode {
527        Mode::Disabled => 0u64,
528        Mode::Monitor => 1,
529        Mode::Audit => 2,
530        Mode::Learning => 3,
531        Mode::Enforce => 4,
532    };
533    let connected = match state.connectivity {
534        crate::mode::Connectivity::Connected => 0u64,
535        crate::mode::Connectivity::Disconnected => 1,
536    };
537    let session = match state.session {
538        None => 0u64,
539        Some(crate::mode::SessionKind::Audit) => 1,
540        Some(crate::mode::SessionKind::FullBypass) => 2,
541    };
542    (mode | connected << 3 | session << 4 | u64::from(state.lockdown) << 6)
543        .wrapping_mul(0x9e37_79b9_7f4a_7c15)
544}
