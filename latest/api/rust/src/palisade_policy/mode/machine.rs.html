1//! The mode state machine: which mode changes are legal, and when a
2//! time-boxed mode ends (#36).
3//!
4//! A [`ModeMachine`] holds a device's base mode ([`DeviceState::mode`](super::DeviceState::mode))
5//! and, when the device was moved to a weaker mode for a limited time, the
6//! [`TimeBox`] that moves it back. It is pure: every call takes the time
7//! from the caller ([`Now`], usually from a [`Clock`]), and the agent
8//! stores the machine between calls (#69).
9//!
10//! # Legal transitions
11//!
12//! | From | To |
13//! |---|---|
14//! | `disabled` | any |
15//! | `monitor` | `audit`, `learning`, `enforce`, `disabled` |
16//! | `audit` | `learning`, `enforce`, `monitor`, `disabled` |
17//! | `learning` | `audit`, `enforce`, `disabled` |
18//! | `enforce` | `audit`, `learning`, `monitor`, `disabled`, each time-boxed |
19//!
20//! A mode never transitions to itself. Every other change is rejected
21//! ([`TransitionError::Illegal`]), and a rejected request changes nothing.
22//!
23//! # Time boxes
24//!
25//! Leaving `enforce` for a weaker mode, and entering `disabled` from any
26//! mode, needs a `reason`, a ticket reference when the tenant requires
27//! one, and an `expires_at` at most 30 days ahead. Only `enforce` to
28//! `audit` may instead be `permanent`, and only for an actor with the
29//! elevated permission ([`Governance::may_weaken_permanently`]).
30//!
31//! When a time box ends, the machine moves back to the mode it left (its
32//! rollback target, [`TimeBox::prior`]), with no server involved, so an
33//! offline device reverts too. A later change to a mode at least as strict
34//! as the target ends the box early. Any other change keeps it: the
35//! device still reverts to the target at the same time, and the change
36//! cannot set an expiry of its own. A change inside a box to a mode weaker
37//! than the target still needs a reason (and a ticket where required).
38//!
39//! A time box is a target for clock rollback: setting the clock back would
40//! keep a weaker mode alive. So the end is checked against two clocks, and
41//! the first to pass ends the box (ADR 0012):
42//!
43//! - the wall clock reaches `expires_at`; or
44//! - the monotonic clock has advanced by the box's budget (`expires_at`
45//!   minus the wall time when the box started), whatever the wall clock
46//!   says.
47//!
48//! Setting the wall clock back therefore cannot extend a box, and setting
49//! it forward only ends one early, which is the strict direction. A
50//! monotonic reading below the box's start (the agent restarted, and its
51//! monotonic clock with it) also ends the box. Carrying the remaining
52//! budget across a restart in protected storage, as ADR 0012 does for
53//! override sessions, is the agent's part (#69, #220).
54//!
55//! # Readiness gate
56//!
57//! A requested move to `enforce` can be gated on readiness (#41): the
58//! caller runs [`readiness`](crate::trace::readiness::readiness) over the
59//! recent events, and when its gate says a `readiness_ack` is required
60//! ([`ReadinessGate::ack_required`](crate::trace::readiness::ReadinessGate::ack_required)),
61//! sets [`Governance::readiness_ack_required`].
62//! The request must then carry [`Request::readiness_ack`], or it is rejected
63//! ([`TransitionError::ReadinessAckRequired`]). The gate never applies to a
64//! change that only returns a device to `enforce` from a time box that left
65//! it, nor to the revert when a box ends: restoring enforcement is never
66//! held back.
67//!
68//! Every change, requested or by expiry, gives a [`ModeChanged`] record
69//! for the `mode_changed` event (#16).

78/// One reading of the agent's two clocks.
79#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
80pub struct Now {
81    /// Wall-clock time, in seconds since 1970-01-01T00:00:00Z. On the
82    /// agent this is anchored time where it has it (ADR 0012).
83    pub wall_secs: i64,
84    /// Monotonic time, in seconds since an arbitrary origin. It only goes
85    /// forwards while the agent runs.
86    pub monotonic_secs: u64,
87}

92    /// The current reading.

96/// Whether `from` may change to `to`, ignoring the time-box requirements.
97#[must_use]
98pub const fn is_legal(from: Mode, to: Mode) -> bool {
99    use Mode::{Audit, Disabled, Enforce, Learning, Monitor};
100    match (from, to) {
101        (Disabled, Disabled)
102        | (Monitor, Monitor)
103        | (Audit, Audit)
104        | (Learning, Learning)
105        | (Enforce, Enforce)
106        | (Learning, Monitor) => false,
107        (Disabled | Monitor | Audit | Learning | Enforce, _) => true,
108    }
109}

111/// Whether a change from `from` to `to` needs a time box: leaving
112/// `enforce` for a weaker mode, or entering `disabled`.
113#[must_use]
114pub const fn needs_time_box(from: Mode, to: Mode) -> bool {
115    (matches!(from, Mode::Enforce) && !matches!(to, Mode::Enforce)) || matches!(to, Mode::Disabled)
116}

169/// Why a request was rejected. The machine is unchanged.
170#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
171pub enum TransitionError {
172    /// The request names no actor.
173    ActorRequired,
174    /// The device is already in the mode.
175    SameMode,
176    /// The change is not in the table of legal transitions.
177    Illegal {
178        /// The current mode.
179        from: Mode,
180        /// The requested mode.
181        to: Mode,
182    },
183    /// The change needs a reason that is not empty.
184    ReasonRequired,
185    /// The tenant requires a ticket reference for the change.
186    TicketRequired,
187    /// The change needs `expires_at` (or `permanent`, where allowed).
188    ExpiryRequired,
189    /// `expires_at` is not after the current wall-clock time.
190    ExpiryInPast,
191    /// `expires_at` is more than 30 days ahead.
192    ExpiryTooFar,
193    /// `permanent` is only allowed from `enforce` to `audit`, for an actor
194    /// with the elevated permission, and never with `expires_at`.
195    PermanentNotAllowed,
196    /// `expires_at` or `permanent` was given for a change that starts no
197    /// time box: one that does not need one, or one made while a time box
198    /// is already running. Time boxes on other changes are #69.
199    TimeBoxNotAllowed,
200    /// The move to `enforce` needs a `readiness_ack`: the readiness check
201    /// found more would-block events than the threshold.
202    ReadinessAckRequired,
203}
204
205impl core::fmt::Display for TransitionError {
206    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
207        match self {
208            Self::ActorRequired => f.write_str("the change needs an actor"),
209            Self::SameMode => f.write_str("the device is already in this mode"),
210            Self::Illegal { from, to } => write!(f, "{from:?} cannot change to {to:?}"),
211            Self::ReasonRequired => f.write_str("this change needs a reason"),
212            Self::TicketRequired => f.write_str("this change needs a ticket reference"),
213            Self::ExpiryRequired => f.write_str("this change needs an expiry"),
214            Self::ExpiryInPast => f.write_str("the expiry is not in the future"),
215            Self::ExpiryTooFar => f.write_str("the expiry is more than 30 days ahead"),
216            Self::PermanentNotAllowed => f.write_str("this change cannot be permanent"),
217            Self::TimeBoxNotAllowed => f.write_str("this change cannot set an expiry"),
218            Self::ReadinessAckRequired => {
219                f.write_str("the move to enforce needs a readiness acknowledgement")
220            }
221        }
222    }
223}

227/// A weaker mode that ends by itself.
228#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
229pub struct TimeBox {
230    /// The mode to return to: the rollback target.
231    pub prior: Mode,
232    /// When it ends, in wall-clock seconds.
233    pub expires_at: i64,
234    /// How long it lasts on the monotonic clock, in seconds.
235    pub budget_secs: u64,
236    /// The monotonic time it started at.
237    pub started_monotonic: u64,
238}
239
240impl TimeBox {
241    /// Whether the box has ended at `now`: the wall clock has reached
242    /// `expires_at`, or the monotonic clock has run for the budget or gone
243    /// backwards.
244    #[must_use]
245    pub const fn has_ended(&self, now: Now) -> bool {
246        now.wall_secs >= self.expires_at
247            || now.monotonic_secs < self.started_monotonic
248            || now.monotonic_secs - self.started_monotonic >= self.budget_secs
249    }
250}

261/// One mode change, for the `mode_changed` event (#16). The scope
262/// (device or group) is added by the caller (#37, #38).
263#[derive(Clone, Debug, PartialEq, Eq, Hash)]
264pub struct ModeChanged {
265    /// The mode before.
266    pub old: Mode,
267    /// The mode after.
268    pub new: Mode,
269    /// Why it changed.
270    pub cause: Cause,
271    /// Who asked; `None` on expiry.
272    pub actor: Option<String>,
273    /// The request's reason.
274    pub reason: Option<String>,
275    /// The request's ticket reference.
276    pub ticket: Option<String>,
277    /// When the new mode ends, if it is time-boxed.
278    pub expires_at: Option<i64>,
279    /// The device left `learning`, so the learning-ended summary is due
280    /// (#5, #69).
281    pub learning_ended: bool,
282    /// The change was a gated move to `enforce`, acknowledged with
283    /// `readiness_ack`.
284    pub readiness_acked: bool,
285}

287/// A device's base mode and its time box.
288#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
289pub struct ModeMachine {
290    mode: Mode,
291    time_box: Option<TimeBox>,
292}
293
294impl ModeMachine {
295    /// A machine in `mode`, with no time box.
296    #[must_use]
297    pub const fn new(mode: Mode) -> Self {
298        Self {
299            mode,
300            time_box: None,
301        }
302    }
303
304    /// The mode as last stored. It may be a time-boxed mode that has ended
305    /// but not yet been observed; use [`current`](Self::current) to read the
306    /// mode in force.
307    #[must_use]
308    pub const fn stored(&self) -> Mode {
309        self.mode
310    }
311
312    /// The running time box, if any.
313    #[must_use]
314    pub const fn time_box(&self) -> Option<&TimeBox> {
315        self.time_box.as_ref()
316    }
317
318    /// The mode in force at `now`, with the expiry record if this read is
319    /// the one that observed the time box ending.
320    ///
321    /// Observing an expiry is final: the box is closed and the rollback
322    /// target becomes the stored mode, exactly as [`tick`](Self::tick) does.
323    /// A later read with the wall clock set back, or before the monotonic
324    /// budget has run out, can therefore never bring the weaker mode back.
325    /// Reading the mode is how the expiry is noticed, so it needs `&mut`.
326    #[must_use = "the expiry record, if any, is the `mode_changed` event (#16)"]
327    pub fn current(&mut self, now: Now) -> (Mode, Option<ModeChanged>) {
328        let expired = self.tick(now);
329        (self.mode, expired)
330    }
331
332    /// Ends the time box if it has ended at `now`, and reports the change.
333    /// Once a box has ended it stays ended: the box is removed and its
334    /// rollback target is stored.
335    #[must_use = "the expiry record is the `mode_changed` event (#16)"]
336    pub fn tick(&mut self, now: Now) -> Option<ModeChanged> {
337        let b = self.time_box.filter(|b| b.has_ended(now))?;
338        let old = self.mode;
339        self.mode = b.prior;
340        self.time_box = None;
341        Some(ModeChanged {
342            old,
343            new: b.prior,
344            cause: Cause::Expired,
345            actor: None,
346            reason: None,
347            ticket: None,
348            expires_at: None,
349            learning_ended: old == Mode::Learning && b.prior != Mode::Learning,
350            readiness_acked: false,
351        })
352    }
353
354    /// Applies a request at `now`. A time box that has ended is closed
355    /// first, and the request is checked against the mode that results.
356    /// Returns every change, in order: the expiry, if any, then the
357    /// requested change.
358    ///
359    /// # Errors
360    ///
361    /// Returns [`TransitionError`] if the request is not allowed. The
362    /// machine is then unchanged, even if a time box has ended; call
363    /// [`tick`](Self::tick) for that.
364    pub fn transition(
365        &mut self,
366        request: &Request,
367        governance: &Governance,
368        now: Now,
369    ) -> Result<Vec<ModeChanged>, TransitionError> {
370        let mut next = *self;
371        let mut changes: Vec<ModeChanged> = next.tick(now).into_iter().collect();
372        let change = next.apply(request, governance, now)?;
373        changes.push(change);
374        *self = next;
375        Ok(changes)
376    }
377
378    fn apply(
379        &mut self,
380        r: &Request,
381        governance: &Governance,
382        now: Now,
383    ) -> Result<ModeChanged, TransitionError> {
384        let from = self.mode;
385        if r.actor.trim().is_empty() {
386            return Err(TransitionError::ActorRequired);
387        }
388        if from == r.to {
389            return Err(TransitionError::SameMode);
390        }
391        if !is_legal(from, r.to) {
392            return Err(TransitionError::Illegal { from, to: r.to });
393        }
394        // Returning to `enforce` from a box that left it restores
395        // enforcement, and is never held back by the gate.
396        let restores = self.time_box.is_some_and(|b| b.prior == Mode::Enforce);
397        let gated = r.to == Mode::Enforce && governance.readiness_ack_required && !restores;
398        if gated && !r.readiness_ack {
399            return Err(TransitionError::ReadinessAckRequired);
400        }
401        // A change that needs a time box starts one, unless one is already
402        // running: then the device keeps its rollback target and end.
403        let boxed = needs_time_box(from, r.to) && self.time_box.is_none();
404        let wants_box = r.expires_at.is_some() || r.permanent;
405        if !boxed && wants_box {
406            return Err(TransitionError::TimeBoxNotAllowed);
407        }
408        // Weakening needs a reason, also inside a running box.
409        let weakening = needs_time_box(from, r.to)
410            || self
411                .time_box
412                .is_some_and(|b| strictness(r.to) < strictness(b.prior));
413        if weakening {
414            if r.reason.as_deref().is_none_or(|s| s.trim().is_empty()) {
415                return Err(TransitionError::ReasonRequired);
416            }
417            if governance.require_ticket && r.ticket.as_deref().is_none_or(|s| s.trim().is_empty())
418            {
419                return Err(TransitionError::TicketRequired);
420            }
421        }
422        let time_box = match (boxed, r.expires_at, r.permanent) {
423            (false, _, _) => None,
424            (true, Some(_), true) => return Err(TransitionError::PermanentNotAllowed),
425            (true, None, true) => {
426                if from == Mode::Enforce && r.to == Mode::Audit && governance.may_weaken_permanently
427                {
428                    None
429                } else {
430                    return Err(TransitionError::PermanentNotAllowed);
431                }
432            }
433            (true, None, false) => return Err(TransitionError::ExpiryRequired),
434            (true, Some(at), false) => {
435                let ahead = at.saturating_sub(now.wall_secs);
436                if ahead <= 0 {
437                    return Err(TransitionError::ExpiryInPast);
438                }
439                if ahead > MAX_TIME_BOX_SECS {
440                    return Err(TransitionError::ExpiryTooFar);
441                }
442                Some(TimeBox {
443                    prior: from,
444                    expires_at: at,
445                    budget_secs: ahead.unsigned_abs(),
446                    started_monotonic: now.monotonic_secs,
447                })
448            }
449        };
450        // A change to a mode at least as strict as the rollback target
451        // ends the running box; any other change keeps it.
452        if let Some(b) = self.time_box {
453            if strictness(r.to) >= strictness(b.prior) {
454                self.time_box = None;
455            }
456        } else {
457            self.time_box = time_box;
458        }
459        self.mode = r.to;
460        Ok(ModeChanged {
461            old: from,
462            new: r.to,
463            cause: Cause::Requested,
464            actor: Some(r.actor.clone()),
465            reason: r.reason.clone(),
466            ticket: r.ticket.clone(),
467            expires_at: self.time_box.map(|b| b.expires_at),
468            learning_ended: from == Mode::Learning,
469            readiness_acked: gated,
470        })
471    }
472}
