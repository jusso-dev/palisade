1//! The options of a policy's precedence model (#35).
2//!
3//! [`PrecedenceModel`](super::PrecedenceModel) chooses how the matching
4//! rules of the ordinary rule sets are combined. [`PrecedenceOptions`] holds
5//! the settings that tune each model. Every option has a default, and an
6//! option that the selected model does not use is ignored, with the warning
7//! `precedence_option_unused`. The semantics are in [`crate::precedence`].
8
9use schemars::JsonSchema;
10use serde::{Deserialize, Serialize};
11
12/// Settings of the precedence models. Each is optional in a document.
13#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, JsonSchema)]
14#[serde(deny_unknown_fields)]
15pub struct PrecedenceOptions {
16    /// Which allow decides when several allow rules of different kinds
17    /// could (`deny_wins`, `allow_wins` and `specificity`). The default,
18    /// `most_restrictive`, ranks `allow_contained` over `allow` over
19    /// `allow_elevated`.
20    #[serde(default, skip_serializing_if = "AllowTiebreak::is_default")]
21    pub allow_tiebreak: AllowTiebreak,
22    /// `allow_wins` only: within one rule set, a rule with a more specific
23    /// identity beats a less specific one before allow-over-deny applies,
24    /// so a denied file in an allowed folder stays denied. Off by default.
25    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
26    pub granularity_tiebreak: bool,
27    /// `first_match` only: rules of curated (read-only) rule sets are
28    /// evaluated before every custom rule, whatever their `order`. On by
29    /// default.
30    #[serde(default = "default_true", skip_serializing_if = "is_true")]
31    pub curated_first: bool,
32    /// `specificity` only: which action wins between rules of the same
33    /// specificity. `deny` by default.
34    #[serde(default, skip_serializing_if = "TierTiebreak::is_default")]
35    pub same_tier_tiebreak: TierTiebreak,
36    /// `priority` only: whether the lowest or the highest `priority`
37    /// number wins. `lower_wins` by default.
38    #[serde(default, skip_serializing_if = "PriorityDirection::is_default")]
39    pub priority_direction: PriorityDirection,
40}
41
42impl Default for PrecedenceOptions {
43    fn default() -> Self {
44        Self {
45            allow_tiebreak: AllowTiebreak::default(),
46            granularity_tiebreak: false,
47            curated_first: true,
48            same_tier_tiebreak: TierTiebreak::default(),
49            priority_direction: PriorityDirection::default(),
50        }
51    }
52}
53
54impl PrecedenceOptions {
55    /// Whether every option has its default.
56    #[must_use]
57    pub fn is_default(&self) -> bool {
58        self == &Self::default()
59    }
60}
61
62const fn default_true() -> bool {
63    true
64}

66/// serde's `skip_serializing_if` passes a reference.
67const fn is_true(b: &bool) -> bool {
68    *b
69}

71/// Which allow decides among allows of different kinds.
72#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize, JsonSchema)]
73#[serde(rename_all = "snake_case")]
74pub enum AllowTiebreak {
75    /// `allow_contained`, then `allow`, then `allow_elevated` (with
76    /// `condition: always` before `if_required`). The default.
77    #[default]
78    MostRestrictive,
79    /// The reverse: `allow_elevated` (`if_required` before `always`), then
80    /// `allow`, then `allow_contained`.
81    MostPermissive,
82}
83
84impl AllowTiebreak {
85    const fn is_default(&self) -> bool {
86        matches!(self, Self::MostRestrictive)
87    }
88}

90/// Which action wins between rules of the same specificity.
91#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize, JsonSchema)]
92#[serde(rename_all = "snake_case")]
93pub enum TierTiebreak {
94    /// A deny beats an allow. The default.
95    #[default]
96    Deny,
97    /// An allow beats a deny.
98    Allow,
99}
100
101impl TierTiebreak {
102    const fn is_default(&self) -> bool {
103        matches!(self, Self::Deny)
104    }
105}

107/// Which `priority` number wins.
108#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize, JsonSchema)]
109#[serde(rename_all = "snake_case")]
110pub enum PriorityDirection {
111    /// The lowest number wins. The default.
112    #[default]
113    LowerWins,
114    /// The highest number wins.
115    HigherWins,
116}
117
118impl PriorityDirection {
119    const fn is_default(&self) -> bool {
120        matches!(self, Self::LowerWins)
121    }
