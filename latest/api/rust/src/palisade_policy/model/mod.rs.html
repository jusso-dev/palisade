1//! The canonical policy object model (#30).
2//!
3//! A [`PolicyDocument`] is the unit that is stored, exchanged and versioned.
4//! It holds one [`Policy`] and the [`RuleSet`]s that the policy references by
5//! ID. Every type here maps one-to-one to the JSON Schema in
6//! `schemas/policy/v1.json` ([`crate::schema`]).
7//!
8//! Deserialising gives a structurally valid document. It is not yet safe to
9//! evaluate: [`PolicyDocument::validate`] checks limits, references and
10//! field types, and returns the [`ValidatedPolicy`](crate::validate::ValidatedPolicy)
11//! that later stages accept.
12//!
13//! All structs reject unknown fields (strict mode). Enums are `snake_case`
14//! strings.

94/// The decision part of a policy: which rule sets apply, and how.
95#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]

98    /// Stable ID. Renaming the policy never changes it.
99    pub id: PolicyId,
100    /// Display name: 1 to 128 characters, trimmed, no control characters.
101    /// Uniqueness within a tenant is checked by the server (#10).
102    #[schemars(length(min = 1, max = 128))]
103    pub name: String,
104    /// Free-text description, at most 4096 characters.
105    #[serde(default, skip_serializing_if = "Option::is_none")]
106    #[schemars(length(max = 4096))]
107    pub description: Option<String>,
108    /// Monotonic revision, raised by the server on every change (#10).
109    pub revision: u64,
110    /// How matching rules are combined (#35, ADR 0004).
111    #[serde(default)]
112    pub precedence_model: PrecedenceModel,
113    /// Settings of the precedence models (#35). An option the selected
114    /// model does not use is ignored, with a warning.
115    #[serde(default, skip_serializing_if = "PrecedenceOptions::is_default")]
116    pub precedence_options: PrecedenceOptions,
117    /// The base enforcement mode (#36). `disabled` is not allowed: only a
118    /// device can be disabled, with a reason and an expiry.
119    pub mode: Mode,
120    /// The modes per object type and while disconnected (#36).
121    #[serde(default, skip_serializing_if = "ModeOptions::is_default")]
122    pub mode_options: ModeOptions,
123    /// Typed settings. Every field has a documented default.
124    #[serde(default)]
125    pub settings: PolicySettings,
126    /// What happens when no rule decides (#39). The default blocks every
127    /// unknown object.
128    #[serde(default, skip_serializing_if = "UnknownPolicy::is_default")]
129    pub unknown: UnknownPolicy,
130    /// Governance settings: justification, review and expiry rules, the
131    /// policy assistant's severities and counter reporting (#40).
132    #[serde(default, skip_serializing_if = "GovernanceSettings::is_default")]
133    pub governance: GovernanceSettings,
134    /// The rule sets that apply, in order, by ID.
135    #[serde(default, skip_serializing_if = "Vec::is_empty")]
136    #[schemars(length(max = 1000))]
137    pub rule_sets: Vec<RuleSetId>,
138    /// Who, where and when the whole policy applies (#38). Every rule must
139    /// also be in this scope.
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    pub scope: Option<Scope>,
142    /// The scopes of rule set assignments (#38): each rule of the set must
143    /// also be in every one of them. At most one entry per rule set.
144    #[serde(default, skip_serializing_if = "Vec::is_empty")]
145    #[schemars(length(max = 1000))]
146    pub rule_set_scopes: Vec<RuleSetScope>,
147    /// Named network locations that scopes test with `location_profile`
148    /// (#38).
149    #[serde(default, skip_serializing_if = "Vec::is_empty")]

151    pub location_profiles: Vec<LocationProfile>,
152    /// Governance metadata (#40).
153    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
154    pub metadata: Metadata,
155}

157/// How the matching rules of the ordinary rule sets are combined (#35).
158///
159/// The mandatory-deny tier and the implicit deny for unmatched files apply
160/// under every model (ADR 0004). The semantics of each model are in
161/// [`crate::precedence`].
162#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
163#[serde(rename_all = "snake_case")]
164pub enum PrecedenceModel {
165    /// Any matching deny blocks; otherwise any matching allow allows. The default.
166    #[default]
167    DenyWins,
168    /// Any matching allow allows. Not recommended for Essential Eight.
169    AllowWins,
170    /// The rule with the most specific identity wins.
171    Specificity,
172    /// Rules are ordered, and the first match decides.
173    FirstMatch,
174    /// The rule with the winning numeric priority decides.
175    Priority,
176}

178/// A named, reusable, ordered collection of rules.
179#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
180#[serde(deny_unknown_fields)]
181pub struct RuleSet {
182    /// Stable ID.
183    pub id: RuleSetId,
184    /// Display name, with the same rules as a policy name.
185    #[schemars(length(min = 1, max = 128))]
186    pub name: String,
187    /// Free-text description, at most 4096 characters.
188    #[serde(default, skip_serializing_if = "Option::is_none")]
189    #[schemars(length(max = 4096))]
190    pub description: Option<String>,
191    /// The content version of this rule set, for display (for example `3.0`).
192    #[serde(default, skip_serializing_if = "Option::is_none")]
193    #[schemars(length(min = 1, max = 128))]
194    pub version: Option<String>,
195    /// What the rule set is for.
196    pub kind: RuleSetKind,
197    /// Only for `blocklist`: evaluated in the non-overridable mandatory-deny
198    /// tier (#35, ADR 0004).
199    #[serde(default)]
200    pub mandatory: bool,
201    /// The operating system the rules are written for.
202    #[serde(default)]
203    pub os: Os,
204    /// Curated sets are read-only: edit a duplicate instead.
205    #[serde(default)]
206    pub read_only: bool,
207    /// Hidden from default console views (staging).
208    #[serde(default)]
209    pub hidden: bool,
210    /// Only for a blocklist that is not mandatory: the mode its rules
211    /// decide in, replacing the mode for the object's type, for example to
212    /// try a new blocklist in `audit` (#36). `disabled` is not allowed.
213    #[serde(default, skip_serializing_if = "Option::is_none")]
214    pub mode_override: Option<Mode>,
215    /// Whether a group below the one that assigns this rule set may disable
216    /// one of its rules with `suppress_inherited` (#37). Unset means `true`
217    /// for an allowlist or a baseline and `false` for every other kind. A
218    /// mandatory rule set can never be suppressed (`suppress_not_allowed`).
219    #[serde(default, skip_serializing_if = "Option::is_none")]
220    pub allow_child_suppress: Option<bool>,
221    /// The category, as `Category` or `Category/Subcategory`: each part 1 to
222    /// 64 characters, trimmed, with no `/` or control characters. Assigning
223    /// a category assigns every rule set library in it (#37).
224    #[serde(default, skip_serializing_if = "Option::is_none")]
225    #[schemars(regex(pattern = r"^[^/]+(/[^/]+)?$"))]
226    pub category_path: Option<String>,
227    /// The `first_match` band of the hierarchy level that assigned this rule
228    /// set (#37): rules of a lower band come first, whatever their `order`.
229    /// The hierarchy writes it into an effective policy and replaces any
230    /// value an authored policy gives, so a local administrator cannot move
231    /// a rule set ahead of global policy. 0, the default, outside a
232    /// hierarchy.
233    #[serde(default, skip_serializing_if = "is_zero")]
234    pub band: u8,
235    /// The rules, in order.
236    #[serde(default, skip_serializing_if = "Vec::is_empty")]
237    pub rules: Vec<Rule>,
238    /// Governance metadata (#40).
239    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
240    pub metadata: Metadata,
241}

449/// A document could not be read.
450#[derive(Debug)]
451pub enum LoadError {
452    /// The input is not JSON, or does not match the model.
453    Json(serde_json::Error),
454    /// `schema_version` is missing or not a version string.
455    MissingSchemaVersion,
456    /// The document's major version is not one this reader implements.
457    UnsupportedSchemaVersion(SchemaVersion),
458}
459
460impl core::fmt::Display for LoadError {
461    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
462        match self {
463            Self::Json(e) => write!(f, "invalid policy document: {e}"),
464            Self::MissingSchemaVersion => {
465                f.write_str("invalid policy document: missing or invalid `schema_version`")
466            }
467            Self::UnsupportedSchemaVersion(v) => write!(
468                f,
469                "unsupported schema version {v}: this reader implements {}",
470                crate::schema::CURRENT
471            ),
472        }
473    }
474}

476impl std::error::Error for LoadError {
477    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
478        match self {
479            Self::Json(e) => Some(e),
480            _ => None,
481        }
482    }
483}

485impl PolicyDocument {
486    /// Reads a document from JSON in strict mode.
487    ///
488    /// The schema version is checked first, so a document from another major
489    /// version gets a version error rather than a confusing field error.
490    /// Unknown fields are rejected, including fields from a newer minor
491    /// version (a lenient reader is future work in #30).
492    ///
493    /// # Errors
494    ///
495    /// Returns [`LoadError`] if the input is not a document this reader
496    /// understands.
497    pub fn from_json(input: &str) -> Result<Self, LoadError> {
498        #[derive(Deserialize)]
499        struct Header {
500            schema_version: Option<serde_json::Value>,
501        }
502        let header: Header = serde_json::from_str(input).map_err(LoadError::Json)?;
503        let version: SchemaVersion = header
504            .schema_version
505            .and_then(|v| v.as_str()?.parse().ok())
506            .ok_or(LoadError::MissingSchemaVersion)?;
507        if !version.is_readable_by(crate::schema::CURRENT) {
508            return Err(LoadError::UnsupportedSchemaVersion(version));
509        }
510        serde_json::from_str(input).map_err(LoadError::Json)
511    }
512
513    /// Writes the document as pretty-printed JSON.
514    ///
515    /// # Errors
516    ///
517    /// Only fails if serialisation itself fails, which the model's types do
518    /// not do.
519    pub fn to_json_pretty(&self) -> Result<String, serde_json::Error> {
520        serde_json::to_string_pretty(self)
521    }
