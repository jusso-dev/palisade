1//! The canonical policy object model (#30).
2//!
3//! A [`PolicyDocument`] is the unit that is stored, exchanged and versioned.
4//! It holds one [`Policy`] and the [`RuleSet`]s that the policy references by
5//! ID. Every type here maps one-to-one to the JSON Schema in
6//! `schemas/policy/v1.json` ([`crate::schema`]).
7//!
8//! Deserialising gives a structurally valid document. It is not yet safe to
9//! evaluate: [`PolicyDocument::validate`] checks limits, references and
10//! field types, and returns the [`ValidatedPolicy`](crate::validate::ValidatedPolicy)
11//! that later stages accept.
12//!
13//! All structs reject unknown fields (strict mode). Enums are `snake_case`
14//! strings.

73/// A versioned policy document: one policy and the rule sets it uses.
74#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
75#[serde(deny_unknown_fields)]
76pub struct PolicyDocument {
77    /// The version of this schema the document was written for. A reader
78    /// accepts the same major version and any minor version.
79    pub schema_version: SchemaVersion,
80    /// The policy.
81    pub policy: Policy,
82    /// The rule sets the policy references, in any order. The policy's
83    /// `rule_sets` list gives the order that applies.
84    #[serde(default, skip_serializing_if = "Vec::is_empty")]
85    #[schemars(length(max = 1000))]
86    pub rule_sets: Vec<RuleSet>,
87}

89/// The decision part of a policy: which rule sets apply, and how.
90#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
91#[serde(deny_unknown_fields)]
92pub struct Policy {
93    /// Stable ID. Renaming the policy never changes it.
94    pub id: PolicyId,
95    /// Display name: 1 to 128 characters, trimmed, no control characters.
96    /// Uniqueness within a tenant is checked by the server (#10).
97    #[schemars(length(min = 1, max = 128))]
98    pub name: String,
99    /// Free-text description, at most 4096 characters.
100    #[serde(default, skip_serializing_if = "Option::is_none")]
101    #[schemars(length(max = 4096))]
102    pub description: Option<String>,
103    /// Monotonic revision, raised by the server on every change (#10).
104    pub revision: u64,
105    /// How matching rules are combined (#35, ADR 0004).
106    #[serde(default)]
107    pub precedence_model: PrecedenceModel,
108    /// Settings of the precedence models (#35). An option the selected
109    /// model does not use is ignored, with a warning.
110    #[serde(default, skip_serializing_if = "PrecedenceOptions::is_default")]
111    pub precedence_options: PrecedenceOptions,
112    /// The base enforcement mode (#36). `disabled` is not allowed: only a
113    /// device can be disabled, with a reason and an expiry.
114    pub mode: Mode,
115    /// The modes per object type and while disconnected (#36).
116    #[serde(default, skip_serializing_if = "ModeOptions::is_default")]
117    pub mode_options: ModeOptions,
118    /// Typed settings. Every field has a documented default.
119    #[serde(default)]
120    pub settings: PolicySettings,
121    /// What happens when no rule decides (#39). The default blocks every
122    /// unknown object.
123    #[serde(default, skip_serializing_if = "UnknownPolicy::is_default")]
124    pub unknown: UnknownPolicy,
125    /// The rule sets that apply, in order, by ID.
126    #[serde(default, skip_serializing_if = "Vec::is_empty")]
127    #[schemars(length(max = 1000))]
128    pub rule_sets: Vec<RuleSetId>,
129    /// Who, where and when the whole policy applies (#38). Every rule must
130    /// also be in this scope.
131    #[serde(default, skip_serializing_if = "Option::is_none")]
132    pub scope: Option<Scope>,
133    /// The scopes of rule set assignments (#38): each rule of the set must
134    /// also be in every one of them. At most one entry per rule set.
135    #[serde(default, skip_serializing_if = "Vec::is_empty")]
136    #[schemars(length(max = 1000))]
137    pub rule_set_scopes: Vec<RuleSetScope>,
138    /// Named network locations that scopes test with `location_profile`
139    /// (#38).
140    #[serde(default, skip_serializing_if = "Vec::is_empty")]
141    #[schemars(length(max = 64))]
142    pub location_profiles: Vec<LocationProfile>,
143    /// Governance metadata (#40).
144    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
145    pub metadata: Metadata,
146}

148/// How the matching rules of the ordinary rule sets are combined (#35).
149///
150/// The mandatory-deny tier and the implicit deny for unmatched files apply
151/// under every model (ADR 0004). The semantics of each model are in
152/// [`crate::precedence`].
153#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
154#[serde(rename_all = "snake_case")]
155pub enum PrecedenceModel {
156    /// Any matching deny blocks; otherwise any matching allow allows. The default.
157    #[default]
158    DenyWins,
159    /// Any matching allow allows. Not recommended for Essential Eight.
160    AllowWins,
161    /// The rule with the most specific identity wins.
162    Specificity,
163    /// Rules are ordered, and the first match decides.
164    FirstMatch,
165    /// The rule with the winning numeric priority decides.
166    Priority,
167}

169/// A named, reusable, ordered collection of rules.
170#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
171#[serde(deny_unknown_fields)]
172pub struct RuleSet {
173    /// Stable ID.
174    pub id: RuleSetId,
175    /// Display name, with the same rules as a policy name.
176    #[schemars(length(min = 1, max = 128))]
177    pub name: String,
178    /// Free-text description, at most 4096 characters.
179    #[serde(default, skip_serializing_if = "Option::is_none")]
180    #[schemars(length(max = 4096))]
181    pub description: Option<String>,
182    /// The content version of this rule set, for display (for example `3.0`).
183    #[serde(default, skip_serializing_if = "Option::is_none")]
184    #[schemars(length(min = 1, max = 128))]
185    pub version: Option<String>,
186    /// What the rule set is for.
187    pub kind: RuleSetKind,
188    /// Only for `blocklist`: evaluated in the non-overridable mandatory-deny
189    /// tier (#35, ADR 0004).
190    #[serde(default)]
191    pub mandatory: bool,
192    /// The operating system the rules are written for.
193    #[serde(default)]
194    pub os: Os,
195    /// Curated sets are read-only: edit a duplicate instead.
196    #[serde(default)]
197    pub read_only: bool,
198    /// Hidden from default console views (staging).
199    #[serde(default)]
200    pub hidden: bool,
201    /// Only for a blocklist that is not mandatory: the mode its rules
202    /// decide in, replacing the mode for the object's type, for example to
203    /// try a new blocklist in `audit` (#36). `disabled` is not allowed.
204    #[serde(default, skip_serializing_if = "Option::is_none")]
205    pub mode_override: Option<Mode>,
206    /// Whether a group below the one that assigns this rule set may disable
207    /// one of its rules with `suppress_inherited` (#37). Unset means `true`
208    /// for an allowlist or a baseline and `false` for every other kind. A
209    /// mandatory rule set can never be suppressed (`suppress_not_allowed`).
210    #[serde(default, skip_serializing_if = "Option::is_none")]
211    pub allow_child_suppress: Option<bool>,
212    /// The category, as `Category` or `Category/Subcategory`: each part 1 to
213    /// 64 characters, trimmed, with no `/` or control characters. Assigning
214    /// a category assigns every rule set library in it (#37).
215    #[serde(default, skip_serializing_if = "Option::is_none")]
216    #[schemars(regex(pattern = r"^[^/]+(/[^/]+)?$"))]
217    pub category_path: Option<String>,
218    /// The `first_match` band of the hierarchy level that assigned this rule
219    /// set (#37): rules of a lower band come first, whatever their `order`.
220    /// The hierarchy writes it into an effective policy and replaces any
221    /// value an authored policy gives, so a local administrator cannot move
222    /// a rule set ahead of global policy. 0, the default, outside a
223    /// hierarchy.
224    #[serde(default, skip_serializing_if = "is_zero")]
225    pub band: u8,
226    /// The rules, in order.
227    #[serde(default, skip_serializing_if = "Vec::is_empty")]
228    pub rules: Vec<Rule>,
229    /// Governance metadata (#40).
230    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
231    pub metadata: Metadata,
232}

264/// One rule: a match tree and the action taken when it matches.
265#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
266#[serde(deny_unknown_fields)]
267pub struct Rule {
268    /// Stable ID, never reused.
269    pub id: RuleId,
270    /// Human short ID from the tenant's sequence, `R-` and 6 to 12 digits.
271    #[serde(default, skip_serializing_if = "Option::is_none")]
272    #[schemars(regex(pattern = r"^R-[0-9]{6,12}$"))]
273    pub display_id: Option<String>,
274    /// Display name, with the same rules as a policy name.
275    #[schemars(length(min = 1, max = 128))]
276    pub name: String,
277    /// Free-text description, at most 4096 characters.
278    #[serde(default, skip_serializing_if = "Option::is_none")]
279    #[schemars(length(max = 4096))]
280    pub description: Option<String>,
281    /// Disabled rules are kept, but never decide.
282    #[serde(default = "default_true")]
283    pub enabled: bool,
284    /// The object classes the rule applies to. Empty means every class.
285    #[serde(default, skip_serializing_if = "Vec::is_empty")]
286    pub object_types: Vec<ObjectType>,
287    /// When the rule matches (#33).
288    #[serde(rename = "match")]
289    pub match_: Match,
290    /// What happens when it matches (#34).
291    pub action: Action,
292    /// Logging, upload, notification, severity and tags (#34). Unset
293    /// fields take the action's defaults.
294    #[serde(default, skip_serializing_if = "Effects::is_empty")]
295    pub effects: Effects,
296    /// Position for the `first_match` model (#35): 0 to 1000000, lowest
297    /// first. Rules of custom (not read-only) rule sets use 100001 to
298    /// 999989; the other ranges are reserved for system, curated and
299    /// catch-all rules.
300    #[serde(default, skip_serializing_if = "Option::is_none")]
301    #[schemars(range(max = 1_000_000))]
302    pub order: Option<u32>,
303    /// Priority for the `priority` model (#35).
304    #[serde(default, skip_serializing_if = "Option::is_none")]
305    pub priority: Option<i32>,
306    /// Per-rule enforcement override (#36). Only for `deny` rules.
307    #[serde(default)]
308    pub enforcement_override: EnforcementOverride,
309    /// Who, where and when the rule applies (#38). Unset means everyone,
310    /// everywhere, always, within the policy's and the rule set's scopes.
311    #[serde(default, skip_serializing_if = "Option::is_none")]
312    pub scope: Option<Scope>,
313    /// Governance metadata (#40).
314    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
315    pub metadata: Metadata,
316}

370/// A document could not be read.
371#[derive(Debug)]
372pub enum LoadError {
373    /// The input is not JSON, or does not match the model.
374    Json(serde_json::Error),
375    /// `schema_version` is missing or not a version string.
376    MissingSchemaVersion,
377    /// The document's major version is not one this reader implements.
378    UnsupportedSchemaVersion(SchemaVersion),
379}
380
381impl core::fmt::Display for LoadError {
382    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
383        match self {
384            Self::Json(e) => write!(f, "invalid policy document: {e}"),
385            Self::MissingSchemaVersion => {
386                f.write_str("invalid policy document: missing or invalid `schema_version`")
387            }
388            Self::UnsupportedSchemaVersion(v) => write!(
389                f,
390                "unsupported schema version {v}: this reader implements {}",
391                crate::schema::CURRENT
392            ),
393        }
394    }
395}

397impl std::error::Error for LoadError {
398    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
399        match self {
400            Self::Json(e) => Some(e),
401            _ => None,
402        }
403    }

406impl PolicyDocument {
407    /// Reads a document from JSON in strict mode.
408    ///
409    /// The schema version is checked first, so a document from another major
410    /// version gets a version error rather than a confusing field error.
411    /// Unknown fields are rejected, including fields from a newer minor
412    /// version (a lenient reader is future work in #30).
413    ///
414    /// # Errors
415    ///
416    /// Returns [`LoadError`] if the input is not a document this reader
417    /// understands.
418    pub fn from_json(input: &str) -> Result<Self, LoadError> {
419        #[derive(Deserialize)]
420        struct Header {
421            schema_version: Option<serde_json::Value>,
422        }
423        let header: Header = serde_json::from_str(input).map_err(LoadError::Json)?;
424        let version: SchemaVersion = header
425            .schema_version
426            .and_then(|v| v.as_str()?.parse().ok())
427            .ok_or(LoadError::MissingSchemaVersion)?;
428        if !version.is_readable_by(crate::schema::CURRENT) {
429            return Err(LoadError::UnsupportedSchemaVersion(version));
430        }
431        serde_json::from_str(input).map_err(LoadError::Json)
432    }
433
434    /// Writes the document as pretty-printed JSON.
435    ///
436    /// # Errors
437    ///
438    /// Only fails if serialisation itself fails, which the model's types do
439    /// not do.
440    pub fn to_json_pretty(&self) -> Result<String, serde_json::Error> {
441        serde_json::to_string_pretty(self)
442    }
