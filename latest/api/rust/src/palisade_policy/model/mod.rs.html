1//! The canonical policy object model (#30).
2//!
3//! A [`PolicyDocument`] is the unit that is stored, exchanged and versioned.
4//! It holds one [`Policy`] and the [`RuleSet`]s that the policy references by
5//! ID. Every type here maps one-to-one to the JSON Schema in
6//! `schemas/policy/v1.json` ([`crate::schema`]).
7//!
8//! Deserialising gives a structurally valid document. It is not yet safe to
9//! evaluate: [`PolicyDocument::validate`] checks limits, references and
10//! field types, and returns the [`ValidatedPolicy`](crate::validate::ValidatedPolicy)
11//! that later stages accept.
12//!
13//! All structs reject unknown fields (strict mode); [`crate::lenient`]
14//! reads newer minor versions. Enums are `snake_case`
15//! strings.

95/// The decision part of a policy: which rule sets apply, and how.
96#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
97#[serde(deny_unknown_fields)]
98pub struct Policy {
99    /// Stable ID. Renaming the policy never changes it.
100    pub id: PolicyId,
101    /// Display name: 1 to 128 characters, trimmed, no control characters.
102    /// Uniqueness within a tenant is checked by the server (#10).
103    #[schemars(length(min = 1, max = 128))]
104    pub name: String,
105    /// Free-text description, at most 4096 characters.
106    #[serde(default, skip_serializing_if = "Option::is_none")]
107    #[schemars(length(max = 4096))]
108    pub description: Option<String>,
109    /// Monotonic revision, raised by the server on every change (#10).
110    pub revision: u64,
111    /// How matching rules are combined (#35, ADR 0004).
112    #[serde(default)]
113    pub precedence_model: PrecedenceModel,
114    /// Settings of the precedence models (#35). An option the selected
115    /// model does not use is ignored, with a warning.
116    #[serde(default, skip_serializing_if = "PrecedenceOptions::is_default")]
117    pub precedence_options: PrecedenceOptions,
118    /// The base enforcement mode (#36). `disabled` is not allowed: only a
119    /// device can be disabled, with a reason and an expiry.
120    pub mode: Mode,
121    /// The modes per object type and while disconnected (#36).
122    #[serde(default, skip_serializing_if = "ModeOptions::is_default")]
123    pub mode_options: ModeOptions,
124    /// Typed settings. Every field has a documented default.
125    #[serde(default)]
126    pub settings: PolicySettings,
127    /// What happens when no rule decides (#39). The default blocks every
128    /// unknown object.
129    #[serde(default, skip_serializing_if = "UnknownPolicy::is_default")]
130    pub unknown: UnknownPolicy,
131    /// Governance settings: justification, review and expiry rules, the
132    /// policy assistant's severities and counter reporting (#40).
133    #[serde(default, skip_serializing_if = "GovernanceSettings::is_default")]
134    pub governance: GovernanceSettings,
135    /// The rule sets that apply, in order, by ID.
136    #[serde(default, skip_serializing_if = "Vec::is_empty")]
137    #[schemars(length(max = 1000))]
138    pub rule_sets: Vec<RuleSetId>,
139    /// Who, where and when the whole policy applies (#38). Every rule must
140    /// also be in this scope.
141    #[serde(default, skip_serializing_if = "Option::is_none")]
142    pub scope: Option<Scope>,
143    /// The scopes of rule set assignments (#38): each rule of the set must
144    /// also be in every one of them. At most one entry per rule set.
145    #[serde(default, skip_serializing_if = "Vec::is_empty")]
146    #[schemars(length(max = 1000))]
147    pub rule_set_scopes: Vec<RuleSetScope>,
148    /// Named network locations that scopes test with `location_profile`
149    /// (#38).
150    #[serde(default, skip_serializing_if = "Vec::is_empty")]
151    #[schemars(length(max = 64))]
152    pub location_profiles: Vec<LocationProfile>,
153    /// The confirmation a catch-all rule that lets objects run needs
154    /// (#35, #329): exactly [`CONFIRM_PERMIT_ALL`]. Without it, such a rule
155    /// is rejected (`permit_all_unconfirmed`).
156    #[serde(default, skip_serializing_if = "Option::is_none")]
157    #[schemars(length(min = 1, max = 128))]
158    pub confirm_permit_all: Option<String>,
159    /// Governance metadata (#40).
160    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
161    pub metadata: Metadata,
162}

168/// How the matching rules of the ordinary rule sets are combined (#35).
169///
170/// The mandatory-deny tier and the implicit deny for unmatched files apply
171/// under every model (ADR 0004). The semantics of each model are in
172/// [`crate::precedence`].
173#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
174#[serde(rename_all = "snake_case")]
175pub enum PrecedenceModel {
176    /// Any matching deny blocks; otherwise any matching allow allows. The default.
177    #[default]
178    DenyWins,
179    /// Any matching allow allows. Not recommended for Essential Eight.
180    AllowWins,
181    /// The rule with the most specific identity wins.
182    Specificity,
183    /// Rules are ordered, and the first match decides.
184    FirstMatch,
185    /// The rule with the winning numeric priority decides.
186    Priority,
187}

189/// A named, reusable, ordered collection of rules.
190#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
191#[serde(deny_unknown_fields)]
192pub struct RuleSet {
193    /// Stable ID.
194    pub id: RuleSetId,
195    /// Display name, with the same rules as a policy name.
196    #[schemars(length(min = 1, max = 128))]
197    pub name: String,
198    /// Free-text description, at most 4096 characters.
199    #[serde(default, skip_serializing_if = "Option::is_none")]
200    #[schemars(length(max = 4096))]
201    pub description: Option<String>,
202    /// The content version of this rule set, for display (for example `3.0`).
203    #[serde(default, skip_serializing_if = "Option::is_none")]
204    #[schemars(length(min = 1, max = 128))]
205    pub version: Option<String>,
206    /// What the rule set is for.
207    pub kind: RuleSetKind,
208    /// Only for `blocklist`: evaluated in the non-overridable mandatory-deny
209    /// tier (#35, ADR 0004).
210    #[serde(default)]
211    pub mandatory: bool,
212    /// The operating system the rules are written for.
213    #[serde(default)]
214    pub os: Os,
215    /// Curated sets are read-only: edit a duplicate instead.
216    #[serde(default)]
217    pub read_only: bool,
218    /// Hidden from default console views (staging).
219    #[serde(default)]
220    pub hidden: bool,
221    /// Only for a blocklist that is not mandatory: the mode its rules
222    /// decide in, replacing the mode for the object's type, for example to
223    /// try a new blocklist in `audit` (#36). `disabled` is not allowed.
224    #[serde(default, skip_serializing_if = "Option::is_none")]
225    pub mode_override: Option<Mode>,
226    /// Whether a group below the one that assigns this rule set may disable
227    /// one of its rules with `suppress_inherited` (#37). Unset means `true`
228    /// for an allowlist or a baseline and `false` for every other kind. A
229    /// mandatory rule set can never be suppressed (`suppress_not_allowed`).
230    #[serde(default, skip_serializing_if = "Option::is_none")]
231    pub allow_child_suppress: Option<bool>,
232    /// The category, as `Category` or `Category/Subcategory`: each part 1 to
233    /// 64 characters, trimmed, with no `/` or control characters. Assigning
234    /// a category assigns every rule set library in it (#37).
235    #[serde(default, skip_serializing_if = "Option::is_none")]
236    #[schemars(regex(pattern = r"^[^/]+(/[^/]+)?$"))]
237    pub category_path: Option<String>,
238    /// The `first_match` band of the hierarchy level that assigned this rule
239    /// set (#37): rules of a lower band come first, whatever their `order`.
240    /// The hierarchy writes it into an effective policy and replaces any
241    /// value an authored policy gives, so a local administrator cannot move
242    /// a rule set ahead of global policy. 0, the default, outside a
243    /// hierarchy.
244    #[serde(default, skip_serializing_if = "is_zero")]
245    pub band: u8,
246    /// The rules, in order.
247    #[serde(default, skip_serializing_if = "Vec::is_empty")]
248    pub rules: Vec<Rule>,
249    /// Governance metadata (#40).
250    #[serde(default, skip_serializing_if = "Metadata::is_empty")]
251    pub metadata: Metadata,
252}

285#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
286#[serde(deny_unknown_fields)]
287#[schemars(transform = empty_match_only_for_catch_all)]
288pub struct Rule {
289    /// Stable ID, never reused.
290    pub id: RuleId,
291    /// Human short ID from the tenant's sequence, `R-` and 6 to 12 digits.
292    #[serde(default, skip_serializing_if = "Option::is_none")]
293    #[schemars(regex(pattern = r"^R-[0-9]{6,12}$"))]
294    pub display_id: Option<String>,
295    /// Display name, with the same rules as a policy name.
296    #[schemars(length(min = 1, max = 128))]
297    pub name: String,
298    /// Free-text description, at most 4096 characters.
299    #[serde(default, skip_serializing_if = "Option::is_none")]
300    #[schemars(length(max = 4096))]
301    pub description: Option<String>,
302    /// Disabled rules are kept, but never decide.
303    #[serde(default = "default_true")]
304    pub enabled: bool,
305    /// The rule is absent before this instant (#40).
306    #[serde(default, skip_serializing_if = "Option::is_none")]
307    pub activates_at: Option<Timestamp>,
308    /// The rule is absent from this instant on (#40). An expired allow
309    /// never revives, even if the clock is set back
310    /// ([`crate::lifecycle`]).
311    #[serde(default, skip_serializing_if = "Option::is_none")]
312    pub expires_at: Option<Timestamp>,
313    /// What happens when the rule expires (#40).
314    #[serde(default, skip_serializing_if = "OnExpire::is_default")]
315    pub on_expire: OnExpire,
316    /// The object classes the rule applies to. Empty means every class.
317    #[serde(default, skip_serializing_if = "Vec::is_empty")]
318    pub object_types: Vec<ObjectType>,
319    /// When the rule matches (#33).
320    #[serde(rename = "match")]
321    pub match_: Match,
322    /// A catch-all rule (#33, #35, #329): the default of its rule set, which
323    /// decides only when no other rule of stage 4 matched, under every
324    /// precedence model. It is the only rule that may have no identity
325    /// anchor, and the only one whose `match` may be an empty `all`, which
326    /// matches every object. At most one per rule set; its name starts
327    /// with `Default` or contains `Catch-all`; it allows, denies or audits;
328    /// and it is never in a mandatory rule set. Unset means `false`.
329    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
330    pub catch_all: bool,
331    /// What happens when it matches (#34).
332    pub action: Action,
333    /// Logging, upload, notification, severity and tags (#34). Unset
334    /// fields take the action's defaults.
335    #[serde(default, skip_serializing_if = "Effects::is_empty")]
336    pub effects: Effects,
337    /// Position for the `first_match` model (#35): 0 to 1000000, lowest
338    /// first. Rules of custom (not read-only) rule sets use 100001 to
339    /// 999989; the other ranges are reserved for system, curated and
340    /// catch-all rules.
341    #[serde(default, skip_serializing_if = "Option::is_none")]
342    #[schemars(range(max = 1_000_000))]
343    pub order: Option<u32>,
344    /// Priority for the `priority` model (#35).
345    #[serde(default, skip_serializing_if = "Option::is_none")]
346    pub priority: Option<i32>,
347    /// Per-rule enforcement override (#36). Only for `deny` rules.
348    #[serde(default)]
349    pub enforcement_override: EnforcementOverride,
350    /// Who, where and when the rule applies (#38). Unset means everyone,
351    /// everywhere, always, within the policy's and the rule set's scopes.
352    // Boxed: most rules have none, and a large hash set has tens of
353    // thousands of rules, so an inline scope would dominate its memory (#42).
354    #[serde(default, skip_serializing_if = "Option::is_none")]
355    pub scope: Option<Box<Scope>>,
356    /// Governance metadata (#40).
357    #[serde(default, skip_serializing_if = "RuleMetadata::is_empty")]
358    pub metadata: RuleMetadata,
359}

365/// Adds to the generated schema of [`Rule`] what the derive cannot say: a
366/// `catch_all` rule's `match` may also be an empty `all`, which matches
367/// every object (#329). Everywhere else a group needs a child.
368fn empty_match_only_for_catch_all(schema: &mut schemars::Schema) {
369    let empty_all = serde_json::json!({
370        "type": "object",
371        "properties": { "all": { "type": "array", "maxItems": 0 } },
372        "required": ["all"],
373        "additionalProperties": false
374    });
375    if let Some(serde_json::Value::Object(properties)) = schema.get_mut("properties")
376        && let Some(tree) = properties.remove("match")
377    {
378        properties.insert(
379            "match".to_owned(),
380            serde_json::json!({ "anyOf": [tree, empty_all.clone()] }),
381        );
382    }
383    let rule = serde_json::json!({
384        "if": {
385            "properties": { "catch_all": { "const": true } },
386            "required": ["catch_all"]
387        },
388        "then": true,
389        "else": { "properties": { "match": { "not": empty_all } } }
390    });
391    schema.insert("allOf".to_owned(), serde_json::Value::Array(vec![rule]));
392}

510/// A rule's [`Metadata`], kept out of line and only when it is set (#42).
511///
512/// Reads and writes go through `Deref` and `DerefMut`, so it is used like a
513/// `Metadata`; writing to an empty one allocates it. Most rules of a large
514/// hash set carry no metadata, and an inline `Metadata` on each would
515/// dominate the set's memory. Its JSON form and schema are `Metadata`'s.
516#[derive(Clone, Default)]
517pub struct RuleMetadata(Option<Box<Metadata>>);
518
519/// What an empty [`RuleMetadata`] reads as.
520static EMPTY_METADATA: std::sync::LazyLock<Metadata> = std::sync::LazyLock::new(Metadata::default);
521
522impl RuleMetadata {
523    /// Whether no field is set.
524    #[must_use]
525    pub fn is_empty(&self) -> bool {
526        self.0.as_deref().is_none_or(Metadata::is_empty)
527    }
528}
529
530impl From<Metadata> for RuleMetadata {
531    fn from(m: Metadata) -> Self {
532        Self((!m.is_empty()).then(|| Box::new(m)))
533    }
534}

536impl core::ops::Deref for RuleMetadata {
537    type Target = Metadata;
538    fn deref(&self) -> &Metadata {
539        self.0.as_deref().unwrap_or(&EMPTY_METADATA)
540    }

590/// A document could not be read.
591#[derive(Debug)]
592pub enum LoadError {
593    /// The input is not JSON, or does not match the model.
594    Json(serde_json::Error),
595    /// `schema_version` is missing or not a version string.
596    MissingSchemaVersion,
597    /// The document's major version is not one this reader implements.
598    UnsupportedSchemaVersion(SchemaVersion),
599}
600
601impl core::fmt::Display for LoadError {
602    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
603        match self {
604            Self::Json(e) => write!(f, "invalid policy document: {e}"),
605            Self::MissingSchemaVersion => {
606                f.write_str("invalid policy document: missing or invalid `schema_version`")
607            }
608            Self::UnsupportedSchemaVersion(v) => write!(
609                f,
610                "unsupported schema version {v}: this reader implements {}",
611                crate::schema::CURRENT
612            ),
613        }
614    }
615}
616
617impl std::error::Error for LoadError {
618    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
619        match self {
620            Self::Json(e) => Some(e),
621            _ => None,
622        }
623    }
624}
625
626impl PolicyDocument {
627    /// Reads a document from JSON in strict mode.
628    ///
629    /// The schema version is checked first, so a document from another major
630    /// version gets a version error rather than a confusing field error.
631    /// Unknown fields are rejected, including fields from a newer minor
632    /// version. This is the mode for authoring. An agent that receives a
633    /// newer minor reads it with [`crate::lenient::read_json`], which skips
634    /// only what can never relax a decision.
635    ///
636    /// # Errors
637    ///
638    /// Returns [`LoadError`] if the input is not a document this reader
639    /// understands.
640    pub fn from_json(input: &str) -> Result<Self, LoadError> {
641        #[derive(Deserialize)]
642        struct Header {
643            schema_version: Option<serde_json::Value>,
644        }
645        let header: Header = serde_json::from_str(input).map_err(LoadError::Json)?;
646        let version: SchemaVersion = header
647            .schema_version
648            .and_then(|v| v.as_str()?.parse().ok())
649            .ok_or(LoadError::MissingSchemaVersion)?;
650        if !version.is_readable_by(crate::schema::CURRENT) {
651            return Err(LoadError::UnsupportedSchemaVersion(version));
652        }
653        serde_json::from_str(input).map_err(LoadError::Json)
654    }
655
656    /// Writes the document as pretty-printed JSON.
657    ///
658    /// # Errors
659    ///
660    /// Only fails if serialisation itself fails, which the model's types do
661    /// not do.
662    pub fn to_json_pretty(&self) -> Result<String, serde_json::Error> {
663        serde_json::to_string_pretty(self)
664    }
665}
