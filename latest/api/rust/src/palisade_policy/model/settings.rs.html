51/// Settings that apply to the whole policy. Every field is optional in a
52/// document and has the default shown in the schema.
53#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
54#[serde(default, deny_unknown_fields)]
55pub struct PolicySettings {
56    /// Control level for scripts of a kind not listed in `script_control_by_kind`.
57    pub script_control: ControlLevel,
58    /// Control level per script kind, overriding `script_control`.
59    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
60    #[serde(deserialize_with = "unique_keys")]
61    #[schemars(transform = script_kind_keys)]
62    pub script_control_by_kind: BTreeMap<ScriptKind, ControlLevel>,
63    /// Control level for libraries.
64    pub dll_control: ControlLevel,
65    /// Control level for drivers.
66    pub driver_control: ControlLevel,
67    /// Whether the agent records process command lines. Off by default for
68    /// privacy (#18).
69    pub capture_command_line: bool,
70    /// Agent check-in interval, in seconds.
71    #[schemars(range(min = POLL_INTERVAL_MIN, max = POLL_INTERVAL_MAX))]
72    pub poll_interval_secs: u32,
73    /// How long a decision may be cached, in seconds. `0` disables the cache.
74    #[schemars(range(max = DECISION_CACHE_TTL_MAX))]
75    pub decision_cache_ttl_secs: u32,
76    /// Durations, in minutes, offered for override sessions (#9). At most 5
77    /// slots, each at most 10080. `0` marks a disabled slot.
78    #[serde(skip_serializing_if = "Vec::is_empty")]
79    #[schemars(length(max = OVERRIDE_SLOTS_MAX), inner(range(max = OVERRIDE_MINUTES_MAX)))]
80    pub override_durations: Vec<u32>,
81    /// Which clock schedules use, unless a schedule sets its own (#38).
82    pub time_zone_basis: TimeZoneBasis,
83    /// How old cached group memberships that are not in the token
84    /// (cloud-directory groups) may be before scopes treat them as unknown,
85    /// in seconds (#38). From 3600 to 86400.
86    #[schemars(range(min = GROUP_CACHE_MAX_AGE_MIN, max = GROUP_CACHE_MAX_AGE_MAX))]
87    pub group_cache_max_age_secs: u32,
88    /// How many ancestors process criteria may inspect (#31).
89    #[schemars(range(min = ANCESTOR_DEPTH_MIN, max = ANCESTOR_DEPTH_MAX))]
90    pub max_ancestor_depth: u8,
91    /// Most criteria in one rule's match tree (#33): 1 to 32, the fixed
92    /// maximum and the default. A policy can tighten the limit, never
93    /// raise it.
94    #[schemars(range(min = 1, max = CRITERIA_PER_RULE_MAX))]
95    #[serde(skip_serializing_if = "is_max_criteria_per_rule_default")]
96    pub max_criteria_per_rule: u8,
97    /// Deepest match tree (#33), where a single criterion has depth 1: 1 to
98    /// 8, the fixed maximum and the default.
99    #[schemars(range(min = 1, max = TREE_DEPTH_MAX))]
100    #[serde(skip_serializing_if = "is_max_tree_depth_default")]
101    pub max_tree_depth: u8,
102    /// Most values in one criterion (#33): 1 to 25, the fixed maximum and
103    /// the default.
104    #[schemars(range(min = 1, max = VALUES_PER_CRITERION_MAX))]
105    #[serde(skip_serializing_if = "is_max_values_per_criterion_default")]
106    pub max_values_per_criterion: u8,
107    /// Engineering settings, checked against a registry of known keys. The
108    /// registry is empty in this version, so any key is rejected unless
109    /// `allow_unknown_advanced` is set. Keys are added only with the feature
110    /// that reads them (#330), and never relax a decision.
111    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
112    #[serde(deserialize_with = "unique_keys")]
113    #[schemars(transform = advanced_keys)]
114    pub advanced: BTreeMap<String, AdvancedValue>,
115    /// Accept keys in `advanced` that the registry does not know.
116    pub allow_unknown_advanced: bool,
117}

119/// Most bytes in an `advanced` key.
120pub const ADVANCED_KEY_MAX_BYTES: usize = 128;
121
122/// The schema of `advanced` keys: printable ASCII identifiers. Validation
123/// checks the same rule ([`is_valid_advanced_key`]).
124pub const ADVANCED_KEY_PATTERN: &str = "^[a-z][a-z0-9_.-]{0,127}$";
125
126/// Whether `key` is a well-formed `advanced` key: a lowercase ASCII letter,
127/// then up to 127 lowercase letters, digits, `_`, `.` or `-`.
128#[must_use]
129pub fn is_valid_advanced_key(key: &str) -> bool {
130    let mut bytes = key.bytes();
131    bytes.next().is_some_and(|b| b.is_ascii_lowercase())
132        && key.len() <= ADVANCED_KEY_MAX_BYTES
133        && bytes.all(|b| {
134            b.is_ascii_lowercase() || b.is_ascii_digit() || matches!(b, b'_' | b'.' | b'-')
135        })
136}

138/// Reads a map and rejects repeated keys. serde's own map reader keeps the
139/// last value, which would let two tools disagree about one document.
140pub(crate) fn unique_keys<'de, D, K, V>(deserializer: D) -> Result<BTreeMap<K, V>, D::Error>
141where
142    D: serde::Deserializer<'de>,
143    K: Deserialize<'de> + Ord + core::fmt::Debug,
144    V: Deserialize<'de>,
145{
146    struct Visitor<K, V>(core::marker::PhantomData<(K, V)>);
147    impl<'de, K, V> serde::de::Visitor<'de> for Visitor<K, V>
148    where
149        K: Deserialize<'de> + Ord + core::fmt::Debug,
150        V: Deserialize<'de>,
151    {
152        type Value = BTreeMap<K, V>;
153        fn expecting(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
154            f.write_str("a map with unique keys")
155        }
156        fn visit_map<A: serde::de::MapAccess<'de>>(
157            self,
158            mut map: A,
159        ) -> Result<Self::Value, A::Error> {
160            let mut out = BTreeMap::new();
161            while let Some((key, value)) = map.next_entry::<K, V>()? {
162                if out.contains_key(&key) {
163                    return Err(serde::de::Error::custom(format_args!(
164                        "duplicate key {key:?}"
165                    )));
166                }
167                out.insert(key, value);
168            }
169            Ok(out)
170        }
171    }
172    deserializer.deserialize_map(Visitor(core::marker::PhantomData))
173}

175// schemars describes a map's values but not its keys, and its `length`
176// attribute does not apply to maps, so these add the missing constraints.
177
178fn script_kind_keys(schema: &mut schemars::Schema) {
179    let names: Vec<&str> = ScriptKind::ALL.iter().map(|k| k.as_str()).collect();
180    schema.insert(
181        "propertyNames".to_owned(),
182        serde_json::json!({ "enum": names }),
183    );
184}
185
186fn advanced_keys(schema: &mut schemars::Schema) {
187    schema.insert("maxProperties".to_owned(), ADVANCED_KEYS_MAX.into());
188    schema.insert(
189        "propertyNames".to_owned(),
190        serde_json::json!({ "pattern": ADVANCED_KEY_PATTERN }),
191    );
192}

217/// Whether a class of object is controlled.
218#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
219#[serde(rename_all = "snake_case")]
220pub enum ControlLevel {
221    /// Not evaluated.
222    Off,
223    /// Evaluated, and would-block decisions are reported only.
224    Audit,
225    /// Evaluated and enforced.
226    Enforce,
227}
228
229/// The clock that schedules are interpreted in.
230#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
231#[serde(rename_all = "snake_case")]
232pub enum TimeZoneBasis {
233    /// The device's local time zone.
234    DeviceLocal,
235    /// Coordinated Universal Time.
236    Utc,
237    /// The time zone of the session's user, if the agent can read it, and
238    /// otherwise the device's (#38).
239    UserLocal,
240}

242/// A scalar value in `advanced`.
243#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
244#[serde(untagged)]
245pub enum AdvancedValue {
246    /// A boolean.
247    Bool(bool),
248    /// A signed integer.
249    Int(i64),
250    /// A string of at most 2048 bytes, with no control characters.
251    Text(#[schemars(length(max = 2048))] String),
252}
