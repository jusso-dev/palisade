1//! Presets: match trees for the recommended rule forms (#31).
2//!
3//! A preset builds an ordinary match tree, so a policy that uses one is
4//! validated, evaluated and ranked by precedence like any other.
5
6use super::{Criterion, Field, Match, Operator};
7
8/// The publisher rule `publisher_product` (#31, ISM-1471): a signer's
9/// publisher **and** a product name, optionally narrowed by the original
10/// file name and a file-version range. This is the recommended form of an
11/// allow rule: a publisher alone trusts everything the publisher signs.
12///
13/// ```
14/// use palisade_policy::model::preset::PublisherProduct;
15///
16/// let tree = PublisherProduct::new("Example Software Ltd", "Example Editor")
17///     .original_filename("editor.exe")
18///     .min_version("2.0")
19///     .into_match();
20/// ```
21#[derive(Clone, Debug, PartialEq, Eq)]
22#[must_use]
23pub struct PublisherProduct {
24    /// `signer.publisher equals`: the leaf subject's common name.
25    pub publisher: String,
26    /// `meta.product_name equals`.
27    pub product_name: String,
28    /// `meta.original_filename equals`, if set.
29    pub original_filename: Option<String>,
30    /// The lowest `meta.file_version`, inclusive, if set.
31    pub min_version: Option<String>,
32    /// The highest `meta.file_version`, inclusive, if set.
33    pub max_version: Option<String>,
34}
35
36impl PublisherProduct {
37    /// A publisher and a product name.
38    pub fn new(publisher: impl Into<String>, product_name: impl Into<String>) -> Self {
39        Self {
40            publisher: publisher.into(),
41            product_name: product_name.into(),
42            original_filename: None,
43            min_version: None,
44            max_version: None,
45        }
46    }
47
48    /// Also requires this original file name.
49    pub fn original_filename(mut self, name: impl Into<String>) -> Self {
50        self.original_filename = Some(name.into());
51        self
52    }
53
54    /// Also requires a file version of at least `version`.
55    pub fn min_version(mut self, version: impl Into<String>) -> Self {
56        self.min_version = Some(version.into());
57        self
58    }
59
60    /// Also requires a file version of at most `version`.
61    pub fn max_version(mut self, version: impl Into<String>) -> Self {
62        self.max_version = Some(version.into());
63        self
64    }
65
66    /// The match tree: an `all` of the publisher, the product name, then
67    /// the original file name and the version test that are set. Values
68    /// are trimmed of leading and trailing white space, which a copied name
69    /// often carries and a fact never matches. Values are not otherwise
70    /// checked here: validating the policy rejects an empty value or a
71    /// malformed version, as for any criterion.
72    #[must_use]
73    pub fn into_match(self) -> Match {
74        let test = |field, operator, values: Vec<String>| {
75            let values = values.into_iter().map(|v| v.trim().to_owned());
76            Match::Criterion(Criterion::new(field, operator, values))
77        };
78        let mut all = vec![
79            test(
80                Field::SignerPublisher,
81                Operator::Equals,
82                vec![self.publisher],
83            ),
84            test(
85                Field::MetaProductName,
86                Operator::Equals,
87                vec![self.product_name],
88            ),
89        ];
90        if let Some(name) = self.original_filename {
91            all.push(test(
92                Field::MetaOriginalFilename,
93                Operator::Equals,
94                vec![name],
95            ));
96        }
97        let version = match (self.min_version, self.max_version) {
98            (Some(min), Some(max)) => Some((Operator::VersionBetween, vec![min, max])),
99            (Some(min), None) => Some((Operator::VersionGe, vec![min])),
100            (None, Some(max)) => Some((Operator::VersionLe, vec![max])),
101            (None, None) => None,
102        };
103        if let Some((operator, values)) = version {
104            all.push(test(Field::MetaFileVersion, operator, values));
105        }
106        Match::All(all)
107    }
108}
