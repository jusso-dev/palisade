1//! Match trees and identity criteria.
2//!
3//! This module defines the shape of a match tree (#30, #33), the identity
4//! fields a criterion can test with their strength (#31), the context fields
5//! that describe the user, device and session (#33), and the operators with
6//! their compatibility table (#32). How a criterion is matched against facts
7//! is in [`crate::ops`], and how a whole tree is evaluated is in
8//! [`crate::condition`]. New fields and operators are added as enum variants.
9
10use schemars::JsonSchema;
11use serde::{Deserialize, Serialize};

13/// A condition tree (#33).
14///
15/// Each node is an object with exactly one key: `all`, `any`, `not`,
16/// `criterion` or `ref`.
17#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
18#[serde(rename_all = "snake_case", deny_unknown_fields)]
19pub enum Match {
20    /// Matches when every child matches. Needs at least one child.
21    All(#[schemars(length(min = 1))] Vec<Match>),
22    /// Matches when any child matches. Needs at least one child.
23    Any(#[schemars(length(min = 1))] Vec<Match>),
24    /// Matches when the child does not.
25    Not(Box<Match>),
26    /// One test of one fact.
27    Criterion(Criterion),
28    /// A reference to a reusable match fragment or application definition
29    /// in a library (#37), by ID and an optional version pin. A policy
30    /// document that is evaluated never holds one: the hierarchy replaces
31    /// every reference with the tree it names when it builds the effective
32    /// policy, and validation rejects one that is left (`unresolved_ref`).
33    Ref(crate::library::LibraryRef),
34}
35
36impl Match {
37    /// The nesting depth: a single criterion has depth 1.
38    ///
39    /// Iterative, so a hostile, very deep tree cannot overflow the stack.
40    #[must_use]
41    pub fn depth(&self) -> usize {
42        let mut deepest = 0;
43        let mut stack = vec![(self, 1usize)];
44        while let Some((node, depth)) = stack.pop() {
45            deepest = deepest.max(depth);
46            for child in node.children() {
47                stack.push((child, depth.saturating_add(1)));
48            }
49        }
50        deepest
51    }
52
53    /// The direct children of this node.
54    #[must_use]
55    pub fn children(&self) -> &[Match] {
56        match self {
57            Self::All(children) | Self::Any(children) => children,
58            Self::Not(child) => core::slice::from_ref(child),
59            Self::Criterion(_) | Self::Ref(_) => &[],
60        }
61    }
62}
63
64// The derived `Drop` would recurse through `Box` and `Vec`. JSON input is
65// bounded by serde_json's recursion limit, but trees built in code or by
66// other deserialisers are not, so drop iteratively.
67impl Drop for Match {
68    fn drop(&mut self) {
69        let mut stack = Vec::new();
70        take_children(self, &mut stack);
71        while let Some(mut node) = stack.pop() {
72            take_children(&mut node, &mut stack);
73        }
74    }
75}
76
77/// Moves a node's children onto `stack`, leaving the node childless.
78fn take_children(node: &mut Match, stack: &mut Vec<Match>) {
79    match node {
80        Match::All(children) | Match::Any(children) => stack.append(children),
81        Match::Not(child) => stack.push(core::mem::replace(child.as_mut(), Match::All(Vec::new()))),
82        Match::Criterion(_) | Match::Ref(_) => {}
83    }
84}

86/// A test of one fact about the object or its context: `field operator
87/// values`.
88///
89/// Values are combined with OR for positive operators and with AND for
90/// negative ones (#32): `not_contains [a, b]` means neither `a` nor `b`.
91#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
92#[serde(deny_unknown_fields)]
93#[schemars(transform = values_by_operator)]
94pub struct Criterion {
95    /// The fact being tested.
96    pub field: Field,
97    /// How it is compared.
98    pub operator: Operator,
99    /// 1 to 25 values. Each is at most 2048 bytes of UTF-8; the schema can
100    /// only bound the length in characters, so validation checks the bytes.
101    /// `is_true` and `is_false` take no values, and the key may be left out.
102    #[serde(default, skip_serializing_if = "Vec::is_empty")]
103    #[schemars(length(max = 25), inner(length(min = 1, max = 2048)))]
104    pub values: Vec<String>,
105    /// Compare strings case-sensitively. Unset means the field's default
106    /// (#32), which is case-insensitive for every field defined so far.
107    /// Only text, paths, file names, account names and tags have a case;
108    /// other fields (hashes, versions, SIDs, group sets, yes-or-no facts and
109    /// session types) reject this option.
110    #[serde(default, skip_serializing_if = "Option::is_none")]
111    pub case_sensitive: Option<bool>,
112    /// Only for `regex` and `not_regex`: `false` lets the pattern match
113    /// anywhere in the fact instead of the whole fact (#32). Unset means
114    /// `true`.
115    #[serde(default, skip_serializing_if = "Option::is_none")]
116    pub anchored: Option<bool>,
117    /// Only for positive tests of `path.full`: `true` lets a value that
118    /// names no alternate data stream also match a stream of a file it
119    /// matches, so `C:\\Tools\\*` matches `C:\\Tools\\a.exe:s` (#32).
120    /// Unset means `false`: a stream is its own object.
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub match_ads: Option<bool>,
123    /// Only for `user.ou` and `device.ou`: `true` also matches objects in
124    /// every organisational unit below the named one (#33). Unset means
125    /// `false`: the object must be directly in it.
126    #[serde(default, skip_serializing_if = "Option::is_none")]
127    pub include_sub_ous: Option<bool>,
128    /// Only in a mandatory blocklist: when the fact is unknown, this
129    /// criterion counts towards the deny applying (`match`, or `no_match`
130    /// under an odd number of `not` nodes), so the shipped blocklists fail
131    /// closed (#32, #87). Unset means `false`.
132    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
133    pub safety_critical: bool,
134}
135
136impl Criterion {
137    /// A criterion with every option unset. Build criteria with this, so
138    /// that options added in later minor versions do not break the code.
139    #[must_use]
140    pub fn new<S: Into<String>>(
141        field: Field,
142        operator: Operator,
143        values: impl IntoIterator<Item = S>,
144    ) -> Self {
145        Self {
146            field,
147            operator,
148            values: values.into_iter().map(Into::into).collect(),
149            case_sensitive: None,
150            anchored: None,
151            match_ads: None,
152            include_sub_ous: None,
153            safety_critical: false,
154        }
155    }
156}

388/// The type of a field's value, which decides the operators it accepts and
389/// how values are compared (#32).
390#[derive(Clone, Copy, Debug, PartialEq, Eq)]
391pub enum FieldClass {
392    /// A hexadecimal digest of a fixed length.
393    Hash {
394        /// The number of hex digits.
395        hex_len: usize,
396    },
397    /// A full path or a directory: normalised before comparison, and the
398    /// only class with path variables and the recursive `**` wildcard.
399    Path,
400    /// One path component, such as a file name or an extension.
401    FileName,
402    /// Free text, such as a signer name.
403    Text,
404    /// A dotted version number.
405    Version,
406    /// A principal's account name (#33): text, without regular
407    /// expressions (#32).
408    PrincipalName,
409    /// One security identifier, such as `S-1-5-32-544` (#33).
410    Sid,
411    /// A set of principals: SIDs, and cloud-directory object IDs (#33).
412    PrincipalSet,
413    /// A set of tags (#33).
414    TagSet,
415    /// A yes-or-no fact (#33).
416    Bool,
417    /// The type of a session (#33).
418    SessionType,
419    /// A whole number, such as a build number or a session ID (#32).
420    Number,
421    /// A token integrity level, ordered from `untrusted` to `protected` (#33).
422    IntegrityLevel,
423    /// A directory distinguished name (#33): compared component by component.
424    DistinguishedName,
425    /// One IP address (#32).
426    Ip,
427    /// A set of IP addresses (#32).
428    IpSet,
429    /// A set of MAC addresses (#33).
430    MacSet,
431    /// A set of DNS suffixes or names (#33).
432    NameSet,
433    /// A day of the week (#32).
434    Weekday,
435    /// A time of day (#32).
436    TimeOfDay,
437    /// A calendar date (#32).
438    Date,
439}

694/// How a criterion compares the fact with its values (#32). Positive
695/// operators match if any value matches; negative operators (`not_*`)
696/// match if no value does, and never match a fact that is unknown.
697#[derive(
698    Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
699)]
700#[serde(rename_all = "snake_case")]
701pub enum Operator {
702    /// Equal to any value.
703    Equals,
704    /// Equal to none of the values.
705    NotEquals,
706    /// Starts with any value. On paths, the value must end at a component
707    /// boundary: `C:\Tools` does not match `C:\ToolsX`.
708    StartsWith,
709    /// Ends with any value.
710    EndsWith,
711    /// Contains any value.
712    Contains,
713    /// Contains none of the values.
714    NotContains,
715    /// Matches any wildcard pattern.
716    Wildcard,
717    /// Matches none of the wildcard patterns.
718    NotWildcard,
719    /// Matches any regular expression. Anchored at both ends, and run by a
720    /// linear-time engine without look-around or backreferences.
721    Regex,
722    /// Matches none of the regular expressions.
723    NotRegex,
724    /// Version equal to any value.
725    VersionEq,
726    /// Version greater than the value.
727    VersionGt,
728    /// Version greater than or equal to the value.
729    VersionGe,
730    /// Version less than the value.
731    VersionLt,
732    /// Version less than or equal to the value.
733    VersionLe,
734    /// Version within `[min, max]`, both inclusive. Takes exactly two
735    /// values, the minimum first.
736    VersionBetween,
737    /// The set holds any of the values (#33).
738    MemberOf,
739    /// The set holds none of the values (#33).
740    NotMemberOf,
741    /// The fact is true. Takes no values (#33).
742    IsTrue,
743    /// The fact is false. Takes no values (#33). On an unknown fact it is
744    /// unknown, like every other test.
745    IsFalse,
746    /// Number, level or date equal to any value (#32).
747    Eq,
748    /// Number, level or date equal to none of the values (#32).
749    Ne,
750    /// Greater than the value (#32).
751    Gt,
752    /// Greater than or equal to the value (#32).
753    Ge,
754    /// Less than the value (#32).
755    Lt,
756    /// Less than or equal to the value (#32).
757    Le,
758    /// Within `[min, max]`, both inclusive. Takes exactly two values, the
759    /// minimum first (#32).
760    Between,
761    /// The address is in any of the CIDR blocks, such as `10.0.0.0/8` or
762    /// `2001:db8::/32` (#32). On a set of addresses, any address may be.
763    IpInCidr,
764    /// The address is in any of the ranges, written `first-last` or, for
765    /// IPv4, `a.b.c.first-last` (#32).
766    IpBetween,
767    /// The IPv4 address matches any whole-octet pattern, such as `10.*.*.*`
768    /// (#32).
769    IpWildcard,
770    /// The set holds at least one of the values (#32). The same test as
771    /// `member_of`.
772    AnyOf,
773    /// The set holds every one of the values (#32).
774    AllOf,
775    /// The day of the week is any of the values, `mon` to `sun` (#32).
776    WeekdayIn,
777    /// The time of day is within `[start, end)`, both `HH:MM`. A window
778    /// whose end is before its start wraps midnight: `22:00` to `06:00`
779    /// (#32).
780    TimeBetween,
781    /// The date is within `[from, to]`, both `YYYY-MM-DD` and inclusive
782    /// (#32).
783    DateBetween,
784}

786impl Operator {
787    /// Whether a match means the fact equals or resembles a value, as opposed
788    /// to a negative test such as `not_equals`. Only positive tests identify
789    /// an object.
790    #[must_use]
791    pub const fn is_positive(self) -> bool {
792        !matches!(
793            self,
794            Self::NotEquals
795                | Self::NotContains
796                | Self::NotWildcard
797                | Self::NotRegex
798                | Self::NotMemberOf
799                | Self::Ne
800        )
801    }
802
803    /// The positive operator that this one negates, or the operator itself
804    /// if it is positive. `not_contains` gives `contains`.
805    #[must_use]
806    pub const fn positive(self) -> Self {
807        match self {
808            Self::NotEquals => Self::Equals,
809            Self::NotContains => Self::Contains,
810            Self::NotWildcard => Self::Wildcard,
811            Self::NotRegex => Self::Regex,
812            Self::NotMemberOf => Self::MemberOf,
813            Self::Ne => Self::Eq,
814            other => other,
815        }
816    }
817
818    /// Whether the operator compares versions.
819    #[must_use]
820    pub const fn is_version(self) -> bool {
821        matches!(
822            self,
823            Self::VersionEq
824                | Self::VersionGt
825                | Self::VersionGe
826                | Self::VersionLt
827                | Self::VersionLe
828                | Self::VersionBetween
829        )
830    }
831
832    /// Whether the operator compares text: the operators of path, file
833    /// name and text fields.
834    #[must_use]

836        matches!(
837            self,
838            Self::Equals
839                | Self::NotEquals
840                | Self::StartsWith
841                | Self::EndsWith
842                | Self::Contains
843                | Self::NotContains
844                | Self::Wildcard
845                | Self::NotWildcard
846                | Self::Regex
847                | Self::NotRegex
848        )
849    }

851    /// Whether the operator orders values: `eq` to `between`.
852    #[must_use]
853    pub const fn is_ordered(self) -> bool {
854        matches!(
855            self,
856            Self::Eq | Self::Ne | Self::Gt | Self::Ge | Self::Lt | Self::Le | Self::Between
857        )
858    }

860    /// Whether the operator tests IP addresses.

864    }

866    /// Whether the operator tests membership of a set.

870            self,
871            Self::MemberOf | Self::NotMemberOf | Self::AnyOf | Self::AllOf
872        )
873    }
874
875    /// Whether the operator takes regular expressions.
876    #[must_use]
877    pub const fn is_regex(self) -> bool {
878        matches!(self, Self::Regex | Self::NotRegex)
879    }
880
881    /// The smallest and largest number of values the operator takes.
882    #[must_use]
883    pub const fn value_count(self) -> (usize, usize) {
884        match self {
885            Self::VersionGt
886            | Self::VersionGe
887            | Self::VersionLt
888            | Self::VersionLe
889            | Self::Gt
890            | Self::Ge
891            | Self::Lt
892            | Self::Le => (1, 1),
893            Self::VersionBetween | Self::Between | Self::TimeBetween | Self::DateBetween => (2, 2),
894            Self::IsTrue | Self::IsFalse => (0, 0),
895            _ => (1, super::limits::MAX_VALUES_PER_CRITERION),
896        }
897    }
898
899    /// Whether the operator is valid on a field of this class: the static
900    /// compatibility table of #32.
901    #[must_use]
902    pub const fn applies_to(self, class: FieldClass) -> bool {
903        match class {
904            FieldClass::Hash { .. } | FieldClass::Sid | FieldClass::SessionType => {
905                matches!(self, Self::Equals | Self::NotEquals)
906            }
907            FieldClass::Path | FieldClass::FileName | FieldClass::Text => self.is_text(),
908            FieldClass::PrincipalName => self.is_text() && !self.is_regex(),
909            FieldClass::Version => {
910                self.is_version() || matches!(self, Self::Equals | Self::NotEquals)
911            }
912            FieldClass::PrincipalSet
913            | FieldClass::TagSet
914            | FieldClass::NameSet
915            | FieldClass::MacSet => self.is_set(),
916            FieldClass::Bool => matches!(self, Self::IsTrue | Self::IsFalse),
917            FieldClass::Number | FieldClass::IntegrityLevel => self.is_ordered(),
918            FieldClass::Date => self.is_ordered() || matches!(self, Self::DateBetween),
919            FieldClass::DistinguishedName => matches!(self, Self::Equals | Self::NotEquals),
920            FieldClass::Ip => self.is_network() || matches!(self, Self::Equals | Self::NotEquals),
921            FieldClass::IpSet => self.is_network() || self.is_set(),
922            FieldClass::Weekday => matches!(self, Self::WeekdayIn),
923            FieldClass::TimeOfDay => matches!(self, Self::TimeBetween),
924        }
925    }

927    /// Whether the operator is valid on this field: the class table of
928    /// [`Self::applies_to`].
929    #[must_use]
930    pub const fn applies_to_field(self, field: Field) -> bool {
931        self.applies_to(field.class())
932    }

974/// Adds to the generated schema of [`Criterion`] what the derive cannot
975/// say: `is_true` and `is_false` take no values, and every other operator
976/// needs at least one. Validation checks the exact count per operator.
977fn values_by_operator(schema: &mut schemars::Schema) {
978    let rule = serde_json::json!({
979        "if": {
980            "properties": { "operator": { "enum": ["is_true", "is_false"] } },
981            "required": ["operator"]
982        },
983        "then": { "properties": { "values": { "maxItems": 0 } } },
984        "else": {
985            "properties": { "values": { "minItems": 1 } },
986            "required": ["values"]
987        }
988    });
989    schema.insert("allOf".to_owned(), serde_json::Value::Array(vec![rule]));
