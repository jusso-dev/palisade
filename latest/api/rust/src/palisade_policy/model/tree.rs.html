1//! Match trees and identity criteria.
2//!
3//! This module defines the shape of a match tree (#30, #33), the identity
4//! fields a criterion can test with their strength (#31), the context fields
5//! that describe the user, device and session (#33), and the operators with
6//! their compatibility table (#32). How a criterion is matched against facts
7//! is in [`crate::ops`], and how a whole tree is evaluated is in
8//! [`crate::condition`]. New fields and operators are added as enum variants.
9
10use schemars::JsonSchema;
11use serde::{Deserialize, Serialize};
12
13/// A condition tree (#33).
14///
15/// Each node is an object with exactly one key: `all`, `any`, `not`,
16/// `criterion` or `ref`.
17#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
18#[serde(rename_all = "snake_case", deny_unknown_fields)]
19pub enum Match {
20    /// Matches when every child matches. Needs at least one child.
21    All(#[schemars(length(min = 1))] Vec<Match>),
22    /// Matches when any child matches. Needs at least one child.
23    Any(#[schemars(length(min = 1))] Vec<Match>),
24    /// Matches when the child does not.
25    Not(Box<Match>),
26    /// One test of one fact.
27    Criterion(Criterion),
28    /// A reference to a reusable match fragment or application definition
29    /// in a library (#37), by ID and an optional version pin. A policy
30    /// document that is evaluated never holds one: the hierarchy replaces
31    /// every reference with the tree it names when it builds the effective
32    /// policy, and validation rejects one that is left (`unresolved_ref`).
33    Ref(crate::library::LibraryRef),
34}
35
36impl Match {
37    /// The nesting depth: a single criterion has depth 1.
38    ///
39    /// Iterative, so a hostile, very deep tree cannot overflow the stack.
40    #[must_use]
41    pub fn depth(&self) -> usize {
42        let mut deepest = 0;
43        let mut stack = vec![(self, 1usize)];
44        while let Some((node, depth)) = stack.pop() {
45            deepest = deepest.max(depth);
46            for child in node.children() {
47                stack.push((child, depth.saturating_add(1)));
48            }
49        }
50        deepest
51    }
52
53    /// The direct children of this node.
54    #[must_use]
55    pub fn children(&self) -> &[Match] {
56        match self {
57            Self::All(children) | Self::Any(children) => children,
58            Self::Not(child) => core::slice::from_ref(child),
59            Self::Criterion(_) | Self::Ref(_) => &[],
60        }
61    }
62}
63
64// The derived `Drop` would recurse through `Box` and `Vec`. JSON input is
65// bounded by serde_json's recursion limit, but trees built in code or by
66// other deserialisers are not, so drop iteratively.
67impl Drop for Match {
68    fn drop(&mut self) {
69        let mut stack = Vec::new();
70        take_children(self, &mut stack);
71        while let Some(mut node) = stack.pop() {
72            take_children(&mut node, &mut stack);
73        }
74    }
75}
76
77/// Moves a node's children onto `stack`, leaving the node childless.
78fn take_children(node: &mut Match, stack: &mut Vec<Match>) {
79    match node {
80        Match::All(children) | Match::Any(children) => stack.append(children),
81        Match::Not(child) => stack.push(core::mem::replace(child.as_mut(), Match::All(Vec::new()))),
82        Match::Criterion(_) | Match::Ref(_) => {}
83    }
84}

86/// A test of one fact about the object or its context: `field operator
87/// values`.
88///
89/// Values are combined with OR for positive operators and with AND for
90/// negative ones (#32): `not_contains [a, b]` means neither `a` nor `b`.
91#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
92#[serde(deny_unknown_fields)]
93#[schemars(transform = values_by_operator)]
94pub struct Criterion {
95    /// The fact being tested.
96    pub field: Field,
97    /// How it is compared.
98    pub operator: Operator,
99    /// 1 to 25 values. Each is at most 2048 bytes of UTF-8; the schema can
100    /// only bound the length in characters, so validation checks the bytes.
101    /// `is_true` and `is_false` take no values, and the key may be left out.
102    #[serde(default, skip_serializing_if = "Vec::is_empty")]
103    #[schemars(length(max = 25), inner(length(min = 1, max = 2048)))]
104    pub values: Vec<String>,
105    /// Compare strings case-sensitively. Unset means the field's default
106    /// (#32), which is case-insensitive for every field defined so far.
107    /// Only text, paths, file names, account names and tags have a case;
108    /// other fields (hashes, versions, SIDs, group sets, yes-or-no facts and
109    /// session types) reject this option.
110    #[serde(default, skip_serializing_if = "Option::is_none")]
111    pub case_sensitive: Option<bool>,
112    /// Only for `regex` and `not_regex`: `false` lets the pattern match
113    /// anywhere in the fact instead of the whole fact (#32). Unset means
114    /// `true`.
115    #[serde(default, skip_serializing_if = "Option::is_none")]
116    pub anchored: Option<bool>,
117    /// Only for positive tests of `path.full`: `true` lets a value that
118    /// names no alternate data stream also match a stream of a file it
119    /// matches, so `C:\\Tools\\*` matches `C:\\Tools\\a.exe:s` (#32).
120    /// Unset means `false`: a stream is its own object.
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub match_ads: Option<bool>,
123    /// Only for `user.ou` and `device.ou`: `true` also matches objects in
124    /// every organisational unit below the named one (#33). Unset means
125    /// `false`: the object must be directly in it.
126    #[serde(default, skip_serializing_if = "Option::is_none")]
127    pub include_sub_ous: Option<bool>,
128    /// Only in a mandatory blocklist: when the fact is unknown, this
129    /// criterion counts towards the deny applying (`match`, or `no_match`
130    /// under an odd number of `not` nodes), so the shipped blocklists fail
131    /// closed (#32, #87). Unset means `false`.
132    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
133    pub safety_critical: bool,
134    /// Only for `signer.*` fields: which signatures take part, and what
135    /// each must also satisfy (#31). The options bind to the signature the
136    /// field is read from, so `min_key_bits` on a publisher test needs the
137    /// publisher's own signature to have the key size. Unset means every
138    /// option's default.
139    #[serde(default, skip_serializing_if = "Option::is_none")]
140    pub signer: Option<Box<SignerOptions>>,
141}
142
143impl Criterion {
144    /// A criterion with every option unset. Build criteria with this, so
145    /// that options added in later minor versions do not break the code.
146    #[must_use]
147    pub fn new<S: Into<String>>(
148        field: Field,
149        operator: Operator,
150        values: impl IntoIterator<Item = S>,
151    ) -> Self {
152        Self {
153            field,
154            operator,
155            values: values.into_iter().map(Into::into).collect(),
156            case_sensitive: None,
157            anchored: None,
158            match_ads: None,
159            include_sub_ous: None,
160            safety_critical: false,
161            signer: None,
162        }
163    }
164}

166/// The options of a signer criterion (#31).
167///
168/// A signer criterion matches if any signature that takes part satisfies
169/// it. A signature takes part when its status is accepted, it comes from
170/// the chosen source and has the chosen index, and it meets every
171/// constraint here (`signed_after`, `countersigned`, `min_key_bits` and
172/// `disallowed_algorithms`). A constraint whose fact is unknown makes that
173/// signature's result unknown, never a match.
174#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
175#[serde(deny_unknown_fields)]
176pub struct SignerOptions {
177    /// Only for `signer.publisher`: the part of the leaf subject compared.
178    /// Unset means `cn`.
179    #[serde(default, skip_serializing_if = "Option::is_none")]
180    pub component: Option<SubjectComponent>,
181    /// Only for `signer.tbs_hash`: where in the chain the certificate must
182    /// be. Unset means `any`.
183    #[serde(default, skip_serializing_if = "Option::is_none")]
184    pub position: Option<TbsPosition>,
185    /// The signature statuses that can match. Unset means `valid`,
186    /// `valid_expired_timestamped` and `catalog_signed`. `digest_mismatch`
187    /// cannot be listed: see `allow_digest_mismatch`. Validation warns when
188    /// a rule that lets an object run accepts any other status.
189    #[serde(default, skip_serializing_if = "Option::is_none")]
190    #[schemars(length(min = 1, max = 6))]
191    pub accept_status: Option<Vec<crate::facts::SignerStatus>>,
192    /// Let a signature whose signed digest does not match the file take
193    /// part: the file was changed after it was signed. Validation always
194    /// warns about it. Unset means `false`.
195    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
196    pub allow_digest_mismatch: bool,
197    /// Where the signature must come from. Unset means `any`.
198    #[serde(default, skip_serializing_if = "Option::is_none")]
199    pub source: Option<SignerSource>,
200    /// Only the signature at this index, counted from 0 among the
201    /// signatures of `source`: 0 is the primary embedded signature, 1 the
202    /// first nested one. Unset means any signature.
203    #[serde(default, skip_serializing_if = "Option::is_none")]
204    #[schemars(range(max = MAX_SIGNATURE_INDEX))]
205    pub signature_index: Option<u8>,
206    /// A trusted timestamp must show that the signature was made at or
207    /// after this instant: an RFC 3339 time with an offset, such as
208    /// `2024-01-01T00:00:00Z`. A signature with no trusted timestamp is
209    /// unknown.
210    #[serde(default, skip_serializing_if = "Option::is_none")]
211    #[schemars(length(min = 1, max = 64))]
212    pub signed_after: Option<String>,
213    /// The signature must carry an RFC 3161 timestamp that verifies.
214    /// Unset means `false`: no requirement.
215    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
216    pub countersigned: bool,
217    /// Every key in the signature's chain must be at least this strong, in
218    /// RSA-equivalent bits (see [`crate::facts::KeyFact::rsa_equivalent_bits`]).
219    #[serde(default, skip_serializing_if = "Option::is_none")]
220    #[schemars(range(min = 1, max = MAX_KEY_BITS))]
221    pub min_key_bits: Option<u32>,
222    /// The signature must use none of these digest algorithms, in the
223    /// file digest, the signer information or any certificate of its chain.
224    #[serde(default, skip_serializing_if = "Vec::is_empty")]
225    #[schemars(length(max = 6))]
226    pub disallowed_algorithms: Vec<crate::facts::DigestAlgorithm>,
227}

922/// How a criterion compares the fact with its values (#32). Positive
923/// operators match if any value matches; negative operators (`not_*`)
924/// match if no value does, and never match a fact that is unknown.
925#[derive(
926    Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
927)]
928#[serde(rename_all = "snake_case")]
929pub enum Operator {
930    /// Equal to any value.
931    Equals,
932    /// Equal to none of the values.
933    NotEquals,
934    /// Starts with any value. On paths, the value must end at a component
935    /// boundary: `C:\Tools` does not match `C:\ToolsX`.
936    StartsWith,
937    /// Ends with any value.
938    EndsWith,
939    /// Contains any value.
940    Contains,
941    /// Contains none of the values.
942    NotContains,
943    /// Matches any wildcard pattern.
944    Wildcard,
945    /// Matches none of the wildcard patterns.
946    NotWildcard,
947    /// Matches any regular expression. Anchored at both ends, and run by a
948    /// linear-time engine without look-around or backreferences.
949    Regex,
950    /// Matches none of the regular expressions.
951    NotRegex,
952    /// Version equal to any value.
953    VersionEq,
954    /// Version greater than the value.
955    VersionGt,
956    /// Version greater than or equal to the value.
957    VersionGe,
958    /// Version less than the value.
959    VersionLt,
960    /// Version less than or equal to the value.
961    VersionLe,
962    /// Version within `[min, max]`, both inclusive. Takes exactly two
963    /// values, the minimum first.
964    VersionBetween,
965    /// The set holds any of the values (#33).
966    MemberOf,
967    /// The set holds none of the values (#33).
968    NotMemberOf,
969    /// The fact is true. Takes no values (#33).
970    IsTrue,
971    /// The fact is false. Takes no values (#33). On an unknown fact it is
972    /// unknown, like every other test.
973    IsFalse,
974    /// Number, level or date equal to any value (#32).
975    Eq,
976    /// Number, level or date equal to none of the values (#32).
977    Ne,
978    /// Greater than the value (#32).
979    Gt,
980    /// Greater than or equal to the value (#32).
981    Ge,
982    /// Less than the value (#32).
983    Lt,
984    /// Less than or equal to the value (#32).
985    Le,
986    /// Within `[min, max]`, both inclusive. Takes exactly two values, the
987    /// minimum first (#32).
988    Between,
989    /// The address is in any of the CIDR blocks, such as `10.0.0.0/8` or
990    /// `2001:db8::/32` (#32). On a set of addresses, any address may be.
991    IpInCidr,
992    /// The address is in any of the ranges, written `first-last` or, for
993    /// IPv4, `a.b.c.first-last` (#32).
994    IpBetween,
995    /// The IPv4 address matches any whole-octet pattern, such as `10.*.*.*`
996    /// (#32).
997    IpWildcard,
998    /// The set holds at least one of the values (#32). The same test as
999    /// `member_of`.
1000    AnyOf,
1001    /// The set holds every one of the values (#32).
1002    AllOf,
1003    /// The day of the week is any of the values, `mon` to `sun` (#32).
1004    WeekdayIn,
1005    /// The time of day is within `[start, end)`, both `HH:MM`. A window
1006    /// whose end is before its start wraps midnight: `22:00` to `06:00`
1007    /// (#32).
1008    TimeBetween,
1009    /// The date is within `[from, to]`, both `YYYY-MM-DD` and inclusive
1010    /// (#32).
1011    DateBetween,
1012}

1206/// Adds to the generated schema of [`Criterion`] what the derive cannot
1207/// say: `is_true` and `is_false` take no values, and every other operator
1208/// needs at least one. Validation checks the exact count per operator.
1209fn values_by_operator(schema: &mut schemars::Schema) {
1210    let rule = serde_json::json!({
1211        "if": {
1212            "properties": { "operator": { "enum": ["is_true", "is_false"] } },
1213            "required": ["operator"]
1214        },
1215        "then": { "properties": { "values": { "maxItems": 0 } } },
1216        "else": {
1217            "properties": { "values": { "minItems": 1 } },
1218            "required": ["values"]
1219        }
1220    });
1221    schema.insert("allOf".to_owned(), serde_json::Value::Array(vec![rule]));
