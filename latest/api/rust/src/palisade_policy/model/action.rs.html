1//! Rule actions and their options (#34).
2//!
3//! An [`Action`] says what a matching rule asks the engine to do. It is a
4//! closed set: every action is an object with a `type`, and the options of
5//! that action as optional keys. The directives that some actions carry
6//! (containment, elevation, trust source) are references and settings that
7//! the engine passes through unchanged; their meaning belongs to the
8//! epics that own them (#9, #12, #14).
9//!
10//! The directive types are typed and strict, like the rest of the model, so
11//! a misspelt option is an error rather than a silently ignored key. The
12//! owning epics add options as optional fields in a minor schema version.
13//!
14//! [`ActionKind::valid_for_object_type`] and
15//! [`ActionKind::valid_in_rule_set`] are the static validity matrix that
16//! validation applies.
17

24/// Defines a reference to an object that another epic owns, by its ID.

36            fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {

66/// The action of a rule (#34).
67///
68/// Each action is an object with a `type`, and its options as optional
69/// keys, so options can be added later without a breaking change.
70#[derive(Clone, Debug, PartialEq, Eq, Serialize, JsonSchema)]
71#[serde(tag = "type", rename_all = "snake_case", deny_unknown_fields)]
72pub enum Action {
73    /// Permit execution.
74    Allow {},
75    /// Block execution.
76    Deny(DenyOptions),
77    /// Record the match without changing the outcome. A non-terminal
78    /// annotation, used for detection and tagging.
79    Audit {},
80    /// Permit execution inside a containment profile (#14). The engine only
81    /// carries the reference.
82    AllowContained(ContainmentDirective),
83    /// Permit execution with an elevated token (#12). The engine only
84    /// carries the directive.
85    AllowElevated(ElevationDirective),
86    /// Mark the matching process as a trusted installer, updater or source
87    /// (#9, #114). This does not permit the process itself to run; an allow
88    /// rule decides that. The engine returns the directive.
89    TrustSource(TrustSourceDirective),
90    /// Skip the object entirely: no decision is logged. Only for performance
91    /// exclusions, never for executables or drivers, and validation always
92    /// warns.
93    Ignore {},
94    /// No opinion. A first-match model continues with the next rule (#35).
95    Defer {},
96}
97
98impl Action {
99    /// A deny with default options.
100    #[must_use]
101    pub fn deny() -> Self {
102        Self::Deny(DenyOptions::default())
103    }
104
105    /// Which action this is, without its options.
106    #[must_use]
107    pub const fn kind(&self) -> ActionKind {
108        match self {
109            Self::Allow {} => ActionKind::Allow,
110            Self::Deny(_) => ActionKind::Deny,
111            Self::Audit {} => ActionKind::Audit,
112            Self::AllowContained(_) => ActionKind::AllowContained,
113            Self::AllowElevated(_) => ActionKind::AllowElevated,
114            Self::TrustSource(_) => ActionKind::TrustSource,
115            Self::Ignore {} => ActionKind::Ignore,
116            Self::Defer {} => ActionKind::Defer,
117        }
118    }
119}

121/// The options of a `deny` action.
122#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
123#[serde(deny_unknown_fields)]
124pub struct DenyOptions {
125    /// Terminate already-running instances that match when the policy is
126    /// applied. Not valid for drivers or libraries.
127    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
128    pub kill_running: bool,
129    /// The end user may raise an approval request (#11). Needs a
130    /// notification, and is not valid in a mandatory rule set.
131    #[serde(default, skip_serializing_if = "core::ops::Not::not")]
132    pub allow_request: bool,
133    /// The approval request template to use. Only with `allow_request`.
134    #[serde(default, skip_serializing_if = "Option::is_none")]
135    pub request_template: Option<RequestTemplateRef>,
136    /// Whether the rule blocks when its match tree is `Unknown` because a
137    /// fact is missing or cannot be trusted (#32, #35). `no_match`, the
138    /// default, means an unknown fact does not block; `match` makes the
139    /// deny fail closed.
140    #[serde(default, skip_serializing_if = "UnknownAs::is_no_match")]
141    pub unknown_as: UnknownAs,
142    /// Whether a version fact that is not a version counts towards the
143    /// rule applying (#32, #35). With `match`, each version criterion on
144    /// such a fact is taken in whichever direction makes the deny apply,
145    /// also under `not`. The rest of the tree is evaluated as usual.
146    /// `no_match`, the default, leaves the criterion `Unknown`, which
147    /// `unknown_as` then decides.
148    #[serde(default, skip_serializing_if = "UnknownAs::is_no_match")]
149    pub on_unparseable: UnknownAs,
150}

304    /// The option keys this action accepts, besides `type`.
305    #[must_use]
306    pub const fn option_keys(self) -> &'static [&'static str] {
307        match self {
308            Self::Deny => &[
309                "kill_running",
310                "allow_request",
311                "request_template",
312                "unknown_as",
313                "on_unparseable",
314            ],
315            Self::AllowContained => &["profile_id"],
316            Self::AllowElevated => &["token_profile", "condition", "reason_prompt", "auth"],
317            Self::TrustSource => &["mode", "scope_paths", "tag_output"],
318            Self::Allow | Self::Audit | Self::Ignore | Self::Defer => &[],

322    /// Whether a matching rule with this action decides the outcome.
323    ///
324    /// `audit` rules are collected as annotations and `defer` rules have no
325    /// opinion. A `trust_source` rule only attaches its directive: whether
326    /// the process may run is decided by another rule (#114). How the
327    /// terminal rules are combined is #35.
328    #[must_use]
329    pub const fn is_terminal(self) -> bool {
330        !matches!(self, Self::Audit | Self::Defer | Self::TrustSource)
331    }
332
333    /// Whether the action lets the object run when it decides.
334    #[must_use]
335    pub const fn permits(self) -> bool {
336        matches!(
337            self,
338            Self::Allow | Self::AllowContained | Self::AllowElevated
339        )
340    }
341
342    /// Whether a rule with this action must identify the object (the
343    /// identity-anchor and weak-hash checks, ADR 0008): every action that
344    /// lets an object run, trusts it, or excludes it. A deny, an audit or
345    /// a defer only narrows or observes, so it may match broadly.
346    #[must_use]
347    pub const fn needs_identity(self) -> bool {
348        self.permits() || matches!(self, Self::TrustSource | Self::Ignore)
349    }
350
351    /// Whether the action may apply to an object type. Part of the static
352    /// validity matrix.
353    ///
354    /// - `allow_elevated`: executables, installers, scripts and control
355    ///   panel applets only. Only those start a process that can be given a
356    ///   token.
357    /// - `allow_contained`: not drivers or libraries. A driver runs in the
358    ///   kernel, and a library runs inside its host process, so neither can
359    ///   be contained on its own.
360    /// - `trust_source`: executables and installers only, the processes
361    ///   that install and update software.
362    /// - `ignore`: not executables or drivers. Whether a path is
363    ///   user-writable is only known at evaluation (#54), so an exclusion
364    ///   for code that starts a process or runs in the kernel is never
365    ///   accepted.
366    /// - Every other action applies to every type. Deny's `kill_running`
367    ///   option is checked by [`kill_running_valid_for`](Self::kill_running_valid_for).
368    #[must_use]
369    pub const fn valid_for_object_type(self, object_type: ObjectType) -> bool {
370        use ObjectType as T;
371        match self {
372            Self::AllowElevated => matches!(
373                object_type,
374                T::Executable | T::Installer | T::Script(_) | T::ControlPanelApplet
375            ),
376            Self::AllowContained => !matches!(object_type, T::Driver | T::Library),
377            Self::TrustSource => matches!(object_type, T::Executable | T::Installer),
378            Self::Ignore => !matches!(object_type, T::Executable | T::Driver),
379            Self::Allow | Self::Deny | Self::Audit | Self::Defer => true,
380        }
381    }
382
383    /// Whether deny's `kill_running` option may apply to an object type. A
384    /// driver or a library is not a process of its own, so there is nothing
385    /// to terminate.
386    #[must_use]
387    pub const fn kill_running_valid_for(object_type: ObjectType) -> bool {
388        !matches!(object_type, ObjectType::Driver | ObjectType::Library)
389    }
390
391    /// Whether the action may appear in a rule set of this kind. Part of the
392    /// static validity matrix. `audit` and `defer` are valid everywhere;
393    /// `mixed` accepts every action. A mandatory rule set must also deny
394    /// (ADR 0004).
395    #[must_use]
396    pub const fn valid_in_rule_set(self, kind: RuleSetKind) -> bool {
397        use RuleSetKind as K;
398        match self {
399            Self::Audit | Self::Defer => true,
400            Self::Allow => matches!(kind, K::Allowlist | K::Baseline | K::Mixed),
401            Self::Deny => matches!(kind, K::Blocklist | K::Mixed),
402            Self::AllowContained => matches!(kind, K::Allowlist | K::Containment | K::Mixed),
403            Self::AllowElevated => matches!(kind, K::Elevation | K::Mixed),
404            Self::TrustSource => matches!(kind, K::Trust | K::Mixed),
405            Self::Ignore => matches!(kind, K::Mixed),
406        }
407    }
408}

416/// Reads an [`Action`] key by key, as a derived struct reader does.
417///
418/// The derived reader for an internally tagged enum buffers the whole map
419/// before it looks at the keys. That accepts a sequence (`["deny"]`, read as
420/// its first item), takes the tag by variant index (`type: 1` as `deny`),
421/// and reports an unknown key only after the map is read, without that
422/// key's location. This reader checks each key as it reads it: a key that
423/// no action has, or one that the action named by an earlier `type` does
424/// not have, is reported at the key. A key that comes before `type` and
425/// turns out not to belong to the action is reported at the action.
426impl<'de> Deserialize<'de> for Action {
427    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
428        deserializer.deserialize_map(de::ActionVisitor)
429    }
430}

513    impl<'de> DeserializeSeed<'de> for KeySeed {

521    impl Visitor<'_> for KeySeed {
522        type Value = Key;
523
524        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
525            f.write_str("an action key")
526        }
527
528        fn visit_str<E: de::Error>(self, v: &str) -> Result<Key, E> {
529            let key = KEYS
530                .iter()
531                .find(|(name, _)| *name == v)
532                .map(|(_, k)| *k)
533                .ok_or_else(|| E::unknown_field(v, &KEY_NAMES))?;
534            match self.0 {
535                Some(kind) if !key.belongs_to(kind) => Err(unknown_for(v, kind)),
536                _ => Ok(key),
537            }
538        }
539    }

541    fn unknown_for<E: de::Error>(key: &str, kind: ActionKind) -> E {
542        let options = kind.option_keys();
543        if options.is_empty() {
544            E::custom(format_args!(
545                "unknown field `{key}`: action `{kind}` takes no options"
546            ))
547        } else {
548            E::custom(format_args!(
549                "unknown field `{key}` for action `{kind}`, expected one of {}",
550                options
551                    .iter()
552                    .map(|o| format!("`{o}`"))
553                    .collect::<Vec<_>>()
554                    .join(", ")
555            ))
556        }
557    }

636    impl<'de> Visitor<'de> for ActionVisitor {
637        type Value = Action;
638
639        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
640            f.write_str("a map with a `type` key naming the action")
641        }
642
643        fn visit_map<A: MapAccess<'de>>(self, mut map: A) -> Result<Action, A::Error> {
644            let mut kind: Option<ActionKind> = None;
645            let mut seen: Vec<Key> = Vec::new();
646            let mut o = Options::default();
647            while let Some(key) = map.next_key_seed(KeySeed(kind))? {
648                if seen.contains(&key) {
649                    return Err(de::Error::duplicate_field(key.name()));
650                }
651                seen.push(key);
652                let m = &mut map;
653                match key {
654                    Key::Type => kind = Some(m.next_value::<Kind>()?.0),
655                    Key::KillRunning => read(m, &mut o.kill_running)?,
656                    Key::AllowRequest => read(m, &mut o.allow_request)?,
657                    Key::RequestTemplate => read_nullable(m, &mut o.request_template)?,
658                    Key::ProfileId => read(m, &mut o.profile_id)?,
659                    Key::TokenProfile => read(m, &mut o.token_profile)?,
660                    Key::Condition => read(m, &mut o.condition)?,
661                    Key::ReasonPrompt => read_nullable(m, &mut o.reason_prompt)?,
662                    Key::Auth => read(m, &mut o.auth)?,
663                    Key::Mode => read(m, &mut o.mode)?,
664                    Key::ScopePaths => read(m, &mut o.scope_paths)?,
665                    Key::TagOutput => read(m, &mut o.tag_output)?,
666                    Key::UnknownAs => read(m, &mut o.unknown_as)?,
667                    Key::OnUnparseable => read(m, &mut o.on_unparseable)?,
668                }
669            }
670            let kind = kind.ok_or_else(|| de::Error::missing_field("type"))?;
671            // Keys read before `type` could not be checked as they came.
672            if let Some(key) = seen.iter().find(|k| !k.belongs_to(kind)) {
673                return Err(unknown_for(key.name(), kind));
674            }
675            Ok(match kind {
676                ActionKind::Allow => Action::Allow {},
677                ActionKind::Audit => Action::Audit {},
678                ActionKind::Ignore => Action::Ignore {},
679                ActionKind::Defer => Action::Defer {},
680                ActionKind::Deny => Action::Deny(DenyOptions {
681                    kill_running: o.kill_running.unwrap_or_default(),
682                    allow_request: o.allow_request.unwrap_or_default(),
683                    request_template: o.request_template,
684                    unknown_as: o.unknown_as.unwrap_or_default(),
685                    on_unparseable: o.on_unparseable.unwrap_or_default(),
686                }),
687                ActionKind::AllowContained => Action::AllowContained(ContainmentDirective {
688                    profile_id: required(o.profile_id, "profile_id")?,
689                }),
690                ActionKind::AllowElevated => Action::AllowElevated(ElevationDirective {
691                    token_profile: required(o.token_profile, "token_profile")?,
692                    condition: required(o.condition, "condition")?,
693                    reason_prompt: o.reason_prompt,
694                    auth: o.auth.unwrap_or_default(),
695                }),
696                ActionKind::TrustSource => Action::TrustSource(TrustSourceDirective {
697                    mode: required(o.mode, "mode")?,
698                    scope_paths: required(o.scope_paths, "scope_paths")?,
699                    tag_output: o.tag_output.unwrap_or_default(),
700                }),
701            })
702        }
703    }
704}
