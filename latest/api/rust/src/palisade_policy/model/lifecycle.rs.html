1//! Rule lifecycle and governance types (#40): instants and dates, what
2//! happens when a rule expires, where a rule came from, and the tenant's
3//! governance settings.
4//!
5//! Instants are written as RFC 3339 date-times and dates as RFC 3339 full
6//! dates. Both are always UTC inside the engine: an offset is applied when
7//! a value is read, and values are written with `Z`. Fractions of a second
8//! are dropped, so an instant is a whole number of seconds since
9//! 1970-01-01T00:00:00Z.

18/// Seconds in a day.
19pub const DAY_SECS: i64 = 86_400;
20
21/// Days since 1970-01-01 of a proleptic Gregorian date (H. Hinnant's
22/// `days_from_civil`).
23const fn days_from_civil(y: i64, m: i64, d: i64) -> i64 {
24    let y = if m <= 2 { y - 1 } else { y };
25    let era = if y >= 0 { y } else { y - 399 } / 400;
26    let yoe = y - era * 400;
27    let mp = (m + 9) % 12;
28    let doy = (153 * mp + 2) / 5 + d - 1;
29    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
30    era * 146_097 + doe - 719_468
31}
32
33/// The date of a day number (H. Hinnant's `civil_from_days`).
34const fn civil_from_days(z: i64) -> (i64, i64, i64) {
35    let z = z + 719_468;
36    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
37    let doe = z - era * 146_097;
38    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
39    let y = yoe + era * 400;
40    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
41    let mp = (5 * doy + 2) / 153;
42    let d = doy - (153 * mp + 2) / 5 + 1;
43    let m = if mp < 10 { mp + 3 } else { mp - 9 };
44    (if m <= 2 { y + 1 } else { y }, m, d)
45}

47const fn days_in_month(y: i64, m: i64) -> i64 {
48    match m {
49        2 if (y % 4 == 0 && y % 100 != 0) || y % 400 == 0 => 29,
50        2 => 28,
51        4 | 6 | 9 | 11 => 30,
52        _ => 31,
53    }
54}

56/// Reads exactly `n` ASCII digits.
57fn digits(s: &[u8], n: usize) -> Option<i64> {
58    let part = s.get(..n)?;
59    part.iter().try_fold(0i64, |acc, b| {
60        b.is_ascii_digit().then(|| acc * 10 + i64::from(b - b'0'))
61    })
62}
63
64/// Parses `YYYY-MM-DD` into a day number. Years 0001 to 9999.
65fn parse_date(s: &[u8]) -> Option<i64> {
66    if s.len() != 10 || s[4] != b'-' || s[7] != b'-' {
67        return None;
68    }
69    let y = digits(s, 4)?;
70    let m = digits(&s[5..], 2)?;
71    let d = digits(&s[8..], 2)?;
72    if y < 1 || !(1..=12).contains(&m) || d < 1 || d > days_in_month(y, m) {
73        return None;
74    }
75    Some(days_from_civil(y, m, d))
76}

78/// The first representable instant, 0001-01-01T00:00:00Z.
79const MIN_SECS: i64 = -62_135_596_800;
80/// The last representable instant, 9999-12-31T23:59:59Z.
81const MAX_SECS: i64 = 253_402_300_799;
82
83/// An instant: whole seconds since 1970-01-01T00:00:00Z. Only instants in
84/// years 0001 to 9999 (UTC) are read and produced, so every instant has an
85/// RFC 3339 form that reads back.
86#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
87pub struct Timestamp(pub i64);
88
89impl Timestamp {
90    /// The instant, if it is in years 0001 to 9999 UTC.
91    #[must_use]
92    pub const fn checked(secs: i64) -> Option<Self> {
93        if secs >= MIN_SECS && secs <= MAX_SECS {
94            Some(Self(secs))
95        } else {
96            None
97        }
98    }
99
100    /// The instant `days` days later, or `None` past the representable
101    /// range.
102    #[must_use]
103    pub const fn plus_days(self, days: i64) -> Option<Self> {
104        match days.checked_mul(DAY_SECS) {
105            Some(secs) => match self.0.checked_add(secs) {
106                Some(t) => Self::checked(t),
107                None => None,
108            },
109            None => None,
110        }
111    }
112
113    /// The UTC date the instant falls on.
114    #[must_use]
115    pub const fn date(self) -> Date {
116        Date(self.0.div_euclid(DAY_SECS))
117    }
118
119    /// Reads an RFC 3339 date-time: `YYYY-MM-DDTHH:MM:SS`, an optional
120    /// fraction, and `Z` or an offset `±HH:MM`. `T` and `Z` may be lower
121    /// case. Leap seconds (`:60`) are not accepted.
122    #[must_use]
123    pub fn parse(text: &str) -> Option<Self> {
124        let s = text.as_bytes();
125        let day = parse_date(s.get(..10)?)?;
126        if !matches!(s.get(10)?, b'T' | b't') || s.get(13)? != &b':' || s.get(16)? != &b':' {
127            return None;
128        }
129        let (h, mi, sec) = (
130            digits(&s[11..], 2)?,
131            digits(&s[14..], 2)?,
132            digits(&s[17..], 2)?,
133        );
134        if h > 23 || mi > 59 || sec > 59 {
135            return None;
136        }
137        let mut rest = &s[19..];
138        if let Some(frac) = rest.strip_prefix(b".") {
139            let n = frac.iter().take_while(|b| b.is_ascii_digit()).count();
140            if n == 0 {
141                return None;
142            }
143            rest = &frac[n..];
144        }
145        let offset = match rest {
146            [b'Z' | b'z'] => 0,
147            [sign @ (b'+' | b'-'), oh1, oh2, b':', om1, om2] => {
148                let oh = digits(&[*oh1, *oh2], 2)?;
149                let om = digits(&[*om1, *om2], 2)?;
150                if oh > 23 || om > 59 {
151                    return None;
152                }
153                let o = oh * 3600 + om * 60;
154                if *sign == b'+' { o } else { -o }
155            }
156            _ => return None,
157        };
158        // An offset can move the instant out of years 0001 to 9999.
159        Self::checked(day * DAY_SECS + h * 3600 + mi * 60 + sec - offset)
160    }
161}

163impl core::fmt::Display for Timestamp {
164    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
165        let (y, m, d) = civil_from_days(self.0.div_euclid(DAY_SECS));
166        let secs = self.0.rem_euclid(DAY_SECS);
167        write!(
168            f,
169            "{y:04}-{m:02}-{d:02}T{:02}:{:02}:{:02}Z",
170            secs / 3600,
171            secs % 3600 / 60,
172            secs % 60
173        )
174    }
175}

210/// A calendar date in UTC: days since 1970-01-01.
211#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
212pub struct Date(pub i64);
213
214impl Date {
215    /// The instant the day starts, clamped to the representable range.
216    #[must_use]
217    pub const fn start(self) -> Timestamp {
218        match self.0.checked_mul(DAY_SECS) {
219            Some(t) if t < MIN_SECS => Timestamp(MIN_SECS),
220            Some(t) if t > MAX_SECS => Timestamp(MAX_SECS),
221            Some(t) => Timestamp(t),
222            None if self.0 < 0 => Timestamp(MIN_SECS),
223            None => Timestamp(MAX_SECS),
224        }
225    }
226
227    /// Reads an RFC 3339 full date, `YYYY-MM-DD`.
228    #[must_use]
229    pub fn parse(text: &str) -> Option<Self> {
230        parse_date(text.as_bytes()).map(Self)
231    }
232}
233
234impl core::fmt::Display for Date {
235    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
236        let (y, m, d) = civil_from_days(self.0);
237        write!(f, "{y:04}-{m:02}-{d:02}")
238    }
239}
240
241impl Serialize for Date {
242    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
243        serializer.collect_str(self)
244    }
245}

247impl<'de> Deserialize<'de> for Date {
248    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
249        let s = String::deserialize(deserializer)?;
250        Self::parse(&s)
251            .ok_or_else(|| serde::de::Error::custom("expected a date such as 2026-11-25"))
252    }
253}

270/// What happens when a rule expires.
271#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize, JsonSchema)]
272#[serde(rename_all = "snake_case")]
273pub enum OnExpire {
274    /// The rule behaves as absent. The default.
275    #[default]
276    Deactivate,
277    /// The rule is deleted this many days after it expires (1 to 3650):
278    /// it behaves as absent at once, and [`crate::lifecycle::retire_due`]
279    /// lists it for deletion once the days have passed.
280    DeleteAfterDays(#[schemars(range(min = 1, max = 3650))] u16),
281    /// The rule becomes an `audit` rule: it still records matches, but no
282    /// longer decides.
283    ConvertToAudit,
284}
285
286impl OnExpire {
287    /// Whether this is the default.
288    #[must_use]
289    pub fn is_default(&self) -> bool {
290        *self == Self::Deactivate
291    }
292}

360/// The tenant's governance settings (#40). Every field is optional.
361#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
362#[serde(default, deny_unknown_fields)]
363pub struct GovernanceSettings {
364    /// Governance mode: allow, trust, exclusion and exception rules need a
365    /// justification, allow and trust rules a review date, and exception
366    /// rules an expiry, an approver and a risk rating. Unset means on for
367    /// a policy with a compliance preset (`unknown.preset`) and off
368    /// otherwise.
369    #[serde(skip_serializing_if = "Option::is_none")]
370    pub require_justification: Option<bool>,
371    /// A rule with no hits for this many days, while usage was observed,
372    /// is stale: 1 to 3650, default 90.
373    #[schemars(range(min = 1, max = 3650))]
374    pub stale_after_days: u16,
375    /// A rule that expires within this many days is expiring soon: 1 to
376    /// 365, default 14.
377    #[schemars(range(min = 1, max = 365))]
378    pub expiring_soon_days: u16,
379    /// When the server tells a rule's owner that it expires, in days
380    /// before the expiry: at most 8 values, each at most 365, default
381    /// `[14, 3, 0]`.
382    #[schemars(length(max = 8), inner(range(max = 365)))]
383    pub expiry_notice_days: Vec<u16>,
384    /// How often the agent reports hit counters, in hours: 1 to 168,
385    /// default 24.
386    #[schemars(range(min = 1, max = 168))]
387    pub counter_report_hours: u16,
388    /// The severity of policy-assistant findings, per finding code,
389    /// replacing the default.
390    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
391    #[serde(deserialize_with = "unique_keys")]
392    #[schemars(transform = finding_code_keys)]
393    pub severity: BTreeMap<String, FindingSeverity>,
394}
395
396impl Default for GovernanceSettings {
397    fn default() -> Self {
398        Self {
399            require_justification: None,
400            stale_after_days: 90,
401            expiring_soon_days: 14,
402            expiry_notice_days: vec![14, 3, 0],
403            counter_report_hours: 24,
404            severity: BTreeMap::new(),
405        }
406    }
407}

409impl GovernanceSettings {
410    /// Whether every setting has its default.
411    #[must_use]
412    pub fn is_default(&self) -> bool {
413        self == &Self::default()
414    }
415
416    /// Whether governance mode is on: `require_justification`, or, when
417    /// it is unset, whether the policy has a compliance preset (`preset`).
418    #[must_use]
419    pub fn mode(&self, preset: bool) -> bool {
420        self.require_justification.unwrap_or(preset)
421    }
422
423    /// The same settings with governance mode written out when the policy
424    /// gets it from its compliance preset (`preset`), so that merging
425    /// layers ([`stricter`](Self::stricter)) keeps a mode an outer layer
426    /// gets from its preset. A mode that is off is left as written, so a
427    /// hierarchy where no layer is governed merges to the same settings as
428    /// before.
429    #[must_use]
430    pub fn resolved(&self, preset: bool) -> Self {
431        Self {
432            require_justification: if self.mode(preset) {
433                Some(true)
434            } else {
435                self.require_justification
436            },
437            ..self.clone()
438        }
439    }
440
441    /// The stricter combination of `self`, the nearer layer, and `other`,
442    /// for building an effective policy (#37). Resolve each layer first
443    /// ([`resolved`](Self::resolved)): governance mode is then on if
444    /// either side has it on, by setting or by preset. Rules go stale and
445    /// expiry warnings start at the earlier point; counters report at the
446    /// shorter interval; the nearest expiry notices of either side are
447    /// sent, at most 8; and each finding's severity is the more serious
448    /// one, a side without an entry counting as the code's default. A
449    /// nearer layer can therefore not relax governance that an outer layer
450    /// requires.
451    #[must_use]
452    pub fn stricter(&self, other: &Self) -> Self {
453        let require_justification = match (self.require_justification, other.require_justification)
454        {
455            (Some(true), _) | (_, Some(true)) => Some(true),
456            (Some(false), _) | (_, Some(false)) => Some(false),
457            (None, None) => None,
458        };
459        // The notices closest to expiry are kept: the last warning before a
460        // rule stops deciding matters most.
461        let mut notices: Vec<u16> = self
462            .expiry_notice_days
463            .iter()
464            .chain(&other.expiry_notice_days)
465            .copied()
466            .collect();
467        notices.sort_unstable();
468        notices.dedup();
469        notices.truncate(limits::MAX_EXPIRY_NOTICES);
470        notices.reverse();
471        let mut severity = BTreeMap::new();
472        for key in self.severity.keys().chain(other.severity.keys()) {
473            let default = crate::analysis::FindingCode::from_name(key).map_or(
474                FindingSeverity::Info,
475                crate::analysis::FindingCode::default_severity,
476            );
477            let of = |m: &BTreeMap<String, FindingSeverity>| m.get(key).copied().unwrap_or(default);
478            severity.insert(key.clone(), of(&self.severity).max(of(&other.severity)));
479        }
480        Self {
481            require_justification,
482            stale_after_days: self.stale_after_days.min(other.stale_after_days),
483            expiring_soon_days: self.expiring_soon_days.max(other.expiring_soon_days),
484            expiry_notice_days: notices,
485            counter_report_hours: self.counter_report_hours.min(other.counter_report_hours),
486            severity,
487        }
488    }
489}

508    fn timestamps_read_rfc3339_and_write_utc() {
509        for (text, secs) in [
510            ("1970-01-01T00:00:00Z", 0),
511            ("1970-01-01T00:00:00.999Z", 0),

519        }

521        assert_eq!(Timestamp(-1).to_string(), "1969-12-31T23:59:59Z");
522        for bad in [

524            "2026-11-25T09:30:00",
525            "2026-02-29T00:00:00Z",
526            "2026-13-01T00:00:00Z",
527            "2026-11-25T24:00:00Z",
528            "2026-11-25T23:59:60Z",
529            "2026-11-25T09:30:00.Z",

541    /// F5: a side with no entry for a code counts as the code's default,
542    /// so merging with a layer that never mentions it cannot lower it.
543    /// Notices: the eight nearest to expiry are kept.
544    #[test]
545    fn merging_keeps_default_severities_and_the_nearest_notices() {
546        let outer = GovernanceSettings::default();
547        let mut nearer = GovernanceSettings::default();
548        // As a merged policy could hold before validation rejected it.
549        nearer
550            .severity
551            .insert("missing_justification".into(), FindingSeverity::Info);
552        let merged = nearer.stricter(&outer);
553        assert_eq!(
554            merged.severity.get("missing_justification"),
555            Some(&FindingSeverity::Error)
556        );
557        let merged = outer.stricter(&nearer);
558        assert_eq!(
559            merged.severity.get("missing_justification"),
560            Some(&FindingSeverity::Error)
561        );
562        let a = GovernanceSettings {
563            expiry_notice_days: vec![60, 30, 20, 14, 10],
564            ..GovernanceSettings::default()
565        };
566        let b = GovernanceSettings {
567            expiry_notice_days: vec![90, 7, 5, 3, 1, 0],
568            ..GovernanceSettings::default()
569        };
570        // Before the fix the farthest were kept: 90, 60, 30, 20, 14, 10, 7,
571        // 5, dropping the notices at 3, 1 and 0 days.
572        assert_eq!(

574            vec![20, 14, 10, 7, 5, 3, 1, 0]

577        let off = GovernanceSettings {
578            require_justification: Some(false),

580        };
581        assert_eq!(off.resolved(false).require_justification, Some(false));
582        assert_eq!(off.resolved(true).require_justification, Some(false));
583        assert_eq!(
584            GovernanceSettings::default()
585                .resolved(true)
586                .require_justification,
