321fn canonical(
322    management: Management,
323    doc: Option<&PolicyDocument>,
324    libraries: &BTreeSet<Version>,
325) -> String {
326    let document = doc.map(|d| {
327        let mut d = d.clone();
328        d.policy.revision = 0;
329        schema::sort_keys(serde_json::to_value(&d).unwrap_or_default())
330    });
331    let libraries: Vec<(String, u32)> = libraries
332        .iter()
333        .map(|(id, v)| (id.to_string(), *v))
334        .collect();
335    let value = schema::sort_keys(serde_json::json!({
336        "document": document,
337        "libraries": libraries,
338        "management": management,
339    }));
340    value.to_string()
341}

343fn sha256_hex(text: &str) -> String {
344    use core::fmt::Write as _;
345    let digest = Sha256::digest(text.as_bytes());
346    let mut out = String::with_capacity(64);
347    for b in digest {
348        let _ = write!(out, "{b:02x}");
349    }
350    out

402    /// The effective policy a device would have at `node`: a device, or
403    /// for a dry run a group (including `staging`), a tenant or the global
404    /// node.
405    ///
406    /// # Errors
407    ///
408    /// Every finding: an unknown node or bad membership; a suppression of a
409    /// rule that is not inherited or not suppressible; a precedence
410    /// conflict; two different rule sets with one ID; and every error of
411    /// validating the effective policy, under `/effective`.
412    pub fn effective_policy_as(
413        &self,
414        node: NodeId,
415        membership: &Membership,
416    ) -> Result<EffectivePolicy, Vec<HierarchyError>> {
417        let target = self.nodes.get(&node).ok_or_else(|| {
418            vec![herr(
419                HierarchyCode::UnknownNode,
420                "",
421                format!("node {node} does not exist"),
422            )]
423        })?;
424        if matches!(
425            target.node.kind,
426            NodeKind::DynamicGroup | NodeKind::Template
427        ) {
428            return Err(vec![herr(
429                HierarchyCode::UnknownNode,
430                target.path(),
431                format!(
432                    "`{}` is a `{}`, which has no effective policy of its own",
433                    target.node.name,
434                    target.node.kind.as_str()
435                ),
436            )]);
437        }
438        let dynamic = self.dynamic_groups(target, membership)?;
439        let unmanaged = target.node.builtin == Some(Builtin::Unmanaged)
440            || (target.node.kind == NodeKind::Device
441                && target
442                    .node
443                    .parent
444                    .and_then(|p| self.nodes.get(&p))
445                    .is_some_and(|p| p.node.builtin == Some(Builtin::Unmanaged)));
446        if unmanaged {
447            let libraries = BTreeSet::new();
448            let text = canonical(Management::Unmanaged, None, &libraries);
449            return Ok(EffectivePolicy {
450                node,
451                management: Management::Unmanaged,
452                policy: None,
453                rules: BTreeMap::new(),
454                rule_sets: BTreeMap::new(),
455                settings: BTreeMap::new(),
456                mode: None,
457                precedence: None,
458                libraries,
459                content_hash: sha256_hex(&text),
460            });
461        }
462        let (level_order, merge) = self.options(target.tenant);
463        let layers = self.layers(target, &dynamic, level_order);
464        Builder::new(self, &layers, merge).build(node, true)
465    }

467    /// Checks the effective policy a device would have at `node`, as

469    /// hashing it. Validation runs this for every configured structural

472        let Some(target) = self.nodes.get(&node) else {

474        };

482    /// The dynamic groups of a device at `target`, checked, sorted from the
483    /// outermost (highest rank) to the nearest (lowest rank).
484    fn dynamic_groups(
485        &self,
486        target: &NodeInfo,
487        membership: &Membership,
488    ) -> Result<Vec<&NodeInfo>, Vec<HierarchyError>> {
489        let ids: BTreeSet<NodeId> = membership.dynamic_groups.iter().copied().collect();
490        if ids.len() > MAX_DYNAMIC_GROUPS {
491            return Err(vec![herr(
492                HierarchyCode::TooMany,
493                "/membership",
494                format!(
495                    "{} dynamic groups; at most {MAX_DYNAMIC_GROUPS} are allowed",
496                    ids.len()
497                ),
498            )]);
499        }
500        let mut out = Vec::new();
501        let mut errors = Vec::new();
502        for id in ids {
503            match self.nodes.get(&id) {
504                Some(g)
505                    if g.node.kind == NodeKind::DynamicGroup
506                        && target.tenant.is_some()
507                        && g.tenant == target.tenant =>
508                {
509                    out.push(g);
510                }
511                _ => errors.push(herr(
512                    HierarchyCode::InvalidMembership,
513                    "/membership",
514                    format!("node {id} is not a dynamic group of the device's tenant"),
515                )),
516            }
517        }
518        if !errors.is_empty() {
519            return Err(errors);
520        }
521        // The highest rank first, so the lowest rank is nearest; equal ranks
522        // by ID, so the order never depends on the input.
523        out.sort_by(|a, b| {
524            (b.node.rank.unwrap_or(0), b.node.id).cmp(&(a.node.rank.unwrap_or(0), a.node.id))
525        });
526        Ok(out)
527    }

529    /// The layers of a device at `target`, from the outermost to the
530    /// nearest, with each linked template just before its node.
531    fn layers<'h>(
532        &'h self,
533        target: &'h NodeInfo,
534        dynamic: &[&'h NodeInfo],
535        order: LevelOrder,
536    ) -> Vec<Layer<'h>> {
537        let mut chain: Vec<&NodeInfo> = Vec::new();
538        let mut at = Some(target);
539        while let Some(info) = at {
540            chain.push(info);
541            at = info.node.parent.and_then(|p| self.nodes.get(&p));
542        }
543        chain.reverse();
544        let device = (target.node.kind == NodeKind::Device)
545            .then(|| chain.pop())
546            .flatten();
547        let mut nodes: Vec<(&NodeInfo, u8)> = Vec::new();
548        let mut depth = 0u8;
549        for info in chain {
550            let band = match info.node.kind {
551                NodeKind::Global => BAND_GLOBAL,
552                NodeKind::Tenant => BAND_TENANT,
553                _ => {
554                    depth = depth.saturating_add(1);
555                    BAND_TENANT.saturating_add(depth)
556                }
557            };
558            nodes.push((info, band));
559        }
560        nodes.extend(dynamic.iter().map(|g| (*g, BAND_DYNAMIC)));
561        if let Some(d) = device {
562            nodes.push((d, BAND_DEVICE));
563        }
564        let mut layers = Vec::new();
565        for (info, band) in nodes {
566            let band = match order {
567                LevelOrder::TopDown => band,
568                LevelOrder::BottomUp => BAND_DEVICE.saturating_sub(band),
569                LevelOrder::Flat => 0,
570            };
571            if let Some(t) = info.node.template.and_then(|t| self.nodes.get(&t)) {
572                layers.push(Layer {
573                    info: t,
574                    template_for: Some(info.node.id),
575                    band,
576                });
577            }
578            layers.push(Layer {
579                info,
580                template_for: None,
581                band,
582            });
583        }
584        layers
585    }

629            .assignments

642    /// Builds the effective policy. With `hash` false (a validation check)
643    /// the content hash is left empty.
644    fn build(mut self, node: NodeId, hash: bool) -> Result<EffectivePolicy, Vec<HierarchyError>> {
645        self.collect_rule_sets();
646        let rules = self.suppressions();
647        let precedence = self.precedence();
648        let (mut settings, mut settings_prov) = self.settings();
649        let (mut mode_options, mode_prov) = self.mode_options();
650        let mut mode = self.mode();
651        let location_profiles = self.location_profiles();
652        self.scope_set_versions();
653        if !self.errors.is_empty() {
654            return Err(self.errors);
655        }
656        let Some((model, options, source)) = precedence else {
657            return Err(self.errors);
658        };
659        // The floor.
660        let mut adjustments = Vec::new();
661        let mut sets: BTreeMap<RuleSetId, RuleSet> = BTreeMap::new();
662        for (id, entry) in &self.sets {
663            let mut set = entry.set.clone();
664            set.band = entry.band;
665            for rule in &mut set.rules {
666                if let Some(p) = rules.get(&rule.id) {
667                    if matches!(p.status, RuleStatus::Suppressed { .. }) {
668                        rule.enabled = false;
669                    }
670                }
671            }
672            sets.insert(*id, set);
673        }
674        if let Some((floor, _)) = mode.floor {
675            apply_floor(
676                floor,
677                &mut mode,
678                &mut mode_options,
679                &mut settings,
680                &mut sets,
681                &mut adjustments,
682                &mut settings_prov,
683            );
684        }
685        // Unset `mode_disconnected` means the effective mode itself, so a
686        // merged one that is not at least as strict is dropped.
687        if let Some(offline) = mode_options.mode_disconnected {
688            if offline != mode.mode && strictness(offline) <= strictness(mode.mode) {
689                adjustments.push(ModeAdjustment {
690                    path: "/policy/mode_options/mode_disconnected".to_owned(),
691                    from: serde_json::to_value(offline).unwrap_or_default(),
692                    to: serde_json::Value::Null,
693                });
694                mode_options.mode_disconnected = None;
695            }
696        }
697        mode.adjustments = adjustments;
698        // The provenance of the mode options shows their final values.
699        let mut mode_prov = mode_prov;
700        for (key, p) in &mut mode_prov {
701            let now = if let Some(t) = key.strip_prefix("mode_options.mode_by_type.") {
702                t.parse::<ObjectType>()
703                    .ok()
704                    .and_then(|t| mode_options.mode_by_type.get(&t))
705                    .map(|m| serde_json::to_value(m).unwrap_or_default())
706            } else if key == "mode_options.mode_disconnected" {
707                mode_options
708                    .mode_disconnected
709                    .map(|m| serde_json::to_value(m).unwrap_or_default())
710            } else {
711                mode_options.disconnected_after_secs.map(Into::into)
712            }
713            .unwrap_or(serde_json::Value::Null);
714            if now != p.value {
715                p.raised_by_floor = true;
716                p.value = now;
717            }
718        }
719        settings_prov.extend(mode_prov);
720
721        let target = self.h.nodes.get(&node);
722        let name = target.map_or_else(|| node.to_string(), |t| t.node.name.clone());
723        // Each contributing policy's scopes now sit on its rule sets, so the
724        // effective policy has no scope of its own (#38). Location profiles
725        // are the union of the contributing policies'; two with one name
726        // and different tests cannot both hold.
727        let rule_set_scopes = self
728            .sets
729            .iter()
730            .filter(|(_, e)| !e.scopes.is_empty())
731            .map(|(id, e)| crate::scope::RuleSetScope {
732                rule_set: *id,
733                scopes: e.scopes.clone(),
734            })
735            .collect();
736        let doc = PolicyDocument {
737            schema_version: schema::CURRENT,
738            policy: Policy {
739                id: PolicyId(node.0),
740                name,
741                description: None,
742                revision: self.h.revision,
743                precedence_model: model,
744                precedence_options: options,
745                mode: mode.mode,
746                mode_options,
747                settings,
748                unknown: self.unknown(),
749                rule_sets: sets.keys().copied().collect(),
750                scope: None,
751                rule_set_scopes,
752                location_profiles,
753                metadata: Metadata::default(),
754            },
755            rule_sets: sets.into_values().collect(),
756        };
757        // Every rule is a copy of one in a validated policy or rule set
758        // library, so its criteria are not compiled again.
759        let validated = doc.validate_assembled().map_err(|findings| {
760            findings
761                .into_iter()
762                .filter(|f| f.severity == Severity::Error)
763                .map(|f| HierarchyError {
764                    code: HierarchyCode::Policy(f.code),
765                    path: format!("/effective{}", f.path),
766                    message: f.message,
767                })
768                .collect::<Vec<_>>()
769        })?;
770        let content_hash = if hash {
771            sha256_hex(&canonical(
772                Management::Managed,
773                Some(validated.document()),
774                &self.libraries,
775            ))
776        } else {
777            String::new()
778        };
779        let rule_sets = self
780            .sets
781            .iter()
782            .map(|(id, e)| {
783                (
784                    *id,
785                    e.contributions.iter().map(|(_, c)| c.clone()).collect(),
786                )
787            })
788            .collect();
789        Ok(EffectivePolicy {
790            node,
791            management: Management::Managed,
792            policy: Some(validated),
793            rules,
794            rule_sets,
795            settings: settings_prov,
796            mode: Some(mode),
797            precedence: Some(source),
798            libraries: self.libraries,
799            content_hash,
800        })
801    }

803    /// One version of each library set the scopes use (#38), as of each
804    /// rule set library and policy: two pins of one set would leave a
805    /// device-set membership fact, which names the set only, ambiguous.
806    fn scope_set_versions(&mut self) {
807        let mut seen: BTreeMap<LibraryId, (crate::library::Pin, String)> = BTreeMap::new();
808        let mut found = Vec::new();
809        // The policies' scopes also limit their unknown settings (#39).
810        for layer in self.layers {
811            for (_, doc) in self.policies(layer) {
812                let u = &doc.policy.unknown;
813                let scopes = doc.policy.scope.iter().chain(&u.scopes).chain(
814                    u.scope_limits
815                        .iter()
816                        .flat_map(|l| l.all_of.iter().chain(&l.any_of)),
817                );
818                for scope in scopes {
819                    for (_, r, _) in crate::scope::set_refs(scope) {
820                        found.push((
821                            r,
822                            format!("/effective/policy/unknown/scope_limits ({})", doc.policy.id),
823                        ));
824                    }
825                }
826            }
827        }
828        for (id, e) in &self.sets {
829            let scopes = e
830                .scopes
831                .iter()
832                .chain(e.set.rules.iter().filter_map(|r| r.scope.as_ref()));
833            for scope in scopes {
834                for (_, r, _) in crate::scope::set_refs(scope) {
835                    found.push((r, format!("/effective/rule_sets/{id}")));
836                }
837            }
838        }
839        for (r, at) in found {
840            match seen.get(&r.id) {
841                Some((pin, first)) if *pin != r.pin => self.errors.push(herr(
842                    HierarchyCode::LibraryVersionConflict,
843                    at,
844                    format!(
845                        "library set {} is used at version {} and at version {pin} ({first}); one device uses one version of each set",
846                        r.id, r.pin
847                    ),
848                )),
849                Some(_) => {}
850                None => {
851                    seen.insert(r.id, (r.pin, at));
852                }
853            }
854        }
855    }

896    /// Adds one rule set from layer `pos`.
897    fn add_set(
898        &mut self,
899        pos: usize,
900        set: &RuleSet,
901        library: Option<Version>,
902        scopes: Vec<crate::scope::Scope>,
903        contribution: Contribution,
904        at: &str,
905    ) {
906        let band = self.layers[pos].band;
907        if let Some((id, version)) = library {
908            match self.library_versions.get(&id) {
909                Some((v, first)) if *v != version => {
910                    self.errors.push(herr(
911                        HierarchyCode::LibraryVersionConflict,
912                        at,
913                        format!(
914                            "rule set library {id} would apply at version {version} and at version {v} ({first}); one device uses one version of each rule set"
915                        ),
916                    ));
917                    return;
918                }
919                _ => {
920                    self.library_versions.insert(id, (version, at.to_owned()));
921                }
922            }
923        }
924        match self.sets.get_mut(&set.id) {
925            Some(entry) => {
926                let mut a = entry.set.clone();
927                let mut b = set.clone();
928                a.band = 0;
929                b.band = 0;
930                if a == b && entry.scopes != scopes {
931                    self.errors.push(herr(
932                        HierarchyCode::ConflictingScope,
933                        at,
934                        format!(
935                            "rule set {} (`{}`) is also contributed with a different scope; give it one scope, or contribute it once",
936                            set.id, set.name
937                        ),
938                    ));
939                    return;
940                }
941                if a != b {
942                    self.errors.push(herr(
943                        HierarchyCode::RuleSetConflict,
944                        at,
945                        format!(
946                            "rule set {} (`{}`) is also contributed with different content; a rule set ID names one rule set",
947                            set.id, set.name
948                        ),
949                    ));
950                    return;
951                }
952                entry.band = entry.band.min(band);
953                entry.contributions.push((pos, contribution));
954            }
955            None => {
956                self.sets.insert(
957                    set.id,
958                    SetEntry {
959                        set: set.clone(),
960                        scopes,
961                        contributions: vec![(pos, contribution)],
962                        band,
963                    },
964                );
965            }
966        }
967    }

971        for (pos, layer) in layers.iter().enumerate() {
972            let info = layer.info;
973            let base = info.path();

975                let by_id: BTreeMap<RuleSetId, &RuleSet> =

1034                );
1035            }
1036            for (i, category) in info.node.categories.iter().enumerate() {
1037                for lib in self.h.catalogue.in_category(category, info.visibility()) {

1048                        category: category.clone(),
1049                        library: lib.id,
1050                        version: lib.version,
1051                    };
1052                    let set = entry.rule_set.clone();
1053                    self.add_set(
1054                        pos,
1055                        &set,
1056                        Some(version),
1057                        Vec::new(),
1058                        layer.contribution(via),
1059                        &format!("{base}/categories/{i}"),
1060                    );
1061                }
1062            }
1063        }

1066    /// Every rule's provenance, after the suppressions.
1067    fn suppressions(&mut self) -> BTreeMap<RuleId, RuleProvenance> {
1068        let mut out = BTreeMap::new();
1069        for (set_id, entry) in &self.sets {
1070            for rule in &entry.set.rules {
1071                out.entry(rule.id).or_insert_with(|| RuleProvenance {
1072                    rule_id: rule.id,
1073                    rule_set_id: *set_id,
1074                    contributions: entry.contributions.iter().map(|(_, c)| c.clone()).collect(),
1075                    status: if rule.enabled {
1076                        RuleStatus::Active
1077                    } else {
1078                        RuleStatus::Disabled
1079                    },
1080                });
1081            }
1082        }
1083        // The nearest suppression of each rule decides.
1084        let mut nearest: BTreeMap<RuleId, (usize, NodeId)> = BTreeMap::new();
1085        for (pos, layer) in self.layers.iter().enumerate() {
1086            let info = layer.info;
1087            for (i, id) in info.node.suppress_inherited.iter().enumerate() {
1088                let path = format!("{}/suppress_inherited/{i}", info.path());
1089                let Some(p) = out.get(id) else {
1090                    self.errors.push(herr(
1091                        HierarchyCode::SuppressUnknownRule,
1092                        path,
1093                        format!(
1094                            "rule {id} is not in the effective policy of a device below `{}`",
1095                            info.node.name
1096                        ),
1097                    ));
1098                    continue;
1099                };
1100                let Some(entry) = self.sets.get(&p.rule_set_id) else {
1101                    continue;
1102                };
1103                if !entry.contributions.iter().any(|(at, _)| *at < pos) {
1104                    self.errors.push(herr(
1105                        HierarchyCode::SuppressUnknownRule,
1106                        path,
1107                        format!(
1108                            "rule {id} is not inherited by `{}`: its rule set `{}` is contributed only here or below; remove it instead",
1109                            info.node.name, entry.set.name
1110                        ),
1111                    ));
1112                    continue;
1113                }
1114                if !suppressible(&entry.set) {
1115                    let why = if entry.set.mandatory {
1116                        "it is in a mandatory rule set, which blocks under every policy below it (ADR 0004)".to_owned()
1117                    } else {
1118                        format!(
1119                            "its rule set `{}` does not allow it (`allow_child_suppress`, which is false by default for a `{}`)",
1120                            entry.set.name,
1121                            kind_name(entry.set.kind)
1122                        )
1123                    };
1124                    self.errors.push(herr(
1125                        HierarchyCode::SuppressNotAllowed,
1126                        path,
1127                        format!("rule {id} cannot be suppressed: {why}; shadow it with a rule that wins under the precedence model instead"),
1128                    ));
1129                    continue;
1130                }
1131                nearest.insert(*id, (pos, info.node.id));
1132            }
1133        }
1134        for (id, (pos, by)) in nearest {
1135            let Some(p) = out.get_mut(&id) else { continue };
1136            let Some(entry) = self.sets.get(&p.rule_set_id) else {
1137                continue;
1138            };
1139            // A rule its rule set disables stays disabled: a suppression
1140            // can neither enable it nor make it look active.
1141            if p.status == RuleStatus::Disabled {
1142                continue;
1143            }
1144            let inherited_only = entry.contributions.iter().all(|(at, _)| *at < pos);
1145            p.status = if inherited_only {
1146                RuleStatus::Suppressed { by }
1147            } else {
1148                RuleStatus::Reasserted { suppressed_by: by }
1149            };
1150        }
1151        out
1152    }

1154    /// The model, its options and who chose them. `None` after an error.
1155    fn precedence(&mut self) -> Option<(PrecedenceModel, PrecedenceOptions, PrecedenceSource)> {
1156        let chosen = self
1157            .layers
1158            .iter()
1159            .rev()
1160            .find_map(|l| l.info.node.precedence.map(|p| (p, l.info.node.id)));
1161        let mut ranked: Vec<(usize, u32, &PolicyDocument, &NodeInfo)> = Vec::new();
1162        for (pos, layer) in self.layers.iter().enumerate() {
1163            for (slot, doc) in self.policies(layer) {
1164                ranked.push((pos, slot, doc, layer.info));
1165            }
1166        }
1167        let (model, options, source) = match chosen {
1168            Some((p, node)) => (p.model, p.options, PrecedenceSource::Node(node)),
1169            None => match ranked
1170                .iter()
1171                .min_by_key(|(pos, slot, doc, _)| (core::cmp::Reverse(*pos), *slot, doc.policy.id))
1172            {
1173                Some((_, _, doc, _)) => (
1174                    doc.policy.precedence_model,
1175                    doc.policy.precedence_options,
1176                    PrecedenceSource::Policy(doc.policy.id),
1177                ),
1178                None => (
1179                    PrecedenceModel::DenyWins,
1180                    PrecedenceOptions::default(),
1181                    PrecedenceSource::Default,
1182                ),
1183            },
1184        };
1185        let mut ok = true;
1186        let mut seen = BTreeSet::new();
1187        for (_, _, doc, info) in &ranked {
1188            let differs =
1189                doc.policy.precedence_model != model || doc.policy.precedence_options != options;
1190            if differs && seen.insert(doc.policy.id) {
1191                ok = false;
1192                let what = if doc.policy.precedence_model == model {
1193                    format!("other `{}` options", name_of(&model))
1194                } else {
1195                    format!("`{}`", name_of(&doc.policy.precedence_model))
1196                };
1197                self.errors.push(herr(
1198                    HierarchyCode::PrecedenceConflict,
1199                    format!("{}/assignments", info.path()),
1200                    format!(
1201                        "policy `{}` uses {what}, but the effective policy uses `{}` ({}); one effective policy uses one model with one set of options",
1202                        doc.policy.name,
1203                        name_of(&model),
1204                        match source {
1205                            PrecedenceSource::Node(n) => format!("chosen by node {n}"),
1206                            PrecedenceSource::Policy(p) => format!("the model of the highest-ranked policy, {p}"),
1207                            PrecedenceSource::Default => "the default".to_owned(),
1208                        }
1209                    ),
1210                ));
1211            }
1212        }
1213        ok.then_some((model, options, source))
1214    }

1216    /// The unknown-application policy (#39): the strictest combination of
1217    /// every assigned policy's ([`UnknownPolicy::stricter`](crate::model::UnknownPolicy::stricter)),
1218    /// so no layer can make unknown objects run more freely than another
1219    /// layer allows. Policies are folded from the outermost layer in, and
1220    /// within a layer from the highest `slot_order` down, so where a value
1221    /// cannot be compared the nearest layer's lowest slot gives it, as for
1222    /// every other merge here. With no policy, the default (block).
1223    ///
1224    /// A policy's `scope` and its `unknown.scopes` limit where its unknown
1225    /// settings apply (#38): `scope` and any of `scopes`. The effective
1226    /// policy has no scope of its own, so that limit moves, unexpanded, into
1227    /// one entry of `unknown.scope_limits`
1228    /// ([`UnknownPolicy::scope_limit_of`](crate::model::UnknownPolicy::scope_limit_of)):
1229    /// at most one entry per contributing policy.
1230    /// The merged settings apply to a subject in at least one contributing
1231    /// policy's limit (everyone if one has none), exactly where that
1232    /// policy's own evaluation would apply its settings, and the built-in
1233    /// implicit deny decides for any other. Where several limits overlap,
1234    /// the merged settings are at least as strict as each policy's.
1235    ///
1236    /// The fold is linear in the contributing policies. Levels and settings
1237    /// merge pairwise ([`UnknownPolicy::stricter_settings`](crate::model::UnknownPolicy::stricter_settings)),
1238    /// which costs nothing that grows with the limits, and the limits are
1239    /// collected in one pass, de-duplicated by a canonical form, so no limit
1240    /// is cloned or compared more than once.
1241    fn unknown(&self) -> crate::model::UnknownPolicy {
1242        let mut acc: Option<crate::model::UnknownPolicy> = None;
1243        let mut limits: Vec<crate::model::ScopeLimit> = Vec::new();
1244        let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
1245        let mut everywhere = false;
1246        for layer in self.layers {
1247            for (_, d) in self.policies(layer).into_iter().rev() {
1248                match d
1249                    .policy
1250                    .unknown
1251                    .scope_limit_of(d.policy.scope.as_ref(), stage_six_scope)
1252                {
1253                    // A policy with no limit applies everywhere, which
1254                    // absorbs every other limit.
1255                    None => everywhere = true,
1256                    Some(l) if !everywhere => {
1257                        if seen.insert(l.canonical()) {
1258                            limits.push(l);
1259                        }
1260                    }
1261                    Some(_) => {}
1262                }
1263                acc = Some(match acc {
1264                    None => d.policy.unknown.without_limits(),
1265                    Some(outer) => d.policy.unknown.stricter_settings(&outer),
1266                });
1267            }
1268        }
1269        let mut u = acc.unwrap_or_default();
1270        u.scopes = Vec::new();
1271        u.scope_limits = if everywhere { Vec::new() } else { limits };
1272        u
1273    }

1275    /// The merged settings and their provenance.
1276    fn settings(&mut self) -> (PolicySettings, BTreeMap<String, SettingProvenance>) {
1277        let default = super::validate::default_settings();
1278        let keys = super::validate::setting_keys();
1279        let mut current: BTreeMap<String, Option<serde_json::Value>> = keys
1280            .iter()
1281            .map(|k| (k.clone(), default.get(k).cloned()))
1282            .collect();
1283        let mut prov: BTreeMap<String, SettingProvenance> = keys
1284            .iter()
1285            .map(|k| {
1286                (
1287                    k.clone(),
1288                    SettingProvenance {
1289                        value: default.get(k).cloned().unwrap_or(serde_json::Value::Null),
1290                        source: None,
1291                        locked_at: None,
1292                        raised_by_floor: false,
1293                        steps: Vec::new(),
1294                    },
1295                )
1296            })
1297            .collect();
1298        let mut locks: BTreeMap<String, NodeId> = BTreeMap::new();
1299        for layer in self.layers {
1300            let policies: Vec<(u32, serde_json::Value, PolicyId)> = self
1301                .policies(layer)
1302                .into_iter()
1303                .map(|(slot, d)| {
1304                    (
1305                        slot,
1306                        serde_json::to_value(&d.policy.settings).unwrap_or_default(),
1307                        d.policy.id,
1308                    )
1309                })
1310                .collect();
1311            for key in keys {
1312                let mut offers: Vec<(Contribution, serde_json::Value)> = Vec::new();
1313                if let Some(v) = layer.info.node.settings.get(key) {
1314                    offers.push((layer.contribution(Via::Node), v.clone()));
1315                }
1316                for (slot, value, policy) in &policies {
1317                    // A field left at its default is not written, so a
1318                    // missing key is the default too.
1319                    let v = value.get(key).or_else(|| default.get(key));
1320                    if v != default.get(key) {
1321                        offers.push((
1322                            layer.contribution(Via::Policy {
1323                                policy: *policy,
1324                                slot_order: *slot,
1325                            }),
1326                            v.cloned().unwrap_or(serde_json::Value::Null),
1327                        ));
1328                    }
1329                }
1330                let Some(p) = prov.get_mut(key) else { continue };
1331                for (i, (from, value)) in offers.into_iter().enumerate() {
1332                    // A lock above blocks every offer of the layer; without
1333                    // one, the first offer applies and the rest lose to it.
1334                    let outcome = if let Some(&locked_at) = locks.get(key) {
1335                        StepOutcome::BlockedByLock { locked_at }
1336                    } else if i > 0 {
1337                        StepOutcome::Overridden
1338                    } else {
1339                        current.insert(key.clone(), Some(value.clone()));
1340                        StepOutcome::Applied
1341                    };
1342                    p.steps.push(SettingStep {
1343                        from,
1344                        value,
1345                        outcome,
1346                    });
1347                }
1348            }
1349            for key in &layer.info.node.locked_settings {
1350                locks.entry(key.clone()).or_insert(layer.info.node.id);
1351            }
1352        }
1353        for (key, p) in &mut prov {
1354            let last = p
1355                .steps
1356                .iter()
1357                .rposition(|s| s.outcome == StepOutcome::Applied);
1358            for (i, step) in p.steps.iter_mut().enumerate() {
1359                if step.outcome == StepOutcome::Applied && Some(i) != last {
1360                    step.outcome = StepOutcome::Overridden;
1361                }
1362            }
1363            p.source = last.map(|i| p.steps[i].from.node);
1364            p.locked_at = locks.get(key).copied();
1365            p.value = current
1366                .get(key)
1367                .cloned()
1368                .flatten()
1369                .unwrap_or(serde_json::Value::Null);
1370        }
1371        let map: serde_json::Map<String, serde_json::Value> = current
1372            .into_iter()
1373            .filter_map(|(k, v)| v.map(|v| (k, v)))
1374            .collect();
1375        let settings =
1376            match serde_json::from_value::<PolicySettings>(serde_json::Value::Object(map)) {
1377                Ok(s) => s,
1378                Err(e) => {
1379                    self.errors.push(herr(
1380                        HierarchyCode::InvalidSetting,
1381                        "/effective/policy/settings",
1382                        format!("the merged settings are not valid: {e}"),
1383                    ));
1384                    PolicySettings::default()
1385                }
1386            };
1387        (settings, prov)
1388    }

1390    /// The merged mode options, nearest layer and lowest slot first.
1391    ///
1392    /// A key in a node's `locked_settings` (`mode_options.mode_disconnected`,
1393    /// `mode_options.disconnected_after_secs` or `mode_options.mode_by_type`,
1394    /// which locks every type) keeps the value it has after that node's
1395    /// layer; later offers are `blocked_by_lock`.
1396    fn mode_options(&self) -> (ModeOptions, BTreeMap<String, SettingProvenance>) {
1397        let mut out = ModeOptions::default();
1398        let mut prov: BTreeMap<String, SettingProvenance> = BTreeMap::new();
1399        let mut locks: BTreeMap<String, NodeId> = BTreeMap::new();
1400        let lock_of = |locks: &BTreeMap<String, NodeId>, key: &str| {
1401            locks.get(key).copied().or_else(|| {
1402                key.starts_with("mode_options.mode_by_type.")
1403                    .then(|| locks.get("mode_options.mode_by_type").copied())
1404                    .flatten()
1405            })
1406        };
1407        let mut offer = |locks: &BTreeMap<String, NodeId>,
1408                         key: String,
1409                         from: Contribution,
1410                         value: serde_json::Value| {
1411            let locked_at = lock_of(locks, &key);
1412            let p = prov.entry(key).or_insert_with(|| SettingProvenance {
1413                value: serde_json::Value::Null,
1414                source: None,
1415                locked_at: None,
1416                raised_by_floor: false,
1417                steps: Vec::new(),
1418            });
1419            p.locked_at = p.locked_at.or(locked_at);
1420            p.steps.push(SettingStep {
1421                from,
1422                value,
1423                outcome: match locked_at {
1424                    Some(locked_at) => StepOutcome::BlockedByLock { locked_at },
1425                    None => StepOutcome::Overridden,
1426                },
1427            });
1428        };
1429        for layer in self.layers {
1430            for (slot, doc) in self.policies(layer) {
1431                let from = layer.contribution(Via::Policy {
1432                    policy: doc.policy.id,
1433                    slot_order: slot,
1434                });
1435                let o = &doc.policy.mode_options;
1436                for (t, m) in &o.mode_by_type {
1437                    offer(
1438                        &locks,
1439                        format!("mode_options.mode_by_type.{t}"),
1440                        from.clone(),
1441                        serde_json::to_value(m).unwrap_or_default(),
1442                    );
1443                }
1444                if let Some(m) = o.mode_disconnected {
1445                    offer(
1446                        &locks,
1447                        "mode_options.mode_disconnected".to_owned(),
1448                        from.clone(),
1449                        serde_json::to_value(m).unwrap_or_default(),
1450                    );
1451                }
1452                if let Some(s) = o.disconnected_after_secs {
1453                    offer(
1454                        &locks,
1455                        "mode_options.disconnected_after_secs".to_owned(),
1456                        from.clone(),
1457                        s.into(),
1458                    );
1459                }
1460            }
1461            for key in &layer.info.node.locked_settings {
1462                if key.starts_with("mode_options.") {
1463                    locks.entry(key.clone()).or_insert(layer.info.node.id);
1464                }
1465            }
1466        }
1467        // A locked key is reported even if no layer offered it (it is then
1468        // locked at unset); a lock on `mode_by_type` covers every type
1469        // offered anywhere.
1470        for key in locks.keys() {
1471            if key != "mode_options.mode_by_type" {
1472                prov.entry(key.clone())
1473                    .or_insert_with(|| SettingProvenance {
1474                        value: serde_json::Value::Null,
1475                        source: None,
1476                        locked_at: None,
1477                        raised_by_floor: false,
1478                        steps: Vec::new(),
1479                    });
1480            }
1481        }
1482        // Within a layer the lowest slot wins, across layers the nearest:
1483        // the winner is the first offer of the last layer that offered and
1484        // was not blocked by a lock.
1485        for (key, p) in &mut prov {
1486            p.locked_at = lock_of(&locks, key);
1487            let open = |s: &SettingStep| !matches!(s.outcome, StepOutcome::BlockedByLock { .. });
1488            let Some(last_node) = p
1489                .steps
1490                .iter()
1491                .rev()
1492                .find(|s| open(s))
1493                .map(|s| (s.from.node, s.from.template_for))
1494            else {
1495                continue;
1496            };
1497            let Some(win) = p
1498                .steps
1499                .iter()
1500                .position(|s| open(s) && (s.from.node, s.from.template_for) == last_node)
1501            else {
1502                continue;
1503            };
1504            p.steps[win].outcome = StepOutcome::Applied;
1505            p.source = Some(p.steps[win].from.node);
1506            p.value = p.steps[win].value.clone();
1507            let value = p.value.clone();
1508            if let Some(t) = key.strip_prefix("mode_options.mode_by_type.") {
1509                if let (Ok(t), Ok(m)) = (
1510                    t.parse::<ObjectType>(),
1511                    serde_json::from_value::<Mode>(value),
1512                ) {
1513                    out.mode_by_type.insert(t, m);
1514                }
1515            } else if key == "mode_options.mode_disconnected" {
1516                out.mode_disconnected = serde_json::from_value(value).ok();
1517            } else {
1518                out.disconnected_after_secs = serde_json::from_value(value).ok();
1519            }
1520        }
1521        (out, prov)
1522    }

1524    /// Each layer's mode, merged, and the strictest floor.
1525    fn mode(&self) -> ModeProvenance {
1526        let mut current: Option<(Mode, NodeId)> = None;
1527        let mut floor: Option<(Mode, NodeId)> = None;
1528        for layer in self.layers {
1529            let info = layer.info;
1530            let own = info.node.mode.or_else(|| {
1531                let modes: Vec<Mode> = self
1532                    .policies(layer)
1533                    .into_iter()
1534                    .map(|(_, d)| d.policy.mode)
1535                    .collect();
1536                match self.merge {
1537                    ModeMerge::Closest => modes.first().copied(),
1538                    // The first of the strictest, by slot.
1539                    ModeMerge::Strictest => modes
1540                        .iter()
1541                        .copied()
1542                        .reduce(|a, b| if strictness(b) > strictness(a) { b } else { a }),
1543                }
1544                .or_else(|| (info.node.builtin == Some(Builtin::Unassigned)).then_some(Mode::Audit))
1545            });
1546            if let Some(m) = own {
1547                current = match (self.merge, current) {
1548                    (ModeMerge::Strictest, Some((c, n))) if strictness(c) >= strictness(m) => {
1549                        Some((c, n))
1550                    }
1551                    _ => Some((m, info.node.id)),
1552                };
1553            }
1554            if info.node.lock_mode {
1555                let at = current.map_or(Mode::Audit, |(m, _)| m);
1556                if floor.is_none_or(|(f, _)| strictness(at) > strictness(f)) {
1557                    floor = Some((at, info.node.id));
1558                }
1559            }
1560        }
1561        ModeProvenance {
1562            mode: current.map_or(Mode::Audit, |(m, _)| m),
1563            source: current.map(|(_, n)| n),
1564            merge: self.merge,
1565            floor,
1566            adjustments: Vec::new(),
1567        }
1568    }
1569}

1583/// Raises everything below `floor` to it. See the module documentation.
1584fn apply_floor(
1585    floor: Mode,
1586    mode: &mut ModeProvenance,
1587    options: &mut ModeOptions,
1588    settings: &mut PolicySettings,
1589    sets: &mut BTreeMap<RuleSetId, RuleSet>,
1590    adjustments: &mut Vec<ModeAdjustment>,
1591    settings_prov: &mut BTreeMap<String, SettingProvenance>,
1592) {
1593    let json = |v: &dyn erased::Ser| v.to_json();
1594    let f = strictness(floor);
1595    if strictness(mode.mode) < f {
1596        adjustments.push(ModeAdjustment {
1597            path: "/policy/mode".to_owned(),
1598            from: json(&mode.mode),
1599            to: json(&floor),
1600        });
1601        mode.mode = floor;
1602    }
1603    for (t, m) in &mut options.mode_by_type {
1604        if strictness(*m) < f {
1605            adjustments.push(ModeAdjustment {
1606                path: format!(
1607                    "/policy/mode_options/mode_by_type/{}",
1608                    pointer_escape(&t.to_string())
1609                ),
1610                from: json(m),
1611                to: json(&floor),
1612            });
1613            *m = floor;
1614        }
1615    }
1616    // Control levels cap the mode: `audit` at audit, `off` at monitor.
1617    let needed = if f >= strictness(Mode::Enforce) {
1618        Some(ControlLevel::Enforce)
1619    } else if f >= strictness(Mode::Audit) {
1620        Some(ControlLevel::Audit)
1621    } else {
1622        None
1623    };
1624    if let Some(needed) = needed {
1625        let rank = |c: ControlLevel| match c {
1626            ControlLevel::Off => 0,
1627            ControlLevel::Audit => 1,
1628            ControlLevel::Enforce => 2,
1629        };
1630        let mut raise = |key: &str, path: String, level: &mut ControlLevel| {
1631            if rank(*level) < rank(needed) {
1632                adjustments.push(ModeAdjustment {
1633                    path,
1634                    from: json(level),
1635                    to: json(&needed),
1636                });
1637                *level = needed;
1638                if let Some(p) = settings_prov.get_mut(key) {
1639                    p.raised_by_floor = true;
1640                }
1641            }
1642        };
1643        raise(
1644            "script_control",
1645            "/policy/settings/script_control".to_owned(),
1646            &mut settings.script_control,
1647        );
1648        raise(
1649            "dll_control",
1650            "/policy/settings/dll_control".to_owned(),
1651            &mut settings.dll_control,
1652        );
1653        raise(
1654            "driver_control",
1655            "/policy/settings/driver_control".to_owned(),
1656            &mut settings.driver_control,
1657        );
1658        for (kind, level) in &mut settings.script_control_by_kind {
1659            raise(
1660                "script_control_by_kind",
1661                format!("/policy/settings/script_control_by_kind/{}", kind.as_str()),
1662                level,
1663            );
1664        }
1665        for key in [
1666            "script_control",
1667            "dll_control",
1668            "driver_control",
1669            "script_control_by_kind",
1670        ] {
1671            if let (Some(p), Ok(Some(v))) = (
1672                settings_prov.get_mut(key),
1673                serde_json::to_value(&*settings).map(|s| s.get(key).cloned()),
1674            ) {
1675                p.value = v;
1676            }
1677        }
1678    }
1679    for (i, set) in sets.values_mut().enumerate() {
1680        if let Some(m) = set.mode_override {
1681            if strictness(m) < f {
1682                adjustments.push(ModeAdjustment {
1683                    path: format!("/rule_sets/{i}/mode_override"),
1684                    from: json(&m),
1685                    to: json(&floor),
1686                });
1687                set.mode_override = Some(floor);
1688            }
1689        }
1690        if floor == Mode::Enforce {
1691            for (j, rule) in set.rules.iter_mut().enumerate() {
1692                if rule.enforcement_override == EnforcementOverride::NeverEnforce
1693                    && matches!(rule.action, Action::Deny(_))
1694                {
1695                    adjustments.push(ModeAdjustment {
1696                        path: format!("/rule_sets/{i}/rules/{j}/enforcement_override"),
1697                        from: json(&rule.enforcement_override),
1698                        to: json(&EnforcementOverride::Inherit),
1699                    });
1700                    rule.enforcement_override = EnforcementOverride::Inherit;
1701                }
1702            }
1703        }
1704    }
1705}
