1//! Group provenance for traces and the effective view (#41, #37).
2//!
3//! A device's policy is built from the group hierarchy
4//! ([`crate::hierarchy::EffectivePolicy`]): each rule set comes from one or
5//! more nodes (the global node, a tenant, groups, dynamic groups,
6//! templates or the device), through a policy assigned to the node, a rule
7//! set library or a category, and a node below may have suppressed one of
8//! its rules. [`GroupProvenance`] carries that record, with the names of
9//! the nodes, into a [`Tracer`](super::Tracer) and
10//! [`effective_view`](super::effective::effective_view), so a trace says
11//! which groups a deciding rule came from and the view says it for every
12//! rule.
13//!
14//! The group steps go before the policy, rule set and rule steps, outermost
15//! node first: [`InheritanceStep::node`] is the node's kind (`global`,
16//! `tenant`, `group`, `dynamic_group`, `template` or `device`), `via` says
17//! how the node contributed, and `outcome` is `contributed`, or for the
18//! node that suppressed the rule `suppressed` or `suppression_overridden`.
19//!
20//! Names follow the trace's privacy settings: every name passes the
21//! [`Redactor`](super::Redactor) and is cut to
22//! [`MAX_VALUE_CHARS`](super::privacy::MAX_VALUE_CHARS) characters, as
23//! policy, rule set and rule names do, and the ID is always written. A
24//! node is named only by its configured name; group membership of the user
25//! or device is never part of it.
26

29use crate::hierarchy::{

37/// Where every rule of a device's effective policy came from, by node,
38/// with the names of the nodes.
39#[derive(Clone, Debug, Default, PartialEq, Eq)]
40pub struct GroupProvenance {
41    rule_sets: BTreeMap<RuleSetId, Vec<Contribution>>,
42    rules: BTreeMap<RuleId, RuleStatus>,
43    nodes: BTreeMap<NodeId, (&'static str, String)>,
44}
45
46impl GroupProvenance {
47    /// The provenance of `effective`, naming its nodes from `hierarchy`. A
48    /// node the hierarchy does not hold is named by its ID.
49    #[must_use]
50    pub fn new(effective: &EffectivePolicy, hierarchy: &ValidatedHierarchy) -> Self {
51        let mut out = Self {
52            rule_sets: BTreeMap::new(),
53            rules: effective
54                .rules()
55                .iter()
56                .map(|(id, p)| (*id, p.status))
57                .collect(),
58            nodes: BTreeMap::new(),
59        };
60        for p in effective.rules().values() {
61            out.rule_sets
62                .entry(p.rule_set_id)
63                .or_insert_with(|| p.contributions.clone());
64        }
65        let mut name = |id: NodeId| {
66            out.nodes.entry(id).or_insert_with(|| {
67                hierarchy.node(id).map_or_else(
68                    || ("node", id.to_string()),
69                    |n| (n.kind.as_str(), n.name.clone()),
70                )
71            });
72        };
73        let ids: Vec<NodeId> = out
74            .rule_sets
75            .values()
76            .flatten()
77            .flat_map(|c| core::iter::once(c.node).chain(c.template_for))
78            .chain(out.rules.values().filter_map(|s| match s {
79                RuleStatus::Suppressed { by } => Some(*by),
80                RuleStatus::Reasserted { suppressed_by } => Some(*suppressed_by),
81                RuleStatus::Active | RuleStatus::Disabled => None,
82            }))
83            .collect();
84        for id in ids {
85            name(id);
86        }
87        out
88    }
89
90    /// The group steps of one rule, outermost first, then the step of the
91    /// node that suppressed it, if any. Empty for a rule set or rule this
92    /// provenance does not hold.
93    #[must_use]
94    pub fn steps(&self, set: RuleSetId, rule: RuleId, privacy: &Privacy) -> Vec<InheritanceStep> {
95        let mut out: Vec<InheritanceStep> = self
96            .rule_sets
97            .get(&set)
98            .into_iter()
99            .flatten()
100            .map(|c| self.step(c.node, Some(via(c)), "contributed", c.template_for, privacy))
101            .collect();
102        match self.rules.get(&rule) {
103            Some(RuleStatus::Suppressed { by }) => {
104                out.push(self.step(*by, None, "suppressed", None, privacy));
105            }
106            Some(RuleStatus::Reasserted { suppressed_by }) => {
107                out.push(self.step(
108                    *suppressed_by,
109                    None,
110                    "suppression_overridden",
111                    None,
112                    privacy,
113                ));
114            }
115            _ => {}
116        }
117        out
118    }
119
120    /// Whether a hierarchy node suppressed `rule` (`suppress_inherited`),
121    /// so that it is disabled in the effective policy.
122    #[must_use]
123    pub fn suppressed(&self, rule: RuleId) -> bool {
124        matches!(self.rules.get(&rule), Some(RuleStatus::Suppressed { .. }))
125    }
126
127    fn step(
128        &self,
129        id: NodeId,
130        via: Option<String>,
131        outcome: &'static str,
132        template_for: Option<NodeId>,
133        privacy: &Privacy,
134    ) -> InheritanceStep {
135        let (kind, name) = self
136            .nodes
137            .get(&id)
138            .map_or(("node", id.to_string()), |(k, n)| (*k, n.clone()));
139        InheritanceStep {
140            node: kind,
141            id: id.to_string(),
142            name: privacy.text(&name).0,
143            via,
144            outcome: Some(outcome),
145            template_for: template_for.map(|t| t.to_string()),
146        }
147    }
148}
149
150/// How a node contributed a rule set, as the trace writes it.
151fn via(c: &Contribution) -> String {
152    match &c.via {
153        Via::Policy { policy, slot_order } => format!("policy {policy} (slot {slot_order})"),
154        Via::Library { library, version } => format!("library {library} v{version}"),
155        Via::Category {
156            category,
157            library,
158            version,
159        } => format!("category {category}: library {library} v{version}"),
160        Via::Node => "node settings".to_owned(),
161    }
162}
