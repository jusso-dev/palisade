1//! The effective-policy view of one device (#41): what it enforces.
2//!
3//! [`effective_view`] flattens the policy for a device in a given state:
4//! every rule with where it comes from and whether it takes part, the
5//! settings with their sources, the effective mode per object type, the
6//! active override session and lockdown, and the protection status (#36).
7//! The view reads one policy document, so every rule's provenance is that
8//! document's rule set. A device's effective policy from the group
9//! hierarchy (#37, `hierarchy::EffectivePolicy`) records group provenance
10//! too; showing it here is a follow-up under #41.
11

23/// One rule as the device sees it.
24#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
25pub struct EffectiveRule {
26    /// The rule.
27    pub rule: RuleRef,
28    /// Its name.
29    pub name: String,
30    /// Its action.
31    pub action: ActionKind,
32    /// Its stage, or `null` for `defer`.
33    pub stage: Option<RuleStage>,
34    /// Whether it takes part in evaluation: it is enabled and the device
35    /// is not disabled, as [`Evaluator::evaluate_in`] decides.
36    ///
37    /// [`Evaluator::evaluate_in`]: crate::evaluate::Evaluator::evaluate_in
38    pub active: bool,
39    /// Why not, if it does not: `disabled` (the rule is disabled) or
40    /// `device_disabled` (the device mode is `disabled`, so no rule runs).
41    pub inactive_reason: Option<&'static str>,
42    /// Its rule set's `mode_override`.
43    pub rule_set_mode: Option<Mode>,
44    /// Where it comes from, outermost first.
45    pub provenance: Vec<InheritanceStep>,
46}

69/// What a device enforces.
70#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
71pub struct EffectiveView {
72    /// The policy.
73    pub policy_id: PolicyId,
74    /// Its revision.
75    pub policy_revision: u64,
76    /// The device mode (#36).
77    pub device_mode: Mode,
78    /// The protection status.
79    pub protection: ProtectionStatus,
80    /// Emergency lockdown is set.
81    pub lockdown: bool,
82    /// The active override session: `audit`, `full_bypass` or `null`.
83    pub override_session: Option<&'static str>,
84    /// The device is disconnected.
85    pub disconnected: bool,
86    /// The mode of every object type.
87    pub modes: Vec<TypeMode>,
88    /// The settings, by key.
89    pub settings: Vec<EffectiveSetting>,
90    /// Every rule of the referenced rule sets, in policy order.
91    pub rules: Vec<EffectiveRule>,
92}

94/// The effective view of a device in `state` under `policy`.
95#[must_use]
96pub fn effective_view(policy: &ValidatedPolicy, state: &DeviceState) -> EffectiveView {
97    let doc = policy.document();
98    let p = &doc.policy;
99    let sets: BTreeMap<_, _> = doc.rule_sets.iter().map(|s| (s.id, s)).collect();
100    // The evaluator runs no rule on a disabled device. The resolved mode is
101    // used, because lockdown overrides a disabled device mode.
102    let device_mode = mode::device_mode(p, state);
103    let device_disabled = device_mode == Mode::Disabled;
104    let mut rules = Vec::new();
105    for id in &p.rule_sets {
106        let Some(set) = sets.get(id).copied() else {
107            continue;
108        };
109        for rule in &set.rules {
110            rules.push(EffectiveRule {
111                rule: RuleRef::new(p, set, rule),
112                name: rule.name.clone(),
113                action: rule.action.kind(),
114                stage: RuleStage::of(set, rule),
115                active: rule.enabled && !device_disabled,
116                inactive_reason: if !rule.enabled {
117                    Some("disabled")
118                } else if device_disabled {
119                    Some("device_disabled")
120                } else {
121                    None
122                },
123                rule_set_mode: set.mode_override,
124                provenance: vec![
125                    InheritanceStep {
126                        node: "policy",
127                        id: p.id.to_string(),
128                        name: p.name.clone(),
129                    },
130                    InheritanceStep {
131                        node: "rule_set",
132                        id: set.id.to_string(),
133                        name: set.name.clone(),
134                    },
135                ],
136            });
137        }
138    }
139    let modes = ObjectType::ALL
140        .into_iter()
141        .map(|t| TypeMode {
142            object_type: t,
143            mode: mode::object_mode(p, state, t),
144        })
145        .collect();
146    EffectiveView {
147        policy_id: p.id,
148        policy_revision: p.revision,
149        device_mode,
150        protection: ProtectionStatus::of(p, state),
151        lockdown: state.lockdown,
152        override_session: state.session.map(|s| match s {
153            SessionKind::Audit => "audit",
154            SessionKind::FullBypass => "full_bypass",
155        }),
156        disconnected: state.connectivity == mode::Connectivity::Disconnected,
157        modes,
158        settings: settings(p),
159        rules,
160    }
161}

163/// The settings, each marked `policy` if it differs from the default.
164fn settings(p: &crate::model::Policy) -> Vec<EffectiveSetting> {
165    let actual = serde_json::to_value(&p.settings).unwrap_or_default();
166    let default = serde_json::to_value(crate::model::PolicySettings::default()).unwrap_or_default();
167    let mut out = Vec::new();
168    if let serde_json::Value::Object(map) = &actual {
169        for (k, v) in map {
170            out.push(EffectiveSetting {
171                key: k.clone(),
172                value: v.clone(),
173                source: if default.get(k) == Some(v) {
174                    "default"
175                } else {
176                    "policy"
177                },
178            });
179        }
180    }
181    // Keys the default writes but the policy's settings leave out (empty
182    // maps) are still settings.
183    if let serde_json::Value::Object(map) = &default {
184        for (k, v) in map {
185            if actual.get(k).is_none() {
186                out.push(EffectiveSetting {
187                    key: k.clone(),
188                    value: v.clone(),
189                    source: "default",
190                });
191            }
192        }
193    }
194    out.push(EffectiveSetting {
195        key: "mode".to_owned(),
196        value: serde_json::to_value(p.mode).unwrap_or_default(),
197        source: "policy",
198    });
199    out.push(EffectiveSetting {
200        key: "precedence_model".to_owned(),
201        value: serde_json::to_value(p.precedence_model).unwrap_or_default(),
202        source: if p.precedence_model == crate::model::PrecedenceModel::default() {
203            "default"
204        } else {
205            "policy"
206        },
207    });
208    out.sort_by(|a, b| a.key.cmp(&b.key));
209    out
210}
