1//! Enforcement readiness (#41, #36): how many would-block events each
2//! device had over a period, and whether a move to `enforce` is safe.
3//!
4//! A would-block is an event that enforce mode would block but the
5//! device's mode let run (`allowed_audit`), recomputed by the simulator
6//! from the recorded facts, so it reflects the policy as it is now, not as
7//! it was when the event was recorded. An event blocked anyway is not a
8//! would-block: enforcing changes nothing for it.
9//!
10//! [`readiness`] reports, over the events in `[until − days, until)`:
11//!
12//! - every device, with its would-block count, ready when the count is at
13//!   most `max_would_block`;
14//! - the objects with the most would-blocks, with a draft allow rule for
15//!   each (the trace's suggestion; the recommendation flow is #5);
16//! - the gate of #36: "N would-block events across M devices in the last D
17//!   days", and whether a `readiness_ack` is required before the
18//!   transition, which is when N exceeds the configured threshold.
19

31/// The most top objects listed.
32pub const MAX_TOP_OBJECTS: usize = 25;

34/// The most devices listed.
35pub const MAX_DEVICES: usize = 10_000;

37/// What a readiness check counts.
38#[derive(Clone, Copy, Debug, PartialEq, Eq)]
39pub struct ReadinessOptions {
40    /// The end of the window, in wall-clock seconds (exclusive).
41    pub until: i64,
42    /// The window's length, in days.
43    pub days: u32,
44    /// A device is ready with at most this many would-blocks.
45    pub max_would_block: u64,
46    /// The gate requires a `readiness_ack` when the total would-block count
47    /// is above this. `None`: never required.
48    pub ack_threshold: Option<u64>,
49    /// How many objects to list.
50    pub top_objects: usize,
51}

53/// One device.
54#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
55pub struct DeviceReadiness {
56    /// The device.
57    pub device_id: String,
58    /// Events in the window.
59    pub events: u64,
60    /// Would-blocks in the window.
61    pub would_block: u64,
62    /// `would_block <= max_would_block`.
63    pub ready: bool,
64}

66/// One object with would-blocks.
67#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
68pub struct TopObject {
69    /// The object: `sha256:…` or `path:…`.
70    pub object: String,
71    /// Its would-blocks.
72    pub would_block: u64,
73    /// On how many devices.
74    pub devices: u64,
75    /// The rule enforce mode would block it by, or `null` for the implicit
76    /// deny.
77    pub blocked_by: Option<String>,
78    /// A draft rule that would allow it, for the recommendation flow (#5).
79    pub recommended_rule: Option<RuleDraft>,
80}

82/// The gate before a move to `enforce` (#36).
83#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
84pub struct ReadinessGate {
85    /// Would-blocks in the window.
86    pub would_block_events: u64,
87    /// Devices with at least one.
88    pub devices_affected: u64,
89    /// The window, in days.
90    pub days: u32,
91    /// A `readiness_ack` is required before the transition.
92    pub ack_required: bool,
93    /// "N would-block events across M devices in the last D days".
94    pub message: String,
95}

97/// The report.
98#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
99pub struct ReadinessReport {
100    /// The window, in wall-clock seconds: `[from, until)`.
101    pub window: (i64, i64),
102    /// Events in the window.
103    pub events: u64,
104    /// Events left out: outside the window or without a timestamp.
105    pub events_outside_window: u64,
106    /// Devices, by device ID.
107    pub devices: Vec<DeviceReadiness>,
108    /// Devices past [`MAX_DEVICES`].
109    pub devices_omitted: u64,
110    /// Ready devices.
111    pub ready_devices: u64,
112    /// The objects with the most would-blocks.
113    pub top_objects: Vec<TopObject>,
114    /// The gate.
115    pub gate: ReadinessGate,
116}

118/// Checks how ready the devices in `snapshots` are for `enforce` under
119/// `policy`.
120///
121/// # Errors
122///
123/// [`SimError`] if a snapshot cannot be used or the policy cannot be
124/// compiled.
125pub fn readiness(
126    policy: &ValidatedPolicy,
127    snapshots: &[FactsSnapshot],
128    options: &ReadinessOptions,
129) -> Result<ReadinessReport, SimError> {
130    // Every snapshot is checked, inside the window or not, so the result
131    // never depends on the window hiding a bad input.
132    super::simulate::check_snapshots(snapshots).map_err(SimError::Snapshot)?;
133    let sim = Simulator::new(policy).map_err(build)?;
134    let from = options
135        .until
136        .saturating_sub(i64::from(options.days).saturating_mul(86_400));
137    let in_window: Vec<&FactsSnapshot> = snapshots
138        .iter()
139        .filter(|s| s.timestamp.is_some_and(|t| t >= from && t < options.until))
140        .collect();
141    let owned: Vec<FactsSnapshot> = in_window.iter().map(|s| (*s).clone()).collect();
142    let decisions = sim.decide_all(&owned).map_err(SimError::Snapshot)?;
143
144    let mut devices: BTreeMap<String, (u64, u64)> = BTreeMap::new();
145    let mut objects: BTreeMap<String, ObjectTally> = BTreeMap::new();
146    let mut total = 0u64;
147    for (i, (s, d)) in owned.iter().zip(&decisions).enumerate() {
148        let device = s
149            .device_id
150            .clone()
151            .unwrap_or_else(|| "(unknown)".to_owned());
152        let e = devices.entry(device.clone()).or_default();
153        e.0 += 1;
154        let would = d
155            .as_ref()
156            .is_some_and(|d| d.outcome() == Outcome::AllowedAudit && d.would_have_been().stops());
157        if would {
158            e.1 += 1;
159            total += 1;
160            let key = object_key(s).unwrap_or_else(|| "(unknown)".to_owned());
161            let o = objects
162                .entry(key)
163                .or_insert_with(|| (0, BTreeSet::new(), None, i));
164            o.0 += 1;
165            o.1.insert(device);
166            if o.2.is_none() {
167                o.2 = d
168                    .as_ref()
169                    .and_then(|d| d.matched_rule())
170                    .map(|r| r.rule_id.to_string());
171            }
172        }
173    }
174    let affected = devices.values().filter(|(_, w)| *w > 0).count() as u64;
175    let ready_devices = devices
176        .values()
177        .filter(|(_, w)| *w <= options.max_would_block)
178        .count() as u64;
179    let devices_omitted = devices.len().saturating_sub(MAX_DEVICES) as u64;
180    let device_list = devices
181        .into_iter()
182        .take(MAX_DEVICES)
183        .map(|(device_id, (events, would_block))| DeviceReadiness {
184            device_id,
185            events,
186            would_block,
187            ready: would_block <= options.max_would_block,
188        })
189        .collect();
190
191    let mut top: Vec<(String, ObjectTally)> = objects.into_iter().collect();
192    top.sort_by(|a, b| b.1.0.cmp(&a.1.0).then_with(|| a.0.cmp(&b.0)));
193    top.truncate(options.top_objects.min(MAX_TOP_OBJECTS));
194    let mut top_objects = Vec::with_capacity(top.len());
195    for (object, (would_block, devs, blocked_by, first)) in top {
196        let recommended_rule = sim
197            .trace(&owned[first], &TraceOptions::at(TraceLevel::Full))
198            .ok()
199            .and_then(|t| t.allow_suggestion);
200        top_objects.push(TopObject {
201            object,
202            would_block,
203            devices: devs.len() as u64,
204            blocked_by,
205            recommended_rule,
206        });
207    }
208    let ack_required = options.ack_threshold.is_some_and(|t| total > t);
209    Ok(ReadinessReport {
210        window: (from, options.until),
211        events: owned.len() as u64,
212        events_outside_window: (snapshots.len() - owned.len()) as u64,
213        devices: device_list,
214        devices_omitted,
215        ready_devices,
216        top_objects,
217        gate: ReadinessGate {
218            would_block_events: total,
219            devices_affected: affected,
220            days: options.days,
221            ack_required,
222            message: format!(
223                "{total} would-block {} across {affected} {} in the last {} {}",
224                if total == 1 { "event" } else { "events" },
225                if affected == 1 { "device" } else { "devices" },
226                options.days,
227                if options.days == 1 { "day" } else { "days" },
228            ),
229        },
230    })
231}
