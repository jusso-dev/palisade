1//! The trace's own view of every rule: its match tree as an arena of
2//! compiled criteria, evaluated leaf by leaf without short-circuiting, so
3//! the trace can say which criteria matched, which did not, and why.
4//!
5//! The evaluator ([`crate::evaluate`]) stays the authority on which rules
6//! applied: the trace takes that set from it. This view only explains each
7//! result. It uses the same compiled criteria ([`CompiledCriterion`]) and
8//! the same Kleene logic as [`crate::condition`], so the two agree; the
9//! tests check that over random policies.
10

22/// A three-valued result, as a trace writes it.
23#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
24#[serde(rename_all = "snake_case")]
25pub enum TriState {
26    /// The test holds.
27    Match,
28    /// The test does not hold.
29    NoMatch,
30    /// A fact the test needs is missing or cannot be trusted.
31    Unknown,
32}
33
34impl From<Tri> for TriState {
35    fn from(t: Tri) -> Self {
36        match t {
37            Tri::Match => Self::Match,
38            Tri::NoMatch => Self::NoMatch,
39            Tri::Unknown => Self::Unknown,
40        }
41    }
42}

44/// Which pipeline stage a rule takes part in (#35).
45#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
46#[serde(rename_all = "snake_case")]
47pub enum RuleStage {
48    /// Stage 1: a rule of a mandatory rule set.
49    MandatoryDeny,
50    /// Stage 2: an `ignore` rule.
51    Exclusion,
52    /// Stage 4: an allow or deny for the precedence model.
53    Policy,
54    /// An `audit` rule: an annotation, never a decision.
55    Annotation,
56    /// A `trust_source` rule: a grant if the object is allowed.
57    TrustSource,
58}
59
60impl RuleStage {
61    /// The stage of a rule, exactly as the evaluator assigns it. `None` for
62    /// `defer`, which the evaluator never evaluates.
63    pub(crate) const fn of(set: &RuleSet, rule: &Rule) -> Option<Self> {
64        if set.mandatory {
65            return Some(Self::MandatoryDeny);
66        }
67        Some(match rule.action {
68            Action::Ignore {} => Self::Exclusion,
69            Action::Audit {} => Self::Annotation,
70            Action::TrustSource(_) => Self::TrustSource,
71            Action::Defer {} => return None,
72            Action::Allow {}
73            | Action::Deny(_)
74            | Action::AllowContained(_)
75            | Action::AllowElevated(_) => Self::Policy,
76        })
77    }
78}

100/// One rule of the policy, enabled or not, ready to be explained.
101#[derive(Debug)]
102pub(crate) struct TracedRule<'p> {
103    pub set: &'p RuleSet,
104    pub rule: &'p Rule,
105    pub stage: RuleStage,
106    pub spec: Spec,
107    pub reference: RuleRef,
108    unknown_as: UnknownAs,
109    unparseable: Unparseable,
110    nodes: Vec<Node>,
111    pub leaves: Vec<Leaf<'p>>,
112}

114/// The result of one criterion.
115#[derive(Clone, Copy, Debug)]
116pub(crate) struct LeafResult {
117    /// The criterion's own result.
118    pub raw: Tri,
119    /// The result the tree used: `raw`, unless the rule reads unparseable
120    /// versions as matching (`on_unparseable: match`), or the criterion is
121    /// `safety_critical` in a mandatory deny.
122    pub effective: Tri,
123    pub notes: Notes,
124}
125
126impl LeafResult {
127    /// Whether the criterion pulled the rule towards matching: `Match`, or
128    /// `NoMatch` under an odd number of `not` nodes.
129    pub fn satisfied(&self, negated: bool) -> bool {
130        self.effective == if negated { Tri::NoMatch } else { Tri::Match }
131    }
132}

134/// The result of evaluating one traced rule against one subject.
135#[derive(Clone, Debug)]
136pub(crate) struct RuleResult {
137    /// Whether the rule's `object_types` include the subject's type.
138    pub type_applies: bool,
139    /// The tree's result. `NoMatch` when the type does not apply.
140    pub tree: Tri,
141    /// Whether the rule applies: the type applies and the tree is `Match`,
142    /// or `Unknown` for a deny with `unknown_as: match`.
143    pub applies: bool,
144    nodes: Vec<Tri>,
145    /// One result per leaf, in [`TracedRule::leaves`] order. Empty when
146    /// the type does not apply.
147    pub leaves: Vec<LeafResult>,
148}

150impl<'p> TracedRule<'p> {
151    /// Builds the traced form of `rule`. Returns `Ok(None)` for `defer`.
152    pub fn build(
153        policy: &Policy,
154        set: &'p RuleSet,
155        rule: &'p Rule,
156    ) -> Result<Option<Self>, BuildError> {
157        let Some(stage) = RuleStage::of(set, rule) else {
158            return Ok(None);
159        };
160        // The same compilation and limits as the evaluator, so a tree the
161        // evaluator accepts is within the depth that `add` recurses to.
162        CompiledMatch::compile(&rule.match_).map_err(|errors| BuildError {
163            rule_id: rule.id,
164            errors,
165        })?;
166        let (unknown_as, unparseable) = match &rule.action {
167            Action::Deny(o) => (
168                o.unknown_as,
169                match o.on_unparseable {
170                    UnknownAs::Match => Unparseable::FavourMatch,
171                    UnknownAs::NoMatch => Unparseable::Unknown,
172                },
173            ),
174            _ => (UnknownAs::NoMatch, Unparseable::Unknown),
175        };
176        let mut out = Self {
177            set,
178            rule,
179            stage,
180            spec: Spec::of(&rule.match_),
181            reference: RuleRef::new(policy, set, rule),
182            unknown_as,
183            unparseable,
184            nodes: Vec::new(),
185            leaves: Vec::new(),
186        };
187        out.add(&rule.match_, "/match".to_owned(), false)
188            .map_err(|errors| BuildError {
189                rule_id: rule.id,
190                errors,
191            })?;
192        Ok(Some(out))
193    }
194
195    fn add(
196        &mut self,
197        node: &'p Match,
198        pointer: String,
199        negated: bool,
200    ) -> Result<u32, Vec<crate::condition::TreeError>> {
201        let entry = match node {
202            Match::All(items) | Match::Any(items) => {
203                let key = if matches!(node, Match::All(_)) {
204                    "all"
205                } else {
206                    "any"
207                };
208                let mut ids = Vec::with_capacity(items.len());
209                for (i, child) in items.iter().enumerate() {
210                    ids.push(self.add(child, format!("{pointer}/{key}/{i}"), negated)?);
211                }
212                if matches!(node, Match::All(_)) {
213                    Node::All(ids)
214                } else {
215                    Node::Any(ids)
216                }
217            }
218            Match::Not(child) => Node::Not(self.add(child, format!("{pointer}/not"), !negated)?),
219            // As the evaluator: the hierarchy resolves every reference
220            // before evaluation (#37), and validation rejects one left over.
221            Match::Ref(_) => {
222                return Err(vec![crate::condition::TreeError::UnresolvedRef {
223                    pointer: format!("{pointer}/ref"),
224                }]);
225            }
226            Match::Criterion(c) => {
227                let compiled = CompiledCriterion::compile(c).map_err(|errors| {
228                    vec![crate::condition::TreeError::Criterion {
229                        pointer: format!("{pointer}/criterion"),
230                        errors,
231                    }]
232                })?;
233                self.leaves.push(Leaf {
234                    criterion: c,
235                    compiled,
236                    pointer: format!("{pointer}/criterion"),
237                    negated,
238                });
239                Node::Leaf(index(self.leaves.len() - 1))
240            }
241        };
242        self.nodes.push(entry);
243        Ok(index(self.nodes.len() - 1))
244    }

246    /// The rule's action.
247    pub fn action(&self) -> ActionKind {
248        self.rule.action.kind()
249    }

251    /// Evaluates every criterion of the rule, without short-circuiting.
252    pub fn evaluate(&self, subject: &Subject<'_>) -> RuleResult {
253        if !self.rule.object_types.is_empty()
254            && !self.rule.object_types.contains(&subject.object_type)
255        {
256            return RuleResult {
257                type_applies: false,
258                tree: Tri::NoMatch,
259                applies: false,
260                nodes: Vec::new(),
261                leaves: Vec::new(),
262            };
263        }
264        let ctx = EvalContext {
265            action: self.rule.action.kind(),
266            variables: subject.variables,
267            context: subject.context,
268        };
269        let mut nodes = vec![Tri::Unknown; self.nodes.len()];
270        let mut leaves = Vec::with_capacity(self.leaves.len());
271        for leaf in &self.leaves {
272            let e = leaf.compiled.eval(subject.facts, &ctx);
273            // As the evaluator (`CompiledMatch::eval_mandatory`): a
274            // safety-critical criterion of a mandatory deny that is unknown
275            // counts towards the deny applying (#32, #87).
276            let fail_closed = self.stage == RuleStage::MandatoryDeny
277                && e.result == Tri::Unknown
278                && leaf.compiled.safety_critical();
279            let effective = if fail_closed
280                || (e.result == Tri::Unknown
281                    && self.unparseable == Unparseable::FavourMatch
282                    && e.notes.contains(Note::VersionUnparseable))
283            {
284                Tri::from_bool(!leaf.negated)
285            } else {
286                e.result
287            };
288            leaves.push(LeafResult {
289                raw: e.result,
290                effective,
291                notes: e.notes,
292            });
293        }
294        // Children always come before their parent in the arena.
295        for (i, node) in self.nodes.iter().enumerate() {
296            nodes[i] = match node {
297                Node::All(c) => c
298                    .iter()
299                    .fold(Tri::Match, |acc, &n| acc.and(nodes[n as usize])),
300                Node::Any(c) => c
301                    .iter()
302                    .fold(Tri::NoMatch, |acc, &n| acc.or(nodes[n as usize])),
303                Node::Not(n) => nodes[*n as usize].not(),
304                Node::Leaf(l) => leaves[*l as usize].effective,
305            };
306        }
307        let tree = nodes.last().copied().unwrap_or(Tri::Unknown);
308        let applies = match tree {
309            Tri::Match => true,
310            Tri::Unknown => self.unknown_as == UnknownAs::Match,
311            Tri::NoMatch => false,
312        };
313        RuleResult {
314            type_applies: true,
315            tree,
316            applies,
317            nodes,
318            leaves,
319        }
320    }

322    /// The leaves that kept the rule from matching, each with the result it
323    /// needed (`Match`, or `NoMatch` under `not`), in document order. Empty
324    /// when the tree matched.
325    pub fn failing(&self, result: &RuleResult) -> Vec<(usize, Tri)> {
326        let mut out = Vec::new();
327        if result.type_applies && !self.nodes.is_empty() {
328            self.collect_failing(index(self.nodes.len() - 1), Tri::Match, result, &mut out);
329        }
330        out.sort_by_key(|(leaf, _)| *leaf);
331        out
332    }
333
334    fn collect_failing(&self, id: u32, want: Tri, r: &RuleResult, out: &mut Vec<(usize, Tri)>) {
335        if r.nodes[id as usize] == want {
336            return;
337        }
338        match &self.nodes[id as usize] {
339            // `all` wanted `Match`: every child that is not `Match` failed.
340            // `all` wanted `NoMatch` (under `not`): no child was `NoMatch`,
341            // so every child failed. `any` is the mirror image. A child that
342            // already gives `want` returns at once.
343            Node::All(children) | Node::Any(children) => {
344                for &c in children {
345                    self.collect_failing(c, want, r, out);
346                }
347            }
348            Node::Not(child) => {
349                let flipped = if want == Tri::Match {
350                    Tri::NoMatch
351                } else {
352                    Tri::Match
353                };
354                self.collect_failing(*child, flipped, r, out);
355            }
356            Node::Leaf(l) => out.push((*l as usize, want)),
357        }
358    }
359
360    /// For each leaf, whether its result is part of why the tree has its
361    /// result: walking down from the root, a node's children count when
362    /// their result is the node's (every child of an `all` that matched or
363    /// an `any` that did not; the matching children of an `any`, the
364    /// failing children of an `all`; the unknown children of an unknown
365    /// node), and a `not` node's child always counts. A leaf in a branch
366    /// that Kleene logic absorbed does not count.
367    pub fn contributing(&self, result: &RuleResult) -> Vec<bool> {
368        let mut out = vec![false; self.leaves.len()];
369        if result.type_applies && !self.nodes.is_empty() {
370            self.mark(index(self.nodes.len() - 1), result, &mut out);
371        }
372        out
373    }
374
375    fn mark(&self, id: u32, r: &RuleResult, out: &mut [bool]) {
376        let mine = r.nodes[id as usize];
377        match &self.nodes[id as usize] {
378            Node::All(children) | Node::Any(children) => {
379                for &c in children {
380                    if r.nodes[c as usize] == mine {
381                        self.mark(c, r, out);
382                    }
383                }
384            }
385            Node::Not(child) => self.mark(*child, r, out),
386            Node::Leaf(l) => {
387                if let Some(slot) = out.get_mut(*l as usize) {
388                    *slot = true;
389                }
390            }
391        }
392    }
393
394    /// Whether the rule has a positive identity criterion (an object fact,
395    /// not under `not`) that matched: the anchor of a near miss.
396    pub fn anchor_matched(&self, result: &RuleResult) -> bool {
397        self.leaves.iter().zip(&result.leaves).any(|(leaf, r)| {
398            !leaf.negated
399                && leaf.criterion.operator.is_positive()
400                && leaf.criterion.field.strength().is_some()
401                && r.effective == Tri::Match
402        })
403    }
404
405    /// The number of criteria that pulled the rule towards matching.
406    pub fn satisfied_count(&self, result: &RuleResult) -> usize {
407        self.leaves
408            .iter()
409            .zip(&result.leaves)
410            .filter(|(leaf, r)| r.satisfied(leaf.negated))
411            .count()
412    }
413}

416pub(crate) fn field_name(field: Field) -> String {

418        Ok(serde_json::Value::String(s)) => s,
