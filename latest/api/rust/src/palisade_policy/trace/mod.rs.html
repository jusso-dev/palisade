1//! The decision trace, explanations and policy simulation (#41).
2//!
3//! Every decision can be explained. This module has three parts:
4//!
5//! - **Trace** ([`Tracer`], [`Trace`]): a structured, bounded record of one
6//!   evaluation. It lists the stages that ran, the rules that matched and
7//!   the part each played, the rules that nearly matched and the criteria
8//!   that failed them, the rules that would have matched but are disabled,
9//!   the facts that were missing and what that did to the decision, the
10//!   warnings, the effective mode with each step of its resolution, and the
11//!   decision itself. Its JSON form is versioned
12//!   ([`TRACE_SCHEMA_VERSION`]) and snapshot-tested.
13//! - **Explain** ([`explain`]): turns a trace into paragraphs and a table
14//!   for people, rendered as plain text, Markdown or one short sentence,
15//!   with localisable message templates.
16//! - **Simulate** ([`simulate`], [`readiness`], [`policy_test`],
17//!   [`effective`]): runs a policy, or a proposed change to it, over
18//!   recorded facts and reports what would change, how ready devices are
19//!   for enforce mode, whether policy-as-code tests pass, and what a device
20//!   enforces.
21//!
22//! # The trace is the evaluator's
23//!
24//! [`Tracer::trace`] calls [`Evaluator::evaluate_in`], and the trace
25//! embeds that decision unchanged. The set of rules that matched is taken
26//! from the evaluator too. The trace then explains each rule by evaluating
27//! its criteria one by one, with the same compiled criteria and the same
28//! three-valued logic, without stopping early. It never decides anything
29//! itself. Tests check over random policies, facts and device states that
30//! the trace and the evaluator agree.
31//!
32//! # Levels
33//!
34//! | Level | Contents |
35//! |---|---|
36//! | `minimal` | The decision, the deciding rule and the effective mode. Built from the decision alone, with no further evaluation. |
37//! | `standard` | Adds the stages, every rule that matched, the deciding rule's criteria, disabled rules that would have matched, rules whose scope excluded the subject, unknown facts, warnings, mode resolution steps, provenance, and a draft rule that would allow a blocked object. |
38//! | `full` | Adds near misses, with the failing criteria and the observed and expected values, and every fact the evaluation read. |
39//!
40//! [`LevelPolicy`] chooses the level the agent records: `minimal` for an
41//! object that runs, `standard` for a block or a would-block, and `full`
42//! with a per-device debug flag or during a time-boxed diagnostic session.
43//! The simulator always records `full`.
44//!
45//! # Bounds
46//!
47//! A trace never grows with the size of the policy. Each list has a limit
48//! ([`limits`]), and what was left out is counted in [`Trace::omitted`]
49//! and in each entry's `*_omitted` field. Every value is cut to
50//! [`privacy::MAX_VALUE_CHARS`] characters, and at most
51//! [`privacy::MAX_VALUES`] values of one fact are written. The JSON of a
52//! trace is at most [`limits::MAX_TRACE_BYTES`]; a test builds the largest
53//! trace the limits allow and checks it.
54//!
55//! # Privacy
56//!
57//! Facts are written as [`Privacy`] allows, when the trace is built: see
58//! [`privacy`]. The defaults write object facts and paths, withhold the
59//! user, device and session, and redact nothing.

138/// How an agent chooses the trace level of each decision (#41).
139#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
140pub struct LevelPolicy {
141    /// The device's debug flag: every decision is traced at `full`.
142    pub debug_full: bool,
143    /// The end of a diagnostic session, on the agent's monotonic clock in
144    /// seconds. Until then, every decision is traced at `full`.
145    pub diagnostic_until: Option<u64>,
146}
147
148impl LevelPolicy {
149    /// A diagnostic session from `now_monotonic` for `secs` seconds
150    /// ([`DIAGNOSTIC_SESSION_SECS`] by default).
151    #[must_use]
152    pub const fn with_diagnostic_session(mut self, now_monotonic: u64, secs: u64) -> Self {
153        self.diagnostic_until = Some(now_monotonic.saturating_add(secs));
154        self
155    }
156
157    /// The level for a decision made at `now_monotonic`: `full` with the
158    /// debug flag or in a diagnostic session; otherwise `standard` for a
159    /// block or a would-block, and `minimal` for anything else. A device
160    /// that is disabled has no decision and gets `minimal`.
161    #[must_use]
162    pub fn level_for(&self, decision: Option<&Decision>, now_monotonic: u64) -> TraceLevel {
163        if self.debug_full || self.diagnostic_until.is_some_and(|end| now_monotonic < end) {
164            return TraceLevel::Full;
165        }
166        match decision {
167            Some(d) if d.outcome().stops() || d.would_have_been().stops() => TraceLevel::Standard,
168            _ => TraceLevel::Minimal,
169        }
170    }
171}

239/// One evaluation, explained. Build one with [`Tracer::trace`].
240///
241/// Every field is always present, in declaration order: lists are empty

244pub struct Trace {
245    /// [`TRACE_SCHEMA_VERSION`].
246    pub trace_schema_version: &'static str,
247    /// The trace ID, the same as the decision's.
248    pub trace_id: TraceId,

250    pub level: TraceLevel,

252    pub policy_id: PolicyId,
253    /// The policy's revision.
254    pub policy_revision: u64,
255    /// The bundle hash, if the caller supplied one (#43).
256    pub bundle_hash: Option<String>,
257    /// The object's type.
258    pub object_type: ObjectType,
259    /// The effective mode of the decision: `disabled` when the device is
260    /// disabled and nothing was decided.
261    pub effective_mode: Mode,
262    /// How the effective mode was resolved, step by step (#36). Standard
263    /// and full.
264    pub mode_resolution: Vec<ModeStep>,
265    /// The policy's precedence model (#35).
266    pub precedence_model: PrecedenceModel,
267    /// The pipeline stages, in order. Standard and full.
268    pub stages: Vec<StageTrace>,
269    /// The deciding rule, or `null` for the implicit deny or a disabled
270    /// device. Its criteria are listed at standard and full.
271    pub decisive: Option<DecisiveRule>,
272    /// Every rule that matched, by rule ID. Standard and full.
273    pub matched: Vec<MatchedRule>,
274    /// Rules whose identity anchor matched but another criterion did not,

279    /// Rules whose tree matched but whose scope excluded the subject.

282    /// absent, so it is in no other list.
283    pub scope_rejections: Vec<ScopeRejection>,
284    /// Facts the evaluation needed but did not have, and what that did to
285    /// the decision. Standard and full.
286    pub unknown_facts: Vec<UnknownFact>,
287    /// Every fact the evaluated criteria read, by field. Full.
288    pub facts_used: Vec<Observed>,
289    /// Warnings, by code. Standard and full.
290    pub warnings: Vec<Warning>,
291    /// Where the deciding rule comes from, outermost first (#37). Standard
292    /// and full.
293    pub inheritance: Vec<InheritanceStep>,
294    /// The smallest allow rule that would match a blocked object, as a
295    /// draft for approvals and recommendations (#5, #11). Standard and
296    /// full, for a block or a would-block only. The trace never creates a
297    /// rule.
298    pub allow_suggestion: Option<RuleDraft>,
299    /// What the limits left out.
300    pub omitted: Omitted,
301    /// The decision, exactly as the evaluator made it, or `null` for a
302    /// disabled device. Its JSON is the decision's stable form (#34), in
303    /// the same field order, with
304    /// `overridden`, `annotations` and `directives.trust_sources` cut to
305    /// [`limits::MAX_DECISION_LIST`] entries and the number left out added
306    /// as `overridden_omitted`, `annotations_omitted` and
307    /// `trust_sources_omitted`, so a decision over many rules cannot make
308    /// the trace unbounded. The Rust value is never cut.
309    #[serde(serialize_with = "bounded_decision")]
310    pub decision: Option<Decision>,
311}

637/// Builds traces for one policy. It holds an [`Evaluator`], which makes
638/// every decision, and its own view of every rule for the explanations.
639#[derive(Debug)]
640pub struct Tracer<'p> {
641    evaluator: Evaluator<'p>,
642    policy: &'p Policy,
643    rules: Vec<TracedRule<'p>>,
644    by_id: BTreeMap<RuleId, usize>,
645}
646
647impl<'p> Tracer<'p> {
648    /// Compiles the policy for evaluation and tracing.
649    ///
650    /// # Errors
651    ///
652    /// Returns [`BuildError`] if a tree does not compile, which validation
653    /// rules out.
654    pub fn new(validated: &'p ValidatedPolicy) -> Result<Self, BuildError> {
655        Self::with_library(validated, &crate::scope::NoLibrary)
656    }
657
658    /// Compiles the policy for evaluation and tracing, resolving the
659    /// library sets scopes reference through `library` (#37, #38), as
660    /// [`Evaluator::with_library`] does.
661    ///
662    /// # Errors
663    ///
664    /// Returns [`BuildError`] if a tree does not compile, which validation
665    /// rules out.
666    pub fn with_library(
667        validated: &'p ValidatedPolicy,
668        library: &dyn crate::scope::ScopeLibrary,
669    ) -> Result<Self, BuildError> {
670        let evaluator = Evaluator::with_library(validated, library)?;
671        let doc = validated.document();
672        let policy = &doc.policy;
673        let sets: BTreeMap<_, _> = doc.rule_sets.iter().map(|s| (s.id, s)).collect();
674        let mut rules = Vec::new();
675        for id in &policy.rule_sets {
676            let Some(set) = sets.get(id).copied() else {
677                continue;
678            };
679            for rule in &set.rules {
680                if let Some(traced) = TracedRule::build(policy, set, rule)? {
681                    rules.push(traced);
682                }
683            }
684        }
685        let by_id = rules
686            .iter()
687            .enumerate()
688            .map(|(i, r)| (r.rule.id, i))
689            .collect();
690        Ok(Self {
691            evaluator,
692            policy,
693            rules,
694            by_id,
695        })
696    }

762
763    fn trace_inner(
764        &self,
765        subject: &Subject<'_>,
766        extras: &ObjectExtras,
767        state: &DeviceState,
768        trace_id: TraceId,
769        options: &TraceOptions,
770        clock: Option<&dyn NanoClock>,
771    ) -> Trace {
772        let decision = self.evaluator.evaluate_in(subject, state, trace_id);
773        let decider = decision
774            .as_ref()
775            .and_then(|d| d.matched_rule())
776            .and_then(|r| self.traced(r.rule_id));
777        let mut trace = Trace {
778            trace_schema_version: TRACE_SCHEMA_VERSION,
779            trace_id,
780            level: options.level,
781            policy_id: self.policy.id,
782            policy_revision: self.policy.revision,
783            bundle_hash: options.bundle_hash.as_deref().map(|h| clip(h).0),
784            object_type: subject.object_type,
785            effective_mode: decision.as_ref().map_or(Mode::Disabled, Decision::mode),
786            mode_resolution: Vec::new(),
787            precedence_model: self.policy.precedence_model,
788            stages: Vec::new(),
789            decisive: decider.map(|r| DecisiveRule {
790                rule: r.reference,
791                name: clip(&r.rule.name).0,
792                stage: r.stage,
793                action: r.action(),
794                specificity_tier: r.spec.tier.number(),
795                criteria: Vec::new(),
796            }),
797            matched: Vec::new(),
798            near_misses: Vec::new(),
799            inactive_matches: Vec::new(),
800            scope_rejections: Vec::new(),
801            unknown_facts: Vec::new(),
802            facts_used: Vec::new(),
803            warnings: Vec::new(),
804            inheritance: Vec::new(),
805            allow_suggestion: None,
806            omitted: Omitted::default(),
807            decision: None,
808        };
809        if options.level == TraceLevel::Minimal {
810            trace.decision = decision;
811            return trace;
812        }
813        // A disabled device consults no rule (#36): there is nothing to
814        // explain but the mode.
815        if decision.is_none() {
816            trace.mode_resolution = modes::disabled_steps(state);
817            trace.stages = stages(None, None, &[], state, &BTreeMap::new());
818            return trace;
819        }
820        let full = options.level == TraceLevel::Full;
821        let privacy = &options.privacy;
822
823        // The recording pass: every rule, enabled or not, in stage order so
824        // that each stage can be timed.
825        let mut results: Vec<Option<RuleResult>> = vec![None; self.rules.len()];
826        let mut durations: BTreeMap<RuleStage, u64> = BTreeMap::new();
827        for stage in [
828            RuleStage::MandatoryDeny,
829            RuleStage::Exclusion,
830            RuleStage::Policy,
831            RuleStage::Annotation,
832            RuleStage::TrustSource,
833        ] {
834            let start = clock.map(NanoClock::now_ns);
835            for (i, r) in self.rules.iter().enumerate() {
836                if r.stage == stage {
837                    results[i] = Some(r.evaluate(subject));
838                }
839            }
840            if let (Some(c), Some(s)) = (clock, start) {
841                durations.insert(stage, c.now_ns().saturating_sub(s));
842            }
843        }
844        let result = |i: usize| results[i].as_ref();
845        // The scopes of every enabled rule whose type applies (#38), as
846        // the evaluator read them. A rule they take out is absent.
847        let scopes: Vec<Option<crate::scope::ScopeEvaluation>> = self
848            .rules
849            .iter()
850            .enumerate()
851            .map(|(i, r)| {
852                let applies = r.rule.enabled && result(i).is_some_and(|x| x.type_applies);
853                if applies {
854                    self.evaluator.scope_of(r.rule.id, subject)
855                } else {
856                    None
857                }
858            })
859            .collect();
860        let out_of_scope = |i: usize| scopes[i].as_ref().is_some_and(|s| s.result == Tri::NoMatch);
861        // A rule whose scopes are not `Match` cannot be a near miss: its
862        // criteria are not what kept it from applying.
863        let scope_open = |i: usize| scopes[i].as_ref().is_none_or(|s| s.result == Tri::Match);
864        let scope_unknown = |i: usize| scopes[i].as_ref().is_some_and(|s| s.result == Tri::Unknown);
865
866        let matched_ids: BTreeSet<RuleId> =
867            self.evaluator.matching_ids(subject).into_iter().collect();
868        let decision_ref = decision.as_ref();
869
870        // The deciding rule's criteria.
871        if let (Some(d), Some(r)) = (trace.decisive.as_mut(), decider) {
872            if let Some(res) = self.by_id.get(&r.rule.id).and_then(|&i| result(i)) {
873                d.criteria = criteria(r, res, subject, privacy, full, None);
874            }
875        }
876
877        // Matched rules, by rule ID.
878        let mut matched: Vec<&TracedRule<'p>> = matched_ids
879            .iter()
880            .filter_map(|id| self.traced(*id))
881            .collect();
882        matched.sort_by_key(|r| r.rule.id);
883        trace.omitted.matched = count(matched.len().saturating_sub(MAX_MATCHED));
884        trace.matched = matched
885            .iter()
886            .take(MAX_MATCHED)
887            .map(|r| {
888                let i = self.by_id.get(&r.rule.id).copied();
889                let res = i.and_then(result);
890                // It applied only through an unknown: its tree's, or its
891                // scopes' (a deny with `unknown_as: match`).
892                let via_unknown = res.is_some_and(|x| x.applies && x.tree == Tri::Unknown)
893                    || i.is_some_and(scope_unknown);
894                MatchedRule {
895                    rule: r.reference,
896                    name: clip(&r.rule.name).0,
897                    stage: r.stage,
898                    action: r.action(),
899                    specificity_tier: r.spec.tier.number(),
900                    specificity_depth: r.spec.depth,
901                    order: r.rule.order,
902                    priority: r.rule.priority,
903                    role: role_of(r.reference, decision_ref),
904                    via_unknown,
905                }
906            })
907            .collect();
908
909        // Stages.
910        trace.stages = stages(decision_ref, decider, &matched, state, &durations);
911
912        // Inactive matches: rules outside their activity window (disabled,
913        // not yet active or expired) whose tree matches.
914        let mut inactive: Vec<&TracedRule<'p>> = self
915            .rules
916            .iter()
917            .enumerate()
918            .filter(|(i, r)| {
919                !self.active(r.rule) && result(*i).is_some_and(|x| x.tree == Tri::Match)
920            })
921            .map(|(_, r)| r)
922            .collect();
923        inactive.sort_by_key(|r| r.rule.id);
924        trace.omitted.inactive_matches = count(inactive.len().saturating_sub(MAX_INACTIVE));
925        trace.inactive_matches = inactive
926            .iter()
927            .take(MAX_INACTIVE)
928            .map(|r| InactiveMatch {
929                rule: r.reference,
930                name: clip(&r.rule.name).0,
931                action: r.action(),
932                reason: match self.evaluator.activity_of(r.rule) {
933                    crate::lifecycle::Activity::NotYetActive => InactiveReason::NotYetActive,
934                    crate::lifecycle::Activity::Expired => InactiveReason::Expired,
935                    crate::lifecycle::Activity::Disabled | crate::lifecycle::Activity::Active => {
936                        InactiveReason::Disabled
937                    }
938                },
939            })
940            .collect();
941
942        // Scope rejections: the tree matched, the scopes did not.
943        let mut rejected: Vec<(&TracedRule<'p>, crate::scope::ScopeFailure)> = self
944            .rules
945            .iter()
946            .enumerate()
947            .filter(|(i, _)| result(*i).is_some_and(|x| x.tree == Tri::Match))
948            .filter_map(|(i, r)| Some((r, scopes[i].as_ref()?.failure?)))
949            .collect();
950        rejected.sort_by_key(|(r, _)| r.rule.id);
951        trace.omitted.scope_rejections = count(rejected.len().saturating_sub(MAX_INACTIVE));
952        trace.scope_rejections = rejected
953            .iter()
954            .take(MAX_INACTIVE)
955            .map(|(r, f)| ScopeRejection {
956                rule: r.reference,
957                name: clip(&r.rule.name).0,
958                level: scope_level_name(f.level),
959                dimension: f.dimension.map(dimension_name),
960            })
961            .collect();
962
963        // Unknown facts and warnings, over the enabled rules in scope.
964        let mut unknown: BTreeMap<Field, UnknownTally> = BTreeMap::new();
965        let mut warn: BTreeMap<WarningCode, BTreeMap<RuleId, RuleRef>> = BTreeMap::new();
966        // A signer field read with several components or positions is
967        // reported once for each.
968        let mut used: BTreeSet<(Field, Selector)> = BTreeSet::new();
969        for (i, r) in self.rules.iter().enumerate() {
970            let Some(res) = result(i) else { continue };
971            if !self.active(r.rule) || !res.type_applies {
972                continue;
973            }
974            // A scope note is worth a warning on a rule whose tree did not
975            // rule it out: the note is why it applied or not.
976            if let Some(s) = scopes[i].as_ref().filter(|_| res.tree != Tri::NoMatch) {
977                for (note, code) in [
978                    (crate::scope::ScopeNote::ScopeStale, WarningCode::ScopeStale),
979                    (
980                        crate::scope::ScopeNote::ClockUntrusted,
981                        WarningCode::ClockUntrusted,
982                    ),
983                ] {
984                    if s.notes.contains(note) {
985                        warn.entry(code)
986                            .or_default()
987                            .insert(r.reference.rule_id, r.reference);
988                    }
989                }
990            }
991            if out_of_scope(i) {
992                continue;
993            }
994            let contributing = r.contributing(res);
995            for ((leaf, lr), counts) in r.leaves.iter().zip(&res.leaves).zip(contributing) {
996                used.insert((leaf.criterion.field, Selector::of(leaf.criterion)));
997                if lr.notes.contains(Note::PathAclUnsafe) {
998                    warn.entry(WarningCode::PathAclUnsafe)
999                        .or_default()
1000                        .insert(r.reference.rule_id, r.reference);
1001                }
1002                if lr.notes.contains(Note::ShortNameUnresolved) {
1003                    warn.entry(WarningCode::ShortNameUnresolved)
1004                        .or_default()
1005                        .insert(r.reference.rule_id, r.reference);
1006                }
1007                // Only an unknown that left the rule's result unknown had an
1008                // effect on it: one Kleene logic did not absorb.
1009                if counts && lr.effective == Tri::Unknown && res.tree == Tri::Unknown {
1010                    let e = unknown.entry(leaf.criterion.field).or_default();
1011                    e.0.extend(lr.notes.iter().map(Note::as_str));
1012                    e.1.insert(unknown_effect(r));
1013                    e.2.insert(r.reference.rule_id, r.reference);
1014                }
1015            }
1016        }
1017        trace.unknown_facts = unknown
1018            .into_iter()
1019            .map(|(field, (notes, effects, rules))| {
1020                let (rules, rules_omitted) = capped_rules(rules);
1021                UnknownFact {
1022                    field,
1023                    notes: notes.into_iter().collect(),
1024                    effects: effects.into_iter().collect(),
1025                    rules,
1026                    rules_omitted,
1027                }
1028            })
1029            .collect();
1030        let mut warnings: Vec<Warning> = warn
1031            .into_iter()
1032            .map(|(code, rules)| {
1033                let (rules, rules_omitted) = capped_rules(rules);
1034                Warning {
1035                    code,
1036                    rules,
1037                    rules_omitted,
1038                    details: Vec::new(),
1039                }
1040            })
1041            .collect();
1042        if !extras.signature_relaxations.is_empty() {
1043            let mut details: Vec<String> = extras
1044                .signature_relaxations
1045                .iter()
1046                .take(MAX_RULES_PER_ENTRY)
1047                .map(|x| {
1048                    format!(
1049                        "signature {}: {} suppressed {} ({})",
1050                        x.signature,
1051                        relaxation_name(x.flag),
1052                        x.suppressed.as_str(),
1053                        x.suppressed.code()
1054                    )
1055                })
1056                .collect();
1057            details.sort();
1058            details.dedup();
1059            // The rules that rest on a signer: every matched rule that has a
1060            // signer criterion that matched.
1061            let signer_rules: BTreeMap<RuleId, RuleRef> = matched
1062                .iter()
1063                .filter(|r| {
1064                    self.by_id
1065                        .get(&r.rule.id)
1066                        .and_then(|&i| result(i))
1067                        .is_some_and(|res| {
1068                            r.leaves.iter().zip(&res.leaves).any(|(l, lr)| {
1069                                l.criterion.field.is_signer() && lr.effective == Tri::Match
1070                            })
1071                        })
1072                })
1073                .map(|r| (r.rule.id, r.reference))
1074                .collect();
1075            let (rules, rules_omitted) = capped_rules(signer_rules);
1076            warnings.push(Warning {
1077                code: WarningCode::SignatureRelaxed,
1078                rules,
1079                rules_omitted,
1080                details,
1081            });
1082            warnings.sort_by_key(|w| w.code);
1083        }
1084        trace.warnings = warnings;
1085
1086        // Mode resolution.
1087        if let Some(d) = decision_ref {
1088            trace.mode_resolution = modes::steps(
1089                self.policy,
1090                subject,
1091                decider.map(|r| (r.set, r.rule)),
1092                state,
1093                d,
1094            );
1095        } else {
1096            trace.mode_resolution = modes::disabled_steps(state);
1097        }
1098
1099        // Provenance of the deciding rule within the policy document. The
1100        // group steps of #37's effective policy are a follow-up under #41.
1101        if let Some(r) = decider {
1102            trace.inheritance = vec![
1103                InheritanceStep {
1104                    node: "policy",
1105                    id: self.policy.id.to_string(),
1106                    name: clip(&self.policy.name).0,
1107                },
1108                InheritanceStep {
1109                    node: "rule_set",
1110                    id: r.set.id.to_string(),
1111                    name: clip(&r.set.name).0,
1112                },
1113                InheritanceStep {
1114                    node: "rule",
1115                    id: r.rule.id.to_string(),
1116                    name: clip(&r.rule.name).0,
1117                },
1118            ];
1119        }
1120
1121        // A draft allow for a block or a would-block, a prompt included.
1122        // Not for an instance limit (#38): an allow already decided, and
1123        // another allow would not change what limits it.
1124        if decision_ref.is_some_and(|d| {
1125            d.would_have_been().stops()
1126                && d.reason_code() != crate::decision::ReasonCode::InstanceLimit
1127        }) {
1128            trace.allow_suggestion = suggest(subject, privacy, &matched, extras);
1129        }
1130
1131        if full {
1132            let in_scope: Vec<Option<RuleResult>> = results
1133                .iter()
1134                .enumerate()
1135                .map(|(i, r)| if scope_open(i) { r.clone() } else { None })
1136                .collect();
1137            trace.near_misses = self.near_misses(
1138                &in_scope,
1139                &matched_ids,
1140                subject,
1141                privacy,
1142                &mut trace.omitted,
1143            );
1144            trace.facts_used = used
1145                .into_iter()
1146                .map(|(f, sel)| Observed::new(f, observe(f, sel, subject), privacy))
1147                .collect();
1148        }
1149        trace.decision = decision;
1150        trace
1151    }

1153    fn near_misses(
1154        &self,
1155        results: &[Option<RuleResult>],
1156        matched: &BTreeSet<RuleId>,
1157        subject: &Subject<'_>,
1158        privacy: &Privacy,
1159        omitted: &mut Omitted,
1160    ) -> Vec<NearMiss> {
1161        let mut candidates: Vec<(usize, &TracedRule<'p>, &RuleResult)> = self
1162            .rules
1163            .iter()
1164            .zip(results)
1165            .filter_map(|(r, res)| Some((r, res.as_ref()?)))
1166            .filter(|(r, res)| {
1167                self.active(r.rule)
1168                    && res.type_applies
1169                    && !matched.contains(&r.rule.id)
1170                    && r.anchor_matched(res)
1171            })
1172            .map(|(r, res)| (r.satisfied_count(res), r, res))
1173            .collect();
1174        // Most matched criteria first, then the lowest rule ID.
1175        candidates.sort_by(|a, b| b.0.cmp(&a.0).then(a.1.rule.id.cmp(&b.1.rule.id)));
1176        omitted.near_misses = count(candidates.len().saturating_sub(MAX_NEAR_MISSES));
1177        candidates
1178            .into_iter()
1179            .take(MAX_NEAR_MISSES)
1180            .map(|(n, r, res)| {
1181                let failing = r.failing(res);
1182                let failing_omitted = count(failing.len().saturating_sub(MAX_FAILING));
1183                let keep: BTreeSet<usize> =
1184                    failing.iter().take(MAX_FAILING).map(|(i, _)| *i).collect();
1185                NearMiss {
1186                    rule: r.reference,
1187                    name: clip(&r.rule.name).0,
1188                    stage: r.stage,
1189                    action: r.action(),
1190                    tree: res.tree.into(),
1191                    matched_criteria: count(n),
1192                    total_criteria: count(r.leaves.len()),
1193                    failing: criteria(r, res, subject, privacy, true, Some(&keep)),
1194                    failing_omitted,
1195                }
1196            })
1197            .collect()
1198    }

1228/// The traced criteria of `rule`: all of them, or those in `only`.
1229fn criteria(
1230    rule: &TracedRule<'_>,
1231    res: &RuleResult,
1232    subject: &Subject<'_>,
1233    privacy: &Privacy,
1234    with_observed: bool,
1235    only: Option<&BTreeSet<usize>>,
1236) -> Vec<CriterionTrace> {
1237    let contributing = rule.contributing(res);
1238    rule.leaves
1239        .iter()
1240        .zip(&res.leaves)
1241        .enumerate()
1242        .filter(|(i, _)| only.is_none_or(|o| o.contains(i)))
1243        .map(|(i, (leaf, lr))| {
1244            let c = leaf.criterion;
1245            let mut expected_truncated = c.values.len() > privacy::MAX_VALUES;
1246            let expected = c
1247                .values
1248                .iter()
1249                .take(privacy::MAX_VALUES)
1250                .map(|v| {
1251                    // The policy's values pass the same redactor as the
1252                    // facts: a path criterion can name a user as well.
1253                    let (t, cut) = privacy.text(v);
1254                    expected_truncated |= cut;
1255                    t
1256                })
1257                .collect();
1258            CriterionTrace {
1259                pointer: leaf.pointer.clone(),
1260                field: c.field,
1261                operator: c.operator,
1262                under_not: leaf.negated,
1263                result: lr.effective.into(),
1264                needed: if leaf.negated {
1265                    TriState::NoMatch
1266                } else {
1267                    TriState::Match
1268                },
1269                raw_result: lr.raw.into(),
1270                notes: lr.notes.iter().map(Note::as_str).collect(),
1271                expected,
1272                expected_truncated,
1273                contributing: contributing.get(i).copied().unwrap_or(false),
1274                observed: with_observed.then(|| {
1275                    Observed::new(c.field, observe(c.field, Selector::of(c), subject), privacy)
1276                }),
1277            }
1278        })
1279        .collect()
1280}

1282/// The raw values of a fact, or `None` if it is unknown. `sel` picks the
1283/// part of a signature a signer criterion reads, so the trace shows the
1284/// value that was compared.
1285fn observe(field: Field, sel: Selector, subject: &Subject<'_>) -> Option<Vec<String>> {
1286    use crate::context::ContextFact;
1287    let facts = subject.facts;
1288    if field.is_context() {
1289        return Some(match subject.context.fact(field)? {
1290            ContextFact::Text(t) => vec![t.to_owned()],
1291            ContextFact::Names(a, u) => a.into_iter().chain(u).map(str::to_owned).collect(),
1292            ContextFact::Set(s) => s.to_vec(),
1293            ContextFact::Bool(b) => vec![b.to_string()],
1294            ContextFact::Session(s) => vec![s.as_str().to_owned()],
1295            ContextFact::Number(n) => vec![n.to_string()],
1296            ContextFact::Level(l) => vec![l.as_str().to_owned()],
1297            ContextFact::Ip(ip) => vec![ip.to_string()],
1298            ContextFact::Ips(ips) => ips.iter().map(ToString::to_string).collect(),
1299            ContextFact::Macs(macs) => macs
1300                .iter()
1301                .map(|m| {
1302                    m.iter()
1303                        .map(|b| format!("{b:02x}"))
1304                        .collect::<Vec<_>>()
1305                        .join(":")
1306                })
1307                .collect(),
1308            ContextFact::Weekday(d) => vec![d.as_str().to_owned()],
1309            ContextFact::Minute(m) => vec![format!("{:02}:{:02}", m / 60, m % 60)],
1310            ContextFact::Date(d) => vec![d.to_string()],
1311        });
1312    }
1313    match field {
1314        field if field.is_signer() => {
1315            let signers = facts.signers.as_ref()?;
1316            Some(
1317                signers
1318                    .iter()
1319                    .map(|s| {
1320                        let v = crate::ops::signer::observed(field, sel, s);
1321                        let v = v.as_deref().unwrap_or("(none)");
1322                        if s.status.is_trusted() {
1323                            v.to_owned()
1324                        } else {
1325                            format!("{v} ({})", signer_status(s.status))
1326                        }
1327                    })
1328                    .collect(),
1329            )
1330        }
1331        Field::PathFull | Field::PathDirectory | Field::PathFileName | Field::PathExtension => {
1332            let p = facts.path.as_ref()?;
1333            let pick = |n: &palisade_core::path::NormalizedPath| -> String {
1334                match field {
1335                    Field::PathFull => n.as_str().to_owned(),
1336                    Field::PathDirectory => n.directory().unwrap_or("").to_owned(),
1337                    Field::PathFileName => n.file_name().unwrap_or("").to_owned(),
1338                    _ => n.extension().unwrap_or("").to_owned(),
1339                }
1340            };
1341            Some(std::iter::once(&p.path).chain(&p.alias).map(pick).collect())
1342        }
1343        _ => facts.get(field).map(|v| vec![v.to_owned()]),
1344    }
1345}

1399fn stages(
1400    decision: Option<&Decision>,
1401    decider: Option<&TracedRule<'_>>,
1402    matched: &[&TracedRule<'_>],
1403    state: &DeviceState,
1404    durations: &BTreeMap<RuleStage, u64>,
1405) -> Vec<StageTrace> {
1406    let n = |s: RuleStage| count(matched.iter().filter(|r| r.stage == s).count());
1407    let decided = match (decision, decider) {
1408        (None, _) => None,
1409        (Some(_), Some(r)) => Some(match r.stage {
1410            RuleStage::MandatoryDeny => StageName::MandatoryDeny,
1411            RuleStage::Exclusion => StageName::Exclusion,
1412            _ => StageName::Policy,
1413        }),
1414        // A pinned Palisade image is allowed right after stage 1 (#39).
1415        (Some(d), None) if d.reason_code() == crate::decision::ReasonCode::SelfAllow => {
1416            Some(StageName::Exclusion)
1417        }
1418        (Some(_), None) => Some(StageName::Unknown),
1419    };
1420    let order = [
1421        StageName::MandatoryDeny,
1422        StageName::Exclusion,
1423        StageName::SessionOverride,
1424        StageName::Policy,
1425        StageName::Attribute,
1426        StageName::Unknown,
1427    ];
1428    let position = decided.and_then(|d| order.iter().position(|s| *s == d));
1429    order
1430        .iter()
1431        .enumerate()
1432        .map(|(i, &name)| {
1433            let outcome = match position {
1434                None => StageOutcome::Skipped,
1435                Some(p) if i == p => StageOutcome::Decided,
1436                // Attribute rules do not exist yet (#33), and a session
1437                // stage only runs with a session.
1438                Some(p) if i < p => match name {
1439                    StageName::Attribute => StageOutcome::Skipped,
1440                    // Lockdown suspends every session (#36).
1441                    StageName::SessionOverride if state.session.is_none() || state.lockdown => {
1442                        StageOutcome::Skipped
1443                    }
1444                    _ => StageOutcome::Passed,
1445                },
1446                Some(_) => StageOutcome::Skipped,
1447            };
1448            let (candidates, duration_ns) = match name {
1449                StageName::MandatoryDeny => (
1450                    n(RuleStage::MandatoryDeny),
1451                    durations.get(&RuleStage::MandatoryDeny).copied(),
1452                ),
1453                StageName::Exclusion => (
1454                    n(RuleStage::Exclusion),
1455                    durations.get(&RuleStage::Exclusion).copied(),
1456                ),
1457                StageName::Policy => (
1458                    n(RuleStage::Policy),
1459                    durations.get(&RuleStage::Policy).copied(),
1460                ),
1461                _ => (0, None),
1462            };
1463            StageTrace {
1464                name,
1465                outcome,
1466                candidates,
1467                duration_ns,
1468            }
1469        })
1470        .collect()
1471}

1473/// The smallest allow rule that identifies the subject, from the strongest
1474/// identity the privacy settings let the trace reveal.
1475fn suggest(
1476    subject: &Subject<'_>,
1477    privacy: &Privacy,
1478    matched: &[&TracedRule<'_>],
1479    extras: &ObjectExtras,
1480) -> Option<RuleDraft> {
1481    use crate::model::{Criterion, Match};
1482    let facts = subject.facts;
1483    // A criterion is offered only if validation would accept it: the value
1484    // is revealed and unchanged by redaction, not empty, has no control
1485    // character, is within the size limit, and the criterion compiles. If
1486    // not, the next identity is tried.
1487    let clean = |field: Field, v: &str| {
1488        privacy.reveals(field)
1489            && privacy.redactor.redact(v) == v
1490            && !v.is_empty()
1491            && !v.chars().any(char::is_control)
1492            && v.len() <= crate::model::limits::MAX_VALUE_BYTES
1493    };
1494    let criterion = |field: Field, operator: Operator, v: &str| -> Option<Match> {
1495        let c = Criterion::new(field, operator, [v]);
1496        (clean(field, v) && crate::ops::CompiledCriterion::compile(&c).is_ok())
1497            .then_some(Match::Criterion(c))
1498    };
1499    let hash = [
1500        Field::HashSha256,
1501        Field::HashAuthenticodeSha256,
1502        Field::HashSha384,
1503        Field::HashSha512,
1504    ]
1505    .into_iter()
1506    .find_map(|f| {
1507        // The fact must be what the matcher accepts: bare hex digits of the
1508        // field's length (a `0x` prefix makes it malformed, so no criterion
1509        // could match it). The criterion is written in lowercase.
1510        let raw = facts.get(f)?;
1511        match f.class() {
1512            crate::model::FieldClass::Hash { hex_len }
1513                if raw.len() == hex_len && raw.bytes().all(|b| b.is_ascii_hexdigit()) => {}
1514            _ => return None,
1515        }
1516        if !clean(f, raw) {
1517            return None;
1518        }
1519        criterion(f, Operator::Equals, &raw.to_ascii_lowercase())
1520    });
1521    // A signer is only offered if no check of its signature was relaxed.
1522    let relaxed: BTreeSet<usize> = extras
1523        .signature_relaxations
1524        .iter()
1525        .map(|r| r.signature)
1526        .collect();
1527    let publisher = facts.signers.as_ref().and_then(|s| {
1528        s.iter()
1529            .enumerate()
1530            .filter(|(i, s)| s.status.is_trusted() && !relaxed.contains(i))
1531            .find_map(|(_, s)| {
1532                criterion(
1533                    Field::SignerPublisher,
1534                    Operator::Equals,
1535                    s.publisher.as_deref()?,
1536                )
1537            })
1538    });
1539    let product = facts
1540        .get(Field::MetaProductName)
1541        .and_then(|v| criterion(Field::MetaProductName, Operator::Equals, v));
1542    // The recorded version must itself be revealed and unredacted; the
1543    // draft uses its numeric part.
1544    let version = facts
1545        .get(Field::MetaFileVersion)
1546        .filter(|v| clean(Field::MetaFileVersion, v))
1547        .and_then(crate::ops::Version::parse_fact)
1548        .and_then(|(v, _)| criterion(Field::MetaFileVersion, Operator::VersionGe, &v.to_string()));
1549    let (identity, tree) = if let Some(h) = hash {
1550        (DraftIdentity::Hash, h)
1551    } else if let Some(p) = publisher {
1552        match (product, version) {
1553            (Some(pr), Some(v)) => (
1554                DraftIdentity::PublisherProductVersion,
1555                Match::All(vec![p, pr, v]),
1556            ),
1557            (Some(pr), None) => (DraftIdentity::PublisherProduct, Match::All(vec![p, pr])),
1558            _ => (DraftIdentity::Publisher, p),
1559        }
1560    } else {
1561        let p = facts.path.as_ref().filter(|p| {
1562            p.user_writable == Some(false) && !p.short_name_unresolved && p.path.is_resolved()
1563        })?;
1564        (
1565            DraftIdentity::Path,
1566            criterion(Field::PathFull, Operator::Equals, p.path.as_str())?,
1567        )
1568    };
1569    let mut caveats = Vec::new();
1570    if matched.iter().any(|r| r.stage == RuleStage::MandatoryDeny) {
1571        caveats.push("mandatory_deny_matched");
1572    }
1573    if matched
1574        .iter()
1575        .any(|r| r.stage == RuleStage::Policy && r.action() == ActionKind::Deny)
1576    {
1577        caveats.push("deny_rule_matched");
1578    }
1579    Some(RuleDraft {
1580        action: ActionKind::Allow,
1581        object_types: vec![subject.object_type],
1582        identity,
1583        r#match: tree,
1584        caveats,
1585    })
1586}

1588/// The decision's JSON with its lists cut ([`Trace::decision`]): the
1589/// decision's own fields in their stable order, then the counts.
1590fn bounded_decision<S: serde::Serializer>(d: &Option<Decision>, s: S) -> Result<S::Ok, S::Error> {
1591    #[derive(Serialize)]
1592    struct Bounded {
1593        #[serde(flatten)]
1594        decision: Decision,
1595        overridden_omitted: usize,
1596        annotations_omitted: usize,
1597        trust_sources_omitted: usize,
1598    }
1599    match d {
1600        None => s.serialize_none(),
1601        Some(d) => {
1602            let (decision, [o, a, t]) = d.with_lists_cut(limits::MAX_DECISION_LIST);
1603            Bounded {
1604                decision,
1605                overridden_omitted: o,
1606                annotations_omitted: a,
1607                trust_sources_omitted: t,
1608            }
1609            .serialize(s)
1610        }
1611    }
1612}
