1//! The mode resolution steps of one decision (#36), for the trace.
2//!
3//! [`crate::mode::resolve`] gives the effective mode. These steps restate
4//! its order, one step per rule it applies, so a reader can see which
5//! setting set the mode. The last step is the result of `resolve` itself,
6//! and the tests check that the steps lead to it.
7

12use crate::evaluate::Subject;
13use crate::mode::{self, Connectivity, DeviceState, SessionKind};

15    Action, ControlLevel, DriverBootPolicy, EnforcementOverride, Mode, ObjectType, Policy, Rule,
16    RuleSet,

19/// What one resolution step looked at.
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
21#[serde(rename_all = "snake_case")]
22pub enum ModeStepKind {
23    /// Emergency lockdown forces `enforce`.
24    Lockdown,
25    /// The device's base mode.
26    DeviceMode,
27    /// A mandatory deny always enforces.
28    MandatoryDeny,
29    /// `mode_disconnected` while the device is disconnected.
30    Disconnected,
31    /// `mode_by_type` for the object's type.
32    ObjectType,
33    /// The class's control level caps the mode.
34    ControlLevel,
35    /// The deciding rule set's `mode_override`.
36    RuleSetOverride,
37    /// The deciding rule's `enforcement_override`.
38    RuleOverride,
39    /// An active override session.
40    Session,
41    /// A driver loaded during early boot, decided by the unknown-application
42    /// policy, follows `driver_boot_policy` (#39).
43    BootPhase,
44    /// The effective mode.
45    Effective,
46    /// What the mode did to the outcome.
47    Applied,
48}
49
50/// One step of mode resolution.
51#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
52pub struct ModeStep {
53    /// The step.
54    pub kind: ModeStepKind,
55    /// Whether it changed the mode (or, for `lockdown`, `mandatory_deny`
56    /// and `effective`, fixed it).
57    pub applied: bool,
58    /// The mode after the step.
59    pub mode: Mode,
60    /// What the step read, such as `script.powershell: audit`.
61    pub detail: String,
62}
63
64fn step(kind: ModeStepKind, applied: bool, mode: Mode, detail: impl Into<String>) -> ModeStep {
65    ModeStep {
66        kind,
67        applied,
68        mode,
69        detail: detail.into(),
70    }
71}
72
73fn name(m: Mode) -> &'static str {
74    match m {
75        Mode::Disabled => "disabled",
76        Mode::Monitor => "monitor",
77        Mode::Audit => "audit",
78        Mode::Learning => "learning",
79        Mode::Enforce => "enforce",
80    }
81}

83/// The steps for a disabled device, which decides nothing.
84pub(crate) fn disabled_steps(state: &DeviceState) -> Vec<ModeStep> {
85    vec![
86        step(
87            ModeStepKind::DeviceMode,
88            true,
89            Mode::Disabled,
90            format!("device mode {}", name(state.mode)),
91        ),
92        step(
93            ModeStepKind::Effective,
94            true,
95            Mode::Disabled,
96            "the engine is not consulted",
97        ),
98    ]
99}

101/// The steps that led to `decision`'s mode, for an object of type `t`
102/// decided by `rule` (or the implicit deny). Each step's mode is the mode
103/// after it, computed with the functions of [`crate::mode`]; the
104/// `effective` step is [`mode::resolve`] itself, and the tests check that
105/// the step before it agrees.
106pub(crate) fn steps(
107    policy: &Policy,
108    subject: &Subject<'_>,
109    rule: Option<(&RuleSet, &Rule)>,
110    state: &DeviceState,
111    decision: &Decision,
112) -> Vec<ModeStep> {
113    let t: ObjectType = subject.object_type;
114    let mut out = Vec::new();
115    let mut effective = mode::resolve(policy, t, rule, state);
116    // As the evaluator does: an early-boot driver decided at stage 6 under
117    // `driver_boot_policy: audit` is only reported, except in lockdown.
118    let boot_audit = decision.unhandled()
119        && t == ObjectType::Driver
120        && subject.context.device.boot_phase == Some(BootPhase::Early)
121        && policy.unknown.driver_boot_policy == DriverBootPolicy::Audit
122        && !state.lockdown;
123    if state.lockdown {
124        out.push(step(
125            ModeStepKind::Lockdown,
126            true,
127            Mode::Enforce,
128            "emergency lockdown",
129        ));
130    } else if rule.is_some_and(|(set, _)| set.mandatory) {
131        out.push(step(
132            ModeStepKind::DeviceMode,
133            false,
134            state.mode,
135            format!("device mode {}", name(state.mode)),
136        ));
137        out.push(step(
138            ModeStepKind::MandatoryDeny,
139            true,
140            Mode::Enforce,
141            "a mandatory deny enforces in every mode",
142        ));
143    } else {
144        out.push(step(
145            ModeStepKind::DeviceMode,
146            false,
147            state.mode,
148            format!("device mode {}", name(state.mode)),
149        ));
150        let device = mode::device_mode(policy, state);
151        if state.connectivity == Connectivity::Disconnected {
152            out.push(step(
153                ModeStepKind::Disconnected,
154                device != state.mode,
155                device,
156                match policy.mode_options.mode_disconnected {
157                    Some(m) => format!("disconnected; mode_disconnected {}", name(m)),
158                    None => "disconnected; no mode_disconnected".to_owned(),
159                },
160            ));
161        }
162        let object = mode::object_mode(policy, state, t);
163        let by_type = policy.mode_options.mode_by_type.get(&t).copied();
164        let level = mode::control_level(policy, t);
165        if by_type.is_some() || level != ControlLevel::Enforce {
166            let mut detail = Vec::new();
167            if let Some(m) = by_type {
168                detail.push(format!("mode_by_type {t}: {}", name(m)));
169            }
170            match level {
171                ControlLevel::Enforce => {}
172                ControlLevel::Audit => detail.push("control level audit".to_owned()),
173                ControlLevel::Off => detail.push("control level off".to_owned()),
174            }
175            out.push(step(
176                if by_type.is_some() {
177                    ModeStepKind::ObjectType
178                } else {
179                    ModeStepKind::ControlLevel
180                },
181                object != device,
182                object,
183                detail.join("; "),
184            ));
185        }
186        let mut current = object;
187        let mut always = false;
188        if let Some((set, r)) = rule {
189            if let Some(o) = set.mode_override {
190                // `disabled` below a device is read as `monitor`.
191                let m = if o == Mode::Disabled {
192                    Mode::Monitor
193                } else {
194                    o
195                };
196                out.push(step(
197                    ModeStepKind::RuleSetOverride,
198                    m != current,
199                    m,
200                    format!("rule set mode_override {}", name(o)),
201                ));
202                current = m;
203            }
204            if matches!(r.action, Action::Deny(_)) {
205                match r.enforcement_override {
206                    EnforcementOverride::Inherit => {}
207                    EnforcementOverride::AlwaysEnforce => {
208                        always = true;
209                        out.push(step(
210                            ModeStepKind::RuleOverride,
211                            current != Mode::Enforce,
212                            Mode::Enforce,
213                            "enforcement_override always_enforce",
214                        ));
215                        current = Mode::Enforce;
216                    }
217                    EnforcementOverride::NeverEnforce => {
218                        let m = mode::capped(current, Mode::Audit);
219                        out.push(step(
220                            ModeStepKind::RuleOverride,
221                            m != current,
222                            m,
223                            "enforcement_override never_enforce",
224                        ));
225                        current = m;
226                    }
227                }
228            }
229        }
230        if let Some(s) = state.session {
231            let full = s == SessionKind::FullBypass;
232            let kind = if full { "full_bypass" } else { "audit" };
233            if always && !full {
234                out.push(step(
235                    ModeStepKind::Session,
236                    false,
237                    current,
238                    format!("override session {kind}; always_enforce is not relaxed"),
239                ));
240            } else {
241                let m = mode::capped(current, Mode::Audit);
242                out.push(step(
243                    ModeStepKind::Session,
244                    m != current,
245                    m,
246                    format!("override session {kind}"),
247                ));
248            }
249        }
250    }
251    if boot_audit {
252        let m = mode::capped(effective.mode, Mode::Audit);
253        out.push(step(
254            ModeStepKind::BootPhase,
255            m != effective.mode,
256            m,
257            "early boot; driver_boot_policy audit",
258        ));
259        effective.mode = m;
260        effective.by_session = false;
261    }
262    out.push(step(
263        ModeStepKind::Effective,
264        true,
265        effective.mode,
266        if effective.by_session {
267            "lowered by the override session"
268        } else {
269            ""
270        },
271    ));
272    out.push(step(
273        ModeStepKind::Applied,
274        decision.outcome() != decision.would_have_been(),
275        decision.mode(),
276        applied_detail(decision),
277    ));
278    out
279}

281fn applied_detail(d: &Decision) -> &'static str {
282    let enforce = match d.would_have_been() {
283        Outcome::PendingUser => "would have asked the user",
284        _ => "would have blocked",
285    };
286    match (d.would_have_been(), d.outcome(), d.reason_code()) {
287        (Outcome::Blocked | Outcome::PendingUser, Outcome::AllowedAudit, _) => {
288            if enforce == "would have asked the user" {
289                "would have asked the user; runs because the mode does not enforce"
290            } else {
291                "would have blocked; runs because the mode does not enforce"
292            }
293        }
294        (Outcome::Blocked | Outcome::PendingUser, Outcome::AllowedOverride, _) => {
295            if enforce == "would have asked the user" {
296                "would have asked the user; runs because of the override session"
297            } else {
298                "would have blocked; runs because of the override session"
299            }
300        }
301        (Outcome::Blocked, Outcome::Blocked, ReasonCode::MandatoryDeny) => {
302            "blocked; a mandatory deny is never relaxed"
303        }
304        (Outcome::Blocked, Outcome::Blocked, ReasonCode::ImplicitDenyUnknown) => {
305            "blocked; no rule allows it, and the mode enforces"
306        }
307        (Outcome::Blocked, Outcome::Blocked, ReasonCode::RestrictUnavailable) => {
308            "blocked; it could not be contained, and the mode enforces"
309        }
310        (Outcome::PendingUser, Outcome::PendingUser, _) => {
311            "waits for the user; no rule decides it, and the mode enforces"
312        }
313        (Outcome::Blocked, Outcome::Blocked, _) => "blocked; the deciding rule enforces",
314        _ => "the outcome is the enforce-mode outcome",
315    }
