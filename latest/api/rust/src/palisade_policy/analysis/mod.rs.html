1//! The policy assistant: static analysis of a policy (#40).
2//!
3//! [`analyse`] runs on every save, on bundle compile ([`publish_gate`]) and
4//! on demand (the CLI's `pal policy lint`, #15). Each [`Finding`] has a
5//! stable [`FindingCode`], a severity, the rules it is about, a message and
6//! an optional suggested fix. Findings are sorted, so the output is a
7//! deterministic function of the policy, the facts and the time.
8//!
9//! A policy can raise a finding's severity (`policy.governance.severity`),
10//! never lower it below the code's default: validation rejects that, and
11//! the assistant reports the higher of the two. `error` findings block
12//! publishing unless a tenant administrator overrides them with a reason
13//! ([`publish_gate`]).
14//!
15//! # Conflicts
16//!
17//! `shadowed`, `contradiction` and `ineffective_vs_mandatory` need a proof
18//! that one rule matches whenever another does ([`subsume::covers`]). The
19//! proof is conservative: a finding is only made when it is certain, so a
20//! rule reported as shadowed or ineffective truly never decides (a property
21//! test checks this against the evaluator). Rules are compared only when
22//! they share an identity value (an exact hash, publisher or path), which
23//! keeps the analysis near linear on large policies.
24//!
25//! - `ineffective_vs_mandatory`: an allow that a mandatory deny always
26//!   blocks first (stage 1).
27//! - `shadowed`: a rule that another rule of the same verdict always
28//!   outranks, under the policy's model, so it never decides. Only proven
29//!   for `deny_wins` among denies and for `first_match`, where the outcome
30//!   order is fixed. Other cases are left to the decision trace (#41).
31//! - `contradiction`: an allow and a deny where one covers the other, so
32//!   for some objects both match, and the precedence model alone decides.
33//!   Under `deny_wins`, an allow covered by a deny never decides, which is
34//!   reported as `shadowed` instead.

54/// The stable code of a finding. The `snake_case` name is part of the API.
55#[derive(
56    Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, serde::Deserialize,
57)]
58#[serde(rename_all = "snake_case")]
59pub enum FindingCode {
60    /// An allow on a weak identity: a file name, metadata, an extension,
61    /// SHA-1 or MD5 only, or a path whose ACL safety is not proven (#31).
62    WeakIdentity,
63    /// A path allow on a location the writable-path scanner flagged.
64    UserWritablePathAllow,
65    /// A publisher-only allow for a restricted publisher (#31).
66    RestrictedPublisherBroad,
67    /// A very broad wildcard or regular expression.
68    BroadWildcard,
69    /// A rule that never decides, because another always outranks it.
70    Shadowed,
71    /// An allow and a deny that both match some objects.
72    Contradiction,
73    /// An allow that a mandatory deny always blocks first.
74    IneffectiveVsMandatory,
75    /// Two rules with the same normalised match, action and scope.
76    DuplicateRule,
77    /// A principal in both the positive and the negative test of a rule,
78    /// or repeated in one test.
79    DuplicatePrincipal,
80    /// An expired rule still in the policy.
81    ExpiredRulePresent,
82    /// A rule that expires soon.
83    ExpiringSoon,
84    /// A rule whose review date has passed.
85    ReviewOverdue,
86    /// A rule with no hits for `stale_after_days` while devices in scope
87    /// were online.
88    StaleRule,
89    /// A publisher or thumbprint rule whose certificate is revoked or
90    /// expired.
91    RevokedOrExpiredSigner,
92    /// An allow covering a version range that a mandatory or curated
93    /// blocklist marks vulnerable.
94    VulnerableVersionAllowed,
95    /// A rule that needs a justification has none (governance mode).
96    MissingJustification,
97    /// An exception without an expiry, approver or risk rating, or with an
98    /// expiry more than a year away (governance mode).
99    ExceptionWithoutExpiry,
100    /// More inline hash values than the engine keeps in a list (#42).
101    TooManyHashRules,
102    /// A regular expression near the size limit.
103    RegexCost,
104    /// `approved_at`, `created_at` or `last_reviewed_at` is in the future.
105    FutureDatedMetadata,
106    /// A rule that requires a strong file hash also requires another
107    /// identity criterion (#31, #328). The hash already names one file;
108    /// the rest only narrows, and stops the rule matching that file if the
109    /// other fact changes or is unknown.
110    HashWithOtherCriteria,
111}

307/// One rule, with where it is.
308#[derive(Clone, Copy)]
309struct Entry<'d> {
310    set: &'d RuleSet,
311    rule: &'d Rule,
312}
313
314impl Entry<'_> {
315    fn r(&self) -> RuleRef {
316        RuleRef {
317            rule_set_id: self.set.id,
318            rule_id: self.rule.id,
319        }
320    }
321
322    fn verdict(&self) -> Option<Verdict> {
323        // An attribute rule is stage 5, not one the model decides (#329).
324        Verdict::of(&self.rule.action).filter(|_| !self.rule.attribute)
325    }
326
327    /// The scopes on the way to the rule (#38): its rule set's assignment
328    /// scopes, then its own. The policy's scope covers every rule alike.
329    fn scopes<'a>(&'a self, doc: &'a PolicyDocument) -> Vec<&'a crate::scope::Scope> {
330        doc.policy
331            .rule_set_scopes
332            .iter()
333            .filter(|a| a.rule_set == self.set.id)
334            .flat_map(|a| &a.scopes)
335            .chain(self.rule.scope.as_deref())
336            .collect()
337    }
338
339    fn applies_to_all_types_of(&self, other: &Self) -> bool {
340        self.rule.object_types.is_empty()
341            || (!other.rule.object_types.is_empty()
342                && other
343                    .rule
344                    .object_types
345                    .iter()
346                    .all(|t| self.rule.object_types.contains(t)))
347    }
348}

356/// [`analyse`], with options.
357#[must_use]
358pub fn analyse_with(
359    doc: &PolicyDocument,
360    facts: &Facts,
361    now: Timestamp,
362    options: &Options,
363) -> Vec<Finding> {
364    let mut a = Analyser {
365        doc,
366        facts,
367        now,
368        exempt: crate::lifecycle::managed_baseline(doc),
369        clock: RuleClock::at(now),
370        options,
371        out: Vec::new(),
372    };
373    let entries: Vec<Entry<'_>> = crate::lifecycle::referenced(doc)
374        .flat_map(|set| set.rules.iter().map(move |rule| (set, rule)))
375        .map(|(set, rule)| Entry { set, rule })
376        .collect();
377    for e in &entries {
378        a.rule_checks(e);
379    }
380    let active: Vec<Entry<'_>> = entries
381        .iter()
382        .copied()
383        .filter(|e| activity(e.rule, &a.clock) == Activity::Active)
384        .collect();
385    a.duplicates(&active);
386    a.conflicts(&active);
387    a.hash_counts();
388    a.finish()
389}

391struct Analyser<'a> {
392    doc: &'a PolicyDocument,
393    facts: &'a Facts,
394    now: Timestamp,
395    /// The managed `system_baseline`, if the document holds the one the
396    /// engine built.
397    exempt: Option<RuleSetId>,
398    clock: RuleClock,
399    options: &'a Options,
400    out: Vec<Finding>,
401}
402
403impl Analyser<'_> {
404    fn push(&mut self, code: FindingCode, refs: Vec<RuleRef>, message: String, fix: Option<Fix>) {
405        let severity = self
406            .doc
407            .policy
408            .governance
409            .severity
410            .get(code.as_str())
411            .copied()
412            .map_or_else(
413                || code.default_severity(),
414                |s| s.max(code.default_severity()),
415            );
416        self.out.push(Finding {
417            code,
418            severity,
419            rule_refs: refs,
420            message,
421            fix,
422        });
423    }
424
425    fn finish(mut self) -> Vec<Finding> {
426        self.out.sort_by(|a, b| {
427            (a.rule_refs.first(), a.code, &a.rule_refs, &a.message).cmp(&(
428                b.rule_refs.first(),
429                b.code,
430                &b.rule_refs,
431                &b.message,
432            ))
433        });
434        self.out.dedup();
435        self.out
436    }
437
438    fn governance(&self) -> bool {
439        crate::validate::governance_mode(self.doc)
440    }
441
442    /// Whether the rule's set is exempt from governance and reviews: only
443    /// the managed `system_baseline` the engine built.
444    fn exempt(&self, set: &RuleSet) -> bool {
445        self.exempt == Some(set.id)
446    }
447
448    /// The checks of one rule on its own.
449    fn rule_checks(&mut self, e: &Entry<'_>) {
450        let rule = e.rule;
451        let kind = rule.action.kind();
452        let me = vec![e.r()];
453        let g = &self.doc.policy.governance;
454
455        // Lifecycle.
456        match activity(rule, &self.clock) {
457            // An expired `convert_to_audit` rule still records matches:
458            // removing it would lose that audit trail, so it is reported
459            // without a fix.
460            Activity::Expired if rule.on_expire == crate::model::OnExpire::ConvertToAudit => self
461                .push(
462                    FindingCode::ExpiredRulePresent,
463                    me.clone(),
464                    format!(
465                        "expired at {}; it no longer decides, but still audits its matches (`on_expire: convert_to_audit`)",
466                        rule.expires_at.map_or_else(String::new, |t| t.to_string())
467                    ),
468                    None,
469                ),
470            Activity::Expired => self.push(
471                FindingCode::ExpiredRulePresent,
472                me.clone(),
473                format!(
474                    "expired at {}; it no longer decides",
475                    rule.expires_at.map_or_else(String::new, |t| t.to_string())
476                ),
477                Some(Fix::RemoveRule),
478            ),
479            Activity::Active => {
480                if let Some(e_at) = rule.expires_at
481                    && self
482                        .now
483                        .plus_days(i64::from(g.expiring_soon_days))
484                        .is_some_and(|soon| e_at <= soon)
485                    {
486                        self.push(
487                            FindingCode::ExpiringSoon,
488                            me.clone(),
489                            format!("expires at {e_at}"),
490                            None,
491                        );
492                    }
493            }
494            Activity::Disabled | Activity::NotYetActive => {}
495        }
496        if rule.enabled && !self.exempt(e.set) {
497            if let Some(due) = review_by(&rule.metadata, Some(kind))
498                && due < self.now.date()
499            {
500                self.push(
501                    FindingCode::ReviewOverdue,
502                    me.clone(),
503                    format!("review was due on {due}"),
504                    None,
505                );
506            }
507            // A trust source grants rather than decides, so its use is not
508            // counted as a hit (#9 reports grants).
509            let usage = self
510                .facts
511                .usage
512                .get(&rule.id)
513                .filter(|_| kind != ActionKind::TrustSource);
514            if let Some(u) = usage {
515                let window = self.now.plus_days(-i64::from(g.stale_after_days));
516                let old = window.is_some_and(|w| {
517                    u.last_matched_at.is_none_or(|t| t < w)
518                        && rule.metadata.created_at.is_none_or(|c| c < w)
519                });
520                if old && u.devices_online {
521                    self.push(
522                        FindingCode::StaleRule,
523                        me.clone(),
524                        format!(
525                            "no match in {} days while devices in scope were online",
526                            g.stale_after_days
527                        ),
528                        Some(Fix::ConvertToAudit),
529                    );
530                }
531            }
532        }
533
534        // Identity.
535        if kind.permits() {
536            match weakest_identity(&rule.match_) {
537                Guaranteed::None | Guaranteed::WeakHash | Guaranteed::Weak => self.push(
538                    FindingCode::WeakIdentity,
539                    me.clone(),
540                    "this allow identifies objects only by a file name, metadata, an extension, or SHA-1 or MD5; anyone can produce such a file".to_owned(),
541                    Some(Fix::Narrow { hint: "add a SHA-256, or a publisher with a product name" }),
542                ),
543                Guaranteed::Path => {
544                    let flagged = self.flagged_path(&rule.match_);
545                    if let Some(dir) = flagged {
546                        self.push(
547                            FindingCode::UserWritablePathAllow,
548                            me.clone(),
549                            format!("this allow trusts the path `{dir}`, which the writable-path scanner flagged as writable by non-administrators"),
550                            Some(Fix::Narrow { hint: "replace the path with a hash or publisher identity" }),
551                        );
552                    } else {
553                        self.push(
554                            FindingCode::WeakIdentity,
555                            me.clone(),
556                            "this allow identifies objects only by their path; it is only as safe as the path's ACL".to_owned(),
557                            Some(Fix::Narrow { hint: "add a hash or a publisher" }),
558                        );
559                    }
560                }
561                Guaranteed::Publisher => {
562                    if let Some(p) = self.restricted_publisher_only(&rule.match_) {
563                        self.push(
564                            FindingCode::RestrictedPublisherBroad,
565                            me.clone(),
566                            format!("this allow trusts everything `{p}` signs; that publisher's binaries include tools that run arbitrary code"),
567                            Some(Fix::Narrow { hint: "add a product or file-name constraint" }),
568                        );
569                    }
570                }
571                Guaranteed::Strong => {}
572            }
573        }
574        if let Some(other) = hash_with_other(&rule.match_) {
575            self.push(
576                FindingCode::HashWithOtherCriteria,
577                me.clone(),
578                format!("this rule already names one file by its hash; `{other}` only narrows it, and stops the rule matching that file if it changes or is unknown"),
579                None,
580            );
581        }
582        self.signers(e);
583        self.patterns(e);
584        self.principals(e);
585
586        // Metadata dated in the future: a creation, approval or review that
587        // has not happened yet, which could stretch a review or an
588        // exception's year.
589        if rule.enabled {
590            let m = &rule.metadata;
591            for (field, at) in [
592                ("approved_at", m.approved_at),
593                ("created_at", m.created_at),
594                ("last_reviewed_at", m.last_reviewed_at),
595            ] {
596                if let Some(at) = at.filter(|t| *t > self.now) {
597                    self.push(
598                        FindingCode::FutureDatedMetadata,
599                        me.clone(),
600                        format!("`{field}` ({at}) is in the future"),
601                        Some(Fix::AddMetadata {
602                            fields: vec![field],
603                        }),
604                    );
605                }
606            }
607        }
608
609        // Governance, as validation applies it: not to the managed
610        // `system_baseline` the engine built.
611        if self.governance() && rule.enabled && !self.exempt(e.set) {
612            let exception = is_exception(rule);
613            let needs_justification = kind.permits()
614                || matches!(kind, ActionKind::TrustSource | ActionKind::Ignore)
615                || exception;
616            if needs_justification
617                && rule
618                    .metadata
619                    .justification
620                    .as_deref()
621                    .is_none_or(|j| j.trim().is_empty())
622            {
623                self.push(
624                    FindingCode::MissingJustification,
625                    me.clone(),
626                    format!("governance mode needs a justification for this `{kind}` rule"),
627                    Some(Fix::AddMetadata {
628                        fields: vec!["justification"],
629                    }),
630                );
631            }
632            if exception {
633                let mut missing = Vec::new();
634                let start = crate::lifecycle::exception_start(&rule.metadata);
635                match (rule.expires_at, start) {
636                    (None, None) => missing.extend(["expires_at", "approved_at"]),
637                    (None, Some(_)) => missing.push("expires_at"),
638                    (Some(_), None) => missing.push("approved_at"),
639                    (Some(t), Some(start)) => {
640                        // From no later than now, so an approval dated in
641                        // the future cannot stretch the year.
642                        let start = start.min(self.now);
643                        if start.plus_days(MAX_EXCEPTION_DAYS).is_none_or(|l| t > l) {
644                            missing.push("expires_at (at most 365 days)");
645                        }
646                    }
647                }
648                if rule.metadata.approved_by.is_none() {
649                    missing.push("approved_by");
650                }
651                if rule.metadata.risk_rating.is_none() {
652                    missing.push("risk_rating");
653                }
654                if !missing.is_empty() {
655                    self.push(
656                        FindingCode::ExceptionWithoutExpiry,
657                        me.clone(),
658                        format!(
659                            "this rule is an exception; governance mode needs {}",
660                            missing.join(", ")
661                        ),
662                        Some(Fix::AddMetadata { fields: missing }),
663                    );
664                }
665            }

715    /// The restricted publisher that some way the tree can match trusts
716    /// with nothing else narrowing it: a branch whose positive criteria
717    /// name a restricted publisher and test no product, file name, full
718    /// path or hash. A test under `not` excludes files and narrows nothing.
719    /// A tree too wide to expand is reported if it names a restricted
720    /// publisher at all.
721    fn restricted_publisher_only(&self, tree: &Match) -> Option<String> {
722        if self.facts.restricted_publishers.is_empty() {
723            return None;
724        }
725        // Each restricted publisher as a validly signed object's facts, so
726        // a publisher criterion is tested as the evaluator tests it: with
727        // its operator and options (`starts_with "Example"` names
728        // `Example Ltd`), not by comparing values. A result the bare
729        // signer cannot settle (a signer option it has no fact for) could
730        // match, so it counts.
731        let objects: Vec<(&String, crate::facts::ObjectFacts)> = self
732            .facts
733            .restricted_publishers
734            .iter()
735            .map(|p| {
736                let mut signer = crate::facts::SignerFact::new(
737                    crate::facts::SignerStatus::Valid,
738                    crate::facts::SignatureSource::Embedded,
739                );
740                signer.publisher = Some(p.clone());
741                (
742                    p,
743                    crate::facts::ObjectFacts::new().with_signers(vec![signer]),
744                )
745            })
746            .collect();
747        let variables = crate::facts::PathVariables::new();
748        let context = crate::context::EvaluationContext::new();
749        let ctx = crate::ops::EvalContext {
750            action: ActionKind::Allow,
751            variables: &variables,
752            context: &context,
753        };
754        let named = |c: &Criterion| {
755            if c.field != Field::SignerPublisher || !c.operator.is_positive() {
756                return None;
757            }
758            let compiled = crate::ops::CompiledCriterion::compile(c).ok()?;
759            objects
760                .iter()
761                .find(|(_, facts)| compiled.eval(facts, &ctx).result != crate::ops::Tri::NoMatch)
762                .map(|(p, _)| (*p).clone())
763        };
764        let narrows = |c: &Criterion| {
765            matches!(
766                c.field,
767                Field::MetaProductName
768                    | Field::MetaOriginalFilename
769                    | Field::MetaInternalName
770                    | Field::PathFileName
771                    | Field::PathFull
772            ) || matches!(c.field.class(), FieldClass::Hash { .. })
773                || c.field == Field::SignerLeafThumbprint
774        };
775        match identity::branches(tree) {
776            Some(branches) => branches.iter().find_map(|b| {
777                if b.iter().any(|c| narrows(c)) {
778                    None
779                } else {
780                    b.iter().find_map(|c| named(c))
781                }
782            }),
783            None => identity::literals(tree)
784                .into_iter()
785                .filter(|(_, positive)| *positive)
786                .find_map(|(c, _)| named(c)),
787        }
788    }

790    fn signers(&mut self, e: &Entry<'_>) {
791        if self.facts.certificates.is_empty() || !e.rule.action.kind().needs_identity() {
792            return;
793        }
794        let certs: BTreeMap<String, CertificateState> = self
795            .facts
796            .certificates
797            .iter()
798            .map(|(k, v)| {
799                (
800                    crate::ops::normalize_hex(k)
801                        .filter(|h| h.len() == 64)
802                        .unwrap_or_else(|| crate::ops::fold(k, false)),
803                    *v,
804                )
805            })
806            .collect();
807        // Only a signer the rule trusts: a positive test outside `not`.
808        for (c, positive) in identity::literals(&e.rule.match_) {
809            if !positive || !c.operator.is_positive() {
810                continue;
811            }
812            for v in &c.values {
813                let key = match c.field {
814                    Field::SignerPublisher => crate::ops::fold(v, false),
815                    Field::SignerLeafThumbprint => crate::ops::normalize_hex(v).unwrap_or_default(),
816                    _ => continue,
817                };
818                if let Some(state) = certs.get(&key) {
819                    let what = match state {
820                        CertificateState::Revoked => "revoked",
821                        CertificateState::Expired => "expired",
822                    };
823                    self.push(
824                        FindingCode::RevokedOrExpiredSigner,
825                        vec![e.r()],
826                        format!("this rule trusts `{v}`, whose certificate is {what}; files it signs no longer match a trusted signature"),
827                        Some(Fix::Narrow { hint: "replace the signer with hashes of the files still needed" }),
828                    );
829                    return;
830                }
831            }
832        }

835    /// `broad_wildcard` and `regex_cost`. Each is reported at most once
836    /// per rule, independently.
837    fn patterns(&mut self, e: &Entry<'_>) {
838        let literals = identity::literals(&e.rule.match_);
839        if let Some((c, _)) = literals.iter().find(|(c, _)| {
840            matches!(c.operator, Operator::Regex | Operator::NotRegex)
841                && c.values
842                    .iter()
843                    .any(|v| regex_near_limit(v, c.case_sensitive.unwrap_or(false)))
844        }) {
845            let _ = c;
846            self.push(
847                FindingCode::RegexCost,
848                vec![e.r()],
849                format!("a regular expression compiles to more than half the engine's limit of {REGEX_SIZE_LIMIT} bytes; matching it is slow and a small edit can push it over"),
850                Some(Fix::Narrow { hint: "simplify the expression or use several values" }),

852        }
853        if !e.rule.action.kind().needs_identity() {

855        }

857        // `not`. An excluded pattern makes the rule narrower.

860                || !c.operator.is_positive()

864            }

867            {

870                    vec![e.r()],
871                    format!("`{v}` matches almost every file"),
872                    Some(Fix::Narrow {
873                        hint: "name a specific directory",
874                    }),
875                );
876                return;
877            }
878            if matches!(c.operator, Operator::Wildcard | Operator::Regex)
879                && !self.facts.path_corpus.is_empty()
880                && let Some(share) = corpus_share(c, &self.facts.path_corpus)
881                && share >= u64::from(self.options.broad_corpus_percent)
882            {
883                self.push(
884                    FindingCode::BroadWildcard,
885                    vec![e.r()],
886                    format!("this pattern matches {share}% of the sample paths"),
887                    Some(Fix::Narrow {
888                        hint: "name a specific directory",
889                    }),
890                );
891                return;
892            }
893        }

896    /// `duplicate_principal`: a principal that a rule both requires and
897    /// excludes, or lists twice on the same field. Polarity is the
898    /// operator's, flipped under each `not`.
899    fn principals(&mut self, e: &Entry<'_>) {
900        let mut include: BTreeMap<Field, BTreeSet<String>> = BTreeMap::new();
901        let mut exclude: BTreeMap<Field, BTreeSet<String>> = BTreeMap::new();
902        for (c, positive) in identity::literals(&e.rule.match_) {
903            if !matches!(
904                c.field.class(),
905                FieldClass::PrincipalSet | FieldClass::Sid | FieldClass::PrincipalName
906            ) {
907                continue;
908            }
909            let side = if positive { &mut include } else { &mut exclude };
910            let set = side.entry(c.field).or_default();
911            for v in &c.values {
912                if !set.insert(principal_key(v)) {
913                    self.push(
914                        FindingCode::DuplicatePrincipal,

917                        None,
918                    );
919                    return;
920                }
921            }
922        }
923        for (field, inc) in &include {
924            if let Some(exc) = exclude.get(field)
925                && let Some(p) = inc.intersection(exc).next()
926            {
927                self.push(
928                    FindingCode::DuplicatePrincipal,
929                    vec![e.r()],
930                    format!("`{p}` is both included and excluded"),
931                    None,

933                return;
934            }

938    /// `duplicate_rule`: rules with the same normalised match, action,
939    /// object types, enforcement, mode, activity window and expiry
940    /// behaviour, so they mean the same. Each group is reported once, on
941    /// every rule but the one that keeps deciding: the first under
942    /// `first_match` and `priority` (by the model's own order), otherwise
943    /// the lowest rule ID. So the result does not depend on rule order, and
944    /// removing the reported rules changes no decision.
945    fn duplicates(&mut self, entries: &[Entry<'_>]) {
946        let model = self.doc.policy.precedence_model;
947        let mut groups: BTreeMap<String, Vec<&Entry<'_>>> = BTreeMap::new();
948        for e in entries {
949            let Ok(key) = serde_json::to_string(&(
950                normalized(&e.rule.match_),
951                &e.rule.action,
952                sorted_types(e.rule),
953                e.rule.enforcement_override,
954                e.set.mandatory,
955                e.set.mode_override,
956                (e.rule.activates_at, e.rule.expires_at, e.rule.on_expire),
957                // Under the ordered models, position matters too.
958                match model {
959                    PrecedenceModel::FirstMatch => (e.rule.order.map(i64::from), e.set.read_only),
960                    PrecedenceModel::Priority => (e.rule.priority.map(i64::from), false),
961                    _ => (None, false),
962                },
963                e.scopes(self.doc),
964            )) else {
965                continue;
966            };
967            groups.entry(key).or_default().push(e);
968        }
969        for group in groups.values_mut().filter(|g| g.len() > 1) {
970            group.sort_by_key(|e| e.r());
971            let keep = group[0].r();
972            for e in &group[1..] {
973                self.push(
974                    FindingCode::DuplicateRule,
975                    vec![e.r(), keep],
976                    "this rule has the same match, action, scope and timing as another".to_owned(),

979            }
980        }

983    /// `shadowed`, `contradiction` and `ineffective_vs_mandatory`.
984    fn conflicts(&mut self, entries: &[Entry<'_>]) {
985        // Only pairs that can be proven are compared. If `cover` covers
986        // `victim`, then every criterion `cover` requires (one that every
987        // match of `cover` must satisfy) is implied by a criterion of
988        // `victim` on the same field, with a value among the required
989        // criterion's values ([`subsume::criterion_covers`]). So each rule
990        // is filed as a cover under the keys of one required criterion,
991        // and looked up as a victim by the keys of all its criteria. A
992        // rule that requires no criterion is compared with every rule.
993        let mut covers_by_key: BTreeMap<String, Vec<usize>> = BTreeMap::new();
994        let mut broad: Vec<usize> = Vec::new();
995        for (j, e) in entries.iter().enumerate() {
996            match identity::best_required(&e.rule.match_) {
997                Some(c) => {
998                    for k in cover_keys(c) {
999                        covers_by_key.entry(k).or_default().push(j);
1000                    }
1001                }
1002                None => broad.push(j),
1003            }
1004        }
1005        // Pairs are checked as they are found, never collected, and each
1006        // victim is compared with at most `MAX_CANDIDATES` covers, most
1007        // selective keys first: a check that gives up only misses findings,
1008        // which the conservative design allows.
1009        let mut reported: BTreeSet<(FindingCode, usize)> = BTreeSet::new();
1010        let mut budget = self.options.max_pair_checks;
1011        for (i, e) in entries.iter().enumerate() {
1012            let Some(others) = candidates(&e.rule.match_, &covers_by_key, &broad, &mut budget)
1013            else {
1014                return;
1015            };
1016            for j in others {
1017                if j == i {
1018                    continue;
1019                }
1020                if budget == 0 {
1021                    return;
1022                }
1023                budget -= 1;
1024                self.conflict(entries, &mut reported, i, j);
1025            }
1026        }
1027    }

1029    /// Checks whether `entries[j]` makes `entries[i]` shadowed,
1030    /// ineffective or contradicted.
1031    fn conflict(
1032        &mut self,
1033        entries: &[Entry<'_>],
1034        reported: &mut BTreeSet<(FindingCode, usize)>,
1035        i: usize,
1036        j: usize,
1037    ) {
1038        let model = self.doc.policy.precedence_model;
1039        {
1040            // Does `cover` (j) always match when `victim` (i) does?
1041            let (victim, cover) = (&entries[i], &entries[j]);
1042            // A scoped rule applies only inside its scope (#38). Scopes are
1043            // not compared, so a pair is only proven when both rules are in
1044            // exactly the same scopes.
1045            if cover.scopes(self.doc) != victim.scopes(self.doc) {
1046                return;
1047            }
1048            // A catch-all is a floor (#329): when the victim matches, the
1049            // catch-all never decides over it, so it neither shadows nor
1050            // contradicts it.
1051            if cover.rule.catch_all && !victim.rule.catch_all {
1052                return;
1053            }
1054            if !always_matches(cover, victim) {
1055                return;
1056            }
1057            // A deny that matches on unknown facts is not a plain match.
1058            let cover_mandatory =
1059                cover.set.mandatory && matches!(cover.rule.action, Action::Deny(_));
1060            let Some(v) = victim.verdict().filter(|_| !victim.set.mandatory) else {
1061                return;
1062            };
1063            if cover_mandatory && v.allows() {
1064                if reported.insert((FindingCode::IneffectiveVsMandatory, i)) {
1065                    self.push(
1066                        FindingCode::IneffectiveVsMandatory,
1067                        vec![victim.r(), cover.r()],
1068                        "a mandatory deny always blocks what this allow matches, so it never lets anything run".to_owned(),
1069                        Some(Fix::RemoveRule),
1070                    );
1071                }
1072                return;
1073            }
1074            if cover.set.mandatory
1075                || cover.rule.enforcement_override != victim.rule.enforcement_override
1076            {
1077                return;
1078            }
1079            // Different modes can make the outranked rule decide (#36).
1080            if cover.set.mode_override != victim.set.mode_override {
1081                return;
1082            }
1083            let Some(cv) = cover.verdict() else { return };
1084            let shadowed = match model {
1085                // Any deny beats every allow, and among denies the most
1086                // specific, then the lowest rule ID, decides.
1087                PrecedenceModel::DenyWins => {
1088                    !cv.allows()
1089                        && (v.allows()
1090                            || (
1091                                crate::precedence::Spec::of(&cover.rule.match_),
1092                                cover.rule.id,
1093                            ) < (
1094                                crate::precedence::Spec::of(&victim.rule.match_),
1095                                victim.rule.id,
1096                            ))
1097                }
1098                PrecedenceModel::FirstMatch => first_match_before(self.doc, cover, victim),
1099                _ => false,
1100            };
1101            if shadowed {
1102                if reported.insert((FindingCode::Shadowed, i)) {
1103                    self.push(
1104                        FindingCode::Shadowed,
1105                        vec![victim.r(), cover.r()],
1106                        format!(
1107                            "under `{}`, another rule always matches first and decides, so this rule never does",
1108                            model_name(model)
1109                        ),
1110                        Some(Fix::RemoveRule),
1111                    );
1112                }
1113                return;
1114            }
1115            if v.allows() != cv.allows() {
1116                let (a, b) = if victim.r() < cover.r() {
1117                    (i, j)
1118                } else {
1119                    (j, i)
1120                };
1121                if reported.insert((
1122                    FindingCode::Contradiction,
1123                    a.min(b) * entries.len() + a.max(b),
1124                )) {
1125                    self.push(
1126                        FindingCode::Contradiction,
1127                        vec![entries[a].r(), entries[b].r()],
1128                        "an allow and a deny both match some objects; only the precedence model decides between them".to_owned(),
1129                        None,
1130                    );
1131                }
1132            }
1133        }
1134    }

1136    /// `too_many_hash_rules`, and `vulnerable_version_allowed`.
1137    fn hash_counts(&mut self) {
1138        let referenced: Vec<&RuleSet> = crate::lifecycle::referenced(self.doc).collect();
1139        for set in &referenced {
1140            let n: usize = set
1141                .rules
1142                .iter()
1143                .flat_map(|r| identity::criteria(&r.match_))
1144                .filter(|c| matches!(c.field.class(), FieldClass::Hash { .. }))
1145                .map(|c| c.values.len())
1146                .sum();
1147            if n > self.options.max_inline_hashes
1148                && let Some(first) = set.rules.first()
1149            {
1150                self.push(
1151                        FindingCode::TooManyHashRules,
1152                        vec![RuleRef { rule_set_id: set.id, rule_id: first.id }],
1153                        format!("rule set `{}` holds {n} inline hash values; more than {} should be a hash set (#42)", set.name, self.options.max_inline_hashes),
1154                        None,
1155                    );
1156            }
1157        }
1158        let clock = self.clock;
1159        // Vulnerable versions: a deny in a mandatory or curated blocklist
1160        // with a version test, and an allow whose versions overlap it for
1161        // the same signer or product.
1162        let vulnerable: Vec<(&RuleSet, &Rule)> = referenced
1163            .iter()
1164            .filter(|s| {
1165                s.kind == crate::model::RuleSetKind::Blocklist && (s.mandatory || s.read_only)
1166            })
1167            .flat_map(|s| s.rules.iter().map(move |r| (*s, r)))
1168            .filter(|(_, r)| activity(r, &clock).is_active() && matches!(r.action, Action::Deny(_)))
1169            .collect();
1170        if vulnerable.is_empty() {
1171            return;
1172        }
1173        // Each deny is filed under the publisher and product names it
1174        // requires: an allow can only overlap a deny that shares one
1175        // ([`overlap`]), so it is compared only with those, and each pair
1176        // costs one unit of the pair budget.
1177        let mut by_key: BTreeMap<String, Vec<usize>> = BTreeMap::new();
1178        for (n, (_, r)) in vulnerable.iter().enumerate() {
1179            for k in identity_name_keys(&r.match_) {
1180                by_key.entry(k).or_default().push(n);
1181            }
1182        }
1183        let mut budget = self.options.max_pair_checks;
1184        for set in &referenced {
1185            for rule in set
1186                .rules
1187                .iter()
1188                .filter(|r| activity(r, &clock).is_active() && r.action.kind().permits())
1189            {
1190                let candidates: BTreeSet<usize> = identity_name_keys(&rule.match_)
1191                    .iter()
1192                    .filter_map(|k| by_key.get(k))
1193                    .flatten()
1194                    .copied()
1195                    .collect();
1196                for (vset, vrule) in candidates.into_iter().map(|n| vulnerable[n]) {
1197                    if budget == 0 {
1198                        return;
1199                    }
1200                    budget -= 1;
1201                    let types_meet = rule.object_types.is_empty()
1202                        || vrule.object_types.is_empty()
1203                        || rule
1204                            .object_types
1205                            .iter()
1206                            .any(|t| vrule.object_types.contains(t));
1207                    if !types_meet {
1208                        continue;
1209                    }
1210                    if let Some(range) = overlap(&rule.match_, &vrule.match_) {
1211                        self.push(
1212                            FindingCode::VulnerableVersionAllowed,
1213                            vec![RuleRef { rule_set_id: set.id, rule_id: rule.id }, RuleRef { rule_set_id: vset.id, rule_id: vrule.id }],
1214                            format!("this allow covers versions {range}, which a blocklist marks vulnerable"),
1215                            Some(Fix::Narrow { hint: "raise the allowed minimum version" }),
1216                        );
1217                        break;
1218                    }
1219                }
1220            }
1221        }
1222    }

1225/// The most covers one rule is compared with.
1226const MAX_CANDIDATES: usize = 512;
1227
1228/// The covers a victim with `tree` is compared with: at most
1229/// [`MAX_CANDIDATES`] distinct rules, collected in key order and stopping
1230/// as soon as the cap is reached, so a huge bucket (one publisher shared
1231/// by thousands of rules) is never copied. Each index entry looked at
1232/// costs one unit of `budget`; `None` once it runs out.
1233fn candidates(
1234    tree: &Match,
1235    covers_by_key: &BTreeMap<String, Vec<usize>>,
1236    broad: &[usize],
1237    budget: &mut usize,
1238) -> Option<BTreeSet<usize>> {
1239    let mut others: BTreeSet<usize> = BTreeSet::new();
1240    for bucket in victim_keys(tree)
1241        .iter()
1242        .filter_map(|k| covers_by_key.get(k).map(Vec::as_slice))
1243        .chain(core::iter::once(broad))
1244    {
1245        for &j in bucket {
1246            if others.len() >= MAX_CANDIDATES {
1247                return Some(others);
1248            }
1249            *budget = budget.checked_sub(1)?;
1250            others.insert(j);
1251        }
1252    }
1253    Some(others)
1254}

1267/// Whether `cover` applies, at every time and for every object and
1268/// context, whenever `victim` does. Conservative:
1269///
1270/// - `cover` applies to every object type `victim` does;
1271/// - `cover` is active whenever `victim` is: its activity window contains
1272///   `victim`'s, and it fails in the same direction or more openly;
1273/// - `victim` is not a deny that also applies on unknown or unparseable
1274///   facts, which `cover` might not;
1275/// - a rule that needs an identity trusts a path only where users cannot
1276///   write (#31), so its path tests match less often than a deny's: such
1277///   a `cover` with a path test is only compared with another rule that
1278///   needs an identity;
1279/// - and `cover`'s tree is `Match` whenever `victim`'s is
1280///   ([`subsume::covers`]).
1281fn always_matches(cover: &Entry<'_>, victim: &Entry<'_>) -> bool {
1282    if !cover.applies_to_all_types_of(victim) {
1283        return false;
1284    }
1285    let (c, v) = (cover.rule, victim.rule);
1286    // `victim`'s window inside `cover`'s, and `cover` no stricter about
1287    // time: a permissive cover needs both clock readings in its window,
1288    // which can end it before a deny victim.
1289    let starts = match (c.activates_at, v.activates_at) {
1290        (None, _) => true,
1291        (Some(a), Some(b)) => a <= b,
1292        (Some(_), None) => false,
1293    };
1294    let ends = match (c.expires_at, v.expires_at) {
1295        (None, _) => true,
1296        (Some(a), Some(b)) => a >= b,
1297        (Some(_), None) => false,
1298    };
1299    let timed = c.activates_at.is_some() || c.expires_at.is_some();
1300    let same_direction = c.action.kind().needs_identity() == v.action.kind().needs_identity()
1301        || !c.action.kind().needs_identity();
1302    if !starts || !ends || (timed && !same_direction) {
1303        return false;
1304    }
1305    if let Action::Deny(o) = &v.action
1306        && (o.unknown_as == crate::model::UnknownAs::Match
1307            || o.on_unparseable == crate::model::UnknownAs::Match)
1308    {
1309        return false;
1310    }
1311    let path_acl = c.action.kind().needs_identity() && !v.action.kind().needs_identity();
1312    if path_acl
1313        && identity::criteria(&c.match_)
1314            .iter()
1315            .any(|x| matches!(x.field.class(), FieldClass::Path | FieldClass::FileName))
1316    {
1317        return false;
1318    }
1319    covers(&c.match_, &v.match_)
1320}

1322/// Whether `a` comes before `b` under `first_match`, whatever matches.
1323fn first_match_before(doc: &PolicyDocument, a: &Entry<'_>, b: &Entry<'_>) -> bool {
1324    let curated_first = doc.policy.precedence_options.curated_first;
1325    let key = |e: &Entry<'_>| {
1326        (
1327            curated_first && !e.set.read_only,
1328            e.rule.order.is_none(),
1329            e.rule.order,
1330            core::cmp::Reverse(e.rule.id),
1331        )
1332    };
1333    // The managed baseline (#39) and then catch-alls (#329) are floors:
1334    // a rule of a lower floor never comes first.
1335    let floor = |e: &Entry<'_>| {
1336        u8::from(crate::unknown::baseline::is_system_baseline(e.set))
1337            + 2 * u8::from(e.rule.catch_all)
1338    };
1339    (floor(a), key(a)) < (floor(b), key(b))
1340}

1387/// A tree with each criterion's values in a canonical order and form, for
1388/// comparing rules.
1389fn normalized(tree: &Match) -> Match {
1390    match tree {
1391        Match::All(c) => {
1392            let mut c: Vec<Match> = c.iter().map(normalized).collect();
1393            c.sort_by_key(|m| serde_json::to_string(m).unwrap_or_default());
1394            Match::All(c)
1395        }
1396        Match::Any(c) => {
1397            let mut c: Vec<Match> = c.iter().map(normalized).collect();
1398            c.sort_by_key(|m| serde_json::to_string(m).unwrap_or_default());
1399            Match::Any(c)
1400        }
1401        Match::Not(c) => Match::Not(Box::new(normalized(c))),
1402        Match::Ref(r) => Match::Ref(*r),
1403        Match::Criterion(c) => {
1404            let mut c: Criterion = c.clone();
1405            let sensitive = c.case_sensitive.unwrap_or(false);
1406            c.values = c
1407                .values
1408                .iter()
1409                .map(|v| match c.field.class() {
1410                    FieldClass::Hash { .. } => {
1411                        crate::ops::normalize_hex(v).unwrap_or_else(|| v.clone())
1412                    }
1413                    // A pattern's text is its meaning (`\w` is not `\W`).
1414                    _ if matches!(c.operator, Operator::Regex | Operator::NotRegex) => v.clone(),
1415                    class if class.has_case() && !sensitive => crate::ops::fold(v, false),
1416                    _ => v.clone(),
1417                })
1418                .collect();
1419            // Two-value operators are positional (`time_between 22:00
1420            // 06:00` wraps past midnight; `06:00 22:00` does not), so their
1421            // order is kept. Every other operator's values are a set.
1422            if c.operator.value_count() != (2, 2) {
1423                c.values.sort();
1424                c.values.dedup();
1425            }
1426            c.case_sensitive = Some(sensitive).filter(|s| *s);
1427            Match::Criterion(c)
1428        }
1429    }
1430}

1479/// Whether a positive path criterion can match a file in the directory
1480/// `dir` (normalised): a literal by prefix in either direction, or a
1481/// pattern that matches a file in `dir` or whose fixed part lies under it.
1482fn path_test_reaches(c: &Criterion, dir: &str) -> bool {
1483    let fold = |v: &str| crate::ops::fold_path_value(v, false);
1484    let d = fold(dir.trim_end_matches('\\'));
1485    let under = |a: &str, b: &str| {
1486        a.strip_prefix(b)
1487            .is_some_and(|r| r.is_empty() || r.starts_with('\\'))
1488    };
1489    match c.operator {
1490        Operator::Equals | Operator::StartsWith => c.values.iter().any(|v| {
1491            let v = fold(
1492                palisade_core::path::normalize(v)
1493                    .as_str()
1494                    .trim_end_matches('\\'),
1495            );
1496            under(&v, &d) || (c.operator == Operator::StartsWith && under(&d, &v))
1497        }),
1498        Operator::Wildcard | Operator::Regex => {
1499            let Ok(compiled) = crate::ops::CompiledCriterion::compile(c) else {
1500                return false;
1501            };
1502            let variables = crate::facts::PathVariables::new();
1503            let context = crate::context::EvaluationContext::new();
1504            let ctx = crate::ops::EvalContext {
1505                action: ActionKind::Deny,
1506                variables: &variables,
1507                context: &context,
1508            };
1509            let probe = format!("{}\\probe.exe", dir.trim_end_matches('\\'));
1510            let facts = crate::facts::ObjectFacts::new().with_path(
1511                crate::facts::PathFact::new(&probe, &palisade_core::path::VolumeMap::new())
1512                    .with_user_writable(false)
1513                    .with_volume(crate::unknown::VolumeKind::Fixed),
1514            );
1515            if compiled.eval(&facts, &ctx).result == crate::ops::Tri::Match {
1516                return true;
1517            }
1518            // The fixed part of a wildcard below the directory.
1519            c.operator == Operator::Wildcard
1520                && c.values.iter().any(|v| {
1521                    let n = palisade_core::path::normalize(v);
1522                    let text = n.as_str();
1523                    let cut = text.find(['*', '?', '[']).unwrap_or(text.len());
1524                    let fixed = fold(text[..cut].trim_end_matches('\\'));
1525                    !fixed.is_empty() && under(&fixed, &d)
1526                })
1527        }
1528        _ => false,
1529    }
1530}

1553/// The versions an allow covers that a vulnerable deny also covers. Both
1554/// trees must require the same publisher or product, and must not
1555/// require disjoint values of any other identity field (product name,
1556/// file names, hashes). On each version field, a tree's versions are the
1557/// intersection of every version criterion it requires (a union over the
1558/// values of one criterion). Conservative: an `any` or a criterion it
1559/// cannot read leaves a tree's versions unbounded, but only when the
1560/// trees share an identity, which keeps the finding to the same product.
1561fn overlap(allow: &Match, deny: &Match) -> Option<String> {
1562    let ra = identity::required_all(allow);
1563    let rd = identity::required_all(deny);
1564    let values = |cs: &[&Criterion], field: Field| -> Option<BTreeSet<String>> {
1565        let sets: Vec<BTreeSet<String>> = cs
1566            .iter()
1567            .filter(|c| c.field == field && c.operator == Operator::Equals)
1568            .map(|c| c.values.iter().map(|v| key(field, v)).collect())
1569            .collect();
1570        sets.into_iter()
1571            .reduce(|a, b| a.intersection(&b).cloned().collect())
1572    };
1573    let shared = |field: Field| match (values(&ra, field), values(&rd, field)) {
1574        (Some(a), Some(d)) => Some(!a.is_disjoint(&d)),
1575        _ => None,
1576    };
1577    if shared(Field::SignerPublisher) != Some(true) && shared(Field::MetaProductName) != Some(true)
1578    {
1579        return None;
1580    }
1581    for field in [
1582        Field::MetaProductName,
1583        Field::MetaOriginalFilename,
1584        Field::MetaInternalName,
1585        Field::PathFileName,
1586        Field::HashSha256,
1587        Field::SignerLeafThumbprint,
1588    ] {
1589        if shared(field) == Some(false) {
1590            return None;
1591        }
1592    }
1593    // The deny must say which versions are vulnerable: a deny with no
1594    // version test is not a vulnerable-version entry, and the allow alone
1595    // is no evidence. Every field either side constrains must overlap, so
1596    // a file-version overlap with disjoint product versions is none.
1597    let mut evidence = None;
1598    for field in [Field::MetaFileVersion, Field::MetaProductVersion] {
1599        let a = versions(&ra, field);
1600        let d = versions(&rd, field);
1601        if a.is_none() && d.is_none() {
1602            continue;
1603        }
1604        let full = vec![(0u64, u64::MAX)];
1605        let constrained_by_deny = d.is_some();
1606        let a = a.unwrap_or_else(|| full.clone());
1607        let d = d.unwrap_or(full);
1608        let shared = a.iter().find_map(|(allow_lo, allow_hi)| {
1609            d.iter().find_map(|(deny_lo, deny_hi)| {
1610                let lo = (*allow_lo).max(*deny_lo);
1611                let hi = (*allow_hi).min(*deny_hi);
1612                (lo <= hi).then_some((lo, hi))
1613            })
1614        });
1615        let (lo, hi) = shared?;
1616        if constrained_by_deny && evidence.is_none() {
1617            evidence = Some(format!("{} to {}", show(lo), show(hi)));
1618        }
1619    }
1620    evidence
1621}

1639/// The versions on `field` that every match of the criteria allows, as
1640/// closed intervals (empty when none can match); `None` if no criterion
1641/// constrains the field.
1642fn versions(cs: &[&Criterion], field: Field) -> Option<Vec<(u64, u64)>> {
1643    let mut out: Option<Vec<(u64, u64)>> = None;
1644    for c in cs.iter().filter(|c| c.field == field) {
1645        let Some(these) = version_ranges(c) else {
1646            continue;
1647        };
1648        out = Some(match out {
1649            None => these,
1650            Some(prev) => prev
1651                .iter()
1652                .flat_map(|(a, b)| {
1653                    these.iter().filter_map(move |(c, d)| {
1654                        let lo = (*a).max(*c);
1655                        let hi = (*b).min(*d);
1656                        (lo <= hi).then_some((lo, hi))
1657                    })
1658                })
1659                .collect(),
1660        });
1661    }
1662    out
1663}

1674/// The versions one criterion allows, as closed intervals: a union over
1675/// its values, empty when no version can match (`version_lt 0.0.0.0`,
1676/// `version_gt` the largest version, or a reversed `version_between`).
1677/// `None` if it is not a version test this reads.
1678fn version_ranges(c: &Criterion) -> Option<Vec<(u64, u64)>> {
1679    let v = |s: &String| crate::ops::Version::parse_strict(s).map(as_u64);
1680    let one = |i: usize| c.values.get(i).and_then(v);
1681    Some(match c.operator {
1682        Operator::VersionEq | Operator::Equals => c
1683            .values
1684            .iter()
1685            .map(|s| v(s).map(|x| (x, x)))
1686            .collect::<Option<Vec<_>>>()?,
1687        Operator::VersionGe => vec![(one(0)?, u64::MAX)],
1688        Operator::VersionGt => one(0)?
1689            .checked_add(1)
1690            .map(|lo| (lo, u64::MAX))
1691            .into_iter()
1692            .collect(),
1693        Operator::VersionLe => vec![(0, one(0)?)],
1694        Operator::VersionLt => one(0)?
1695            .checked_sub(1)
1696            .map(|hi| (0, hi))
1697            .into_iter()
1698            .collect(),
1699        Operator::VersionBetween => {
1700            let (lo, hi) = (one(0)?, one(1)?);
1701            if lo <= hi { vec![(lo, hi)] } else { Vec::new() }
1702        }
1703        _ => return None,
1704    })
1705}

1771            && !c.field.is_signer()

1773            && matches!(c.field.class(), FieldClass::Hash { .. })
