1//! A conservative proof that one match tree matches whenever another does
2//! (#40): `covers(a, b)` is true only if every object and context for
3//! which `b` is `Match` also makes `a` `Match`. It never claims more, so a
4//! rule the policy assistant calls shadowed or ineffective truly never
5//! decides. It does not find every such pair: an unproven pair is just not
6//! reported.
7//!
8//! The proof works on the trees' shapes. `a` covers `b` when:
9//!
10//! - `b` is an `any`: `a` covers each child of `b`;
11//! - `a` is an `all`: each child of `a` covers `b`;
12//! - `a` is an `any`: some child of `a` covers `b`;
13//! - `b` is an `all`: some child of `b` is covered by `a`;
14//! - both are criteria, and the criterion `a` accepts every fact value the
15//!   criterion `b` accepts ([`criterion_covers`]).
16//!
17//! `not` is only covered by an identical `not`. The order of these rules
18//! matters only for how much is proven, never for soundness.
19
20use crate::model::{Criterion, Field, FieldClass, Match, Operator};
21use crate::ops::{Version, fold};
22
23/// Whether `a` is `Match` whenever `b` is.
24#[must_use]
25pub fn covers(a: &Match, b: &Match) -> bool {
26    covers_depth(a, b, 0)
27}
28
29/// The search is bounded: trees are at most 8 deep (#33), and the bound
30/// keeps a hand-built deeper tree from recursing without limit.
31const MAX_DEPTH: usize = 32;
32
33fn covers_depth(a: &Match, b: &Match, depth: usize) -> bool {
34    if depth > MAX_DEPTH {
35        return false;
36    }
37    let d = depth + 1;
38    // An unresolved library reference is never proven to match anything:
39    // its tree is only known once the hierarchy expands it (#37).
40    if matches!(a, Match::Ref(_)) || matches!(b, Match::Ref(_)) {
41        return false;
42    }
43    if a == b && !contains_ref(a) {
44        return true;
45    }
46    if let Match::Any(bs) = b {
47        return !bs.is_empty() && bs.iter().all(|bc| covers_depth(a, bc, d));
48    }
49    if let Match::All(as_) = a {
50        return !as_.is_empty() && as_.iter().all(|ac| covers_depth(ac, b, d));
51    }
52    if let Match::Any(as_) = a
53        && as_.iter().any(|ac| covers_depth(ac, b, d))
54    {
55        return true;
56    }
57    if let Match::All(bs) = b
58        && bs.iter().any(|bc| covers_depth(a, bc, d))
59    {
60        return true;
61    }
62    match (a, b) {
63        (Match::Criterion(ca), Match::Criterion(cb)) => criterion_covers(ca, cb),
64        _ => false,
65    }
66}
67
68/// Whether a tree holds a library reference anywhere.
69fn contains_ref(tree: &Match) -> bool {
70    let mut stack = vec![tree];
71    while let Some(node) = stack.pop() {
72        match node {
73            Match::Ref(_) => return true,
74            Match::All(c) | Match::Any(c) => stack.extend(c),
75            Match::Not(c) => stack.push(c),
76            Match::Criterion(_) => {}
77        }
78    }
79    false
80}

82/// Whether the criterion `a` is `Match` whenever `b` is.
83///
84/// Only cases whose meaning is fully known are proven: the same field, and
85/// either the same operator with `a`'s values a superset of `b`'s (for a
86/// positive operator, where any value matching suffices), or a version
87/// range that contains another. Case sensitivity must agree, or `a` must
88/// be case-insensitive while `b` is not, because folding only widens a
89/// match. Likewise each option that widens a match (`anchored: false`,
90/// `match_ads`, `include_sub_ous`, `safety_critical`, #32) set on `b` must
91/// be set on `a`. Anything else is not proven.
92#[must_use]
93pub fn criterion_covers(a: &Criterion, b: &Criterion) -> bool {
94    if a.field != b.field {
95        return false;
96    }
97    // `a` insensitive and `b` sensitive is fine; the reverse is not.
98    let a_sensitive = a.case_sensitive.unwrap_or(false);
99    let b_sensitive = b.case_sensitive.unwrap_or(false);
100    if a_sensitive && !b_sensitive {
101        return false;
102    }
103    // Options that widen a match (#32): `a` must widen wherever `b` does.
104    // An unanchored regex matches substrings, `match_ads` also matches a
105    // file's streams, `include_sub_ous` also matches units below, and
106    // `safety_critical` makes an unknown fact count as a match.
107    let widens = |x: &Criterion| {
108        [
109            x.anchored == Some(false),
110            x.match_ads == Some(true),
111            x.include_sub_ous == Some(true),
112            x.safety_critical,
113        ]
114    };
115    if widens(b).iter().zip(widens(a)).any(|(&wb, wa)| wb && !wa) {
116        return false;
117    }
118    // Signer options (#31) restrict which signatures take part and what
119    // each must satisfy (`min_key_bits`, `accept_status`, the signature's
120    // source and index, ...). They are not compared by strength, so a
121    // proof needs them equal: otherwise `a` could reject a signature `b`
122    // accepts, and a rule reported as covered would still decide.
123    if a.signer != b.signer {
124        return false;
125    }
126    let class = a.field.class();
127    if matches!(class, FieldClass::Version)
128        && let (Some(ra), Some(rb)) = (range(a), range(b))
129    {
130        return ra.0 <= rb.0 && rb.1 <= ra.1;
131    }
132    if a.operator != b.operator || !a.operator.is_positive() {
133        return false;
134    }
135    match a.operator {
136        Operator::IsTrue | Operator::IsFalse => true,
137        // Patterns are compared as written: folding their text could change
138        // what they mean (`\w` and `\W`). Case must then agree too.
139        Operator::Wildcard | Operator::Regex => {
140            a_sensitive == b_sensitive && b.values.iter().all(|v| a.values.contains(v))
141        }
142        Operator::Equals
143        | Operator::MemberOf
144        | Operator::EndsWith
145        | Operator::Contains
146        | Operator::StartsWith => {
147            // Any value of `b` that matches is also a value of `a`.
148            let key = |v: &str| value_key(a.field, v, a_sensitive);
149            let theirs: Vec<String> = a.values.iter().map(|v| key(v)).collect();
150            b.values.iter().all(|v| theirs.contains(&key(v)))
151        }
152        _ => false,
153    }
154}

156/// How a value compares to another of the same field: hashes as lowercase
157/// hex, paths after normalisation, text after case folding where the
158/// match folds it.
159fn value_key(field: Field, v: &str, case_sensitive: bool) -> String {
160    match field.class() {
161        FieldClass::Hash { .. } => crate::ops::normalize_hex(v).unwrap_or_else(|| v.to_owned()),
162        FieldClass::Path => fold(&v.replace('/', "\\"), case_sensitive),
163        _ if field.class().has_case() => fold(v, case_sensitive),
164        _ => v.to_owned(),
165    }
166}

168/// The closed version interval a version criterion accepts, if it is one
169/// that can be compared: `version_eq` with one value, `version_ge`,
170/// `version_le` and `version_between`. Strict bounds (`gt`, `lt`) are left
171/// out, so no off-by-one can make a proof wrong.
172fn range(c: &Criterion) -> Option<(Version, Version)> {
173    const MIN: Version = Version([0; 4]);
174    const MAX: Version = Version([u16::MAX; 4]);
175    let v = |i: usize| c.values.get(i).and_then(|s| Version::parse_strict(s));
176    Some(match c.operator {
177        Operator::VersionEq if c.values.len() == 1 => (v(0)?, v(0)?),
178        Operator::VersionGe => (v(0)?, MAX),
179        Operator::VersionLe => (MIN, v(0)?),
180        Operator::VersionBetween => (v(0)?, v(1)?),
181        _ => return None,
182    })
183}

189    fn c(field: Field, op: Operator, values: &[&str]) -> Match {
190        Match::Criterion(Criterion::new(field, op, values.iter().copied()))

193    #[test]
194    fn values_and_structure() {
195        let pub_ab = c(Field::SignerPublisher, Operator::Equals, &["A", "B"]);
196        let pub_a = c(Field::SignerPublisher, Operator::Equals, &["a"]);
197        let hash = c(Field::HashSha256, Operator::Equals, &["ab"]);
198        assert!(covers(&pub_ab, &pub_a));
199        assert!(!covers(&pub_a, &pub_ab));
200        assert!(covers(
201            &pub_a,
202            &Match::All(vec![pub_a.clone(), hash.clone()])
203        ));
204        assert!(!covers(
205            &Match::All(vec![pub_a.clone(), hash.clone()]),
206            &pub_a
207        ));
208        assert!(covers(
209            &Match::Any(vec![hash.clone(), pub_a.clone()]),
210            &pub_a
211        ));
212        assert!(covers(
213            &pub_ab,
214            &Match::Any(vec![
215                pub_a.clone(),
216                c(Field::SignerPublisher, Operator::Equals, &["B"])
217            ])
218        ));
219        let not = Match::Not(Box::new(hash.clone()));
220        assert!(covers(&not, &not));
221        assert!(!covers(&not, &hash));
222        // Negative operators are never proven by value sets.
223        let ne = c(Field::SignerPublisher, Operator::NotEquals, &["A", "B"]);
224        assert!(!covers(
225            &ne,
226            &c(Field::SignerPublisher, Operator::NotEquals, &["A"])
227        ));
228    }

230    #[test]
231    fn versions_and_case() {
232        let ge = c(Field::MetaFileVersion, Operator::VersionGe, &["1.0"]);
233        let between = c(
234            Field::MetaFileVersion,
235            Operator::VersionBetween,
236            &["1.2", "3"],
237        );
238        let gt = c(Field::MetaFileVersion, Operator::VersionGt, &["1.0"]);
239        assert!(covers(&ge, &between));
240        assert!(!covers(&between, &ge));
241        assert!(!covers(&ge, &gt));
242        let mut sensitive = Criterion::new(Field::MetaProductName, Operator::Equals, ["X"]);
243        sensitive.case_sensitive = Some(true);
244        let insensitive = Criterion {
245            case_sensitive: None,
246            ..sensitive.clone()
247        };
248        assert!(criterion_covers(&insensitive, &sensitive));
249        assert!(!criterion_covers(&sensitive, &insensitive));
250        sensitive.values = vec!["x".into()];
251        assert!(criterion_covers(&insensitive, &sensitive));
252        // Regular expressions are never folded.
253        let re = |p: &str| Criterion::new(Field::MetaProductName, Operator::Regex, [p]);
254        assert!(!criterion_covers(&re("\\W+"), &re("\\w+")));
255        assert!(criterion_covers(&re("a|b"), &re("a|b")));
256    }

259    fn an_option_that_widens_b_must_widen_a() {
260        let re = |anchored| {
261            let mut c = Criterion::new(Field::MetaProductName, Operator::Regex, ["tool"]);
262            c.anchored = anchored;
263            c

265        assert!(criterion_covers(&re(Some(false)), &re(None)));
266        assert!(!criterion_covers(&re(None), &re(Some(false))));
267        assert!(!criterion_covers(&re(Some(true)), &re(Some(false))));
268        let path = |ads| {
269            let mut c = Criterion::new(Field::PathFull, Operator::Equals, ["C:\\a.exe"]);
270            c.match_ads = ads;
271            c
272        };
273        assert!(criterion_covers(&path(Some(true)), &path(None)));
274        assert!(!criterion_covers(&path(None), &path(Some(true))));
275        let ou = |sub| {
276            let mut c = Criterion::new(Field::DeviceOu, Operator::Equals, ["OU=A,DC=corp"]);
277            c.include_sub_ous = sub;
278            c
279        };
280        assert!(criterion_covers(&ou(Some(true)), &ou(None)));
281        assert!(!criterion_covers(&ou(Some(false)), &ou(Some(true))));
282        let mut critical = Criterion::new(Field::MetaFileVersion, Operator::VersionLe, ["3.2"]);
283        critical.safety_critical = true;
284        let plain = Criterion::new(Field::MetaFileVersion, Operator::VersionLe, ["3.2"]);
285        assert!(criterion_covers(&critical, &plain));
286        assert!(!criterion_covers(&plain, &critical));
287    }
