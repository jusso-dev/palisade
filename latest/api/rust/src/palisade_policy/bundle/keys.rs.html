1//! The stable registry of bundle keys (#43).
2//!
3//! Every struct field and enum variant name the bundle encodes has a small
4//! integer here, used in place of the name. An ID is never reused or
5//! renumbered: a new field or variant gets the next free ID, and a removed
6//! one keeps its entry. `tests/golden/bundle-keys.txt` holds every ID
7//! ever published, and a test fails if one changes meaning.
8//!
9//! The sign of an ID says what a reader that does not know it must do when
10//! it appears as a map key:
11//!
12//! - **critical** (0 and up): the reader refuses the bundle, because it
13//!   cannot tell how the field changes decisions;
14//! - **ignorable** (below 0): the reader skips the entry. Only fields that
15//!   never change a decision, such as descriptions, are ignorable.
16//!
17//! As a value (an enum variant), an unknown ID is always refused.
18//!
19//! Names are one namespace: `name` is the same key in a rule, a rule set
20//! and a location profile. A name that is critical anywhere is critical
21//! everywhere.
22
23use std::collections::BTreeMap;
24use std::sync::LazyLock;
25
26/// One registered key.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub struct Key {
29    /// The integer written in the bundle.
30    pub id: i64,
31    /// The field or variant name in the policy model, as serde writes it.
32    pub name: &'static str,
33    /// The first bundle format version that has it.
34    pub since: u16,
35}
36
37impl Key {
38    /// Whether a reader that does not know the key must refuse the bundle.
39    #[must_use]
40    pub const fn critical(&self) -> bool {
41        self.id >= 0
42    }
43}
44
45const fn k(id: i64, name: &'static str) -> Key {
46    Key { id, name, since: 1 }
47}

604static BY_NAME: LazyLock<BTreeMap<&'static str, i64>> =
605    LazyLock::new(|| KEYS.iter().map(|k| (k.name, k.id)).collect());
606
607static BY_ID: LazyLock<BTreeMap<i64, &'static str>> =
608    LazyLock::new(|| KEYS.iter().map(|k| (k.id, k.name)).collect());

610/// The ID of a name, if it is registered.
611#[must_use]
612pub fn id(name: &str) -> Option<i64> {
613    BY_NAME.get(name).copied()
614}
615
616/// The name of an ID, if it is registered.
617#[must_use]
618pub fn name(id: i64) -> Option<&'static str> {
