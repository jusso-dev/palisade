1//! The bundle's byte format (#43): a header that decodes on its own, then
2//! the body.
3//!
4//! A bundle is one canonical CBOR array of two items, `[header, body]`:
5//!
6//! - the **header** names what the bundle is for and when it may run:
7//!   format and schema version, tenant, target, `policy_version`,
8//!   `min_agent_version`, `generated_at` and an optional `expires_at`.
9//!   [`decode_header`] reads it without reading the body, so the signature
10//!   envelope (#225, in `palisade-proto`) can check the target and version
11//!   cheaply. It is safe on untrusted input;
12//! - the **body** is the effective policy, the library sets its scopes
13//!   use, the source revisions and the digest of the governance metadata
14//!   that stays on the server.
15//!
16//! Every map key is a [registry](super::keys) ID. The encoding is
17//! canonical at two levels: the CBOR is deterministic ([`super::cbor`]),
18//! and the body is in canonical order ([`Body::new`]: rule sets, rules,
19//! assignments and profiles sorted by ID or name, governance metadata
20//! removed). [`PolicyBundle::decode`] refuses input that is not in that
21//! form, so one content has exactly one byte string, and
22//! [`bundle_hash`] identifies it.
23//!
24//! [`PolicyBundle::decision_hash`] covers only what decisions depend on,
25//! for the decision cache (#42): it leaves out the header, the source
26//! revisions, the metadata digest, the policy revision, and every name,
27//! description and display-only field.
28

115/// The header: everything the envelope checks need before the body.
116#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
117pub struct Header {
118    /// The bundle format, [`FORMAT_VERSION`].
119    pub format_version: u16,
120    /// The policy schema version of the body.
121    pub schema_version: SchemaVersion,
122    /// The tenant and target.
123    pub target: Target,
124    /// Monotonic per target, from 1, assigned by the server.
125    pub policy_version: u64,

129    /// Informational: not in the decision hash.

228/// A 32-byte digest, written as a CBOR byte string.
229#[derive(Clone, Copy, Debug, PartialEq, Eq)]
230struct Bytes32(Sha256Digest);
231
232impl Serialize for Bytes32 {
233    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
234        s.serialize_bytes(&self.0)
235    }
236}
237
238impl<'de> Deserialize<'de> for Bytes32 {
239    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
240        struct V;
241        impl serde::de::Visitor<'_> for V {
242            type Value = Bytes32;
243            fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
244                f.write_str("32 bytes")
245            }
246            fn visit_bytes<E: serde::de::Error>(self, v: &[u8]) -> Result<Bytes32, E> {
247                v.try_into()
248                    .map(Bytes32)
249                    .map_err(|_| E::invalid_length(v.len(), &self))
250            }
251        }
252        d.deserialize_bytes(V)
253    }
254}

266/// The body: the policy and what it needs, in canonical order.
267#[derive(Clone, Debug, PartialEq, Eq)]
268pub struct Body {
269    document: PolicyDocument,
270    libraries: Vec<InlineLibrary>,
271    source_revisions: SourceRevisions,
272    metadata_digest: Sha256Digest,
273}
274
275impl Body {
276    /// The canonical body of `document`: its governance metadata is
277    /// replaced by [`Body::metadata_digest`], and rule sets, rules, rule set
278    /// assignments, location profiles, libraries and sources are sorted.
279    ///
280    /// Evaluation does not depend on any of these orders (the precedence
281    /// models are property-tested to be order-independent), so sorting
282    /// changes no decision and makes two servers write the same bytes.
283    /// Text is put in NFC, as the bytes hold it, so a body equals the body
284    /// decoded from its bytes.
285    ///
286    /// # Errors
287    ///
288    /// [`serial::Error`] if the document holds a name with no registered
289    /// key.
290    pub fn new(
291        mut document: PolicyDocument,
292        mut libraries: Vec<InlineLibrary>,
293        mut source_revisions: SourceRevisions,
294    ) -> Result<Self, serial::Error> {
295        let metadata_digest = take_metadata(&mut document);
296        sort_document(&mut document);
297        sort_libraries(&mut libraries);
298        source_revisions.normalise();
299        let body = Self {
300            document,
301            libraries,
302            source_revisions,
303            metadata_digest,
304        };
305        Self::from_value(body.to_value()?)
306    }
307
308    /// The policy, without governance metadata.
309    #[must_use]
310    pub const fn document(&self) -> &PolicyDocument {
311        &self.document
312    }
313
314    /// The inlined library sets, sorted by ID and version.
315    #[must_use]
316    pub fn libraries(&self) -> &[InlineLibrary] {
317        &self.libraries
318    }
319
320    /// Where it came from.
321    #[must_use]
322    pub const fn source_revisions(&self) -> &SourceRevisions {
323        &self.source_revisions
324    }
325
326    /// SHA-256 of the governance metadata the server kept: the policy's,
327    /// and every rule set's and rule's by ID ([`metadata_digest`]).
328    #[must_use]
329    pub const fn metadata_digest(&self) -> &Sha256Digest {
330        &self.metadata_digest
331    }
332
333    fn to_value(&self) -> Result<Value, serial::Error> {
334        serial::to_value(&BodyRepr {
335            document: self.document.clone(),
336            libraries: self.libraries.clone(),
337            source_revisions: self.source_revisions.clone(),
338            metadata_digest: Bytes32(self.metadata_digest),
339        })
340    }
341
342    fn from_value(value: Value) -> Result<Self, serial::Error> {
343        let r: BodyRepr = serial::from_value(value)?;
344        Ok(Self {
345            document: r.document,
346            libraries: r.libraries,
347            source_revisions: r.source_revisions,
348            metadata_digest: r.metadata_digest.0,
349        })
350    }
351
352    /// The same body in canonical order, without recomputing the digest.
353    fn sorted(&self) -> Self {
354        let mut out = self.clone();
355        sort_document(&mut out.document);
356        sort_libraries(&mut out.libraries);
357        out.source_revisions.normalise();
358        out
359    }
360}

362/// SHA-256 over the governance metadata of `doc`: a JSON object with the
363/// policy's metadata, and every non-empty rule set's and rule's by ID.
364/// Field and key order is fixed, so the digest is deterministic.
365#[must_use]
366pub fn metadata_digest(doc: &PolicyDocument) -> Sha256Digest {
367    use std::collections::BTreeMap;
368    #[derive(Serialize)]
369    struct All<'a> {
370        policy: &'a Metadata,
371        rule_sets: BTreeMap<String, &'a Metadata>,
372        rules: BTreeMap<String, &'a Metadata>,
373    }
374    let all = All {
375        policy: &doc.policy.metadata,
376        rule_sets: doc
377            .rule_sets
378            .iter()
379            .filter(|s| !s.metadata.is_empty())
380            .map(|s| (s.id.to_string(), &s.metadata))
381            .collect(),
382        rules: doc
383            .rule_sets
384            .iter()
385            .flat_map(|s| &s.rules)
386            .filter(|r| !r.metadata.is_empty())
387            .map(|r| (r.id.to_string(), &*r.metadata))
388            .collect(),
389    };
390    let json = serde_json::to_vec(&all).unwrap_or_default();
391    Sha256::digest(&json).into()
392}
393
394fn take_metadata(doc: &mut PolicyDocument) -> Sha256Digest {
395    let digest = metadata_digest(doc);
396    doc.policy.metadata = Metadata::default();
397    for set in &mut doc.rule_sets {
398        set.metadata = Metadata::default();
399        for rule in &mut set.rules {
400            rule.metadata = crate::model::RuleMetadata::default();
401        }
402    }
403    digest
404}

406fn sort_document(doc: &mut PolicyDocument) {
407    doc.policy.rule_sets.sort_unstable();
408    doc.policy.rule_set_scopes.sort_by_key(|a| a.rule_set);
409    doc.policy
410        .location_profiles
411        .sort_by(|a, b| a.name.cmp(&b.name));
412    doc.rule_sets.sort_by_key(|s| s.id);
413    for set in &mut doc.rule_sets {
414        set.rules.sort_by_key(|r| r.id);
415    }
416}
417
418fn sort_libraries(libraries: &mut Vec<InlineLibrary>) {
419    libraries.sort_by_key(|l| (l.id, l.resolved_version));
420    libraries.dedup_by_key(|l| (l.id, l.resolved_version));
421}

423/// The part of a document that decisions depend on: every name,
424/// description and display-only field cleared, and the revision 0.
425fn decision_view(doc: &PolicyDocument) -> PolicyDocument {
426    let mut doc = doc.clone();
427    let p = &mut doc.policy;
428    p.name.clear();
429    p.description = None;
430    p.revision = 0;
431    for set in &mut doc.rule_sets {
432        set.name.clear();
433        set.description = None;
434        set.version = None;
435        set.hidden = false;
436        set.category_path = None;
437        set.allow_child_suppress = None;
438        for rule in &mut set.rules {
439            rule.name.clear();
440            rule.description = None;
441        }
442    }
443    doc
444}

478/// Why bytes are not a bundle this crate reads.
479#[derive(Clone, Debug, PartialEq, Eq)]
480pub enum BundleError {
481    /// Larger than the limit.
482    TooLarge {
483        /// The size, in bytes.
484        size: usize,
485        /// The limit.
486        limit: usize,
487    },
488    /// Not canonical CBOR, or over a CBOR limit.
489    Cbor(DecodeError),
490    /// The CBOR does not have a bundle's shape.
491    Shape(&'static str),
492    /// A required header field is missing.
493    Missing(&'static str),
494    /// An integer does not fit its field.
495    Overflow(&'static str),
496    /// An ID is not a UUIDv7.
497    InvalidId(&'static str),
498    /// A key this reader does not know, and that it must not skip.
499    UnknownCriticalKey(i64),
500    /// A bundle format this reader does not implement.
501    UnsupportedFormat(u64),
502    /// A schema major version this reader does not implement.
503    UnsupportedSchema(SchemaVersion),
504    /// The body does not read as a policy.
505    Body(serial::Error),
506    /// The body is not in canonical form: for example, its rules are not
507    /// sorted, or a field holds a default the writer leaves out.
508    NonCanonical,
509}
510
511impl fmt::Display for BundleError {
512    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
513        match self {
514            Self::TooLarge { size, limit } => write!(
515                f,
516                "the bundle is {size} bytes, over the limit of {limit}; split it into a group bundle and a device bundle, or replace hash rules with signer rules"
517            ),
518            Self::Cbor(e) => write!(f, "not canonical CBOR: {e}"),
519            Self::Shape(what) => write!(f, "not a bundle: {what}"),
520            Self::Missing(field) => write!(f, "the header has no `{field}`"),
521            Self::Overflow(field) => write!(f, "`{field}` is out of range"),
522            Self::InvalidId(field) => write!(f, "`{field}` is not a UUIDv7"),
523            Self::UnknownCriticalKey(id) => write!(f, "unknown critical key {id}"),
524            Self::UnsupportedFormat(v) => write!(
525                f,
526                "bundle format {v} is not supported: this reader implements {FORMAT_VERSION}"
527            ),
528            Self::UnsupportedSchema(v) => write!(
529                f,
530                "schema version {v} is not supported: this reader implements {}",
531                crate::schema::CURRENT
532            ),
533            Self::Body(e) => write!(f, "invalid bundle body: {e}"),
534            Self::NonCanonical => f.write_str("the bundle body is not in canonical form"),
535        }
536    }
537}

541fn k(name: &'static str) -> Value {
542    // Every name used here is registered; `keys` tests that.
543    Value::int(keys::id(name).unwrap_or(i64::MIN))
544}

546fn uuid(id: Uuid7) -> Value {
547    Value::Bytes(id.as_bytes().to_vec())
548}

550fn triple(a: u64, b: u64, c: u64) -> Value {
551    Value::Array(vec![Value::Uint(a), Value::Uint(b), Value::Uint(c)])
552}
553
554fn map(entries: Vec<(Value, Value)>) -> Result<Value, serial::Error> {
555    Map::new(entries)
556        .map(Value::Map)
557        .map_err(serial::Error::Encode)
558}

560impl Header {
561    fn to_value(self) -> Result<Value, serial::Error> {
562        let (scope_key, scope_id) = match self.target.scope {
563            TargetScope::Group(id) => ("group", id),
564            TargetScope::Device(id) => ("device", id),
565        };
566        let s = self.schema_version;
567        let a = self.min_agent_version;
568        let mut entries = vec![
569            (k("format_version"), Value::Uint(self.format_version.into())),
570            (k("schema_version"), triple(s.major, s.minor, s.patch)),
571            (k("tenant"), uuid(self.target.tenant)),
572            (k("target"), map(vec![(k(scope_key), uuid(scope_id))])?),
573            (k("policy_version"), Value::Uint(self.policy_version)),
574            (
575                k("min_agent_version"),
576                triple(a.major.into(), a.minor.into(), a.patch.into()),
577            ),
578            (k("generated_at"), Value::int(self.generated_at)),
579        ];
580        if let Some(t) = self.expires_at {
581            entries.push((k("expires_at"), Value::int(t)));
582        }
583        map(entries)
584    }
585
586    fn from_value(value: Value) -> Result<Self, BundleError> {
587        let Value::Map(m) = value else {
588            return Err(BundleError::Shape("the header is not a map"));
589        };
590        // The format version is read first, so a newer format is reported
591        // as such rather than as an unknown key.
592        let format = match m.get(&k("format_version")) {
593            Some(Value::Uint(v)) => *v,
594            Some(_) => return Err(BundleError::Shape("`format_version` is not an integer")),
595            None => return Err(BundleError::Missing("format_version")),
596        };
597        if format != u64::from(FORMAT_VERSION) {
598            return Err(BundleError::UnsupportedFormat(format));
599        }
600        let mut h = HeaderFields::default();
601        for (key, v) in m.into_entries() {
602            let Some(id) = key.as_i64() else {
603                return Err(BundleError::Shape("a header key is not an integer"));
604            };
605            match keys::name(id) {
606                Some("format_version") => {}
607                Some("schema_version") => {
608                    let [a, b, c] = ints(v, "schema_version")?;
609                    h.schema_version = Some(SchemaVersion::new(a, b, c));
610                }
611                Some("tenant") => h.tenant = Some(read_uuid(v, "tenant")?),
612                Some("target") => h.target = Some(read_target(v)?),
613                Some("policy_version") => h.policy_version = Some(uint(v, "policy_version")?),
614                Some("min_agent_version") => {
615                    let [a, b, c] = ints(v, "min_agent_version")?;
616                    let n = |x: u64| {
617                        u32::try_from(x).map_err(|_| BundleError::Overflow("min_agent_version"))
618                    };
619                    h.min_agent_version = Some(AgentVersion::new(n(a)?, n(b)?, n(c)?));
620                }
621                Some("generated_at") => h.generated_at = Some(int(&v, "generated_at")?),
622                Some("expires_at") => h.expires_at = Some(int(&v, "expires_at")?),
623                _ if id < 0 => {}
624                _ => return Err(BundleError::UnknownCriticalKey(id)),
625            }
626        }
627        Ok(Self {
628            format_version: FORMAT_VERSION,
629            schema_version: h
630                .schema_version
631                .ok_or(BundleError::Missing("schema_version"))?,
632            target: Target {
633                tenant: h.tenant.ok_or(BundleError::Missing("tenant"))?,
634                scope: h.target.ok_or(BundleError::Missing("target"))?,
635            },
636            policy_version: h
637                .policy_version
638                .ok_or(BundleError::Missing("policy_version"))?,
639            min_agent_version: h
640                .min_agent_version
641                .ok_or(BundleError::Missing("min_agent_version"))?,
642            generated_at: h.generated_at.ok_or(BundleError::Missing("generated_at"))?,
643            expires_at: h.expires_at,
644        })
645    }
646}

655    generated_at: Option<i64>,
656    expires_at: Option<i64>,
657}

659fn uint(v: Value, field: &'static str) -> Result<u64, BundleError> {
660    match v {
661        Value::Uint(n) => Ok(n),
662        Value::Nint(_) => Err(BundleError::Overflow(field)),
663        _ => Err(BundleError::Shape("an integer field holds something else")),
664    }
665}
666
667fn int(v: &Value, field: &'static str) -> Result<i64, BundleError> {
668    match v {
669        Value::Uint(_) | Value::Nint(_) => v.as_i64().ok_or(BundleError::Overflow(field)),
670        _ => Err(BundleError::Shape("an integer field holds something else")),
671    }
672}

674fn ints(v: Value, field: &'static str) -> Result<[u64; 3], BundleError> {
675    let Value::Array(items) = v else {
676        return Err(BundleError::Shape("a version is not an array"));
677    };
678    let parts: Vec<u64> = items
679        .into_iter()
680        .map(|x| uint(x, field))
681        .collect::<Result<_, _>>()?;
682    parts
683        .try_into()
684        .map_err(|_| BundleError::Shape("a version is not three integers"))
685}

687fn read_uuid(v: Value, field: &'static str) -> Result<Uuid7, BundleError> {
688    let Value::Bytes(b) = v else {
689        return Err(BundleError::Shape("an ID is not a byte string"));
690    };
691    let bytes: [u8; 16] = b.try_into().map_err(|_| BundleError::InvalidId(field))?;
692    Uuid7::from_bytes(bytes).ok_or(BundleError::InvalidId(field))
693}

695fn read_target(v: Value) -> Result<TargetScope, BundleError> {
696    let Value::Map(m) = v else {
697        return Err(BundleError::Shape("the target is not a map"));
698    };
699    let mut entries = m.into_entries();
700    if entries.len() != 1 {
701        return Err(BundleError::Shape("the target names more than one scope"));
702    }
703    let (key, id) = entries.remove(0);
704    match key.as_i64().and_then(keys::name) {
705        Some("group") => Ok(TargetScope::Group(read_uuid(id, "target")?)),
706        Some("device") => Ok(TargetScope::Device(read_uuid(id, "target")?)),
707        _ => match key.as_i64() {
708            Some(id) if id >= 0 => Err(BundleError::UnknownCriticalKey(id)),
709            _ => Err(BundleError::Shape(
710                "the target is neither a group nor a device",
711            )),
712        },
713    }
714}

716/// Whether `written` is `read` with only ignorable map entries (negative
717/// keys) removed: what a reader that drops fields it does not know gives
718/// back. Everything else must be identical.
719fn same_but_dropped(read: &Value, written: &Value) -> bool {
720    match (read, written) {
721        (Value::Map(r), Value::Map(w)) => {
722            let mut w = w.entries().iter().peekable();
723            for (key, value) in r.entries() {
724                match w.peek() {
725                    Some((wk, wv)) if wk == key => {
726                        if !same_but_dropped(value, wv) {
727                            return false;
728                        }
729                        w.next();
730                    }
731                    _ if key.as_i64().is_some_and(|id| id < 0) => {}
732                    _ => return false,
733                }
734            }
735            w.next().is_none()
736        }
737        (Value::Array(r), Value::Array(w)) => {
738            r.len() == w.len() && r.iter().zip(w).all(|(a, b)| same_but_dropped(a, b))
739        }
740        _ => read == written,
741    }
742}

744/// Splits `bytes` into the header item and the rest, checking only the
745/// outer array and the size.
746fn split(bytes: &[u8], limits: &BundleLimits) -> Result<(), BundleError> {
747    if bytes.len() > limits.max_bytes {
748        return Err(BundleError::TooLarge {
749            size: bytes.len(),
750            limit: limits.max_bytes,
751        });
752    }
753    // `[header, body]`: an array of two items, 0x82.
754    match bytes.first() {
755        Some(0x82) => Ok(()),
756        _ => Err(BundleError::Shape("not an array of a header and a body")),
757    }
758}
759
760/// The most data items a header can have, with room for ignorable fields.
761const HEADER_ITEMS: usize = 256;
762
763/// Reads only the header of a bundle. The body is not read or checked, so
764/// this is cheap on a large bundle, and safe on untrusted input: the
765/// header is canonical CBOR and bounded to a few hundred items.
766///
767/// `palisade-proto` calls this for the envelope checks (#225), and
768/// [`PolicyBundle::decode`] after the signature verifies.
769///
770/// # Errors
771///
772/// [`BundleError`] if the bytes are over the size limit, do not start with
773/// a canonical header, or the header is for another format.
774pub fn decode_header(bytes: &[u8], limits: &BundleLimits) -> Result<Header, BundleError> {
775    split(bytes, limits)?;
776    let cbor_limits = Limits {
777        max_bytes: limits.max_bytes,
778        max_items: HEADER_ITEMS,
779        ..Limits::DEFAULT
780    };
781    let (value, _) = cbor::decode_prefix(&bytes[1..], &cbor_limits).map_err(|mut e| {
782        e.offset += 1;
783        BundleError::Cbor(e)
784    })?;
785    Header::from_value(value)
786}

788impl PolicyBundle {
789    /// Builds a bundle.
790    #[must_use]
791    pub const fn new(header: Header, body: Body) -> Self {
792        Self { header, body }
793    }
794
795    /// The canonical bytes.
796    ///
797    /// # Errors
798    ///
799    /// [`serial::Error`] if the body holds a name with no registered key,
800    /// which the key registry tests rule out.
801    pub fn to_bytes(&self) -> Result<Vec<u8>, serial::Error> {
802        let value = Value::Array(vec![self.header.to_value()?, self.body.to_value()?]);
803        cbor::encode(&value).map_err(serial::Error::Encode)
804    }
805
806    /// Reads a bundle, strictly: the CBOR must be canonical, the format
807    /// and schema supported, every critical key known, and the body in
808    /// canonical form, so that the bytes are exactly what
809    /// [`to_bytes`](Self::to_bytes) writes for the result (apart from
810    /// ignorable fields this reader does not know, which it drops).
811    ///
812    /// This does not check `min_agent_version`, and it does not verify a
813    /// signature: verification comes first (#225).
814    ///
815    /// # Errors
816    ///
817    /// [`BundleError`] for anything else.
818    pub fn decode(bytes: &[u8], limits: &BundleLimits) -> Result<Self, BundleError> {
819        split(bytes, limits)?;
820        let cbor_limits = Limits {
821            max_bytes: limits.max_bytes,
822            max_items: limits.max_items,
823            ..Limits::DEFAULT
824        };
825        let value = cbor::decode_with_limits(bytes, &cbor_limits).map_err(BundleError::Cbor)?;
826        let Value::Array(mut items) = value else {
827            return Err(BundleError::Shape("not an array"));
828        };
829        let body_value = items
830            .pop()
831            .ok_or(BundleError::Shape("not an array of a header and a body"))?;
832        let header_value = items
833            .pop()
834            .ok_or(BundleError::Shape("not an array of a header and a body"))?;
835        let header = Header::from_value(header_value.clone())?;
836        if !header.schema_version.is_readable_by(crate::schema::CURRENT) {
837            return Err(BundleError::UnsupportedSchema(header.schema_version));
838        }
839        let body = Body::from_value(body_value.clone()).map_err(|e| match e {
840            serial::Error::UnknownCriticalKey(id) => BundleError::UnknownCriticalKey(id),
841            e => BundleError::Body(e),
842        })?;
843        // The canonical form of what was read must be what was read, apart
844        // from unknown ignorable entries.
845        let again = body.sorted().to_value().map_err(BundleError::Body)?;
846        let header_again = header.to_value().map_err(BundleError::Body)?;
847        if !same_but_dropped(&body_value, &again) || !same_but_dropped(&header_value, &header_again)
848        {
849            return Err(BundleError::NonCanonical);
850        }
851        Ok(Self { header, body })
852    }
853
854    /// Validates the body as a policy (#30): decoding checks the encoding
855    /// and the model's shape, not the rules validation applies. A loader
856    /// calls this after [`decode`](Self::decode) and before it evaluates
857    /// anything.
858    ///
859    /// # Errors
860    ///
861    /// Every finding of [`PolicyDocument::validate`].
862    pub fn validate(
863        &self,
864    ) -> Result<crate::validate::ValidatedPolicy, Vec<crate::validate::ValidationError>> {
865        self.body.document.clone().validate()
866    }
867
868    /// SHA-256 of the canonical bytes ([`bundle_hash`]).
869    ///
870    /// # Errors
871    ///
872    /// As [`to_bytes`](Self::to_bytes).
873    pub fn bundle_hash(&self) -> Result<Sha256Digest, serial::Error> {
874        self.to_bytes().map(|b| bundle_hash(&b))
875    }
876
877    /// SHA-256 over the decision-relevant part of the body: the policy
878    /// without its names, descriptions, display-only rule set fields and
879    /// revision, and the inlined libraries. The header, the source
880    /// revisions and the metadata digest are left out, so a text-only
881    /// edit, a new `policy_version` or a new `generated_at` keeps it. It is
882    /// meant as the policy part of the decision cache key (#42).
883    ///
884    /// A cache keyed by it must not hand back a stored [`Decision`] as is:
885    /// a decision names the policy revision it was made under
886    /// ([`RuleRef::policy_revision`]), and rule names and display IDs, which
887    /// this hash leaves out. Re-stamp the revision (and anything else taken
888    /// from the current policy) when serving a cached decision, or key the
889    /// cache by the bundle hash instead. The engine's cache (#293) keys by
890    /// the snapshot generation today, which changes on every swap, so it is
891    /// not affected.
892    ///
893    /// [`Decision`]: crate::decision::Decision
894    /// [`RuleRef::policy_revision`]: crate::decision::RuleRef::policy_revision
895    ///
896    /// # Errors
897    ///
898    /// As [`to_bytes`](Self::to_bytes).
899    pub fn decision_hash(&self) -> Result<Sha256Digest, serial::Error> {
900        let doc = serial::to_value(&decision_view(&self.body.document))?;
901        let libraries = serial::to_value(&self.body.libraries)?;
902        let value = map(vec![(k("document"), doc), (k("libraries"), libraries)])?;
903        let bytes = cbor::encode(&value).map_err(serial::Error::Encode)?;
904        Ok(Sha256::digest(&bytes).into())
905    }
906}
