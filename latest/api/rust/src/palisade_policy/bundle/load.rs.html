1//! Loading a bundle in one step (#43, #42).
2//!
3//! [`load`] turns bundle bytes into a [`LoadedPolicy`]: it decodes them
4//! strictly ([`PolicyBundle::decode`]), validates the body as a policy
5//! ([`PolicyBundle::validate`]), adds the agent's own `system_baseline`
6//! if it has one, and builds the evaluation indexes with the bundle's
7//! inlined library sets ([`BundleSets`]), so scopes and location profiles
8//! read exactly the sets the server compiled in. Any failure leaves
9//! nothing half-built: the caller keeps its current policy.
10//!
11//! The bytes must already be verified (#225). `load` does not check a
12//! signature, the target or `min_agent_version`; it reads what it is
13//! given, safely, and refuses anything that is not a valid bundle.
14

26/// Why bytes could not be loaded as a policy. The agent keeps its current
27/// policy and reports it.
28#[derive(Clone, Debug, PartialEq, Eq)]
29pub enum LoadError {
30    /// The bytes are not a bundle this reader reads.
31    Decode(BundleError),
32    /// The body is not a valid policy.
33    Invalid(Vec<ValidationError>),
34    /// The agent's system baseline facts do not fit in one rule.
35    Baseline(BaselineError),
36    /// A match tree does not compile, which validation rules out.
37    Build(BuildError),
38}
39
40impl fmt::Display for LoadError {
41    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
42        match self {
43            Self::Decode(e) => write!(f, "the bundle does not decode: {e}"),
44            Self::Invalid(errors) => {
45                write!(
46                    f,
47                    "the bundle's policy is not valid ({} findings)",
48                    errors.len()
49                )?;
50                if let Some(first) = errors.first() {
51                    write!(f, ": {first}")?;
52                }
53                Ok(())
54            }
55            Self::Baseline(e) => write!(f, "the system baseline does not build: {e}"),
56            Self::Build(e) => write!(f, "the bundle's policy does not compile: {e}"),
57        }
58    }
59}
60
61impl std::error::Error for LoadError {}

63/// A bundle, decoded, validated and compiled: ready for
64/// [`Engine::from_loaded`](crate::engine::Engine::from_loaded) or
65/// [`Engine::install`](crate::engine::Engine::install).
66#[derive(Debug)]
67pub struct LoadedPolicy {
68    header: Header,
69    bundle_hash: Sha256Digest,
70    profiles: CompiledProfiles,
71    snapshot: Snapshot,
72}
73
74impl LoadedPolicy {
75    /// The bundle's header.
76    #[must_use]
77    pub const fn header(&self) -> &Header {
78        &self.header
79    }
80
81    /// SHA-256 of the bytes loaded ([`bundle_hash`]), for events (#16).
82    #[must_use]
83    pub const fn bundle_hash(&self) -> &Sha256Digest {
84        &self.bundle_hash
85    }
86
87    /// The validated policy, with the system baseline if one was added.
88    #[must_use]
89    pub fn policy(&self) -> &ValidatedPolicy {
90        self.snapshot.policy()
91    }
92
93    /// The policy's location profiles, compiled with the bundle's sets,
94    /// for the agent's location tracker (#38).
95    #[must_use]
96    pub const fn profiles(&self) -> &CompiledProfiles {
97        &self.profiles
98    }
99
100    /// The compiled snapshot.
101    #[must_use]
102    pub const fn snapshot(&self) -> &Snapshot {
103        &self.snapshot
104    }
105
106    pub(crate) fn into_snapshot(self) -> Snapshot {
107        self.snapshot
108    }
109}

111/// Decodes, validates and compiles a bundle in one step. `baseline` is the
112/// agent's `system_baseline` facts, if it has any
113/// ([`ValidatedPolicy::with_system_baseline`]); a bundle never carries
114/// the baseline itself.
115///
116/// # Errors
117///
118/// [`LoadError`] if the bytes do not decode within `limits`, the policy is
119/// not valid, the baseline does not build, or a tree does not compile.
120pub fn load(
121    bytes: &[u8],
122    limits: &BundleLimits,
123    baseline: Option<&SystemBaseline>,
124) -> Result<LoadedPolicy, LoadError> {
125    let bundle = PolicyBundle::decode(bytes, limits).map_err(LoadError::Decode)?;
126    let mut policy = bundle.validate().map_err(LoadError::Invalid)?;
127    if let Some(facts) = baseline {
128        policy = policy
129            .with_system_baseline(facts)
130            .map_err(LoadError::Baseline)?;
131    }
132    let sets = BundleSets::new(&bundle);
133    let profiles = CompiledProfiles::compile(&policy.document().policy.location_profiles, &sets);
134    // The generation is set when an engine takes the snapshot.
135    let snapshot = Snapshot::with_library(policy, &sets, 0).map_err(LoadError::Build)?;
136    Ok(LoadedPolicy {
137        header: bundle.header,
138        bundle_hash: bundle_hash(bytes),
139        profiles,
140        snapshot,
141    })
142}
