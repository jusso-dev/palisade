1//! Builders for tests: policies, rule sets, rules, match trees, object facts
2//! and evaluation contexts (#25).
3//!
4//! Only compiled with the `test-util` feature, which only
5//! `[dev-dependencies]` turn on (`cargo xtask check-manifests` enforces it),
6//! so none of this reaches a release build. Other crates' tests use it with
7//!
8//! ```toml
9//! [dev-dependencies]
10//! palisade-policy = { workspace = true, features = ["test-util"] }
11//! ```
12//!
13//! Every builder starts from the smallest valid value and changes only what
14//! the test sets, so a test states what matters to it and nothing else.
15//! IDs are deterministic: the same calls give the same document, so
16//! snapshots of it are stable.
17//!
18//! ```
19//! use palisade_policy::model::ObjectType;
20//! use palisade_policy::test_util::{PolicyBuilder, RuleBuilder, RuleSetBuilder, facts, hash, subject};
21//!
22//! let policy = PolicyBuilder::new()
23//!     .rule_set(RuleSetBuilder::allowlist(1).rule(RuleBuilder::allow(1, hash('a'))))
24//!     .validate();
25//! let facts = facts().sha256('a');
26//! let decision = subject(&facts).object_type(ObjectType::Executable).decide(&policy);
27//! assert_eq!(decision.outcome(), palisade_policy::decision::Outcome::Allowed);
28//! ```
29

56/// A UUIDv7 with a fixed time and variant, whose last 12 bits are `kind`
57/// and whose low 32 bits are `n`. IDs of different kinds never collide.
58fn uuid7(kind: u16, n: u32) -> Uuid7 {
59    let text = format!("01936b2e-8c4a-7{:03x}-9f21-0000{n:08x}", kind & 0xfff);
60    text.parse().expect("a valid UUIDv7")
61}

81/// A 64-character SHA-256 made of one repeated hex digit, such as
82/// `aaaa…`. Different digits give different hashes.
83#[must_use]
84pub fn sha256_of(digit: char) -> String {
85    assert!(digit.is_ascii_hexdigit(), "{digit:?} is not a hex digit");
86    std::iter::repeat_n(digit.to_ascii_lowercase(), 64).collect()
87}

89/// One criterion.
90#[must_use]
91pub fn criterion(field: Field, operator: Operator, values: &[&str]) -> Match {
92    Match::Criterion(Criterion {
93        field,
94        operator,
95        values: values.iter().map(|v| (*v).to_owned()).collect(),
96        case_sensitive: None,
97        anchored: None,
98        match_ads: None,
99        include_sub_ous: None,
100        safety_critical: false,
101        signer: None,
102    })
103}

107pub fn hash(digit: char) -> Match {

296impl PolicyBuilder {
297    /// The smallest valid policy.
298    pub fn new() -> Self {
299        Self(PolicyDocument {
300            schema_version: crate::schema::CURRENT,
301            policy: Policy {
302                id: policy_id(1),
303                name: "Test policy".to_owned(),
304                description: None,
305                revision: 1,
306                precedence_model: PrecedenceModel::default(),
307                precedence_options: crate::model::PrecedenceOptions::default(),
308                mode: Mode::Enforce,
309                mode_options: crate::model::ModeOptions::default(),

312                governance: crate::model::GovernanceSettings::default(),
313                rule_sets: Vec::new(),
314                scope: None,
315                rule_set_scopes: Vec::new(),
316                location_profiles: Vec::new(),
317                confirm_permit_all: None,
318                metadata: Metadata::default(),
319            },
320            rule_sets: Vec::new(),
321        })
322    }
323
324    /// The precedence model.
325    pub fn precedence(mut self, model: PrecedenceModel) -> Self {
326        self.0.policy.precedence_model = model;
327        self
328    }
329
330    /// The base mode. [`SubjectBuilder::decide`] decides as enforce mode

344    /// Adds a rule set, and references it from the policy, in order.
345    pub fn rule_set(mut self, set: RuleSetBuilder) -> Self {
346        let set = set.build();
347        self.0.policy.rule_sets.push(set.id);
348        self.0.rule_sets.push(set);
349        self
350    }
351
352    /// The document, unvalidated.
353    pub fn build(self) -> PolicyDocument {
354        self.0
355    }
356
357    /// The document, validated. Panics with the validation errors if it is
358    /// not valid.
359    pub fn validate(self) -> ValidatedPolicy {
360        match self.0.validate() {
361            Ok(policy) => policy,
362            Err(errors) => panic!("the test policy is not valid: {errors:#?}"),
363        }
364    }

367/// Object facts. Starts with no facts.
368#[derive(Clone, Debug, Default)]
369#[must_use]
370pub struct FactsBuilder(ObjectFacts);
371
372/// Starts a [`FactsBuilder`].
373pub fn facts() -> FactsBuilder {
374    FactsBuilder::default()
375}
376
377impl FactsBuilder {
378    /// A hash or version-resource fact (`hash.*`, `meta.*`). Panics for any
379    /// other field, which [`ObjectFacts`] would otherwise drop silently:
380    /// use [`Self::path`] for `path.*`, [`Self::signer`] for `signer.*`, and
381    /// [`context`] for user, device and session fields.
382    pub fn with(mut self, field: Field, value: &str) -> Self {
383        assert!(
384            self.0.set(field, value),
385            "{field:?} is not a stored fact; use .path(..), .signer(..) or context() instead"
386        );
387        self
388    }
389
390    /// `hash.sha256` = [`sha256_of`]`(digit)`.
391    pub fn sha256(self, digit: char) -> Self {
392        self.with(Field::HashSha256, &sha256_of(digit))
393    }
394
395    /// The path, not user-writable, on a fixed volume, with no volume
396    /// mapping: a path an allow can trust.
397    pub fn path(mut self, path: &str) -> Self {
398        self.0 = self.0.with_path(
399            PathFact::new(path, &VolumeMap::new())
400                .with_user_writable(false)
401                .with_volume(crate::unknown::VolumeKind::Fixed),
402        );
403        self
404    }
405
406    /// The path, writable by a standard user.
407    pub fn user_writable_path(mut self, path: &str) -> Self {
408        self.0 = self
409            .0
410            .with_path(PathFact::new(path, &VolumeMap::new()).with_user_writable(true));
411        self
412    }
413
414    /// The kind of volume the path is on (#39). Call after [`Self::path`]
415    /// or [`Self::user_writable_path`]; without one it panics.
416    pub fn volume(mut self, volume: crate::unknown::VolumeKind) -> Self {
417        let path = self.0.path.take().expect("set the path before its volume");
418        self.0 = self.0.with_path(path.with_volume(volume));
419        self
420    }
421
422    /// The origins the agent's origin collection found (#39). An empty list
423    /// means it ran and found nothing: origin `unknown`. Without this call
424    /// the facts have no origin fact at all, which stage 6 treats as missing.
425    pub fn origins(mut self, origins: &[crate::model::OriginClass]) -> Self {
426        self.0 = self.0.with_origin(origins.iter().copied());
427        self
428    }
429
430    /// One valid embedded signature by `publisher`.
431    pub fn signed_by(self, publisher: &str) -> Self {
432        self.signer(SignerStatus::Valid, publisher)
433    }
434
435    /// One embedded signature by `publisher` with `status`.
436    pub fn signer(mut self, status: SignerStatus, publisher: &str) -> Self {
437        let mut signers = self.0.signers.clone().unwrap_or_default();
438        signers.push(SignerFact {
439            status,
440            source: SignatureSource::Embedded,
441            publisher: Some(publisher.to_owned()),
442            issuer: None,
443            leaf_thumbprint_sha256: None,
444            details: Default::default(),
445        });
446        self.0 = self.0.with_signers(signers);
447        self
448    }
449
450    /// The facts.
451    pub fn build(self) -> ObjectFacts {
452        self.0
453    }
454}

499/// What to decide: an object of a type, with facts, in a context. Starts as
500/// an executable, in an empty context, with no path variables.
501#[derive(Debug)]
502#[must_use]
503pub struct SubjectBuilder<'a> {
504    facts: &'a ObjectFacts,
505    object_type: ObjectType,
506    context: EvaluationContext,
507    variables: PathVariables,
508}
509
510/// Starts a [`SubjectBuilder`] over `facts` (a [`FactsBuilder`] or the
511/// [`ObjectFacts`] themselves).
512pub fn subject<F: AsRef<ObjectFacts> + ?Sized>(facts: &F) -> SubjectBuilder<'_> {
513    SubjectBuilder {
514        facts: facts.as_ref(),
515        object_type: ObjectType::Executable,
516        context: EvaluationContext::new(),
517        variables: PathVariables::new(),
518    }
519}
520
521impl AsRef<ObjectFacts> for FactsBuilder {
522    fn as_ref(&self) -> &ObjectFacts {
523        &self.0
524    }
525}
526
527impl<'a> SubjectBuilder<'a> {
528    /// The object type.
529    pub fn object_type(mut self, object_type: ObjectType) -> Self {
530        self.object_type = object_type;
531        self
532    }
533
534    /// The context.
535    pub fn context(mut self, context: ContextBuilder) -> Self {
536        self.context = context.build();
537        self
538    }
539
540    /// The path variables.
541    pub fn variables(mut self, variables: PathVariables) -> Self {
542        self.variables = variables;
543        self
544    }
545
546    fn as_subject(&self) -> Subject<'_> {
547        Subject {
548            object_type: self.object_type,
549            facts: self.facts,
550            variables: &self.variables,
551            context: &self.context,
552        }
553    }
554
555    /// Decides the subject against `policy` as enforce mode sees it
556    /// ([`Evaluator::evaluate`]), with the trace ID [`TRACE`]. The policy's
557    /// modes are not applied; use [`Self::decide_in_mode`] for that.
558    #[must_use]
559    pub fn decide(&self, policy: &ValidatedPolicy) -> Decision {
560        let evaluator = Evaluator::new(policy).expect("a validated policy compiles");
561        evaluator.evaluate(&self.as_subject(), TRACE)
562    }
563
564    /// Decides the subject on a connected device in the policy's own mode,
565    /// with no override session and no lockdown ([`Evaluator::evaluate_in`]
566    /// with [`DeviceState::of`]), with the trace ID [`TRACE`]. `None` when
567    /// the device is disabled.
568    #[must_use]
569    pub fn decide_in_mode(&self, policy: &ValidatedPolicy) -> Option<Decision> {
570        self.decide_on(policy, &DeviceState::of(&policy.document().policy))
571    }
572
573    /// Decides the subject on a device in `state` ([`Evaluator::evaluate_in`]),
574    /// with the trace ID [`TRACE`].
575    #[must_use]
576    pub fn decide_on(&self, policy: &ValidatedPolicy, state: &DeviceState) -> Option<Decision> {
577        let evaluator = Evaluator::new(policy).expect("a validated policy compiles");
578        evaluator.evaluate_in(&self.as_subject(), state, TRACE)
579    }
580}
