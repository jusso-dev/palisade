1//! Candidate indexes, so evaluation is not linear in the number of rules
2//! (#42).
3//!
4//! When an [`Evaluator`] is built, each rule is filed under the keys of
5//! its **anchor** ([`crate::ops`] defines the anchor of one criterion): a
6//! set of keys such that, whenever the rule's match tree is `Match`, the
7//! object's facts contain at least one of them. The anchor of a tree is:
8//!
9//! - a criterion: its own anchor, if it has one;
10//! - `all`: the anchor of one anchored child, since every child must
11//!   match: the one on the cheapest facts (the path and the context come
12//!   before hashes, then signers and the version resource), then the most
13//!   selective (hashes, the leaf thumbprint, the publisher, other text,
14//!   principals, path prefixes);
15//! - `any`: the union of its children's anchors, if every child has one;
16//! - `not`: none.
17//!
18//! The indexes, one map per field:
19//!
20//! - **hash**: decoded digest to rules, per hash field. A rule whose
21//!   criterion lists 25 digests is filed 25 times, so one probe replaces
22//!   scanning every hash rule.
23//! - **signer**: leaf thumbprint, the leaf certificate's TBS hash
24//!   (`signer.tbs_hash` at `position: leaf` only; a CA's TBS hash covers
25//!   every file it signed, so it is evaluated always), and the folded
26//!   publisher and issuer.
27//! - **meta**: folded version-resource strings, file name and extension.
28//! - **path**: folded literal path prefixes of `path.full` and
29//!   `path.directory`, cut at a separator, with values that start with a
30//!   path variable filed under the variable. Wildcards are filed under
31//!   their literal part before the first wildcard character, so
32//!   `C:\Program Files\*\x.exe` is under `C:\PROGRAM FILES`. A probe looks
33//!   up every prefix of the folded fact that ends at a separator.
34//! - **principal**: canonical SIDs and object IDs of `user.sid`,
35//!   `user.primary_group`, `user.groups` and `device.groups`.
36//! - **always**: rules with no anchor, every `deny` with
37//!   `unknown_as: match`, which applies when its tree is `Unknown` and so
38//!   cannot be skipped on a probe that finds nothing, and every rule with a
39//!   `safety_critical` criterion, which counts as matching when its fact
40//!   is unknown (#32). Each is evaluated
41//!   for every object of its types. Validation warns when a policy has
42//!   more than [`ALWAYS_EVAL_WARN`] of them (`unindexed_rules`).
43//!
44//! Mandatory denies have their own set of indexes, probed first.
45//!
46//! Candidate selection is the union of the probes, then the full match
47//! tree is evaluated for each candidate only. A rule that is not a
48//! candidate cannot be `Match`, and is not an always-evaluated deny, so it
49//! does not apply: the matching rules are exactly those that a linear scan
50//! finds, and the decision is the same (a property test checks this
51//! against the linear scan, with the same decision and deciding rule).
52//!
53//! # Lazy facts
54//!
55//! [`Evaluator::evaluate_with`] takes the classes of facts that have been
56//! computed ([`FactSet`]). Probing reads only facts that are there, and
57//! never asks for any. Then:
58//!
59//! - If a rule of the object's type is anchored on a class that has not
60//!   been computed, it could still match, unless the facts computed so far
61//!   already make its tree `NoMatch` (only checked for rules that read
62//!   classes besides their anchor's). Then that class is needed, with the
63//!   classes it reads in that round or an earlier one. Its later rounds
64//!   are asked for once it is a candidate whose tree is still `Unknown`.
65//! - A candidate whose tree is `Unknown` needs the classes it reads that
66//!   have not been computed. A tree that is `Match` or `NoMatch` is
67//!   settled: Kleene logic is monotone, so more facts never change it.
68//!
69//! If anything is needed, the evaluator answers [`NeedFacts`] with the
70//! needed classes of the earliest round ([`FactSet::ROUNDS`]). Every class
71//! it asks for is read by a candidate. A rule only becomes a candidate
72//! later if it was anchored on a class that was missing, and then its
73//! classes were already needed; a settled tree stays settled. So every
74//! class of a round is asked for the first time that round is reached,
75//! and a decision takes at most three rounds after the cheap facts. If no
76//! rule could match given the cheap facts, the decision is made without
77//! asking for anything.

129/// The candidates of one probe, kept inline.
130pub(crate) type Postings = smallvec::SmallVec<[u32; 32]>;

132/// The rules filed under one key: one rule, or a list.
133#[derive(Clone, Debug)]
134enum Posting {
135    One(u32),
136    Many(Vec<u32>),
137}
138
139impl Posting {
140    fn push(&mut self, rule: u32) {
141        match self {
142            Self::One(first) if *first == rule => {}
143            Self::One(first) => *self = Self::Many(vec![*first, rule]),
144            Self::Many(list) => {
145                if list.last() != Some(&rule) {
146                    list.push(rule);
147                }
148            }
149        }
150    }
151
152    fn rules(&self) -> &[u32] {
153        match self {
154            Self::One(rule) => core::slice::from_ref(rule),
155            Self::Many(list) => list,
156        }
157    }
158}
159
160fn file<K: Eq + core::hash::Hash>(map: &mut HashMap<K, Posting>, key: K, rule: u32) {
161    map.entry(key)
162        .and_modify(|p| p.push(rule))
163        .or_insert(Posting::One(rule));
164}

166/// Digests of one fixed length: a sorted array of `(digest, rule)` pairs,
167/// built when the policy is loaded. A probe is a binary search, and one
168/// entry costs the digest and four bytes, with no hash-table overhead, so
169/// a million-entry hash set stays small.
170#[derive(Debug)]
171struct DigestMap {
172    /// The digest length in bytes.
173    len: usize,
174    /// `len` bytes of digest, then the rule as four little-endian bytes,
175    /// per entry: sorted by digest, then rule, without duplicates.
176    entries: Vec<u8>,
177    /// Whether `entries` is sorted (it is after [`Self::seal`]).
178    sealed: bool,
179}
180
181impl DigestMap {
182    fn for_len(len: usize) -> Option<Self> {
183        matches!(len, 16 | 20 | 32 | 48 | 64).then(|| Self {
184            len,
185            entries: Vec::new(),
186            sealed: false,
187        })
188    }
189
190    const fn stride(&self) -> usize {
191        self.len + 4
192    }
193
194    fn insert(&mut self, digest: &[u8], rule: u32) {
195        if digest.len() == self.len {
196            self.entries.extend_from_slice(digest);
197            self.entries.extend_from_slice(&rule.to_le_bytes());
198            self.sealed = false;
199        }
200    }
201
202    /// Sorts the entries and drops duplicates, once every rule is filed.
203    fn seal(&mut self) {
204        let stride = self.stride();
205        let mut rows: Vec<&[u8]> = self.entries.chunks_exact(stride).collect();
206        rows.sort_unstable();
207        rows.dedup();
208        let mut sorted = Vec::with_capacity(rows.len() * stride);
209        for row in rows {
210            sorted.extend_from_slice(row);
211        }
212        sorted.shrink_to_fit();
213        self.entries = sorted;
214        self.sealed = true;
215    }
216
217    fn rows(&self) -> usize {
218        self.entries.len() / self.stride()
219    }
220
221    fn digest(&self, i: usize) -> &[u8] {
222        let at = i * self.stride();
223        &self.entries[at..at + self.len]
224    }
225
226    fn rule(&self, i: usize) -> u32 {
227        let at = i * self.stride() + self.len;
228        let mut b = [0u8; 4];
229        b.copy_from_slice(&self.entries[at..at + 4]);
230        u32::from_le_bytes(b)
231    }
232
233    /// Adds to `out` the rules filed under the digest written as `hex`.
234    /// Returns `false` if `hex` is not a digest of this length. Does not
235    /// allocate.
236    fn probe_hex(&self, hex: &str, out: &mut Postings) -> bool {
237        let mut key = [0u8; 64];
238        let Some(key) = key.get_mut(..self.len) else {
239            return false;
240        };
241        if decode_hex_into(hex, key).is_none() {
242            return false;
243        }
244        debug_assert!(self.sealed, "probed before sealing");
245        // The first row whose digest is not below the key.
246        let (mut lo, mut hi) = (0, self.rows());
247        while lo < hi {
248            let mid = lo + (hi - lo) / 2;
249            if self.digest(mid) < &*key {
250                lo = mid + 1;
251            } else {
252                hi = mid;
253            }
254        }
255        while lo < self.rows() && self.digest(lo) == &*key {
256            out.push(self.rule(lo));
257            lo += 1;
258        }
259        true
260    }
261
262    const fn digest_len(&self) -> usize {
263        self.len
264    }
265
266    /// The number of distinct digests.
267    fn len(&self) -> usize {
268        let mut n = 0;
269        for i in 0..self.rows() {
270            if i == 0 || self.digest(i) != self.digest(i - 1) {
271                n += 1;
272            }
273        }
274        n
275    }
276}

278/// Every index of one tier: the mandatory denies, or the other rules.
279#[derive(Debug, Default)]
280struct IndexSet {
281    hash: Vec<(Field, DigestMap)>,
282    thumbprint: HashMap<[u8; 32], Posting>,
283    /// Leaf TBS hashes, as lower-case hex.
284    leaf_tbs: HashMap<String, Posting>,
285    text: Vec<(Field, HashMap<String, Posting>)>,
286    path: Vec<(Field, HashMap<String, Posting>)>,
287    path_variable: Vec<(Field, PathVariable, Vec<u32>)>,
288    principal: Vec<(Field, HashMap<String, Posting>)>,
289    always: Vec<u32>,
290}
291
292impl IndexSet {
293    /// Files `rule` under the keys of `anchor`. Returns `false`, filing
294    /// nothing, if the anchor cannot be filed (a digest of a length no
295    /// hash has), so the caller evaluates the rule always instead.
296    fn file(&mut self, anchor: &LeafAnchor, rule: u32) -> bool {
297        match anchor {
298            LeafAnchor::Hash(field, digests) => {
299                let len = digests.first().map_or(0, Vec::len);
300                if digests.iter().any(|d| d.len() != len) {
301                    return false;
302                }
303                let Some(i) = self.slot_hash(*field, len) else {
304                    return false;
305                };
306                for d in digests {
307                    self.hash[i].1.insert(d, rule);
308                }
309            }
310            LeafAnchor::Thumbprint(prints) => {
311                for p in prints {
312                    file(&mut self.thumbprint, *p, rule);
313                }
314            }
315            LeafAnchor::LeafTbs(hashes) => {
316                for h in hashes {
317                    file(&mut self.leaf_tbs, h.clone(), rule);
318                }
319            }
320            LeafAnchor::Text(field, values) => {
321                let map = slot(&mut self.text, *field);
322                for v in values {
323                    file(map, v.clone(), rule);
324                }
325            }
326            LeafAnchor::Path {
327                field,
328                prefixes,
329                variables,
330            } => {
331                let map = slot(&mut self.path, *field);
332                for p in prefixes {
333                    file(map, p.clone(), rule);
334                }
335                for v in variables {
336                    match self
337                        .path_variable
338                        .iter_mut()
339                        .find(|(f, var, _)| f == field && var == v)
340                    {
341                        Some((_, _, rules)) => {
342                            if rules.last() != Some(&rule) {
343                                rules.push(rule);
344                            }
345                        }
346                        None => self.path_variable.push((*field, *v, vec![rule])),
347                    }
348                }
349            }
350            LeafAnchor::Principal(field, values) => {
351                let map = slot(&mut self.principal, *field);
352                for v in values {
353                    file(map, v.clone(), rule);
354                }
355            }
356        }
357        true
358    }
359
360    /// Whether [`Self::file`] can file `anchor`: every digest has the
361    /// length of a hash, the same for one field.
362    fn fileable(anchor: &LeafAnchor) -> bool {
363        match anchor {
364            LeafAnchor::Hash(_, digests) => {
365                let len = digests.first().map_or(0, Vec::len);
366                DigestMap::for_len(len).is_some() && digests.iter().all(|d| d.len() == len)
367            }
368            _ => true,
369        }
370    }
371
372    fn slot_hash(&mut self, field: Field, len: usize) -> Option<usize> {
373        if let Some(i) = self.hash.iter().position(|(f, _)| *f == field) {
374            return (self.hash[i].1.digest_len() == len).then_some(i);
375        }
376        self.hash.push((field, DigestMap::for_len(len)?));
377        Some(self.hash.len() - 1)
378    }
379
380    /// Sorts the digest arrays, once every rule is filed.
381    fn seal(&mut self) {
382        for (_, map) in &mut self.hash {
383            map.seal();
384        }
385        for (_, map) in &mut self.text {
386            map.shrink_to_fit();
387        }
388        for (_, map) in &mut self.path {
389            map.shrink_to_fit();
390        }
391    }
392
393    fn keys(&self) -> usize {
394        self.hash.iter().map(|(_, m)| m.len()).sum::<usize>()
395            + self.thumbprint.len()
396            + self.leaf_tbs.len()
397            + self.text.iter().map(|(_, m)| m.len()).sum::<usize>()
398            + self.path.iter().map(|(_, m)| m.len()).sum::<usize>()
399            + self.path_variable.len()
400            + self.principal.iter().map(|(_, m)| m.len()).sum::<usize>()
401    }

403    /// Adds to `out` every rule filed under a key the facts hold, reading
404    /// only the classes in `available`. A fact that is malformed (a digest
405    /// of the wrong length) finds nothing: a criterion on it is `Unknown`,
406    /// never `Match`.
407    ///
408    /// Returns whether a fact that rules of this set are filed under is
409    /// missing, malformed or unresolved although its class was computed.
410    /// Those rules are not candidates, because they cannot be `Match`; but
411    /// their trees may be `Unknown`, so the decision rests on an unknown
412    /// fact and must not be cached.
413    fn probe(&self, subject: &Subject<'_>, available: FactSet, out: &mut Postings) -> bool {
414        let facts = subject.facts;
415        let mut unknown = false;
416        for (field, map) in &self.hash {
417            if !available.contains(FactSet::hash(*field)) {
418                continue;
419            }
420            match facts.get(*field) {
421                Some(hex) => unknown |= !map.probe_hex(hex, out),
422                None => unknown = true,
423            }
424        }
425        let signer_fields = !self.thumbprint.is_empty()
426            || !self.leaf_tbs.is_empty()
427            || find(&self.text, Field::SignerPublisher).is_some()
428            || find(&self.text, Field::SignerIssuer).is_some();
429        if signer_fields && available.contains(FactSet::SIGNERS) {
430            if facts.signers.is_none() {
431                unknown = true;
432            }
433            for signer in facts
434                .signers
435                .iter()
436                .flatten()
437                .filter(|s| s.status.is_trusted())
438            {
439                if !self.leaf_tbs.is_empty() {
440                    match signer.details.leaf_tbs_hash.as_deref() {
441                        // As the evaluator reads it: a digest, compared
442                        // ignoring case.
443                        Some(hex)
444                            if matches!(hex.len(), 40 | 64 | 96 | 128)
445                                && hex.bytes().all(|b| b.is_ascii_hexdigit()) =>
446                        {
447                            let mut key = [0u8; 128];
448                            let key = &mut key[..hex.len()];
449                            key.copy_from_slice(hex.as_bytes());
450                            key.make_ascii_lowercase();
451                            if let Some(p) = core::str::from_utf8(key)
452                                .ok()
453                                .and_then(|k| self.leaf_tbs.get(k))
454                            {
455                                out.extend_from_slice(p.rules());
456                            }
457                        }
458                        _ => unknown = true,
459                    }
460                }
461                if !self.thumbprint.is_empty() {
462                    let mut key = [0u8; 32];
463                    match signer.leaf_thumbprint_sha256.as_deref() {
464                        Some(hex) if decode_hex_into(hex, &mut key).is_some() => {
465                            if let Some(p) = self.thumbprint.get(&key) {
466                                out.extend_from_slice(p.rules());
467                            }
468                        }
469                        _ => unknown = true,
470                    }
471                }
472                for (field, fact) in [
473                    (Field::SignerPublisher, signer.publisher.as_deref()),
474                    (Field::SignerIssuer, signer.issuer.as_deref()),
475                ] {
476                    if let Some(map) = find(&self.text, field) {
477                        match fact {
478                            Some(fact) => with_folded(fact, false, |f| lookup(map, f, out)),
479                            None => unknown = true,
480                        }
481                    }
482                }
483            }
484        }
485        for (field, map) in &self.text {
486            let field = *field;
487            if !available.contains(FactSet::of_field(field)) {
488                continue;
489            }
490            match field.class() {
491                FieldClass::FileName => {
492                    unknown |= !for_each_path_form(facts.path.as_ref(), |path| {
493                        // A bare root has no file name: unknown.
494                        let Some(name) = path.file_name() else {
495                            return false;
496                        };
497                        let text = if field == Field::PathFileName {
498                            name
499                        } else {
500                            path.extension().unwrap_or("")
501                        };
502                        with_folded(text, false, |f| lookup(map, f, out));
503                        true
504                    });
505                }
506                _ if matches!(field, Field::SignerPublisher | Field::SignerIssuer) => {}
507                _ => match facts.get(field) {
508                    Some(fact) => with_folded(fact, false, |f| lookup(map, f, out)),
509                    None => unknown = true,
510                },
511            }
512        }
513        if !self.path.is_empty() {
514            unknown |= !for_each_path_form(facts.path.as_ref(), |path| {
515                let mut known = true;
516                for (field, map) in &self.path {
517                    let text = match field {
518                        Field::PathDirectory if path.file_name().is_some() => {
519                            path.directory().unwrap_or("")
520                        }
521                        // A bare root has no directory: unknown.
522                        Field::PathDirectory => {
523                            known = false;
524                            continue;
525                        }
526                        _ => path.as_str(),
527                    };
528                    with_folded(text, false, |folded| {
529                        for prefix in probe_prefixes(folded) {
530                            lookup(map, prefix, out);
531                        }
532                    });
533                }
534                known
535            });
536        }
537        for (_, variable, rules) in &self.path_variable {
538            if subject.variables.get(*variable).is_empty() {
539                // The rule's value cannot be expanded: unknown.
540                unknown = true;
541            } else {
542                out.extend_from_slice(rules);
543            }
544        }
545        if !self.principal.is_empty() {
546            let user = &subject.context.user;
547            let device = &subject.context.device;
548            for (field, map) in &self.principal {
549                match field {
550                    Field::UserSid | Field::UserPrimaryGroup => {
551                        let fact = if *field == Field::UserSid {
552                            user.sid.as_deref()
553                        } else {
554                            user.primary_group.as_deref()
555                        };
556                        match fact {
557                            Some(fact) => with_buffer(|sid| {
558                                if crate::context::normalize_sid_into(fact, sid) {
559                                    lookup(map, sid, out);
560                                } else {
561                                    unknown = true;
562                                }
563                            }),
564                            None => unknown = true,
565                        }
566                    }
567                    Field::UserGroups | Field::DeviceGroups => {
568                        let set = if *field == Field::UserGroups {
569                            user.groups.as_deref()
570                        } else {
571                            device.groups.as_deref()
572                        };
573                        match set {
574                            Some(set) => with_buffer(|m| {
575                                for member in set {
576                                    if crate::context::normalize_principal_into(member, m) {
577                                        lookup(map, m, out);
578                                    } else {
579                                        unknown = true;
580                                    }
581                                }
582                            }),
583                            None => unknown = true,
584                        }
585                    }
586                    _ => {}
587                }
588            }
589        }
590        out.extend_from_slice(&self.always);
591        unknown
592    }
593}

595fn slot<V: Default>(list: &mut Vec<(Field, V)>, field: Field) -> &mut V {
596    let i = match list.iter().position(|(f, _)| *f == field) {
597        Some(i) => i,
598        None => {
599            list.push((field, V::default()));
600            list.len() - 1
601        }
602    };
603    &mut list[i].1
604}
605
606fn find<V>(list: &[(Field, V)], field: Field) -> Option<&V> {
607    list.iter().find(|(f, _)| *f == field).map(|(_, v)| v)
608}

610fn lookup(map: &HashMap<String, Posting>, key: &str, out: &mut Postings) {
611    if let Some(p) = map.get(key) {
612        out.extend_from_slice(p.rules());
613    }
614}

616/// Runs `f` with each spelling of the path that path criteria test: the
617/// path and its alias, if they are drive or UNC paths and the short name
618/// was resolved. Otherwise path criteria are `Unknown`, never `Match`.
619/// Returns whether the path was known: there is one, it is resolved, and
620/// `f` said every spelling was known.
621fn for_each_path_form(
622    path: Option<&PathFact>,
623    mut f: impl FnMut(&palisade_core::path::NormalizedPath) -> bool,
624) -> bool {
625    let Some(path) = path else {
626        return false;
627    };
628    if path.short_name_unresolved || !path.path.is_resolved() {
629        return false;
630    }
631    let mut known = f(&path.path);
632    if let Some(alias) = path.alias.as_ref().filter(|a| a.is_resolved()) {
633        known &= f(alias);
634    }
635    known
636}

638/// The anchor of a match tree: the keys one of which the facts must hold
639/// for the tree to be `Match`. See the module documentation.
640pub(crate) fn tree_anchor(tree: &Match) -> Option<Vec<LeafAnchor>> {
641    match tree {
642        Match::Criterion(c) => leaf_anchor(c).map(|a| vec![a]),
643        Match::All(children) => children
644            .iter()
645            .filter_map(tree_anchor)
646            .min_by_key(|set| (set.iter().map(LeafAnchor::rank).max(), set.len())),
647        Match::Any(children) => {
648            let mut out = Vec::new();
649            for child in children {
650                out.extend(tree_anchor(child)?);
651            }
652            Some(out)
653        }
654        // `not` identifies nothing.
655        _ => None,
656    }
657}

659/// Whether the evaluator files `rule` in an index rather than evaluating it
660/// for every object: it has an anchor that can be filed, and it is not a
661/// deny with `unknown_as: match`. Validation counts the rules that are not
662/// (`unindexed_rules`).
663pub(crate) fn is_indexed(rule: &crate::model::Rule) -> bool {
664    let fails_closed = matches!(&rule.action, Action::Deny(o) if o.unknown_as == UnknownAs::Match)
665        || safety_critical(&rule.match_);
666    !fails_closed
667        && tree_anchor(&rule.match_).is_some_and(|leaves| leaves.iter().all(IndexSet::fileable))
668}
669
670/// Whether a criterion of `tree` is `safety_critical` (#32): in a mandatory
671/// deny it counts as matching when its fact is unknown, so the rule can
672/// apply with none of its anchor's keys in the facts.
673fn safety_critical(tree: &Match) -> bool {
674    let mut stack = vec![tree];
675    while let Some(node) = stack.pop() {
676        if matches!(node, Match::Criterion(c) if c.safety_critical) {
677            return true;
678        }
679        stack.extend(node.children());
680    }
681    false
682}
683
684/// The classes a rule reads: one per criterion field, and the path ACL if
685/// the rule lets an object run, trusts it or excludes it and tests a
686/// path.
687pub(super) fn reads(rule: &crate::model::Rule) -> FactSet {
688    let mut set = FactSet::EMPTY;
689    let mut stack = vec![&rule.match_];
690    while let Some(node) = stack.pop() {
691        if let Match::Criterion(c) = node {
692            set |= FactSet::of_field(c.field);
693            if rule.action.kind().needs_identity()
694                && matches!(c.field.class(), FieldClass::Path | FieldClass::FileName)
695            {
696                set |= FactSet::PATH_ACL;
697            }
698        }
699        stack.extend(node.children());
700    }
701    set
702}

704/// The slot of an object type in per-type tables.
705fn type_slot(t: ObjectType) -> usize {
706    ObjectType::ALL.iter().position(|x| *x == t).unwrap_or(0)
707}

709/// The number of classes a rule can be anchored on: one bit each.
710const CLASS_BITS: usize = 32;
711
712/// The candidate indexes of one policy.
713#[derive(Debug)]
714pub(super) struct RuleIndex {
715    mandatory: IndexSet,
716    ordinary: IndexSet,
717    /// Per rule: the classes it reads.
718    reads: Vec<FactSet>,
719    /// Per tier: the rules anchored on a class that may not have been
720    /// computed yet ([`Pending`]).
721    pending: [Pending; 2],
722    stats: IndexStats,
723    required: RequiredFacts,
724}
725
726/// The rules of one tier anchored on each class, for the lazy fact
727/// protocol: when the class has not been computed, they could still match.
728#[derive(Debug)]
729struct Pending {
730    /// Per object type, per anchor class bit: `None` if no rule of that
731    /// type is anchored on the class, or every such rule reads other
732    /// classes too; otherwise the class and the classes those rules read
733    /// in the same or an earlier round. Asking for those together means no
734    /// rule that surfaces later needs an earlier round again, so a
735    /// decision takes at most three rounds.
736    whole: Vec<[Option<FactSet>; CLASS_BITS]>,
737    /// Per object type, per anchor class bit: the rules anchored on the
738    /// class that also read classes outside it, so the facts computed so
739    /// far may already settle them. Each is evaluated when its anchor
740    /// class is missing, and only needs the class if its tree is `Unknown`:
741    /// `NoMatch` rules it out, and `Match` makes it a candidate already.
742    mixed: Vec<[Vec<u32>; CLASS_BITS]>,
743}
744
745impl Pending {
746    fn new() -> Self {
747        Self {
748            whole: vec![[None; CLASS_BITS]; ObjectType::ALL.len()],
749            mixed: (0..ObjectType::ALL.len())
750                .map(|_| core::array::from_fn(|_| Vec::new()))
751                .collect(),
752        }
753    }
754}
755
756/// The classes of the rounds up to the one `class` is asked for in: none
757/// for a cheap class.
758fn rounds_to(class: FactSet) -> FactSet {
759    FactSet::ROUNDS
760        .iter()
761        .take(class.round())
762        .fold(FactSet::EMPTY, |acc, r| acc | *r)
763}
764
765/// The bit of a single class.
766fn bit(class: FactSet) -> usize {
767    class.bits().trailing_zeros() as usize
768}

770impl RuleIndex {
771    pub(super) fn build(rules: &[Compiled<'_>]) -> Self {
772        let mut mandatory = IndexSet::default();
773        let mut ordinary = IndexSet::default();
774        let mut reads_all = Vec::with_capacity(rules.len());
775        let mut pending = [Pending::new(), Pending::new()];
776        let mut stats = IndexStats::default();
777        let mut required = RequiredFacts {
778            reads: FactSet::EMPTY,
779            anchored: FactSet::EMPTY,
780        };
781        for (i, compiled) in rules.iter().enumerate() {
782            let rule = u32::try_from(i).unwrap_or(u32::MAX);
783            let tier = Tier::of(compiled.role);
784            let set = match tier {
785                Tier::Mandatory => &mut mandatory,
786                Tier::Ordinary => &mut ordinary,
787            };
788            let reads = reads(compiled.rule);
789            reads_all.push(reads);
790            required.reads |= reads;
791            // A deny that applies on `Unknown` cannot be skipped when a
792            // probe finds nothing, and every leaf of an anchor must be
793            // filed, or the rule could be missed.
794            let anchor = if is_indexed(compiled.rule) {
795                tree_anchor(&compiled.rule.match_)
796            } else {
797                None
798            };
799            let Some(anchor) = anchor else {
800                set.always.push(rule);
801                stats.always_eval += 1;
802                continue;
803            };
804            stats.indexed_rules += 1;
805            let mut classes = FactSet::EMPTY;
806            for leaf in &anchor {
807                let filed = set.file(leaf, rule);
808                debug_assert!(filed, "a fileable anchor is filed");
809                classes |= FactSet::of_field(leaf.field());
810            }
811            required.anchored |= classes;
812            let pending = &mut pending[tier as usize];
813            for t in ObjectType::ALL {
814                if !compiled.rule.object_types.is_empty()
815                    && !compiled.rule.object_types.contains(&t)
816                {
817                    continue;
818                }
819                for class in classes.classes() {
820                    // A rule that reads nothing besides this class cannot
821                    // be settled without it; one that reads other classes
822                    // (another branch of an `any`, a cheap criterion of an
823                    // `all`) may be.
824                    if reads.difference(class).is_empty() {
825                        let cell =
826                            pending.whole[type_slot(t)][bit(class)].get_or_insert(FactSet::EMPTY);
827                        *cell |= class | reads.intersection(rounds_to(class));
828                    } else {
829                        pending.mixed[type_slot(t)][bit(class)].push(rule);
830                    }
831                }
832            }
833        }
834        mandatory.seal();
835        ordinary.seal();
836        stats.keys = mandatory.keys() + ordinary.keys();
837        Self {
838            mandatory,
839            ordinary,
840            reads: reads_all,
841            pending,
842            stats,
843            required,
844        }
845    }
846
847    pub(super) const fn stats(&self) -> IndexStats {
848        self.stats
849    }
850
851    pub(super) const fn required(&self) -> RequiredFacts {
852        self.required
853    }
854}
855
856/// The classes computed for `facts`: those `declared`, the cheap ones,
857/// and any class whose facts are present.
858pub(super) fn available(facts: &ObjectFacts, declared: FactSet) -> FactSet {
859    let mut set = declared | FactSet::CHEAP;
860    for field in facts.fields() {
861        set |= FactSet::hash(field);
862    }
863    if facts.signers.is_some() {
864        set |= FactSet::SIGNERS;
865    }
866    if facts
867        .fields()
868        .any(|f| FactSet::of_field(f) == FactSet::METADATA)
869    {
870        set |= FactSet::METADATA;
871    }
872    // Without a path there is no path criterion to decide, so its ACL is
873    // never needed.
874    if facts
875        .path
876        .as_ref()
877        .is_none_or(|p| p.user_writable.is_some())
878    {
879        set |= FactSet::PATH_ACL;
880    }
881    set
882}

884/// The rules selected for one object.
885pub(super) struct Selection<'a, 'p> {
886    /// The rules that apply, in rule order.
887    pub(super) matched: super::Rules<'a, 'p>,
888    /// Whether a candidate's tree or scopes were `Unknown`: the decision
889    /// rests on a fact that is missing or cannot be trusted, so it must not
890    /// be cached.
891    pub(super) unknown: bool,
892    /// The scope notes (#38) of every candidate whose tree was not
893    /// `NoMatch`. A rule that is not a candidate cannot be `Match`, so its
894    /// scopes never kept a matching rule from applying.
895    pub(super) notes: crate::scope::ScopeNotes,
896}

898/// Why selection stopped before it decided.
899#[derive(Clone, Debug, PartialEq, Eq)]
900pub(crate) enum Stop {
901    /// More facts are needed.
902    Need(NeedFacts),
903    /// The watchdog's budget ran out.
904    Timeout(Timeout),
905}
906
907/// Where the watchdog stopped a selection.
908#[derive(Clone, Debug, PartialEq, Eq)]
909pub(crate) struct Timeout {
910    /// The rule whose tree was evaluated last, when the budget was found
911    /// spent.
912    pub(crate) slow_rule: crate::decision::RuleRef,
913    /// Whether every mandatory deny was settled: evaluated, and not
914    /// waiting on a fact that has not been computed.
915    pub(crate) mandatory_done: bool,
916    /// The ordinals of the mandatory denies that matched. A timeout is
917    /// rare, so this may allocate.
918    pub(crate) mandatory: Vec<u32>,
919}
920
921impl<'p> Evaluator<'p> {
922    /// The rules that apply to the subject, found through the indexes,
923    /// with every fact taken as computed.
924    pub(super) fn select_all<'a>(&'a self, subject: &Subject<'_>) -> Selection<'a, 'p> {
925        match self.select(subject, FactSet::ALL, &mut |_| false) {
926            Ok(selection) => selection,
927            // With every class computed nothing is ever needed, and nothing
928            // stops the selection; if it did, the linear scan decides.
929            Err(_) => {
930                let (matched, notes) = self.matching_with_notes(subject);
931                Selection {
932                    matched: matched.into_iter().collect(),
933                    unknown: true,
934                    notes,
935                }
936            }
937        }
938    }
939
940    /// Selects the rules that apply, reading the classes in `declared`
941    /// and those present in the facts. See the module documentation.
942    /// `expired` is asked after each match tree is evaluated, with its
943    /// rule, and stops the selection when it returns `true`.
944    pub(super) fn select<'a>(
945        &'a self,
946        subject: &Subject<'_>,
947        declared: FactSet,
948        expired: &mut dyn FnMut(&crate::decision::RuleRef) -> bool,
949    ) -> Result<Selection<'a, 'p>, Stop> {
950        let available = available(subject.facts, declared);
951        let index = &self.index;
952        let slot = type_slot(subject.object_type);
953
954        let mut missing = FactSet::EMPTY;
955        let mut hits: Postings = smallvec::SmallVec::new();
956        let mut unknown = false;
957        let mut notes = crate::scope::ScopeNotes::default();
958        // The mandatory tier first, then the rest, each in rule order, so a
959        // mandatory deny is always found before the watchdog can stop the
960        // selection on an ordinary rule. The budget is checked after every
961        // match tree.
962        let mut mandatory_done = false;
963        for tier in [Tier::Mandatory, Tier::Ordinary] {
964            let set = match tier {
965                Tier::Mandatory => &index.mandatory,
966                Tier::Ordinary => &index.ordinary,
967            };
968            let stop = |rule: &Compiled<'_>, hits: &Postings, done: bool| {
969                Stop::Timeout(Timeout {
970                    slow_rule: rule.reference,
971                    mandatory_done: done,
972                    mandatory: hits
973                        .iter()
974                        .copied()
975                        .filter(|&h| {
976                            self.rules
977                                .get(h as usize)
978                                .is_some_and(|r| r.role == Role::Mandatory)
979                        })
980                        .collect(),
981                })
982            };
983            let mut candidates: Postings = smallvec::SmallVec::new();
984            unknown |= set.probe(subject, available, &mut candidates);
985            candidates.sort_unstable();
986            candidates.dedup();
987            for &i in &candidates {
988                let Some(rule) = self.rules.get(i as usize) else {
989                    continue;
990                };
991                if (!rule.rule.object_types.is_empty()
992                    && !rule.rule.object_types.contains(&subject.object_type))
993                    || !self.live(rule)
994                {
995                    continue;
996                }
997                // As `Compiled::applies`: the scopes first (#38), and the
998                // tree only if they could match or noticed something.
999                let scope = rule.scope_eval(subject);
1000                let tree = if scope.result == Tri::NoMatch && scope.notes.is_empty() {
1001                    Tri::NoMatch
1002                } else {
1003                    rule.tree_result(subject)
1004                };
1005                if tree != Tri::NoMatch {
1006                    notes.extend(scope.notes);
1007                }
1008                match scope.result.and(tree) {
1009                    Tri::Match => hits.push(i),
1010                    Tri::NoMatch => {}
1011                    Tri::Unknown => {
1012                        // Only the tree reads facts that can be computed;
1013                        // an unknown scope is settled.
1014                        let need = if tree == Tri::Unknown {
1015                            index.reads[i as usize].difference(available)
1016                        } else {
1017                            FactSet::EMPTY
1018                        };
1019                        if need.is_empty() {
1020                            unknown = true;
1021                            if rule.unknown_as == UnknownAs::Match {
1022                                hits.push(i);
1023                            }
1024                        } else {
1025                            missing |= need;
1026                        }
1027                    }
1028                }
1029                if expired(&rule.reference) {
1030                    return Err(stop(rule, &hits, mandatory_done));
1031                }
1032            }
1033            // Rules of this tier anchored on a class that is not computed
1034            // could still match, unless the facts computed so far already
1035            // settle them.
1036            let pending = &index.pending[tier as usize];
1037            for (b, whole) in pending.whole[slot].iter().enumerate() {
1038                let class = FactSet::from_bits(1 << b);
1039                if available.contains(class) {
1040                    continue;
1041                }
1042                if let Some(need) = whole {
1043                    missing |= class | *need;
1044                }
1045            }
1046            for (b, rules) in pending.mixed[slot].iter().enumerate() {
1047                let class = FactSet::from_bits(1 << b);
1048                if available.contains(class) {
1049                    continue;
1050                }
1051                for &i in rules {
1052                    let need = class | index.reads[i as usize].intersection(rounds_to(class));
1053                    if missing.contains(need.difference(available)) {
1054                        continue;
1055                    }
1056                    let Some(rule) = self.rules.get(i as usize) else {
1057                        continue;
1058                    };
1059                    // A rule that is inactive, or that its scopes take out
1060                    // whatever its tree says, needs nothing.
1061                    let out = !self.live(rule) || {
1062                        let scope = rule.scope_eval(subject);
1063                        scope.result == Tri::NoMatch && scope.notes.is_empty()
1064                    };
1065                    if !out && rule.tree_result(subject) == Tri::Unknown {
1066                        missing |= need;
1067                    }
1068                    if expired(&rule.reference) {
1069                        return Err(stop(rule, &hits, mandatory_done));
1070                    }
1071                }
1072            }
1073            if tier == Tier::Mandatory {
1074                // Settled if no mandatory deny waits on a missing fact.
1075                mandatory_done = missing.difference(available).is_empty();
1076            }
1077        }
1078        // Facts the stages after selection read (#39): the self-allow
1079        // check reads the hash and the signers of an object of a pinned
1080        // type, and stage 6 reads the path's ACL for its location class
1081        // unless the agent gave the class. The pin is checked between the
1082        // mandatory tier and the rest, so it is asked for unless a
1083        // mandatory deny already matched.
1084        let mandatory_hit = hits.iter().any(|&h| {
1085            self.rules
1086                .get(h as usize)
1087                .is_some_and(|r| r.role == Role::Mandatory)
1088        });
1089        if !mandatory_hit {
1090            missing |= self.after_selection(subject);
1091        }
1092        missing = missing.difference(available);
1093
1094        if !missing.is_empty() {
1095            return Err(Stop::Need(NeedFacts(missing.first_round())));
1096        }
1097        // Rules of both tiers in rule order, as a linear scan finds them.
1098        hits.sort_unstable();
1099        let matched = hits
1100            .iter()
1101            .filter_map(|&i| self.rules.get(i as usize))
1102            .collect();
1103        Ok(Selection {
1104            matched,
1105            unknown,
1106            notes,
1107        })
1108    }
1109}

1111impl Evaluator<'_> {
1112    /// The classes the self-allow check and stage 6 read for `subject`,
1113    /// besides those of the rules (#39), where they can change the
1114    /// decision. Stage 6 only decides when no rule did, but asking for its
1115    /// facts up front keeps the rounds in order.
1116    fn after_selection(&self, subject: &Subject<'_>) -> FactSet {
1117        let mut set = FactSet::EMPTY;
1118        if self
1119            .self_images
1120            .iter()
1121            .any(|i| i.object_type() == subject.object_type)
1122        {
1123            set |= FactSet::hash(Field::HashSha256) | FactSet::SIGNERS;
1124        }
1125        // A prompt offers `allow_always_local` only with the file's
1126        // SHA-256, which its directive carries.
1127        if self.prompts {
1128            set |= FactSet::hash(Field::HashSha256);
1129        }
1130        // Without the ACL a file is `user_writable_other`, the stricter
1131        // reading ([`crate::unknown::location`]), so the ACL only matters
1132        // when a level depends on the location.
1133        let classified = subject
1134            .facts
1135            .location_class
1136            .as_ref()
1137            .is_some_and(|c| !c.is_empty());
1138        if subject.facts.path.is_some() && !classified && self.by_location {
1139            set |= FactSet::PATH_ACL;
1140        }
1141        set
1142    }
1143}

1145/// The two tiers of candidates, evaluated in this order.
1146#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1147enum Tier {
1148    Mandatory = 0,
1149    Ordinary = 1,
1150}
1151
1152impl Tier {
1153    fn of(role: Role) -> Self {
1154        if role == Role::Mandatory {
1155            Self::Mandatory
1156        } else {
1157            Self::Ordinary
1158        }
1159    }
1160}

1162impl Compiled<'_> {
1163    /// The tree's result for the subject, with this rule's reading of
1164    /// unparseable versions.
1165    fn tree_result(&self, subject: &Subject<'_>) -> Tri {
1166        let ctx = crate::ops::EvalContext {
1167            action: self.rule.action.kind(),
1168            variables: subject.variables,
1169            context: subject.context,
1170        };
1171        self.tree.eval(
1172            subject.facts,
1173            &ctx,
1174            self.unparseable,
1175            self.role == Role::Mandatory,
1176        )
1177    }
